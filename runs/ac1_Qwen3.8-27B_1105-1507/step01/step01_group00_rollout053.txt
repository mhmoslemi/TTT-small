import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def evaluate_objective(a):
    """Compute 2n * max(conv) / sum^2 without calling evaluate_sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    n = len(a)
    conv = np.convolve(a, a)
    M = np.max(conv)
    return 2.0 * n * M / (S * S)

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

    """Compute weighted gradient using top K lags."""
    n = len(a)
    # Get top K lags
    if K >= len(conv):
        top_k = np.arange(len(conv))
    else:
        top_k_idx = np.argsort(conv)[-K:]

    weights = np.exp(conv[top_k_idx] / temperature)
    weights = weights / weights.sum()
    
    grad = np.zeros(n)

        for j in range(n):
            kj = int(k) - j
            if 0 <= kj < n:
                grad[j] += w * 2.0 * a[kj]
    return grad

from scipy.sparse import csr_matrix


def eval_score(a):
    """Fast local evaluation (no security checks)."""
    a = np.asarray(a, dtype=np.float64)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def build_conv_sparse(f):
    """Sparse matrix A: (A@g)[k] = (f*g)[k]."""
    n = len(f)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(f, M):
    """Max sum(g) s.t. g>=0, (f*g)[k]<=M."""
    n = len(f)
    A = build_conv_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def compute_gradient(a, conv, K=8):
    """Weighted gradient using top-K active lags."""
    n = len(a)
    top_k = np.argsort(conv)[-K:]
    # Exponential weights based on proximity to max
    M = conv[top_k[-1]]  # max
    weights = np.exp((conv[top_k] - M) / 0.05)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_k):
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1) + 1
        js = np.arange(j_lo, j_hi)
        grad[js] += w * 2.0 * a[k - js]
    return grad

        M = float(np.max(conv))
        mean_conv = 1.0 / (2 * n - 1)  # Since sum(a)=1, sum(conv)=1
        score = 2.0 * n * M
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if it % 100 == 0:
                print(f"  {prefix}it={it} score={best_score:.6f} M={M:.8f} max/mean={M/mean_conv:.4f}")
        else:
            no_improve += 1
        
        # Determine phase based on how flat conv is
        ratio = M / mean_conv
        if ratio > 5.0:
            phase = 0
        elif ratio > 2.5:
            phase = 1
        else:
            phase = 2
        
        # Stuck detection
        if no_improve > 80:
            # Try LP jolt
            if time.monotonic() < deadline - 15:
                g = lp_direction(a, M * 1.001)
                if g is not None and float(np.sum(g)) > 1e-10:
                    g /= float(np.sum(g))
                    a = 0.6 * a + 0.4 * g
                    a = np.clip(a, 0, None)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a /= s
                    v = np.zeros(n)
                    no_improve = 0
                    A_sparse = build_conv_sparse(a)
                    continue
                else:
                    # Perturb and restart
                    a = a + 0.01 * np.random.randn(n)
                    a = np.clip(a, 0, None)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a /= s
                    else:
                        break
                    v = np.zeros(n)
                    no_improve = 0
                    A_sparse = build_conv_sparse(a)
                    continue
            else:
                break
        
        # --- Main update ---
        if phase == 0:
            # Focus on reducing max: use subgradient on top lags
            grad = compute_topk_gradient(a, conv, K=5, scale=0.05)
            grad -= grad.mean()  # Project to sum=0
        elif phase == 1:
            # Focus on flattening: use QP direction
            if it % qp_every == 0:
                # Run a few QP iterations to get direction
                M_target = (M + mean_conv) / 2.0
                g_qp = qp_projected_gradient(a, M_target, n_iter=50, lr_init=0.005)
                if g_qp is not None and float(np.sum(g_qp)) > 1e-10:
                    # Direction from current a toward g_qp
                    direction = g_qp - a
                    direction -= direction.mean()
                    # Normalize direction
                    d_norm = float(np.linalg.norm(direction))
                    if d_norm > 1e-12:
                        direction /= d_norm
                    else:
                        direction = -compute_topk_gradient(a, conv, K=8, scale=0.1)
                        direction -= direction.mean()
                else:
                    direction = -compute_topk_gradient(a, conv, K=8, scale=0.1)
                    direction -= direction.mean()
            else:
                direction = -compute_topk_gradient(a, conv, K=8, scale=0.1)
                direction -= direction.mean()
        else:
            # Fine-tune: smaller K, smaller scale
            grad = compute_topk_gradient(a, conv, K=3, scale=0.03)
            grad -= grad.mean()
            direction = -grad
        
        # Momentum step
        v = mu * v + direction
        step = a + lr * v
        step = np.clip(step, 0.0, None)
        s = float(np.sum(step))
        if s < 1e-12:
            lr *= 0.5
            v *= 0.5
            continue
        step /= s
        
        # Evaluate
        new_conv = np.convolve(step, step)
        new_score = 2.0 * n * float(np.max(new_conv))
        
        if new_score < score:
            a = step
            conv = new_conv
            M = float(np.max(conv))
            score = new_score
            lr = min(lr * 1.05, 0.05)
        else:
            # Backtracking
            for _ in range(6):
                lr *= 0.6
                step = a + lr * v
                step = np.clip(step, 0.0, None)
                s2 = float(np.sum(step))
                if s2 < 1e-12:
                    break
                step /= s2
                nc = np.convolve(step, step)
                ns = 2.0 * n * float(np.max(nc))
                if ns < score:
                    a = step
                    conv = nc
                    M = float(np.max(conv))
                    score = ns
                    break
                else:
                    break
            if not (new_score < score):
                lr = max(lr, 1e-9)
        
        it += 1
        
        # Periodic LP
        if it % lp_every == 0 and time.monotonic() < deadline - 25:
            g = lp_direction(a, M * 1.0005)
            if g is not None and float(np.sum(g)) > 1e-10:
                g /= float(np.sum(g))
                # Try several blend factors
                for t in [0.15, 0.25, 0.4, 0.5]:
                    cand = (1 - t) * a + t * g
                    cand = np.clip(cand, 0, None)
                    cs = float(np.sum(cand))
                    if cs < 1e-10:
                        continue
                    cand /= cs
                    c_score = 2.0 * n * float(np.max(np.convolve(cand, cand)))
                    if c_score < score:
                        a = cand
                        v = np.zeros(n)
                        conv = np.convolve(a, a)
                        M = float(np.max(conv))
                        score = c_score
                        A_sparse = build_conv_sparse(a)
                        break
        
        # Rebuild sparse matrix periodically (since a changes)
        if it % 100 == 0:
            A_sparse = build_conv_sparse(a)
        
        # Adapt K and scale
        if it % 300 == 0:
            K = max(3, K - 1)
            scale = max(0.02, scale * 0.9)
    
    return best_a, best_score


def propose_candidate(budget_s=1500.0, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 5.0)
    
    best_seq = None
    best_score = float('inf')
    rng = np.random.default_rng(77)
    
    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            s = evaluate_sequence(parent.tolist())
            if np.isfinite(s):
                best_score = s
                best_seq = parent.copy()
                print(f"Parent n={parent.size}: score={s:.6f}")
    except (NameError, Exception) as e:
        print(f"No parent available: {e}")
    
    # Build candidate list
    candidates = []
    
    if parent is not None:
        candidates.append(('parent', parent.copy()))
    
    n = 512
    idx = np.arange(n, dtype=np.float64)
    
    # Various initializations for n=512
    candidates.append(('unif', np.ones(n)))
    candidates.append(('cosine', 0.5 + 0.5 * np.cos(2 * np.pi * idx / n) + 0.3))
    candidates.append(('sine', np.sin(np.pi * idx / (n - 1))**2 + 0.05))
    candidates.append(('rand1', rng.uniform(0.05, 2.0, size=n)))
    candidates.append(('rand2', rng.uniform(0.0, 3.0, size=n)))
    
    # n=1024 candidates
    n2 = 1024
    idx2 = np.arange(n2, dtype=np.float64)
    candidates.append(('unif1024', np.ones(n2)))
    candidates.append(('cos1024', 0.5 + 0.5 * np.cos(2 * np.pi * idx2 / n2) + 0.3))
    candidates.append(('rand1024', rng.uniform(0.05, 2.0, size=n2)))
    
    # Time allocation: give parent and n=512 candidates more time
    # Prioritize: parent > n=512 > n=1024
    n_cands = len(candidates)
    base_time = (deadline - t0 - 60) / n_cands  # 60s reserved for final
    
    for i, (name, init) in enumerate(candidates):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            print(f"Time nearly up. Best: {best_score:.6f}")
            break
        
        # Allocate time based on priority
        if i == 0 and parent is not None:
            alloc = min(base_time * 2.5, remaining - 30)
        elif '1024' not in name:
            alloc = min(base_time * 1.5, remaining - 30)
        else:
            alloc = min(base_time * 0.8, remaining - 30)
        
        alloc = max(alloc, 20)  # Minimum 20s per candidate
        sub_deadline = time.monotonic() + alloc
        
        s0 = eval_score(init)
        print(f"\n[{name}] n={len(init)} init_score={s0:.6f} alloc={alloc:.0f}s")
        
        if s0 < best_score:
            best_score = s0
            best_seq = init.copy()
        
        # Optimize
        opt_a, opt_score = main_optimize(init, sub_deadline, prefix=f"[{name}]")
        
        if opt_score < best_score:
            best_score = opt_score
            best_seq = opt_a.copy()
            print(f"  [{name}] NEW BEST: {opt_score:.6f}")
    
    # Final refinement on the best sequence found
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\n--- Final refinement (n={len(best_seq)}, score={best_score:.6f}) ---")
        b = best_seq.copy()
        s = float(np.sum(b))
        if s > 0:
            b = b / s
        opt_a, opt_score = main_optimize(b, deadline, prefix="[final]")
        if opt_score < best_score:
            best_score = opt_score
            best_seq = opt_a.copy()
            print(f"  Final: {opt_score:.6f}")
    
    if best_seq is None:
        return [1.0] * 128
    
    # Scale to a reasonable range (avoid too small values that might cause issues)
    final_arr = best_seq.copy()
    s = float(np.sum(final_arr))
    if s > 0:
        # Scale so that values are in a reasonable range
        final_arr = final_arr * (s / s)  # Keep as is (sum-invariant)
    
    final = [float(max(0.0, min(1000.0, x))) for x in final_arr]
    
    # Verify
    final_score = evaluate_sequence(final)
    if not np.isfinite(final_score):
        if parent is not None:
            final = [float(x) for x in parent]
            final_score = evaluate_sequence(final)
        else:
            final = [1.0] * 128
            final_score = evaluate_sequence(final)
    
    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(final)}, score={final_score:.6f}, elapsed={elapsed:.1f}s")
    return final
