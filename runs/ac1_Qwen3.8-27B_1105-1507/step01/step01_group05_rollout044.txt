import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute score = 2*n*max(conv)/sum^2 for sequence f."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    S = f.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * M / (S * S)


def compute_score_normalized(f):
    """Compute score assuming f is normalized to sum=1. Returns 2*n*max(conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def solve_lp_direction(f, M=None, tol=1e-8):
    """Solve LP to find direction g that minimizes max conv(f,g).
    
    Minimize t subject to:
      conv(f, g)[k] <= t for all k
      sum(g) = 1
      g >= 0
    
    Returns (g, t) or (None, None) if infeasible.
    """
    n = len(f)
    A = build_conv_matrix(f)
    
    # Variables: g[0], ..., g[n-1], t
    # c = [0, ..., 0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A@g - t <= 0 for each row
    # [A, -I] @ [g; t] <= 0
    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g[i] >= 0, t unbounded below (but will be positive)
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if not res.success or res.x is None:
        return None, None
    
    g = np.clip(res.x[:n], 0, None)
    t = res.x[n]
    
    # Ensure g sums to 1 (numerical fix)
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None, None
    g = g / g_sum
    
    return g, t

def compute_score_normalized(f):
    """Compute 2*n*max(conv(f,f)) assuming f is normalized to sum=1."""
    n = len(f)
    conv = np.convolve(f, f)

def compute_score_full(f):
    """Compute full score = 2*n*max(conv)/sum^2."""
    n = len(f)
    S = f.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * np.convolve(f, f).max() / (S * S)


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def solve_lp_min_peak(f, top_k=1):
    """Solve LP: minimize t s.t. conv(f,g)[k] <= t for top-k indices,
    conv(f,g)[k] <= M for others, sum(g)=1, g>=0.
    Returns (g, t) or (None, None).
    """
    n = len(f)
    conv_ff = np.convolve(f, f)
    M = conv_ff.max()
    
    # Find top-k indices
    if top_k > 1:
        top_indices = np.argsort(conv_ff)[-top_k:]
    else:
        top_indices = np.array([np.argmax(conv_ff)])
    
    A = build_conv_matrix(f)
    
    # Variables: g[0..n-1], t
    # Minimize t
    c = np.zeros(n + 1)

    n_constraints = 2 * n - 1

    b_ub = np.zeros(n_constraints)

    b_eq = np.array([1.0])

            best_score_overall = score_final
            best_overall = f_final.copy()
    
    # Phase 4: Final local search
    t_left = deadline - time.monotonic() - 2
    if best_overall is not None and t_left > 1:
        n_iter = max(1000, int(t_left * 800))
        best_overall, best_score_overall = local_search(best_overall, rng, n_iter=n_iter)
    
    # Return best
    if best_overall is not None:
        final = np.clip(best_overall, 0, 1000).tolist()
        fs = evaluate_sequence(final)
        print(f"\nFinal: score={fs:.6f}, n={len(final)}")
        return final
    
    # Fallback
    return [1.0] * 256
