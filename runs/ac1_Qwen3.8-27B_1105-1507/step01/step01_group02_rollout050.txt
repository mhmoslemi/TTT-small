"""
Optimizes a nonnegative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses smoothed log-sum-exp gradient with FISTA acceleration, mirror descent,
and multi-scale random restarts. FFT-based convolution for efficiency.

import numpy as np
import time


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def fast_convolve_pair(a, b):
    """Compute full linear convolution of a and b using FFT."""
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    result = np.fft.irfft(fa * fb, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_exact_subgradient(a, c, peak_indices):
    """Compute subgradient of max(c[k]) at a, averaged over active peaks."""
    n = len(a)
    g = np.zeros(n)
    # g[j] = (1/|S|) * sum_{k in S} a[k-j] where 0 <= k-j < n
    # Using FFT: g[j] = conv(a_rev, weights)[j + n - 1] where weights is indicator of S
    a_rev = a[::-1]
    weights = np.zeros(2*n-1)
    weights[peak_indices] = 1.0 / len(peak_indices)
    full = fast_convolve_pair(a_rev, weights)
    g = full[n-1 : 2*n-1]
    return g


def compute_smoothed_gradient(a, c, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*c)) w.r.t. a using FFT."""
    n = len(a)
    c_shift = tau * c - tau * np.max(c)  # numerical stability
    weights = np.exp(c_shift)
    weights /= np.sum(weights)
    
    a_rev = a[::-1]
    full = fast_convolve_pair(a_rev, weights)
    g = full[n-1 : 2*n-1]
    return g


def gen_cosine_window(n, alpha=1.0):
    """Generate a raised-cosine windowed sequence."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(np.pi * (2*x - 1)))  # smooth bump
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a / a.sum()


def gen_exponential_window(n, decay=1.0):
    """Generate an exponential windowed sequence."""
    x = np.linspace(0, 1, n)
    center = 0.5
    a = np.exp(-decay * (x - center)**2 / (2 * 0.15**2))
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a / a.sum()


def gen_random_smooth(n, rng, smoothness=3):
    """Generate a random smooth sequence."""
    a = rng.random(n)
    # Smooth with a moving average
    kernel_size = max(1, n // smoothness)
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a / a.sum()


def optimize_fista(a, deadline, tau=10.0, max_iters=200000, verbose=False):
    """
    FISTA optimization of smoothed objective (1/tau)*logsumexp(tau*conv(a,a))
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    
    c_init = fast_convolve(a)
    best_peak = c_init.max()
    best_a = a.copy()
    
    y = a.copy()
    t_mom = 1.0
    
    # Initial step size
    c_y = fast_convolve(y)
    g_y = compute_smoothed_gradient(y, c_y, tau)
    g_norm = np.max(np.abs(g_y))
    if g_norm < 1e-15:
        return best_a
    lr = 0.5 / (tau * np.max(a) + g_norm)
    
    stall = 0
    stall_limit = 5000
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        t_old = t_mom
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old)) / 2.0

def optimize_mirror_descent(a, deadline, max_iters=100000):
    """
    Mirror descent (exponential weights) on the simplex.
    Minimize max(conv(a,a)) using the smoothed gradient.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    
    c_init = fast_convolve(a)
    best_peak = c_init.max()
    best_a = a.copy()
    
    tau = 5.0
    eta = 1.0  # step size
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c = fast_convolve(a)
        g = compute_smoothed_gradient(a, c, tau)
        
        # Mirror descent update: a_new = exp(log(a) - eta*g) / Z
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * g
        log_a_new -= np.max(log_a_new)  # stability

"""
Optimizes a nonnegative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses smoothed log-sum-exp gradient with FISTA acceleration, temperature annealing,
FFT-based convolution, and exact subgradient polishing. Multi-length, multi-init search.
"""

import numpy as np
import time


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve_self(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def fast_convolve_pair(a, b):
    """Compute full linear convolution of a and b using FFT."""
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    result = np.fft.irfft(fa * fb, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_gradient_fft(a, weights):
    """
    Compute g[j] = 2 * sum_{m=0}^{n-1} a[m] * weights[j+m]
    where weights has length 2n-1 (zero outside [0, 2n-2]).
    Uses FFT-based convolution.
    """
    n = len(a)
    a_rev = a[::-1]
    full = fast_convolve_pair(a_rev, weights)
    # g[j] = 2 * full[j + n - 1] for j = 0, ..., n-1
    g = 2.0 * full[n - 1: 2 * n - 1]
    return g


def compute_smoothed_grad(a, c, tau):
    """Compute gradient of (1/tau)*logsumexp(tau*c) w.r.t. a."""
    n = len(a)
    c_shift = tau * c - tau * np.max(c)
    weights = np.exp(c_shift)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    weights /= w_sum
    return compute_gradient_fft(a, weights)


def compute_exact_grad(a, c, peak_indices):
    """Compute exact subgradient of max(c) averaged over peak_indices."""
    n = len(a)
    weights = np.zeros(2 * n - 1)
    weights[peak_indices] = 1.0 / len(peak_indices)
    return compute_gradient_fft(a, weights)


def gen_sin2_window(n):
    """Raised sine-squared window."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def gen_cosine_window(n, power=1.0):
    """Cosine power window with smooth edges."""
    x = np.linspace(0, 1, n)
    a = (0.5 * (1 + np.cos(np.pi * (2 * x - 1)))) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def gen_kaiser_window(n, beta=8.0):
    """Kaiser window."""
    from scipy.special import i0
    x = np.linspace(-1, 1, n)
    a = i0(beta * np.sqrt(np.maximum(1 - x**2, 0))) / i0(beta)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def gen_flat_tapered(n, taper_frac=0.15):
    """Flat middle with smooth taper at edges."""
    a = np.ones(n)
    n_tap = max(1, int(taper_frac * n))
    for i in range(n_tap):
        w = 0.5 * (1 - np.cos(np.pi * i / n_tap))
        a[i] *= w
        a[n - 1 - i] *= w
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def gen_random_smooth(n, rng, sigma_ratio=0.1):
    """Random sequence smoothed with Gaussian kernel."""
    a = rng.standard_normal(n)
    sigma = sigma_ratio * n
    x = np.arange(n)
    center = n / 2
    kernel = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = a * kernel
    # Additional smoothing
    ksize = max(1, n // 10)
    kernel_lin = np.ones(ksize) / ksize
    a = np.convolve(a, kernel_lin, mode='same')
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    return a / a.sum()


def gen_alt_projection(n, rng, n_iter=40):
    """Alternating projection for flat spectrum initialization."""
    N = 1 << (2 * n - 2).bit_length()
    a = np.ones(n) / n
    for _ in range(n_iter):
        F = np.fft.rfft(a, N)
        # Flatten magnitude
        mag = np.abs(F)
        target_mag = np.full(len(mag), mag.mean())
        F = target_mag * np.exp(1j * np.angle(F))
        F[0] = F[0].real + 0j
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n], 0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def optimize_fista_smoothed(a, deadline, tau_start=30.0, tau_end=2.0, max_iters=300000):
    """
    FISTA optimization with temperature annealing on smoothed log-sumexp objective.
    Minimizes (1/tau)*logsumexp(tau*conv(a,a)) on the simplex.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))

    c = fast_convolve_self(a)
    best_peak = c.max()
    best_a = a.copy()

    y = a.copy()
    t_mom = 1.0

    # Initial step size estimation
    c_y = fast_convolve_self(y)
    g_y = compute_smoothed_grad(y, c_y, tau_start)
    g_norm = np.max(np.abs(g_y))
    if g_norm < 1e-15:
        return best_a
    lr = 1.0 / g_norm

    stall = 0
    stall_limit = 3000

    n_total = max_iters
    for it in range(n_total):
        if time.time() > deadline:
            break

        # Anneal temperature
        frac = float(it) / max(1, n_total)
        tau = tau_start * (tau_end / tau_start) ** frac

        t_old = t_mom
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old)) / 2.0

        # Gradient at y
        c_y = fast_convolve_self(y)
        g_y = compute_smoothed_grad(y, c_y, tau)

        # Backtracking line search (simplified: adapt lr based on gradient magnitude)
        g_abs_max = np.max(np.abs(g_y))
        if g_abs_max > 1e-15:
            target_lr = 0.5 / g_abs_max
            lr = 0.7 * lr + 0.3 * target_lr

        # Gradient step
        x = y - lr * g_y
        x = project_simplex(x)

        # Evaluate true objective
        c_x = fast_convolve_self(x)
        peak = c_x.max()

        if peak < best_peak - 1e-14:
            best_peak = peak
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_limit:
            lr *= 0.5
            stall = 0
            if lr < 1e-14:
                break

        # FISTA extrapolation
        beta = (1.0 - t_old) / t_mom
        y = x + beta * (x - a)
        y = np.maximum(y, 0)
        s = y.sum()
        if s < 1e-15:
            y = np.ones(n) / n
        else:
            y = y / s

        a = x

    return best_a


def polish_exact_subgrad(a, deadline, max_iters=100000):
    """Polish with exact subgradient on the true max of conv(a,a)."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))

    c = fast_convolve_self(a)
    best_peak = c.max()
    best_a = a.copy()

    # Adaptive learning rate
    lr = 0.3 / n
    stall = 0
    stall_limit = 3000

    for it in range(max_iters):
        if time.time() > deadline:
            break

        c = fast_convolve_self(a)
        peak = c.max()

        if peak < best_peak - 1e-14:
            best_peak = peak
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_limit:
            lr *= 0.6
            stall = 0
            if lr < 1e-14:
                break

        # Get peak indices
        threshold = peak * (1.0 - 1e-5)
        peak_indices = np.where(c >= threshold)[0]
        if len(peak_indices) > 30:
            top = np.argsort(c[peak_indices])[::-1][:12]
            peak_indices = peak_indices[top]

        g = compute_exact_grad(a, c, peak_indices)

        # Gradient step with projection
        x = a - lr * g
        a = project_simplex(x)

    return best_a


def optimize_two_phase(a, deadline):
    """Two-phase optimization: FISTA smoothed + exact subgradient polish."""
    remaining = deadline - time.time()
    if remaining < 10:

    # Phase 1: FISTA with annealing (60% of time)
    phase1_deadline = time.time() + remaining * 0.6
    a1 = optimize_fista_smoothed(a, phase1_deadline, tau_start=30.0, tau_end=3.0)

    # Phase 3: Exact subgradient polish (20% of time)
    a3 = polish_exact_subgrad(a2, deadline)

    return a3
