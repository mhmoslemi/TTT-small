import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def soft_max(c, T):
    """Compute (1/T) * log(sum(exp(T*c))) with numerical stability."""
    c_max = np.max(c)
    if T * c_max > 700:
        # Avoid overflow
        return c_max + np.log(np.sum(np.exp(T * (c - c_max)))) / T
    return c_max + np.log(np.sum(np.exp(T * (c - c_max)))) / T

def soft_max_grad(c, T):
    """Compute softmax(T*c) which is the gradient of soft_max w.r.t. c."""
    Tc = T * c - T * np.max(c)
    exp_Tc = np.exp(Tc)
    return exp_Tc / np.sum(exp_Tc)

def convolve_grad(c_weights, a):
    """
    Compute gradient of sum_k w_k * c_k w.r.t. a,
    where c_k = (a*a)[k].
    Returns grad of shape (n,).
    """
    n = len(a)
    grad = np.zeros(n)
    # c_k = sum_j a_j * a[k-j]
    # dc_k/da_i = 2*a[k-i] if 0 <= k-i < n (except when k=2i, where it's 2*a_i but that's the same)
    # Actually: dc_k/da_i = 2*a[k-i] for 0 <= k-i < n
    # (When k-i = i, i.e., k=2i, the formula still gives 2*a_i which is correct
    #  because d(a_i^2)/da_i = 2*a_i)
    for k in range(2*n-1):
        w = c_weights[k]
        if w == 0:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo:hi+1] += 2*w * a[k-hi:k-lo+1][::-1]
        # a[k-j] for j in [lo, hi] = a[k-hi : k-lo+1] reversed
        idx = k - hi + np.arange(hi - lo + 1)  # k-hi, k-hi+1, ..., k
        # We need a[k-j] for j from lo to hi, which is a[k-hi] to a[k-lo]
        # But we need to reverse: for j=lo, k-j=k-lo; for j=hi, k-j=k-hi
        # So grad[j] += 2*w*a[k-j], i.e., grad[lo:hi+1] += 2*w * a[k-hi:k-lo+1][::-1]
        grad[lo:hi+1] += 2.0 * w * a[k-hi:k-lo+1][::-1]
    return grad

        m = beta1 * m + (1 - beta1) * grad_z

        lr = 0.1 / np.sqrt(t + 1)  # or adaptive

    """Compute gradient of sum_k w_k * (a*a)[k] w.r.t. a using FFT."""
    n = len(a)
    # grad_i = 2 * sum_j w_{i+j} * a_j for j=0..n-1
    # This is 2 * (cross-correlation of w and a at lags 0..n-1)

    # where N >= 3n-2

    # (w ⋆ a)[i] = sum_j w[j] * a[j-i] ... hmm, need to be careful
    # Actually: IFFT(FFT(w) * conj(FFT(a))) gives circular cross-correlation
    # (w ⋆_circ a)[i] = sum_j w[j] * a[(j-i) mod N]
    # For linear case (with enough zero-padding): = sum_j w[j] * a[j-i] for 0<=j-i<N

    # For real a: = sum_k w[k] * a[k-i]
    # So we want IFFT(Fw * conj(Fa))[:n]
    result = np.fft.irfft(Fw * np.conj(Fa), N)[:n]
    return 2.0 * result

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def softmax(x):
    x = x - np.max(x)
    e = np.exp(x)
    return e / e.sum()

def soft_max_and_weights(c, T):
    """Returns (soft_max_value, weights) where weights = softmax(T*c)."""
    c_shifted = T * (c - np.max(c))
    exp_c = np.exp(c_shifted)
    sum_exp = exp_c.sum()
    soft_max_val = np.max(c) + np.log(sum_exp) / T
    weights = exp_c / sum_exp
    return soft_max_val, weights

def convolve_grad_fft(weights, a):
    """Compute 2 * (w ⋆ a) using FFT."""
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    Fw = np.fft.rfft(weights, N)
    Fa = np.fft.rfft(a, N)
    result = np.fft.irfft(Fw * np.conj(Fa), N)[:n]
    return 2.0 * result

def adam_log_space(a_init, deadline, T_start=5.0, T_end=500.0, 
                   lr=0.05, beta1=0.8, beta2=0.999, eps=1e-8,
                   eval_interval=50, restart_interval=None, rng=None):
    """Optimize in log-space using Adam."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / a_init.sum()
    
    z = np.log(a_init)  # log-parameterization
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a_init.copy()
    best_score = evaluate_sequence(a_init.tolist())
    
    total_deadline = deadline
    start_time = time.time()
    
    while time.time() < total_deadline - 5:
        elapsed = time.time() - start_time
        # Total expected iterations: let's assume ~5000 iterations
        progress = min(1.0, elapsed / 600.0)
        T = T_start * (T_end / T_start) ** progress
        
        a = softmax(z)
        c = fast_convolve(a)
        _, weights = soft_max_and_weights(c, T)
        
        grad_a = convolve_grad_fft(weights, a)
        grad_a *= 2.0 * n  # from the objective 2n * soft_max
        
        # Convert to z-gradient
        mean_grad = np.dot(grad_a, a)
        grad_z = a * (grad_a - mean_grad)
        
        # Sparse gradient handling: if grad_z has very small entries, skip
        t += 1
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adaptive learning rate
        lr_t = lr / (1.0 + 0.01 * np.sqrt(progress))
        z -= lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic evaluation
        if t % eval_interval == 0:
            a_check = softmax(z)
            score = evaluate_sequence(a_check.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a_check.copy()
                print(f"  Adam iter {t}: score={best_score:.6f}, T={T:.1f}")
        
        # Restart perturbation
        if restart_interval is not None and t % restart_interval == 0:
            if time.time() < total_deadline - 30:
                noise_scale = 0.1 * (1.0 - progress)
                z += rng.normal(0, noise_scale, n)
    
    return best_a, best_score

"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses Adam optimizer in log-space (softmax parameterization) with annealed
log-sum-exp approximation of the max. Multiple phases with restarts.
"""

import numpy as np
import time


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def softmax(x):
    """Numerically stable softmax."""
    x = x - np.max(x)
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def soft_max_weights(c, T):
    """Compute weights = softmax(T*c) for gradient of soft-max."""
    Tc = T * c
    Tc_shifted = Tc - np.max(Tc)
    exp_Tc = np.exp(Tc_shifted)
    s = exp_Tc.sum()
    if s < 1e-300:
        # All weights equal (uniform)
        return np.ones(len(c)) / len(c)
    return exp_Tc / s


def convolve_grad_fft(weights, a):
    """
    Compute gradient of sum_k w_k * c_k w.r.t. a, where c_k = (a*a)[k].
    Result is 2 * cross_correlation(w, a) truncated to first n elements.
    Uses FFT for efficiency.
    """
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    Fw = np.fft.rfft(weights, N)
    Fa = np.fft.rfft(a, N)
    result = np.fft.irfft(Fw * np.conj(Fa), N)[:n]
    # Clean tiny negative values
    # Actually the gradient can be negative in principle? No, weights >= 0 and a >= 0,
    # so cross-correlation should be >= 0. But numerically might have tiny negatives.
    return 2.0 * result


def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def alt_projection_init(n, rng, n_iter=60):
    """Alternating projection for flat-spectrum non-negative sequence."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        mag_target = np.ones(len(F_full))
        weights = np.linspace(1.0, 0.8, len(F_full))
        mag_target = weights ** 0.05
        F = mag_target * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j

    return a


def adam_log_space(z_init, deadline, T_start=5.0, T_end=500.0,
                   lr=0.05, beta1=0.8, beta2=0.999, eps=1e-8,
                   eval_interval=30, perturb_interval=500, perturb_scale=0.05,
                   rng=None, phase_label=""):
    """
    Adam optimizer in log-space (softmax parameterization).
    Minimizes 2*n*soft_max(conv(a,a), T) where a = softmax(z).
    T is annealed from T_start to T_end over the course of optimization.
    """
    if rng is None:
        rng = np.random.default_rng()
    
    n = len(z_init)
    z = z_init.copy().astype(np.float64)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    a_init = softmax(z)
    best_a = a_init.copy()
    best_score = evaluate_sequence(a_init.tolist())
    
    start_time = time.time()
    last_print = 0
    
    while time.time() < deadline - 3:
        elapsed = time.time() - start_time
        # Anneal T: use a fixed total duration for annealing
        anneal_duration = 300.0  # seconds
        progress = min(1.0, elapsed / anneal_duration)
        T = T_start * (T_end / T_start) ** progress
        
        a = softmax(z)
        c = fast_convolve(a)
        weights = soft_max_weights(c, T)
        
        # Gradient w.r.t. a
        grad_a = convolve_grad_fft(weights, a)
        grad_a *= 2.0 * n
        
        # Convert to gradient w.r.t. z
        mean_grad = np.dot(grad_a, a)
        grad_z = a * (grad_a - mean_grad)
        
        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        # Learning rate with decay
        lr_t = lr / (1.0 + 0.05 * np.sqrt(t))
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic evaluation
        if t % eval_interval == 0:
            a_check = softmax(z)
            score = evaluate_sequence(a_check.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a_check.copy()

                    print(f"  {phase_label} iter {t}: score={best_score:.6f}, T={T:.1f}, elapsed={elapsed:.0f}s")
                    last_print = time.time()
        
        # Random perturbation to escape local minima

            if time.time() < deadline - 60:
                scale = perturb_scale * (1.0 - progress) + 0.01
                z = z + rng.normal(0, scale, n)
    
    return best_a, best_score, z

"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses Adam optimizer in log-space (softmax parameterization) with annealed
log-sum-exp approximation of the max, FFT-based gradient computation,
multi-phase search with restarts, and spectral shaping initialization.
"""

import numpy as np
import time


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def softmax_stable(x):
    """Numerically stable softmax."""
    x = x - np.max(x)
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def soft_max_weights(c, T):
    """Compute weights = softmax(T*c) for gradient of log-sum-exp."""
    Tc = T * c
    Tc_shifted = Tc - np.max(Tc)
    exp_Tc = np.exp(Tc_shifted)
    s = exp_Tc.sum()
    if s < 1e-300:
        return np.ones(len(c)) / len(c)
    return exp_Tc / s


def convolve_grad_fft(weights, a):
    """
    Compute gradient of sum_k w_k * c_k w.r.t. a, where c = convolve(a,a).
    Uses FFT cross-correlation for O(n log n) computation.
    Returns: 2 * (w cross-correlate a) truncated to n elements.
    """
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    Fw = np.fft.rfft(weights, N)
    Fa = np.fft.rfft(a, N)
    result = np.fft.irfft(Fw * np.conj(Fa), N)[:n]
    return 2.0 * result


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def alt_projection_init(n, rng, n_iter=50):
    """Alternating projection for near-flat-spectrum non-negative sequence."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j
    a = np.ones(n) / n
    for _ in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        mag_target = np.ones(len(F_full))
        w = np.linspace(1.0, 0.85, len(F_full))
        mag_target = w ** 0.05
        F = mag_target * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j
    return a


def adam_log_space(z_init, deadline, T_start=10.0, T_end=500.0,
                   lr=0.04, beta1=0.8, beta2=0.999, eps=1e-8,
                   eval_interval=25, perturb_interval=300, perturb_scale=0.03,
                   anneal_duration=250.0, grad_clip=10.0,
                   rng=None, phase_label=""):
    """
    Adam optimizer in log-space with softmax parameterization.
    Minimizes 2*n*soft_max(conv(a,a), T) where a=softmax(z).
    T is annealed from T_start to T_end over anneal_duration seconds.
    """
    if rng is None:
        rng = np.random.default_rng()

    n = len(z_init)
    z = z_init.copy().astype(np.float64)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    a_init = softmax_stable(z)
    best_a = a_init.copy()
    best_score = evaluate_sequence(a_init.tolist())

    start_time = time.time()
    last_improve_time = start_time
    last_print = start_time

    while time.time() < deadline - 3:
        elapsed = time.time() - start_time
        progress = min(1.0, elapsed / anneal_duration)
        T = T_start * (T_end / T_start) ** progress

        a = softmax_stable(z)
        c = fast_convolve(a)
        weights = soft_max_weights(c, T)

        # Gradient w.r.t. a
        grad_a = convolve_grad_fft(weights, a)
        grad_a *= 2.0 * n

        # Gradient w.r.t. z
        mean_grad = np.dot(grad_a, a)
        grad_z = a * (grad_a - mean_grad)

        # Gradient clipping
        max_gz = np.max(np.abs(grad_z))
        if max_gz > grad_clip:
            grad_z = grad_z * (grad_clip / max_gz)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr_t = lr / (1.0 + 0.02 * np.sqrt(t))
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic evaluation
        if t % eval_interval == 0:
            a_check = softmax_stable(z)
            c_check = fast_convolve(a_check)
            score = evaluate_sequence(a_check.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a_check.copy()
                last_improve_time = time.time()
                if time.time() - last_print > 15:
                    print(f"    {phase_label} iter={t} score={best_score:.6f} T={T:.1f} t={elapsed:.0f}s")
                    last_print = time.time()

        # Stagnation detection and restart
        if perturb_interval > 0 and t % perturb_interval == 0:
            if time.time() < deadline - 60:
                stagnation = time.time() - last_improve_time
                if stagnation > 30:
                    scale = perturb_scale * min(1.0, stagnation / 100.0) + 0.01
                    z = z + rng.normal(0, scale, n)

    a_final = softmax_stable(z)
    final_score = evaluate_sequence(a_final.tolist())
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
        best_a = a_final.copy()

    return best_a, best_score


def run_adam_phase(a_init, deadline, rng, T_start, T_end, lr,
                   perturb_interval, perturb_scale, anneal_duration, phase_label):
    """Run one Adam optimization phase."""
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / a_init.sum()
    z_init = np.log(a_init)
    return adam_log_space(z_init, deadline, T_start=T_start, T_end=T_end,
                          lr=lr, eval_interval=25,
                          perturb_interval=perturb_interval,
                          perturb_scale=perturb_scale,
                          anneal_duration=anneal_duration,
                          rng=rng, phase_label=phase_label)


def try_multi_T(a_init, deadline, rng, phase_label=""):
    """Try multiple T ranges and keep the best result."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / a_init.sum()
    z_init = np.log(a_init)
    best_a = a_init.copy()
    best_score = evaluate_sequence(a_init.tolist())

    # Three T ranges: low (exploration), medium, high (exploitation)
    configs = [
        (5.0, 50.0, 0.05, 300, 0.04, 200.0),
        (20.0, 200.0, 0.03, 250, 0.02, 200.0),
        (100.0, 800.0, 0.015, 0, 0.0, 150.0),
    ]

    # Allocate time proportionally
    total_time = max(0, deadline - time.time() - 5)
    time_fracs = [0.35, 0.35, 0.30]

    for i, (t_s, t_e, lr, pi, ps, ad) in enumerate(configs):
        if time.time() > deadline - 10:

        sub_deadline = time.time() + total_time * time_fracs[i]
        sub_deadline = min(sub_deadline, deadline - 5)
        if sub_deadline - time.time() < 20:
            continue

        a_best, score = run_adam_phase(a_init, sub_deadline, rng,

                                       f"{phase_label}_T{i}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_best.copy()
            # Use best result as starting point for next T range
            a_init = np.maximum(a_best, 1e-15)

def subgradient_polish(a_init, deadline, rng, n_max_iters=200000):
    """
    Projected subgradient method with adaptive step size.
    Used as a polishing step after Adam.
    """
    n = len(a_init)
    a = a_init.copy()
    a = project_simplex(a)

    best_a = a.copy()
    best_peak = np.max(fast_convolve(a))
    best_score = evaluate_sequence(a.tolist())

    base_lr = 0.8 / n
    lr = base_lr
    stall = 0
    it = 0

    while time.time() < deadline - 3 and it < n_max_iters:
        it += 1
        c = fast_convolve(a)
        peak = np.max(c)

        if peak < best_peak * (1 - 1e-8):
            best_peak = peak
            best_a = a.copy()
            score = evaluate_sequence(a.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
            stall = 0
        else:
            stall += 1

        if stall > 500:
            lr = max(lr * 0.6, base_lr * 1e-4)
            stall = 0
            if lr < base_lr * 0.01 and time.time() < deadline - 30:
                a = project_simplex(a + rng.normal(0, 0.03 / np.sqrt(n), n))

        threshold = peak * (1.0 - 1e-5)
        peak_idx = np.where(c >= threshold)[0]
        if len(peak_idx) > 30:
            top = np.argsort(c[peak_idx])[::-1][:12]
            peak_idx = peak_idx[top]

        grad = np.zeros(n)
        for k in peak_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        grad /= max(1, len(peak_idx))

        a = project_simplex(a - lr * grad)

    return best_a, best_score
