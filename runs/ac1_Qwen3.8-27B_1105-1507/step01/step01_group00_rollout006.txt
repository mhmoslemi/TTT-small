import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    # Pre-allocate
    # Number of nonzeros: for each k (0 to 2n-2), the number of j's is min(k+1, n, 2n-1-k)
    # Total nonzeros = sum_{k=0}^{2n-2} min(k+1, n, 2n-1-k)
    # = sum_{k=0}^{n-1} (k+1) + sum_{k=n}^{2n-2} (2n-1-k)
    # = n(n+1)/2 + (n-1)n/2 = n^2
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, active_indices=None):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)[k] <= M for k in active_indices."""
    n = len(f)
    A_full = build_conv_matrix_sparse(f)
    
    if active_indices is not None and len(active_indices) < 2 * n - 1:
        A = A_full[active_indices, :]
        b_ub = np.full(len(active_indices), M)
    else:
        A = A_full
        b_ub = np.full(2 * n - 1, M)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def active_set_lp(f, M, tolerance=0.1, max_active=None):
    """Solve LP using only active constraints (those where conv is close to M)."""
    conv_f = np.convolve(f, f)
    # Active constraints: those where conv_f[k] > M * (1 - tolerance)
    active = np.where(conv_f > M * (1 - tolerance))[0]
    if max_active is not None and len(active) > max_active:
        # Keep only the most active
        top_idx = np.argsort(conv_f[active])[-max_active:]
        active = active[top_idx]
    return solve_lp_direction(f, M, active_indices=active)


def lp_step(f, M, step_size, use_active_set=True, active_tolerance=0.15):
    """Perform one LP step: find direction g and blend."""
    n = len(f)
    
    if use_active_set:
        g = active_set_lp(f, M, tolerance=active_tolerance, max_active=200)
    else:
        g = solve_lp_direction(f, M)
    
    if g is None:
        return f, False
    
    sum_g = np.sum(g)
    if sum_g < 1e-15:
        return f, False
    
    # Rescale to match sum of f
    sum_f = np.sum(f)
    g = g * (sum_f / sum_g)
    
    # Blend
    new_f = (1.0 - step_size) * f + step_size * g
    new_f = np.clip(new_f, 0.0, 1000.0)
    
    return new_f, True

def perturb(f, rng, magnitude=0.1):
    """Add random perturbation to escape local minima."""
    n = len(f)
    # Random smooth perturbation
    noise = np.zeros(n)
    for _ in range(3):
        center = rng.integers(0, n)
        width = rng.integers(n // 10, n // 3)
        x = np.arange(n)
        noise += magnitude * np.exp(-0.5 * ((x - center) / width) ** 2)
    
    perturbed = f + noise
    perturbed = np.clip(perturbed, 0, 1000.0)
    
    # Renormalize
    if np.sum(perturbed) > 0:
        perturbed = perturbed * (np.sum(f) / np.sum(perturbed))
    
    return perturbed


def smooth_max_gradient(f, T=100.0):
    """Compute gradient of soft-max approximation of max(conv(f,f))."""
    n = len(f)
    conv_f = np.convolve(f, f)
    
    # Soft max weights
    # Numerical stability: subtract max
    max_conv = np.max(conv_f)
    weights = np.exp(T * (conv_f - max_conv))
    weights /= np.sum(weights)
    
    # Gradient: d/d f[j] of soft_max = sum_k weights[k] * d(conv[k])/d(f[j])
    # d(conv[k])/d(f[j]) = 2*f[k-j] if 0 <= k-j < n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * weights[k] * f[k - j]
    
    return grad


def smooth_max_descent(f, deadline, T_init=50.0, T_max=2000.0, lr_init=0.01, max_iters=10000):
    """Gradient descent on soft-max approximation."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    T = T_init
    lr = lr_init
    sum_f = np.sum(f)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad = smooth_max_gradient(f, T)
        
        # Project gradient onto the hyperplane sum(f) = const
        # Subtract the mean to maintain the sum
        grad = grad - np.mean(grad)

        new_f = f - lr * grad
        new_f = np.clip(new_f, 0, 1000.0)
        
        # Renormalize to maintain sum

def coordinate_descent(f, deadline, max_rounds=100):
    """Simple coordinate descent: try small perturbations at each position."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    n = len(f)
    sum_f = np.sum(f)
    rng = np.random.default_rng(123)
    
    delta = 0.01 * sum_f / n  # Small perturbation size
    
    for round_idx in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            
            # Try increasing
            f[j] += delta
            new_sum = np.sum(f)
            f[j] = f[j] * (sum_f / new_sum) if new_sum > 0 else 0
            score = evaluate_sequence(f.tolist())
            
            if np.isfinite(score) and score < best_score:
                best_f = f.copy()
                best_score = score
                improved = True
            else:
                f[j] = f[j] * (sum_f / np.sum(f)) if np.sum(f) > 0 else 0  # Restore
                # Try decreasing
                old_val = f[j]
                f[j] = max(0, f[j] - delta)
                new_sum = np.sum(f)
                if new_sum > 0:
                    f = f * (sum_f / new_sum)
                score = evaluate_sequence(f.tolist())
                if np.isfinite(score) and score < best_score:
                    best_f = f.copy()
                    best_score = score
                    improved = True
                else:
                    f = best_f.copy()  # Restore
            
        if not improved:
            delta *= 0.7
            if delta < 1e-8:
                break
    
    return best_f, best_score
