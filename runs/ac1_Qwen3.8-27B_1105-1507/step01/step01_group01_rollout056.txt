```python
def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

"""
Subgradient optimization for minimizing max autocorrelation of non-negative sequences.
Uses softmax-weighted subgradients, simplex projection, multiplicative updates,
and multiple initialization strategies across different sequence lengths.
"""

import time
import numpy as np

# Helper functions (all top-level)

def compute_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 1e-300:
        return float('inf')
    c = np.convolve(a, a)
    m = float(np.max(c))
    return float(2.0 * n * m / (s * s))

def normalize_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s < 1e-300:
        return np.ones(a.size) / a.size
    return a / s

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n <= 1:
        return np.array([max(0.0, float(np.sum(v)))], dtype=np.float64) if n == 1 else v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    r = np.arange(1, n + 1)
    cond = u * r > (cv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, p):
    """Compute grad_j = 2 * sum_m a[m] * p[j+m]."""
    n = a.size
    p_rev = p[::-1]
    full_conv = np.convolve(a, p_rev)
    # grad_j = 2 * full_conv[2n-2-j]
    idx = 2 * n - 2 - np.arange(n)
    return 2.0 * full_conv[idx]

def softmax_weights(c, beta):
    """Compute softmax weights on the convolution values."""
    c_max = float(np.max(c))
    if c_max <= 0:
        return np.ones(c.size) / c.size
    x = beta * c / c_max
    x -= x.max()
    w = np.exp(x)
    s = w.sum()
    if s < 1e-300:
        return np.ones(c.size) / c.size
    return w / s

def topk_weights(c, K):
    """Uniform weights on top-K peaks."""
    n = c.size
    K = min(K, n)
    w = np.zeros(n)
    idx = np.argpartition(c, -K)[-K:]
    w[idx] = 1.0 / K
    return w

def subgradient_step(a, c, p, eta, mode='subtract'):
    """One optimization step."""
    grad = compute_gradient(a, p)
    if mode == 'subtract':
        a_new = a - eta * grad
        a_new = project_simplex(a_new)
    else:  # multiplicative
        max_grad = float(np.max(grad))
        if max_grad > 0:
            a_new = a * np.exp(-eta * grad / max_grad)
        else:
            a_new = a.copy()
        a_new = normalize_sum1(a_new)
    return a_new
