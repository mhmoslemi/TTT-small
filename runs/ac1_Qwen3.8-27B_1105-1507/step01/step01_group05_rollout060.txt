```python
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def compute_score(f):
    """Compute 2*n*max(conv)/sum^2 for sequence f (assumed sum=1)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()

def build_conv_matrix_sparse(f):
    """Build sparse convolution matrix A such that A @ b = conv(f, b)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def lp_direction(f, M):
    """Solve LP to find direction that keeps conv(f, b) <= M."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    return np.clip(res.x, 0, None)

def optimize_sequence(f, time_budget, verbose=True):
    """Main LP-based optimization loop."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f /= f.sum()
    best_f = f.copy()
    best_score = compute_score(f)
    t = 0.3
    start = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Try to find LP direction with current M
        b = lp_direction(f, M)
        if b is None or b.sum() < 1e-15:
            # Try with a tighter M
            M_tight = M * 0.99
            b = lp_direction(f, M_tight)
            if b is None or b.sum() < 1e-15:
                break
        
        b = b / b.sum()
        
        # Line search for step size
        best_t = t
        best_new_score = np.inf
        for trial_t in [t, t*0.5, t*1.5, t*0.1]:
            f_new = (1-trial_t)*f + trial_t*b
            f_new = np.clip(f_new, 0, None)
            if f_new.sum() < 1e-10:
                continue
            f_new /= f_new.sum()
            s = compute_score(f_new)
            if s < best_new_score:
                best_new_score = s
                best_t = trial_t
        
        if best_new_score < best_score - 1e-10:
            f = f.copy()
            f = (1-best_t)*f + best_t*b
            f = np.clip(f, 0, None)
            f /= f.sum()
            best_f = f.copy()
            best_score = compute_score(f)
            stall = 0
            t = min(0.5, best_t * 1.1)
        else:
            t *= 0.7
            stall += 1
            if stall > 30:
                # Random perturbation
                f = f + np.random.normal(0, 0.01, n)
                f = np.clip(f, 0, None)
                f /= f.sum()
                t = 0.3
                stall = 0
        
        it += 1
        if verbose and it % 10 == 0:
            print(f"  it={it}, score={best_score:.6f}, t={t:.4f}")
    
    return best_f, best_score
```

```python
        f0 /= f0.sum()

        time_here = min(frac / total_frac * total_time, deadline - time.monotonic() - 30)
        if time_here < 15:
            continue

        print(f"\n['{name}'] n={len(f0)}, budget={time_here:.0f}s")
        t_start = time.monotonic()

        f_opt, score = optimize_lp(f0, time_here, verbose=True, rng=rng)

        # Peak shaving
        time_left = deadline - time.monotonic()
        if time_left > 10 and np.isfinite(score):
            n_ps = min(5000, int(time_left * 800))
            f_ps, score_ps = peak_shave(f_opt, rng, n_iter=n_ps)
            if score_ps < score:
                f_opt = f_ps
                score = score_ps
                print(f"  After peak shave: {score:.6f}")

        # Local search
        time_left = deadline - time.monotonic()
        if time_left > 5 and np.isfinite(score):
            n_ls = min(10000, int(time_left * 3000))
            f_ls, score_ls = local_search(f_opt, rng, n_iter=n_ls)
            if score_ls < score:
                f_opt = f_ls
                score = score_ls
                print(f"  After local search: {score:.6f}")

        elapsed = time.monotonic() - t_start
        if np.isfinite(score):
            print(f"  [{name}] final: {score:.6f} (took {elapsed:.1f}s)")
            if score < best_score_overall:
                best_score_overall = score
                best_overall = f_opt.copy()
                print(f"  ** NEW OVERALL BEST: {score:.6f} **")

    # Final intensive polishing on the best sequence found
    if best_overall is not None:
        time_left = deadline - time.monotonic()
        print(f"\n=== Final polishing phase, time_left={time_left:.0f}s ===")

        if time_left > 30:
            # Another round of LP on the best
            t_lp = min(time_left * 0.5, 300)
            f_opt, score = optimize_lp(best_overall, t_lp, verbose=True, rng=rng)
            if score < best_score_overall:
                best_overall = f_opt
                best_score_overall = score
                print(f"  After final LP: {score:.6f}")

        # Peak shaving
        time_left = deadline - time.monotonic()
        if time_left > 10:
            n_ps = min(10000, int(time_left * 1000))
            f_ps, score_ps = peak_shave(best_overall, rng, n_iter=n_ps)
            if score_ps < best_score_overall:
                best_overall = f_ps
                best_score_overall = score_ps
                print(f"  After final peak shave: {score_ps:.6f}")

        # Local search
        time_left = deadline - time.monotonic()
        if time_left > 5:
            n_ls = min(50000, int(time_left * 5000))
            f_ls, score_ls = local_search(best_overall, rng, n_iter=n_ls)
            if score_ls < best_score_overall:
                best_overall = f_ls
                best_score_overall = score_ls
                print(f"  After final local search: {score_ls:.6f}")

        # Verify with evaluate_sequence
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal verified score: {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        return final_list

    # Fallback
    print("WARNING: Fallback to uniform sequence")
    return [1.0] * 256
```
