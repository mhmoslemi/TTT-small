"""
Hybrid subgradient + LP optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses projected subgradient descent for fast local search, LP direction finding for
larger moves, and multiple diverse initializations. Combines these within a time budget.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute score = 2*n*max(conv)/sum^2."""
    n = len(f)
    S = f.sum()
    if S < 1e-10:
        return np.inf
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M / (S * S)


def compute_conv(f):
    """Compute full convolution."""
    return np.convolve(f, f)


def subgradient_of_max_conv(f):
    """
    Compute subgradient of max_k (f*f)_k with respect to f.
    Returns (g, active_indices) where g is the subgradient.
    """
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    # Active set: indices where conv is close to max
    tol = 1e-6 * M
    active = np.where(conv >= M - tol)[0]
    
    # Subgradient contribution from each active index
    g = np.zeros(n)
    for k in active:
        for i in range(n):
            idx = k - i
            if 0 <= idx < n:
                g[i] += 2.0 * f[idx]
    g /= len(active)
    return g, active, M


def project_simplex(v):
    """Project vector v onto the simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_descent(f, rng, n_iter=5000, eta0=0.05, decay=0.9999):
    """
    Projected subgradient descent to minimize max(conv(f,f)) subject to f>=0, sum(f)=1.
    """
    f = f.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-10:
        return f, np.inf
    f = f / S
    
    best_f = f.copy()
    best_score = compute_score(f)
    eta = eta0
    momentum = np.zeros_like(f)
    beta = 0.7
    
    for it in range(n_iter):
        g, active, M = subgradient_of_max_conv(f)
        
        # Momentum
        momentum = beta * momentum + g
        step = eta * momentum
        
        # Try multiple step sizes
        best_step_f = None
        best_step_score = np.inf
        for scale in [1.0, 0.5, 0.25, 2.0]:
            f_candidate = f - scale * step
            f_candidate = project_simplex(f_candidate)
            score = compute_score(f_candidate)
            if score < best_step_score:
                best_step_score = score
                best_step_f = f_candidate
        
        if best_step_f is not None and best_step_score < best_score:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
        else:
            eta *= 0.95
        
        # Occasionally add noise to escape local minima
        if it % 500 == 499:
            f = f + eta * 0.1 * rng.normal(0, 1, len(f))
            f = np.clip(f, 0, None)
            S = f.sum()
            if S > 1e-10:
                f = f / S
    
    return best_f, best_score


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_direction(f, M_scale=1.0):
    """
    Solve LP: max sum(b) s.t. convolve(f,b) <= M*M_scale, b >= 0.
    Returns the optimal b or None.
    """
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max() * M_scale
    if M < 1e-15:
        return None
    
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': 10})
    except Exception:
        return None
    
    if not res.success or res.x is None:
        return None
    
    b = np.clip(res.x, 0, None)
    if b.sum() < 1e-15:
        return None
    return b


def lp_optimize(f, time_budget, t0=0.4, M_scales=(1.0, 1.01, 1.05), verbose=False, rng=None):
    """
    LP-based iterative optimization. Alternates between LP directions with different M scales.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-10:
        return f, np.inf
    f = f / S
    
    best_f = f.copy()
    best_score = compute_score(f)
    t = t0
    start = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Try different M scales
        best_g = None
        best_improvement = 0
        for ms in M_scales:
            b = lp_direction(f, M_scale=ms)
            if b is None:
                continue
            b = b / b.sum()
            # Try the direction
            f_new = (1 - t) * f + t * b
            f_new = np.clip(f_new, 0, None)
            S_new = f_new.sum()
            if S_new < 1e-10:
                continue
            f_new = f_new / S_new
            score = compute_score(f_new)
            improvement = best_score - score
            if improvement > best_improvement:
                best_improvement = improvement
                best_g = f_new
        
        if best_g is not None and best_improvement > 1e-12:
            f = best_g
            best_f = f.copy()
            best_score = compute_score(f)
            stall = 0
            if verbose and it % 10 == 0:
                elapsed = time.monotonic() - start

        else:

            stall += 1

                if S_p > 1e-10:

                    if compute_score(f_pert) < best_score:

                        best_f = f.copy()
                        best_score = compute_score(f)
                        t = t0
                        stall = 0
                if time.monotonic() - start > time_budget * 0.9:

    return best_f, best_score


def gradient_polish(f, n_iter=2000, eta0=0.01):
    """
    Fine-grained gradient-based polishing using the subgradient formula.
    Uses diminishing step sizes for convergence.
    """
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-10:
        return f, np.inf
    f = f / S
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    for it in range(n_iter):
        g, active, M = subgradient_of_max_conv(f)
        eta = eta0 / (1 + it * 0.01)
        
        f_new = f - eta * g
        f_new = project_simplex(f_new)
        score = compute_score(f_new)
        
        if score < best_score:
            f = f_new
            best_f = f.copy()
            best_score = score
    
    return best_f, best_score


def random_block_search(f, rng, n_iter=3000):
    """
    Random block perturbation local search.
    """
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()

        return f, np.inf
    f = f / S
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    for it in range(n_iter):
        f_cand = f.copy()
        width = max(1, int(rng.integers(1, max(2, n // 16))))
        start_idx = int(rng.integers(0, n - width))
        scale = np.exp(rng.normal(0, 0.3))
        f_cand[start_idx:start_idx + width] *= scale
        f_cand = np.clip(f_cand, 0, None)
        S_c = f_cand.sum()

        if score < best_score:

            best_score = score
            f = f_cand
    
    return best_f, best_score
