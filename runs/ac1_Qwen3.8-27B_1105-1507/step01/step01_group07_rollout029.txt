```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_lp(a, M):
    n = len(a)
    A = build_A_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None
```

```python
def smooth_objective(a, T=0.1):
    """Smooth approximation of max(conv(a,a)) using logsumexp."""
    n = len(a)
    conv = np.convolve(a, a)
```

```python
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, hstack
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A_sparse(a):
    """Build sparse convolution constraint matrix."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def lp_solve(fixed_a):
    """Solve LP: min M s.t. conv(fixed_a, g)_k <= M for all k, sum(g)=1, g>=0."""
    n = len(fixed_a)
    A_g = build_A_sparse(fixed_a)
    
    # Build [A_g | -1] matrix
    neg_col = -csr_matrix((np.ones(2*n-1), 
                           (np.arange(2*n-1), np.full(2*n-1, n))), 
                          shape=(2*n-1, n+1))
    A_ub = hstack([A_g, neg_col])
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n+1)
    c[n] = 1.0
    
    bounds = [(0, None)] * (n+1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    if res.status == 0:
        return res.x[:n], res.x[n]
    return None, None

def run_alternating_lp(a, max_iter=500, t=0.5, tol=1e-6, time_limit=None, start_time=None):
    """Run alternating LP to minimize max(conv(a,a)) with sum(a)=1."""
    a = a / np.sum(a)
    best_a = a.copy()
    best_M = float(np.max(np.convolve(a, a)))

        g, M_new = lp_solve(a)
        if g is None:

            break
        
        # Update

        a_new = np.clip(a_new, 0, None)

            best_M = M_after

def smooth_max_grad(a, T=0.01):
    """Compute smooth max approximation and its gradient."""
    n = len(a)
    conv = np.convolve(a, a)
    
    # logsumexp
    conv_scaled = conv / T
    conv_max = np.max(conv_scaled)
    lse = T * (conv_max + np.log(np.sum(np.exp(conv_scaled - conv_max))))
    
    # softmax weights
    weights = np.exp(conv_scaled - conv_max)
    weights /= np.sum(weights)
    
    # Gradient of lse w.r.t. a[i]
    # = 2 * sum_k weights[k] * a[k-i] (for valid k-i)
    # = 2 * correlate(weights, a)[i]
    corr = np.correlate(weights, a, mode='full')
    d_lse = 2.0 * corr[:n]
    
    return lse, d_lse

def adam_refine(a, max_iter=1000, lr=0.01, T_start=0.1, T_end=0.001,
                time_limit=None, start_time=None):
    """Refine using Adam with smooth logsumexp objective."""
    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    for it in range(max_iter):
        if time_limit and time.monotonic() - start_time > time_limit:
            break
        
        # Anneal temperature
        progress = it / max_iter
        T = T_start * (T_end / T_start) ** progress
        
        lse, d_lse = smooth_max_grad(a, T)
        sum_a = np.sum(a)
        
        # Full gradient of objective 2n*lse/(sum)^2
        grad = 2.0 * n * (d_lse / (sum_a ** 2) - 2.0 * lse / (sum_a ** 3))
        
        # Adam update
        t_idx = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_idx)
        v_hat = v / (1 - beta2 ** t_idx)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        
        a = a_new
    
    return best_a, best_obj
```

```python
def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_subproblem(fixed_a):
    """Solve: min M s.t. conv(fixed_a, g)_k <= M, sum(g)=1, g>=0."""
    n = len(fixed_a)
    A_g = build_A_sparse(fixed_a)
    neg_col = -csr_matrix(
        (np.ones(2 * n - 1), (np.arange(2 * n - 1), np.full(2 * n - 1, n))),
        shape=(2 * n - 1, n + 1))
    A_ub = hstack([A_g, neg_col])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(0, None)] * (n + 1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n], res.x[n]
    return None, None

    best_M = float(np.max(np.convolve(a, a)))

def smooth_obj_grad(a, T):
    n = len(a)
    conv = np.convolve(a, a)
    cs = conv / T
    cs_max = np.max(cs)
    lse = T * (cs_max + np.log(np.sum(np.exp(cs - cs_max))))
    w = np.exp(cs - cs_max)
    w /= np.sum(w)
    corr = np.correlate(w, a, mode='full')
    d_lse = 2.0 * corr[:n]
    return lse, d_lse


def adam_refine(a_init, max_iter=3000, lr=0.01, T_start=0.05, T_end=0.002,
                time_limit=60.0, start_time=None):

    a = a_init / np.sum(a_init)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    if start_time is None:
        start_time = time.monotonic()
    for it in range(max_iter):
        if time.monotonic() - start_time > time_limit:
            break
        progress = it / max(max_iter, 1)
        T = T_start * (T_end / T_start) ** progress
        lse, d_lse = smooth_obj_grad(a, T)
        s = np.sum(a)
        grad = 2.0 * n * (d_lse / (s * s) - 2.0 * lse / (s * s * s))
        t_i = it + 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t_i)
        vh = v / (1 - b2 ** t_i)
        a_new = a - lr * mh / (np.sqrt(vh) + eps)
        a_new = np.clip(a_new, 0, None)
        s2 = np.sum(a_new)
        if s2 < 1e-10:

        a_new = a_new / s2
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
    return best_a, best_obj
```

    b_eq = np.array([1.0])
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(0, None)] * (n + 1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n], res.x[n]
    return None, None


def run_lp(a_init, max_iter=500, time_limit=60.0, start_time=None):
    a = a_init / np.sum(a_init)
    best_a = a.copy()
    best_M = float(np.max(np.convolve(a, a)))
    if start_time is None:
        start_time = time.monotonic()
    stall = 0
    for it in range(max_iter):
        if time.monotonic() - start_time > time_limit:
            break
        current_M = float(np.max(np.convolve(a, a)))
        g, M_lp = lp_subproblem(a)
        if g is None:
            break
        if M_lp >= current_M - 1e-8:
            stall += 1
            if stall > 5:
                break
            continue
        stall = 0
        best_t = 0.5
        best_trial_M = best_M
        for trial_t in [0.2, 0.4, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - trial_t) * a + trial_t * g
            a_try = np.clip(a_try, 0, None)
            s = np.sum(a_try)
            if s < 1e-10:
                continue
            a_try = a_try / s
            M_try = float(np.max(np.convolve(a_try, a_try)))
            if M_try < best_trial_M:
                best_trial_M = M_try
                best_t = trial_t
        a = (1 - best_t) * a + best_t * g
        a = np.clip(a, 0, None)
        s = np.sum(a)
        if s < 1e-10:
            break
        a = a / s
        M_now = float(np.max(np.convolve(a, a)))
        if M_now < best_M:
            best_M = M_now
            best_a = a.copy()
    return best_a, best_M


def smooth_obj_grad(a, T):
    n = len(a)
    conv = np.convolve(a, a)
    cs = conv / T
    cs_max = np.max(cs)
    lse = T * (cs_max + np.log(np.sum(np.exp(cs - cs_max))))
    w = np.exp(cs - cs_max)
    w /= np.sum(w)
    corr = np.correlate(w, a, mode='full')
    d_lse = 2.0 * corr[:n]
    return lse, d_lse


def adam_refine(a_init, max_iter=3000, lr=0.01, T_start=0.05, T_end=0.002,
                time_limit=60.0, start_time=None):
    n = len(a_init)
    a = a_init / np.sum(a_init)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    if start_time is None:
        start_time = time.monotonic()
    for it in range(max_iter):
        if time.monotonic() - start_time > time_limit:
            break
        progress = it / max(max_iter, 1)
        T = T_start * (T_end / T_start) ** progress
        lse, d_lse = smooth_obj_grad(a, T)
        s = np.sum(a)
        grad = 2.0 * n * (d_lse / (s * s) - 2.0 * lse / (s * s * s))
        t_i = it + 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t_i)
        vh = v / (1 - b2 ** t_i)
        a_new = a - lr * mh / (np.sqrt(vh) + eps)
        a_new = np.clip(a_new, 0, None)
        s2 = np.sum(a_new)
        if s2 < 1e-10:
            break
        a_new = a_new / s2
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
    return best_a, best_obj


def local_search(a_init, rng, max_iter=3000, time_limit=30.0, start_time=None):
    n = len(a_init)
    best_a = a_init.copy()
    best_obj = compute_obj(best_a)
    if start_time is None:
        start_time = time.monotonic()
    for it in range(max_iter):
        if time.monotonic() - start_time > time_limit:
            break
        width = max(1, n // 40)
        start = int(rng.integers(0, n - width + 1))
        cand = best_a.copy()
        sigma = 0.06
        cand[start:start + width] *= np.exp(rng.normal(0, sigma, width))
        cand = np.clip(cand, 0, None)
        s = np.sum(cand)
        if s < 1e-10:
            continue
        cand = cand / s
        obj = compute_obj(cand)
        if obj < best_obj:
            best_a = cand
            best_obj = obj
    return best_a, best_obj


def init_uniform(n):
    return np.ones(n)


def init_random(n, rng):
    a = rng.random(n)
    return a / a.sum()


def init_cosine(n, rng):
    freq = rng.uniform(1.0, 4.0)
    phase = rng.uniform(0, 2 * np.pi)
    a = np.maximum(0.05, np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    return a / a.sum()


def init_periodic(n, rng):
    p = max(2, n // int(rng.integers(5, 15)))
    pattern = np.maximum(0.1, rng.random(p))
    a = np.tile(pattern, n // p + 1)[:n]
    return a / a.sum()


def init_bimodal(n, rng):
    a = np.ones(n)
    a[n // 5:n - n // 5] = rng.uniform(0.2, 0.6)
    return a / a.sum()


def init_from_parent(parent, n, rng):
    if parent is None or parent.size == 0:
        return init_random(n, rng)
    if parent.size >= n:
        a = parent[:n].copy()
    else:
        reps = n // parent.size + 1
        a = np.tile(parent, reps)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return init_random(n, rng)
    noise = np.exp(rng.normal(0, 0.3, n))
    a = a * noise
    s2 = a.sum()
    if s2 < 1e-10:
        return init_random(n, rng)
    return a / s2
