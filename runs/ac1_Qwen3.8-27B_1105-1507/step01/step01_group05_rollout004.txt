import numpy as np
from scipy.optimize import linprog

def compute_score(f):
    """Compute score = 2*n*max(conv)/sum^2."""
    n = len(f)
    S = f.sum()
    if S < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M / (S * S)

def compute_score_normalized(f):
    """Compute score assuming f is already normalized (sum=1)."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M

def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A

def solve_lp_direction(f, M, verbose=False):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    
    if not res.success or res.x is None:
        return None
    return np.clip(res.x, 0, None)

def line_search(f, g, best_score, max_iter=50):

            best_t = d

            best_s = fc
            best_t = c

def score_of_blend(f, g, t):

Hybrid LP + peak-shaving optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP direction finding with line search, direct peak reduction,
gradient polishing, multi-length search, and random restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute score = 2*n*max(conv)/sum^2 for arbitrary f."""
    n = len(f)
    S = f.sum()
    if S < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M / (S * S)


def compute_score_norm(f):
    """Compute score assuming f is normalized (sum=1). Returns 2*n*max_conv."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def solve_lp(f, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if not res.success or res.x is None:
            return None
        g = np.clip(res.x, 0, None)
        if g.sum() < 1e-15:
            return None
        return g / g.sum()
    except Exception:
        return None


def line_search_blend(f, g, best_score, n_iter=30):
    """Golden section search for best t minimizing score of (1-t)*f + t*g."""
    gr = (np.sqrt(5) - 1) / 2
    a, b = 0.0, 1.0
    best_t = 0.0
    best_s = best_score

    for _ in range(n_iter):
        c = b - gr * (b - a)
        d = a + gr * (b - a)
        h_c = np.clip((1 - c) * f + c * g, 0, None)
        h_d = np.clip((1 - d) * f + d * g, 0, None)
        S_c = h_c.sum()
        S_d = h_d.sum()
        if S_c < 1e-15 or S_d < 1e-15:
            break
        s_c = compute_score_norm(h_c / S_c)
        s_d = compute_score_norm(h_d / S_d)
        if s_c < best_s:
            best_s = s_c
            best_t = c
        if s_d < best_s:
            best_s = s_d
            best_t = d
        if s_c < s_d:
            b = d
        else:
            a = c
        if abs(b - a) < 1e-6:
            break
    return best_t, best_s


def peak_shave(f, n_shaves=50, strength=0.05, rng=None):
    """Directly reduce peaks in the convolution."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    f = f / S
    best = f.copy()
    best_score = compute_score_norm(f)

    for _ in range(n_shaves):
        conv = np.convolve(f, f)
        M = conv.max()
        # Find top lags
        top_lags = np.argsort(conv)[-3:]
        for k in top_lags:
            if conv[k] < M * 0.95:
                continue
            # Gradient of (f*f)(k) w.r.t. f
            grad = np.zeros(n)
            for j in range(n):
                partner = k - j
                if 0 <= partner < n:
                    grad[j] += f[partner]
            # Reduce f in direction of gradient (where it contributes most to peak)
            f_new = f - strength * grad / (np.max(grad) + 1e-15)
            f_new = np.clip(f_new, 0, None)
            S_new = f_new.sum()
            if S_new < 1e-10:
                continue
            f_new = f_new / S_new
            s_new = compute_score_norm(f_new)
            if s_new < best_score:
                best = f_new.copy()
                best_score = s_new
                f = f_new
    return best, best_score

        f_opt, score = lp_optimize(f0, time_alloc, t0=0.4, verbose=True, rng=rng,
                                   use_line_search=(n <= 512))

        # Peak shave
        time_left2 = deadline - time.monotonic()
        if time_left2 > 10 and np.isfinite(score):
            f_ps, score_ps = peak_shave(f_opt, n_shaves=min(100, int(time_left2 * 30)),
                                        strength=0.03, rng=rng)
            if score_ps < score:
                f_opt = f_ps
                score = score_ps
                print(f"    After peak shave: {score:.6f}")

        if np.isfinite(score) and score < best_score_overall:
            best_score_overall = score
            best_overall = f_opt.copy()
            print(f"  ** NEW BEST: {score:.6f} (from {name}) **")

    # Phase 3: Polish best solution
    time_left = deadline - time.monotonic()
    if best_overall is not None and time_left > 15:
        print(f"\n=== Phase 3: Polishing (time={time_left:.0f}s) ===")
        half = time_left * 0.5
        f_pol, s_pol = gradient_polish(best_overall, half, rng=rng)
        if s_pol < best_score_overall:
            best_score_overall = s_pol
            best_overall = f_pol.copy()
            print(f"  After gradient polish: {s_pol:.6f}")

        time_left2 = deadline - time.monotonic()
        if time_left2 > 20:
            f_opt2, s2 = lp_optimize(best_overall, time_left2 * 0.7, t0=0.25,
                                     verbose=True, rng=rng, use_line_search=(len(best_overall) <= 512))
            if np.isfinite(s2) and s2 < best_score_overall:
                best_score_overall = s2
                best_overall = f_opt2.copy()
                print(f"  After additional LP: {s2:.6f}")

    # Phase 4: Random restarts
    time_left = deadline - time.monotonic()
    if best_overall is not None and time_left > 25:
        print(f"\n=== Phase 4: Random restarts (time={time_left:.0f}s) ===")
        n_restarts = min(4, int(time_left / 35))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            f0 = best_overall * (1 + 0.25 * rng.normal(0, 1, len(best_overall)))
            f0 = np.clip(f0, 0, None)
            S = f0.sum()
            if S < 1e-10:
                continue
            f0 = f0 / S
            t_left = deadline - time.monotonic() - 10
            if t_left < 15:
                break
            f_opt, score = lp_optimize(f0, t_left, t0=0.35, verbose=False, rng=rng,
                                       use_line_search=(len(f0) <= 512))
            if np.isfinite(score) and score < best_score_overall:
                best_score_overall = score
                best_overall = f_opt.copy()
                print(f"  Restart {r}: NEW BEST {score:.6f}")

    # Final verification
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal verified score: {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        print(f"Total time: {time.monotonic() - t_global_start:.1f}s")
        return final_list

    # Fallback
    print("Fallback: uniform 256")
    return [1.0] * 256
