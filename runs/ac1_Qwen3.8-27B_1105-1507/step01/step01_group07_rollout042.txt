"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max gradient descent, coordinate descent, structured initializations,
and random restarts across multiple sequence lengths.
"""

import numpy as np
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_soft_max(a, tau):
    """Compute gradient of soft-max approximation of the objective."""
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    exp_vals = np.exp((conv - max_conv) / tau)
    w = exp_vals / np.sum(exp_vals)
    L = (max_conv + tau * np.log(np.sum(exp_vals)))  # log-sum-exp
    Ew_conv = np.dot(w, conv)
    # G[i] = sum_k w[k] * a[k-i] (valid indices only)
    # G[i] = sum_{j=0}^{n-1} w[j+i] * a[j]
    G = np.correlate(w, a, mode='valid')
    # Gradient of 2n * L / (sum_a)^2 under sum_a=1:
    # = 2n * (dL/da[i] - 2L)
    # dL/da[i] = (2/tau) * G[i]  ... wait, let me re-derive
    # conv[k] = sum_j a[j]*a[k-j]
    # d(conv[k])/da[i] = 2*a[k-i] (if 0<=k-i<n, else 0)
    # dL/da[i] = (1/tau) * sum_k w[k] * d(conv[k])/da[i] = (2/tau) * sum_k w[k]*a[k-i] = (2/tau)*G[i]
    # d(S)/da[i] where S = L/(sum_a)^2, sum_a=1:
    # = dL/da[i] - 2L = (2/tau)*G[i] - 2L
    # f = 2n*S, so df/da[i] = 2n*((2/tau)*G[i] - 2L)
    grad = 2.0 * n * ((2.0 / tau) * G - 2.0 * L)
    return grad


def grad_hard_max(a):
    """Compute subgradient using the argmax lag."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    # Find all lags achieving max (within tolerance)
    tol = 1e-10 * max_b + 1e-15
    active = conv >= (max_b - tol)
    w = active.astype(float)
    w /= w.sum()
    # G[i] = sum_k w[k]*a[k-i]
    G = np.correlate(w, a, mode='valid')
    # Subgradient of 2n*M/(sum_a)^2 under sum_a=1:
    # = 2n*(grad_M[i] - 2M) where grad_M[i] = 2*G[i]
    grad = 2.0 * n * (2.0 * G - 2.0 * max_b)
    return grad, max_b


def coord_descent_step(a, i):
    """Optimize a single coordinate a[i] via 1D search."""
    n = len(a)
    s_others = a.sum() - a[i]
    # conv[k] as function of x = a[i]:
    # conv[k] = base[k] + 2*x*a[k-i] (for k != 2i, k-i in [0,n-1])
    # conv[2i] = base[2i] + 2*x*a[i] + x^2 - a[i]^2 ... 
    # Actually: old conv[2i] includes a[i]^2. New: (a[i]+(x-a[i]))^2 = x^2
    # So new_conv[2i] = old_conv[2i] - a[i]^2 + x^2
    # For k != 2i: new_conv[k] = old_conv[k] - 2*a[i]*a[k-i] + 2*x*a[k-i]
    a_i = a[i]
    conv_old = np.convolve(a, a)
    
    def eval_x(x):
        conv_new = conv_old.copy()
        for k in range(2 * n - 1):
            ki = k - i
            if 0 <= ki < n:
                if k == 2 * i:
                    conv_new[k] = conv_old[k] - a_i * a_i + x * x
                else:
                    conv_new[k] = conv_old[k] - 2 * a_i * a[ki] + 2 * x * a[ki]
        s_new = s_others + x
        if s_new < 1e-10:
            return np.inf
        return 2.0 * n * np.max(conv_new) / (s_new * s_new)
    
    # Golden section search on [0, hi]
    hi = max(10.0 * a_i + 0.1, 2.0 * s_others / n)
    if a_i > 0:
        lo = 0.0
    else:
        lo = 0.0
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    f1 = eval_x(a1)
    f2 = eval_x(a2)
    
    for _ in range(40):
        if f1 < f2:
            hi = a2
            a2 = a1
            f2 = f1
            a1 = hi - gr * (hi - lo)
            f1 = eval_x(a1)
        else:
            lo = a1
            a1 = a2
            f1 = f2
            a2 = lo + gr * (hi - lo)
            f2 = eval_x(a2)
    
    x_opt = (lo + hi) / 2.0
    return x_opt


def run_coord_descent(a, max_iter, rng, time_limit=None, start_time=None):
    """Run coordinate descent optimization."""
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)
    
    for it in range(max_iter):
        if time_limit is not None and time.monotonic() - start_time > time_limit:
            break
        i = rng.integers(n)
        x_opt = coord_descent_step(best_a, i)
        if x_opt < 0:
            x_opt = 0.0
        new_a = best_a.copy()
        new_a[i] = x_opt
        new_obj = compute_obj(new_a)
        if new_obj < best_obj:
            best_obj = new_obj
            best_a = new_a
    
    return best_a, best_obj


def run_gradient_descent(a, max_iter, tau_init, tau_end, rng, time_limit=None, start_time=None):
    """Run soft-max gradient descent with adaptive step size."""
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)
    momentum = np.zeros(n)
    beta = 0.9
    
    # Annealing schedule for tau
    for it in range(max_iter):
        if time_limit is not None and time.monotonic() - start_time > time_limit:
            break
        # Anneal tau
        frac = it / max(1, max_iter)
        tau = tau_init * (tau_end / tau_init) ** frac
        
        grad = grad_soft_max(best_a, tau)
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad
        momentum_norm = np.linalg.norm(momentum)
        if momentum_norm < 1e-15:
            break
        
        # Step size: backtracking
        lr = 1.0 / (n * momentum_norm) * 0.5
        best_trial = best_a
        best_trial_obj = best_obj
        found_better = False
        
        for back in range(8):
            trial = best_a - lr * momentum
            trial = project_simplex(trial)
            trial_obj = compute_obj(trial)
            if trial_obj < best_obj * (1 - 1e-12):
                best_obj = trial_obj
                best_a = trial
                found_better = True
                break
            lr *= 0.5
        
        if not found_better:
            # Try a small random perturbation
            if it % 50 == 0:
                noise = rng.normal(0, 0.01, n)
                trial = best_a + noise
                trial = project_simplex(trial)
                trial_obj = compute_obj(trial)
                if trial_obj < best_obj:
                    best_obj = trial_obj
                    best_a = trial
            momentum *= 0.5
    
    return best_a, best_obj


def run_subgradient(a, max_iter, rng, time_limit=None, start_time=None):
    """Run hard-max subgradient method with momentum."""
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)
    velocity = np.zeros(n)
    beta = 0.8
    
    for it in range(max_iter):

            break
        
        grad, max_b = grad_hard_max(best_a)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        
        velocity = beta * velocity + (1 - beta) * grad
        vel_norm = np.linalg.norm(velocity)
        if vel_norm < 1e-14:
            break
        
        lr = 0.3 / (n * vel_norm)
        trial = best_a - lr * velocity

            if trial2_obj < best_obj:
                best_obj = trial2_obj
                best_a = trial2

    return best_a, best_obj


def make_init_gaussian(n, rng):
    """Gaussian bump initialization."""
    center = rng.uniform(0.2 * n, 0.8 * n)
    width = rng.uniform(0.1 * n, 0.4 * n)
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def make_init_cosine(n, rng):
    """Cosine-based initialization."""
    phase = rng.uniform(0, 2 * np.pi)
    freq = rng.uniform(0.5, 3.0)
    x = np.arange(n) / n
    a = 1.0 + rng.uniform(0.3, 1.0) * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def make_init_edge_concentrated(n, rng):
    """Concentrated at edges."""
    edge_width = int(rng.integers(max(2, n // 10), max(3, n // 3)))
    a = np.zeros(n)

    a += rng.uniform(0, 0.1, n)
    s = a.sum()
    if s < 1e-10:

def make_init_ramp(n, rng):

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
