Hybrid DE + Spectral Subgradient optimization for minimum autocorrelation peak sequences.
Uses differential evolution for global exploration and FFT-based subgradient refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def score_from_conv(a, conv):
    """Compute score from sequence and its convolution."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return float('inf')
    max_b = np.max(conv)
    return 2.0 * n * max_b / (S * S)

def de_mutation_rand1(pop, i, F, rng):
    """DE/rand/1 mutation."""
    npop = len(pop)
    while True:
        r1, r2, r3 = rng.choice(npop, 3, replace=False)
        if r1 != i and r2 != i and r3 != i:
            break
    return pop[r1] + F * (pop[r2] - pop[r3])

def lp_improve(a, P, n_max_lp=2000):
    """LP step: find b that maximizes sum(b) subject to conv(a,b) <= P, b >= 0."""
    n = len(a)
    if n > n_max_lp:
        return None
    
    try:
        # Build convolution matrix
        A = np.zeros((2*n-1, n))
        for k in range(2*n - 1):
            lo = max(0, k - n + 1)
            hi = min(n-1, k)
            if lo <= hi:
                A[k, lo:hi+1] = a[k-hi:k-lo+1][::-1]
        
        c_obj = -np.ones(n)
        b_ub = np.full(2*n-1, P)
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                      method='highs', options={'time_limit': 5.0})
        if res.success:
            b = res.x
            s = b.sum()
            if s > 1e-8:
                return b * (a.sum() / s)
    except Exception:
        pass
    return None

"""
Hybrid DE + Spectral Subgradient optimization for minimum autocorrelation peak sequences.
Uses differential evolution for global exploration, alternating projection initialization,
FFT-based subgradient refinement, and LP improvement steps. Tries multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = v.astype(np.float64)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def score_from_conv(a, conv):
    """Compute score 2*n*max(conv)/sum(a)^2 from sequence and its convolution."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return float('inf')
    max_b = np.max(conv)
    return 2.0 * n * max_b / (S * S)


def de_mutation_rand1(pop, i, F, rng):
    """DE/rand/1 mutation: v = x_r1 + F*(x_r2 - x_r3)."""
    npop = len(pop)
    while True:
        r1, r2, r3 = rng.choice(npop, 3, replace=False)
        if r1 != i and r2 != i and r3 != i:
            break
    return pop[r1] + F * (pop[r2] - pop[r3])


def de_mutation_best2(pop, i, F, best_idx, rng):
    """DE/best/2 mutation: v = x_best + F1*(x_r1 - x_r2) + F2*(x_r3 - x_r4)."""
    npop = len(pop)
    while True:
        r1, r2, r3, r4 = rng.choice(npop, 4, replace=False)
        if best_idx not in (r1, r2, r3, r4) and i not in (r1, r2, r3, r4):
            break
    F2 = F * 0.5
    return pop[best_idx] + F * (pop[r1] - pop[r2]) + F2 * (pop[r3] - pop[r4])


def de_mutation_current_best1(pop, i, F, best_idx, rng):
    """DE/current-to-best/1: v = x_i + F*(x_best - x_i) + F*(x_r1 - x_r2)."""

    while True:
        r1, r2 = rng.choice(npop, 2, replace=False)
        if r1 != i and r2 != i and r1 != best_idx and r2 != best_idx:
            break
    return pop[i] + F * (pop[best_idx] - pop[i]) + F * (pop[r1] - pop[r2])


def de_crossover_binomial(v, x_i, CR, j_rand, rng):
    """Binomial crossover: u[j] = v[j] if rand < CR or j==j_rand, else x_i[j]."""
    n = len(v)
    mask = rng.random(n) < CR
    mask[j_rand] = True
    return np.where(mask, v, x_i)


def alt_projection_init(n, rng, n_iter=60):
    """Alternating projection for flat-spectrum non-negative sequence."""
    N = 1 << (2 * n - 2).bit_length()
    a = rng.random(n)
    a = a / a.sum()

    for it in range(n_iter):
        F = np.fft.rfft(a, N)
        angle = np.angle(F)
        F_new = np.exp(1j * angle)
        F_new[0] = F_new[0].real + 0j
        a_new = np.fft.irfft(F_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a_new / s
    return a


def alt_projection_tapered(n, rng, n_iter=60, taper=0.3):
    """Alternating projection with tapered target spectrum."""
    N = 1 << (2 * n - 2).bit_length()

    a = a / a.sum()
    freq_idx = np.arange(len(np.fft.rfft(a, N)))

    target_mag = np.maximum(target_mag, 0.1)

    for it in range(n_iter):
        F = np.fft.rfft(a, N)
        angle = np.angle(F)
        F_new = target_mag * np.exp(1j * angle)
        F_new[0] = F_new[0].real + 0j

def subgradient_refine(a, deadline, rng, max_iters=300000):

    n = len(a)
    a = a.astype(np.float64).copy()
    a = project_simplex(a)

    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()

    lr = 3.0 / n
    lr_min = lr * 1e-5
    stall = 0
    max_stall = 400
    it = 0
    best_lr = lr

    while time.time() < deadline and it < max_iters:
        it += 1

        c = fast_convolve(a)
        peak = np.max(c)

        if peak < best_peak * (1.0 - 1e-10):
            best_peak = peak
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            lr = max(lr * 0.75, lr_min)
            stall = 0
            if lr < lr_min * 5:
                if time.time() > deadline - 30:
                    break
                # Perturb to escape
                scale = rng.uniform(0.02, 0.08) / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, scale, n))
                lr = best_lr * 0.5

        # Active peaks
        threshold = peak * (1.0 - 2e-4)
        peak_idx = np.where(c >= threshold)[0]
        if len(peak_idx) > 25:
            top = np.argsort(c[peak_idx])[::-1][:12]

        grad /= max(1, len(peak_idx))

        a = project_simplex(a - lr * grad)

    return best_a

            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * y[k - hi:k - lo + 1][::-1]
        grad /= max(1, len(peak_idx))

        # Update
        x_new = project_simplex(y - lr * grad)
        y = x_new * (1 + beta) - x * beta
        y = project_simplex(y)
        x = x_new

    return best_a


def lp_improve(a, P, time_limit=4.0):
    """LP: maximize sum(b) s.t. conv(a,b) <= P, b >= 0. Returns rescaled b or None."""
    n = len(a)
    if n > 1500:
        return None
    try:
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, P)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': time_limit})
        if res.success:
            b = res.x
            s = b.sum()
            if s > 1e-8:
                return b * (a.sum() / s)
    except Exception:
        pass
    return None


def de_optimize(n, deadline, rng, pop_size=35, init_solutions=None):
    """Differential evolution for fixed sequence length n."""
    pop = []
    scores = []

    # Add initial solutions
    if init_solutions:
        for sol in init_solutions[:pop_size // 3]:
            a = np.array(sol, dtype=np.float64)
            a = np.maximum(a, 0)
            s = a.sum()
            a = a / s if s > 1e-10 else np.ones(n) / n
            pop.append(a)
            c = fast_convolve(a)
            scores.append(score_from_conv(a, c))

    # Fill with diverse initializations
    while len(pop) < pop_size:
        r = rng.random()
        if r < 0.25:
            a = alt_projection_init(n, rng, n_iter=40)
        elif r < 0.4:
            a = alt_projection_tapered(n, rng, n_iter=40, taper=rng.uniform(0.1, 0.5))
        elif r < 0.55:
            # Random with concentration
            a = rng.power(0.5, n)
            a = a / a.sum()
        elif r < 0.7:
            # Cosine-like
            x = np.linspace(0, 1, n)
            a = 0.5 * (1 + np.cos(2 * np.pi * rng.random() * x)) + 0.05 * rng.random(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
        elif r < 0.85:
            # Two-lobe
            a = np.exp(-0.5 * ((np.linspace(0, 1, n) - 0.3) / 0.15) ** 2)
            a += np.exp(-0.5 * ((np.linspace(0, 1, n) - 0.7) / 0.15) ** 2)
            a += 0.01 * rng.random(n)
            a = a / a.sum()
        else:
            a = rng.random(n)
            a = a / a.sum()
        pop.append(a)
        c = fast_convolve(a)
        scores.append(score_from_conv(a, c))

    pop = np.array(pop)
    scores = np.array(scores)
    best_idx = int(np.argmin(scores))
    best_score = float(scores[best_idx])
    best_a = pop[best_idx].copy()

    print(f"    DE init: best={best_score:.6f}")

    gen = 0
    F = 0.7
    CR = 0.7
    stagnant = 0
    max_stagnant = 25

    while time.time() < deadline:
        gen += 1
        F = max(0.25, 0.7 * (0.995 ** gen))
        CR = 0.5 + 0.3 * (0.998 ** gen)

        for i in range(pop_size):
            if time.time() > deadline:
                break

            # Choose mutation strategy
            r = rng.random()
            if r < 0.35:
                v = de_mutation_rand1(pop, i, F, rng)
            elif r < 0.7:
                v = de_mutation_best2(pop, i, F, best_idx, rng)
            else:
                v = de_mutation_current_best1(pop, i, F, best_idx, rng)

            # Crossover
            j_rand = int(rng.integers(n))
            u = de_crossover_binomial(v, pop[i], CR, j_rand, rng)
            u = project_simplex(u)

            # Evaluate
            c = fast_convolve(u)
            s = score_from_conv(u, c)

            if s < scores[i]:
                pop[i] = u
                scores[i] = s
                if s < best_score:
                    best_score = s
                    best_a = u.copy()
                    best_idx = i

        current_best = float(np.min(scores))
        if current_best < best_score * (1 - 1e-9):
            best_score = current_best
            best_idx = int(np.argmin(scores))
            best_a = pop[best_idx].copy()
            stagnant = 0
        else:
            stagnant += 1

        # Re-inject diversity
        if stagnant > max_stagnant:
            stagnant = 0
            n_replace = max(3, pop_size // 6)
            worst = np.argsort(scores)[::-1][:n_replace]
            for wi in worst:
                r = rng.random()
                if r < 0.4:
                    pop[wi] = alt_projection_init(n, rng, n_iter=25)
                elif r < 0.7:
                    pop[wi] = alt_projection_tapered(n, rng, n_iter=25, taper=rng.uniform(0.1, 0.6))
                else:
                    pop[wi] = rng.power(rng.uniform(0.3, 1.5), n)
                    s = pop[wi].sum()
                    pop[wi] = pop[wi] / s if s > 0 else np.ones(n) / n
                c = fast_convolve(pop[wi])
                scores[wi] = score_from_conv(pop[wi], c)

        if gen % 30 == 0:
            print(f"    gen={gen}, best={best_score:.6f}, F={F:.3f}, CR={CR:.3f}")

    return best_a, best_score


def spectral_phase_optimize(a, deadline, rng, n_iter=5000):
    """
    Frequency-domain phase optimization: minimize time-domain peak by adjusting
    the phase of the spectrum while keeping magnitude fixed.
    Then project back to non-negative simplex.
    """
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    a = a.astype(np.float64).copy()
    a = project_simplex(a)

    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()

    lr = 0.1
    for it in range(n_iter):
        if time.time() > deadline:
            break

        # Get current spectrum
        X = np.fft.rfft(a, N)
        mag = np.abs(X)
        mag[0] = 0.0  # Fix DC

        # Current autocorrelation (circular approx)
        ac = np.fft.irfft(mag * mag, N)
        peak_idx = np.argmax(ac[:2 * n - 1])
        peak_val = ac[peak_idx]

        if peak_val < best_peak * 0.999:
            best_peak = peak_val
            best_a = a.copy()

        # Gradient of peak with respect to phase
        # d(ac[k])/d(phi[m]) = -Im(X[m] * conj(X[k-m])) * ... (simplified)
        # Use numerical: perturb phase at peak frequency
        # Simpler approach: just do time-domain subgradient
        c = fast_convolve(a)
        peak = np.max(c)
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        threshold = peak * (1.0 - 1e-3)
        pidx = np.where(c >= threshold)[0]
        if len(pidx) > 15:
            top = np.argsort(c[pidx])[::-1][:8]
            pidx = pidx[top]

        grad = np.zeros(n)
        for k in pidx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        grad /= max(1, len(pidx))

        a = project_simplex(a - lr * grad / n)
        lr = max(lr * 0.999, 0.01)

    return best_a


def propose_candidate(budget_s=1500, **kwargs):
    """
    Hybrid DE + subgradient + LP optimization for minimum autocorrelation peak.
    Uses differential evolution for global search, multiple initialization strategies,
    projected subgradient (standard and accelerated) for local refinement,
    and LP-based improvement. Tries multiple sequence lengths.
    """
    overall_deadline = time.time() + budget_s - 30.0
    rng = np.random.default_rng(777)

    best_score = float('inf')
    best_seq = None

    # Get parent solution
    parent_solutions = []
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000)
        if len(parent) > 0 and parent.sum() > 0.01:
            parent = parent / parent.sum()
            parent_solutions.append(parent)
    except Exception:
        pass

    # Try different lengths (prioritize 512 since previous best was there)
    lengths = [512, 384, 256, 768, 1024]

    # Time allocation: give more time to 512
    total_time = overall_deadline - time.time()
    time_weights = [0.30, 0.18, 0.15, 0.17, 0.20]

    for n_idx, n in enumerate(lengths):
        if time.time() > overall_deadline - 90:
            break

        alloc_time = total_time * time_weights[n_idx]
        print(f"\n{'='*50}")
        print(f"=== n={n}, allocated={alloc_time:.0f}s ===")
        print(f"{'='*50}")

        # Prepare initial solutions
        inits = []
        for ps in parent_solutions:
            if len(ps) >= n:
                a = ps[:n].copy()
            else:
                a = np.zeros(n)
                a[:len(ps)] = ps
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                inits.append(a / s)

        # Phase 1: DE (50% of allocated time)
        de_time = alloc_time * 0.45
        de_deadline = min(time.time() + de_time, overall_deadline - 90)

        if de_deadline < time.time() + 20:
            continue

        try:
            a_de, score_de = de_optimize(n, de_deadline, rng, pop_size=35, init_solutions=inits)
            print(f"  After DE: score={score_de:.6f}")
        except Exception as e:
            print(f"  DE error: {e}")
            a_de = alt_projection_init(n, rng, n_iter=50)
            c = fast_convolve(a_de)
            score_de = score_from_conv(a_de, c)

        if np.isfinite(score_de) and score_de < best_score:
            best_score = score_de
            best_seq = a_de.copy()
            print(f"  >>> New best: {best_score:.6f}")

        current_best = a_de if best_seq is None or len(best_seq) != n else best_seq.copy()

        # Phase 2: Subgradient refinement (30%)
        sg_time = alloc_time * 0.30
        sg_deadline = min(time.time() + sg_time, overall_deadline - 60)

        if sg_deadline > time.time() + 15:
            try:
                a_sg = subgradient_refine(current_best, sg_deadline, rng)
                c = fast_convolve(a_sg)
                score_sg = score_from_conv(a_sg, c)
                print(f"  After SG: score={score_sg:.6f}")
                if np.isfinite(score_sg) and score_sg < best_score:
                    best_score = score_sg
                    best_seq = a_sg.copy()
                    print(f"  >>> New best: {best_score:.6f}")
            except Exception as e:
                print(f"  SG error: {e}")

        # Phase 3: Accelerated subgradient (15%)
        if time.time() < overall_deadline - 45:
            ac_deadline = min(time.time() + alloc_time * 0.15, overall_deadline - 45)
            if ac_deadline > time.time() + 10:
                try:
                    a_ac = accelerated_subgradient(current_best, ac_deadline, rng)
                    c = fast_convolve(a_ac)
                    score_ac = score_from_conv(a_ac, c)
                    print(f"  After AccSG: score={score_ac:.6f}")
                    if np.isfinite(score_ac) and score_ac < best_score:
                        best_score = score_ac
                        best_seq = a_ac.copy()
                        print(f"  >>> New best: {best_score:.6f}")
                except Exception as e:
                    print(f"  AccSG error: {e}")

        # Phase 4: LP improvement (10%, only for n <= 1024)
        if n <= 1024 and time.time() < overall_deadline - 30 and best_seq is not None and len(best_seq) == n:
            try:
                a_best = best_seq.copy()
                c_best = fast_convolve(a_best)
                P_best = np.max(c_best) * 1.005
                b = lp_improve(a_best, P_best, time_limit=3.0)
                if b is not None:
                    for t_mix in [0.2, 0.4, 0.6]:
                        a_mix = project_simplex((1 - t_mix) * a_best + t_mix * b)
                        c_mix = fast_convolve(a_mix)
                        s_mix = score_from_conv(a_mix, c_mix)
                        if np.isfinite(s_mix) and s_mix < best_score:
                            best_score = s_mix
                            best_seq = a_mix.copy()
                            print(f"  >>> LP mix (t={t_mix}): {best_score:.6f}")
            except Exception as e:
                print(f"  LP error: {e}")

    # Final polish on best sequence
    if best_seq is not None and time.time() < overall_deadline - 10:
        print(f"\n=== Final polish (n={len(best_seq)}) ===")
        try:
            a_final = subgradient_refine(best_seq, overall_deadline - 3, rng)
            c = fast_convolve(a_final)
            score_final = score_from_conv(a_final, c)
            print(f"  Final: {score_final:.6f}")
            if np.isfinite(score_final) and score_final < best_score:
                best_score = score_final
                best_seq = a_final
        except Exception as e:
            print(f"  Polish error: {e}")

    # Fallback
    if best_seq is None:
        print("Falling back to uniform sequence")
        best_seq = np.ones(256)

    result = [float(x) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"\n{'='*50}")
    print(f"FINAL: score={final_score:.6f}, length={len(result)}")
    print(f"{'='*50}")
    return result
