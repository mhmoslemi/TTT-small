def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()

                    print(f"  [n={n}, init={idx}] after mid: J={sc:.6f} (t={time.monotonic()-t_start:.0f}s)", flush=True)

                # Phase C: Fine (small T, small lr)
                remaining = deadline - time.monotonic()
                if remaining > 3:
                    n_iters_c = int(min(6000, max(500, remaining * 35)))
                    result_c, val_c = run_subgrad(
                        result_b, n_iters_c, lr_0=0.15, mu=0.97,
                        top_t_start=4, top_t_end=1, deadline=deadline
                    )
                    sc = score_sequence(result_c)
                    if np.isfinite(sc) and sc < best_score:
                        best = result_c.copy()
                        best_score = sc
                        print(f"  [n={n}, init={idx}] after fine: J={sc:.6f} (t={time.monotonic()-t_start:.0f}s)", flush=True)

        elapsed = time.monotonic() - t0
        print(f"  n={n} done in {elapsed:.1f}s, current best={best_score:.6f}", flush=True)

    # === Final polish on the best sequence ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== Final polish (n={best.size}) ===", flush=True)
        t0 = time.monotonic()
        current = best.copy()

        polish_configs = [
            (0.05, 0.99, 2, 1),
            (0.03, 0.995, 1, 1),
            (0.08, 0.98, 3, 1),
            (0.02, 0.998, 1, 1),
            (0.04, 0.99, 2, 2),
        ]

        for lr_p, mu_p, tt_s, tt_e in polish_configs:
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            n_iters = int(min(15000, max(1000, remaining * 40)))
            result, _ = run_subgrad(
                current, n_iters, lr_0=lr_p, mu=mu_p,
                top_t_start=tt_s, top_t_end=tt_e, deadline=deadline
            )
            sc = score_sequence(result)
            if np.isfinite(sc) and sc < best_score:
                best = result.copy()
                best_score = sc
                current = result.copy()
                print(f"  Polish lr={lr_p},mu={mu_p},T={tt_s}->{tt_e}: J={sc:.6f} (t={time.monotonic()-t_start:.0f}s)", flush=True)
            else:
                # Keep the better of current and result
                sc_cur = score_sequence(current)
                if sc < sc_cur:
                    current = result.copy()

        print(f"  Polish done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Try a few different lengths for final comparison ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== Cross-length refinement ===", flush=True)
        t0 = time.monotonic()
        base_n = best.size
        alt_lengths = list(set([base_n, max(128, base_n // 2), min(2048, base_n * 2)]))
        alt_lengths.remove(base_n)

        for alt_n in alt_lengths:
            if time.monotonic() > deadline:
                break
            # Resample best to alt_n
            t_old = np.linspace(0, 1, base_n, endpoint=False)
            t_new = np.linspace(0, 1, alt_n, endpoint=False)
            a0 = np.maximum(np.interp(t_new, t_old, best), 0.001)
            a0 = a0 / a0.sum()

            remaining = deadline - time.monotonic()
            n_iters = int(min(8000, max(1000, remaining * 30)))
            result, _ = run_subgrad(
                a0, n_iters, lr_0=0.2, mu=0.95,
                top_t_start=5, top_t_end=1, deadline=deadline
            )
            sc = score_sequence(result)
            if np.isfinite(sc) and sc < best_score:
                best = result.copy()
                best_score = sc
                print(f"  Cross-length n={alt_n}: J={sc:.6f} (t={time.monotonic()-t_start:.0f}s)", flush=True)

        print(f"  Cross-length done in {time.monotonic()-t0:.1f}s", flush=True)

    if best is None:
        best = np.ones(256)

    total_time = time.monotonic() - t_start
    print(f"\nFINAL: J={best_score:.6f}, n={best.size}, total_time={total_time:.0f}s", flush=True)
    return [float(x) for x in best]
