```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

```python
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)

def project_simplex(v):
    n = v.size
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

    """Compute gradient of (f*f)_k with respect to f."""
    n = f.size
    grad = np.zeros(n)
    # (f*f)_k = sum_j f_j * f_{k-j} for valid j

    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        indices = np.arange(j_start, j_end + 1)
        grad[indices] = f[k - indices]
    return grad

    """Compute weighted multi-peak gradient."""
    n = f.size
    # Get top peaks
    if len(f_squared) <= top_k:
        peaks = np.arange(len(f_squared))
        weights = np.ones(len(f_squared)) / len(f_squared)
    else:
```

Hybrid LP + Subgradient optimization for minimizing 2*n*max(convolve(f,f))/(sum(f))^2.
Uses Frank-Wolfe (LP) directions with subgradient polishing, adaptive step sizes,
momentum, and diverse initializations including the provided sequence.

def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """
    Compute the gradient of (f*f)_k with respect to f.
    (f*f)_k = sum_j f_j * f_{k-j} for valid indices.
    d/dw_j = f_{k-j} if 0 <= k-j < n, else 0.
    But since (f*f)_k = sum_j f_j * f_{k-j}, and both f_j and f_{k-j} appear,
    the derivative is:
      If k-j != j: d/dw_j = f_{k-j} (from the f_j term) + f_{k-j} (from the f_{k-j} term when the summation index hits k-j)
      Wait, let me be careful.
      
      (f*f)_k = sum_i f_i * f_{k-i}
      d/dw_j: the term where i=j gives f_{k-j} (if valid)
      The term where k-i=j, i.e., i=k-j, gives f_{k-j} (if valid and k-j != j)
      If k-j == j (k=2j): the term f_j * f_j has derivative 2*f_j
      
      So: grad[j] = f_{k-j} (if 0<=k-j<n) + f_{k-j} (if 0<=k-j<n and k-j!=j)
                 = 2*f_{k-j} (if 0<=k-j<n and k!=2j)
                 = 2*f_j (if k=2j)  [which is the same as 2*f_{k-j} since k-j=j]
      
      Actually: grad[j] = 2*f_{k-j} if 0 <= k-j < n, else 0.
      (The factor of 2 comes from both terms, and it correctly handles k=2j where both terms are f_j.)
    """
    n = f.size
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        indices = np.arange(j_start, j_end + 1)
        grad[indices] = 2.0 * f[k - indices]
    return grad


def multi_peak_subgradient(f, top_k=5, temperature=0.5):
    """
    Compute a weighted subgradient using the top-K peaks of the autocorrelation.
    """
    n = f.size
    f_sq = np.convolve(f, f)
    m = f_sq.size
    M = float(np.max(f_sq))
    if M <= 0 or not np.isfinite(M):
        return np.zeros(n), f_sq
    
    # Get top peaks
    if m <= top_k:
        peaks = np.arange(m)
        weights = np.ones(m) / m
    else:
        sorted_idx = np.argsort(f_sq)[::-1]
        peaks = sorted_idx[:top_k]
        peak_vals = f_sq[peaks]
        # Softmax weighting
        scaled = (peak_vals - M) / (temperature * M + 1e-300)
        weights = np.exp(scaled)
        weights /= weights.sum()
    
    grad = np.zeros(n)
    for k, w in zip(peaks, weights):
        grad += w * compute_peak_gradient(f, int(k))
    return grad, f_sq


def make_conv_template(n):
    """
    Precompute the CSR matrix structure for convolve(f, g).
    """
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(template, f, n, M, relax=1.0):
    """Solve the LP direction problem for the given f and M."""
    A_template, edge_offsets, c_obj, bounds, m = template
    lp_scale = float(n)

    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale

    b = np.full(m, M * lp_scale * lp_scale * relax, dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")


def interpolate_sequence(x, n_target):
    """Resample a nonnegative sequence to length n_target and normalize it."""
    x = normalize_sequence(x)
    if x is None or n_target <= 0:
        return None
    m = x.size
    if m == n_target:
        return x
    if m == 1:
        return np.ones(n_target, dtype=np.float64) / float(n_target)
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n_target, dtype=np.float64)
    out = np.interp(new, old, x)
    return normalize_sequence(out)
