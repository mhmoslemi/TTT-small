def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()

    if s <= 1e-300:

    x /= s
    return x


def score_normalized(x, n):

def make_conv_template(n):
    """
    Precompute the CSR matrix structure for convolve(f, g).
    """
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    
    # Store the structure for efficient rebuilding
    return {
        'shape': (m, n),
        'row': row,
        'col': col,
        'edge_offsets': edge_offsets,
        'm': m,
        'n': n,
        'total': total,
    }

    A = csr_matrix((data, (row, col)), shape=(m, n), dtype=np.float64)
    return A

    """Project v onto the probability simplex {x : sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Enhanced LP-based and smoothed-gradient search for minimizing
2*n*max(convolve(f,f))/(sum(f))^2 over non-negative sequences.

Combines:
1. LP direction finding (max sum(g) s.t. f*g <= M) with structured restarts
2. FISTA with log-sum-exp smoothing and simplex projection
3. Multiple starting points and time-adaptive scheduling

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1. Returns 2*n*M."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def make_conv_template(n):
    """Precompute sparse matrix structure for convolve(f, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)

    return {
        'shape': (m, n),
        'row': row,
        'col': col,
        'edge_offsets': edge_offsets,
        'm': m,
        'n': n,
        'total': total,
    }


def build_A(template, f):
    """Build sparse A matrix from template and vector f."""
    data = f[template['edge_offsets']]
    return csr_matrix((data, (template['row'], template['col'])),
                      shape=template['shape'], dtype=np.float64)


def solve_lp(template, f, n, M, scale=1.0):
    """Solve: max sum(g) s.t. A*f @ g <= M*scale*n^2, g >= 0."""
    A = build_A(template, f)
    A_scaled = (A * n).tocsr()
    b = np.full(template['m'], M * n * n * scale, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    options = {'presolve': True, 'time_limit': 30.0}
    res = linprog(c_obj, A_ub=A_scaled, b_ub=b, bounds=bounds,
                  method='highs', options=options)
    return res


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(f, n, alpha):
    """
    Compute gradient of smooth_max((f*f)_k, alpha) w.r.t. f.
    Uses FFT for efficiency.
    """
    conv = np.convolve(f, f)
    max_c = float(np.max(conv))
    if max_c <= 0:
        return np.zeros(n)

    # Softmax weights (numerically stable)
    shifted = conv / max_c
    weights = np.exp(alpha * (shifted - 1.0))
    wsum = np.sum(weights)
    if wsum <= 0:
        weights = np.zeros(len(conv))
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= wsum

    # d_i = sum_k w_k * [f_{k-i} + f_{i-k}] (where indices valid)
    # C1[i] = sum_k w_k * f_{k-i} = np.correlate(weights, f, 'full')[i]
    C1 = np.correlate(weights, f, 'full')
    C1 = C1[:n]

    # C2[i] = sum_k w_k * f_{i-k} = np.convolve(f, weights, 'full')[i]
    C2 = np.convolve(f, weights, 'full')
    C2 = C2[:n]

    grad = C1 + C2
    return grad


def optimize_fista(f_init, n, time_budget, rng):
    """
    FISTA with log-sum-exp smoothing for min max_k (f*f)_k on simplex.
    """
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n) / n

    best_f = f.copy()
    best_score = score_normalized(f, n)

    deadline = time.monotonic() + max(0.0, float(time_budget))
    alpha = 5.0
    t_fista = 1.0
    x_old = f.copy()
    y = f.copy()
    L = 4.0 * float(np.max(f)) + 0.1
    stall = 0

    while time.monotonic() < deadline:
        # Compute gradient at y
        grad = compute_smooth_gradient(y, n, alpha)

        # FISTA step
        step = 1.0 / L
        z = y - step * grad
        x_new = project_simplex(z)

        # Update momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
        y = x_new + ((t_fista - 1.0) / t_new) * (x_new - x_old)
        t_fista = t_new
        x_old = x_new

        # Evaluate
        sc = score_normalized(x_new, n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = x_new.copy()
            stall = 0
        else:
            stall += 1

        # Increase alpha periodically
        if stall > 50:
            alpha *= 1.3
            stall = 0
            L = 4.0 * float(np.max(x_new)) * alpha / 5.0 + 0.1
            t_fista = 1.0
            x_old = x_new.copy()
            y = x_new.copy()
            if alpha > 500:
                # Reset with perturbation
                x_new = 0.8 * best_f + 0.2 * project_simplex(
                    best_f + 0.1 * rng.standard_normal(n))
                x_new = project_simplex(x_new)
                x_old = x_new.copy()
                y = x_new.copy()
                t_fista = 1.0
                alpha = 10.0
                L = 4.0 * float(np.max(x_new)) + 0.1

    return best_f, float(best_score)


def structured_perturb(f, best_f, rng, iteration, n):
    """Generate a structured perturbation of f."""
    alpha_mix = 0.6 + 0.2 * min(1.0, iteration / 100.0)
    f_new = alpha_mix * f + (1.0 - alpha_mix) * best_f
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-300:
        f_new /= s
    else:
        return best_f.copy()

    # Add smooth low-frequency perturbation
    eps = 0.03 + 0.04 * (iteration % 4) / 4.0
    freq = rng.integers(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n, dtype=np.float64)
    pert = np.abs(np.sin(2 * np.pi * freq * x / n + phase))
    pert = pert * eps / n
    f_new = f_new + pert
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-300:
        f_new /= s
    else:
        f_new = best_f.copy()
    return f_new


def aggressive_restart(best_f, rng, n):
    """Generate a diverse restart point."""
    ptype = rng.integers(0, 6)
    x = np.arange(n, dtype=np.float64)

    if ptype == 0:
        center = rng.uniform(0.2 * n, 0.8 * n)
        width = rng.uniform(0.05 * n, 0.4 * n)
        pattern = np.exp(-0.5 * ((x - center) / width) ** 2)
    elif ptype == 1:
        t = x / max(n - 1, 1)
        pattern = 1.0 + 3.0 * (2 * t - 1) ** 4
    elif ptype == 2:
        freq = rng.integers(2, 10)
        pattern = 1.0 + 0.7 * np.cos(2 * np.pi * freq * x / n)
    elif ptype == 3:
        # Two bumps
        c1 = rng.uniform(0.15 * n, 0.4 * n)
        c2 = rng.uniform(0.6 * n, 0.85 * n)
        w = rng.uniform(0.05 * n, 0.15 * n)
        pattern = (np.exp(-0.5 * ((x - c1) / w) ** 2) +
                   np.exp(-0.5 * ((x - c2) / w) ** 2))
    elif ptype == 4:
        # Smooth random
        npts = rng.integers(3, 12)
        xs = np.sort(rng.uniform(0, n, npts))
        ys = rng.uniform(0.3, 2.0, npts)
        pattern = np.interp(x, xs, ys)
    else:
        # Random + decay
        pattern = rng.uniform(0.5, 1.5, n)
        decay = np.exp(-x / (n * rng.uniform(0.5, 2.0)))
        pattern = pattern * decay

    pattern = np.maximum(pattern, 0.01)
    p = pattern / pattern.sum()
    alpha = rng.uniform(0.2, 0.6)
    f_new = alpha * best_f + (1 - alpha) * p
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-300:
        f_new /= s
    else:
        f_new = best_f.copy()
    return f_new

    try:
        template = make_conv_template(n)
    except Exception as e:
        print(f"Template failed: {e}", flush=True)
        template = None

    # === Phase 1: Main LP optimization from given start ===
    rem = deadline - time.monotonic()
    if rem > 60 and template is not None:
        alloc = min(700.0, rem * 0.55)
        print(f"\nPhase 1: LP from given, alloc={alloc:.0f}s", flush=True)
        f_res, s_res = optimize_lp_main(n, best_f, alloc, rng, template)
        if np.isfinite(s_res) and s_res < best_score:
            best_score = s_res
            best_f = f_res.copy()
            print(f"  -> score={best_score:.6f}", flush=True)

    # === Phase 2: FISTA smoothing ===
    rem = deadline - time.monotonic()
    if rem > 40:
        alloc = min(200.0, rem * 0.35)
        print(f"\nPhase 2: FISTA, alloc={alloc:.0f}s, current={best_score:.6f}", flush=True)
        f_res, s_res = optimize_fista(best_f, n, alloc, rng)
        if np.isfinite(s_res) and s_res < best_score:
            best_score = s_res
            best_f = f_res.copy()
            print(f"  -> score={best_score:.6f}", flush=True)

    # === Phase 3: LP from FISTA result ===
    rem = deadline - time.monotonic()
    if rem > 40 and template is not None:
        alloc = min(300.0, rem * 0.55)
        print(f"\nPhase 3: LP from FISTA result, alloc={alloc:.0f}s", flush=True)
        f_res, s_res = optimize_lp_main(n, best_f, alloc, rng, template)
        if np.isfinite(s_res) and s_res < best_score:
            best_score = s_res
            best_f = f_res.copy()
            print(f"  -> score={best_score:.6f}", flush=True)

    # === Phase 4: LP from perturbed best ===
    rem = deadline - time.monotonic()
    if rem > 30 and template is not None:
        alloc = min(200.0, rem * 0.6)
        print(f"\nPhase 4: LP from perturbed start, alloc={alloc:.0f}s", flush=True)
        # Create perturbed start
        f_pert = aggressive_restart(best_f, rng, n)
        f_res, s_res = optimize_lp_main(n, f_pert, alloc, rng, template)
        if np.isfinite(s_res) and s_res < best_score:
            best_score = s_res
            best_f = f_res.copy()
            print(f"  -> score={best_score:.6f}", flush=True)

    # === Phase 5: Final polishing ===
    rem = deadline - time.monotonic()
    if rem > 20 and template is not None:
        alloc = min(150.0, rem * 0.8)
        print(f"\nPhase 5: Final polish, alloc={alloc:.0f}s, current={best_score:.6f}", flush=True)
        f_res, s_res = optimize_lp_main(n, best_f, alloc, rng, template)
        if np.isfinite(s_res) and s_res < best_score:
            best_score = s_res
            best_f = f_res.copy()
            print(f"  -> score={best_score:.6f}", flush=True)

    # Final validation
    if best_f is None:
        best_f = np.ones(64, dtype=np.float64) / 64.0

    seq = [float(x) for x in best_f]
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float("inf")

    if not np.isfinite(final_score):
        if given_f is not None:
            seq = [float(x) for x in given_f]
            try:
                final_score = float(evaluate_sequence(seq))
            except Exception:
                pass

    if not np.isfinite(final_score):
        seq = [1.0 / 1024.0] * 1024
        final_score = 2.0

    total_time = time.monotonic() - start_time
    print(f"\nFINAL: n={len(seq)}, score={final_score:.6f}, time={total_time:.1f}s", flush=True)
    return seq
