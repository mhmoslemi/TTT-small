```python
"""
Hybrid LP + mirror descent + coordinate descent search for minimizing
2*n*max(conv(f,f))/(sum(f))^2 over non-negative sequences.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize(arr):
    """Normalize to sum 1, non-negative."""
    x = np.asarray(arr, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-300:
        return None
    return x / s


def score_unit(f):
    """Score for unit-sum non-negative f: 2*n*max(conv(f,f))."""
    n = len(f)
    c = np.convolve(f, f)
    return 2.0 * n * np.max(c)


def score_full(f):
    """Full score: 2*n*max(conv(f,f))/(sum(f))^2."""
    s = np.sum(f)
    if s < 1e-300:
        return float('inf')
    n = len(f)
    c = np.convolve(f, f)
    return 2.0 * n * np.max(c) / (s*s)


def build_conv_matrix(n):
    """
    Build the sparse matrix A such that (A @ g)_k = (f*g)_k for a given f.
    Actually, we need: (f*g)_k = sum_j f_{k-j} * g_j
    So A_{k,j} = f_{k-j} if 0 <= k-j < n, else 0.
    The structure (which entries are nonzero) depends only on n, not on f.
    The values depend on f.
    """
    m = 2 * n - 1
    rows = []
    cols = []
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    # The "offset" for each entry: A_{k,j} = f[k-j], so offset = k - j
    offsets = (rows - cols).astype(np.int32)
    return rows, cols, offsets, m


def solve_lp_direction(f, top_k=200, method='highs'):
    """
    Solve the LP:
      max sum(g)
      s.t. (f*g)_k <= M for all k in top_k_peaks
           g >= 0
    where M = max_k (f*f)_k.
    Returns: g_star, M, active_set
    """
    n = len(f)
    conv_ff = np.convolve(f, f)
    M = np.max(conv_ff)
    
    # Select top-k peaks
    if top_k < len(conv_ff):
        # Get indices of top-k peaks
        thresholds = np.sort(conv_ff)[-top_k]
        active = np.where(conv_ff >= thresholds[0])[0]
    else:
        active = np.arange(len(conv_ff))
    
    # Build the constraint matrix for the selected peaks
    m = len(active)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(m, n))
    
    c = -np.ones(n)
    b = np.full(m, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method=method)
    
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        return g, M, active
    return None, M, active


def mirror_descent_step(f, eta, top_k=50):
    """
    One step of mirror descent (multiplicative weight update).
    f <- f * exp(-eta * g) / sum(f * exp(-eta * g))
    where g is the subgradient of max_k (f*f)_k.
    """
    n = len(f)
    conv_ff = np.convolve(f, f)
    M = np.max(conv_ff)
    
    # Active set: peaks within 1% of max
    threshold = M * 0.99
    active = np.where(conv_ff >= threshold)[0]
    
    # Compute subgradient: average of gradients at active peaks
    # grad_k[i] = 2*f[k-i] if 0 <= k-i < n
    g = np.zeros(n)
    for k in active:
        # grad_k[i] = 2*f[k-i] for i where 0 <= k-i < n
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        k_minus_i = k - np.arange(i_start, i_end + 1)
        g[i_start:i_end+1] += 2.0 * f[k_minus_i]
    g /= len(active)
    
    # Multiplicative update
    log_f = np.log(np.maximum(f, 1e-300))
    new_log_f = log_f - eta * g
    # Normalize
    new_f = np.exp(new_log_f - np.max(new_log_f))
    new_f /= new_f.sum()
    return new_f


def coordinate_descent(f, max_rounds=50, verbose=False):
    """
    Coordinate descent polishing: for each coordinate, find the best value
    that minimizes the score.
    """
    n = len(f)

            improved = True

            break
```

Hmm, the coordinate descent as I've written it is O(n^2) per coordinate (to compute conv_base), which makes it O(n^3) per round. For n=1024, that's about 1 billion operations per round, which is too slow.

Let me optimize: Instead of recomputing the full convolution for each coordinate, I can update it incrementally.

When I change f[i] from old_val to new_val (delta = new_val - old_val):
- (f*f)_k changes by: delta * f[k-i] + f[k-i] * delta + (delta^2 if k = 2i and we're adding to f[i])
  = 2*delta*f[k-i] + delta^2 (if k = 2i)

Wait, more carefully:
Old: (f*f)_k = sum_j f_j f_{k-j}
New: (f*f)_k = sum_j f_j' f_{k-j}' where f_i' = f_i + delta, others unchanged.

The change is:
- Terms with j = i: (f_i + delta) * f_{k-i} - f_i * f_{k-i} = delta * f_{k-i}
- Terms with j = k-i (if k-i != i): f_{k-i} * (f_i + delta) - f_{k-i} * f_i = delta * f_{k-i}
- If k = 2i (so j = i = k-i): the term is (f_i + delta)^2 - f_i^2 = 2*delta*f_i + delta^2

So the update is:
- For k where 0 <= k-i < n and k != 2i: (f*f)_k += 2*delta*f[k-i]
- For k = 2i: (f*f)_k += 2*delta*f[i] + delta^2

This is O(n) per coordinate update! So the full coordinate descent sweep is O(n^2) per round, which is about 1M operations for n=1024. That's very fast.

```python
"""
Hybrid LP + mirror descent + coordinate descent search for minimizing
2*n*max(conv(f,f))/(sum(f))^2 over non-negative sequences.
Uses sparsified LP constraints, adaptive active sets, and fast coordinate
descent polishing to improve upon pure LP-based approaches.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_arr(arr):
    """Normalize to sum 1, non-negative, or None."""
    x = np.asarray(arr, dtype=np.float64).ravel()
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s < 1e-300:
        return None
    return x / s


def score_unit(f):
    """Score for unit-sum non-negative f: 2*n*max(conv(f,f))."""
    n = len(f)
    c = np.convolve(f, f)
    return float(2.0 * n * np.max(c))


def interp_to_length(x, n):
    """Resample non-negative sequence to length n, normalize."""
    x = normalize_arr(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / n
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n)
    out = np.interp(new, old, x)
    return normalize_arr(out)


def build_sparse_conv(n):
    """
    Build sparse structure for computing (f*g)_k via matrix-vector product.
    Returns (row, col, offset, m) where A[row,col] = f[offset] gives (f*g)_col at position row.
    Actually: (f*g)_k = sum_j f[k-j] * g[j], so A[k,j] = f[k-j] if valid.
    """
    m = 2 * n - 1
    rows_list = []
    cols_list = []
    off_list = []
    for k in range(m):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows_list.append(k)
            cols_list.append(j)
            off_list.append(k - j)
    row = np.array(rows_list, dtype=np.int32)
    col = np.array(cols_list, dtype=np.int32)
    off = np.array(off_list, dtype=np.int32)
    return row, col, off, m


def lp_direction(f, row, col, off, m, top_k=300):
    """
    Solve LP: max sum(g) s.t. (f*g)_k <= M for top-k peaks, g >= 0.
    Returns (g, M, active_indices) or (None, M, active_indices).
    """
    n = len(f)
    conv_ff = np.convolve(f, f)
    M = float(np.max(conv_ff))
    if M <= 0 or not np.isfinite(M):
        return None, M, np.array([], dtype=int)

    # Select top-k peaks
    if top_k < m:
        sorted_vals = np.sort(conv_ff)[::-1]
        thresh = sorted_vals[min(top_k, m) - 1]
        active = np.where(conv_ff >= thresh * 0.9999)[0]
    else:
        active = np.arange(m)

    if len(active) == 0:
        return None, M, active

    # Build constraint matrix for selected peaks
    a_rows = []
    a_cols = []
    a_vals = []
    for idx, k in enumerate(active):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            a_rows.append(idx)
            a_cols.append(j)
            a_vals.append(f[k - j])

    if len(a_rows) == 0:
        return None, M, active

    A = csr_matrix((a_vals, (a_rows, a_cols)), shape=(len(active), n))
    c_obj = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None, M, active

    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        return g, M, active
    return None, M, active


def lp_direction_scaled(f, row, col, off, m, top_k=300, scale_n=True):
    """
    LP with scaling for numerical stability.
    Maximize sum(g) subject to (f*g)_k <= M for active k, g >= 0.
    """
    n = len(f)
    conv_ff = np.convolve(f, f)
    M = float(np.max(conv_ff))
    if M <= 0 or not np.isfinite(M):
        return None, M, np.array([], dtype=int)

    if top_k < m:
        sorted_vals = np.sort(conv_ff)[::-1]
        thresh = sorted_vals[min(top_k, m) - 1]
        # Include peaks that are close to the top
        thresh = max(thresh, M * 0.5)
        active = np.where(conv_ff >= thresh)[0]
    else:
        active = np.arange(m)

    active = active[:top_k]  # cap it
    n_act = len(active)
    if n_act == 0:
        return None, M, active

    # Scale: multiply f and g by n to get cleaner numbers
    # (f*n * g*n)_k * (1/n^2) = (f*g)_k
    # So constraint: sum_j (n*f[k-j])*(n*g[j]) <= M * n^2
    nf = f * n  # scaled f

    a_rows = []
    a_cols = []
    a_vals = []
    for idx, k in enumerate(active):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            a_rows.append(idx)
            a_cols.append(j)
            a_vals.append(nf[k - j])

    if not a_rows:
        return None, M, active

    A = csr_matrix((a_vals, (a_rows, a_cols)), shape=(n_act, n))
    c_obj = -np.ones(n)
    b_ub = np.full(n_act, M * n * n)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None, M, active

    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        return g, M, active
    return None, M, active


def safe_step(f, g, M, G):
    """
    Find the maximum safe step t such that
    max_k [(1-t)^2 (f*f)_k + 2(1-t)t (f*g)_k + t^2 (g*g)_k] <= M
    
    We use the bound: H(t) <= (1-t)^2*M + 2(1-t)*t*r_max + t^2*G
    where r_max = max_k (f*g)_k, G = max_k (g*g)_k.
    """
    r = np.convolve(f, g)
    r_max = float(np.max(r))
    if r_max >= M * (1 - 1e-12):
        return 0.0, r_max

    a_c = 2.0 * (r_max - M)  # negative since r_max < M
    b_c = G + M - 2.0 * r_max

    if b_c > 1e-14:
        t_max = -a_c / b_c
    else:
        t_max = 1.0

    if not np.isfinite(t_max) or t_max <= 0:
        return 0.0, r_max
    return min(1.0, t_max), r_max

def mirror_descent_optimize(f_init, time_budget, rng, verbose=True):

    Mirror descent (multiplicative weight) optimization.

    n = len(f_init)
    f = normalize_arr(f_init)
    if f is None:
        f = np.ones(n) / n

    best_f = f.copy()
    best_score = score_unit(f)
    deadline = time.monotonic() + time_budget
    eta = 0.5
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        t0 = time.monotonic()
        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))

        # Active set: peaks within 5% of max
        thresh = M * 0.95
        active = np.where(conv_ff >= thresh)[0]

        # Subgradient
        g = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            km = k - np.arange(i_lo, i_hi + 1)
            g[i_lo:i_hi+1] += 2.0 * f[km]
        g /= len(active)

        # Multiple eta values
        for eta_try in [eta, eta * 0.5, eta * 2.0, 0.1, 0.05]:
            log_f = np.log(np.maximum(f, 1e-30))
            new_log = log_f - eta_try * g
            new_log -= np.max(new_log)
            f_new = np.exp(new_log)
            f_new /= f_new.sum()
            sc = score_unit(f_new)
            if sc < best_score - 1e-12:
                f = f_new
                best_f = f.copy()
                best_score = sc
                stall = 0
                break
        else:
            # No improvement, adapt eta
            eta *= 0.7
            stall += 1
            if stall > 50:
                break
            # Occasionally perturb
            if stall % 10 == 0:
                f = f + rng.normal(0, 0.005, n)
                f = normalize_arr(f)
                if f is None:
                    break

        if it % 20 == 0 and verbose:
            print(f"    MD iter={it} score={best_score:.7f} eta={eta:.4f} dt={time.monotonic()-t0:.3f}s", flush=True)

    return best_f, best_score


def coordinate_descent(f_init, max_rounds=80, n_candidates=25, verbose=True):
    """
    Fast coordinate descent with incremental convolution updates.
    """
    n = len(f_init)
    f = normalize_arr(f_init).copy()
    conv = np.convolve(f, f).copy()
    sum_f = 1.0  # f is normalized
    best_score = 2.0 * n * float(np.max(conv))
    no_improve = 0

    for rnd in range(max_rounds):
        improved = False

        for i in range(n):
            old = f[i]
            if old < 1e-14:
                # Try to activate this coordinate
                # The contribution to conv[k] from f[i]=x is:
                # 2*x*f[k-i] for k in [i, i+n-1], k != 2i
                # x^2 for k = 2i
                # Best x would balance the peak
                pass

            # Compute base (conv with f[i] removed)
            # conv[k] - contribution of f[i]
            # Contribution: 2*old*f[k-i] for k in [max(0,i), min(2n-2, i+n-1)], k != 2i
            #               old^2 for k = 2i
            base = conv.copy()
            i_lo = max(0, i)
            i_hi = min(2 * n - 2, i + n - 1)
            for k in range(i_lo, i_hi + 1):
                if k == 2 * i:
                    base[k] -= old * old
                else:
                    base[k] -= 2.0 * old * f[k - i]

            # Try candidate x values
            # Score(x) = 2n * max_k [base[k] + 2*x*f[k-i] + (x^2 if k==2i)] / (sum_f - old + x)^2
            # We search x in [0, x_max]
            x_max = max(3.0 * old, 3.0 / n, 0.01 / n)
            if old > 0:
                x_candidates = np.concatenate([
                    np.linspace(0, x_max, n_candidates),
                    np.array([old])
                ])
            else:
                x_candidates = np.linspace(0, x_max, n_candidates)

            best_x = old
            best_s = np.inf

            # Precompute the "delta" vectors for speed
            # delta_conv[k] = 2*f[k-i] (base linear coefficient)
            # We need max_k [base[k] + 2*x*f[k-i] + x^2*I(k==2i)]
            # For speed, precompute the f[k-i] values
            f_shift = np.zeros(2 * n - 1)
            for k in range(i_lo, i_hi + 1):
                if 0 <= k - i < n:
                    f_shift[k] = f[k - i]

            for x in x_candidates:
                # conv_new[k] = base[k] + 2*x*f_shift[k] + x^2 (if k==2i)
                # We only need the max, so compute efficiently
                if 2 * i < 2 * n - 1:
                    conv_new_max = np.max(base[i_lo:i_hi+1] + 2.0 * x * f_shift[i_lo:i_hi+1])
                    # Also check k = 2i separately (it's in the range)
                    val_2i = base[2*i] + x * x  # the x^2 term replaces 2*x*f[i]
                    # Actually at k=2i, the formula is base[2i] + x^2 (not 2*x*f[i])
                    # Since f[i] was removed, f_shift[2i] = f[i] = old (but we set f[i] conceptually to x)
                    # The correct formula: at k=2i, contribution is x^2
                    # base[2i] already has the old contribution removed
                    # So conv_new[2i] = base[2i] + x^2
                    conv_new_max = max(conv_new_max, base[2*i] + x * x)
                else:
                    conv_new_max = np.max(base + 2.0 * x * f_shift)

                new_sum = sum_f - old + x
                if new_sum < 1e-300:
                    continue
                s = 2.0 * n * conv_new_max / (new_sum * new_sum)
                if s < best_s:
                    best_s = s
                    best_x = x

            if best_x != old and np.isfinite(best_s):
                delta = best_x - old
                if abs(delta) > 1e-14:
                    # Update conv incrementally
                    for k in range(i_lo, i_hi + 1):
                        if k == 2 * i:
                            conv[k] = base[k] + best_x * best_x
                        else:
                            conv[k] = base[k] + 2.0 * best_x * f[k - i]
                    f[i] = best_x
                    sum_f += delta
                    # Renormalize
                    f /= sum_f
                    sum_f = 1.0
                    conv = np.convolve(f, f)  # Redo for safety after normalization

                if best_s < best_score - 1e-13:
                    best_score = best_s
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and rnd % 20 == 0:
            print(f"    CD round={rnd} score={best_score:.7f}", flush=True)

    f = normalize_arr(f)
    final_score = score_unit(f)
    if final_score < best_score:
        best_score = final_score
    return f, best_score


def choose_starts(n, rng, best_f, given_f, budget):
    """Choose diverse starting points."""
    starts = []

    # Given sequence (upsampled)
    if given_f is not None:
        g = interp_to_length(given_f, n)
        if g is not None:
            starts.append(g)
            # Perturbed version
            g2 = g + rng.normal(0, 0.02, n)
            g2 = normalize_arr(g2)
            if g2 is not None:
                starts.append(g2)

    # Best found so far (upsampled)
    if best_f is not None and (given_f is None or best_f.size != given_f.size):
        b = interp_to_length(best_f, n)
        if b is not None:
            starts.append(b)

    # Uniform
    starts.append(np.ones(n, dtype=np.float64) / n)

    # Random Dirichlet
    starts.append(rng.dirichlet(np.ones(n)))

    if budget > 60:
        # Concentrated at center
        c = np.zeros(n)
        center = n // 2
        width = max(1, n // 8)
        c[center - width:center + width + 1] = 1.0
        # Smooth it
        kernel = np.exp(-0.5 * (np.arange(-width, width + 1) / (width / 3.0))**2)
        c[center - width:center + width + 1] = kernel
        starts.append(normalize_arr(c))

        # Random sparse
        sparse = np.zeros(n)
        n_nonzero = max(4, n // 4)
        idx = rng.choice(n, size=n_nonzero, replace=False)
        sparse[idx] = rng.dirichlet(np.ones(n_nonzero))
        starts.append(sparse)

    # Deduplicate
    unique_starts = []
    for s in starts:
        s = normalize_arr(s)
        if s is None:
            continue
        dup = False
        for u in unique_starts:
            if np.linalg.norm(s - u) < 1e-4:
                dup = True
                break
        if not dup:
            unique_starts.append(s)

    return unique_starts[:5]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a low-scoring non-negative sequence minimizing
    2*n*max(convolve(f,f))/(sum(f))^2.
    
    Strategy:
    1. Quick optimization at smaller lengths to find good basins.
    2. Main optimization at n=1024 using LP + mirror descent + coordinate descent.
    3. Multiple diverse starts including the given sequence.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    start_time = time.monotonic()
    reserve = 25.0
    deadline = start_time + max(1.0, budget_s - reserve)

    best_f = None
    best_score = float('inf')
    given_f = None

    # Try to use the given sequence
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        given_f = normalize_arr(raw)
        if given_f is not None:
            best_f = given_f.copy()
            best_score = score_unit(given_f)
            print(f"Given: n={given_f.size}, score={best_score:.7f}", flush=True)
    except Exception:
        given_f = None

    if best_f is None:
        best_f = np.ones(128, dtype=np.float64) / 128.0
        best_score = 2.0

    # Phase 1: Quick optimization at smaller lengths
    small_plan = [
        (128, 20.0),
        (256, 40.0),
        (512, 80.0),
    ]

    for n, alloc in small_plan:
        rem = deadline - time.monotonic()
        if rem < 10:
            break
        alloc = min(alloc, rem - 5)
        if alloc < 5:
            break

        print(f"\n=== n={n}, budget={alloc:.0f}s, best={best_score:.7f} ===", flush=True)

        try:
            row, col, off, m = build_sparse_conv(n)
        except Exception:
            continue

        starts = choose_starts(n, rng, best_f, given_f, alloc)
        if not starts:
            starts = [np.ones(n) / n]

        # Split time among starts
        n_starts = len(starts)
        t_per = alloc / n_starts

        for si, init in enumerate(starts):
            rem = deadline - time.monotonic()
            if rem < 3:
                break
            tb = min(t_per, rem - 1)
            if tb < 2:
                continue

            # LP phase
            f_res, s_res = optimize_lp_iter(init, row, col, off, m, tb * 0.6, rng, verbose=(si == 0))
            if s_res < best_score:
                best_score = s_res
                best_f = f_res.copy()
                print(f"  n={n} start={si+1} LP score={s_res:.7f}", flush=True)

            # CD polishing
            rem = deadline - time.monotonic()
            if rem > 5 and s_res < best_score + 0.1:
                f_cd, s_cd = coordinate_descent(f_res, max_rounds=30, n_candidates=15, verbose=False)
                if s_cd < best_score:
                    best_score = s_cd
                    best_f = f_cd.copy()
                    print(f"  n={n} start={si+1} CD score={s_cd:.7f}", flush=True)

    # Phase 2: Main optimization at n=1024
    n_main = 1024
    rem = deadline - time.monotonic()
    if rem < 30:
        print("Not enough time for main phase", flush=True)
    else:
        print(f"\n=== Main phase: n={n_main}, budget={rem:.0f}s, best={best_score:.7f} ===", flush=True)

        try:
            row, col, off, m = build_sparse_conv(n_main)
        except Exception as e:
            print(f"Failed to build conv matrix: {e}", flush=True)
            row = col = off = None
            m = 2 * n_main - 1

        starts = choose_starts(n_main, rng, best_f, given_f, rem)
        if not starts:
            starts = [interp_to_length(best_f, n_main)]
            if starts[0] is None:
                starts = [np.ones(n_main) / n_main]

        n_starts = len(starts)
        # Weighted time allocation: first start gets more time
        if n_starts == 1:
            weights = [1.0]
        elif n_starts == 2:
            weights = [0.55, 0.45]
        elif n_starts == 3:
            weights = [0.45, 0.35, 0.20]
        elif n_starts == 4:
            weights = [0.40, 0.30, 0.20, 0.10]
        else:
            weights = [0.35, 0.25, 0.20, 0.12, 0.08]

        for si, init in enumerate(starts):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            t_total = min(rem - 5, (deadline - start_time) * weights[si])
            if t_total < 5:
                continue

            print(f"\n  Start {si+1}/{n_starts}, total_budget={t_total:.0f}s", flush=True)

            # Split into LP, MD, CD phases
            t_lp = t_total * 0.50
            t_md = t_total * 0.20
            t_cd = t_total * 0.30

            f_cur = init.copy()
            s_cur = best_score if si == 0 else score_unit(init)

            # LP phase
            if t_lp > 3 and row is not None:
                f_lp, s_lp = optimize_lp_iter(f_cur, row, col, off, m, t_lp, rng, verbose=True)
                if s_lp < s_cur:
                    f_cur = f_lp
                    s_cur = s_lp
                    if s_cur < best_score:
                        best_score = s_cur
                        best_f = f_cur.copy()
                        print(f"  ** New global best: {best_score:.7f} **", flush=True)

            # Mirror descent phase
            rem = deadline - time.monotonic()
            if t_md > 3 and rem > 10:
                t_md_actual = min(t_md, rem - 10)
                f_md, s_md = mirror_descent_optimize(f_cur, t_md_actual, rng, verbose=True)
                if s_md < s_cur:
                    f_cur = f_md
                    s_cur = s_md
                    if s_cur < best_score:
                        best_score = s_cur
                        best_f = f_cur.copy()
                        print(f"  ** New global best (MD): {best_score:.7f} **", flush=True)

            # Coordinate descent polishing
            rem = deadline - time.monotonic()
            if t_cd > 3 and rem > 8:
                t_cd_actual = min(t_cd, rem - 5)
                n_rounds = max(20, int(t_cd_actual / 0.5))
                f_cd, s_cd = coordinate_descent(f_cur, max_rounds=min(n_rounds, 200), n_candidates=20, verbose=True)
                if s_cd < s_cur:
                    f_cur = f_cd
                    s_cur = s_cd
                    if s_cur < best_score:
                        best_score = s_cur
                        best_f = f_cur.copy()
                        print(f"  ** New global best (CD): {best_score:.7f} **", flush=True)

        # Final round: take the best and do more CD
        rem = deadline - time.monotonic()
        if rem > 10 and best_f is not None:
            print(f"\n  Final polishing, time={rem:.0f}s", flush=True)
            f_final, s_final = coordinate_descent(best_f, max_rounds=100, n_candidates=30, verbose=True)
            if s_final < best_score:
                best_score = s_final
                best_f = f_final.copy()
                print(f"  ** Final best: {best_score:.7f} **", flush=True)

    # Prepare output
    if best_f is None:
        best_f = np.ones(64) / 64.0

    seq = [float(x) for x in best_f]
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float('inf')

    if not np.isfinite(final_score):
        seq = [1.0 / 64.0] * 64
        final_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: n={len(seq)}, local_score={best_score:.7f}, evaluate={final_score:.7f}, time={elapsed:.1f}s", flush=True)
    return seq
```
