```python
def compute_gradient_fft(f, w, T):
    """
    Compute gradient of sum_k w_k * (conv(f,f))_k with respect to f.
    f has length n, w has length 2n-1.
    """
    n = len(f)
    # Pad to avoid circular aliasing
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    F = np.fft.rfft(f, size)
    W = np.fft.rfft(w, size)

    corr = np.fft.irfft(np.conj(F) * W, size)

    grad = 2.0 * corr[:n]
```

```python
def smooth_optimize(f0, T_schedule, max_iters, time_limit):
    """
    Optimize f >= 0 to minimize 2n * softmax_T(conv(f,f)) using gradient descent with Adam.
    """
    f = np.array(f0, dtype=np.float64)
    n = len(f)
    f = np.maximum(f, 1e-10)  # ensure positive for now
    f = f / f.sum()  # normalize
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    best_f = f.copy()
    best_score = evaluate_raw(f)
    
    start = time.monotonic()
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        T = T_schedule[it] if it < len(T_schedule) else T_schedule[-1]
        lr = 0.01 / (1 + it * 0.01)  # decaying learning rate
        
        # Compute h and w
        h = fft_convolve(f, f)
        max_h = h.max()
        w = np.exp(T * (h - max_h))
        w /= w.sum()
        
        # Compute gradient
        grad = compute_gradient_fft(f, w)
        
        # Adam step
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        
        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = np.maximum(f, 0)
        s = f.sum()
        if s < 1e-10:
            f = np.ones(n) / n
        else:
            f = f / s
        
        score = evaluate_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score
```

```python
def smooth_objective_and_gradient(g, T, n):
    """
    Objective and gradient with respect to g, where f = softmax(g).
    This handles the constraint sum(f)=1, f>0 automatically.
    """
    f = softmax(g)
    h = fft_convolve(f, f)
    max_h = h.max()
    shifted = T * (h - max_h)
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum
    
    # Soft-max value
    obj = 2.0 * n * (max_h + np.log(w_sum) / T)
    
    # Gradient wrt f
    grad_f = compute_gradient_fft(f, w)
    
    # Gradient wrt g (chain rule through softmax)
    # df_i/dg_j = f_i * (delta_ij - f_j)
    # dL/dg_j = sum_i dL/df_i * f_i * (delta_ij - f_j)
    #          = f_j * dL/df_j - f_j * sum_i dL/df_i * f_i
    grad_f_dot_f = np.dot(grad_f, f)
    grad_g = f * (grad_f - grad_f_dot_f)
    
    return obj, grad_g
```

```python
"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses multi-stage optimization: L-BFGS-B with soft-max smoothing at increasing temperatures,
followed by mirror descent refinement. Multi-start across various sequence lengths (512-8192)
and initializations (cosine, Gaussian, tapered, parent).
"""

import numpy as np
import time
from scipy.optimize import minimize


def _fft_convolve(f):
    """Full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _compute_grad_fft(f, w):
    """
    Compute gradient of sum_k w_k * (conv(f,f))_k with respect to f.
    f has length n, w has length 2n-1.
    Uses FFT-based cross-correlation.
    """
    n = len(f)
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    W = np.fft.rfft(w, size)
    corr = np.fft.irfft(np.conj(F) * W, size)
    return 2.0 * corr[:n]


def _score(f):
    """Compute score = 2n*max(conv(f,f))/(sum(f))^2."""
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    h = _fft_convolve(f)
    M = float(np.max(h))
    return 2.0 * n * M / (S * S)


def _normalize(f):
    """Normalize f to sum 1, clip negatives."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    f = np.maximum(f, 0.0)
    s = f.sum()
    return f / s if s > 0 else f


def _gaussian_init(n, sigma_frac):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = f.sum()
    return f / s if s > 0 else f


def _tapered_init(n, power):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / n)
    f = x ** power
    f = np.maximum(f, 0.0)
    s = f.sum()
    return f / s if s > 0 else f

    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    width = width_frac * n / 2.0

def _smooth_obj_grad(f, T):
    """
    Compute objective and gradient of the scale-invariant soft-max smoothed objective.
    Objective: 2n * softmax_T(conv(f,f)) / (sum(f))^2
    """
    n = len(f)
    f = np.maximum(f, 0.0)
    S = float(np.sum(f))
    if S < 1e-30:
        return 1e10, np.zeros(n)

    h = _fft_convolve(f)
    m = float(np.max(h))
    shifted = T * (h - m)
    # Stable log-sum-exp
    shifted_max = float(np.max(shifted))
    log_sum = shifted_max + np.log(np.exp(shifted - shifted_max).sum())
    M_T = m + log_sum / T

    # Softmax weights
    w = np.exp(shifted - shifted_max)
    w_sum = w.sum()
    w = w / w_sum

    obj = 2.0 * n * M_T / (S * S)

    grad_obj = (2.0 * n / (S * S)) * (grad_f - 2.0 * M_T / S)

def _lbfgs_stage(f_init, T, max_iter, time_limit, tol=1e-10):
    """
    Run one stage of L-BFGS-B optimization with given temperature T.
    Returns (best_f, best_score, elapsed).
    """
    n = len(f_init)
    f_start = np.maximum(f_init, 0.0)
    S = f_start.sum()
    if S > 0:

    else:
        f_start = np.ones(n) / n

    best_f = f_start.copy()
    best_obj = _score(f_start)

    def fun(x):
        obj, grad = _smooth_obj_grad(x, T)
        return obj, grad

        result = minimize(
            fun, f_start, jac=True, method='L-BFGS-B',
            bounds=[(0, None)] * n,

        return best_f, best_obj, time.monotonic() - start_time

    f_res = np.maximum(result.x, 0.0)
    S_res = f_res.sum()
    if S_res > 0:
        f_res = f_res / S_res

    score_res = _score(f_res)
    if score_res < best_obj:
        best_obj = score_res
        best_f = f_res.copy()

    elapsed = time.monotonic() - start_time

def _mirror_descent(f_init, n_iter, eta_scale, threshold_frac, time_limit=None):
    """
    Mirror descent with multiplicative updates.
    Minimizes 2n*max(conv(f,f)) s.t. sum(f)=1, f>=0.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)

    start_time = time.monotonic()

    for it in range(n_iter):
        if time_limit is not None and time.monotonic() - start_time > time_limit:
            break

        h = _fft_convolve(f)
        M = float(np.max(h))
        if M <= 0:
            break

        threshold = threshold_frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)

        # Compute pressure vector
        p = np.zeros(n, dtype=np.float64)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                lo = k - j_end
                hi = k - j_start
                p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
        p /= float(len(active))

        eta = eta_scale / M
        log_ratio = np.clip(-eta * p, -20.0, 20.0)
        f = f * np.exp(log_ratio)
        f_norm = _normalize(f)
        if f_norm is None:
            f = np.ones(n) / n
        else:
            f = f_norm

        score = _score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

    return best_f, best_score

    """
    Adam optimizer on the smooth soft-max objective with increasing temperature.
    """
    n = len(f_init)
    f = _normalize(f_init)
    if f is None:
        return f_init, float('inf')

    best_f = f.copy()
    best_score = _score(f)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    start_time = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start_time > time_limit:
            break

        # Temperature schedule: exponential ramp
        frac = min(1.0, it / (max_iters * 0.8))
        T = T_start * (T_end / T_start) ** frac

        obj, grad = _smooth_obj_grad(f, T)

        # Adam update

        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr / (1.0 + 0.01 * it)
        f = f - step * m_hat / (np.sqrt(v_hat) + eps)
        f = np.maximum(f, 0.0)
        f_norm = _normalize(f)
        if f_norm is None:
            f = np.ones(n) / n
        else:
            f = f_norm

        score = _score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

        if (it + 1) % 500 == 0:
            print(f"        Adam it={it+1}, T={T:.1f}, score={best_score:.6f}")

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-stage search for optimal non-negative sequence.
    Combines L-BFGS-B (smooth), Adam (smooth), and mirror descent (exact max).
    """
    deadline = time.monotonic() + float(budget_s) - 10.0
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = None

    # Initialize best
    best_f = None
    best_score = float('inf')

    # Build candidate list: (name, initialization_array)
    candidates = []

    if parent is not None and len(parent) > 10:
        p = _normalize(parent)
        if p is not None:
            candidates.append(("parent_n%d" % len(p), p))

    # Various lengths and initializations
    for n in [512, 1024, 2048, 4096, 8192]:
        candidates.append(("cos_%d" % n, _cosine_init(n)))

    for n in [1024, 2048, 4096]:
        candidates.append(("gauss_%d" % n, _gaussian_init(n, 0.35)))
        candidates.append(("tap_%d" % n, _tapered_init(n, 1.5)))
        candidates.append(("bump_%d" % n, _bump_init(n, 0.8)))

    # Random inits
    for n in [1024, 2048]:
        f_r = rng.random(n) + 0.05
        candidates.append(("rand_%d" % n, _normalize(f_r)))

    print(f"Total candidates: {len(candidates)}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # Phase 1: Quick screening with Adam (fast, good for larger n)
    print("\n=== Phase 1: Adam screening ===")
    results = []

    for idx, (name, f0) in enumerate(candidates):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        n = len(f0)
        # Allocate time based on size: larger n gets more time but capped
        time_alloc = min(60.0, max(10.0, remaining * 0.15))
        n_iters = int(min(3000, time_alloc * 200))

        t0 = time.monotonic()
        f_opt, s_opt = _adam_optimize(f0, n_iters, time_alloc, lr=0.08, T_start=5.0, T_end=100.0)
        elapsed = time.monotonic() - t0

        results.append((s_opt, name, f_opt, n))
        print(f"  [{idx+1}/{len(candidates)}] {name} (n={n}): score={s_opt:.6f} ({elapsed:.1f}s)")

        if s_opt < best_score:
            best_score = s_opt
            best_f = f_opt.copy()
            print(f"  ** New best: {best_score:.6f} from {name} **")

    # Sort results by score
    results.sort(key=lambda x: x[0])

    # Phase 2: L-BFGS-B refinement on top candidates
    print("\n=== Phase 2: L-BFGS-B refinement ===")
    top_k = min(6, len(results))

    for r in range(top_k):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        s_cur, name, f_cur, n = results[r]
        time_per_stage = min(30.0, remaining * 0.2)

        if time_per_stage < 5:
            break

        print(f"\n  Refining {name} (n={n}, score={s_cur:.6f}), remaining={remaining:.0f}s")

        # Stage 1: L-BFGS-B with T=10
        if time.monotonic() < deadline - 30:
            f1, s1, _ = _lbfgs_stage(f_cur, T=10.0, max_iter=200, time_limit=time_per_stage)
            if s1 < s_cur:
                s_cur, f_cur = s1, f1
                print(f"    After L-BFGS T=10: {s_cur:.6f}")

        # Stage 2: L-BFGS-B with T=50
        remaining = deadline - time.monotonic()
        if remaining > 20:
            f2, s2, _ = _lbfgs_stage(f_cur, T=50.0, max_iter=200, time_limit=min(25.0, remaining * 0.25))
            if s2 < s_cur:
                s_cur, f_cur = s2, f2
                print(f"    After L-BFGS T=50: {s_cur:.6f}")

        # Stage 3: L-BFGS-B with T=200
        remaining = deadline - time.monotonic()
        if remaining > 15:
            f3, s3, _ = _lbfgs_stage(f_cur, T=200.0, max_iter=150, time_limit=min(20.0, remaining * 0.2))
            if s3 < s_cur:
                s_cur, f_cur = s3, f3
                print(f"    After L-BFGS T=200: {s_cur:.6f}")

        # Stage 4: Mirror descent with exact max
        remaining = deadline - time.monotonic()
        if remaining > 10:
            md_iters = int(min(3000, remaining * 150))
            f4, s4 = _mirror_descent(f_cur, md_iters, 0.4, 0.75, time_limit=min(20.0, remaining * 0.3))
            if s4 < s_cur:
                s_cur, f_cur = s4, f4
                print(f"    After mirror descent: {s_cur:.6f}")

        if s_cur < best_score:
            best_score = s_cur
            best_f = f_cur.copy()
            print(f"  ** New best: {best_score:.6f} **")

    # Phase 3: Further refinement on the overall best
    print("\n=== Phase 3: Final refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 10 and best_f is not None:
        n = len(best_f)
        print(f"  Best so far: n={n}, score={best_score:.6f}, remaining={remaining:.0f}s")

        # Try different threshold fractions with mirror descent
        for tf in [0.85, 0.9, 0.95]:
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            md_iters = int(min(2000, remaining * 100))
            f_t, s_t = _mirror_descent(best_f, md_iters, 0.2, tf, time_limit=min(10.0, remaining * 0.3))
            if s_t < best_score:
                best_score = s_t
                best_f = f_t.copy()
                print(f"    Mirror descent tf={tf}: {best_score:.6f}")

        # Try Adam with higher learning rate and more aggressive temperature
        remaining = deadline - time.monotonic()
        if remaining > 10:
            f_a, s_a = _adam_optimize(best_f, int(min(2000, remaining * 80)),
                                       min(15.0, remaining * 0.3), lr=0.03, T_start=50.0, T_end=500.0)
            if s_a < best_score:
                best_score = s_a
                best_f = f_a.copy()
                print(f"    Adam refine: {best_score:.6f}")

        # One more L-BFGS-B pass
        remaining = deadline - time.monotonic()
        if remaining > 15:
            f_f, s_f, _ = _lbfgs_stage(best_f, T=500.0, max_iter=100, time_limit=min(15.0, remaining * 0.4))
            if s_f < best_score:
                best_score = s_f
                best_f = f_f.copy()
                print(f"    L-BFGS final: {best_score:.6f}")

    # Phase 4: Try a different-length restart if time permits
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_f is not None:
        n_best = len(best_f)
        # Try next power of 2 up
        n_try = n_best * 2
        if n_try <= 16384:
            print(f"\n=== Phase 4: Try n={n_try} ===")
            # Upsample best_f to n_try
            from numpy import interp
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            f_up = interp(x_new, x_old, best_f)
            f_up = _normalize(f_up)

            time_alloc = min(60.0, remaining * 0.4)
            f_opt, s_opt = _adam_optimize(f_up, int(min(2000, time_alloc * 150)), time_alloc,
                                           lr=0.05, T_start=10.0, T_end=200.0)
            if s_opt < best_score:
                best_score = s_opt
                best_f = f_opt.copy()
                print(f"  Upsampled n={n_try}: {best_score:.6f}")

            # Quick L-BFGS refinement
            remaining = deadline - time.monotonic()
            if remaining > 15:
                f_r, s_r, _ = _lbfgs_stage(f_opt, T=50.0, max_iter=150, time_limit=min(20.0, remaining * 0.4))
                if s_r < best_score:
                    best_score = s_r
                    best_f = f_r.copy()
                    print(f"  Refined upsampled: {best_score:.6f}")

    # Final verification
    print(f"\n=== Final Result ===")
    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"Verified score: {final_score:.6f} (n={len(final_list)})")

    if not np.isfinite(final_score) or final_score > 2.5:
        print("  Score too high, trying parent...")
        if parent is not None:
            parent_score = float(evaluate_sequence([float(x) for x in parent]))
            print(f"  Parent score: {parent_score:.6f}")
            if np.isfinite(parent_score) and parent_score < final_score:
                return [float(x) for x in parent]
        # Fallback: return a cosine sequence
        f_cos = _cosine_init(1024)
        return [float(x) for x in f_cos]

    return final_list
```
