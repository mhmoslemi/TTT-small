```python
"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses differential evolution with FFT-based batch evaluation, combined with
local subgradient refinement and LP-based improvement steps.
"""

import numpy as np
import time
from scipy.optimize import linprog


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def batch_eval(population):
    """
    Batch evaluate score for a population of sequences.
    population: (P, n) ndarray, each row normalized to sum=1.
    Returns: array of P scores.
    """
    P, n = population.shape
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    
    F = np.fft.rfft(population, N, axis=1)
    C = F * F
    results = np.fft.irfft(C, N, axis=1)[:, :size]
    results = np.maximum(results, 0.0)
    
    max_vals = results.max(axis=1)
    # Each row has sum=1 by construction
    scores = 2.0 * n * max_vals
    return scores


def de_evolve(population, scores, deadline, rng, n_pop, F_start=0.9, F_end=0.3, CR=0.7,
              max_generations=2000, local_refine_interval=50):
    """
    Run differential evolution on the population.
    Returns best individual and best score.
    """
    P, n = population.shape
    best_idx = np.argmin(scores)
    best_individual = population[best_idx].copy()
    best_score = scores[best_idx]
    
    gen = 0
    no_improve = 0
    
    while time.time() < deadline and gen < max_generations:
        gen += 1
        F = F_start + (F_end - F_start) * (gen / max_generations)
        
        for i in range(P):
            if time.time() > deadline:
                break
            
            # Mutation: best/1
            r1, r2 = rng.choice(P, 2, replace=False)
            v = population[best_idx] + F * (population[r1] - population[r2])
            
            # Crossover
            trial = np.empty(n)
            j_rand = rng.integers(n)
            for j in range(n):
                if rng.random() < CR or j == j_rand:
                    trial[j] = v[j]
                else:
                    trial[j] = population[i][j]
            
            # Project to simplex
            trial = project_simplex(trial)
            
            # Evaluate
            score = 2.0 * n * np.max(fast_convolve(trial))
            
            if score < scores[i]:
                scores[i] = score
                population[i] = trial
                if score < best_score:
                    best_score = score
                    best_individual = trial.copy()
                    best_idx = i
                    no_improve = 0
            else:
                no_improve += 1
        
        # Periodic local refinement
        if local_refine_interval > 0 and gen % local_refine_interval == 0:
            if time.time() < deadline - 30:
                refined = local_refine(best_individual, deadline - 10, rng, max_iters=200)
                s = 2.0 * n * np.max(fast_convolve(refined))
                if s < best_score:
                    best_score = s
                    best_individual = refined.copy()
                    best_idx = np.argmin(scores)
                    population[best_idx] = refined.copy()
                    scores[best_idx] = s
    
    return best_individual, best_score


def local_refine(a, deadline, rng, max_iters=500):
    """
    Local refinement using subgradient descent on the peak convolution value.
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_peak = np.max(fast_convolve(a))
    
    lr = 1.0 / n
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c = fast_convolve(a)
        peak = np.max(c)
        
        if peak < best_peak * (1 - 1e-10):
            best_peak = peak
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                lr *= 0.8
                if lr < 1e-6:
                    break
                stall = 0
        
        # Find peak positions
        threshold = peak * (1 - 1e-6)
        peak_idx = np.where(c >= threshold)[0]
        if len(peak_idx) > 20:
            peak_idx = peak_idx[np.argsort(c[peak_idx])[::-1][:10]]
        
        # Subgradient
        grad = np.zeros(n)
        for k in peak_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
```

"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses a hybrid approach: mirror descent (exponentiated gradient) with
differential evolution for diversification, LP-based improvement, and
multi-length search. FFT-based convolution for efficiency.
"""

import numpy as np
import time
from scipy.optimize import linprog


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_score(a):
    """Compute the objective function value for sequence a."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * np.max(c) / (s * s)


def compute_subgradient(a, c, peak_indices):
    """Compute subgradient of max_k c[k] at point a."""
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= max(1, len(peak_indices))
    return grad


def mirror_step(a, grad, eta):
    """Exponentiated gradient (mirror descent) step."""
    # a_{new} proportional to a * exp(-grad/eta)
    log_ratio = -grad / eta
    # For numerical stability
    log_ratio -= log_ratio.max()
    new_a = a * np.exp(log_ratio)
    s = new_a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return new_a / s


def mirror_descent_optimize(a, deadline, rng, max_iters=100000, eta0=None):
    """
    Mirror descent optimization for minimizing max(convolve(a,a)) with sum(a)=1.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    if eta0 is None:
        eta0 = 1.0 / np.sqrt(n)
    
    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()
    
    eta = eta0
    stall = 0
    max_stall = 200
    it = 0
    
    while time.time() < deadline and it < max_iters:
        it += 1
        
        c = fast_convolve(a)
        peak = np.max(c)
        
        if peak < best_peak * (1 - 1e-12):
            best_peak = peak
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            eta = max(eta * 0.8, 1e-8)
            if eta < 1e-7:
                break
            # Perturb to escape
            if time.time() < deadline - 5:
                a = project_simplex(a + rng.normal(0, 0.01 / np.sqrt(n), n))
            stall = 0
        
        # Find active peaks
        threshold = peak * (1 - 1e-6)
        peak_idx = np.where(c >= threshold)[0]
        if len(peak_idx) > 30:
            # Take top peaks
            order = np.argsort(c[peak_idx])[::-1][:15]
            peak_idx = peak_idx[order]
        
        grad = compute_subgradient(a, c, peak_idx)
        
        # Mirror descent step
        a = mirror_step(a, grad, eta)
        
        # Occasionally mix with Euclidean projection step
        if it % 100 == 0:
            a_euclid = project_simplex(a - 0.5 * eta * grad)
            t = 0.5
            a = project_simplex((1 - t) * a + t * a_euclid)
    
    return best_a

    LP-based improvement: iteratively solve max sum(b) s.t. a*b <= M, b >= 0.
    """
    n = len(a)
    if n > 3000:
        return a
    
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_peak = np.max(fast_convolve(a))
    
    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break
        
        c = fast_convolve(a)
        M = np.max(c)
        
        # Build convolution matrix (sparse-friendly, but dense for simplicity)
        # A[k, j] = a[k-j] if 0 <= k-j < n
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
        
        # LP: max sum(b) s.t. A @ b <= M, b >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 5.0})
            if res.success and res.x is not None:
                b = np.maximum(res.x, 0)
                b_sum = b.sum()
                if b_sum > 1e-10:
                    b = b / b_sum  # Normalize to sum=1
                    # Blend
                    t = 0.3 + 0.4 * rng.random()
                    new_a = project_simplex((1 - t) * a + t * b)
                    new_peak = np.max(fast_convolve(new_a))
                    if new_peak < best_peak:
                        best_peak = new_peak
                        best_a = new_a.copy()
                        a = new_a
        except Exception:
            pass
    
    return best_a


def de_diversify(population, scores, n, rng, F=0.8, CR=0.7, n_new=5):
    """
    Generate n_new new candidates using DE mutation/crossover.
    Returns list of new candidates.
    """
    P = len(population)
    best_idx = np.argmin(scores)
    new_candidates = []
    
    for _ in range(n_new):
        r1, r2 = rng.choice(P, 2, replace=False)
        v = population[best_idx] + F * (population[r1] - population[r2])
        
        # Crossover
        mask = rng.random(n) < CR
        j_rand = rng.integers(n)
        mask[j_rand] = True
        trial = np.where(mask, v, population[r1])
        trial = project_simplex(trial)
        new_candidates.append(trial)
    
    return new_candidates


def alt_proj_init(n, rng, n_iter=50):
    """Alternating projection for near-flat-spectrum initialization."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    
    a = np.ones(n) / n
    for _ in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n], 0.0)
        s = a.sum()
        a = a / s if s > 1e-10 else np.ones(n) / n
        F_full = np.fft.rfft(a, N)
        F = np.exp(1j * np.angle(F_full))
        F[0] = 1.0 + 0j
    
    return project_simplex(a)
