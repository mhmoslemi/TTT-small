"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses smooth subgradient descent with adaptive temperature, LP-based
direction finding, and random local search. Stalls detection and
adaptive time allocation across multiple sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_conv_matrix_sparse(f):
    n = len(f)
    total = sum(min(k+1, n, 2*n-1-k) for k in range(2*n-1))
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp_direction(f, M):
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

def smooth_subgradient(f, deadline, T_init=None, lr_init=None, momentum=0.9, verbose_prefix=""):
    """Smooth subgradient descent with adaptive temperature and momentum."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = np.sum(f)
    if S <= 0:
        return f, float('inf')
    f = f / S  # normalize to sum 1

        T_init = np.max(np.convolve(f, f)) * 0.5
    if lr_init is None:
        lr_init = 0.01 / n
    
    T = max(T_init, 1e-6)
    lr = lr_init
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    vel = np.zeros(n)
    no_improve = 0
    
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(f, f)
        
        # Softmax weights
        max_c = np.max(conv)
        exp_c = np.exp((conv - max_c) / T)
        w = exp_c / np.sum(exp_c)
        
        # Gradient: 2 * correlate(w, f, 'valid')
        grad = 2.0 * np.correlate(w, f, mode='valid')
        
        # Momentum update
        vel = momentum * vel - lr * grad
        new_f = f + vel
        new_f = np.clip(new_f, 0, 1000)
        new_sum = np.sum(new_f)
        if new_sum > 0:
            new_f = new_f / new_sum  # maintain sum = 1
        
        new_score = evaluate_sequence(new_f.tolist())

"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines smooth subgradient descent with adaptive temperature and momentum,
LP-based direction finding, and randomized restarts. Uses FFT-accelerated
gradient computation and adaptive time allocation across multiple sequence
lengths and optimization methods.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    total = sum(min(k + 1, n, 2 * n - 1 - k) for k in range(2 * n - 1))
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def smooth_subgradient(f, deadline, T_init=None, lr_init=None, momentum=0.85,

def multiplicative_update(f, deadline, num_iters=5000, verbose_prefix=""):
    """Multiplicative update: reduce positions contributing most to peak lag."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = np.sum(f)
    if S <= 0:
        return f, float('inf')
    f = f / S

    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    rate = 0.05

    for it in range(num_iters):
        if time.monotonic() > deadline:
            break

        conv = np.convolve(f, f)
        max_k = int(np.argmax(conv))
        M = conv[max_k]
        if M <= 1e-15:
            break

        # For lag max_k, position j contributes f[j]*f[max_k-j]
        # We want to reduce the peak, so reduce f[j] where it contributes most
        contrib = np.zeros(n)
        for j in range(n):
            k2 = max_k - j
            if 0 <= k2 < n:
                contrib[j] = f[j] * f[k2]
            else:
                contrib[j] = 0.0

        # Also consider second worst lag for robustness
        sorted_conv = np.sort(conv)[::-1]
        if len(sorted_conv) > 1:
            second_k = int(np.argmax(np.where(conv == max_k, 0, conv)))
            for j in range(n):
                k2 = second_k - j
                if 0 <= k2 < n:
                    contrib[j] += 0.5 * f[j] * f[k2]

        # Multiplicative update: reduce positions with high contribution
        max_contrib = np.max(contrib) if np.max(contrib) > 0 else 1.0
        new_f = f * (1.0 - rate * contrib / max_contrib)
        new_f = np.clip(new_f, 1e-10, 1000.0)
        new_sum = float(np.sum(new_f))
        new_f = new_f / new_sum

        new_score = evaluate_sequence(new_f.tolist())
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            rate = max(rate * 0.999, 0.001)
            if (it + 1) % 500 == 0:
                print(f"      {verbose_prefix}MU iter {it+1}: score={best_score:.7f}, rate={rate:.4f}")
        else:
            rate *= 0.8
            if rate < 1e-5:
                break

    return best_f, best_score

            start = rng.integers(0, n - block_size)
            perturb[start:start + block_size] += rng.normal(0, 0.1 / np.sqrt(n), size=block_size)

        new_f = f + perturb
        new_f = np.clip(new_f, 0.0, 1000.0)
        new_sum = float(np.sum(new_f))
        if new_sum > 1e-10:
            new_f = new_f / new_sum

        new_score = evaluate_sequence(new_f.tolist())
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            if (it + 1) % 100 == 0:
                print(f"      {verbose_prefix}random: score={best_score:.7f} at iter {it+1}")

    return best_f, best_score


def binary_like_init(n, rng):
    """Create a binary-like initialization based on random subset."""
    a = rng.choice([0.0, 1.0], size=n, p=[0.3, 0.7])
    a = a + rng.uniform(0, 0.1, size=n)
    return a


def propose_candidate(budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={score:.7f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    rng = np.random.default_rng(42)

    # === Phase 1: Deep optimization on parent ===
    if best_sequence is not None:
        print("\n=== Phase 1: Deep optimization on parent ===")
        n_p = len(best_sequence)

        # 1a: Smooth subgradient (long run)
        print("  [1a] Smooth subgradient...")
        sub_dead = min(deadline, t0 + (deadline - t0) * 0.25)
        f, s = smooth_subgradient(best_sequence, sub_dead, T_init=0.01, lr_init=0.01, momentum=0.85)
        if s < best_score:
            best_score = s
            best_sequence = f.copy()
            print(f"        -> {best_score:.7f}")

        # 1b: LP iteration
        print("  [1b] LP iteration...")
        sub_dead = min(deadline, t0 + (deadline - t0) * 0.45)
        f, s = lp_iterate(best_sequence, sub_dead, step_size_init=0.08)
        if s < best_score:
            best_score = s
            best_sequence = f.copy()
            print(f"        -> {best_score:.7f}")

        # 1c: Another smooth subgradient pass with smaller T
        print("  [1c] Fine smooth subgradient...")
        sub_dead = min(deadline, t0 + (deadline - t0) * 0.60)
        f, s = smooth_subgradient(best_sequence, sub_dead, T_init=0.003, lr_init=0.003, momentum=0.9)
        if s < best_score:
            best_score = s
            best_sequence = f.copy()
            print(f"        -> {best_score:.7f}")

        # 1d: Multiplicative update
        print("  [1d] Multiplicative update...")
        sub_dead = min(deadline, t0 + (deadline - t0) * 0.72)
        f, s = multiplicative_update(best_sequence, sub_dead, num_iters=3000)
        if s < best_score:
            best_score = s
            best_sequence = f.copy()
            print(f"        -> {best_score:.7f}")

    # === Phase 2: Try different lengths and initializations ===
    print("\n=== Phase 2: Explore other configurations ===")

    lengths_to_try = [512, 256, 1024]
    if best_sequence is not None:
        n_p = len(best_sequence)
        if n_p not in lengths_to_try:
            lengths_to_try.insert(0, n_p)

    for n in lengths_to_try:
        if n < 16 or n > 4096:
            continue
        if time.monotonic() > deadline - 40:
            print("  Time running out, stopping length exploration.")
            break

        print(f"\n  Exploring n={n}...")
        time_per_n = max(20.0, (deadline - t0) * 0.08)

        initializations = []

        # Parent-based (interpolated)
        if best_sequence is not None:
            if len(best_sequence) != n:
                init_parent = np.interp(
                    np.linspace(0, 1, n),
                    np.linspace(0, 1, len(best_sequence)),
                    best_sequence
                )
            else:
                init_parent = best_sequence.copy()
            initializations.append(('parent', init_parent))

        # Uniform
        initializations.append(('uniform', np.ones(n)))

        # Cosine
        idx = np.arange(n, dtype=np.float64)
        initializations.append(('cosine', (1.0 + np.cos(np.pi * idx / max(n - 1, 1))) / 2.0 + 0.01))

        # Binary-like
        initializations.append(('binary', binary_like_init(n, rng)))

        # Sine-squared
        initializations.append(('sine', np.sin(np.pi * idx / max(n - 1, 1)) ** 2 + 0.01))

        for label, f_init in initializations:
            if time.monotonic() > deadline - 15:
                break

            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            f_sum = float(np.sum(f_init))
            if f_sum < 0.01:
                continue
            f_init = f_init / f_sum

            score = evaluate_sequence(f_init.tolist())
            if not np.isfinite(score):
                continue

            if score < best_score:
                best_score = score
                best_sequence = f_init.copy()
                print(f"    n={n} {label}: init={score:.7f} (NEW BEST)")
            else:
                print(f"    n={n} {label}: init={score:.7f}")

            # Apply smooth subgradient
            sub_dead = min(deadline - 10, time.monotonic() + time_per_n * 0.6)
            f, s = smooth_subgradient(f_init, sub_dead, T_init=0.005, lr_init=0.005, momentum=0.85)
            if s < best_score:
                best_score = s
                best_sequence = f.copy()
                print(f"    n={n} {label} -> smooth: {best_score:.7f} (NEW BEST)")

            # Apply LP if time allows
            if time.monotonic() < deadline - 20:
                sub_dead2 = min(deadline - 10, time.monotonic() + time_per_n * 0.3)
                f2, s2 = lp_iterate(f_init, sub_dead2, step_size_init=0.06)
                if s2 < best_score:
                    best_score = s2
                    best_sequence = f2.copy()
                    print(f"    n={n} {label} -> LP: {best_score:.7f} (NEW BEST)")

    # === Phase 3: Final refinement on best found ===
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print(f"\n=== Phase 3: Final refinement (n={len(best_sequence)}) ===")

        # Multi-temperature smooth subgradient
        for T_try in [0.5, 0.1, 0.01, 0.001]:
            if time.monotonic() > deadline - 3:
                break
            sub_dead = min(deadline - 2, time.monotonic() + (deadline - t0) * 0.05)
            f, s = smooth_subgradient(best_sequence, sub_dead, T_init=T_try,
                                      lr_init=0.002 / np.sqrt(len(best_sequence)),
                                      momentum=0.9)
            if s < best_score:
                best_score = s
                best_sequence = f.copy()
                print(f"  T={T_try}: {best_score:.7f} (NEW BEST)")

        # LP final push
        if time.monotonic() < deadline - 3:
            f, s = lp_iterate(best_sequence, deadline - 2, step_size_init=0.04)
            if s < best_score:
                best_score = s
                best_sequence = f.copy()
                print(f"  LP final: {best_score:.7f} (NEW BEST)")

        # Random polish
        if time.monotonic() < deadline - 2:
            f, s = random_perturb_search(best_sequence, deadline - 1, num_tries=200)
            if s < best_score:
                best_score = s
                best_sequence = f.copy()
                print(f"  random: {best_score:.7f} (NEW BEST)")

    # Fallback
    if best_sequence is None:
        print("WARNING: No valid sequence found, returning fallback.")
        return [1.0] * 128

    elapsed = time.monotonic() - t0
    n_final = len(best_sequence)
    print(f"\nFINAL RESULT: n={n_final}, score={best_score:.7f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
