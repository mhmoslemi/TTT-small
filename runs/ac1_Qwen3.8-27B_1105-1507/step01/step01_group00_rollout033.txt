import numpy as np
import time

def fft_conv(a):
    """Compute conv(a, a, 'full') using FFT."""
    n = len(a)
    fsize = 1 << (2*n - 1).bit_length()
    fa = np.fft.fft(a, fsize)
    c = np.fft.ifft(fa * fa).real[:2*n-1]
    return c

def fast_score(a):
    """Compute score quickly."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    c = fft_conv(a)
    return 2.0 * n * np.max(c) / (S * S)

def normalize(a, target_sum=None):
    """Normalize sequence to have a specific sum."""
    S = np.sum(a)
    if S <= 0:
        return a
    if target_sum is None:
        target_sum = np.sqrt(2.0 * len(a))
    return a * (target_sum / S)

def subgradient_optimize(a, deadline, K=5, verbose=True, label=""):
    """
    Subgradient optimization using top-K autocorrelation peaks.
    Uses multiplicative updates for positivity.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = fast_score(best_a)
    
    lr = 0.1
    no_improve = 0
    it = 0
    patience = 300
    
    while time.monotonic() < deadline:
        c = fft_conv(a)
        S = np.sum(a)
        
        # Top K lags
        top_k = np.argsort(c)[::-1][:K]
        max_val = c[top_k[0]]
        
        if max_val <= 1.0 + 1e-10:
            break
        
        # Combined gradient from top-K peaks
        grad = np.zeros(n)
        for k in top_k:
            weight = c[k] - np.mean(c)  # excess over average
            if weight <= 0:
                continue
            # d conv[k] / da[j] = 2*a[k-j] if 0 <= k-j < n
            j_min = max(0, k - (n - 1))
            j_max = min(int(k), n - 1)
            for j in range(j_min, j_max + 1):
                grad[j] += weight * 2.0 * a[int(k) - j]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        # Multiplicative update
        a_new = a * np.exp(-lr * grad)
        a_new = np.clip(a_new, 0, None)
        a_new = normalize(a_new)
        
        score_new = fast_score(a_new)
        it += 1
        
        if score_new < best_score - 1e-10:
            best_a = a_new.copy()
            best_score = score_new
            a = a_new
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"  {label} iter {it}: score={best_score:.6f}, lr={lr:.4f}")

            if no_improve % 50 == 0:
                lr *= 0.8
            if lr < 1e-6 or no_improve > patience:

                a = a * (1 + 0.1 * np.random.randn(n))

                a = normalize(a)
                lr = max(lr, 0.05)
                no_improve = 0
                patience = 200
                if lr < 1e-6:
                    break
    
    return best_a, best_score

def fft_conv(a):
    """Compute conv(a, a, 'full') using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    fsize = 1 << (2 * n - 1).bit_length()
    fa = np.fft.fft(a, fsize)
    return np.fft.ifft(fa * fa).real[:2 * n - 1]

        starts.append(('power', (idx / max(n - 1, 1)) ** 2 * (1 - idx / max(n - 1, 1)) ** 2 + 0.1))

        # Limit number of starts based on time
        max_starts = max(2, int(time_per_length / 40))
        starts = starts[:min(len(starts), max_starts)]

        best_for_n = None
        best_score_n = float('inf')

        for slabel, s in starts:
            if time.monotonic() > length_deadline - 5:
                break

            s = np.clip(s.astype(np.float64), 0.0, 1000.0)
            if float(np.sum(s)) < 0.01:
                continue

            s_score = fast_score(s)
            if not np.isfinite(s_score) or s_score >= 2.5:
                print(f"  {slabel}: score={s_score:.4f} (skipping, too high)")
                continue

            if s_score < best_score_n:
                best_score_n = s_score
                best_for_n = s.copy()
            if s_score < best_score:
                best_score = s_score
                best_sequence = s.copy()

            print(f"  {slabel}: init={s_score:.6f}", end="")

            # Phase 1: Variance optimization (40% of time)
            p1_dl = min(length_deadline, time.monotonic() + time_per_length * 0.35)
            va, vs = optimize_variance(s, p1_dl, lr_init=0.05, verbose=False,
                                       label=f"[{n}/{slabel}]")
            if vs < best_score_n:
                best_score_n = vs
                best_for_n = va.copy()
                if vs < best_score:
                    best_score = vs
                    best_sequence = va.copy()
            print(f" -> var: {best_score_n:.6f}", end="")

            # Phase 2: Peak optimization (25% of time)
            p2_dl = min(length_deadline, time.monotonic() + time_per_length * 0.20)
            if time.monotonic() < p2_dl:
                for K in [3, 10]:
                    pa, ps = optimize_peak(best_for_n, p2_dl, K=K, lr_init=0.08,
                                           verbose=False, label=f"[{n}/{slabel}/K{K}]")
                    if ps < best_score_n:
                        best_score_n = ps
                        best_for_n = pa.copy()
                        if ps < best_score:
                            best_score = ps
                            best_sequence = pa.copy()
            print(f" -> peak: {best_score_n:.6f}", end="")

            # Phase 3: Variance polish (remaining time for this start)
            if time.monotonic() < length_deadline - 3:
                va2, vs2 = optimize_variance(best_for_n, length_deadline, lr_init=0.02,
                                             verbose=False, label=f"[{n}/{slabel}/v2]")
                if vs2 < best_score_n:
                    best_score_n = vs2
                    best_for_n = va2.copy()
                    if vs2 < best_score:
                        best_score = vs2
                        best_sequence = va2.copy()
            print(f" -> v2: {best_score_n:.6f}")

    # Final polish on the best sequence
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print(f"\n{'='*50}")
        print(f"=== FINAL POLISH (n={len(best_sequence)}, score={best_score:.6f}) ===")
        final_dl = deadline

        # Variance polish
        fa, fs = optimize_variance(best_sequence, min(deadline, time.monotonic() + 40),
                                   lr_init=0.01, verbose=True, label="[final_var]")
        if fs < best_score:
            best_score = fs
            best_sequence = fa.copy()
            print(f"  After variance polish: {best_score:.6f}")

        # Peak polish with various K
        for K in [1, 3, 7]:
            if time.monotonic() < final_dl - 10:
                pa, ps = optimize_peak(best_sequence, min(final_dl, time.monotonic() + 25),
                                       K=K, lr_init=0.05, verbose=True,
                                       label=f"[final_K{K}]")
                if ps < best_score:
                    best_score = ps
                    best_sequence = pa.copy()
                    print(f"  After peak K={K}: {best_score:.6f}")

        # One more variance pass
        if time.monotonic() < final_dl - 5:
            fa2, fs2 = optimize_variance(best_sequence, final_dl, lr_init=0.005,
                                         verbose=True, label="[final_var2]")
            if fs2 < best_score:
                best_score = fs2
                best_sequence = fa2.copy()
                print(f"  After final variance: {best_score:.6f}")

    # Final verification
    if best_sequence is not None:
        final_eval = evaluate_sequence(best_sequence.tolist())
        fs = fast_score(best_sequence)
        print(f"\n{'='*50}")
        print(f"FINAL: n={len(best_sequence)}")
        print(f"  fast_score = {fs:.8f}")
        print(f"  eval_score = {final_eval:.8f}")
        if np.isfinite(final_eval):
            print(f"  Using eval score: {final_eval:.6f}")

    if best_sequence is None:
        return [1.0] * 512

    return [float(max(0.0, min(1000.0, x))) for x in best_sequence]
