    """Compute LSE score and gradient for f (non-negative, sum=1)."""
    n = len(f)

    max_c = float(np.max(c))

    exp_c = np.exp(shifted)

"""
Multi-strategy optimization for minimizing 2*n*max(conv(f,f))/(sum(f))^2.
Uses LSE gradient descent, LP polishing, and random restarts across multiple
sequence lengths to find a low-scoring non-negative step function.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return nonnegative vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score = 2*n*max(conv(x,x)) assuming sum(x)=1, x>=0."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def lse_score_and_grad(f, s):
    """Compute LSE(s*conv(f,f)) and its gradient w.r.t. f."""
    n = len(f)
    c = np.convolve(f, f)
    max_c = float(np.max(c))
    shifted = s * (c - max_c)
    exp_c = np.exp(shifted)
    sum_exp = float(np.sum(exp_c))
    lse_val = max_c + float(np.log(sum_exp)) / s
    w = s * c * exp_c / sum_exp
    grad = np.correlate(w, f, mode='full')[:n]
    return lse_val, grad


def lse_descent(f_init, s_list, iters_per_s, lr_init, lr_decay, time_budget):
    """LSE projected gradient descent. Returns (best_f, best_score)."""
    f = normalize_sequence(f_init)
    if f is None:
        n = len(f_init) if f_init is not None else 64
        f = np.ones(n) / n
    n = len(f)
    
    start = time.monotonic()
    best_f = f.copy()
    best_score = score_normalized(f, n)
    lr = lr_init
    global_it = 0
    
    for si, s in enumerate(s_list):
        lr = lr_init * (0.5 ** si)  # reduce LR as s increases
        for it in range(iters_per_s):
            if time.monotonic() - start > time_budget:
                break
            global_it += 1
            
            _, grad = lse_score_and_grad(f, s)
            
            # Adaptive step: scale by norm of gradient
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-15:
                break
            
            step = lr / (gnorm + 1e-10)
            f_new = f - step * grad
            f_new = np.maximum(f_new, 0.0)
            ssum = float(np.sum(f_new))
            if ssum < 1e-10:
                f_new = f.copy()
            else:
                f_new /= ssum
            
            sc = score_normalized(f_new, n)
            if sc < best_score:
                best_score = sc
                best_f = f_new.copy()
            
            f = f_new
            lr *= lr_decay
        
        if time.monotonic() - start > time_budget:
            break
    
    return best_f, best_score


def lse_descent_v2(f_init, s_list, iters_per_s, time_budget):

    Improved LSE descent with backtracking line search.
    """
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(len(f_init) if f_init is not None else 64) / 64.0
    n = len(f)
    
    start = time.monotonic()
    best_f = f.copy()
    best_score = score_normalized(f, n)
    
    for s in s_list:
        lr = 1.0 / (s * n * 0.1)  # initial LR heuristic
        for it in range(iters_per_s):
            if time.monotonic() - start > time_budget:
                break
            
            _, grad = lse_score_and_grad(f, s)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-15:
                break
            
            # Backtracking line search
            cur_score = score_normalized(f, n)
            accepted = False
            for trial in range(20):
                step = lr / (gnorm * (1.0 + trial * 0.5))
                f_trial = f - step * grad
                f_trial = np.maximum(f_trial, 0.0)
                ss = float(np.sum(f_trial))
                if ss < 1e-10:
                    continue
                f_trial /= ss
                sc = score_normalized(f_trial, n)
                if sc < cur_score - 1e-12:
                    f = f_trial
                    cur_score = sc
                    if sc < best_score:
                        best_score = sc
                        best_f = f.copy()
                    accepted = True
                    break
                lr *= 0.7

                step = 1e-4 / (gnorm + 1e-10)

                f_trial = np.maximum(f_trial, 0.0)
                ss = float(np.sum(f_trial))
                if ss > 1e-10:
                    f_trial /= ss
                    f = f_trial
            
            lr *= 0.98
    
    return best_f, best_score

def interpolate_to_length(f, new_n):
    """Resample f to new_n and normalize."""
    f = normalize_sequence(f)
    if f is None:
        return None
    m = len(f)
    if m == new_n:
        return f
    if m == 1:
        return np.ones(new_n) / new_n
    old = np.linspace(0, 1, m)
    new = np.linspace(0, 1, new_n)
    out = np.interp(new, old, f)
    return normalize_sequence(out)


def random_perturb(f, rng, frac=0.1):
    """Perturb a fraction of the sequence randomly."""
    n = len(f)
    f_new = f.copy()
    n_perturb = max(1, int(n * frac))
    indices = rng.choice(n, size=n_perturb, replace=False)
    # Add random positive noise to those positions
    noise = rng.exponential(scale=f[indices].mean() + 1e-6, size=n_perturb)
    f_new[indices] += noise
    f_new = np.maximum(f_new, 0)
    s = float(np.sum(f_new))
    if s > 0:
        f_new /= s
    return f_new


def block_swap(f, rng):
    """Swap two random blocks in the sequence."""
    n = len(f)
    if n < 4:
        return f.copy()
    block_len = max(1, n // 16)
    i = rng.integers(0, n - 2 * block_len)
    j = rng.integers(0, n - 2 * block_len)
    if abs(i - j) < block_len:
        return f.copy()
    f_new = f.copy()
    f_new[i:i+block_len], f_new[j:j+block_len] = f_new[j:j+block_len].copy(), f_new[i:i+block_len].copy()
    return f_new
