from numpy.lib.stride_tricks import sliding_window_view

    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s <= 1e-300:
        return None
    return x / s

def score_normalized(x, n):
    conv = np.convolve(x, x)
    m = np.max(conv)
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * n * m)

def compute_smoothed_obj_and_grad(f, alpha):
    n = len(f)
    c = np.convolve(f, f)
    c_max = c.max()
    shifted = alpha * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    w = exp_shifted / sum_exp
    log_sum_exp_val = alpha * c_max + np.log(sum_exp)
    
    windows = sliding_window_view(w, n)
    grad_f = 2.0 * (windows @ f)
    
    obj = (1.0 / alpha) * log_sum_exp_val
    return obj, grad_f

def simplex_project(z):
    n = len(z)
    if n <= 1:
        return np.ones(n, dtype=np.float64) / max(n, 1)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n, dtype=np.float64) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def flattop_window(n):
    """5-term flattop window."""
    k = np.arange(n, dtype=np.float64)
    if n <= 1:
        return np.ones(n) / max(n, 1)
    x = 2.0 * np.pi * k / (n - 1)
    w = (0.21557895 - 0.41663158 * np.cos(x) + 0.277263158 * np.cos(2*x)
         - 0.083578947 * np.cos(3*x) + 0.006947368 * np.cos(4*x))
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return w / s

def blackman_harris_window(n):
    """4-term Blackman-Harris window."""
    k = np.arange(n, dtype=np.float64)
    if n <= 1:
        return np.ones(n) / max(n, 1)
    x = 2.0 * np.pi * k / (n - 1)
    w = (0.36358115 - 0.48917754 * np.cos(x) + 0.13659959 * np.cos(2*x)
         - 0.01064114 * np.cos(3*x))
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return w / s

def nuttall_window(n):
    """Nuttall (minimum 4-term) window."""
    k = np.arange(n, dtype=np.float64)
    if n <= 1:
        return np.ones(n) / max(n, 1)
    x = 2.0 * np.pi * k / (n - 1)
    a = [0.3635819, 0.4891775, 0.1366027, 0.0106410]
    w = np.zeros(n)
    for i, coeff in enumerate(a):
        if i % 2 == 0:
            w += coeff * np.cos(i * x)
        else:
            w -= coeff * np.cos(i * x)
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return w / s

def make_conv_template(n):
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())
    
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    
    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)
    
    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)
    
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m

def lp_refine(f_init, time_budget, template=None):
    """LP-based refinement (similar to the previous approach but simplified)."""
    n = len(f_init)
    f = normalize_sequence(f_init).copy()
    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_time = time.monotonic()
    
    if template is None:
        template = make_conv_template(n)
    
    A_template, edge_offsets, c_obj, bounds, m = template
    
    stall = 0
    while time.monotonic() - start_time < time_budget:
        conv_ff = np.convolve(f, f)
        M = np.max(conv_ff)
        if M <= 0:
            break
        
        # Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0
        A = A_template.copy()
        A.data[:] = f[edge_offsets]
        
        b = np.full(m, M, dtype=np.float64)
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        except Exception:
            break
        
        if res.x is None:
            stall += 1
            if stall > 10:
                break
            continue
        
        g = np.maximum(res.x, 0.0)
        s_g = g.sum()
        if s_g <= 1e-300:
            stall += 1
            continue
        
        g = g / s_g
        r = np.convolve(f, g)
        r_max = np.max(r)
        
        if r_max >= M * (1 - 1e-10):
            stall += 1
            if stall > 10:
                break
            # Perturb and continue
            f = simplex_project(f + 0.1 * np.random.randn(n))
            continue
        
        # Find step size
        G = np.max(np.convolve(g, g))
        a_coef = 2.0 * (r_max - M)
        b_coef = G + M - 2.0 * r_max
        if b_coef > 1e-14:
            t_max = -a_coef / b_coef
        else:
            t_max = 1.0
        t_max = min(1.0, t_max)
        
        # Try several step sizes
        best_t = None
        best_cand_score = best_score
        for tv in [0.8 * t_max, 0.5 * t_max, 0.25 * t_max, 0.1 * t_max, 0.01 * t_max]:
            if tv <= 0:
                continue
            f_new = (1 - tv) * f + tv * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-300:
                continue
            f_new /= s
            sc = score_normalized(f_new, n)
            if sc < best_cand_score:
                best_cand_score = sc
                best_t = tv
        
        if best_t is not None and best_cand_score < best_score:
            f_new = (1 - best_t) * f + best_t * g
            f_new = np.maximum(f_new, 0)
            f_new /= f_new.sum()
            f = f_new
            best_f = f.copy()
            best_score = best_cand_score
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break
            # Perturb
            f = simplex_project(f + 0.05 * np.random.randn(n))
    
    return best_f, best_score
