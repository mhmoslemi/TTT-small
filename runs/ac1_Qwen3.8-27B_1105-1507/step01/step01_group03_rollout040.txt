"""
Smoothed gradient descent (in softmax z-space) for minimizing
2*n*max(convolve(f,f))/(sum(f))^2 with f>=0.

Uses log-sum-exp smoothing of the max operator, temperature annealing,
and multiple restarts at various sequence lengths. Includes LP-based
polishing as a final refinement step.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _normalize(arr):
    """Normalize to non-negative vector with sum 1."""
    x = np.asarray(arr, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-300:
        return None
    return x / s


def _score(f):
    """Score = 2*n*max(f*f) for sum(f)=1, f>=0."""
    q = np.convolve(f, f)
    return float(2.0 * len(f) * q.max())


def _smooth_grad_z(f, T):
    """
    Compute the natural gradient (in z-space) of log-sum-exp(q/T).
    Returns (g_z, obj_value) where g_z has the same length as f.
    """
    q = np.convolve(f, f)
    qmax = q.max()
    exp_q = np.exp((q - qmax) / T)
    denom = exp_q.sum()
    if denom < 1e-300:
        return np.zeros_like(f), float(qmax)
    w = exp_q / denom
    g_f = 2.0 * np.correlate(w, f, mode='valid')
    # Convert to z-space gradient
    fg = float(np.dot(f, g_f))
    g_z = f * (g_f - fg)
    obj = float(qmax + T * np.log(denom))
    return g_z, obj


def _upsample(f, n):
    """Upsample to length n."""
    f = _normalize(f)
    if f is None:
        return np.ones(n) / n
    if f.size == n:
        return f.copy()
    if f.size == 1:
        return np.ones(n) / n
    old = np.linspace(0, 1, f.size)
    new = np.linspace(0, 1, n)
    out = np.interp(new, old, f)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return out / s
