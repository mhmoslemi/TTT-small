"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses multi-phase approach: mirror descent for coarse optimization, 
cutting-plane LP Frank-Wolfe for refinement, and L-BFGS-B polishing.
"""

import numpy as np
import time


def _fft_conv_full(f):
    """Full linear autocorrelation via FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _score_norm(f):
    """Score for normalized f (sum=1, f>=0): 2*n*max(conv(f,f))."""
    n = len(f)
    h = _fft_conv_full(f)
    return 2.0 * n * float(np.max(h))


def _normalize(f):
    """Clip to non-negative and normalize to sum=1."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    return _normalize(f)


def _gauss_init(n, sigma_frac):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    return _normalize(f)


def _taper_init(n, power):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = np.maximum(x ** power, 0.0)
    return _normalize(f)


def _asym_init(n, alpha):
    """Asymmetric initialization: exponential tilt."""
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    tilt = np.exp(-alpha * (j / float(n) - 0.5))
    f = np.maximum(x ** 2 * tilt, 0.0)
    return _normalize(f)


def _mirror_descent(f_init, n_iter, eta_scale, thresh_start, thresh_end, decay_rate):
    """Mirror descent with adaptive threshold and step size."""
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_norm(f)

    for it in range(n_iter):
        h = _fft_conv_full(f)
        M = float(np.max(h))
        if M <= 1e-30:
            break

        # Adaptive threshold
        frac = thresh_start + (thresh_end - thresh_start) * (it / max(1, n_iter - 1))
        # Adaptive eta
        progress = it / max(1, n_iter)
        eta = eta_scale / (M * (1.0 + progress * decay_rate))

        threshold = frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)

        # Vectorized pressure computation using FFT trick
        # p[j] = (1/|K|) * sum_{k in K} f[k-j] for valid indices
        # This is equivalent to cross-correlation of f with indicator of K
        K = np.zeros(len(h), dtype=np.float64)
        K[active] = 1.0
        # p[j] = (1/|K|) * sum_k K[k] * f[k-j] = cross-corr of K and f
        # cross-corr(K, f)[j] = sum_k K[k] * f[k-j]
        # Use FFT: cross-corr = ifft(conj(FFT(K)) * FFT(f))
        size = 1
        while size < 2 * n:
            size <<= 1
        FK = np.fft.rfft(K, size)
        Ff = np.fft.rfft(f, size)
        cc = np.fft.irfft(np.conj(FK) * Ff, size)
        # cc[j] = sum_k K[k]*f[(k-j) mod size], we want the linear version
        # For our range, cc[j] for j in [0, n-1] gives sum_k K[k]*f[k-j] (with wrap)
        # Since K is zero outside [0, 2n-2], and f is zero outside [0, n-1],
        # the linear cross-corr is correct for j in [0, n-1]
        p = cc[:n] / float(len(active))

        log_ratio = np.clip(-eta * p, -30.0, 30.0)
        f = f * np.exp(log_ratio)
        f = _normalize(f)
        if f is None:
            f = np.ones(n) / n
            break

        score = _score_norm(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

    return best_f, best_score


def _lp_frank_wolfe(f_init, time_budget, max_constraints=800, step_mode='adaptive'):
    """
    LP-based Frank-Wolfe with cutting planes.
    Only solves LP with active (near-tight) constraints.
    """
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix

    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_norm(f)
    start_time = time.monotonic()
    n_lags = 2 * n - 1

    # Active constraint set
    active_set = set()

    for it in range(500):
        if time.monotonic() - start_time > time_budget:
            break

        h = np.convolve(f, f)
        M = float(np.max(h))

        # Update active set every few iterations
        if it % 3 == 0:
            # Keep constraints where h[k] > 0.4*M
            candidates = np.where(h > 0.4 * M + 1e-12)[0]
            if len(candidates) > max_constraints:
                # Keep the top max_constraints
                top_idx = np.argsort(h)[::-1][:max_constraints]
                active_set = set(top_idx.tolist())
            else:
                active_set = set(candidates.tolist())
            # Always include argmax
            active_set.add(int(np.argmax(h)))
        else:
            # Add newly violated
            argmax_k = int(np.argmax(h))
            if h[argmax_k] > M * (1 + 1e-6):
                active_set.add(argmax_k)

        if not active_set:
            break

        # Build sparse matrix for active constraints
        active_list = sorted(active_set)
        m_c = len(active_list)

        row_idx = []
        col_idx = []
        vals = []
        for i, k in enumerate(active_list):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                # coeff of b[j] in (f*b)_k is f[k-j]
                js = np.arange(j_start, j_end + 1)
                row_idx.extend([i] * len(js))
                col_idx.extend(js.tolist())
                vals.extend(f[k - js].tolist())

        if not row_idx:
            break

        C = coo_matrix((vals, (row_idx, col_idx)), shape=(m_c, n)).tocsr()
        c_obj = -np.ones(n)
        b_ub = np.full(m_c, M * (1 + 1e-9))

        res = linprog(c_obj, A_ub=C, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')

        if not res.success or res.x is None:
            # Fallback: use gradient step
            p = np.zeros(n)
            for k in active_list[:50]:
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                if j_end >= j_start:
                    p[j_start:j_end + 1] += 2.0 * f[k - j_end:k - j_start + 1][::-1]
            p /= min(len(active_list), 50)
            g = np.maximum(f - 0.1 * p, 0.0)
            g = _normalize(g)
            if g is None:
                break
        else:
            g = res.x
            Sg = float(np.sum(g))
            if Sg < 1e-10:
                break
            g = g / Sg

        # Step size
        if step_mode == 'adaptive':
            # Decreasing step with restart
            t = 2.0 / (it + 2)
            # With a cap to avoid too-small steps
            t = max(t, 0.02)
        elif step_mode == 'constant':
            t = 0.15
        else:
            t = 1.0 / (1.0 + 0.3 * it)

        f_new = (1.0 - t) * f + t * g
        f = _normalize(f_new)
        if f is None:
            break

        score = _score_norm(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

        if (it + 1) % 10 == 0:
            print(f"    FW it={it+1}: score={best_score:.6f}, M={M:.6f}, |AC|={len(active_list)}")

    return best_f, best_score


def _lbfgs_polish(f_init, time_budget, beta=20.0):
    """Polish with L-BFGS-B using smoothed max (Gumbel-softmax)."""
    from scipy.optimize import minimize

    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_norm(f)
    start_time = time.monotonic()

    # Log parameterization
    f_clip = np.maximum(f, 1e-15)
    x0 = np.log(f_clip - f_clip.min() + 1e-30)

    def obj_and_grad(x):
        # Softmax parameterization
        x_shift = x - np.max(x)
        f_x = np.exp(x_shift)
        sx = np.sum(f_x)
        f_x = f_x / sx
        h = _fft_conv_full(f_x)
        M = float(np.max(h))
        # Smoothed max via Gumbel-softmax
        exp_h = np.exp(beta * (h - M) / max(M, 1e-10))
        w = exp_h / np.sum(exp_h)
        smooth_max = float(np.dot(w, h))
        score = 2.0 * n * smooth_max
        if time.monotonic() - start_time > time_budget:
            return score, None

        # Gradient of smooth_max wrt f
        grad_f = np.zeros(n)
        for k in range(len(h)):
            if w[k] < 1e-10:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                grad_f[j_start:j_end + 1] += 2.0 * w[k] * f_x[k - j_end:k - j_start + 1][::-1]

        # Chain rule: f_x = exp(x)/sum(exp(x))
        # d score / dx_i = 2*n * sum_j (d smooth_max / d f_j) * (df_j/dx_i)
        # df_j/dx_i = f_j * (delta_ji - f_i)
        grad_x = f_x * (grad_f - np.dot(grad_f, f_x)) * 2.0 * n
        return score, grad_x

    x_cur = x0.copy()
    n_calls = 0

    while time.monotonic() - start_time < time_budget:
        res = minimize(obj_and_grad, x_cur, method='L-BFGS-B',
                       options={'maxiter': 50, 'maxfun': 200,
                                'gtol': 1e-4, 'maxls': 20})
        if res.x is not None:
            x_cur = res.x
            # Compute actual (non-smoothed) score
            f_x = np.exp(x_cur - np.max(x_cur))
            f_x = f_x / np.sum(f_x)
            actual_score = _score_norm(f_x)
            if actual_score < best_score:
                best_score = actual_score
                best_f = f_x.copy()
                print(f"    L-BFGS: score={best_score:.6f}")
        else:
            break
        # Reduce beta for sharper max
        beta *= 1.5
        if beta > 100:
            break

    return best_f, best_score


def _try_resample(f_orig, target_n):
    """Resample f to target_n using cubic interpolation, then normalize."""
    from scipy.interpolate import interp1d
    n_orig = len(f_orig)
    x_orig = np.linspace(0, 1, n_orig)
    x_new = np.linspace(0, 1, target_n)
    try:
        f_new = interp1d(x_orig, f_orig, kind='cubic', fill_value='extrapolate')(x_new)
        f_new = np.maximum(f_new, 0.0)
        return _normalize(f_new)
    except:
        return None
