4. Perturbation-based local search

    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def build_A_matrix(a):
    """Build (2n-1) x n matrix where row k is the k-th row of the convolution operator."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh+1] = a[k-jh:k-jl+1][::-1]
    return A

def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = r.x
        if g.sum() > 1e-10:
            return np.maximum(g, 0)
    return None

    """One step of spectral flatness: blend power spectrum towards flat."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a_padded = np.zeros(N)
    a_padded[:n] = a
    A = np.fft.fft(a_padded)
    total_sum = a.sum()
    target_power = np.full(N, total_sum**2 / N)
    current_power = np.abs(A)**2
    new_power = (1.0 - alpha) * current_power + alpha * target_power
    new_mag = np.sqrt(np.maximum(new_power, 0.0))
    new_phase = np.angle(A)
    A_new = new_mag * np.exp(1j * new_phase)
    a_new = np.fft.ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

    """Log-sum-exp smoothed objective for use with scipy.minimize."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()

    ax = ax / s
    b = fast_conv(ax)
    eps = _obj_eps
    b_max = b.max()
    return float(b_max + eps * np.log(np.sum(np.exp((b - b_max) / eps))))

def lp_mirror_descent(a, max_iters, deadline):
    """Run LP-based mirror descent iterations. Returns improved a."""
    a = np.clip(a, 0.01, None).astype(np.float64)
    a = a / a.sum()
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:

        g = g / g.sum()
        t_mix = max(0.03, 0.5 * (1.0 - it / max_iters))
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:

        a = a_new / s
    return a

def spectral_optimize(a, deadline, max_total_iters=200):
    """Run spectral flatness optimization with decreasing alpha."""
    alphas = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03]

            s = a.sum()
            if s > 1e-10:
