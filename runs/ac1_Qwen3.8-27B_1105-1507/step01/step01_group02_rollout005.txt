def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def compute_subgradient(a, peak_indices, c):
    """Compute subgradient of max_k c[k] at point a, weighted by activity."""
    n = len(a)
    grad = np.zeros(n)
    # Weight by how close each peak is to the max
    peak_vals = c[peak_indices]
    max_val = peak_vals.max()
    # Exponential weighting: peaks closer to max get more weight
    weights = np.exp(10 * (peak_vals - max_val) / max_val) if max_val > 0 else np.ones(len(peak_vals))
    weights /= weights.sum()

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            seg = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * weights[i] * seg
    return grad

"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses smooth max approximation with FISTA (accelerated projected gradient)
and adaptive temperature annealing. Combines with LP refinement and
multiple initialization strategies for global exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_mask = u * np.arange(1, n + 1) > cv
    rho_candidates = np.where(rho_mask)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def smooth_max_and_grad(a, tau):
    """
    Compute smooth max approximation of max(conv(a,a)) and its gradient.
    The smooth max is: tau * log(sum(exp(c_k / tau))) where c = conv(a,a).
    Gradient computed via FFT-based cross-correlation.
    Returns (smooth_value, gradient_array).
    """
    n = len(a)
    c = fast_convolve(a)
    
    # Numerically stable softmax
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    p = exp_c / exp_c.sum()
    smooth_val = c_max + tau * np.log(exp_c.sum())
    
    # Gradient via FFT cross-correlation
    # grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    size = 2 * n - 1
    N = 1 << (2 * size - 2).bit_length()
    fp = np.fft.rfft(p, N)
    fa_r = np.fft.rfft(a[::-1], N)
    conv_pa = np.fft.irfft(fp * fa_r, N)[:2 * size - 1]
    grad = 2.0 * conv_pa[n - 1: 2 * n - 1]
    
    return smooth_val, grad

def fista_optimize(a_init, deadline, rng, tau_init=0.01, tau_min=0.001, max_iters=100000):
    """
    FISTA (accelerated projected gradient) on the smooth max approximation.
    Anneals tau from tau_init to tau_min.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy().astype(np.float64))
    a_prev = a.copy()
    t = 1.0
    
    c = fast_convolve(a)
    best_peak = c.max()
    best_a = a.copy()
    
    # Step size (will be adapted)
    step = 1.0 / (2.0 * n)  # Initial guess
    step_min = 1e-8
    
    # Annealing schedule for tau
    total_iters_planned = max_iters
    it = 0
    stall = 0
    max_stall = 300
    
    while time.time() < deadline and it < max_iters:
        it += 1
        
        # Anneal tau
        progress = min(1.0, it / total_iters_planned)
        tau = tau_init * (tau_min / tau_init) ** progress
        
        # FISTA momentum
        if it > 1:
            t_prev = t
            t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev))
            y = a + (t_prev - 1.0) / t * (a - a_prev)
            y = np.maximum(y, 0.0)  # Keep in feasible region for gradient eval
        else:
            y = a.copy()
            t = 1.0
        
        # Compute gradient at y
        _, grad = smooth_max_and_grad(y, tau)
        
        # Gradient step with projection
        a_new = project_simplex(y - step * grad)
        
        # Backtracking line search (simplified)
        c_new = fast_convolve(a_new)
        peak_new = c_new.max()
        
        if peak_new > best_peak * (1.0 + 0.01):
            # Step was too large, reduce
            step = max(step * 0.5, step_min)
            a_new = project_simplex(y - step * grad)
        
        if peak_new < best_peak * (1.0 - 1e-10):
            best_peak = peak_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a_prev = a
        a = a_new
        
        # Handle stalls
        if stall > max_stall:
            step = max(step * 0.5, step_min)
            stall = 0
            if step < step_min * 10 and time.time() < deadline - 10:
                # Perturb to escape
                strength = 0.05 / np.sqrt(n)
                a = project_simplex(a + strength * rng.standard_normal(n))
                a_prev = a.copy()
                t = 1.0
    
    return best_a

def subgradient_optimize(a_init, deadline, rng, max_iters=200000):
    """
    Projected subgradient descent with adaptive step size.
    Direct minimization of max(conv(a,a)) without smooth approximation.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy().astype(np.float64))
    
    c = fast_convolve(a)
    best_peak = c.max()
    best_a = a.copy()
    
    # Polyak step size
    base_lr = 1.0 / n
    lr = base_lr
    lr_min = base_lr * 1e-6
    
    # Momentum
    v = np.zeros(n)
    beta = 0.5
    it = 0
    stall = 0
    
    while time.time() < deadline and it < max_iters:
        it += 1
        
        c = fast_convolve(a)
        peak = c.max()
        
        if peak < best_peak * (1.0 - 1e-10):
            best_peak = peak
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive learning rate
        if stall > 100:
            lr = max(lr * 0.8, lr_min)
            stall = 0
            if lr < lr_min * 100 and time.time() < deadline - 10:
                a = project_simplex(a + 0.05 / np.sqrt(n) * rng.standard_normal(n))
                v = np.zeros(n)
        
        # Active set: top peaks
        threshold = peak * (1.0 - 0.005)
        peak_idx = np.where(c >= threshold)[0]
        if len(peak_idx) > 30:
            top = np.argsort(c[peak_idx])[::-1][:15]
            peak_idx = peak_idx[top]
        
        # Weighted subgradient
        grad = np.zeros(n)
        peak_vals = c[peak_idx]
        # Linear weighting
        w = (peak_vals - peak_vals.min() + 1e-10)
        w /= w.sum()
        
        for i, k in enumerate(peak_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi+1] += 2.0 * w[i] * a[k-hi:k-lo+1][::-1]
        
        # Momentum step
        v = beta * v - lr * grad
        a = project_simplex(a + v)
    
    return best_a

def lp_refine(a, deadline, max_calls=8):
    """LP-based refinement: maximize sum(b) s.t. conv(a,b) <= max(conv(a,a)), b >= 0."""
    n = len(a)
    if n > 1500:
        return a
    
    best_a = a.copy()
    best_peak = fast_convolve(best_a).max()
    calls = 0
    
    while time.time() < deadline - 3 and calls < max_calls:
        calls += 1
        P = fast_convolve(best_a).max()
        
        # Build A matrix: A[k, j] = a[k-j] if valid
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[k, lo:hi+1] = best_a[k-hi:k-lo+1][::-1]
        
        try:
            res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, P),
                         bounds=[(0, None)]*n, method='highs',
                         options={'time_limit': 8})
            if res.success and res.x is not None:
                b = np.maximum(res.x, 0)
                b_sum = b.sum()
                if b_sum > 1e-8:
                    b = b * (best_a.sum() / b_sum)
                    for t_mix in [0.2, 0.4, 0.6]:
                        a_try = project_simplex((1-t_mix)*best_a + t_mix*b)
                        p_try = fast_convolve(a_try).max()
                        if p_try < best_peak:
                            best_a = a_try
                            best_peak = p_try
                            break
        except Exception:
            pass
    
    return best_a

def alt_proj_init(n, rng, n_iter=50):
    """Alternating projection initialization for flat-spectrum sequences."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0, 2*np.pi, N//2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0
    
    a = np.ones(n) / n
    for _ in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n], 0.0)
        s = a.sum()
        a = a / s if s > 1e-10 else np.ones(n) / n
        F_full = np.fft.rfft(a, N)
        w = np.linspace(1.0, 0.85, len(F_full))
        F = (w ** 0.05) * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real
    return a

def resample_sequence(a, target_n):
    """Resample a sequence to a target length using linear interpolation."""
    if len(a) == target_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, target_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    return a_new / s if s > 1e-10 else np.ones(target_n) / target_n
