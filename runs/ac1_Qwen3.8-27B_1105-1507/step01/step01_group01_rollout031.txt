"""
Search for low-score non-negative step-function coefficients using smooth gradient descent.

The objective is score = 2*n*max(convolve(a,a)) / sum(a)^2.
We use a logsumexp smooth approximation of max, compute the exact gradient via
correlation, and perform projected gradient descent onto the simplex.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def normalize_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s <= 0:
        return np.ones_like(a) / len(a)
    return a / s


def compute_score(a):
    """Compute the exact score."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return float('inf')
    a = np.clip(a, 0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def smooth_score_grad(a, beta):
    """Compute smooth score and its gradient using logsumexp.
    
    Returns (smooth_score, gradient) where gradient is w.r.t. a (unconstrained).
    We work with sum(a)=1 normalization.
    """
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    c = np.convolve(a, a)  # length 2n-1
    
    # Stable logsumexp
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    sum_exp = exp_c.sum()
    p = exp_c / sum_exp  # softmax weights
    
    # smooth_max = c_max + log(sum_exp) / beta
    # score = 2*n*smooth_max (since sum(a)=1)
    smooth_max = c_max + np.log(sum_exp) / beta
    score = 2.0 * n * smooth_max
    
    # gradient of smooth_max w.r.t. a[j] = 2 * correlate(p, a, 'valid')[j]
    # p has length 2n-1, a has length n
    # correlate(p, a, 'valid')[j] = sum_{m=0}^{n-1} a[m] * p[m+j]
    grad = 2.0 * np.correlate(p, a, mode='valid')
    
    return score, grad


def simplex_projection(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # If all positive and sum is 1, just normalize
    # Otherwise, find threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_gradient_optimize(a, max_time, seed, beta_start=1.0, beta_end=100.0, 
                               n_anneal=200, lr_init=0.01):
    """Optimize using smooth gradient descent with beta annealing."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    a = simplex_projection(a)
    
    best = a.copy()
    best_score = compute_score(best)
    
    if max_time <= 0:
        return best, best_score
    
    t0 = time.monotonic()
    deadline = t0 + max_time
    rng = np.random.default_rng(seed)
    
    lr = lr_init
    momentum = np.zeros(n)
    beta = beta_start
    stall = 0
    
    while time.monotonic() < deadline:
        # Anneal beta
        progress = min(1.0, (time.monotonic() - t0) / max(n_anneal, 1))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute gradient
        score, grad = smooth_score_grad(a, beta)
        
        # Momentum update
        momentum = 0.9 * momentum + 0.1 * grad
        a_new = a - lr * momentum
        a_new = simplex_projection(a_new)
        
        # Evaluate actual score
        actual_score = compute_score(a_new)
        
        if actual_score < best_score:
            best = a_new.copy()
            best_score = actual_score
            stall = 0
            # Maybe increase lr slightly
            lr = min(lr * 1.05, 0.5)
        else:
            a = best.copy()  # revert
            stall += 1
            if stall > 50:
                lr *= 0.5
                if lr < 1e-6:
                    # Restart with perturbation
                    noise = np.exp(rng.normal(0, 0.1, n))
                    a = simplex_projection(best * noise)
                    best = a.copy()
                    best_score = compute_score(best)
                    lr = lr_init * 0.5
                    momentum = np.zeros(n)
                    stall = 0
        
        a = a_new if np.isfinite(actual_score) and actual_score < best_score else best.copy()
    
    return best, best_score

"""
Search for low-score non-negative step-function coefficients using smooth gradient descent
with logsumexp, spectral compression, and LP-augmented direction finding.

Key differences from prior approach:
- Exact gradient via correlate (not just top-K peaks)
- Smooth annealing of beta (crease trick)
- Spectral compression steps to flatten autocorrelation in frequency domain
- Multiplicative (mirror descent) updates for non-negativity
- Adaptive learning rate with backtracking
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def simplex_finalize(a):
    """Ensure a is non-negative, normalized to sum 1, finite."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1e6)
    if not np.isfinite(a).all():
        return None
    s = a.sum()
    if s < 1e-300:
        return None
    return a / s


def score_of(a):
    """Compute exact score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1e6)
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def smooth_grad(a, beta):
    """Compute gradient of 2*n*smooth_max(conv(a,a)) w.r.t. a, assuming sum(a)=1.
    
    Returns (smooth_score, grad) where grad has same shape as a.
    """
    c = np.convolve(a, a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    if sum_exp < 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_shifted / sum_exp
    smooth_max = c_max + np.log(sum_exp) / beta
    score_val = 2.0 * len(a) * smooth_max
    # grad[j] = 2 * correlate(p, a, 'valid')[j]
    # p has length 2n-1, a has length n
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return score_val, grad


def multiplicative_step(a, grad, lr):
    """Apply multiplicative (mirror descent) update: a[j] *= exp(lr*(mean_grad - grad[j]))."""
    n = a.size
    mean_g = grad.mean()
    # Scale gradient for numerical stability
    g_max = np.abs(grad).max()
    if g_max < 1e-15:
        return a.copy()
    scaled = (mean_g - grad) / g_max
    # Clip to avoid overflow
    scaled = np.clip(scaled, -10.0, 10.0)
    a_new = a * np.exp(lr * scaled)
    a_new = np.clip(a_new, 0.0, 1e6)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s


def spectral_step(a, alpha):
    """Apply spectral compression: flatten |FFT(a)|^2 in frequency domain.
    
    alpha < 1 compresses the spectrum (flattens autocorrelation).
    """
    n = a.size
    # Pad to 2n-1 for proper aperiodic autocorrelation via FFT
    L = 2 * n - 1
    A = np.fft.rfft(a, L)
    power = np.abs(A) ** 2
    # Compress: power_new = power^alpha (alpha in (0,1])
    power = np.maximum(power, 0.0)
    pmax = power.max()
    if pmax < 1e-300:
        return a.copy()
    normalized_power = power / pmax
    compressed = np.power(normalized_power, alpha)
    # Reconstruct: we want |A_new|^2 proportional to compressed
    # Keep phase of A, modify magnitude
    phase = np.angle(A)
    mag_new = np.sqrt(np.maximum(compressed, 0.0))
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, L)[:n]
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s


def lp_direction(a, M_mult=1.0):
    """Solve LP to find a good direction b: max sum(b) s.t. conv(a,b) <= M, b in [0,1].
    
    Returns b normalized, or None.
    """
    if not HAVE_SCIPY:
        return None
    n = a.size
    c_aa = np.convolve(a, a)
    M = c_aa.max() * M_mult
    if M < 1e-15:
        return None
    # Build constraint matrix: A_ub[k, j] such that sum_j A_ub[k,j]*b[j] <= M
    # conv(a,b)[k] = sum_j a[k-j]*b[j]
    A_ub = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A_ub[k, j] = a[k - j]
    b_ub = np.full(2 * n - 1, M * 1.0001)
    c_obj = -np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception:
        return None
    if res is None or not res.success:
        return None
    b = np.clip(res.x, 0.0, 1.0)
    s = b.sum()
    if s < 1e-10:
        return None
    return b / s


def make_initial(n, seed, kind=None):
    """Generate a diverse initial candidate."""
    rng = np.random.default_rng(seed)
    if kind is None:
        kind = int(rng.integers(0, 8))
    
    if kind == 0:
        # Uniform with noise
        a = rng.uniform(0.5, 1.5, n)
    elif kind == 1:
        # Beta distributed
        a = rng.beta(2.0, 2.0, n)
    elif kind == 2:
        # Bimodal (peaks at edges)
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * (0.5 + 0.5 * np.cos(2 * np.pi * x))
    elif kind == 3:
        # Multi-modal
        nmod = int(rng.integers(2, max(3, n // 64)))
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for m in range(nmod):
            center = (m + 0.5) / nmod
            width = 1.0 / (nmod * 3)
            a += np.exp(-0.5 * ((x - center) / width) ** 2)
    elif kind == 4:
        # Smooth random (low-pass filtered)
        a = rng.normal(0, 1, n)
        # Simple moving average
        w = max(3, n // 32)
        kernel = np.ones(w) / w
        a = np.convolve(a, kernel, mode='same')
        a = np.clip(a, 0.01, None)
    elif kind == 5:
        # Power law decay from center
        x = np.linspace(-1, 1, n)
        p = float(rng.uniform(0.5, 2.0))
        a = np.exp(-p * np.abs(x))
    elif kind == 6:
        # Alternating pattern (sparse)
        a = np.zeros(n)
        a[::2] = 1.0
        a += rng.uniform(0, 0.3, n)
    else:
        # Random permutation of smooth values
        a = rng.uniform(0.1, 1.0, n)
        a = np.convolve(a, np.ones(max(3, n // 16)) / max(3, n // 16), mode='same')
        a = np.clip(a, 0.01, None)
    
    return simplex_finalize(a)


def optimize_single(a, max_time, seed, use_lp=True):
    """Main optimization loop for a single starting point at fixed length.
    
    Combines smooth gradient descent (multiplicative), spectral compression,
    and optional LP steps.
    """
    a = simplex_finalize(a)
    if a is None:
        return None, float('inf')
    n = a.size
    
    best = a.copy()
    best_sc = score_of(best)
    
    if max_time <= 0.5:
        return best, best_sc
    
    t0 = time.monotonic()
    deadline = t0 + max_time
    rng = np.random.default_rng(seed)
    
    lr = 0.3
    beta = 5.0
    stall_count = 0
    
    it = 0
    last_lp = 0
    last_spec = 0
    spec_cooldown = max(10, n // 50)
    lp_cooldown = max(50, n // 5)
    
    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 0.1:
            break
        
        it += 1
        
        # Anneal beta: start smooth, become sharp
        progress = min(1.0, it / max(100, n // 2))
        beta = 5.0 + 95.0 * (progress ** 1.5)
        
        # --- Gradient step ---
        _, grad = smooth_grad(a, beta)
        a_new = multiplicative_step(a, grad, lr)
        sc_new = score_of(a_new)
        
        if sc_new < best_sc - 1e-12:
            best = a_new.copy()
            best_sc = sc_new
            a = a_new
            stall_count = 0
            # Slowly increase lr
            lr = min(lr * 1.01, 2.0)
        elif sc_new < best_sc + 0.01 * best_sc:
            # Accept slightly worse (simulated annealing)
            a = a_new
            stall_count += 1
        else:
            stall_count += 1
            if stall_count > 30:
                lr *= 0.7
                if lr < 0.001:
                    # Strong restart
                    noise = np.exp(rng.normal(0, 0.3, n))
                    a = simplex_finalize(best * noise)
                    if a is None:
                        a = best.copy()
                    best = a.copy()
                    best_sc = score_of(best)
                    lr = 0.3
                    stall_count = 0
        
        # --- Spectral step (periodic) ---
        if it % spec_cooldown == 0:
            alpha = float(rng.uniform(0.3, 0.7))
            a_spec = spectral_step(a, alpha)
            sc_spec = score_of(a_spec)
            if sc_spec < best_sc:
                best = a_spec.copy()
                best_sc = sc_spec
                a = a_spec
                stall_count = 0
                last_spec = it
            # Also try blending
            for t in (0.3, 0.5, 0.7):
                a_blend = simplex_finalize((1 - t) * best + t * a_spec)
                if a_blend is not None:
                    sc_b = score_of(a_blend)
                    if sc_b < best_sc:
                        best = a_blend.copy()
                        best_sc = sc_b
                        a = a_blend
                        stall_count = 0
        
        # --- LP step (infrequent, expensive) ---
        if use_lp and HAVE_SCIPY and it - last_lp > lp_cooldown and rem > 10.0 and n <= 1024:
            b = lp_direction(best, M_mult=1.0)
            if b is not None:
                for t in (0.1, 0.2, 0.3, 0.5):
                    a_lp = simplex_finalize((1 - t) * best + t * b)
                    if a_lp is not None:
                        sc_lp = score_of(a_lp)
                        if sc_lp < best_sc:
                            best = a_lp.copy()
                            best_sc = sc_lp
                            a = a_lp
                            stall_count = 0
                            last_lp = it
                            break
        
        # --- Random perturbation (when stuck) ---
        if stall_count > 80:
            # Local perturbation
            w = max(1, n // 20)
            start = int(rng.integers(0, n - w + 1))
            a_pert = a.copy()
            a_pert[start:start + w] *= np.exp(rng.normal(0, 0.2, w))
            a_pert = simplex_finalize(a_pert)
            if a_pert is not None:
                sc_p = score_of(a_pert)
                if sc_p < best_sc:
                    best = a_pert.copy()
                    best_sc = sc_p
                    a = a_pert
            stall_count = 40  # partial reset
            lr = max(lr, 0.1)
    
    return best, best_sc


def optimize_length(n, budget, seed, parent=None):
    """Try multiple initializations at length n, return best."""
    if n <= 1 or budget <= 1.0:
        return None, float('inf')
    
    t0 = time.monotonic()
    deadline = t0 + budget
    best = None
    best_sc = float('inf')
    
    # Try parent first if available
    if parent is not None and parent.size == n:
        p = simplex_finalize(parent)
        if p is not None:
            sc = score_of(p)
            # Quick optimize
            t_rem = deadline - time.monotonic()
            b_local, sc_local = optimize_single(p, min(0.3 * t_rem, 60.0), seed, use_lp=(n <= 512))
            if b_local is not None and sc_local < sc:
                best, best_sc = b_local, sc_local
            else:
                best, best_sc = p, sc
    
    # Try diverse random starts
    n_starts = min(6, max(2, int(budget / 30)))
    for i in range(n_starts):
        t_rem = deadline - time.monotonic()
        if t_rem < 5.0:
            break
        # Allocate time proportionally
        alloc = (t_rem - 5.0) / (n_starts - i) * 0.9
        start_kinds = [2, 3, 0, 4, 1, 5]  # bias toward structured starts
        
        # If we have a parent, also try interpolating with random
        a0 = make_initial(n, seed + 1000 * i + start_kinds[i % len(start_kinds)], kind=start_kinds[i % len(start_kinds)])
        
        # Blend with parent if available
        if best is not None and i < 2:
            a0 = simplex_finalize(0.5 * a0 + 0.5 * best)
            if a0 is None:
                a0 = make_initial(n, seed + 2000 * i, kind=0)
        
        b_local, sc_local = optimize_single(a0, alloc, seed + 7777 * (i + 1), use_lp=(n <= 512))
        if b_local is not None and sc_local < best_sc:
            best = b_local
            best_sc = sc_local
            print(f"  length={n} start={i} score={best_sc:.6f}", flush=True)
    
    return best, best_sc


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main entry point: search for the best sequence within the time budget."""
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 1500.0
    deadline = start_time + max(10.0, budget) - 15.0  # leave 15s for final polish/validation
    
    # Try to get parent
    parent = None
    try:
        parent = globals().get('height_sequence_1', None)
    except Exception:
        parent = None
    parent_arr = None
    if parent is not None:
        try:
            parent_arr = np.asarray(parent, dtype=np.float64)
            parent_arr = np.clip(parent_arr, 0.0, 1000.0)
            if parent_arr.size == 0 or not np.isfinite(parent_arr).all():
                parent_arr = None
        except Exception:
            parent_arr = None
    
    best_seq = None
    best_score = float('inf')
    
    # Initialize with parent
    if parent_arr is not None:
        p = simplex_finalize(parent_arr)
        if p is not None:
            best_seq = p.copy()
            best_score = score_of(p)
            print(f"Parent: n={p.size} score={best_score:.6f}", flush=True)
    
    # Length plan: focus on lengths where we expect good scores
    # Based on analysis, n=512-2048 tends to work well
    if budget >= 1200:
        plan = [
            (256, 40, 0),
            (512, 300, 1),
            (512, 200, 2),
            (1024, 250, 0),
            (2048, 200, 0),
            (512, 200, 3),
        ]
    elif budget >= 600:
        plan = [
            (512, 200, 0),
            (512, 150, 1),
            (1024, 150, 0),
            (2048, 100, 0),
        ]
    else:
        plan = [
            (256, 80, 0),
            (512, 200, 0),
            (512, 150, 1),
        ]
    
    seed_cur = int(seed)
    for n, alloc, variant in plan:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        b = min(float(alloc), rem - 10.0)
        if b < 10:
            b = max(10, 0.4 * rem)
        
        par = None
        if parent_arr is not None and parent_arr.size == n:
            par = parent_arr
        
        try:
            cand, sc = optimize_length(n, b, seed_cur + variant * 99999, parent=par)
            if cand is not None and np.isfinite(sc) and sc < best_score:
                best_seq = cand.copy()
                best_score = sc
                print(f"IMPROVED: n={n} variant={variant} score={best_score:.6f}", flush=True)
            else:
                print(f"n={n} variant={variant} best_here={sc:.6f} (global={best_score:.6f})", flush=True)
        except Exception as e:
            print(f"n={n} variant={variant} error: {e}", flush=True)
        seed_cur += 50000
    
    # Final polish on best sequence
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 10:
            # Intensive optimization on the best
            n_best = best_seq.size
            polish_time = min(rem * 0.6, 120.0)
            try:
                c2, sc2 = optimize_single(best_seq, polish_time, seed_cur + 1, use_lp=(n_best <= 512))
                if c2 is not None and sc2 < best_score:
                    best_seq = c2.copy()
                    best_score = sc2
                    print(f"After polish: score={best_score:.6f}", flush=True)
            except Exception as e:
                print(f"Polish error: {e}", flush=True)
        
        # Try spectral refinement
        rem2 = deadline - time.monotonic()
        if rem2 > 5:
            for alpha in (0.4, 0.5, 0.6, 0.7, 0.8):
                a_spec = spectral_step(best_seq, alpha)
                sc_sp = score_of(a_spec)
                if sc_sp < best_score:
                    best_seq = a_spec.copy()
                    best_score = sc_sp
                    print(f"Spectral alpha={alpha}: score={best_score:.6f}", flush=True)
                # Blend
                for t in (0.2, 0.4, 0.6):
                    a_bl = simplex_finalize((1 - t) * best_seq + t * a_spec)
                    if a_bl is not None:
                        sc_bl = score_of(a_bl)
                        if sc_bl < best_score:
                            best_seq = a_bl.copy()
                            best_score = sc_bl
        
        # One more round of gradient optimization if time allows
        rem3 = deadline - time.monotonic()
        if rem3 > 15:
            try:
                c3, sc3 = optimize_single(best_seq, rem3 * 0.7, seed_cur + 2, use_lp=False)
                if c3 is not None and sc3 < best_score:
                    best_seq = c3.copy()
                    best_score = sc3
                    print(f"Final gradient: score={best_score:.6f}", flush=True)
            except Exception as e:
                print(f"Final gradient error: {e}", flush=True)
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(128, dtype=np.float64) / 128.0
        best_score = score_of(best_seq)
    
    # Final validation
    final = simplex_finalize(best_seq)
    if final is None:
        final = np.ones(128, dtype=np.float64) / 128.0
    
    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
        else:
            # Fallback to parent
            if parent_arr is not None:
                fp = simplex_finalize(parent_arr)
                if fp is not None:
                    try:
                        off2 = float(evaluate_sequence([float(x) for x in fp]))
                        if np.isfinite(off2):
                            final = fp
                            best_score = off2
                    except Exception:
                        pass
    except Exception:
        pass
    
    print(f"FINAL: score={best_score:.6f} n={len(final)}", flush=True)
    return [float(x) for x in np.clip(np.asarray(final, dtype=np.float64), 0.0, 1000.0)]
