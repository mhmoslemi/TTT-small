def soft_max_objective_and_grad(f, n, alpha):
    """Compute soft-max loss and gradient using FFT."""
    conv = np.convolve(f, f)
    max_c = np.max(conv)
    w = np.exp(alpha * (conv - max_c))
    Z = np.sum(w)
    loss = max_c * alpha + np.log(Z)
    # Gradient: (alpha/Z) * correlate(w, f, 'valid')
    # correlate(w, f, 'valid')[i] = sum_j w[j+i] * f[j]
    grad = (alpha / Z) * np.correlate(w, f, mode='valid')
    return loss, grad

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Bisection on tau
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n+1) > (css[-1] - 1))[0][-1]
    theta = (css[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def fista_optimize(f_init, n, time_budget, alpha=500.0):
    """Optimize using FISTA with soft-max objective."""
    f = proj_simplex(f_init.copy())
    y = f.copy()
    t = 1.0
    best_f = f.copy()
    best_score = score_normalized(f, n)
    
    # Lipschitz constant estimate

"""
Hybrid search for non-negative sequences minimizing 2n*max(autocorr)/(sum)^2.
Uses three complementary methods: multiplicative updates (fast), LP iterations
(precise), and FISTA with soft-max loss (global refinement).
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Normalize to sum 1, non-negative."""
    x = np.asarray(arr, dtype=np.float64).ravel()
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    return x / s


def score_of(f):
    """Score = 2n * max(conv(f,f)) assuming sum(f)=1."""
    n = len(f)
    conv = np.convolve(f, f)
    return float(2.0 * n * np.max(conv))


def proj_simplex(v):
    """Project onto probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (css[-1] - 1.0)
    rho_val = np.max(rho[cond]) if np.any(cond) else 0
    theta = (css[rho_val - 1] - 1.0) / rho_val if rho_val > 0 else 0.0
    return np.maximum(v - theta, 0.0)


def make_conv_template(n):
    """Build sparse template for convolve constraints in LP."""
    m = 2 * n - 1
    k = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    row = np.repeat(k, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)
    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_lp_direction(template, f, n, M):
    """Solve LP: max sum(g) s.t. A@f'*g <= M*(scale)^2, g >= 0."""
    A_template, edge_offsets, c_obj, bounds, m = template
    scale = float(n)
    A = A_template.copy()
    A.data[:] = f[edge_offsets] * scale
    b = np.full(m, M * scale * scale, dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")


def lp_optimize(f_init, n, time_budget, rng, template):
    """LP-based iterative optimization."""
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n) / n
    best_f = f.copy()
    best_score = score_of(f)
    deadline = time.monotonic() + time_budget
    stall = 0
    
    while time.monotonic() < deadline:
        t0 = time.monotonic()
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if not np.isfinite(M) or M <= 0:
            break
        
        try:
            res = solve_lp_direction(template, f, n, M)
        except Exception:
            break
        
        x = getattr(res, 'x', None)
        if x is None:
            stall += 1
            if stall > 15:
                break
            f = proj_simplex(f + 0.1 * (rng.random(n) - 0.5))
            continue
        
        x = np.maximum(np.asarray(x, dtype=np.float64), 0.0)
        s_star = float(x.sum())
        if s_star <= 1e-12:
            stall += 1
            if stall > 15:
                break
            f = proj_simplex(f + 0.1 * (rng.random(n) - 0.5))
            continue
        
        g = x / s_star
        r = np.convolve(f, g)
        r_max = float(np.max(r))
        
        if r_max >= M * (1 - 1e-10):
            stall += 1
            if stall > 15:
                break
            f = proj_simplex(f + 0.1 * (rng.random(n) - 0.5))
            continue
        
        G = float(np.max(np.convolve(g, g)))
        a_c = 2.0 * (r_max - M)
        b_c = G + M - 2.0 * r_max
        t_max = -a_c / b_c if b_c > 1e-14 else 1.0
        if not np.isfinite(t_max) or t_max <= 0:
            t_max = 1e-6
        t_max = min(t_max, 1.0)
        
        # Try multiple step sizes
        best_sc = best_score
        best_f_new = None
        for frac in [0.8, 0.6, 0.4, 0.2, 0.1, 0.05, 0.01]:
            t = frac * t_max
            f_new = proj_simplex((1 - t) * f + t * g)

            best_score = best_sc
            stall = 0

            f = proj_simplex(f + 0.1 * (rng.random(n) - 0.5))

def multiplicative_optimize(f_init, n, time_budget, beta=0.3, power_p=5.0):
    """Sinkhorn-like multiplicative update optimization."""
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n) / n
    best_f = f.copy()
    best_score = score_of(f)
    deadline = time.monotonic() + time_budget
    stall = 0
    
    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        max_c = float(np.max(conv))
        if max_c <= 0 or not np.isfinite(max_c):
            break
        
        # Soft selection of "active" constraints
        w = (conv / max_c) ** power_p
        w_sum = float(np.sum(w))
        if w_sum <= 0:
            break
        w = w / w_sum
        
        # Pressure at each position
        g = np.correlate(w, f, mode='valid')
        g = np.maximum(g, 1e-30)
        lam = float(np.mean(g))
        
        # Multiplicative update
        f_new = f * (lam / g) ** beta
        f_new = proj_simplex(f_new)
        
        sc = score_of(f_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            # Try smaller beta
            if stall == 5:
                beta *= 0.5
            if stall > 30:
                break
    
    return best_f, best_score
