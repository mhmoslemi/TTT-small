    p = np.exp(logits)

Uses adaptive softmax temperature for peak focusing and momentum for faster convergence.

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2 for non-negative a."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def score_normalized(a):
    """Score assuming sum(a)=1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * c.max()


def subgradient_optimize(a0, time_budget, rng, n_peaks=10, verbose=False):
    """
    Subgradient descent to minimize max(conv(a,a)) subject to sum(a)=1, a>=0.
    Uses softmax-weighted peak gradients with adaptive temperature and momentum.
    """
    n = len(a0)
    a = np.array(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a0.copy(), np.inf
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)

    # Parameters
    step_size = 0.01
    min_step = 1e-7
    max_step = 0.1
    beta = 0.7  # momentum
    momentum = np.zeros(n)
    temp_factor = 1.0

    start_time = time.monotonic()
    it = 0
    stall = 0
    best_it = 0

    while time.monotonic() - start_time < time_budget:
        c = np.convolve(a, a)
        c_max = c.max()
        c_mean = c.mean()
        spread = c_max - c_mean

        if spread < 1e-12:
            break

        # Adaptive temperature: focus more on peaks as we progress
        # Start broad, narrow down
        temp = temp_factor * spread
        # Softmax weights over peaks
        logits = (c - c_max) / temp
        # Clip to avoid overflow
        logits = np.clip(logits, -50, 0)
        p = np.exp(logits)
        p_sum = p.sum()
        if p_sum < 1e-15:
            break
        p = p / p_sum

        # Only use significant peaks for efficiency
        thresh = 1e-6
        active = p > thresh
        if not active.any():
            active = np.zeros_like(p, dtype=bool)
            active[np.argmax(p)] = True

        # Compute weighted subgradient
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            pk = p[k]
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * pk * a[k - idx]

        # Mean subtraction to preserve sum constraint
        grad_centered = grad - grad.mean()

        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * grad_centered
        a_new = a - step_size * momentum
        a_new = np.clip(a_new, 0, None)
        s_new = a_new.sum()
        if s_new < 1e-10:
            step_size = max(min_step, step_size * 0.5)
            momentum *= 0.5
            continue
        a_new = a_new / s_new

        new_score = score_normalized(a_new)

        if new_score < best_score:
            best_a = a_new.copy()
            best_score = new_score
            a = a_new
            stall = 0
            best_it = it
            # Slightly increase step on success
            step_size = min(max_step, step_size * 1.02)
            # Decrease temperature (focus more)
            temp_factor = max(0.1, temp_factor * 0.995)
        else:
            stall += 1
            step_size = max(min_step, step_size * 0.97)
            momentum *= 0.5
            if stall > 300:
                # Escape: perturb from best
                a = best_a * (1 + 0.05 * rng.normal(0, 1, n))
                a = np.clip(a, 0, None)
                if a.sum() > 1e-10:
                    a = a / a.sum()
                else:
                    a = best_a.copy()
                momentum = np.zeros(n)
                step_size = 0.01
                temp_factor = 1.0
                stall = 0

        it += 1
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"      sg it={it}, score={best_score:.6f}, step={step_size:.5f}, temp={temp_factor:.3f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_optimize(a0, time_budget, t0=0.4, verbose=False, rng=None):
    """LP-based direction method. a0 should have sum=1."""
    if rng is None:
        rng = np.random.default_rng(42)

    n = len(a0)
    a = np.array(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a0.copy(), np.inf
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t = t0

    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        c = np.convolve(a, a)
        M = c.max()

        A_mat = build_conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        g = b / b_sum
        a_new = (1 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        s_new = a_new.sum()
        if s_new < 1e-10:
            t *= 0.5
            continue
        a_new = a_new / s_new

        new_score = score_normalized(a_new)

        if new_score < best_score - 1e-13:
            best_a = a_new.copy()
            best_score = new_score
            a = a_new
            stall = 0
            # Can increase t
            t = min(0.8, t * 1.05)
            if verbose and it % 3 == 0:
                elapsed = time.monotonic() - start
                print(f"      lp it={it}, score={best_score:.6f}, t={t:.3f}, elapsed={elapsed:.1f}s")
        else:
            t *= 0.7
            stall += 1
            if stall > 80:
                a = best_a * (1 + 0.03 * rng.normal(0, 1, n))
                a = np.clip(a, 0, None)
                if a.sum() > 1e-10:
                    a = a / a.sum()
                    if score_normalized(a) < best_score:
                        best_a = a.copy()
                        best_score = score_normalized(a)
                t = max(0.1, t0 * 0.5)
                stall = 0
                if time.monotonic() - start > time_budget * 0.8:
                    break

        it += 1

    return best_a, best_score


def local_search(a0, rng, n_iter=5000):
    """Block perturbation local search for final polishing."""
    n = len(a0)
    a = np.array(a0, dtype=np.float64).copy()
    best = a.copy()
    best_score = score_normalized(a)

    for _ in range(n_iter):
        a_cand = a.copy()
        width = max(1, int(rng.integers(1, max(2, n // 16))))
        start = int(rng.integers(0, n - width))
        scale = np.exp(rng.normal(0, 0.15))
        a_cand[start:start + width] *= scale
        a_cand = np.clip(a_cand, 0, None)
        if a_cand.sum() < 1e-10:
            continue
        a_cand = a_cand / a_cand.sum()
        sc = score_normalized(a_cand)
        if sc < best_score:
            best = a_cand.copy()
            best_score = sc
            a = a_cand

    return best, best_score


def make_m_sequence(n):
    """Create a non-negative sequence inspired by m-sequences (LFSR)."""
    # Use a simple LFSR to generate a pseudo-binary sequence
    # Then map to non-negative values
    m = max(4, int(np.log2(n)) + 1)
    while (1 << m) - 1 < n:
        m += 1
    L = (1 << m) - 1

    # Simple LFSR with primitive polynomial (using numpy for portability)
    # Use tap positions for common m
    tap_map = {
        4: [3, 1], 5: [3, 1], 6: [5, 3], 7: [6, 5], 8: [4, 3, 2, 1],
        9: [5, 3], 10: [7, 5], 11: [9, 5], 12: [6, 4], 13: [12, 11],
        14: [13, 12], 15: [10, 7], 16: [14, 13, 12, 11], 17: [14, 13],
        18: [17, 14], 19: [18, 17], 20: [17, 14], 21: [20, 18],
        22: [21, 20], 23: [18, 15], 24: [16, 14], 25: [22, 21],
        26: [25, 24], 27: [21, 20], 28: [25, 24], 29: [28, 27],
        30: [29, 27],
    }
    if m > 30:
        m = 30
    L = (1 << m) - 1
    taps = tap_map.get(m, [m-1, m-2])

    # Generate LFSR sequence
    state = 1
    seq = []
    for _ in range(L):
        seq.append(1 if (state & 1) else 0)
        if m <= 30:
            bit = 0
            for tapp in taps:
                bit ^= (state >> tapp) & 1
            state = (state >> 1) | (bit << (m - 1))

    seq = np.array(seq[:n], dtype=np.float64)
    # Make non-negative with some spread
    a = 0.1 + 0.9 * seq
    a = a / a.sum()
    return a
