Let me now write the full implementation. I'll focus on making the mirror descent more effective:

Key improvements over the previous program:
1. **Softmax-smoothed active set** instead of hard threshold (avoids discontinuities in the subgradient)
2. **Geometric step-size decay** (fast start, slow finish)
3. **Nesterov momentum** on the multiplicative updates
4. **Adaptive temperature** (beta increases over time to focus on the true active set)
5. **Periodic restarts** with small perturbations
6. **Try n=512 and n=1024** (upsample the parent to 1024 by interpolation)

Let me also add a **cyclic coordinate descent** step: fix all but one coordinate and optimize that coordinate in closed form. This can help with fine-tuning.

For coordinate j, the objective is max_k (f*f)_k. The coordinate j appears in:
(f*f)_k = ... + f[j]*f[k-j] + ... (for the term where i=j)
(f*f)_k = ... + f[k-j]*f[j] + ... (for the term where k-i=j)

So changing f[j] by delta changes (f*f)_k by delta * f[k-j] (for each k where k-j is valid).

If we fix all other coordinates and want to minimize max_k (f*f)_k by adjusting f[j], this is a 1D problem. The optimal f[j] (subject to f[j] >= 0 and sum(f) = 1) can be found by bisection.

But this is complex to implement efficiently. Let me skip this and focus on the mirror desert improvements.

Let me write the final code:

```python
"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses smoothed mirror descent with geometric step-size decay, softmax-based
active set weighting, Nesterov momentum, and LP-based Frank-Wolfe refinement.
"""

import numpy as np
import time

def _fft_convolve_full(f):
    """Full linear convolution via FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score(f):
    """Score = 2n * max(conv(f,f)) / (sum(f))^2."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    """Clip to non-negative and normalize to sum=1."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _compute_grad_fft(f, w):
    """
    Compute grad[j] = 2 * sum_k w[k] * f[k-j] for j=0..n-1.
    Uses FFT cross-correlation.
    """
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    W = np.fft.rfft(w, size)
    F = np.fft.rfft(f, size)
    R = np.fft.irfft(W * np.conj(F), size)[:n]
    return 2.0 * R

def _smoothed_md(f_init, n_iter, beta_start, beta_end, eta_start, eta_end,
                 label="", verbose=True, time_limit=None):
    """
    Smoothed mirror descent with geometric schedules for beta and eta.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    t0 = time.monotonic()
    
    if verbose:
        print(f"  SM-D start ({label}): n={n}, score={best_score:.6f}")
    
    # Precompute size
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    # Geometric decay factors
    if n_iter > 1:
        beta_decay = (beta_end / beta_start) ** (1.0 / n_iter) if beta_start > 0 else 1.0
        eta_decay = (eta_end / eta_start) ** (1.0 / n_iter)
    else:
        beta_decay = 1.0
        eta_decay = 1.0
    
    beta = beta_start
    eta = eta_start
    
    for it in range(n_iter):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution
        F = np.fft.rfft(f, size)
        h = np.fft.irfft(F * F, size)[:2*n-1]
        h_max = float(np.max(h))
        
        # Softmax weights
        w = np.exp(beta * (h - h_max))
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.zeros(len(h))
            w[int(np.argmax(h))] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient
        W = np.fft.rfft(w, size)
        grad = 2.0 * np.fft.irfft(W * np.conj(F), size)[:n]
        
        # Update: f * exp(-eta * grad / scale)
        scale = float(np.max(np.abs(grad))) + 1e-30
        log_update = -eta * grad / scale
        log_update = np.clip(log_update, -30.0, 30.0)
        f = f * np.exp(log_update)
        f_norm = _normalize(f)
        if f_norm is None:
            f = np.ones(n) / n
        else:
            f = f_norm
        
        # Track best
        s = _score(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()
        
        # Update schedules
        beta *= beta_decay
        eta *= eta_decay
        
        if verbose and (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SM-D it={it+1}: score={best_score:.6f}, beta={beta:.1f}, eta={eta:.2f}, t={elapsed:.1f}s")
    
    if verbose:
        print(f"  SM-D done ({label}): best={best_score:.6f}, time={time.monotonic()-t0:.1f}s")
    
    return best_f, best_score

def _fista_logit(f_init, n_iter, step_start, step_end, label="", verbose=True, time_limit=None):
    """
    FISTA on the logit parameterization.
    f = softmax(g) where g are the free parameters.
    Objective: max_k (f*f)_k (with f = softmax(g), so sum(f)=1 automatically).
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    
    n = len(f)
    # Initialize g = log(f) (logit parameterization)
    g = np.log(np.maximum(f, 1e-30))
    g -= np.max(g)  # for numerical stability
    
    f_prev = _normalize(np.exp(g)).copy()
    f_curr = f_prev.copy()
    g_curr = g.copy()
    
    best_f = f_curr.copy()
    best_score = _score(f_curr)
    t0 = time.monotonic()
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    if verbose:
        print(f"  FISTA start ({label}): n={n}, score={best_score:.6f}")
    
    decay = (step_end / step_start) ** (1.0 / max(n_iter, 1))
    step = step_start
    t_momentum = 1.0  # Nesterov momentum parameter
    
    for it in range(n_iter):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum)) / 2.0
        theta = (t_momentum - 1.0) / t_new
        g_y = g_curr + theta * (g_curr - g_prev if it > 0 else np.zeros(n))
        
        # Compute f from g_y
        f_y = np.exp(g_y - np.max(g_y))
        f_y /= np.sum(f_y)
        
        # Compute gradient of max_k (f*f)_k w.r.t. f
        F_y = np.fft.rfft(f_y, size)
        h = np.fft.irfft(F_y * F_y, size)[:2*n-1]
        h_max = float(np.max(h))
        active = int(np.argmax(h))
        
        # Gradient w.r.t. f at the active lag
        # grad_f[j] = 2 * f_y[active - j] (if 0 <= active-j < n)
        grad_f = np.zeros(n)
        j_min = max(0, active - n + 1)
        j_max = min(n - 1, active)
        if j_max >= j_min:
            f_indices = active - np.arange(j_min, j_max + 1)
            grad_f[j_min:j_max+1] = 2.0 * f_y[f_indices]
        
        # Gradient w.r.t. g (chain rule through softmax)
        # d(f_j)/d(g_k) = f_j * (delta_jk - f_k)
        # grad_g[j] = grad_f[j] * f_y[j] - f_y[j] * sum_k grad_f[k] * f_y[k]
        grad_g = grad_f * f_y - f_y * np.dot(grad_f, f_y)
        
        # Gradient step on g
        g_new = g_y - step * grad_g
        
        # No projection needed (g is unconstrained)
        g_prev = g_curr.copy()
        g_curr = g_new
        f_curr = np.exp(g_new - np.max(g_new))
        f_curr /= np.sum(f_curr)
        
        s = _score(f_curr)
        if s < best_score:
            best_score = s
            best_f = f_curr.copy()
        
        t_momentum = t_new
        step *= decay
        
        if verbose and (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA it={it+1}: score={best_score:.6f}, step={step:.4f}, t={elapsed:.1f}s")
    
    if verbose:
        print(f"  FISTA done ({label}): best={best_score:.6f}, time={time.monotonic()-t0:.1f}s")
    
    return best_f, best_score

def _lp_frank_wolfe(f_init, max_iters, time_budget):
    """LP-based Frank-Wolfe refinement."""
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix
    
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start_time = time.monotonic()
    
    if n > 2000:
        return best_f, best_score  # LP too slow for large n
    
    print(f"    LP start: n={n}, score={best_score:.6f}")
    
    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break
        
        h = np.convolve(f, f)
        M = float(np.max(h))
        
        # Build constraint matrix
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            rows_j = np.arange(j, j + n)
            row_idx.extend(rows_j.tolist())
            col_idx.extend([j] * n)
            vals.extend(f.tolist())
        
        C = coo_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))
        
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=C, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)]*n, method='highs')
        
        if not res.success or res.x is None:
            break
        
        g = res.x
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg
        
        t = 2.0 / (it + 2)
        f_new = (1.0 - t) * f + t * g
        f_n = _normalize(f_new)
        if f_n is None:
            break
        f = f_n
        
        score = _score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if (it + 1) % 5 == 0:
            print(f"    LP it={it+1}: score={best_score:.6f}")
    
    return best_f, best_score

def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _upsample(f, new_n):
    """Upsample f to new_n using linear interpolation."""
    old_n = len(f)
    if new_n == old_n:
        return f.copy()
    old_x = np.arange(old_n)
    new_x = np.linspace(0, old_n - 1, new_n)
    return np.interp(new_x, old_x, f)
