The subgradient of M(u) at u is: `delta[k*] + 2*u*d_{k*}` where k* = argmax_k [base_conv[k] + u*delta[k] + u^2*d_k].

Now, let me also think about the LP approach improvements:

The LP direction method works as follows:
1. Given f (sum=1), compute M = max(f*f).
2. Solve: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0.
3. The optimal b, rescaled to sum=1, gives a direction g.
4. Update f ← (1-t)*f + t*g.

The key insight is that the LP finds a direction that "spreads out" the convolution while keeping it bounded by M.

Improvements:
1. **Warm-start the LP** with the previous solution.
2. **Use multiple M values**: solve the LP for slightly different M values and combine.
3. **Adaptive step size**: use backtracking line search.
4. **Polish with coordinate descent** after each LP iteration.

Let me also consider a completely different approach: **Frank-Wolfe with away steps**.

The away-step Frank-Wolfe method allows moving away from vertices that are "bad." This can be much more effective than the standard FW.

Algorithm:
1. Start with f (a point on the simplex).
2. Compute subgradient s = ∇ max_k (f*f)(k).
3. Linear minimization: v = e_{argmin s[i]} (the vertex with smallest subgradient component).
4. Away step: check if moving away from a current vertex w helps. Find w in support(f) that maximizes s[w] (the "worst" vertex).
5. Take a step toward v and/or away from w.

def coordinate_decent(f, n_sweeps=10, n_grid=300, rng=None):
    """Perform coordinate descent with exact 1D minimization."""
    n = len(f)
    f = f.copy().astype(np.float64)
    S = f.sum()
    if S < 1e-15:
        return f, np.inf
    f = f / S  # normalize
    
    best_score = compute_score_normalized(f)
    
    for sweep in range(n_sweeps):
        improved = False
        # Cycle through coordinates (randomize order)
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        for j in order:
            f_j = f[j]
            S_other = S - f_j
            if S_other < 0:
                S_other = 0
            
            # Precompute delta
            delta = np.zeros(2 * n - 1)
            for k in range(max(0, j), min(2*n-1, j+n)):
                kj = k - j
                if 0 <= kj < n:
                    delta[k] = 2.0 * f[kj]
            
            quad_idx = 2 * j
            has_quad = quad_idx < 2 * n - 1
            
            base_conv = np.convolve(f, f)
            
            # Minimize over x (new value of f[j])
            # h(x) = 2*n*max_k[base_conv[k] + (x-f_j)*delta[k] + (x-f_j)^2*(1 if k==quad_idx)] / (S_other+x)^2
            
            # Use scipy minimize_scalar for 1D optimization
            def obj(x):
                u = x - f_j
                new_max = (base_conv + u * delta).max()
                if has_quad:
                    val_at_quad = base_conv[quad_idx] + u * delta[quad_idx] + u * u
                    new_max = max(new_max, val_at_quad)
                denom = S_other + x
                if denom <= 0:
                    return np.inf
                return new_max / (denom * denom)
            
            # Search range: x in [0, S] (can't be more than total sum meaningfully)
            res = minimize_scalar(obj, bounds=(0, S), method='bounded', 
                                  options={'xatol': 1e-10})
            if res.success and res.fun < obj(f_j) * (1 - 1e-12):
                f[j] = res.x
                S += f[j] - f_j
                new_score = compute_score_normalized(f)
                if new_score < best_score:
                    best_score = new_score
                    improved = True
        
        if not improved:
            break
    
    return f, best_score

    f = f.copy().astype(np.float64)
    S = f.sum()
    if S < 1e-15:
        return f, np.inf
    f = f / S
    
    conv = np.convolve(f, f).copy()
    best_score = 2.0 * n * conv.max()
    
    start_time = time.monotonic()
    
    for sweep in range(n_sweeps):
        if time_limit and time.monotonic() - start_time > time_limit:
            break
        
        order = rng.permutation(n) if rng else np.arange(n)
        improved = False

            # quad at k = 2j

def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_direction_step(f, M, n, verbose=False):
    """Solve LP to find direction g that spreads convolution."""
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 10})
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    b = np.clip(res.x, 0, None)
    b_sum = b.sum()
    if b_sum < 1e-15:
        return None
    return b / b_sum

def random_sequence(n, rng, style='uniform'):
    """Generate a random starting sequence."""
    if style == 'uniform':
        f = np.ones(n) * (1.0 + 0.1 * rng.standard_normal(n))
    elif style == 'bimodal':
        center = n // 2
        width = n // 4
        f = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        f += 0.5 * np.exp(-0.5 * ((np.arange(n) - n // 4) / (width // 2)) ** 2)
    elif style == 'periodic':
        period = max(2, n // 8)
        f = 0.5 + 0.3 * np.sin(2 * np.pi * np.arange(n) / period)
        f += 0.1 * rng.standard_normal(n)
    elif style == 'ramp':
        f = np.linspace(0.5, 1.5, n)
    elif style == 'noise':
        f = np.abs(rng.standard_normal(n)) + 0.1
    else:
        f = np.ones(n)

    f = np.clip(f, 0, None)
    s = f.sum()
    if s > 0:
        f = f / s
    return f


def sim_anneal_step(f, T, rng):
    """One simulated annealing step."""
    n = len(f)
    f_new = f.copy()
    # Random perturbation
    n_perturb = max(1, n // 20)
    idx = rng.integers(0, n, size=n_perturb)
    scale = np.exp(rng.normal(0, 0.5, size=n_perturb))
    f_new[idx] *= scale
    f_new = np.clip(f_new, 0, None)
    s = f_new.sum()
    if s < 1e-10:
        return f, compute_score_normalized(f)
    f_new = f_new / s
    return f_new, compute_score_normalized(f_new)
