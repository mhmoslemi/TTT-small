"""
Genetic-algorithm-inspired search combined with Frank-Wolfe and LP refinement
for minimizing 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for sequence a (assumed sum=1 for speed)."""
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()


def fw_step(a, t=0.0):
    """One Frank-Wolfe iteration. Returns updated a and score."""
    n = len(a)
    conv = np.convolve(a, a)
    k_star = np.argmax(conv)
    
    # Gradient of c[k_star] with respect to a
    # dc[k]/da[j] = 2*a[k-j] if 0 <= k-j < n, else 0
    offset = np.arange(n) - k_star
    valid = (offset >= 0) & (offset < n)
    g = np.zeros(n)
    g[valid] = 2.0 * a[offset[valid]]
    
    # Linear minimization: min g[j] subject to a_i >= 0, sum = 1
    # Solution: s = e_{argmin g}
    j_min = np.argmin(g)
    
    # Line search (a few evaluations)
    best_t = t
    best_score = compute_score_normalized(a)
    for t_candidate in [0.5, 0.3, 0.1, 0.05, 0.02, 0.01]:
        a_new = (1 - t_candidate) * a + t_candidate * np.zeros(n)
        a_new[j_min] = t_candidate
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new /= s
        sc = compute_score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_t = t_candidate
    
    a_new = (1 - best_t) * a + best_t * np.zeros(n)
    a_new[j_min] = best_t
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new, best_score
```

def fw_update_conv(conv, a, t, j, n):
    """Update convolution after FW step. conv is modified in place."""
    alpha = 1 - t
    alpha2 = alpha * alpha
    alpha_beta_2 = 2 * alpha * t
    
    # Scale entire conv by alpha^2
    conv *= alpha2
    
    # Add 2*alpha*t*a[k-j] for k in [j, j+n-1]
    k_start = j
    k_end = min(j + n, 2*n - 1)
    for k in range(k_start, k_end):
        conv[k] += alpha_beta_2 * a[k - j]
    
    # Add t^2 at position 2j
    if 2*j < 2*n - 1:
        conv[2*j] += t * t
    
    return conv

def fw_line_search(a, conv, j, n, max_evals=10):
    """Find best step size t for FW step toward vertex j."""
    # Evaluate at t = 1, 0.5, 0.25, 0.1, 0.05, 0.01
    best_t = 0.0
    best_score = np.inf
    
    for t in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
        # Compute conv at this t
        alpha = 1 - t
        conv_trial = alpha * alpha * conv.copy()
        k_end = min(j + n, 2*n - 1)
        conv_trial[j:k_end] += 2 * alpha * t * a[:k_end - j]
        if 2*j < 2*n - 1:
            conv_trial[2*j] += t * t
        score = conv_trial.max()
        if score < best_score:
            best_score = score
            best_t = t
    
    return best_t
```

"""
Frank-Wolfe + LP refinement for minimizing 2n*max(convolve(a,a))/sum(a)^2.
Uses incremental convolution updates for O(n) FW iterations, combined with
periodic LP refinement and local search polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.sparse import diags
import time

def fast_convolve(a, b=None):
    """Compute a*a (or a*b) convolution using direct or FFT method."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    if n * m < 2 * (n + m) * (n + m):
        return np.convolve(a, b)
    else:
        size = 1
        while size < n + m - 1:
            size *= 2
        fa = np.fft.rfft(a, size)
        fb = np.fft.rfft(b, size)
        return np.fft.irfft(fa * fb)[:n + m - 1]

def compute_score(a):
    """Compute 2n*max(conv)/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

    """Score assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()
