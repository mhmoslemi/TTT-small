import numpy as np
import time
from scipy.fft import fft, ifft, next_fast_len

def fast_evaluate(f):
    """Faster evaluation without the security checks."""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    if n == 0:
        return float('inf')
    c = np.convolve(f, f)
    max_c = c.max()
    sum_f = f.sum()
    if sum_f < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (sum_f ** 2)

def gs_direction(f, n_iter=500, relax=0.5, deadline=None):
    """Gerchberg-Saxton to flatten autocorrelation."""
    n = len(f)
    L = next_fast_len(2 * n - 1)
    f = f.astype(np.float64).copy()
    S = f.sum()
    best_f = f.copy()
    best_score = fast_evaluate(f)
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        f_pad = np.zeros(L)
        f_pad[:n] = f
        F = fft(f_pad)
        
        # Flatten magnitude
        F_new = np.ones(L, dtype=np.complex128)
        F_new *= np.exp(1j * np.angle(F))
        
        f_new = np.real(ifft(F_new))[:n]
        f_new = np.maximum(0, f_new)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new *= S / s_new
        
        f = (1 - relax) * f + relax * f_new
        
        score = fast_evaluate(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score

def gradient_descent(f, deadline, lr_init=0.01, max_iter=10000):
    """Subgradient descent on max autocorrelation."""
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    best_f = f.copy()
    best_score = fast_evaluate(f)
    lr = lr_init
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(f, f)
        k_star = int(np.argmax(c))
        
        # Subgradient of max c[k] at f
        grad = np.zeros(n)
        for j in range(n):
            if 0 <= k_star - j < n:
                grad[j] = 2.0 * f[k_star - j]
        
        # Move in negative gradient direction, project to simplex-like constraint
        f_new = f - lr * grad
        f_new = np.maximum(0, f_new)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new *= S / s_new
        
        score = fast_evaluate(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.9
            if no_improve > 50 or lr < 1e-8:
                break
    
    return best_f, best_score

"""
Hybrid GS + gradient optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses Gerchberg-Saxton spectral iterations for fast initial progress,
then adaptive subgradient descent for refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft, next_fast_len

def fast_conv(f):
    """Compute linear convolution of f with itself using FFT."""
    n = len(f)
    L = next_fast_len(2 * n - 1)
    f_pad = np.zeros(L)
    f_pad[:n] = f
    F = fft(f_pad)
    c = np.real(ifft(F * F))
    return c[:2*n-1]

def fast_score(f):
    """Fast score computation."""
    n = len(f)
    c = fast_conv(f)
    max_c = c.max()
    S = f.sum()
    if S < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (S * S)

def gs_iterate(f, n_iter, relax, deadline):
    """Gerchberg-Saxton iterations to flatten the spectrum."""
    n = len(f)
    L = next_fast_len(2 * n - 1)
    f = f.astype(np.float64).copy()
    S = f.sum()
    best_f = f.copy()
    best_score = fast_score(f)
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        f_pad = np.zeros(L)
        f_pad[:n] = f
        F = fft(f_pad)
        
        # Set magnitude to 1, keep phase
        F_new = np.exp(1j * np.angle(F))
        
        f_new = np.real(ifft(F_new))[:n]
        f_new = np.maximum(0, f_new)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new *= S / s_new
        
        f = (1 - relax) * f + relax * f_new
        
        score = fast_score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score

def subgradient_descent(f, deadline, lr_init=0.05, max_iter=50000):
    """Adaptive subgradient descent."""
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    best_f = f.copy()
    best_score = fast_score(f)
    lr = lr_init
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(f)
        k_star = int(np.argmax(c))
        
        # Subgradient: dc[k*]/df[j] = 2*f[k*-j] if 0 <= k*-j < n
        grad = np.zeros(n)
        j_min = max(0, k_star - (n - 1))
        j_max = min(k_star, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] = 2.0 * f[k_star - j]
        
        f_new = f - lr * grad
        f_new = np.maximum(0, f_new)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new *= S / s_new
        
        score = fast_score(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 100 or lr < 1e-10:
                break
    
    return best_f, best_score

def propose_candidate(budget_s=1500, **kwargs):

    rng = np.random.default_rng(123)
    
    # Lengths to try
    lengths_to_try = [256, 512, 1024]
    if best_seq is not None:

    for n in lengths_to_try:
        if time.monotonic() > deadline - 30:

        print(f"\n=== n={n} ===")
        
        # Generate initializations

        # Uniform

Wait, I think the LP approach works as follows:
- Current f has peak M = max(conv(f,f))
- We want to find a new f' with the same or lower peak but higher sum
- The LP finds g that maximizes sum(g) subject to conv(f, g) <= M (element-wise) and g >= 0
- Then we blend: f' = (1-t)f + t*g (rescaled to maintain sum)

The insight is: if conv(f, g) <= M and conv(f, f) <= M, then for small t:
conv(f', f') = (1-t)^2 conv(f,f) + t(1-t)(conv(f,g) + conv(g,f)) + t^2 conv(g,g)
<= (1-t)^2 M + t(1-t)(M + M) + t^2 conv(g,g)
= M - 2t(1-t)M + t^2 conv(g,g) + t(1-t)M
= M - t(1-t)M + t^2 conv(g,g)

For this to be <= M, we need t^2 conv(g,g) <= t(1-t)M, i.e., t * max(conv(g,g)) <= (1-t) * M.

Now let me reconsider the GS approach. The issue I identified was that making |F| constant leads to a peaked autocorrelation (delta at center). But that's the WRONG direction.

What we actually want is: make the convolution as flat as possible, i.e., all c[k] ≈ S^2/(2n-1).

In the frequency domain: c = IFFT(|F|^2) (circular version). We want IFFT(|F|^2) to be flat. This means |F|^2 should be a delta (concentrated at one frequency). But |F|^2 is non-negative, so the "flattest" IFFT we can get is when |F|^2 is as concentrated as possible.

But |F|^2 = |FFT(f_pad)|^2. The DC component |F[0]|^2 = S^2. The total energy sum(|F[k]|^2) = L * sum(f[j]^2) (by Parseval). So the "spread" of |F|^2 is determined by sum(f^2).

To make IFFT(|F|^2) flat, we want |F|^2 to be as concentrated as possible at one frequency. The most concentrated is |F|[0] = S and |F|[k] = 0 for k ≠ 0, which gives IFFT = (S^2/L) * ones(L), i.e., a flat autocorrelation! But this requires f to be a constant (all elements equal to S/L), which gives the uniform sequence with score 2.

Wait, that's circular. If f is uniform (all = S/n), then f_pad = [S/n, ..., S/n, 0, ..., 0] (n non-zero, L-n zeros). F[0] = S, F[k] = (S/n) * D(k) where D is the Dirichlet kernel. So |F|^2 is NOT a delta - it has the Dirichlet kernel shape.

The issue is that f_pad is NOT periodic with period L in a way that makes F a delta. The zeros in f_pad create sidelobes in F.

So the "flat autocorrelation" ideal is not achievable with a finite-length non-negative sequence padded with zeros. The best we can do is minimize the sidelobes.

This is exactly the problem of designing a window with minimal autocorrelation sidelobes! The optimal windows for this purpose are:
- Dolph-Chebyshev windows (minimax sidelobes)
- Kaiser windows (approximate Chebyshev)
- Optimal windows from convex optimization

So the answer is: use a Dolph-Chebyshev window (or similar) as the sequence f!

Let me verify: for a Dolph-Chebyshev window of length n, the autocorrelation sidelobes are equi-ripple with a specified level. The peak sidelobe level can be made small by increasing the number of taps or choosing a lower sidelobe level.
