def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0

The 1D problem is: minimize max(max_unaffected, max_{k in affected} f_k(x)) subject to x >= 0.

If there's no quadratic term (k=2i is not in the affected range, which happens when 2i >= 2n-1, i.e., i >= n - 0.5, so for i <= n-1, 2i <= 2n-2, so k=2i is always in range... wait, 2i can be up to 2(n-1) = 2n-2, which is the last index. So k=2i is always in the affected range for 0 <= i < n.

OK so there's always a quadratic term. The function g(x) = max(C, x^2 + D, max of linear terms) where C = unaffected_max, D = const_{2i}.

This is a convex function. The minimum occurs where:
- Either g is minimized at x = 0 (if the function is decreasing at 0)
- Or at the point where two or more of the "activating" functions cross

For a practical approach, let me use bisection on the level set: find the smallest M such that there exists x >= 0 with all f_k(x) <= M. This is equivalent to:
- x^2 + D <= M, so x <= sqrt(M - D) (if M >= D)
- coef_k * x + const_k <= M, so x <= (M - const_k) / coef_k (if coef_k > 0)
- x >= 0
- C <= M

So the feasible x range is [0, min(sqrt(M-D), min_k (M-const_k)/coef_k)] and we need this to be non-empty, i.e., min(sqrt(M-D), min_k (M-const_k)/coef_k) >= 0.

The minimum M is the smallest value such that max(0, D, max_k const_k) <= M and sqrt(M-D) >= 0 and (M - const_k)/coef_k >= 0 for all k with coef_k > 0.

def fast_convolve(a):
    """Fast autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    np.maximum(result, 0, out=result)
    return result

def compute_score(a):
    """Compute the evaluation score without calling evaluate_sequence."""
    n = len(a)
    c = fast_convolve(a)
    max_b = np.max(c)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def alt_projection_init(n, rng, n_iter=100, flatness=0.3):
    """Alternating projection initialization with soft flattening."""
    N = 1 << (2 * n - 2).bit_length()
    a = rng.random(n)
    a /= a.sum()
    
    for it in range(n_iter):
        # FFT
        F = np.fft.rfft(a, N)
        mag = np.abs(F)
        # Soft flatten the magnitude spectrum
        mag_flat = mag ** flatness  # Compress the dynamic range
        # Renormalize DC
        F_new = mag_flat * np.exp(1j * np.angle(F))
        F_new[0] = F_new[0].real
        # IFFT
        a_new = np.fft.irfft(F_new, N)[:n]
        # Project to non-negative, sum=1
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a = rng.random(n)
            a /= a.sum()
        else:
            a = a_new / s
    
    return a

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines efficient vectorized coordinate descent, frequency-domain projection,
parameterized initializations, and simulated annealing with periodic refinement.
"""

import numpy as np
import time


def fast_convolve(a):
    """Fast linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    np.maximum(result, 0, out=result)
    return result


def compute_score(a):
    """Compute evaluation score (scale-invariant)."""
    n = len(a)
    c = fast_convolve(a)
    max_b = np.max(c)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def alt_projection_init(n, rng, n_iter=100, flatness=0.3):
    """Alternating projection: non-negativity + frequency flattening."""
    N = 1 << (2 * n - 2).bit_length()
    a = rng.random(n)
    a /= a.sum()
    for _ in range(n_iter):
        F = np.fft.rfft(a, N)
        mag = np.abs(F)
        mag_new = mag ** flatness
        F_new = mag_new * (np.angle(F))
        F_new = mag_new * np.exp(1j * np.angle(F))
        F_new[0] = float(F_new[0].real)
        a_new = np.fft.irfft(F_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a = rng.random(n)
            a /= a.sum()
        else:
            a = a_new / s
    return a


def freq_projection_step(a, n_iter=20, flatness=0.5):
    """Apply frequency-domain soft flattening."""
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    for _ in range(n_iter):
        F = np.fft.rfft(a, N)
        mag = np.abs(F)
        mag_new = mag ** flatness
        F_new = mag_new * np.exp(1j * np.angle(F))
        F_new[0] = float(F_new[0].real)
        a = np.fft.irfft(F_new, N)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            break
    return a


def coordinate_descent_refine(a, deadline, max_sweeps=50, seed=0):
    """Vectorized coordinate descent minimizing max autocorrelation peak."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = fast_convolve(a)
    best_a = a.copy()
    best_max = float(np.max(c))
    rng = np.random.default_rng(seed)

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break

        order = rng.permutation(n)

        for i in order:
            if time.time() > deadline:
                break

            old_val = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            ks = np.arange(k_lo, k_hi + 1)
            js = ks - i

            quad_mask = (ks == 2 * i)
            lin_mask = ~quad_mask

            lin_coefs = np.zeros(len(ks))
            lin_consts = np.zeros(len(ks))
            quad_const = 0.0

            if np.any(lin_mask):
                lin_coefs[lin_mask] = 2.0 * a[js[lin_mask]]
                lin_consts[lin_mask] = c[ks[lin_mask]] - 2.0 * old_val * a[js[lin_mask]]
            if np.any(quad_mask):
                quad_const = c[ks[quad_mask][0]] - old_val * old_val

            unaffected_max = 0.0
            if k_lo > 0:
                unaffected_max = max(unaffected_max, float(np.max(c[:k_lo])))
            if k_hi < 2 * n - 2:
                unaffected_max = max(unaffected_max, float(np.max(c[k_hi + 1:])))

            upper = max(old_val * 3.0, 8.0 / n)
            lo = 0.0
            hi = upper

            for _ in range(22):
                x1 = hi - gr * (hi - lo)
                x2 = lo + gr * (hi - lo)

                g1 = unaffected_max
                g2 = unaffected_max

                if np.any(lin_mask):
                    m1 = np.max(lin_coefs[lin_mask] * x1 + lin_consts[lin_mask])
                    m2 = np.max(lin_coefs[lin_mask] * x2 + lin_consts[lin_mask])
                    if m1 > g1:
                        g1 = m1
                    if m2 > g2:
                        g2 = m2

                if np.any(quad_mask):
                    v1 = x1 * x1 + quad_const
                    v2 = x2 * x2 + quad_const
                    if v1 > g1:
                        g1 = v1
                    if v2 > g2:
                        g2 = v2

                if g1 < g2:
                    hi = x2
                else:
                    lo = x1

            new_val = (lo + hi) * 0.5

            if abs(new_val - old_val) > 1e-14:
                a[i] = new_val
                delta = new_val - old_val
                if np.any(lin_mask):
                    c[ks[lin_mask]] += 2.0 * delta * a[js[lin_mask]]
                if np.any(quad_mask):
                    c[ks[quad_mask]] = new_val * new_val + quad_const
                np.maximum(c, 0, out=c)

                new_max = float(np.max(c))
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()

    s = best_a.sum()
    if s > 1e-10:
        best_a = best_a / s
    return best_a


def param_init(n, rng, style=0):
    """Generate parameterized initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        p = rng.uniform(0.5, 3.0)
        a = np.abs(1.0 - np.abs(2.0 * x - 1.0)) ** p
    elif style == 1:
        mu = rng.uniform(0.2, 0.8)
        sig = rng.uniform(0.1, 0.4)
        a = np.exp(-0.5 * ((x - mu) / sig) ** 2)
    elif style == 2:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x)) + 0.005
    elif style == 3:
        p = rng.uniform(1.0, 5.0)
        a = (1.0 - x) ** p + 0.005
    else:
        a = rng.random(n) ** rng.uniform(0.3, 2.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def anneal_and_refine(a, deadline, rng, base_temp=0.04, cd_every=40, cd_sweeps=5):
    """Simulated annealing with periodic coordinate descent refinement."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    current_score = compute_score(a)
    best_a = a.copy()
    best_score = current_score
    counter = 0

    while time.time() < deadline:
        counter += 1
        temp = base_temp * (0.9995 ** counter)
        if temp < 0.001:
            temp = 0.001

        a_pert = a + rng.normal(0, temp / np.sqrt(n), n)
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s

        pert_score = compute_score(a_pert)

        if pert_score <= current_score:
            a = a_pert
            current_score = pert_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        else:
            accept_prob = np.exp(-(pert_score - current_score) / (temp * 2.0))
            if rng.random() < accept_prob:
                a = a_pert
                current_score = pert_score

        if counter % cd_every == 0 and time.time() < deadline - 3:
            a = coordinate_descent_refine(a, time.time() + 10, max_sweeps=cd_sweeps,
                                         seed=counter)
            current_score = compute_score(a)
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()

    return best_a
