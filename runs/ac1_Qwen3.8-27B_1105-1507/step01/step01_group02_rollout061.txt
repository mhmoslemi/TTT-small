def fast_convolve(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    result[result < 0] = 0.0
    return result


def get_smooth_grad(a, tau):
    """
    Compute gradient of tau*logsumexp(conv(a,a)/tau) and the convolution.
    Returns (grad, conv, peak).
    """
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    
    fa = np.fft.rfft(a, N)
    
    # Convolution
    c = np.fft.irfft(fa * fa, N)[:size]
    c[c < 0] = 0.0
    peak = c.max()
    
    # Softmax weights
    c_shifted = (c - peak) / tau
    c_shifted = np.maximum(c_shifted, -50)  # prevent underflow
    p = np.exp(c_shifted)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.zeros(size)
        p[np.argmax(c)] = 1.0
    else:
        p /= p_sum
    
    # Gradient via cross-correlation
    s = np.zeros(N)
    s[:size] = p
    s_ref = np.zeros(N)
    s_ref[:size] = s[size - 1::-1]
    
    fsr = np.fft.rfft(s_ref, N)
    conv_result = np.fft.irfft(fa * fsr, N)
    grad = 2.0 * conv_result[2 * n - 2 : n - 2 : -1]
    
    return grad, c, peak


def normalize_nonneg(a):
    """Project to non-negative simplex (sum=1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def alt_proj_init(n, rng, n_iter=40):
    """Alternating projection for flat-spectrum initialization."""
    N = 1 << (2 * n - 2).bit_length()
    F = np.ones(N // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    phases[0] = 0
    F = np.exp(1j * phases)
    
    a = np.ones(n) / n
    for _ in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n], 0.0)
        a = normalize_nonneg(a)
        F = np.fft.rfft(a, N)
        mag = np.maximum(np.abs(F), 1e-10)
        # Target: flat magnitude with slight low-freq boost
        mean_mag = np.mean(mag[1:]) if len(mag) > 1 else mag[0]
        target = np.ones(len(mag)) * mean_mag
        # Slight boost at low freq
        target[:len(target)//4] *= 1.2
        F = (target / mag) * F
        F[0] = target[0]
    
    return normalize_nonneg(a)


def structured_init(n, rng, method='cosine'):
    """Structured initialization."""
    if method == 'cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.05
    elif method == 'edges':
        x = np.linspace(0, 1, n)
        a = x**2 + (1-x)**2 + 0.05
    elif method == 'uniform':
        a = np.ones(n)
    elif method == 'random':
        a = rng.random(n) + 0.01
    elif method == 'bimodal':
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.15) / 0.1)**2) + np.exp(-((x - 0.85) / 0.1)**2) + 0.01
    else:
        a = rng.random(n) + 0.01
    return normalize_nonneg(a)


def lp_polish(a, max_time=8.0):
    """LP-based polishing step."""
    n = len(a)
    if n > 1200:
        return None
    
    c = fast_convolve(a)
    P = c.max()
    sum_a = a.sum()
    
    # Build constraint matrix (sparse would be better but dense is fine for n <= 1200)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, P),
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': max_time})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0)
            if b.sum() > 1e-8:
                return b * (sum_a / b.sum())
    except Exception:
        pass
    return None


def optimize_smooth(a, deadline, rng, use_lp=True):
    """
    Main optimization: smooth objective with continuation + mirror descent.
    """
    n = len(a)
    a = normalize_nonneg(a.copy())
    
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    best_score = 2.0 * n * best_peak / (a.sum() ** 2)
    
    # Continue method: start with large tau, decrease
    c0 = fast_convolve(a)
    peak0 = c0.max()
    
    # Sequence of tau values (as fraction of current peak)
    tau_fracs = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    total_time = deadline - time.time()
    
    for frac in tau_fracs:
        if time.time() > deadline - 20:
            break
        
        tau = frac * peak0
        if tau < 1e-8:
            tau = 1e-8
        
        # Adaptive eta
        eta = 1.0
        eta_min = 0.001
        stall = 0
        max_stall = 200
        
        local_deadline = min(deadline - 20, time.time() + total_time * max(0.05, 0.4 / (len(tau_fracs) - 1 + tau_fracs.index(frac))))
        # Actually, let's just use remaining time proportionally
        remaining = tau_fracs[len(tau_fracs) - 1 - tau_fracs.index(frac):]
        # Simpler: allocate equal time to each tau level
        time_per_level = (deadline - 20 - time.time()) / max(1, len(tau_fracs) - tau_fracs.index(frac))
        local_deadline = min(deadline - 20, time.time() + time_per_level)
        
        while time.time() < local_deadline:
            grad, c, peak = get_smooth_grad(a, tau)
            
            if peak < best_peak * (1 - 1e-10):
                best_peak = peak
                best_a = a.copy()
                score = 2.0 * n * peak / (a.sum() ** 2)
                best_score = score
                stall = 0
            else:
                stall += 1
            
            # Mirror descent step
            grad_norm = np.max(np.abs(grad))
            if grad_norm < 1e-15:
                break
            # Scale eta based on gradient magnitude
            eta_effective = eta / (1.0 + eta * grad_norm)
            a_new = a * np.exp(-eta_effective * grad)
            a_new = np.maximum(a_new, 1e-30)
            a = normalize_nonneg(a_new)
            
            # Adaptive eta
            if stall > max_stall:
                eta = max(eta * 0.8, eta_min)
                stall = 0
                if eta < eta_min * 5:
                    # Perturb
                    if time.time() < local_deadline - 5:
                        a = normalize_nonneg(a + rng.normal(0, 0.05 / np.sqrt(n), n))
        
        # LP polish at the end of each tau level
        if use_lp and n <= 1024 and time.time() < deadline - 30:
            b = lp_polish(a, max_time=5.0)
            if b is not None:
                t = 0.5
                a = normalize_nonneg((1 - t) * a + t * b)
                c = fast_convolve(a)
                peak = c.max()
                if peak < best_peak:
                    best_peak = peak
                    best_a = a.copy()
                    best_score = 2.0 * n * peak / (a.sum() ** 2)
    
    return best_a, best_score

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses smooth log-sum-exp approximation with continuation method, mirror descent
(multiplicative weights), FFT-based gradients, LP polishing, and simulated annealing.
The key advantage over subgradient methods is the smooth gradient enabling 10-100x
faster convergence per unit time.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_smooth_grad(a, tau, N=None, fa=None):
    """
    Compute gradient of tau*logsumexp(conv(a,a)/tau) using FFT cross-correlation.
    Returns (grad, conv, peak, fa) where fa can be reused.
    grad[i] = 2 * sum_k softmax[c[k]/tau][k] * a[k-i] for valid k-i.
    """
    n = len(a)
    size = 2 * n - 1
    if N is None:
        N = 1 << (size - 1).bit_length()
    if fa is None:
        fa = np.fft.rfft(a, N)

    # Self-convolution
    c = np.fft.irfft(fa * fa, N)[:size]
    c[c < 0] = 0.0
    peak = c.max()

    # Softmax weights (numerically stable)
    c_shifted = (c - peak) / tau
    c_shifted = np.clip(c_shifted, -50, 0)
    p = np.exp(c_shifted)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.zeros(size)
        p[np.argmax(c)] = 1.0
    else:
        p /= p_sum

    # Gradient via cross-correlation: r[i] = sum_j a[j] * p[i+j]
    # r[i] = conv(a, p_ref)[2n-2-i] where p_ref = p[::-1]
    s_ref = np.zeros(N)
    s_ref[:size] = p[::-1]
    fsr = np.fft.rfft(s_ref, N)
    conv_result = np.fft.irfft(fa * fsr, N)
    grad = 2.0 * conv_result[2 * n - 2 : n - 2 : -1]

    return grad, c, peak, fa


def normalize_nonneg(a):
    """Project to non-negative simplex (sum=1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def alt_proj_init(n, rng, n_iter=40):
    """Alternating projection for approximately flat-spectrum non-negative sequence."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    phases[0] = 0.0
    F = np.exp(1j * phases)

    a = np.ones(n) / n
    for _ in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n], 0.0)
        a = normalize_nonneg(a)
        F = np.fft.rfft(a, N)
        mag = np.maximum(np.abs(F), 1e-10)
        mean_mag = np.mean(mag[1:]) if len(mag) > 1 else mag[0]
        target = np.ones(len(mag)) * mean_mag
        target[:max(1, len(target) // 5)] *= 1.15
        F = (target / mag) * F
        F[0] = target[0] + 0j
    return normalize_nonneg(a)


def structured_init(n, rng, method='random'):
    """Generate structured initializations."""
    if method == 'cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.05
    elif method == 'edges':
        x = np.linspace(0, 1, n)
        a = (x ** 2 + (1 - x) ** 2) + 0.05
    elif method == 'bimodal':
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.15) / 0.08) ** 2) + np.exp(-((x - 0.85) / 0.08) ** 2) + 0.02
    elif method == 'trimodal':
        x = np.linspace(0, 1, n)
        a = (np.exp(-((x - 0.1) / 0.07) ** 2) + np.exp(-((x - 0.5) / 0.07) ** 2)
             + np.exp(-((x - 0.9) / 0.07) ** 2)) + 0.01
    elif method == 'v_shape':
        x = np.linspace(0, 1, n)
        a = np.abs(x - 0.5) * 2 + 0.1
    elif method == 'random':
        a = rng.random(n) + 0.01
    elif method == 'logistic':
        x = np.linspace(-5, 5, n)
        a = 1.0 / (1 + np.exp(-x)) * (1.0 / (1 + np.exp(x))) + 0.01
    else:
        a = np.ones(n)
    return normalize_nonneg(a)


def lp_polish(a, max_time=8.0):
    """LP-based local improvement: maximize sum(b) s.t. conv(a,b) <= P, b>=0."""
    n = len(a)
    if n > 1200:
        return None

    c = fast_convolve(a)
    P = c.max()
    sum_a = a.sum()

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, P),
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': max_time, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0)
            if b.sum() > 1e-8:
                return b * (sum_a / b.sum())
    except Exception:
        pass
    return None


def optimize_continuation(a, deadline, rng, use_lp=True, verbose=False):

    Starts with large tau (smooth objective) and gradually decreases to sharp.

    best_a = a.copy()

    c0 = fast_convolve(a)
    peak0 = c0.max()

        # Step size parameters

        eta_min = 0.01 / np.sqrt(n)

        max_stall = 300

def simulated_annealing(a, deadline, rng, max_iters=50000):
    """Simulated annealing to escape local minima."""
    n = len(a)
    a = normalize_nonneg(a.copy())
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    best_score = 2.0 * n * best_peak

    T = best_peak * 0.05
    T_floor = best_peak * 1e-5
    cooling = 0.9995

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # Perturb a random subset of coordinates
        num_perturb = max(1, n // 16)
        idx = rng.choice(n, size=num_perturb, replace=False)
        a_perturbed = a.copy()
        a_perturbed[idx] *= np.exp(rng.normal(0, T / best_peak, num_perturb))
        a_perturbed = normalize_nonneg(a_perturbed)

        c_new = fast_convolve(a_perturbed)
        peak_new = c_new.max()
        score_new = 2.0 * n * peak_new

        if score_new < best_score:
            best_score = score_new
            best_a = a_perturbed.copy()
            best_peak = peak_new

        # Metropolis accept/reject
        current_peak = fast_convolve(a).max()
        delta = (2.0 * n * peak_new) - (2.0 * n * current_peak)
        if delta < 0 or (T > T_floor and rng.random() < np.exp(-delta / (T + 1e-15))):
            a = a_perturbed

        T = max(T * cooling, T_floor)

    return best_a, best_score
