```python
import time
import numpy as np
from scipy.optimize import linprog

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def compute_score(a):
    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * c.max()

def smooth_objective_grad(a, tau):
    """Compute smooth objective and gradient."""
    c = np.convolve(a, a)
    # Softmax with temperature
    c_max = c.max()
    w = np.exp((c - c_max) / tau)
    w /= w.sum()
    obj = tau * (np.log(w.sum()) + c_max / tau)  # = tau * log(sum(exp(c/tau)))
    # Gradient: grad[j] = 2 * sum_m w[m+j] * a[m]
    n = len(a)
    m_idx = np.arange(n)
    idx = m_idx[None, :] + m_idx[:, None]
    grad = 2.0 * (w[idx] @ a)
    return obj, grad, w

def smooth_gd(a, budget, seed):
    """Smooth gradient descent with temperature annealing."""
    a = normalize(a.copy())
    if a is None:
        return None, float('inf')
    
    best = a.copy()
    best_score = compute_score(best)
    t0 = time.monotonic()
    deadline = t0 + budget
    
    rng = np.random.default_rng(seed)
    n = len(a)
    
    # Temperature schedule
    c = np.convolve(a, a)
    tau_max = c.max() * 0.5
    tau_min = c.max() * 0.01
    
    lr = 0.1
    stuck = 0
    
    while time.monotonic() < deadline:
        # Anneal temperature
        progress = (time.monotonic() - t0) / budget
        tau = tau_max * (tau_min / tau_max) ** (progress ** 0.7)
        
        obj, grad, w = smooth_objective_grad(a, tau)
        
        # Exponentiated gradient update
        a_new = a * np.exp(-lr * grad)
        a_new = normalize(a_new)
        if a_new is None:
            break
        
        # Check improvement
        sc = compute_score(a_new)
        if sc < best_score:
            best = a_new.copy()
            best_score = sc
            stuck = 0
            lr = min(lr * 1.01, 1.0)
        else:
            stuck += 1
            if stuck % 50 == 0:
                lr *= 0.8
                if lr < 0.01:
                    # Restart with noise
                    a = best * np.exp(rng.normal(0, 0.1, n))
                    a = normalize(a)
                    lr = 0.1
                    stuck = 0
        
        a = a_new if a_new is not None else a
    
    return best, best_score
```
