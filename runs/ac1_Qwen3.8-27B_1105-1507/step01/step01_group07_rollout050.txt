"""
Hybrid gradient + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp gradient descent with temperature annealing, combined with
LP-based alternation for final refinement. Multiple sequence lengths and restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def norm_sum1(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def grad_softmax_conv(a, T):
    """
    Compute gradient of soft_max(conv(a,a)) w.r.t. a.
    soft_max = (1/T) * log(sum(exp(T*conv_k)))
    
    d(soft_max)/d(a_i) = 2 * sum_j p_{i+j} * a_j
    where p_k = exp(T*conv_k) / sum(exp(T*conv_k))
    """
    n = len(a)
    conv = np.convolve(a, a)
    Tconv = T * conv
    Tconv -= Tconv.max()
    exp_T = np.exp(Tconv)
    p = exp_T / exp_T.sum()
    
    # grad_i = 2 * sum_{j=0}^{n-1} p[i+j] * a_j
    grad = np.zeros(n)
    for i in range(n):
        # p[i : i+n] dot a[0:n]
        grad[i] = 2.0 * np.dot(p[i:i+n], a)
    return grad


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1} (Euclidean projection)."""
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def grad_descent_step(a, n_iter=200, T_start=1.0, T_end=50.0, lr_start=0.1, lr_end=0.01, seed=0):
    """
    Gradient descent with log-sum-exp soft max and temperature annealing.
    a is assumed to be normalized (sum=1, a>=0).
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    total_iters = n_iter
    for it in range(total_iters):
        # Anneal temperature
        frac = it / max(1, total_iters - 1)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        # Compute gradient
        grad = grad_softmax_conv(a, T)
        
        # Gradient step
        a_new = a - lr * grad
        a_new = simplex_project(a_new)
        
        #偶尔 add noise to escape local optima
        if it % 50 == 49:
            noise_scale = 0.01 * (1 - frac)
            a_new = a_new + noise_scale * rng.standard_normal(n)
            a_new = simplex_project(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj


def build_A_matrix(a):
    """Build constraint matrix A for LP: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def lp_step(a, upper_bound=100.0):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, 0 <= g_i <= ub."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, M
    return None, M


def lp_optimize(a_init, max_iter=200, t=0.5, seed=0):
    """LP-based alternating optimization."""
    rng = np.random.default_rng(seed)
    a = norm_sum1(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        g, M = lp_step(a)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum <= 1.0 + 1e-6:
            # Stuck
            a = best_a * np.exp(rng.normal(0, 0.3, len(a)))
            a = norm_sum1(a)
            continue
        
        g = norm_sum1(g)
        a_new = norm_sum1((1 - t) * a + t * g)
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj


def coordinate_descent(a, max_iter=5000, seed=0):
    """Coordinate descent: randomly pick a coordinate, optimize it."""
    n = len(a)
    rng = np.random.default_rng(seed)
    best_a = norm_sum1(a.copy())
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        i = rng.integers(0, n)
        # Try increasing or decreasing a[i]
        for delta in [0.01, -0.01, 0.05, -0.05, 0.1, -0.1]:
            candidate = best_a.copy()
            candidate[i] += delta
            if candidate[i] < 0:
                continue
            candidate = norm_sum1(candidate)
            obj = compute_obj(candidate)
            if obj < best_obj:
                best_obj = obj
                best_a = candidate
                break
    
    return best_a, best_obj


def random_block_refine(a, max_iter=5000, seed=0):
    """Random block multiplicative perturbation."""
    n = len(a)
    rng = np.random.default_rng(seed)
    best_a = norm_sum1(a.copy())
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        width = max(1, n // 50)
        start = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        candidate[start:start+width] *= np.exp(rng.normal(0, 0.1, width))
        candidate = norm_sum1(candidate)
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_obj = obj
            best_a = candidate
    
    return best_a, best_obj
