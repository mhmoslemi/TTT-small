    n = len(a)
    a = a.copy()
    S = a.sum()

One more thought: the LP approach finds a direction `g` such that `conv(f, g) <= M`. The update is `f_new = (1-t)*f + t*g`. The key insight is that if `g` has a lower peak autocorrelation than `f`, then mixing them can reduce the peak.

Actually, let me think about why this works. If `conv(f, g) <= M` (elementwise), where `M = max(conv(f,f))`, then:
`conv((1-t)f + tg, (1-t)f + tg) = (1-t)^2 conv(f,f) + 2t(1-t) conv(f,g) + t^2 conv(g,g)`

We know `conv(f,f) <= M` and `conv(f,g) <= M`. But we don't know about `conv(g,g)`. If `conv(g,g) <= M` as well, then the new peak is at most `M[(1-t)^2 + 2t(1-t) + t^2] = M`. So it doesn't necessarily decrease.

The actual benefit comes from the fact that the peaks of `conv(f,f)`, `conv(f,g)`, and `conv(g,g)` are at different positions, so they don't all add up at the same location.

"""
Combined LP + spectral + SA optimization for minimizing autocorrelation peak.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (f.sum() ** 2)


def compute_score_normalized(f):
    """Assumes f is normalized to sum=1."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    n = len(f)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    valid = (idx >= 0) & (idx < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[valid] = f[idx[valid]]
    return A


def lp_direction(f, M, rng):
    """Solve LP to find direction g maximizing sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.clip(res.x, 0, None)
    except:
        pass
    return None


def spectral_shape(f, n_iter=100, rng=None):
    """Spectral shaping to reduce peak autocorrelation."""
    n = len(f)
    f = f.copy()
    S = f.sum()
    if S < 1e-10:
        return f
    f = f / S
    
    best = f.copy()
    best_score = compute_score_normalized(f)
    
    N = 2 * n  # FFT size (zero-padded)
    
    for i in range(n_iter):
        A = np.fft.rfft(f, N)
        power = np.abs(A) ** 2
        
        # Compute autocorrelation from spectrum
        r = np.fft.irfft(power, N)
        # r[0] is the circular autocorrelation at lag 0 = sum(f^2)
        # For linear autocorrelation, we need to handle the wrap-around
        # Actually, with zero-padding to N >= 2n-1, the circular autocorrelation
        # equals the linear autocorrelation for lags 0 to n-1
        # and the negative lags are the conjugate
        # So r[0:n] gives the linear autocorrelation at lags 0, 1, ..., n-1
        # and r[N-n+1:N] gives lags -(n-1), ..., -1
        
        # The full linear autocorrelation (same as conv(f,f) for real f):
        # conv(f,f)[k] for k=0..2n-2
        # In the zero-padded FFT: r[k] for k=0..n-1 is conv(f,f)[k]
        # and r[N-k] for k=1..n-1 is conv(f,f)[2n-2-k]... 
        
        # Actually let me just compute conv directly for the peak
        conv = np.convolve(f, f)
        peak = conv.max()
        k_star = np.argmax(conv)
        
        # Modify spectrum to reduce peak at k_star
        # The peak at lag k_star is: (1/N) * sum_m |A[m]|^2 * exp(-2*pi*i*m*k_star/N)
        # To reduce it, reduce |A[m]|^2 where exp(-2*pi*i*m*k_star/N) is real positive
        phase_factor = np.cos(2 * np.pi * np.arange(len(A)) * k_star / N)
        
        # Reduce power where phase factor is positive
        reduction = 1.0 - 0.1 * np.maximum(phase_factor, 0)
        new_power = power * reduction
        new_power = np.maximum(new_power, 1e-20)
        
        # Reconstruct with same phase
        new_A = A * np.sqrt(new_power / np.maximum(power, 1e-20))
        f_new = np.fft.irfft(new_A, N)[:n]
        f_new = np.clip(f_new, 0, None)
        S_new = f_new.sum()
        if S_new < 1e-10:
            continue
        f_new = f_new / S_new
        
        score = compute_score_normalized(f_new)
        if score < best_score:
            best = f_new.copy()
            best_score = score
            f = f_new
    
    return best


def local_search(f, rng, n_iter=5000):
    """Local search with block scaling and Gaussian perturbation."""
    n = len(f)
    f = f.copy()
    S = f.sum()
    f = f / S
    best = f.copy()
    best_score = compute_score_normalized(f)
    
    for i in range(n_iter):
        f_c = f.copy()
        method = rng.integers(3)
        
        if method == 0:
            # Block scaling
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, n - width))
            scale = np.exp(rng.normal(0, 0.3))
            f_c[start:start + width] *= scale
        elif method == 1:

            f_c += rng.normal(0, 0.05, n)
        else:

            idx = int(rng.integers(n))
            f_c[idx] *= np.exp(rng.normal(0, 0.5))

            best_score = score
            f = f_c

def simulated_annealing(f, rng, time_budget):
    """Simulated annealing with block operations."""
    n = len(f)
    f = f.copy()
    S = f.sum()
    f = f / S
    
    best = f.copy()
    best_score = compute_score_normalized(f)
    
    T = 0.1
    T_min = 0.001
    cooling = 0.9995
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        f_c = f.copy()
        method = rng.integers(4)
        
        if method == 0:
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start_i = int(rng.integers(0, n - width))
            scale = np.exp(rng.normal(0, 0.5))
            f_c[start_i:start_i + width] *= scale
        elif method == 1:
            f_c += rng.normal(0, 0.1, n)
        elif method == 2:
            # Swap two elements
            i, j = rng.integers(0, n, 2)
            f_c[i], f_c[j] = f_c[j], f_c[i]
        else:
            # Reverse a segment
            i = int(rng.integers(0, n - 1))
            j = int(rng.integers(i + 1, n))
            f_c[i:j] = f_c[i:j][::-1]
        
        f_c = np.clip(f_c, 0, None)
        S_c = f_c.sum()
        if S_c < 1e-10:
            continue
        f_c = f_c / S_c
        score = compute_score_normalized(f_c)
        
        delta = score - compute_score_normalized(f)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            f = f_c
            if score < best_score:
                best = f_c.copy()
                best_score = score
        
        T = max(T * cooling, T_min)
    
    return best, best_score

Actually, a better spectral approach would be:

1. Compute the autocorrelation `r = conv(f, f)`
2. Compute the "excess" above a target level: `excess[k] = max(0, r[k] - target)`
3. Compute the gradient of `sum(excess^2)` w.r.t. `f`: this is `2 * sum_k excess[k] * dr[k]/df`
4. Take a gradient step

The gradient `dr[k]/df[j] = f[j+k] + f[j-k]` (with bounds).

This is a smooth approximation to the max, and gradient descent on it should work well.

def gradient_descent_autocorr(f, n_iter=1000, lr=0.01, rng=None):

"""
Combined optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses LP-based direction finding, gradient descent on smooth max, spectral shaping,
parametric initialization, and simulated annealing with multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute the score for a sequence."""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (s ** 2)


def compute_score_norm(f):
    """Compute score assuming f is normalized to sum=1."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    valid = (idx >= 0) & (idx < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[valid] = f[idx[valid]]
    return A


def lp_solve(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None


def grad_subgradient(f, top_k=3):
    """Compute subgradient of max autocorrelation using top-k peaks."""
    n = len(f)
    r = np.convolve(f, f)
    peak_val = r.max()
    
    # Find peaks that are close to the maximum
    threshold = peak_val * 0.85
    peak_indices = np.where(r >= threshold)[0]
    
    # Take top-k
    if len(peak_indices) > top_k:
        peak_indices = peak_indices[np.argsort(r[peak_indices])[-top_k:]]
    
    grad = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        if 0 <= k < 2 * n - 1:
            # grad[j] += f[j+k] (if valid) + f[j-k] (if valid)
            if k < n:
                grad[:n - k] += f[k:]
                grad[k:] += f[:n - k]
            else:
                # k >= n, then j+k >= n always (since j >= 0), and j-k < 0 always
                # So both terms are zero... actually let me handle this properly
                # f[j+k] valid when j+k < n, i.e. j < n-k (but k>=n so n-k<=0, empty)
                # f[j-k] valid when j-k >= 0, i.e. j >= k (but k>=n so j>=n, empty)
                pass
    
    return grad


def gradient_descent(f, n_iter=2000, lr_init=0.05, top_k=5):
    """Gradient descent using subgradient of peak autocorrelation."""
    n = len(f)
    f = f.copy()
    S = f.sum()
    if S < 1e-10:
        return f
    f = f / S
    
    best = f.copy()
    best_score = compute_score_norm(f)
    lr = lr_init
    
    for i in range(n_iter):
        grad = grad_subgradient(f, top_k=top_k)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        f_new = f - lr * grad / max(grad_norm, 1e-10)
        f_new = np.clip(f_new, 0, None)
        S_new = f_new.sum()
        if S_new < 1e-10:
            lr *= 0.5
            continue
        f_new = f_new / S_new
        
        score = compute_score_norm(f_new)
        if score < best_score:
            best = f_new.copy()
            best_score = score
            f = f_new
            lr = min(lr * 1.02, 0.2)
        else:
            lr *= 0.95
            if lr < 0.001:
                break
    
    return best


def spectral_refine(f, n_iter=100):
    """Spectral shaping: reduce power at frequencies causing peak autocorrelation."""
    n = len(f)
    f = f.copy()
    S = f.sum()
    if S < 1e-10:
        return f
    f = f / S
    
    best = f.copy()
    best_score = compute_score_norm(f)
    N = 2 * n
    
    for i in range(n_iter):
        r = np.convolve(f, f)
        k_star = np.argmax(r)
        
        A = np.fft.rfft(f, N)
        power = np.abs(A) ** 2
        
        # Reduce power where it contributes constructively to the peak
        freqs = np.arange(len(A))
        phase_factor = np.cos(2.0 * np.pi * freqs * k_star / N)
        
        # Adaptive reduction strength
        strength = 0.15 + 0.1 * (i / n_iter)
        reduction = 1.0 - strength * np.maximum(phase_factor, 0)
        new_power = np.maximum(power * reduction, 1e-20)
        
        # Scale to maintain energy
        scale = np.sqrt(power.sum() / max(new_power.sum(), 1e-20))
        new_power = new_power * scale
        
        new_A = A * np.sqrt(new_power / np.maximum(power, 1e-20))
        f_new = np.fft.irfft(new_A, N)[:n]
        f_new = np.clip(f_new, 0, None)
        S_new = f_new.sum()
        if S_new < 1e-10:
            continue
        f_new = f_new / S_new
        
        score = compute_score_norm(f_new)
        if score < best_score:
            best = f_new.copy()
            best_score = score
            f = f_new
    
    return best


def local_search(f, rng, n_iter=5000):
    """Local search with diverse perturbation strategies."""
    n = len(f)
    f = f.copy()
    S = f.sum()
    f = f / S
    best = f.copy()
    best_score = compute_score_norm(f)
    accept_count = 0
    
    for i in range(n_iter):
        f_c = f.copy()
        method = rng.integers(5)
        
        if method == 0:
            # Block scaling
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, n - width))
            scale = np.exp(rng.normal(0, 0.3))
            f_c[start:start + width] *= scale
        elif method == 1:
            # Gaussian noise
            sigma = 0.03 + 0.02 * (accept_count / max(n_iter, 1))
            f_c += rng.normal(0, sigma, n)
        elif method == 2:
            # Single element
            idx = int(rng.integers(n))
            f_c[idx] *= np.exp(rng.normal(0, 0.5))
        elif method == 3:
            # Pair swap
            i, j = rng.integers(0, n, 2)
            f_c[i], f_c[j] = f_c[j], f_c[i]
        else:
            # Segment reversal
            i = int(rng.integers(0, n - 2))
            j = int(rng.integers(i + 2, n))
            f_c[i:j] = f_c[i:j][::-1]
        
        f_c = np.clip(f_c, 0, None)
        S_c = f_c.sum()
        if S_c < 1e-10:
            continue
        f_c = f_c / S_c
        score = compute_score_norm(f_c)
        
        if score < best_score:
            best = f_c.copy()
            best_score = score
            f = f_c
            accept_count += 1
    
    return best, best_score


def simulated_annealing(f, rng, time_budget):
    """Simulated annealing for broader exploration."""
    n = len(f)
    f = f.copy()
    S = f.sum()
    f = f / S
    
    best = f.copy()
    best_score = compute_score_norm(f)
    T = 0.05
    T_min = 0.0001
    cooling = 0.9998
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        f_c = f.copy()
        method = rng.integers(4)
        
        if method == 0:
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start_i = int(rng.integers(0, n - width))
            scale = np.exp(rng.normal(0, 0.4))
            f_c[start_i:start_i + width] *= scale
        elif method == 1:
            f_c += rng.normal(0, 0.08, n)
        elif method == 2:
            i, j = rng.integers(0, n, 2)
            f_c[i], f_c[j] = f_c[j], f_c[i]
        else:
            i = int(rng.integers(0, n - 1))
            j = int(rng.integers(i + 1, n))
            f_c[i:j] = f_c[i:j][::-1]
        
        f_c = np.clip(f_c, 0, None)
        S_c = f_c.sum()
        if S_c < 1e-10:
            continue
        f_c = f_c / S_c
        score = compute_score_norm(f_c)
        
        curr_score = compute_score_norm(f)
        delta = score - curr_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            f = f_c
            if score < best_score:
                best = f_c.copy()
                best_score = score
        
        T = max(T * cooling, T_min)
    
    return best, best_score


def lp_optimize(f, time_budget, rng, verbose=False):

        if S_new < 1e-10:

            stall += 1

                                best_score = c_score

                            break

    return best, best_score


def parametric_init(n, rng):
    """Generate diverse parametric initial sequences."""
    candidates = []
    
    # Gaussian
    sigma = rng.uniform(0.2, 0.5)
    center = n * 0.5
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - center) / (sigma * n)) ** 2)
    candidates.append(a)
    
    # Raised cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    candidates.append(a)
    
    # Exponential decay from center
    a = np.exp(-np.abs(x - center) / (n * 0.3))
    candidates.append(a)
    
    # Two-bump
    c1 = n * rng.uniform(0.2, 0.4)
    c2 = n * rng.uniform(0.6, 0.8)
    s1 = rng.uniform(0.1, 0.2)
    s2 = rng.uniform(0.1, 0.2)
    a = np.exp(-0.5 * ((x - c1) / (s1 * n)) ** 2) + np.exp(-0.5 * ((x - c2) / (s2 * n)) ** 2)
    candidates.append(a)
    
    # Cosine window with phase
    phase = rng.uniform(0, np.pi)
    a = np.abs(np.cos(np.pi * x / n + phase)) ** rng.uniform(1, 3)
    candidates.append(a)
    
    # Random smooth (low-pass filtered noise)
    a = np.ones(n)
    a += rng.uniform(0.3, 0.7) * np.sin(2 * np.pi * rng.uniform(1, 4) * x / n + phase)
    a += rng.uniform(0.2, 0.5) * np.sin(2 * np.pi * rng.uniform(4, 8) * x / n + phase * 2)
    a = np.abs(a)
    candidates.append(a)
    
    # Uniform with small perturbation
    a = np.ones(n) * (1 + 0.1 * rng.normal(0, 1, n))
    a = np.clip(a, 0.1, None)
    candidates.append(a)
    
    return candidates

                best_score = score
                best = f_opt.copy()
                print(f"  Restart {r}: {score:.6f} ** NEW BEST **")
        
        # Phase 5: SA + local search polish
        print(f"\n=== Phase 5: Polish ===")
        time_left = deadline - time.monotonic()
        
        if time_left > 15:
            t_sa = min(time_left * 0.4, 60)
            f_sa, sa_score = simulated_annealing(best, rng, t_sa)
            if sa_score < best_score:
                best_score = sa_score
                best = f_sa.copy()
                print(f"  After SA: {sa_score:.6f} ** NEW BEST **")
        
        time_left = deadline - time.monotonic()
        if time_left > 5:
            n_iter = min(15000, int(time_left * 400))
            f_pol, pol_score = local_search(best, rng, n_iter=n_iter)
            if pol_score < best_score:
                best_score = pol_score
                best = f_pol.copy()
                print(f"  After local search: {pol_score:.6f} ** NEW BEST **")
        
        # Final gradient polish
        time_left = deadline - time.monotonic()
        if time_left > 3:
            f_grad = gradient_descent(best, n_iter=min(500, int(time_left * 100)), top_k=5)
            g_score = compute_score_norm(f_grad)
            if g_score < best_score:
                best_score = g_score
                best = f_grad.copy()
                print(f"  Final gradient: {g_score:.6f} ** NEW BEST **")
    
    # Final verification
    if best is not None:
        best = np.clip(best, 0, 1000)
        final_list = best.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal verified score: {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        return final_list
    
    print("Falling back to uniform")
    return [1.0] * 256
