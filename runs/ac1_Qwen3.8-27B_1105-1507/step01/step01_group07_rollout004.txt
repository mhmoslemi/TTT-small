def build_A_matrix(a):
    """Build the Toeplitz constraint matrix A for the LP.
    A[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A

def solve_lp_for_g(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, 0 <= g_i.
    Returns (g, sum_g) or (None, None) if infeasible.
    """
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None

"""
Combined smoothed-subgradient + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated convolution, log-sum-exp smoothed gradients for fast local optimization,
and alternating LP for global improvements. Explores multiple sequence lengths (30-500).
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute autocorrelation conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smoothed_gradient(conv, a, tau):
    """Compute smoothed gradient of max(conv) using log-sum-exp weights.
    Returns gradient of length n."""
    n = len(a)
    max_conv = np.max(conv)
    # For numerical stability
    shifted = tau * (conv - max_conv)
    # Clip to avoid underflow/overflow
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    # grad[i]/2 = sum_j a[j] * weights[i+j] = np.correlate(weights, a, 'full')[i]
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[:n]
    return grad


def subgradient_optimize(a, max_iter=20000, tau_start=5.0, tau_end=50.0,
                         step_init=0.1, verbose=False):
    """Run smoothed subgradient method to minimize max(conv(a,a)) with fixed sum."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    step = step_init
    stall_count = 0
    patience = 200

    for it in range(max_iter):
        # Animate tau from tau_start to tau_end
        frac = it / max(1, max_iter)
        tau = tau_start + (tau_end - tau_start) * frac

        conv = compute_conv(a)
        current_max = float(np.max(conv))

        if current_max < best_max - 1e-10:
            best_max = current_max
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                # Decrement step and try to escape
                step *= 0.7
                if step < 1e-8:

                # Add small noise

                a = a + noise * step
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s < 1e-10:

                a = a / s
                continue

        # Center gradient (for sum constraint)
        grad_centered = grad - np.mean(grad)

        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)

        a_new = a_new / s
        a = a_new

def build_A_matrix(a):
    """Build the Toeplitz constraint matrix for the LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def solve_lp_for_g(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None


def lp_optimize(a, max_iter=100, t=0.3, verbose=False):

    a = a.copy()
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    stall = 0
    for it in range(max_iter):
        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_for_g(a, M)
        if g is None:
            break

        if sum_g <= 1.0 + 1e-4:
            stall += 1
            if stall > 20:
                break
            continue
        stall = 0

        g = g / sum_g
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print(f"      LP it={it}: obj={best_obj:.6f}")
        a = a_new

    return best_a, best_obj

    # Try parent sequence
    parent, parent_obj = try_parent_sequence()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # Sequence lengths to explore
    n_values = [30, 40, 50, 60, 70, 80, 100, 120, 150, 180, 200, 250, 300, 400]

    # If we have a parent, start with its length and neighbors
    if parent is not None:
        n_parent = len(parent)
        n_values = sorted(set(n_values + [n_parent, max(10, n_parent - 20), min(600, n_parent + 20)]))

    print(f"Exploring n values: {n_values}")

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Time allocation
        if n <= 60:
            alloc = min(25.0, remaining * 0.08)
        elif n <= 120:
            alloc = min(50.0, remaining * 0.12)
        elif n <= 200:
            alloc = min(80.0, remaining * 0.15)
        elif n <= 300:
            alloc = min(120.0, remaining * 0.2)
        else:
            alloc = min(180.0, remaining * 0.25)

        alloc = max(10.0, alloc)
        print(f"N={n}: {remaining:.0f}s remaining, allocating {alloc:.0f}s")

        a_opt, obj = run_combined(n, seed=seed + n * 7919, max_time=alloc,
                                  verbose=(n in [50, 100, 150, 200]))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  UPDATE: n={n}, obj={best_score:.6f}")

    # Final intensive polishing on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nPolishing best sequence (n={n_best}), {remaining:.0f}s remaining")

        a_cur = best_seq.copy()
        polish_round = 0

        while time.monotonic() < deadline - 5:
            polish_round += 1
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break

            # Subgradient polish with high tau (focus on the peak)
            sg_iters = int(min(20000, remaining * 3))
            a_cur, _ = subgradient_optimize(a_cur, max_iter=sg_iters,
                                             tau_start=30.0, tau_end=100.0,
                                             step_init=0.03)
            obj = compute_obj(a_cur)
            if obj < best_score:
                best_score = obj
                best_seq = a_cur.copy()
                print(f"  Polish {polish_round} sg: obj={best_score:.6f}")

            # LP step if time allows and n is reasonable
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 350:
                a_lp, obj_lp = lp_optimize(a_cur, max_iter=50, t=0.25)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy().astype(np.float64)
                    print(f"  Polish {polish_round} LP: obj={best_score:.6f}")
                else:
                    a_cur = a_cur

            # Random restart perturbation if stuck
            if remaining > 20:
                noise_scale = 0.05
                a_perturb = a_cur * np.exp(rng.normal(0, noise_scale, n_best))
                a_perturb = np.maximum(a_perturb, 0)
                s_p = np.sum(a_perturb)
                if s_p > 1e-10:
                    a_perturb = a_perturb / s_p
                    obj_p = compute_obj(a_perturb)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_perturb.copy()
                        a_cur = a_perturb.copy()
                        print(f"  Polish {polish_round} perturb: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation != best_score and validation < np.inf:
        print(f"  Validation discrepancy: computed={best_score:.6f}, validated={validation:.6f}")
        best_score = validation

    return result
