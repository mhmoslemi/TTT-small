def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mirror_descent_step(a, grad, eta):
    """
    Multiplicative (mirror descent) update.
    a_new[j] = a[j] * exp(-eta * grad[j])
    Then normalize to sum=1.
    """
    log_update = -eta * grad
    # Clip to prevent overflow/underflow
    log_update = np.clip(log_update, -50, 50)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-15)  # Soft floor
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a_new / s


def lp_step(a, P, max_time=5.0):
    """
    LP step: maximize sum(b) subject to convolve(a,b)[k] <= P, b >= 0.
    Returns rescaled solution or None.
    """
    n = len(a)
    if n > 2048:
        return None

    # Build convolution matrix A of shape (2n-1, x n)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    # LP: min -1^T b s.t. A @ b <= P*ones, b >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n, 
                      method='highs', options={'time_limit': max_time})
        if res.success:
            b = res.x
            b_sum = b.sum()
            if b_sum > 1e-8:
                a_sum = a.sum()
                b = b * (a_sum / b_sum)
                return b
    except Exception:
        pass
    return None

def freq_smooth_step(a, alpha=0.1):
    """
    Frequency-domain smoothing: blend the power spectrum with its smoothed version,
    then recover a time-domain signal and clip to non-negative.
    """
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, N)
    power = np.abs(fa) ** 2
    
    # Smooth the power spectrum (circular smoothing)
    kernel_size = max(3, N // 100)
    if kernel_size % 2 == 0:
        kernel_size += 1
    kernel = np.ones(kernel_size) / kernel_size
    power_smooth = np.convolve(power, kernel, mode='same')
    
    # Blend
    power_new = (1 - alpha) * power + alpha * power_smooth
    
    # Recover time domain (need sqrt of power, with phase from original)
    mag_new = np.sqrt(power_new)
    phase = np.angle(fa)
    fa_new = mag_new * np.exp(1j * phase)
    fa_new[0] = fa_new[0].real + 0j
    
    a_full = np.fft.irfft(fa_new, N)
    a_new = a_full[:n].copy()
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses adaptive mirror descent with soft-max gradient weighting, frequency-domain
smoothing, LP-based refinement, and diverse multi-start initializations.
Employs coarse-to-fine strategy with iterative peak-directed perturbation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_weighted(a, c, temperature=0.03):
    """
    Compute weighted gradient using softmax over convolution values.
    grad[j] = sum_k w[k] * 2 * a[k-j] where 0<=k-j<n and w[k] ~ softmax(c/k).
    """
    n = len(a)
    max_c = np.max(c)
    if max_c < 1e-15:
        return np.zeros(n)
    # Softmax weights (scaled to avoid overflow)
    scaled = (c - max_c) / (temperature * max_c + 1e-12)
    weights = np.exp(scaled)
    weights = weights / weights.sum()
    # Only use lags with non-negligible weight
    threshold = weights.max() * 0.005
    active = np.where(weights > threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    grad = np.zeros(n)
    for k in active:
        w = weights[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * w * a[k - hi:k - lo + 1][::-1]
    return grad


def mirror_descent_step(a, grad, eta):
    """Multiplicative update: a_new = a * exp(-eta*grad), normalized."""
    log_update = np.clip(-eta * grad, -50, 50)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-15)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a_new / s


def lp_step(a, P, max_time=5.0):
    """LP: max sum(b) s.t. conv(a,b) <= P, b >= 0. Returns rescaled b or None."""
    n = len(a)
    if n > 2000:
        return None
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method='highs', options={'time_limit': max_time})
        if res.success:
            b = res.x
            b_sum = b.sum()
            if b_sum > 1e-8:
                a_sum = a.sum()
                b = b * (a_sum / b_sum)
                return b
    except Exception:
        pass
    return None


def alt_projection_init(n, rng, n_iter=50):
    """Alternating projection for near-flat spectrum non-negative sequence."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j
    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        weights = np.linspace(1.0, 0.85, len(F_full))
        mag_target = weights ** 0.05
        F = mag_target * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j
    return a


def freq_smooth_step(a, alpha=0.08):
    """Smooth power spectrum in frequency domain, recover non-negative time signal."""
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, N)
    power = np.abs(fa) ** 2
    ks = max(3, N // 80)
    if ks % 2 == 0:
        ks += 1
    kernel = np.ones(ks) / ks
    power_smooth = np.convolve(power, kernel, mode='same')
    power_new = (1 - alpha) * power + alpha * power_smooth
    mag_new = np.sqrt(np.maximum(power_new, 0))
    phase = np.angle(fa)
    fa_new = mag_new * np.exp(1j * phase)
    fa_new[0] = fa_new[0].real + 0j
    a_full = np.fft.irfft(fa_new, N)
    a_new = a_full[:n].copy()
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def peak_directed_perturb(a, c, rng, strength=0.05):
    """Perturb a to reduce peak: decrease values contributing to peak, increase others."""
    n = len(a)
    peak_idx = np.argmax(c)
    # Compute contribution of each a[j] to the peak
    contrib = np.zeros(n)
    k = peak_idx
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        contrib[lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    # Also consider near-peak lags
    top_lags = np.argsort(c)[::-1][:5]
    for k2 in top_lags:
        if k2 == peak_idx:
            continue
        lo2 = max(0, k2 - n + 1)
        hi2 = min(n - 1, k2)
        if lo2 <= hi2:
            contrib[lo2:hi2 + 1] += 0.3 * a[k2 - hi2:k2 - lo2 + 1][::-1]
    # Perturbation: reduce high-contribution positions, increase low ones
    pert = -strength * contrib
    pert += rng.normal(0, strength * 0.1 / np.sqrt(n), n)
    a_new = a + pert
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s

    frequency smoothing, peak-directed perturbation, and LP refinement.

    a = project_simplex(a.copy().astype(np.float64))
    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()
    # Adaptive parameters
    eta = 3.0 / np.sqrt(n)
    eta_min = 0.01 / np.sqrt(n)

    temp_min = 0.005
    it = 0
    stall_count = 0
    max_stall = 400
    last_improvement_it = 0

        it += 1
        c = fast_convolve(a)
        peak = np.max(c)
        if peak < best_peak * (1.0 - 1e-10):
            best_peak = peak
            best_a = a.copy()
            stall_count = 0
            last_improvement_it = it
        else:
            stall_count += 1
        # Adaptive step size
        if stall_count > max_stall:

            temp = max(temp * 0.6, temp_min)
            stall_count = 0

                r = rng.random()
                if r < 0.3:
                    a = peak_directed_perturb(a, c, rng, strength=0.08)
                elif r < 0.6:
                    perturb = rng.normal(0, 0.04 / np.sqrt(n), n)
                    a = project_simplex(a + perturb)
                else:
                    a_s = freq_smooth_step(a, alpha=0.15)
                    a = 0.5 * a + 0.5 * a_s
        elif it - last_improvement_it > 100:
            eta = min(eta * 1.02, eta_max)
        # Compute gradient
        grad = compute_gradient_weighted(a, c, temperature=temp)
        grad_max = np.max(np.abs(grad))
        if grad_max < 1e-15:
            break
        grad_normed = grad / grad_max
        # Mirror descent step
        a_new = mirror_descent_step(a, grad_normed, eta)
        # Occasional frequency smoothing
        if it % 800 == 0 and time.time() < deadline - 30:
            a_s = freq_smooth_step(a_new, alpha=0.04)
            a_new = 0.7 * a_new + 0.3 * a_s
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
        a = a_new
        # LP step

            b = lp_step(a, best_peak, max_time=4.0)
            if b is not None:
                t_mix = 0.35 + 0.15 * rng.random()
                a = (1 - t_mix) * a + t_mix * b
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
    return best_a


def try_sequence(n, init_type, rng, deadline, use_lp=True):
    """Initialize and optimize a sequence of given length."""
    if init_type == 'alt_proj':
        a = alt_projection_init(n, rng, n_iter=50)
    elif init_type == 'random':
        a = rng.random(n)
        a = a / a.sum()
    elif init_type == 'cosine':
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x)) + 0.003
        a = a / a.sum()
    elif init_type == 'cosine2':
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * (1.0 + np.cos(4.0 * np.pi * x)) + 0.003
        a = a / a.sum()
    elif init_type == 'gaussian':
        x = np.linspace(0.0, 1.0, n)
        center = 0.5
        sigma = rng.uniform(0.18, 0.32)
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2) + 0.002
        a = a / a.sum()
    elif init_type == 'bimodal':
        x = np.linspace(0.0, 1.0, n)
        sigma = rng.uniform(0.08, 0.14)
        pos1 = rng.uniform(0.25, 0.35)
        pos2 = 1.0 - pos1
        a = np.exp(-0.5 * ((x - pos1) / sigma) ** 2) + np.exp(-0.5 * ((x - pos2) / sigma) ** 2) + 0.002
        a = a / a.sum()
    elif init_type == 'ushaped':
        x = np.linspace(0.0, 1.0, n)
        a = (x - 0.5) ** 2 + 0.01
        a = a / a.sum()
    elif init_type == 'parent':
        try:
            parent = np.array(height_sequence_1, dtype=np.float64)
            parent = np.clip(parent, 0.0, 1000.0)
            if len(parent) >= n:
                a = parent[:n].copy()
            else:
                a = np.zeros(n)
                a[:len(parent)] = parent
                if len(parent) < n:
                    fracs = np.linspace(0, 1, n - len(parent))
                    a[len(parent):] = parent[-1] * fracs
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s if s > 1e-10 else np.ones(n) / n
        except Exception:
            a = rng.random(n)
            a = a / a.sum()
    elif init_type == 'broadband':
        a = alt_projection_init(n, rng, n_iter=25)
        a = a + 0.08 * rng.random(n)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    elif init_type == 'triangular':
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 - np.abs(2.0 * x - 1.0) + 0.005
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()
    a_opt = optimize_sequence(a, deadline, rng, use_lp=use_lp)
    score = evaluate_sequence(a_opt.tolist())
    return a_opt, score


def propose_candidate(budget_s=1500, **kwargs):
    """
    Multi-phase search: Phase 1 multi-start (35%), Phase 2 intensive refinement (50%),
    Phase 3 final polish (15%). Uses mirror descent, freq smoothing, LP refinement.
    """
    overall_deadline = time.time() + budget_s - 15.0
    rng = np.random.default_rng(77)
    best_score = float('inf')
    best_seq = None
    # Phase 1: Multi-start exploration
    phase1_deadline = time.time() + (budget_s - 15.0) * 0.35
    configs_phase1 = [
        (512, 'parent'),
        (512, 'alt_proj'),
        (640, 'alt_proj'),
        (384, 'alt_proj'),
        (512, 'ushaped'),
        (768, 'alt_proj'),
        (448, 'alt_proj'),
        (512, 'bimodal'),
        (1024, 'alt_proj'),
    ]
    num_configs = len(configs_phase1)
    time_per_config = max(15.0, (phase1_deadline - time.time()) / num_configs)
    print(f"Phase 1: {num_configs} configs, ~{time_per_config:.0f}s each")
    for i, (n, init_type) in enumerate(configs_phase1):
        remaining = phase1_deadline - time.time()
        if remaining < 12:
            break
        config_deadline = time.time() + min(time_per_config, remaining)
        use_lp = n <= 1500
        try:
            a_opt, score = try_sequence(n, init_type, rng, config_deadline, use_lp=use_lp)
        except Exception as e:
            print(f"  Error ({n}, {init_type}): {e}")
            continue
        print(f"  P1: n={n}, init={init_type}, score={score:.6f}, remain={overall_deadline - time.time():.0f}s")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  >>> New best: {best_score:.6f} (n={n})")
    # Phase 2: Intensive refinement
    phase2_deadline = time.time() + (overall_deadline - time.time()) * 0.55
    print(f"Phase 2 starts, deadline in {phase2_deadline - time.time():.0f}s")
    if best_seq is not None and time.time() < phase2_deadline - 15:
        n_best = len(best_seq)
        use_lp = n_best <= 1500
        a_curr = best_seq.copy()
        for pass_num in range(4):
            if time.time() > phase2_deadline - 20:
                break
            passes_left = 4 - pass_num
            pass_deadline = time.time() + min(
                (phase2_deadline - time.time()) / passes_left,
                overall_deadline - time.time() - 10
            )
            a_refined = optimize_sequence(a_curr, pass_deadline, rng, use_lp=use_lp)
            a_curr = a_refined
            score = evaluate_sequence(a_refined.tolist())
            print(f"  P2 pass {pass_num}: score={score:.6f}")
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_refined.copy()
                print(f"  >>> New best: {best_score:.6f}")
        # Dedicated LP refinement
        if n_best <= 1200 and time.time() < phase2_deadline - 30:
            a_curr = project_simplex(best_seq.copy())
            c = fast_convolve(a_curr)
            peak = np.max(c)
            print(f"  LP refinement starting, peak={peak:.8f}")
            for lp_iter in range(8):
                if time.time() > phase2_deadline - 15:
                    break
                b = lp_step(a_curr, peak, max_time=6.0)
                if b is None:
                    break
                t_mix = 0.45
                a_curr = (1 - t_mix) * a_curr + t_mix * b
                a_curr = np.maximum(a_curr, 0)
                s = a_curr.sum()
                if s > 1e-10:
                    a_curr = a_curr / s
                c_new = fast_convolve(a_curr)
                new_peak = np.max(c_new)
                score = evaluate_sequence(a_curr.tolist())
                print(f"    LP iter {lp_iter}: peak={new_peak:.8f}, score={score:.6f}")
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = a_curr.copy()
                    peak = new_peak
                    print(f"    >>> New best: {best_score:.6f}")
                elif new_peak > peak * 1.01:
                    break
    # Phase 3: Final polish
    if best_seq is not None and time.time() < overall_deadline - 10:
        n_final = len(best_seq)
        a_polished = optimize_sequence(best_seq, overall_deadline, rng,
                                       use_lp=n_final <= 1200, max_iters=800000)
        score = evaluate_sequence(a_polished.tolist())
        print(f"Phase 3 (polish): score={score:.6f}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_polished
    # Fallback
    if best_seq is None:
        print("FALLBACK: uniform sequence")
        best_seq = np.ones(512)
    result = [float(x) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"FINAL: score={final_score:.6f}, length={len(result)}")
    return result
