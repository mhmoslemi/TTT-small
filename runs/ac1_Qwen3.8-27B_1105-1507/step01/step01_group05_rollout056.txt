import numpy as np
from scipy.signal import correlate, convolve
import time

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)

def compute_score_normalized(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a) = 1."""
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()

def smooth_max_value(conv, beta):
    """Log-sum-exp approximation of max."""
    c = conv.max()
    return c + np.log(np.sum(np.exp(beta * (conv - c)))) / beta

def smooth_max_gradient(a, beta):
    """Gradient of 2n * smooth_max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = np.convolve(a, a)
    c = conv.max()
    exp_conv = np.exp(beta * (conv - c))
    w = exp_conv / exp_conv.sum()
    
    # T1[i] = sum_j a[j] * w[i+j] = correlate(w, a, 'full')[i+n-1]
    T1_full = correlate(w, a, 'full')
    T1 = T1_full[n-1 : 2*n-1]
    
    # T2[i] = sum_j w[j] * a[i-j] = convolve(w, a, 'full')[i]
    T2_full = convolve(w, a, 'full')
    T2 = T2_full[:n]
    
    grad = 2.0 * n * (T1 + T2)
    return grad

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Sort and find threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.sum(u * np.arange(1, n+1) > (cssv - 1))
    theta = (cssv[rho-1] - 1) / rho if rho > 0 else 0
    proj = np.maximum(v - theta, 0)
    # Ensure sum is exactly 1
    proj /= proj.sum()
    return proj
