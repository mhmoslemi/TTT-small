2. Compute gradient: `grad[i] = 2*f[k*-i]` (or `f[i]` if `i = k*/2`)

`∇ = (1/sum_k exp(β*(f*f)_k)) * sum_k exp(β*(f*f)_k) * ∇(f*f)_k`

This is a weighted average of the gradients at each index, with weights proportional to `exp(β*(f*f)_k)`. As `β → ∞`, this approaches the subgradient at the maximum.

        return None

    shifted = conv - c_max

    # = sum_k w_k * grad_k
    # where grad_k[i] = 2*f[k-i] if 0<=k-i<n and k!=2i, else f[i] if k=2i
    # = 2 * (cross_corr of w and f at lags 0..n-1) - f * w[0, 2, 4, ...]

    diag = f * w[0:2*n-1:2]  # w[0], w[2], w[4], ..., w[2n-2]
    
    grad = 2.0 * cross_lags - diag
    return smooth_max, grad

    f = normalize_sequence(f)

        return f_init, score_normalized(f_init, n)
    
    best_f = f.copy()
    best_score = score_normalized(f, n)
    deadline = time.monotonic() + max(0.0, float(time_budget))

    lr = lr_init
    t = 1
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Nesterov momentum
        momentum = (t - 1) / (t + 2)
        y = f + momentum * (f - f_prev)
        y = np.maximum(y, 0.0)
        s = float(y.sum())

        _, grad = smooth_score_grad(y, n, beta)

            else:

        # Adapt beta
