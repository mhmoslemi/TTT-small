Optimized search for low-score non-negative step-function coefficients.
Uses FFT-accelerated subgradient descent with Adam optimizer, 
temperature-based active set selection, and simulated annealing.

import time
import numpy as np

def fft_convolve(a, b):
    """Compute full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    Fa = np.fft.fft(a, size)
    Fb = np.fft.fft(b, size)
    return np.real(np.fft.ifft(Fa * Fb))[:n]

def compute_score(a):
    """Compute the evaluation score."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = fft_convolve(a, a)
    M = float(np.max(c))
    if M < 0:
        M = 0.0  # numerical precision
    return float(2.0 * n * M / (s * s))

def subgradient_convolution(a, temperature=0.0, top_k=None):
    """
    Compute subgradient of max_k (a*a)_k with temperature-controlled active set.
    Returns (gradient, max_value, active_count).
    """
    n = len(a)
    c = fft_convolve(a, a)
    M = float(np.max(c))
    if M <= 1e-30:
        return np.zeros(n), M, 0

        # Soft active set: weight by exp(c/M / temperature - 1)
        # This gives weight ~1 to entries near M and less to others
        weights = np.exp(c / M / temperature - 1.0)
        # Clip to avoid overflow
        weights = np.clip(weights, 0.0, 1e6)
        total_w = float(np.sum(weights))
        if total_w < 1e-30:
            weights = np.zeros_like(c)
            weights[np.argmax(c)] = 1.0
            total_w = 1.0
        weights /= total_w
    else:
        if top_k is not None and top_k < len(c):
            idx = np.argpartition(c, -top_k)[-top_k:]
            weights = np.zeros(len(c))
            weights[idx] = 1.0 / top_k
            total_w = 1.0

            threshold = 0.99 * M
            active = c >= threshold
            if not np.any(active):
                active[np.argmax(c)] = True
            weights = active.astype(np.float64) / float(np.sum(active))
            total_w = 1.0

    g = np.zeros(n)

        # a[k-i] is valid for i in [max(0, k-(n-1)), min(n-1, k)]

        e = min(n - 1, k)
        if s <= e:
            i_range = np.arange(s, e + 1)
            g[i_range] += wk * a[k - i_range]
    
    return g, M, len(nz)

def adam_multiplicative_update(a, g, m, v, t, lr, beta1, beta2, eps):
    """Adam-style update in multiplicative (log) space."""
    n = len(a)
    m = beta1 * m + (1 - beta1) * g
    v = beta2 * v + (1 - beta2) * g * g
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)

    else:

                m = np.zeros(n)
                v = np.zeros(n)

    a = np.asarray(a, dtype=np.float64)
    n = len(a)

    while time.monotonic() < deadline:
        # Temperature schedule (exponential decay)
        frac = (time.monotonic() - t0) / max_time

        pert = rng.random()

            if sc < best_sc:
                best = cur.copy()
                best_sc = float(sc)

"""
Search for minimum-score non-negative step-function coefficients.
Score = 2*n*max(convolve(a,a))/sum(a)^2.

Algorithm:
- FFT-accelerated mirror descent (multiplicative subgradient) with Adam scaling
- Temperature-controlled soft active set for adaptive peak focusing
- Frequency-domain initialization (flat |DFT| magnitude via GS-like iterations)
- Simulated annealing with block perturbations for escaping local minima
- Population-based crossover to combine good sequences
- Multiple sequence lengths with adaptive time allocation
- Nesterov momentum acceleration
"""

import time
import numpy as np

# --- Core utilities ---

def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    Fa = np.fft.rfft(a, size)
    Fb = np.fft.rfft(b, size)
    return np.fft.irfft(Fa * Fb, size)[:n]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = fft_convolve(a, a)
    M = float(np.max(c))
    if M < 0:
        M = 0.0
    return float(2.0 * n * M / (s * s))


def subgradient(a, temp=0.0, top_k=None):
    """
    Compute subgradient of max_k (conv(a,a)_k).
    Returns (grad, max_val).
    """
    n = len(a)
    c = fft_convolve(a, a)
    M = float(np.max(c))
    if M <= 1e-30:
        return np.zeros(n), M

    if temp > 0:
        # Soft weights: exp((c - M) / (M * temp)) normalized
        # This gives weight ~1 to peaks, decaying for others
        exponents = (c - M) / (M * temp)
        exponents = np.clip(exponents, -50.0, 0.0)
        weights = np.exp(exponents)
        tw = float(np.sum(weights))
        if tw < 1e-30:
            weights = np.zeros_like(c)
            weights[int(np.argmax(c))] = 1.0
            tw = 1.0
        weights /= tw
    else:
        if top_k is not None and 0 < top_k < len(c):
            idx = np.argpartition(c, -top_k)[-top_k:]
            weights = np.zeros(len(c))
            weights[idx] = 1.0 / len(idx)
        else:
            thr = 0.95 * M
            active = np.where(c >= thr)[0]
            if active.size == 0:
                active = np.array([int(np.argmax(c))])
            weights = np.zeros(len(c))
            weights[active] = 1.0 / active.size

    # Compute g[i] = sum_k weights[k] * 2 * a[k-i]  (valid where 0 <= k-i < n)
    g = np.zeros(n)
    nz = np.where(weights > 1e-12)[0]
    for k in nz:
        wk2 = 2.0 * float(weights[k])
        s = max(0, k - (n - 1))
        e = min(n - 1, k)
        if s <= e:
            i_idx = np.arange(s, e + 1)
            g[i_idx] += wk2 * a[k - i_idx]

    return g, M


# --- Initialization strategies ---

def init_uniform(n, rng):
    return np.ones(n) / n


def init_random(n, rng, alpha=1.0):
    v = rng.random(n) ** alpha
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return v / s


def init_two_bump(n, rng):
    """Mass in two separated bumps."""
    w = int(rng.integers(n // 4, n // 2))
    s1 = int(rng.integers(0, n - w))
    a = np.zeros(n)
    a[s1:s1 + w] = 1.0
    s2 = int(rng.integers(0, n - w))
    a[s2:s2 + w] += 1.0
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    # Blur
    k = int(rng.integers(3, max(4, n // 20)))
    kern = np.ones(k) / k
    a = fft_convolve(a, kern)[:n]
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_freq_flat(n, rng, n_iter=200):
    """
    Gerchberg-Saxton style: iterate between clipping to [0,inf) in time domain
    and flattening |DFT| in frequency domain.
    """
    a = rng.random(n) + 0.1
    a = np.abs(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a /= a.max()

    for _ in range(n_iter):
        # FFT
        A = np.fft.rfft(a)
        # Flatten magnitude: set all |A[k]| to the mean (except DC)
        mag = np.abs(A)
        # Keep DC, flatten the rest
        mean_mag = float(np.mean(mag[1:])) if len(mag) > 1 else mag[0]
        phase = np.angle(A)
        new_mag = np.full_like(mag, mean_mag)
        new_mag[0] = mag[0]  # preserve DC (sum)
        A_new = new_mag * np.exp(1j * phase)
        a = np.fft.irfft(A_new, n)
        # Clip to non-negative
        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-15:
            break
        a /= s

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_from_parent(parent, n, rng):
    """Resample/interpolate parent to length n with perturbation."""
    p = np.asarray(parent, dtype=np.float64)
    p = np.clip(p, 0, None)
    sp = float(np.sum(p))
    if sp < 1e-15:
        return init_random(n, rng)
    p /= sp
    # Interpolate to new length
    old_x = np.linspace(0, 1, p.size)
    new_x = np.linspace(0, 1, n)
    a = np.interp(new_x, old_x, p)
    a = np.clip(a, 0, None)
    a *= np.exp(rng.normal(0, 0.3, n))
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


# --- Optimization methods ---

def mirror_descent_optimize(a, max_time, seed, lr=0.4, temp_init=0.08, temp_min=0.005):
    """
    Mirror descent (multiplicative subgradient) with Adam-style adaptive scaling.
    """
    rng = np.random.default_rng(seed)
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best = a.copy()
    best_sc = compute_score(best)
    t0 = time.monotonic()
    deadline = t0 + max_time

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    # Nesterov momentum
    a_prev = a.copy()
    mu = 0.4

    temp = temp_init
    lr_cur = lr
    stuck = 0
    it = 0

    while time.monotonic() < deadline:
        t += 1

        # Nesterov lookahead
        a_look = a + mu * (a - a_prev)
        a_look = np.clip(a_look, 0, None)
        sl = float(np.sum(a_look))
        if sl < 1e-15:
            a_look = a.copy()
        else:
            a_look = a_look / sl

        # Subgradient at lookahead point
        g, M = subgradient(a_look, temp=temp, top_k=max(10, n // 8))

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -3.0, 3.0)

        a_prev = a.copy()
        a = a * np.exp(-step)
        a = np.clip(a, 0.0, 1000.0)
        s = float(np.sum(a))
        if s < 1e-15:
            a = best.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        a /= s

        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_sc:
            best = a.copy()
            best_sc = float(sc)
            stuck = 0
            # Slightly increase lr on progress
            lr_cur = min(lr_cur * 1.005, 2.0)
        else:
            stuck += 1

        # Adaptive schedule
        if stuck > 30:
            temp = max(temp * 0.97, temp_min)
            lr_cur = max(lr_cur * 0.95, 0.05)
        if stuck > 150:
            # Restart with noise from best
            noise = np.exp(rng.normal(0, 0.15, n))
            a = best * noise
            a = np.clip(a, 0, None)
            s = a.sum()
            if s > 1e-15:
                a /= s
            else:
                a = best.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr_cur = lr
            temp = temp_init
            mu = 0.3
            stuck = 0

        # Periodically adjust momentum
        if it % 500 == 0:
            mu = 0.3 + 0.3 * float(rng.random())

        it += 1

    return best, best_sc


def simulated_annealing_optimize(a, max_time, seed, T0=0.3, T1=0.001):
    """Simulated annealing with diverse perturbation strategies."""
    rng = np.random.default_rng(seed)
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best = a.copy()
    best_sc = compute_score(best)
    cur = a.copy()
    cur_sc = best_sc

    t0 = time.monotonic()
    deadline = t0 + max_time
    elapsed_total = max_time if max_time > 0 else 1.0

    while time.monotonic() < deadline:
        frac = min((time.monotonic() - t0) / elapsed_total, 1.0)
        T = T0 * (T1 / T0) ** frac

        r = rng.random()
        if r < 0.4:
            # Block multiplicative perturbation
            w = int(rng.integers(max(1, n // 60), max(2, n // 4)))
            start = int(rng.integers(0, n - w + 1))
            cur[start:start + w] *= np.exp(rng.normal(0, 0.4, w))
        elif r < 0.6:
            # Gaussian noise on subset
            frac_sel = float(rng.uniform(0.1, 0.5))
            idx = rng.choice(n, size=int(n * frac_sel), replace=False)
            cur[idx] += rng.normal(0, 0.1 * np.mean(cur), len(idx))
            cur[idx] = np.clip(cur[idx], 0, None)
        elif r < 0.8:
            # Swap-based: move mass from high to low region
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                amt = min(cur[i], cur[j]) * float(rng.uniform(0.1, 0.5))
                cur[i] -= amt
                cur[j] += amt
        else:
            # Global: blend with random
            other = init_random(n, rng)
            alpha = float(rng.uniform(0.05, 0.3))
            cur = (1 - alpha) * cur + alpha * other

        cur = np.clip(cur, 0.0, None)
        s = float(np.sum(cur))
        if s < 1e-15:
            cur = best.copy()
            cur_sc = best_sc
            continue
        cur /= s

        sc = compute_score(cur)
        if not np.isfinite(sc):
            cur = best.copy()
            cur_sc = best_sc
            continue

        delta = sc - cur_sc
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_sc = sc
        if sc < best_sc:
            best = cur.copy()
            best_sc = float(sc)

    return best, best_sc


def crossover(parents, rng):
    """Generate a child by blending two parent sequences (same length)."""
    a = np.asarray(parents[0], dtype=np.float64)
    b = np.asarray(parents[1], dtype=np.float64)
    n = a.size
    alpha = float(rng.uniform(0.2, 0.8))
    child = alpha * a + (1 - alpha) * b
    child *= np.exp(rng.normal(0, 0.15, n))
    child = np.clip(child, 0, None)
    s = child.sum()
    if s < 1e-15:
        return a.copy()
    return child / s


# --- Main search ---

def optimize_single_length(n, budget, seed, parent=None):
    """Run all optimization phases for a single length."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget

    best = None
    best_sc = float('inf')

    # Generation pool for crossover
    pool = []

    # Phase 1: Generate diverse initializations
    init_time = min(5.0, 0.05 * budget)
    inits = []

    if parent is not None and len(parent) > 0:
        inits.append(init_from_parent(parent, n, rng))
    inits.append(init_random(n, rng, alpha=1.0))
    inits.append(init_random(n, rng, alpha=2.0))
    inits.append(init_two_bump(n, rng))
    inits.append(init_freq_flat(n, rng, n_iter=100))
    inits.append(init_uniform(n, rng))

    for i, init in enumerate(inits):
        if time.monotonic() > deadline:
            break
        sc = compute_score(init)
        if np.isfinite(sc) and sc < best_sc:
            best = init.copy()
            best_sc = float(sc)
        pool.append(init.copy())
        print(f"  init {i}: score={sc:.6f}", flush=True)

    # Phase 2: Mirror descent on top initializations
    phase2_time = 0.45 * budget
    for i, init in enumerate(inits[:4]):
        rem = deadline - time.monotonic()
        if rem < 5.0:
            break
        b = min(phase2_time / max(1, len(inits[:4])), rem - 2.0)
        if b < 2.0:
            b = max(1.0, 0.5 * rem)
        c, sc = mirror_descent_optimize(init, b, seed + 100 + i)
        if c is not None and np.isfinite(sc) and sc < best_sc:
            best = c.copy()
            best_sc = float(sc)
            print(f"  mirror_descent init{i}: score={sc:.6f}", flush=True)
        pool.append(c.copy() if c is not None else init.copy())

    # Phase 3: Simulated annealing on best
    if time.monotonic() < deadline - 5.0 and best is not None:
        rem = deadline - time.monotonic()
        sa_b = min(0.25 * budget, rem - 3.0)
        if sa_b > 3.0:
            c, sc = simulated_annealing_optimize(best, sa_b, seed + 500)
            if c is not None and np.isfinite(sc) and sc < best_sc:
                best = c.copy()
                best_sc = float(sc)
                print(f"  simulated_annealing: score={sc:.6f}", flush=True)
            pool.append(c.copy() if c is not None else best.copy())

    # Phase 4: Crossover + re-optimize
    if time.monotonic() < deadline - 10.0 and len(pool) >= 2:
        rem = deadline - time.monotonic()
        n_cross = min(3, int(rem // 20))
        for ci in range(n_cross):
            if time.monotonic() > deadline - 5.0:
                break
            i = int(rng.integers(0, len(pool)))
            j = int(rng.integers(0, len(pool)))
            if i == j:
                j = (j + 1) % len(pool)
            child = crossover([pool[i], pool[j]], rng)
            b = min(10.0, 0.5 * rem)
            if b < 2.0:
                b = max(1.0, rem * 0.3)
            c, sc = mirror_descent_optimize(child, b, seed + 900 + ci)
            if c is not None and np.isfinite(sc) and sc < best_sc:
                best = c.copy()
                best_sc = float(sc)
                print(f"  crossover {ci}: score={sc:.6f}", flush=True)
            pool.append(c.copy() if c is not None else child.copy())

    # Phase 5: Final polish
    if best is not None and time.monotonic() < deadline - 3.0:
        rem = deadline - time.monotonic()
        b = min(0.2 * budget, rem - 1.0)
        if b > 2.0:
            c, sc = mirror_descent_optimize(best, b, seed + 777, lr=0.2, temp_init=0.03, temp_min=0.002)
            if c is not None and np.isfinite(sc) and sc < best_sc:
                best = c.copy()
                best_sc = float(sc)
                print(f"  final_polish: score={sc:.6f}", flush=True)

    return best, best_sc


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main entry point: search for the best sequence within the time budget."""
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 1500.0
    deadline = start_time + budget - 15.0  # 15s safety margin

    # Get parent sequence
    parent = globals().get('height_sequence_1', None)
    parent_arr = None
    if parent is not None:
        try:
            parent_arr = np.asarray(parent, dtype=np.float64)
            parent_arr = np.clip(parent_arr, 0.0, 1000.0)
            if parent_arr.size == 0 or not np.isfinite(parent_arr).all():
                parent_arr = None
        except Exception:
            parent_arr = None

    # Initialize best from parent
    best_seq = None
    best_score = np.inf
    if parent_arr is not None:
        p = parent_arr.copy()
        s = float(np.sum(p))
        if s > 1e-15:
            p /= s
            best_seq = p
            best_score = compute_score(p)
            print(f"Parent: length={parent_arr.size}, score={best_score:.6f}", flush=True)

    # Length plan: (length, fraction_of_budget, use_parent)
    # Allocate more time to lengths most likely to improve
    if budget >= 1200:
        plan = [
            (256, 0.04, True),
            (512, 0.22, True),   # parent length - focus here
            (768, 0.10, True),
            (1024, 0.25, True),
            (1536, 0.15, True),
            (2048, 0.14, False),
            (4096, 0.10, False),
        ]
    elif budget >= 600:
        plan = [
            (256, 0.05, True),
            (512, 0.30, True),
            (1024, 0.35, True),
            (2048, 0.18, False),
        ]
    else:
        plan = [
            (256, 0.10, True),
            (512, 0.40, True),
            (1024, 0.42, True),
        ]

    seed_cur = int(seed)
    for n, frac, use_par in plan:
        rem = deadline - time.monotonic()
        if rem < 15.0:
            break
        b = min(frac * budget, rem - 10.0)
        if b < 10.0:
            b = max(5.0, 0.5 * rem)

        par = parent_arr if (use_par and parent_arr is not None) else None
        print(f"\n=== Length {n}, budget {b:.0f}s ===", flush=True)

        try:
            cand, sc = optimize_single_length(n, b, seed_cur, par)
            if cand is not None and np.isfinite(sc) and sc < best_score:
                best_seq = cand.copy()
                best_score = float(sc)
                print(f"  ** NEW BEST: length={n}, score={sc:.6f} **", flush=True)
            else:
                print(f"  best_here={sc:.6f}", flush=True)
        except Exception as exc:
            print(f"  Error at n={n}: {exc}", flush=True)

        seed_cur += 7919

    # Final: validate and return
    if best_seq is None:
        best_seq = np.ones(64) / 64.0
        best_score = compute_score(best_seq)

    # Final validation with official evaluator
    try:
        official = float(evaluate_sequence([float(x) for x in best_seq]))
        if np.isfinite(official):
            best_score = official
        else:
            # Fallback to parent
            if parent_arr is not None:
                p = np.asarray(parent_arr, dtype=np.float64)
                s = p.sum()
                if s > 1e-15:
                    p = p / s
                    off2 = float(evaluate_sequence([float(x) for x in p]))
                    if np.isfinite(off2):
                        best_seq = p
                        best_score = off2
    except Exception:
        pass

    print(f"\nFINAL: score={best_score:.6f}, length={len(best_seq)}", flush=True)
    return [float(x) for x in np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)]
