def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(a, lam):
    """
    Compute gradient of log(sum_k exp(lam*(a*a)_k)) w.r.t. a.
    Returns (gradient, b) where b = a*a.

    # Softmax weights with log-sum-exp trick

    w = np.exp(b_shifted)

def lp_mirror_descent(a0, n_steps=100, t_start=0.5, t_end=0.05):
    """LP mirror descent."""
    a = np.array(a0, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)
    
    for step in range(n_steps):
        b = fast_conv(a)
        M = float(b.max())
        g = lp_step(a, M)
        if g is None:
            break
        g = g / g.sum()
        t = t_start * (t_end / t_start) ** (step / max(1, n_steps - 1))
        a = (1 - t) * a + t * g
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
    
    return a


def resize_sequence(a, new_n):
    """Resize sequence to new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    new_a = np.maximum(np.interp(t_new, t_old, a), 0.0)
    s = new_a.sum()
    if s < 1e-10:
        return np.ones(new_n)
    return new_a / s


def make_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine U-shape
    inits.append(np.maximum(1.0 + 0.8 * np.cos(2 * np.pi * t), 0.01))
    
    # Parabolic U-shape
    inits.append(np.maximum((t - 0.5) ** 2 + 0.05, 0.01))
    
    # Arcline (singular at ends)
    inits.append(1.0 / (np.abs(t - 0.5) + 0.05))
    
    # Raised cosine (smooth U)
    inits.append(np.maximum(1.0 - 0.7 * np.cos(2 * np.pi * t), 0.01))
    
    # Quadratic with sharp ends
    inits.append(np.maximum((2 * t - 1) ** 4 + 0.03, 0.01))
    
    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    
    # Two Gaussian peaks at ends
    sigma = 0.08
    inits.append(np.exp(-((t - 0) ** 2) / (2 * sigma ** 2)) + 
                 np.exp(-((t - 1) ** 2) / (2 * sigma ** 2)) + 0.05)
    
    # Power-law U-shape
    inits.append(np.maximum(np.power(np.abs(t - 0.5) * 2, 0.5) + 0.1, 0.01))
    
    # Normalize all
    inits = [u / u.sum() for u in inits]
    return inits

$\nabla f = 2\lambda \sum_k w_k P_k(a)$

where $w_k$ are softmax weights and $P_k(a)$ is the reflected-shifted $a$.

The L2 norm of this gradient is:
$\|\nabla f\|_2 = 2\lambda \|\sum_k w_k P_k(a)\|_2 \leq 2\lambda \max_k \|P_k(a)\|_2 = 2\lambda \|a\|_2$

(since $\sum_k w_k = 1$ and $\|P_k(a)\|_2 = \|a\|_2$)

With $\|a\|_2 \leq \|a\|_1 = 1$ (since $\sum a = 1$ and $a \geq 0$), we have $\|\nabla f\|_2 \leq 2\lambda$.
