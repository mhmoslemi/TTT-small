import numpy as np
from scipy.optimize import linprog
import time

def compute_score(f):
    """Compute score = 2*n*max(conv)/sum^2."""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (s * s)

def compute_score_normalized(f):
    """Compute score for f with sum=1: 2*n*max(conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()

def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A

def solve_lp_direction(f, M):
    """Solve LP: maximize sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g normalized."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': 10})
    except Exception:
        return None
    
    if not res.success or res.x is None:
        return None
    
    g = np.clip(res.x, 0, None)
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s

def compute_score_norm(f):

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        A[k, lo:hi] = f[k - (hi - 1): k - lo + 1][::-1]
    return A

def solve_direction(f, M):
    """LP: max sum(g) s.t. A@f@g <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    A = build_conv_matrix(f)

        return None
    if not res.success:
        return None

    return g / s

import numpy as np
from scipy.optimize import linprog
import time


def compute_score_norm(f):
    """Score = 2*n*max(conv) for f with sum=1."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build A such that A@g = np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[k, j] = f[k - j]
    return A


def solve_direction(f, M):
    """LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g."""
    n = len(f)
    A = build_conv_matrix(f)
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    g = np.clip(res.x, 0, None)
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s


def line_search(f, g, n_points=20, center=0.5, spread=0.4):
    """Adaptive line search: geometric grid centered around 'center'."""
    best_t = center
    best_sc = np.inf
    lo = max(0.005, center - spread)
    hi = min(0.995, center + spread)
    ts = np.geomspace(lo, hi, n_points)
    for t in ts:
        fn = (1 - t) * f + t * g
        fn = np.clip(fn, 0, None)
        s = fn.sum()
        if s < 1e-12:
            continue
        fn = fn / s
        sc = compute_score_norm(fn)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    return best_t, best_sc


def coordinate_descent(f, n_sweeps=5, rng=None):
    """Coordinate descent: try modifying one element at a time."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = np.clip(f, 0, None)
    f /= f.sum()
    best_sc = compute_score_norm(f)

    for sweep in range(n_sweeps):
        changed = False
        order = np.arange(n)
        if rng is not None:
            order = rng.permutation(n)
        for i in order:
            # Try several multiplicative factors
            for frac in [0.0, 0.2, 0.5, 0.8, 1.2, 1.5, 2.0, 3.0]:
                f_new = f.copy()
                f_new[i] = f[i] * frac
                s = f_new.sum()
                if s < 1e-12:
                    continue
                f_new /= s
                sc = compute_score_norm(f_new)
                if sc < best_sc - 1e-13:
                    f = f_new
                    best_sc = sc
                    changed = True
                    break
        if not changed:
            break
    return f, best_sc


def structured_perturb(f, rng, scale=0.05):
    """Apply a structured perturbation (mix of sinusoidal and random)."""
    n = len(f)
    x = np.arange(n, dtype=np.float64)
    # Random sinusoidal component
    freq = rng.uniform(1, 5)
    phase = rng.uniform(0, 2 * np.pi)
    sin_comp = np.sin(2 * np.pi * freq * x / n + phase) * scale
    # Random smooth component (low-pass filtered noise)
    noise = np.cumsum(rng.normal(0, scale * 0.3, n))
    noise = noise - noise.mean()
    # Combine
    pert = sin_comp + noise
    f_new = f * np.exp(pert)
    f_new = np.clip(f_new, 0, None)
    s = f_new.sum()
    if s < 1e-10:
        return None
    return f_new / s


def optimize_sequence(f0, time_budget, rng, verbose=True):
    """Main optimization: LP direction + line search + CD + structured escape."""
    n = len(f0)
    f = f0.astype(np.float64).copy()
    f = np.clip(f, 0, None)
    f /= f.sum()

    best_f = f.copy()
    best_sc = compute_score_norm(f)
    start = time.monotonic()
    it = 0
    stall = 0
    center_t = 0.5
    spread = 0.4

    if verbose:
        print(f"    Start: n={n}, score={best_sc:.8f}")

    while time.monotonic() - start < time_budget:
        rem = time_budget - (time.monotonic() - start)
        if rem < 1.5:
            break

        conv = np.convolve(f, f)
        M = conv.max()
        g = solve_direction(f, M)
        if g is None:
            break

        # Line search
        best_t, step_sc = line_search(f, g, n_points=20, center=center_t, spread=spread)

        if step_sc < best_sc - 1e-13:
            fn = (1 - best_t) * f + best_t * g
            fn = np.clip(fn, 0, None)
            fn /= fn.sum()
            f = fn
            best_f = f.copy()
            best_sc = step_sc
            # Adapt center toward successful step
            center_t = 0.7 * center_t + 0.3 * best_t
            spread = max(0.1, spread * 0.95)
            stall = 0
            it += 1
            if verbose and it % 10 == 0:
                el = time.monotonic() - start
                print(f"    it={it}, sc={best_sc:.8f}, t={best_t:.4f}, el={el:.1f}s")
        else:
            stall += 1
            # Try CD
            f_cd, sc_cd = coordinate_descent(f, n_sweeps=2, rng=rng)
            if sc_cd < best_sc - 1e-13:
                f = f_cd
                best_f = f.copy()
                best_sc = sc_cd
                stall = 0
                if verbose:
                    print(f"    CD: {best_sc:.8f}")
            elif stall > 15:
                # Structured perturbation to escape
                escaped = False
                for attempt in range(5):
                    f_p = structured_perturb(f, rng, scale=0.03 + 0.02 * attempt)
                    if f_p is None:
                        continue
                    sc_p = compute_score_norm(f_p)
                    if sc_p < best_sc:
                        f = f_p
                        best_f = f.copy()
                        best_sc = sc_p
                        stall = 0
                        center_t = 0.5
                        spread = 0.4
                        escaped = True
                        if verbose:
                            print(f"    Escaped: {best_sc:.8f}")
                        break
                if not escaped:
                    # Pure random kick
                    f_r = f + rng.normal(0, 0.01, n)
                    f_r = np.clip(f_r, 0, None)
                    s = f_r.sum()
                    if s > 1e-10:
                        f_r /= s
                        sc_r = compute_score_norm(f_r)
                        if sc_r < best_sc:
                            f = f_r
                            best_f = f.copy()
                            best_sc = sc_r
                            stall = 0
                    if stall > 40:
                        break

    # Final CD polish
    f_cd, sc_cd = coordinate_descent(best_f, n_sweeps=15, rng=rng)
    if sc_cd < best_sc:
        best_f = f_cd
        best_sc = sc_cd

    return best_f, best_sc

        for freq in [2.0, 4.0, 8.0]:
            mod = parent * (1 + 0.03 * np.sin(2 * np.pi * freq * x / n_p))
            mod = np.clip(mod, 0, None)
            if mod.sum() > 1e-10:
                candidates.append((f"hsin{freq}", mod, 1.5))

    # Various lengths and shapes
    for n in [256, 384, 512]:
        x = np.arange(n, dtype=np.float64)
        # Uniform + noise
        u = np.ones(n) * (1 + 0.02 * rng.normal(0, 1, n))
        candidates.append((f"u{n}", np.clip(u, 0.05, None), 1.2))
        # Triangle
        tri = np.minimum(x + 1, n - x).astype(np.float64)
        candidates.append((f"T{n}", tri, 0.8))
        # Gaussian
        g = np.exp(-0.5 * ((x - n / 2) / (n / 5.0)) ** 2)
        candidates.append((f"G{n}", g, 0.8))
        # W-shape
        w = (np.exp(-0.5 * ((x - n * 0.2) / (n * 0.07)) ** 2) +

        candidates.append((f"W{n}", w, 0.8))
        # Cosine bump
        c = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        candidates.append((f"C{n}", c, 0.8))

    # Time allocation
    total_w = sum(w for _, _, w in candidates)
    total_time = min(budget_s - 50, deadline - time.monotonic())
    print(f"Candidates: {len(candidates)}, total_time={total_time:.0f}s")

    for name, f0, w in candidates:
