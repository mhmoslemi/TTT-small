def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v

    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses Differential Evolution for global search, Gerchberg-Saxton spectral
flattening for local refinement, and projected subgradient for final polish.
"""

import numpy as np
import time

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_mask = u * np.arange(1, n + 1) > cv
    if not np.any(rho_mask):
        return np.ones(n) / n
    rho = np.max(np.where(rho_mask)[0])
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_score(a):
    """Compute score using FFT convolution."""
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    conv[conv < 0] = 0.0
    max_c = np.max(conv)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    return 2.0 * n * max_c / (s * s)

def gs_flatten(a, n_iter=200, rng=None, alpha=0.5):
    """Gerchberg-Saxton style iteration to flatten the magnitude spectrum."""
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        target = np.mean(mag)
        # Relaxed magnitude update
        new_mag = (1.0 - alpha) * mag + alpha * target
        # Avoid division by zero
        safe_mag = np.maximum(mag, 1e-15)
        A_new = new_mag * A / safe_mag
        # Inverse FFT
        a = np.fft.irfft(A_new, N)[:n]
        # Project to non-negative
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    
    return a

def subgradient_polish(a, deadline, rng, max_iters=100000):
    """Projected subgradient descent polish."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_score = fast_score(a)
    
    lr = 0.5 / n
    lr_min = 1e-6
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        N = 1 << (2 * n - 2).bit_length()
        fa = np.fft.rfft(a, N)
        conv = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        conv[conv < 0] = 0.0
        peak = np.max(conv)
        cur_score = 2.0 * n * peak
        # (sum is 1, so score = 2n * max_conv)
        
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        else:
            stall += 1
        
        if stall > 200:
            lr = max(lr * 0.8, lr_min)
            stall = 0
            if lr < lr_min * 3:
                a = project_simplex(a + rng.normal(0, 0.05 / np.sqrt(n), n))
                continue
        
        # Find peak indices
        thresh = peak * (1.0 - 1e-4)
        peak_idx = np.where(conv >= thresh)[0]
        if len(peak_idx) > 30:
            top = np.argsort(conv[peak_idx])[::-1][:10]
            peak_idx = peak_idx[top]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in peak_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        grad /= max(1, len(peak_idx))
        
        a = project_simplex(a - lr * grad)
    
    return best_a

    best_f = fitness[best_idx]

"""
Hybrid Differential Evolution + Gerchberg-Saxton spectral flattening + 
subgradient polishing to minimize 2*n*max(conv(a,a))/sum(a)^2 
for non-negative sequences.
"""

import numpy as np
import time


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_mask = u * np.arange(1, n + 1) > cv
    if not np.any(rho_mask):
        return np.ones(n) / n
    rho = np.max(np.where(rho_mask)[0])
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 using FFT."""
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    conv[conv < 0] = 0.0
    max_c = np.max(conv)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    return 2.0 * n * max_c / (s * s)


def gs_flatten(a, n_iter=100, alpha=0.5, rng=None):
    """Gerchberg-Saxton style iteration to flatten magnitude spectrum."""
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        target = np.mean(mag)
        new_mag = (1.0 - alpha) * mag + alpha * target
        safe_mag = np.maximum(mag, 1e-15)
        A_new = new_mag * A / safe_mag
        a = np.fft.irfft(A_new, N)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s

    return a


def subgradient_polish(a, deadline, rng, max_iters=200000):
    """Projected subgradient descent to minimize max convolution peak."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_score = fast_score(a)
    N = 1 << (2 * n - 2).bit_length()

    lr = 0.5 / n
    lr_min = 1e-7
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        fa = np.fft.rfft(a, N)
        conv = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        conv[conv < 0] = 0.0
        peak = np.max(conv)
        cur_score = 2.0 * n * peak

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 300:
            lr = max(lr * 0.7, lr_min)
            stall = 0
            if lr < lr_min * 5:
                a = project_simplex(a + rng.normal(0, 0.03 / np.sqrt(n), n))
                continue

        thresh = peak * (1.0 - 1e-5)
        peak_idx = np.where(conv >= thresh)[0]
        if len(peak_idx) > 30:
            top = np.argsort(conv[peak_idx])[::-1][:15]
            peak_idx = peak_idx[top]

        grad = np.zeros(n)
        for k in peak_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        grad /= max(1, len(peak_idx))

        a = project_simplex(a - lr * grad)

    return best_a


def de_search(n, pop_size, deadline, rng, initial_members=None, use_gs=True):
    """
    Differential evolution on the simplex.
    Returns (best_x, best_score).
    """
    pop = np.zeros((pop_size, n))

    start_idx = 0
    if initial_members is not None:
        for m in initial_members:
            if start_idx < pop_size:
                pop[start_idx] = project_simplex(np.asarray(m, dtype=np.float64))
                start_idx += 1

    while start_idx < pop_size:
        pop[start_idx] = project_simplex(rng.random(n))
        start_idx += 1

    fitness = np.array([fast_score(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(fitness))
    best_x = pop[best_idx].copy()
    best_f = fitness[best_idx]

    print(f"    DE init: best={best_f:.6f}, pop_mean={np.mean(fitness):.6f}")

    F = 0.7
    CR = 0.5
    gen = 0
    no_improve = 0
    last_best_time = time.time()

    while time.time() < deadline:
        gen += 1
        gen_improved = False

        for i in range(pop_size):
            if time.time() > deadline:
                break

            # Select 3 distinct indices different from i
            r1 = i
            r2 = i
            r3 = i
            while r2 == i or r2 == r1:

            while r3 == i or r3 == r1 or r3 == r2:

            v = pop[r1] + F * (pop[r2] - pop[r3])

                cross[j] = True
            else:
                trial[cross] = v[cross]
            trial = project_simplex(trial)

            f_trial = fast_score(trial)
            if f_trial < fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                gen_improved = True
                if f_trial < best_f:
                    best_f = f_trial
                    best_x = trial.copy()
                    last_best_time = time.time()

            # Periodic GS injection for diversity in the local region

                f_gs = fast_score(gs_cand)
                if f_gs < fitness[i]:
                    pop[i] = gs_cand
                    fitness[i] = f_gs
                    gen_improved = True
                    if f_gs < best_f:
                        best_f = f_gs
                        best_x = gs_cand.copy()
                        last_best_time = time.time()

        if not gen_improved:
            no_improve += 1
        else:
            no_improve = 0

        # Adaptive parameters
        if no_improve > 30:
            F = max(0.35, F * 0.95)
            CR = min(0.85, CR + 0.02)
            no_improve = 0

            for _ in range(n_reset):

                if rng.random() < 0.5:
                    pop[idx] = project_simplex(rng.random(n))
                else:
                    base = pop[int(np.argmin(fitness))]
                    pop[idx] = project_simplex(base + rng.normal(0, 0.2 / np.sqrt(n), n))
                fitness[idx] = fast_score(pop[idx])
                if fitness[idx] < best_f:
                    best_f = fitness[idx]
                    best_x = pop[idx].copy()
                    last_best_time = time.time()
            print(f"    Gen {gen}: diversity injection, best={best_f:.6f}, time_left={deadline - time.time():.0f}s")

        if gen % 100 == 0:
            print(f"    Gen {gen}: best={best_f:.6f}, F={F:.3f}, CR={CR:.3f}, time_left={deadline - time.time():.0f}s")

    return best_x, best_f

    members = members[:pop_size]
    while len(members) < pop_size:
        members.append(rng.random(n))

    a_opt, score = de_search(n, pop_size, deadline, rng,
                             initial_members=members, use_gs=True)
    return a_opt, score


def propose_candidate(budget_s=1500):
    """
    Multi-phase hybrid search:
    Phase 1: Quick exploration across different lengths.
    Phase 2: Intensive DE search on the best length.
    Phase 3: Final polishing with subgradient + GS.
    """
    overall_deadline = time.time() + budget_s - 20.0
    rng = np.random.default_rng(777)

    best_score = float('inf')
    best_seq = None

    # Try to load parent sequence
    parent_seq = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        s = parent.sum()
        if s > 0.01:
            parent_seq = parent / s
            print(f"Loaded parent sequence: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.6f}")
    except Exception:
        pass

    # Phase 1: Quick exploration
    print("=" * 60)
    print("Phase 1: Quick exploration of different lengths")
    print("=" * 60)
    phase1_deadline = time.time() + 80

    test_lengths = [384, 512, 640, 768]
    phase1_results = []

    for n in test_lengths:
        if time.time() > phase1_deadline:
            break
        alloc = min(25.0, phase1_deadline - time.time())
        if alloc < 8:
            break
        d = time.time() + alloc
        print(f"\n  Trying n={n}...")
        a_opt, score = try_length(n, d, rng, pop_size=32, parent_seq=parent_seq)
        ev_score = evaluate_sequence(a_opt.tolist())
        print(f"  n={n}: score={ev_score:.6f}")
        phase1_results.append((n, a_opt, ev_score))
        if np.isfinite(ev_score) and ev_score < best_score:
            best_score = ev_score
            best_seq = a_opt.copy()
            print(f"  >>> New best: {best_score:.6f} (n={n})")

    # Fallback
    if best_seq is None:
        n = 512
        best_seq = gs_flatten(rng.random(n), n_iter=100, alpha=0.5, rng=rng)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"  Fallback: n={n}, score={best_score:.6f}")

    # Phase 2: Main intensive DE
    print("\n" + "=" * 60)
    print(f"Phase 2: Intensive DE (best n={len(best_seq)}, score={best_score:.6f})")
    print("=" * 60)
    n_best = len(best_seq)
    phase2_deadline = min(overall_deadline - 80, time.time() + 1000)

    if phase2_deadline > time.time() + 30:
        # Build rich initial population
        members = [best_seq.copy()]

        # GS variants of best
        for alpha in [0.3, 0.5, 0.7, 0.9]:
            members.append(gs_flatten(best_seq, n_iter=60, alpha=alpha, rng=rng))

        # Perturbed best + GS
        for _ in range(4):
            pert = best_seq + rng.normal(0, 0.15 / np.sqrt(n_best), n_best)
            members.append(gs_flatten(pert, n_iter=40, alpha=0.5, rng=rng))

        # Random
        for _ in range(6):
            members.append(rng.random(n_best))

        # Structured
        x = np.linspace(0.0, 1.0, n_best)
        for freq in [1, 2, 3, 5]:
            members.append(0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x)) + 0.01)

        # GS from random
        for _ in range(4):
            members.append(gs_flatten(rng.random(n_best), n_iter=50, alpha=0.6, rng=rng))

        # Parent
        if parent_seq is not None:
            if len(parent_seq) >= n_best:
                members.append(parent_seq[:n_best].copy())
            else:
                pad = np.zeros(n_best)
                pad[:len(parent_seq)] = parent_seq
                s = pad.sum()
                if s > 0.01:
                    members.append(pad / s)

        pop_size = 64
        while len(members) < pop_size:
            members.append(rng.random(n_best))
        members = members[:pop_size]

        a_opt, score = de_search(n_best, pop_size, phase2_deadline, rng,
                                 initial_members=members, use_gs=True)
        ev_score = evaluate_sequence(a_opt.tolist())
        print(f"\n  Phase 2 result: score={ev_score:.6f}")
        if np.isfinite(ev_score) and ev_score < best_score:
            best_score = ev_score
            best_seq = a_opt.copy()
            print(f"  >>> New best: {best_score:.6f} (n={n_best})")

    # Try alternative length if time allows
    remaining = overall_deadline - time.time()
    if remaining > 250:
        n_alt = n_best + 128
        alloc = min(180, remaining - 100)
        if alloc > 40:
            d = time.time() + alloc
            print(f"\n  Trying alternative n={n_alt}...")
            try:
                a_alt, score_alt = try_length(n_alt, d, rng, pop_size=40, parent_seq=parent_seq)
                ev_alt = evaluate_sequence(a_alt.tolist())
                print(f"  n={n_alt}: score={ev_alt:.6f}")
                if np.isfinite(ev_alt) and ev_alt < best_score:
                    best_score = ev_alt
                    best_seq = a_alt.copy()
                    print(f"  >>> New best: {best_score:.6f} (n={n_alt})")
            except Exception as e:
                print(f"  Alt length error: {e}")

    # Phase 3: Final polishing
    print("\n" + "=" * 60)
    print(f"Phase 3: Final polishing (score={best_score:.6f}, n={len(best_seq)})")
    print("=" * 60)
    n_p = len(best_seq)
    polish_deadline = overall_deadline

    if polish_deadline > time.time() + 15:
        # Subgradient polish
        sub_deadline = min(polish_deadline - 15, time.time() + (polish_deadline - time.time()) * 0.5)
        a_sub = subgradient_polish(best_seq, sub_deadline, rng)
        s_sub = evaluate_sequence(a_sub.tolist())
        print(f"  After subgradient: {s_sub:.6f}")
        if np.isfinite(s_sub) and s_sub < best_score:
            best_score = s_sub
            best_seq = a_sub
            print(f"  >>> New best: {best_score:.6f}")

    if polish_deadline > time.time() + 10:
        # GS polish
        a_gs = gs_flatten(best_seq, n_iter=300, alpha=0.3, rng=rng)
        s_gs = evaluate_sequence(a_gs.tolist())
        print(f"  After GS: {s_gs:.6f}")
        if np.isfinite(s_gs) and s_gs < best_score:
            best_score = s_gs
            best_seq = a_gs
            print(f"  >>> New best: {best_score:.6f}")

    if polish_deadline > time.time() + 10:
        # Combined: GS then subgradient
        a_c = gs_flatten(best_seq, n_iter=150, alpha=0.4, rng=rng)
        a_c = subgradient_polish(a_c, polish_deadline, rng)
        s_c = evaluate_sequence(a_c.tolist())
        print(f"  After GS+subgrad: {s_c:.6f}")
        if np.isfinite(s_c) and s_c < best_score:
            best_score = s_c
            best_seq = a_c
            print(f"  >>> New best: {best_score:.6f}")

    # One more DE polish on the best
    if polish_deadline > time.time() + 30:
        small_pop = 24
        members = [best_seq.copy()]
        for _ in range(8):
            pert = best_seq + rng.normal(0, 0.05 / np.sqrt(n_p), n_p)
            members.append(gs_flatten(pert, n_iter=30, alpha=0.5, rng=rng))
        while len(members) < small_pop:
            members.append(best_seq + rng.normal(0, 0.1 / np.sqrt(n_p), n_p))
        members = members[:small_pop]
        a_final, s_final = de_search(n_p, small_pop, polish_deadline, rng,
                                      initial_members=members, use_gs=True)
        s_final = evaluate_sequence(a_final.tolist())
        print(f"  After final DE: {s_final:.6f}")
        if np.isfinite(s_final) and s_final < best_score:
            best_score = s_final
            best_seq = a_final
            print(f"  >>> New best: {best_score:.6f}")

    # Construct final result
    result = [float(max(0.0, x)) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.6f}, length={len(result)}")
    print(f"{'=' * 60}")
    return result
