Improved LP-based iterative search for nonnegative step functions minimizing
2*n*max(convolve(f,f))/(sum(f))^2.

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def cosine_window(n, alpha=1.0):
    """Raised cosine window."""
    j = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(alpha * 2.0 * np.pi * j / (n - 1)))


def gaussian_window(n, sigma_frac=0.3):
    """Gaussian window."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((j - center) / sigma) ** 2)


def triangular_window(n):
    """Triangular (Bartlett) window."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(j - center) / (center + 1e-10))


def power_window(n, p=2.0):
    """Power-law window."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = 1.0 - np.abs(j - center) / (center + 1e-10)
    return x ** p


def make_conv_template(n):
    """Precompute the CSR matrix structure for convolve(f, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(template, f, n, M):
    """Solve the LP direction problem for the given f and M."""
    A_template, edge_offsets, c_obj, bounds, m = template
    lp_scale = float(n)

    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale

    b = np.full(m, M * lp_scale * lp_scale, dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")


def peak_targeted_perturbation(f, rng, num_peaks=5, strength=0.3):
    """Perturb f by reducing values near the peaks of f*f."""
    n = f.size
    conv = np.convolve(f, f)
    # Find the peak locations
    top_k = np.argsort(conv)[-num_peaks:]
    # For each peak at position k, the contributing indices j satisfy 0 <= j < n and 0 <= k-j < n
    # We reduce f at indices that contribute to peaks
    f_new = f.copy()
    for k in top_k:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # Reduce f[j] for j in [js, je]
        f_new[js:je+1] *= (1.0 - strength * 0.1)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-300:
        f_new /= s
    return f_new


def mix_noisy(base, rng, eps):
    """Mix a base vector with a random Dirichlet vector."""
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    if eps <= 1e-12:
        return f
    try:
        mix = rng.dirichlet(np.ones(n, dtype=np.float64))
    except Exception:
        mix = np.ones(n, dtype=np.float64) / float(n)
    if eps >= 1.0:
        f = mix
    else:
        f = (1.0 - eps) * f + eps * mix
    return normalize_sequence(f)


def interpolate_sequence(x, n):
    """Resample a nonnegative sequence to length n and normalize it."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, x)
    return normalize_sequence(out)

def structured_perturbation(f, rng, K=10, strength=0.1):
    """Perturb f by reducing contributions to the top K peaks."""
    n = f.size
    conv = np.convolve(f, f)
    top_k = np.argsort(conv)[-K:]
    
    # Count how many times each index j appears in peak-contributing pairs
    blame = np.zeros(n)
    for k in top_k:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            blame[j] += conv[k] * f[j] * f[k-j]  # weighted blame
    
    # Reduce f at high-blame positions
    max_blame = blame.max() if blame.max() > 0 else 1
    f_new = f * (1.0 - strength * blame / max_blame)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-300:
        f_new /= s
    return f_new

def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def ramp_window(n, p=1.0):
    """Power ramp: f_j = (j+1)^p, normalized."""
    j = np.arange(n, dtype=np.float64) + 1.0
    return j ** p


def exponential_window(n, alpha=2.0):
    """Exponential ramp: f_j = exp(alpha * j / n), normalized."""
    j = np.arange(n, dtype=np.float64)
    return np.exp(alpha * j / (n - 1))


def sine_half(n):
    """Half sine: f_j = sin(pi*(j+1)/(n+1))."""
    j = np.arange(n, dtype=np.float64)
    return np.sin(np.pi * (j + 1.0) / (n + 1.0))

    """Cosine taper: f_j = 0.5*(1+cos(2*pi*beta*j/(n-1)))."""
    j = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * beta * j / (n - 1.0)))


def asymmetric_gaussian(n, mu_frac=0.3, sigma_frac=0.2):
    """Asymmetric Gaussian centered at mu_frac*n."""
    j = np.arange(n, dtype=np.float64)
    mu = mu_frac * n
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((j - mu) / sigma) ** 2)


def make_conv_template(n):
    """Precompute CSR matrix structure for convolve(f, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(template, f, n, M):
    """Solve the LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    A_template, edge_offsets, c_obj, bounds, m = template
    lp_scale = float(n)

    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale

    b = np.full(m, M * lp_scale * lp_scale, dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")


def subgradient_step(f, avg_step, active_set=None, top_k=5):
    """
    Perform one subgradient descent step on max(f*f).
    Returns updated f (normalized to sum 1, nonnegative).
    """
    n = f.size
    conv = np.convolve(f, f)

    if active_set is None:
        # Use top-k peaks as active set
        top_indices = np.argsort(conv)[-top_k:]
    else:
        top_indices = active_set

    # Compute subgradient: grad_i = sum_{k in S} 2*f_{k-i} (with edge handling)
    grad = np.zeros(n, dtype=np.float64)
    for k in top_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            # Contribution to grad[j] from peak at k
            ki = k - j
            if ki == j:
                grad[j] += f[ki]
            else:
                grad[j] += 2.0 * f[ki]

    # Normalize gradient
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-30:
        grad /= gnorm

    # Take step
    alpha = avg_step * n
    f_new = f - alpha * grad
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s <= 1e-300:
        return f.copy()
    f_new /= s
    return f_new


def structured_perturbation(f, rng, K=10, strength=0.15):
    """Reduce f at indices that contribute most to peak values of f*f."""
    n = f.size
    conv = np.convolve(f, f)
    top_k = np.argsort(conv)[-K:]

    blame = np.zeros(n, dtype=np.float64)
    for k in top_k:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            ki = k - j
            blame[j] += f[j] * f[ki]

    max_blame = blame.max()
    if max_blame <= 0:
        return f.copy()

    f_new = f * (1.0 - strength * blame / max_blame)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-300:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def mix_noisy(base, rng, eps):
    """Mix a base vector with random noise."""
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    if eps <= 1e-12:
        return f
    noise = rng.uniform(0.0, 1.0, size=n)
    f = (1.0 - eps) * f + eps * noise
    return normalize_sequence(f)


def interpolate_sequence(x, n):
    """Resample a nonnegative sequence to length n and normalize."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, x)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s <= 0:
        return None
    out /= s
    return out


def optimize_lp(n, f_init, time_budget, rng, template, aggressive=False):
    """
    Run LP-based iteration from f_init. Returns (best_f, best_score).
    """
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n, dtype=np.float64) / float(n)

    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_score = best_score

    deadline = time.monotonic() + max(0.0, float(time_budget))
    stall = 0
    iteration = 0
    sg_iter_count = 0
    sg_step_size = 0.3

    while time.monotonic() < deadline:
        iteration += 1
        t0 = time.monotonic()

        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        if not np.isfinite(M) or M <= 0.0:
            break

        # Try LP direction
        try:
            res = solve_direction_lp(template, f, n, M)
        except Exception:
            res = None

        lp_success = False
        if res is not None:
            x = getattr(res, "x", None)
            if x is not None:
                x = np.asarray(x, dtype=np.float64).ravel()
                if x.size == n:
                    x = np.where(np.isfinite(x), x, 0.0)
                    x = np.maximum(x, 0.0)
                    s_star = float(x.sum())
                    if s_star > 1e-300:
                        s_ratio = s_star / float(n)
                        if s_ratio > 1.0 + 1e-8:
                            g = x / s_star
                            r = np.convolve(f, g)
                            r_max = float(np.max(r))
                            if np.isfinite(r_max) and r_max > 0.0:
                                if r_max < M * (1.0 - 1e-10):
                                    lp_success = True
                                    S_eff = M / r_max
                                    G = float(np.max(np.convolve(g, g)))
                                    if not np.isfinite(G):
                                        lp_success = False

        if not lp_success:
            # Use subgradient or perturbation to escape
            stall += 1
            sg_iter_count += 1

            if stall <= 5:
                # Subgradient steps
                f = subgradient_step(f, sg_step_size, top_k=8)
                sc = score_normalized(f, n)
                if np.isfinite(sc) and sc < best_score:
                    best_f = f.copy()
                    best_score = sc
                    stall = 0
                    sg_iter_count = 0
            elif stall <= 10:
                # Structured perturbation
                f = structured_perturbation(best_f, rng, K=8, strength=0.2)
            elif stall <= 18:
                # More aggressive noise
                eps = 0.3 + 0.1 * (stall - 10)
                nf = mix_noisy(best_f, rng, min(eps, 0.8))
                f = nf if nf is not None else best_f.copy()
            else:
                # Big random restart
                if stall > 25:
                    break
                nf = mix_noisy(best_f, rng, 0.9)
                f = nf if nf is not None else best_f.copy()

            if iteration % 15 == 0:
                print(f"      n={n} iter={iteration} stuck, stall={stall}, best={best_score:.6f}", flush=True)
            continue

        # LP direction found - do line search
        G = float(np.max(np.convolve(g, g)))
        a_coef = 2.0 * (r_max - M)
        b_coef = G + M - 2.0 * r_max
        if b_coef > 1e-14:
            t_max = -a_coef / b_coef
        else:
            t_max = 1.0

        if not np.isfinite(t_max) or t_max <= 0.0:
            t_max = 1e-8
        base_t = min(1.0, (0.9 if aggressive else 0.8) * t_max)
        if base_t < 1e-12:
            base_t = 1e-12

        # Line search with multiple candidates
        if aggressive:
            raw_ts = [base_t, 0.9*base_t, 0.7*base_t, 0.5*base_t, 0.3*base_t,
                      0.15*base_t, 0.05*base_t, 0.02*base_t, 0.005*base_t]
        else:
            raw_ts = [base_t, 0.7*base_t, 0.5*base_t, 0.3*base_t, 0.1*base_t,
                      0.03*base_t, 0.01*base_t]

        ts = []
        last_tv = 2.0
        for tv in sorted((float(v) for v in raw_ts), reverse=True):
            if tv < 1e-14:
                continue
            if tv > 1.0:
                tv = 1.0
            if last_tv - tv < 1e-12:
                continue
            last_tv = tv
            ts.append(tv)
            if len(ts) >= 12:
                break

        best_cand_f = None
        best_cand_score = best_score
        best_t_used = None

        for tv in ts:
            f_new = (1.0 - tv) * f + tv * g
            f_new = np.maximum(f_new, 0.0)
            ssum = float(f_new.sum())
            if ssum <= 1e-300:
                continue
            f_new /= ssum
            sc = score_normalized(f_new, n)
            if np.isfinite(sc) and sc < best_cand_score:
                best_cand_score = sc
                best_cand_f = f_new
                best_t_used = tv

        if best_cand_f is not None and best_cand_score < best_score:
            f = best_cand_f
            best_f = f.copy()
            best_score = best_cand_score
            stall = 0
            sg_iter_count = 0
            if iteration <= 3 or iteration % 25 == 0 or best_score < start_score - 0.02:
                print(
                    f"      n={n} iter={iteration} score={best_score:.6f} "
                    f"S_eff={S_eff:.4f} t={best_t_used:.6f} "
                    f"dt={time.monotonic()-t0:.3f}s",
                    flush=True,
                )
        else:
            # LP direction didn't help; use subgradient
            stall += 1
            f = subgradient_step(f, sg_step_size, top_k=10)
            sc = score_normalized(f, n)
            if np.isfinite(sc) and sc < best_score:
                best_f = f.copy()
                best_score = sc
                stall = 0
            if stall > 25:
                break

    return best_f, float(best_score)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a low-scoring nonnegative sequence using LP-based iteration
    with asymmetric starts, subgradient escape, and multi-scale refinement.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    total_budget = max(1.0, float(budget_s))
    reserve = 12.0 if total_budget > 60.0 else 3.0
    start_time = time.monotonic()
    deadline = start_time + max(1.0, total_budget - reserve)

    best_f = None
    best_score = float("inf")
    given_f = None

    # Load given sequence
    if total_budget >= 30.0:
        try:
            raw = np.asarray(height_sequence_1, dtype=np.float64)
            given_f = normalize_sequence(raw)
            if given_f is not None:
                best_f = given_f.copy()
                best_score = score_normalized(given_f, given_f.size)
                print(f"given sequence: n={given_f.size}, score={best_score:.6f}", flush=True)
        except Exception:
            given_f = None

    if best_f is None:
        # Fallback: use a ramp
        n_fb = 1024
        best_f = normalize_sequence(ramp_window(n_fb, p=1.0))
        best_score = score_normalized(best_f, n_fb)
        print(f"fallback ramp: n={n_fb}, score={best_score:.6f}", flush=True)

    # Plan: focus on large n where the given solution lives
    # First do quick improvement at the given n, then try larger
    giv_n = best_f.size if best_f is not None else 1024

    # Phase 1: Quick improvement at given size
    plan = []
    if giv_n <= 1100:
        plan.append((giv_n, 80))
    else:
        plan.append((1024, 60))

    # Phase 2: Larger sizes
    plan.append((1536, 100))
    plan.append((2048, 400))
    plan.append((3072, 350))

    # Adjust plan based on remaining time
    for n, alloc in plan:
        rem = deadline - time.monotonic()
        if rem < 15.0:
            break
        alloc = min(alloc, rem - 8.0)
        if alloc <= 8.0:
            continue

        print(f"\n== n={n}, alloc={alloc:.1f}s, current_best={best_score:.6f}", flush=True)

        try:
            template = make_conv_template(n)
        except MemoryError:
            print(f"  memory error for n={n}; skipping", flush=True)
            continue
        except Exception as e:
            print(f"  template build failed for n={n}: {e}", flush=True)
            continue

        # Build diverse starting points
        starts = []

        # 1. Upsampled best
        if best_f is not None:
            up = interpolate_sequence(best_f, n)
            if up is not None:
                starts.append(up)

        # 2. Given sequence upsampled (if different from best)
        if given_f is not None and (best_f is None or np.linalg.norm(best_f - given_f) > 1e-4):
            g_up = interpolate_sequence(given_f, n)
            if g_up is not None:
                starts.append(g_up)

        # 3. Linear ramp
        rw = normalize_sequence(ramp_window(n, p=1.0))
        if rw is not None:
            starts.append(rw)

        # 4. Exponential ramp
        ew = normalize_sequence(exponential_window(n, alpha=1.5))
        if ew is not None:
            starts.append(ew)

        # 5. Power ramp (p=2)
        pw = normalize_sequence(ramp_window(n, p=2.0))
        if pw is not None:
            starts.append(pw)

        # 6. Sine half-wave
        sw = normalize_sequence(sine_half(n))
        if sw is not None:
            starts.append(sw)

        # 7. Asymmetric Gaussian
        ag = normalize_sequence(asymmetric_gaussian(n, mu_frac=0.35, sigma_frac=0.25))
        if ag is not None:
            starts.append(ag)

        # 8. Noisy upsampled best
        if best_f is not None:
            up2 = interpolate_sequence(best_f, n)
            if up2 is not None:
                noisy = mix_noisy(up2, rng, 0.1)
                if noisy is not None:
                    starts.append(noisy)

        # Clean starts: remove None and duplicates
        clean_starts = []
        for s in starts:
            if s is not None and s.size == n:
                is_dup = False
                for cs in clean_starts:
                    if np.linalg.norm(s - cs) < 1e-5:
                        is_dup = True
                        break
                if not is_dup:
                    clean_starts.append(s)

        if not clean_starts:
            clean_starts = [np.ones(n, dtype=np.float64) / float(n)]

        # Score all starts and sort
        scored_starts = []
        for s in clean_starts:
            sc = score_normalized(s, n)
            scored_starts.append((sc, s))
        scored_starts.sort(key=lambda x: x[0])

        # Print initial scores
        print(f"  {len(scored_starts)} starts, best init score={scored_starts[0][0]:.6f}", flush=True)

        # Time allocation: weighted by quality
        m = len(scored_starts)
        # Give most time to best initial points
        if m == 1:
            weights = [1.0]
        elif m == 2:
            weights = [0.65, 0.35]
        elif m == 3:
            weights = [0.45, 0.35, 0.20]
        elif m == 4:
            weights = [0.35, 0.30, 0.20, 0.15]
        elif m == 5:
            weights = [0.30, 0.25, 0.20, 0.15, 0.10]
        else:
            weights = [0.25, 0.20, 0.18, 0.15, 0.10, 0.07, 0.05][:m]
            wsum = sum(weights)
            weights = [w / wsum for w in weights]

        # Run optimization from each start
        for idx, (sc_init, init) in enumerate(scored_starts):
            rem = deadline - time.monotonic()
            if rem < 8.0:
                break
            tb = alloc * weights[idx]
            tb = min(tb, rem - 4.0)
            if tb < 4.0:
                continue

            # Use aggressive mode for promising starts
            aggressive = (sc_init <= best_score * 1.01)

            print(f"  start {idx+1}/{m} (init={sc_init:.6f}), budget={tb:.1f}s, aggr={aggressive}", flush=True)
            f_res, s_res = optimize_lp(n, init, tb, rng, template, aggressive=aggressive)

            if np.isfinite(s_res) and s_res < best_score:
                best_score = float(s_res)
                best_f = np.asarray(f_res, dtype=np.float64).copy()
                print(f"  *** new global best: n={best_f.size}, score={best_score:.6f} ***", flush=True)

        try:
            del template
        except Exception:
            pass

    # Final polishing at the best n found
    if best_f is not None:
        n_best = best_f.size
        rem = deadline - time.monotonic()
        if rem > 25.0:
            print(f"\n== Final polish: n={n_best}, budget={rem-5:.1f}s, score={best_score:.6f}", flush=True)
            try:
                template = make_conv_template(n_best)
                f_res, s_res = optimize_lp(n_best, best_f, rem - 8.0, rng, template, aggressive=True)
                if np.isfinite(s_res) and s_res < best_score:
                    best_score = float(s_res)
                    best_f = np.asarray(f_res, dtype=np.float64).copy()
                    print(f"  polished: score={best_score:.6f}", flush=True)
                del template
            except Exception as e:
                print(f"  polish failed: {e}", flush=True)

    # Final output
    if best_f is None:
        best_f = np.ones(64, dtype=np.float64) / 64.0

    seq = [float(x) for x in best_f]
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float("inf")

    if not np.isfinite(final_score):
        if given_f is not None:
            seq = [float(x) for x in given_f]
            try:
                final_score = float(evaluate_sequence(seq))
            except Exception:
                final_score = float("inf")

    if not np.isfinite(final_score):
        seq = [1.0 / 64.0] * 64

    print(f"\nFINAL: n={len(seq)}, local_best={best_score:.6f}, evaluate={final_score:.6f}", flush=True)
    return seq
