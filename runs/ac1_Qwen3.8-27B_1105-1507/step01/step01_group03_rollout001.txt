def make_conv_template(n):
    """
    Precompute the CSR matrix structure for convolve(f, g).
    """
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    except Exception:
        mix = np.ones(n, dtype=np.float64) / float(n)
    if eps >= 1.0:
        f = mix
    else:
        f = (1.0 - eps) * f + eps * mix
    return normalize_sequence(f)


def smart_restart(base, rng, strength):
    """Restart that preserves structure: keep top coordinates, perturb the rest."""
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    # Keep a fraction of the mass in the "important" coordinates
    threshold = np.sort(f)[max(0, int(n * strength) - 1)] if n > 1 else 0
    mask = f >= threshold
    if not mask.any():
        mask = np.ones(n, dtype=bool)
    
    f_new = f.copy()
    # Perturb the non-important coordinates
    non_important = ~mask
    if non_important.any():
        f_new[non_important] *= rng.uniform(0.1, 2.0, size=int(non_important.sum()))
        f_new[non_important] = np.maximum(f_new[non_important], 0.0)
    
    # Also slightly perturb important ones
    if mask.any():
        f_new[mask] *= rng.uniform(0.8, 1.2, size=int(mask.sum()))
    
    f_new = np.maximum(f_new, 0.0)
    return normalize_sequence(f_new)


def interpolate_sequence(x, n):
    """Resample a nonnegative sequence to length n and normalize it."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def make_conv_template(n):
    """Precompute the CSR matrix structure for convolve(f, g)."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(template, f, n, M, relax=1.0, time_limit=60.0):
    """Solve the LP direction problem."""
    A_template, edge_offsets, c_obj, bounds, m = template
    lp_scale = float(n)

    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale

    b = np.full(m, M * lp_scale * lp_scale * relax, dtype=np.float64)
    
    options = {"presolve": True, "time_limit": time_limit}
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs", options=options)


def compute_line_search_score(f, g, n, tv):
    """Compute the score at interpolation parameter tv between f and g."""
    f_new = (1.0 - tv) * f + tv * g
    f_new = np.maximum(f_new, 0.0)
    ssum = float(f_new.sum())
    if ssum <= 1e-300:
        return float('inf')
    f_new /= ssum
    return score_normalized(f_new, n)


def binary_line_search(f, g, n, tv_lo, tv_hi, max_iters=15):
    """Binary search for the optimal interpolation parameter."""
    best_tv = tv_lo
    best_sc = compute_line_search_score(f, g, n, tv_lo)
    
    mid = (tv_lo + tv_hi) / 2.0
    sc_mid = compute_line_search_score(f, g, n, mid)
    if sc_mid < best_sc:
        best_sc = sc_mid
        best_tv = mid
    
    # Determine search direction
    if sc_mid > best_sc:
        hi = mid
    else:
        lo = mid
    
    for _ in range(max_iters):
        mid = (tv_lo + tv_hi) / 2.0
        sc_mid = compute_line_search_score(f, g, n, mid)
        if sc_mid < best_sc:
            best_sc = sc_mid
            best_tv = mid
        # We need to determine which half to search
        # Sample at 1/4 and 3/4 points
        q1 = tv_lo + (tv_hi - tv_lo) * 0.25
        q3 = tv_lo + (tv_hi - tv_lo) * 0.75
        sc_q1 = compute_line_search_score(f, g, n, q1)
        sc_q3 = compute_line_search_score(f, g, n, q3)
        if sc_q1 < sc_q3:
            tv_hi = q3
        else:
            tv_lo = q1
    
    return best_tv, best_sc


def subgradient_polish(f, n, max_iters=50, lr0=0.01):
    """Fast subgradient-based polishing on the simplex."""
    f = f.copy()
    best_f = f.copy()
    best_score = score_normalized(f, n)
    
    for it in range(max_iters):
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        k_star = int(np.argmax(conv))
        
        # Subgradient of (f*f)(k_star) w.r.t. f
        subgrad = np.zeros(n)
        js = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
        idx = k_star - js
        subgrad[js] = f[idx]
        
        # Also consider the "flat" part: we want to reduce the peak,
        # so move mass from high-subgradient to low-subgradient coordinates
        # Projected gradient step
        lr = lr0 / (1.0 + 0.05 * it)
        f_new = f - lr * subgrad
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s <= 1e-300:
            break
        f_new /= s
        
        score_new = score_normalized(f_new, n)
        if score_new < best_score:
            best_f = f_new.copy()
            best_score = score_new
            f = f_new
        else:
            # Backtrack
            lr2 = lr * 0.1
            f_new2 = f - lr2 * subgrad
            f_new2 = np.maximum(f_new2, 0.0)
            s2 = f_new2.sum()
            if s2 <= 1e-300:
                break
            f_new2 /= s2
            score_new2 = score_normalized(f_new2, n)
            if score_new2 < best_score:
                best_f = f_new2.copy()
                best_score = score_new2
                f = f_new2
            else:
                # Try even smaller
                lr3 = lr * 0.01
                f_new3 = f - lr3 * subgrad
                f_new3 = np.maximum(f_new3, 0.0)
                s3 = f_new3.sum()
                if s3 <= 1e-300:
                    break
                f_new3 /= s3
                score_new3 = score_normalized(f_new3, n)
                if score_new3 < best_score:
                    best_f = f_new3.copy()
                    best_score = score_new3
                    f = f_new3
                else:
                    break
    
    return best_f, best_score


def momentum_subgradient_polish(f, n, max_iters=100, lr0=0.01, momentum=0.9):
    """Momentum-based subgradient polishing."""
    f = f.copy()
    best_f = f.copy()
    best_score = score_normalized(f, n)
    velocity = np.zeros(n)
    
    for it in range(max_iters):
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        k_star = int(np.argmax(conv))
        
        subgrad = np.zeros(n)
        js = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
        idx = k_star - js
        subgrad[js] = f[idx]
        
        # Normalize subgradient
        sg_norm = np.linalg.norm(subgrad)
        if sg_norm > 1e-300:
            subgrad /= sg_norm
        
        lr = lr0 / (1.0 + 0.02 * it)
        velocity = momentum * velocity - lr * subgrad
        
        f_new = f + velocity
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s <= 1e-300:
            break
        f_new /= s
        
        score_new = score_normalized(f_new, n)
        if score_new < best_score:
            best_f = f_new.copy()
            best_score = score_new
            f = f_new
        else:
            # Decay velocity
            velocity *= 0.5
    
    return best_f, best_score
