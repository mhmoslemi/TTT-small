"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses binomial coefficient initialization (theoretically motivated by the
flat-spectrum property of 1/sqrt(1-x)), LP-based alternating minimization,
and FFT-accelerated subgradient refinement.

import numpy as np
import time
from scipy.optimize import linprog

def binomial_seq(n):
    """Generate C(2k,k)/4^k for k=0,...,n-1 using recurrence."""
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2.0*k - 1.0) / (2.0*k)
    return a

def symmetrize(a):
    """Create a symmetric version: a_sym[k] = a[k] * a[n-1-k] normalized."""
    n = len(a)
    b = a * a[::-1]
    s = b.sum()
    if s > 0:
        return b / s
    return a / a.sum()

def fast_self_conv(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    result[result < 0] = 0.0
    return result

def compute_score_fast(a):
    """Compute score using FFT convolution."""
    n = len(a)
    c = fast_self_conv(a)
    max_c = np.max(c)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (sum_a * sum_a))

def build_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix for conv(a, .)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # A[k, j] = a[k - j] for j in [lo, hi]
        # So A[k, lo:hi+1] = a[k-hi : k-lo+1][::-1]
        A[k, lo:hi+1] = a[k-hi:k-lo+1][::-1]
    return A

def lp_solve(a, M, time_limit=5.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def lp_iteration(a, M, mix=1.0):
    """One LP iteration: solve LP and mix with current a."""
    b = lp_solve(a, M)
    if b is None:
        return a
    b_sum = b.sum()
    a_sum = a.sum()
    if b_sum < 1e-10:
        return a
    b = b * (a_sum / b_sum)
    return (1.0 - mix) * a + mix * b

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses binomial coefficient initialization (C(2k,k)/4^k, which gives a flat
self-convolution in the infinite limit), LP-based alternating minimization
(maximize sum(b) s.t. conv(a,b)<=M, b>=0), and FFT-accelerated subgradient
refinement. Tries multiple lengths and initialization strategies.
"""

import numpy as np
import time
from scipy.optimize import linprog


def binomial_seq(n):
    """Generate C(2k,k)/4^k for k=0,...,n-1 using stable recurrence."""
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return a


def symmetrize_binomial(n):
    """Create symmetric sequence from binomial coefficients."""
    a = binomial_seq(n)
    b = a * a[::-1]
    s = b.sum()
    if s > 1e-30:
        return b / s
    return a / a.sum()


def fast_self_conv(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_score_fast(a):
    """Compute score using FFT convolution (sum(a) assumed = 1)."""
    n = len(a)
    c = fast_self_conv(a)
    max_c = np.max(c)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (sum_a * sum_a))


def build_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix: (A@b)[k] = conv(a,b)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return A


def lp_solve(a, M, time_limit=5.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M (element-wise), b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def subgradient_step(a, lr, rng=None):
    """Projected subgradient step for minimizing max(conv(a,a)) with sum(a)=1."""
    n = len(a)
    c = fast_self_conv(a)
    peak = np.max(c)

    # Find active constraints
    threshold = peak * 0.995
    peak_idx = np.where(c >= threshold)[0]
    if len(peak_idx) > 25:
        vals = c[peak_idx]
        top = np.argsort(vals)[::-1][:20]
        peak_idx = peak_idx[top]

    # Compute subgradient: d max(c)/d a[m] = (2/n_active) * sum_{k active} a[k-m]
    grad = np.zeros(n)
    for k in peak_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= max(1, len(peak_idx))

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-30:
        a_new = a_new / s
    else:
        a_new = a
    return a_new, peak


def optimize_lp_dominant(a, deadline, rng, lp_interval=5, sg_every=3):
    """
    Main optimization: LP-based alternating with periodic subgradient steps.
    """
    n = len(a)
    a = np.maximum(a, 0)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score_fast(a)

    lr = 0.3 / np.sqrt(n)
    lr_min = 0.01 / np.sqrt(n)
    stall = 0
    it = 0
    max_iters = 50000

    while time.time() < deadline and it < max_iters:

        score = 2.0 * n * peak  # sum(a) = 1

            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Periodic LP step

            remaining = deadline - time.time() - 10
            tl = min(4.0, max(0.5, remaining * 0.1))
            M = peak

            if b is not None and b.sum() > 1e-10:

                # Try mixing

                    a_new = (1.0 - t) * a + t * b
                    a_new = np.maximum(a_new, 0)

                    if sn < 1e-10:

                    a_new = a_new / sn
                    s_new = compute_score_fast(a_new)
                    if s_new < score:

                        score = s_new

        # Subgradient step

            # Adaptive learning rate

                    a = np.maximum(a, 0)
                    a = a / a.sum()

    """Two-phase: first subgradient (coarse), then LP (fine)."""
    n = len(a)

    phase1_deadline = time.time() + (deadline - time.time()) * phase1_frac

    # Phase 1: Subgradient
    a, _ = _subgradient_only(a, phase1_deadline, rng)

    # Phase 2: LP refinement

def _subgradient_only(a, deadline, rng):
    """Pure subgradient descent."""
    n = len(a)
    best_a = a.copy()
    best_score = compute_score_fast(a)
    lr = 0.5 / np.sqrt(n)
    stall = 0

    while time.time() < deadline:

            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                lr = max(lr * 0.8, 0.01 / np.sqrt(n))
