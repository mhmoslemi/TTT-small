import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    N = 2 * n - 1
    A = np.zeros((N, n), dtype=np.float64)
    for k in range(N):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # A[k, j] = a[k - j] for j in [j_start, j_end]
        A[k, j_start:j_end+1] = a[k - j_end : k - j_start + 1][::-1]
    return A

def solve_lp(a, T):
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, T)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_alternate(a, max_iter=200, verbose=False):
    n = len(a)
    a = a.copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    t = 0.7  # step size
    for it in range(max_iter):
        T = float(np.max(np.convolve(a, a)))
        g = solve_lp(a, T)
        if g is None or np.sum(g) < 1e-10:
            break
        g = g / np.sum(g)
        
        # Check improvement
        obj_g = compute_obj(g)
        
        # Blend
        a_new = (1 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        # Also check pure g
        if obj_g < best_obj:
            best_obj = obj_g
            best_a = g.copy()
            a = g.copy()
        else:
            a = a_new
        
        if verbose and it % 20 == 0:
            print(f"    it={it}: obj={best_obj:.6f}, T={T:.6f}")
    
    return best_a, best_obj

def peak_shave(a, max_steps=500, verbose=False):
    """Local refinement by reducing the convolution peak."""
    a = a.copy()
    a = a / np.sum(a)
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for step in range(max_steps):
        conv = np.convolve(a, a)
        k_peak = int(np.argmax(conv))
        
        # Sensitivity: d(conv[k_peak])/d(a[j]) = a[k_peak - j] (for valid j)
        sens = np.zeros(n)
        for j in range(n):
            idx = k_peak - j
            if 0 <= idx < n:
                sens[j] = a[idx]
        
        # Reduce a where sensitivity is high, increase where low
        # Use a "push-pull" approach
        mean_sens = np.mean(sens)
        # Elements with high sensitivity should be decreased
        # Elements with low sensitivity should be increased
        
        # Simple gradient step
        lr = 0.1 / (np.max(sens) + 1e-10)
        a_new = a - lr * sens
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_new2 = a - lr * 0.1 * sens
            a_new2 = np.clip(a_new2, 0, None)
            s2 = np.sum(a_new2)
            if s2 > 1e-10:
                a_new2 = a_new2 / s2
                obj2 = compute_obj(a_new2)
                if obj2 < best_obj:
                    best_obj = obj2
                    best_a = a_new2.copy()
                    a = a_new2
                else:
                    break  # Local minimum
            else:
                break
    
    return best_a, best_obj

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_conv_matrix(a):
    """Build matrix A where A @ g = conv(a, g) (full convolution)."""
    n = len(a)
    N = 2 * n - 1
    A = np.zeros((N, n))
    for k in range(N):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je+1] = a[k-je:k-js+1][::-1]
    return A

    """Solve: max sum(g) s.t. A@g <= T, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, T), bounds=[(0, None)]*n, method='highs')

        return res.x
    return None

"""
Optimized LP-alternation with diverse initializations and local search
for minimizing 2*n*max(conv(a,a))/(sum(a))^2 with a >= 0.
Uses staircase and two-block structures inspired by the observation that
'monotone' sequences have flatter self-convolutions than uniform.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    n = len(a)
    N = 2 * n - 1
    A = np.zeros((N, n), dtype=np.float64)
    for k in range(N):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je+1] = a[k - je : k - js + 1][::-1]
    return A


def lp_max_sum(a, T):
    """max sum(g) s.t. A@g <= T, g >= 0"""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, T)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_min_peak(a):
    """min t s.t. A@g <= t, sum(g)=1, g>=0. Returns (g, t)."""
    n = len(a)
    A = build_A(a)
    # Variables: [g_0, ..., g_{n-1}, t], total n+1
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize t
    
    # A_ub @ x <= b_ub: A@g - t <= 0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    # A_eq @ x = b_eq: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n], float(res.x[n])
    return None, None
