def _fft_convolve_full(f):
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

    """Compute pressure vector using FFT.
    p[j] = sum_k q[k] * f[k-j] for valid indices, where q is active_mask.
