def smoothed_grad(a, c, mu):
    """
    Compute gradient of the log-sum-exp smoothed max of c.
    f_mu = (1/mu) * log(sum(exp(mu*c)))
    grad = sum_k w_k * grad_c_k where w_k = exp(mu*(c_k - max_c)) / sum(exp(mu*(c_j - max_c)))
    """
    n = len(a)
    c_max = np.max(c)
    exp_vals = np.exp(mu * (c - c_max))
    Z = np.sum(exp_vals)
    w = exp_vals / Z  # weights over 2n-1 positions
    
    # grad[m] = 2 * sum_k w[k] * a[k-m] (with a[i]=0 for i outside [0,n-1])
    # = 2 * conv(w, a_reversed)[n-1 : 2n-1]
    a_rev = a[::-1]
    size_w = len(w)  # 2n-1
    size_conv = size_w + n - 1
    N_fft = 1 << (size_conv - 1).bit_length()
    fw = np.fft.rfft(w, N_fft)
    fa = np.fft.rfft(a_rev, N_fft)
    conv_result = np.fft.irfft(fw * fa, N_fft)
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    grad = np.maximum(grad, 0)  # gradient should be non-negative for non-neg a
    
    f_val = (c_max + np.log(Z)) / mu
    return f_val, grad


def alt_projection_init(n, rng, n_iter=100, target_type='flat'):
    """
    Alternating projection: project between non-negativity in time and target magnitude in freq.
    """
    N = 1 << (2 * n - 2).bit_length()
    a = np.ones(n) / n
    
    for it in range(n_iter):
        # FFT to frequency domain
        F = np.fft.rfft(a, N)
        # Set target magnitude
        if target_type == 'flat':
            target_mag = np.ones(len(F))
        elif target_type == 'tapered':
            # Slight taper: more weight at low frequencies
            freqs = np.linspace(0, 1, len(F))
            target_mag = 1.0 - 0.3 * freqs
        elif target_type == 'raised_cosine':
            freqs = np.linspace(0, 1, len(F))
            target_mag = 0.5 * (1.0 + np.cos(np.pi * freqs)) + 0.1
        else:
            target_mag = np.ones(len(F))
        
        # Preserve phase, set magnitude
        phase = np.angle(F)
        F_new = target_mag * np.exp(1j * phase)
        F_new[0] = np.abs(F_new[0]) + 0j  # Keep DC real and positive
        
        # IFFT back to time domain
        a_full = np.fft.irfft(F_new, N)
        a = a_full[:n]
        # Project to non-negative and normalize
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a = a / s
    
    return a


def nesterov_smoothed_optimize(a, deadline, mu_init=20.0, mu_decay=0.999, 
                                base_lr=None, max_iters=200000, rng=None):
    """
    Nesterov-accelerated gradient descent on the log-sum-exp smoothed objective.
    Minimize max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    if rng is None:
        rng = np.random.default_rng()
    
    a = project_simplex(a.copy().astype(np.float64))
    
    # Initial evaluation
    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()
    
    if base_lr is None:
        base_lr = 0.8 / np.sqrt(n)
    
    lr = base_lr
    mu = mu_init
    
    # Nesterov momentum
    beta = 1.0
    a_prev = a.copy()
    
    it = 0
    stall_count = 0
    max_stall = 500
    
    while time.time() < deadline and it < max_iters:
        it += 1
        
        # Nesterov extrapolation
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = a + (beta / beta_new - 1.0) * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-12:
            y = a.copy()
        else:
            y = y / s_y
        beta = beta_new
        
        # Compute smoothed gradient at y
        c_y = fast_convolve(y)
        f_val, grad = smoothed_grad(y, c_y, mu)
        
        # Gradient step
        a_new = y - lr * grad
        a_new = project_simplex(a_new)
        
        a_prev = a
        a = a_new
        
        # Evaluate
        if it % 5 == 0:
            c = fast_convolve(a)
            peak = np.max(c)
            if peak < best_peak * (1.0 - 1e-9):
                best_peak = peak
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
            
            # Adaptive learning rate
            if stall_count > max_stall:
                lr = max(lr * 0.75, base_lr * 1e-4)
                stall_count = 0
                if lr < base_lr * 0.01:
                    # Perturb to escape
                    if time.time() < deadline - 10:
                        noise = rng.normal(0, 0.03 / np.sqrt(n), n)
                        a = project_simplex(a + noise)
            
            # Decay mu toward smaller values (less smoothing = more accurate)
            mu = max(mu * mu_decay, 0.5)
        
        # Multiplicative weights step every 50 iterations (for diversity)
        if it % 50 == 0 and time.time() < deadline - 5:
            c = fast_convolve(a)
            peak = np.max(c)
            # Use multiplicative update to reduce peak contributions
            c_max = np.max(c)
            w = np.exp(-0.5 * c / c_max)  # dampen high-convolution positions
            a_mw = a * w
            a_mw = project_simplex(a_mw)
            a = project_simplex(0.7 * a + 0.3 * a_mw)
    
    return best_a


def multiplicative_weights_optimize(a, deadline, max_iters=300000, rng=None):
    """
    Pure multiplicative weights (exponential tilting) optimization.
    """
    n = len(a)
    if rng is None:
        rng = np.random.default_rng()
    
    a = project_simplex(a.copy().astype(np.float64))
    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()
    
    lr = 1.0 / np.sqrt(n)
    it = 0
    stall_count = 0
    
    while time.time() < deadline and it < max_iters:
        it += 1
        
        c = fast_convolve(a)
        peak = np.max(c)
        
        if peak < best_peak * (1.0 - 1e-9):
            best_peak = peak
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Get subgradient
        grad, _ = subgradient_max_conv(a, c, tol=1e-5)
        
        # Multiplicative weights update
        a_mw = a * np.exp(-lr * grad)
        s = a_mw.sum()
        if s < 1e-12:
            a_mw = np.ones(n) / n
        else:
            a_mw = a_mw / s
        
        a = a_mw
        
        # Adaptive lr
        if stall_count > 400:
            lr = max(lr * 0.7, 1e-6)
            stall_count = 0
            if lr < 1e-5:
                noise = rng.normal(0, 0.05 / np.sqrt(n), n)
                a = project_simplex(a + noise)
    
    return best_a


def lp_polish(a, deadline, max_time_per_lp=5.0):
    """
    LP-based polishing: maximize sum(b) subject to conv(a,b) <= P, b >= 0.
    Mix the solution with the current best.
    """
    n = len(a)
    if n > 1500:
        return a
    
    c = fast_convolve(a)
    P = np.max(c)
    a_sum = a.sum()
    
    # Build constraint matrix A: A[k, j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = a[k - hi:k - lo + 1]
    
    try:
        t_start = time.time()
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, P),
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': max_time_per_lp})
        if res.success and res.x is not None:
            b = res.x
            b_sum = b.sum()
            if b_sum > 1e-8:
                b = b * (a_sum / b_sum)
                # Mix
                t_mix = 0.5
                a_new = project_simplex((1 - t_mix) * a + t_mix * b)
                c_new = fast_convolve(a_new)
                if np.max(c_new) < np.max(c):
                    return a_new
    except Exception:
        pass
    
    return a


def spectral_equalize(a, n_steps=10):
    """
    Apply spectral equalization: adjust phases to reduce autocorrelation peak.
    """
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    
    for _ in range(n_steps):
        F = np.fft.rfft(a, N)
        # Compute current autocorrelation
        c = np.fft.irfft(np.abs(F) ** 2, N)[:2 * n - 1]
        c = np.maximum(c, 0)
        peak = np.max(c)
        
        if peak < 1e-10:
            break
        
        # Target: reduce peak by adjusting phases
        # Use a simple gradient on phases
        # d(peak)/d(phase_k) ... this is complex. Instead, use a heuristic:
        # Reduce the magnitude of high-frequency components that contribute to the peak
        freqs = np.arange(len(F))
        # Dampen high frequencies slightly
        damp = 1.0 - 0.02 * (freqs / len(F)) ** 2
        F = F * damp
        
        a_full = np.fft.irfft(F, N)
        a = np.maximum(a_full[:n], 0.0)
        s = a.sum()
        if s < 1e-12:
            break
        a = a / s
    
    return a


def try_optimize(n, init_method, deadline, rng, method='nesterov'):
    """Initialize and optimize a sequence of given length."""
    # Initialize
    if init_method == 'alt_flat':
        a = alt_projection_init(n, rng, n_iter=80, target_type='flat')
    elif init_method == 'alt_tapered':
        a = alt_projection_init(n, rng, n_iter=80, target_type='tapered')
    elif init_method == 'alt_cosine':
        a = alt_projection_init(n, rng, n_iter=80, target_type='raised_cosine')
    elif init_method == 'random':
        a = rng.random(n)
        a = a / a.sum()
    elif init_method == 'parent':
        try:
            parent = np.array(height_sequence_1, dtype=np.float64)
            parent = np.clip(parent, 0.0, 1000.0)
            if len(parent) >= n:
                a = parent[:n].copy()
            else:
                a = np.zeros(n)
                a[:len(parent)] = parent
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s if s > 1e-10 else np.ones(n) / n
        except Exception:
            a = rng.random(n)
            a = a / a.sum()
    elif init_method == 'broadband':
        a = alt_projection_init(n, rng, n_iter=40, target_type='flat')
        a = a + 0.2 * rng.random(n)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    elif init_method == 'alternating_pattern':
        # Start with alternating high/low
        a = np.ones(n)
        a[np.arange(0, n, 2)] = 2.0
        a[np.arange(1, n, 2)] = 1.0
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()
    
    # Optimize
    if method == 'nesterov':
        a_opt = nesterov_smoothed_optimize(a, deadline, mu_init=20.0, 
                                            mu_decay=0.9995, rng=rng)
    elif method == 'mw':
        a_opt = multiplicative_weights_optimize(a, deadline, rng=rng)
    else:
        a_opt = nesterov_smoothed_optimize(a, deadline, rng=rng)
    
    # Spectral equalization
    a_opt = spectral_equalize(a_opt, n_steps=5)
    
    # LP polish
    if n <= 1024 and time.time() < deadline - 3:
        a_opt = lp_polish(a_opt, deadline, max_time_per_lp=min(3.0, deadline - time.time() - 2))
    
    score = evaluate_sequence(a_opt.tolist())
    return a_opt, score


def propose_candidate(budget_s=1500, **kwargs):
    """
    Multi-phase search combining Nesterov-accelerated smoothed gradient descent,
    multiplicative weights, alternating projection initialization, spectral
    equalization, and LP polishing. Tries multiple sequence lengths.
    """
    overall_start = time.time()
    overall_deadline = overall_start + budget_s - 20.0
    rng = np.random.default_rng(12345)
    
    best_score = float('inf')
    best_seq = None
    
    # Phase 1: Try several configurations
    # (length, init_method, method)
    configs = [
        (512, 'parent', 'nesterov'),
        (512, 'alt_flat', 'nesterov'),
        (512, 'alt_tapered', 'nesterov'),
        (1024, 'alt_flat', 'nesterov'),
        (384, 'alt_flat', 'nesterov'),
        (512, 'broadband', 'nesterov'),
        (768, 'alt_tapered', 'nesterov'),
        (512, 'alt_cosine', 'nesterov'),
        (256, 'alt_flat', 'nesterov'),
        (640, 'alt_flat', 'nesterov'),
        (512, 'random', 'mw'),
        (384, 'alternating_pattern', 'nesterov'),
    ]
    
    # Allocate time: give more to early configs
    num_configs = len(configs)
    # First 4 get 35% each of total, rest split remaining 10%
    # Actually let's be more balanced
    total_time = overall_deadline - time.time()
    # First 5 configs get 20% each, rest get 5% each
    times = [0.20] * 5 + [0.05] * (num_configs - 5)
    # Normalize
    t_sum = sum(times)
    times = [t * total_time / t_sum for t in times]
    
    for i, (n, init_method, method) in enumerate(configs):
        if time.time() > overall_deadline - 30:
            break
        
        config_deadline = min(overall_deadline - 10, time.time() + times[i])
        if config_deadline < time.time() + 10:
            continue
        
        print(f"[{time.time()-overall_start:.0f}s] Trying n={n}, init={init_method}, method={method} "
              f"(time budget: {times[i]:.0f}s)")
        
        try:
            a_opt, score = try_optimize(n, init_method, config_deadline, rng, method=method)
        except Exception as e:
            print(f"  Error: {e}")
            continue
        
        print(f"  n={n}, init={init_method}, method={method}: score={score:.6f}")
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  >>> New best: {best_score:.6f} (n={n}, init={init_method})")
    
    # Phase 2: Polish the best sequence with remaining time
    if best_seq is not None and time.time() < overall_deadline - 10:
        print(f"\nPolishing best sequence (score={best_score:.6f})...")
        polish_deadline = overall_deadline
        
        # Multiple polish passes
        for polish_iter in range(3):
            if time.time() > polish_deadline - 5:
                break
            try:
                a_polished = nesterov_smoothed_optimize(
                    best_seq, polish_deadline - 30,
                    mu_init=5.0, mu_decay=0.999, rng=rng)
                a_polished = spectral_equalize(a_polished, n_steps=5)
                if len(a_polished) <= 1024:
                    a_polished = lp_polish(a_polished, polish_deadline - 3, max_time_per_lp=3.0)
                score = evaluate_sequence(a_polished.tolist())
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = a_polished.copy()
                    print(f"  Polish {polish_iter}: score={best_score:.6f}")
                else:
                    print(f"  Polish {polish_iter}: no improvement ({score:.6f})")
            except Exception as e:
                print(f"  Polish error: {e}")
                break
        
        # Final micro-polish with LP
        if len(best_seq) <= 1024 and time.time() < polish_deadline - 3:
            a_lp = lp_polish(best_seq, polish_deadline, max_time_per_lp=5.0)
            score_lp = evaluate_sequence(a_lp.tolist())
            if np.isfinite(score_lp) and score_lp < best_score:
                best_score = score_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: score={best_score:.6f}")
    
    # Fallback
    if best_seq is None:
        print("Falling back to uniform sequence")
        best_seq = np.ones(256) / 256
    
    result = [float(x) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"\nFINAL: score={final_score:.6f}, length={len(result)}, "
          f"time={time.time()-overall_start:.0f}s")
    
    return result
