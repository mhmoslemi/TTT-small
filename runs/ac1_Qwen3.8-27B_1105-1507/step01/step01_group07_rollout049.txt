import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def gradient_step(a, lr=0.01, t_smooth=5.0):
    """One step of smoothed gradient descent."""
    n = len(a)
    conv = np.convolve(a, a)
    S = np.sum(a)
    
    # Softmax weights
    t = t_smooth
    shifted = t * conv - t * np.max(conv)  # for numerical stability
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # Gradient of smoothed max w.r.t. a[i] = sum_k weights[k] * 2*a[k-i] (if valid)
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = weights[k]
        if wk < 1e-10:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # grad[j] += wk * 2 * a[k-j]
        # k-j ranges from k-j_end to k-j_start
        grad[j_start:j_end+1] += wk * 2 * a[k - j_end : k - j_start + 1][::-1]
    
    # Full gradient of f = 2n * M_smooth / S^2
    # df/da[i] = 2n/S^2 * grad_M[i] - 4n*M_smooth/S^3
    M_smooth = float(np.dot(weights, conv))
    obj_grad = (2.0 * n / S**2) * grad - (4.0 * n * M_smooth / S**3)
    
    # Projected gradient step
    a_new = a - lr * obj_grad
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a
    return a_new

"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines LP-based alternating optimization, subgradient methods, and simulated annealing.

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp(a, M):
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, np.inf)]*n, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_optimize(a_init, n, seed, max_iter=1000, deadline=None):
    """LP-based alternating optimization with adaptive step size."""
    rng = np.random.default_rng(seed)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    t = 0.5  # blending parameter
    stall = 0
    t_history = [0.5, 0.5, 0.4, 0.3, 0.2, 0.1]
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        g = solve_lp(a, M)
        if g is None:
            break
        
        g_sum = float(np.sum(g))
        if g_sum < 1e-10:
            break
        
        # Adaptive step size
        if it < len(t_history):
            t = t_history[it]
        else:
            t = max(0.05, t * 0.995)
        
        g = g / g_sum
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0

            stall += 1

                a = best_a * noise
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s < 1e-10:
                    a = rng.random(n)
                    s = np.sum(a)
                a = a / s
                stall = 0
                t = 0.5
        a = a_new
    
    return best_a, best_obj


def subgradient_refine(a_init, n, seed, max_iter=5000, deadline=None):
    """Subgradient method for the non-smooth max convolution."""
    rng = np.random.default_rng(seed)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    lr = 0.1 / np.sqrt(n)
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv = np.convolve(a, a)
        max_b = float(np.max(conv))
        S = float(np.sum(a))
        
        # Find peak positions
        peak_thresh = 0.9 * max_b
        peak_mask = conv >= peak_thresh
        peak_indices = np.where(peak_mask)[0]
        
        # Subgradient: for each peak k, grad[i] += 2*a[k-i] (if valid)
        grad = np.zeros(n)
        for k in peak_indices:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            grad[j_start:j_end+1] += 2.0 * a[k - j_end:k - j_start + 1][::-1]
        
        # Normalize subgradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        # Objective gradient: 2n/S^2 * grad_M - 4n*M/S^3
        obj_grad = (2.0 * n / S**2) * grad - (4.0 * n * max_b / S**3)
        
        a_new = a - lr * obj_grad
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        # Decrease learning rate
        lr *= 0.999
    
    return best_a, best_obj


def simulated_annealing(a_init, n, seed, max_iter=20000, deadline=None):
    """Simulated annealing with block moves."""
    rng = np.random.default_rng(seed)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    curr_obj = best_obj
    
    T = 0.1
    T_min = 0.001
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Block move
        width = max(1, min(n // 10, int(rng.exponential(3)) + 1))
        start = int(rng.integers(0, n - width + 1))
        
        candidate = a.copy()
        candidate[start:start+width] *= np.exp(rng.normal(0, 0.3, width))
        candidate = np.maximum(candidate, 0)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s
        
        cand_obj = compute_obj(candidate)
        delta = cand_obj - curr_obj
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            a = candidate
            curr_obj = cand_obj
            if cand_obj < best_obj:
                best_obj = cand_obj
                best_a = candidate.copy()
        
        T = max(T_min, T * 0.9995)
    
    return best_a, best_obj


def coordinate_descent(a_init, n, seed, max_iter=10000, deadline=None):
    """Coordinate descent: optimize one element at a time."""
    rng = np.random.default_rng(seed)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    order = rng.permutation(n)
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        i = order[it % n]
        
        # Try different scales for a[i]
        old_val = a[i]
        for scale in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
            candidate = a.copy()
            candidate[i] = old_val * scale
            candidate = np.maximum(candidate, 0)
            s = np.sum(candidate)
            if s < 1e-10:
                continue
            obj = compute_obj(candidate)
            if obj < best_obj:
                best_obj = obj
                best_a = candidate.copy()
                a = candidate
                break
    
    return best_a, best_obj

"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines LP-based alternating optimization with adaptive step sizes,
subgradient refinement, and simulated annealing across multiple sequence lengths.
Key improvements over baseline: adaptive LP parameters, active-set monitoring,
log-space gradient steps, and smarter restarts guided by convolution structure.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix A where (A @ g)[k] = (a*g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_for_partner(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, np.inf)] * n, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_optimize_adaptive(a_init, n, seed, max_iter=5000, deadline=None):
    """LP-based alternating optimization with adaptive step and restarts."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    t = 0.5
    stall = 0
    prev_obj = best_obj

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = np.convolve(a, a)
        M = float(np.max(conv))

        # Solve LP for partner
        g = solve_lp_for_partner(a, M)
        if g is None:
            # Perturb to escape
            a = best_a * np.exp(rng.normal(0, 0.4, n))
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            continue

        g_sum = float(np.sum(g))
        if g_sum < 1e-10:
            break

        g = g / g_sum

        # Adaptive step size
        distance = float(np.linalg.norm(g - a)) / np.sqrt(n)
        if distance < 0.005:
            t = 0.05
        elif distance < 0.02:
            t = 0.15
        elif distance < 0.1:
            t = 0.4
        else:
            t = 0.8

        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-9:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                # Escape: perturb based on convolution structure
                conv_best = np.convolve(best_a, best_a)
                M_best = float(np.max(conv_best))
                # Identify peak positions

                s = np.sum(a)

        a = a_new

    return best_a, best_obj


def subgradient_refine(a_init, n, seed, max_iter=20000, deadline=None):
    """Subgradient method targeting peak reduction."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    lr = 0.5 / np.sqrt(n)

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = np.convolve(a, a)
        max_b = float(np.max(conv))
        S = float(np.sum(a))

        # Use soft peak identification
        peak_frac = 0.9 + 0.05 * np.sin(it * 0.01)
        peak_thresh = peak_frac * max_b
        peak_indices = np.where(conv >= peak_thresh)[0]

        if len(peak_indices) == 0:
            peak_indices = np.array([int(np.argmax(conv))])

        # Accumulate subgradient
        grad = np.zeros(n)
        for k in peak_indices:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            grad[j_start:j_end + 1] += 2.0 * a[k - j_end:k - j_start + 1][::-1]

        # Objective subgradient (scale invariant, so work with S=1)
        # df/da[i] ~ (2n/S^2) * grad_M[i] - (4n*M/S^3) * 1
        obj_grad = (2.0 * n / (S * S)) * grad - (4.0 * n * max_b / (S * S * S))

        grad_norm = float(np.linalg.norm(obj_grad))
        if grad_norm < 1e-15:
            break

        a_new = a - lr * obj_grad
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr = max(0.01 / n, lr * 0.9995)

    return best_a, best_obj


def simulated_annealing_refine(a_init, n, seed, max_iter=50000, deadline=None):
    """Simulated annealing with adaptive block sizes."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    curr_obj = best_obj

    T = 0.05
    T_min = 0.0001
    best_T = T

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Adaptive block size
        width = max(1, min(n // 5, int(rng.exponential(2.0)) + 1))
        start = int(rng.integers(0, n - width + 1))

        candidate = a.copy()
        sigma = 0.2 + 0.3 * T
        candidate[start:start + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.maximum(candidate, 0)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s

        cand_obj = compute_obj(candidate)
        delta = cand_obj - curr_obj

        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            a = candidate
            curr_obj = cand_obj
            if cand_obj < best_obj:
                best_obj = cand_obj
                best_a = candidate.copy()
                best_T = T

        T = max(T_min, T * 0.9998)

    return best_a, best_obj


def log_space_gradient(a_init, n, seed, max_iter=5000, deadline=None):
    """Gradient descent in log-space for smooth optimization."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init, 1e-10)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n) + 0.01
        s = np.sum(a)
    a = a / s

    b = np.log(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    lr = 0.05
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        a = np.exp(b)
        conv = np.convolve(a, a)
        max_b = float(np.max(conv))
        S = float(np.sum(a))

        # Softmax weights for smooth max
        t_smooth = 10.0 + it * 0.1
        shifted = t_smooth * conv - t_smooth * max_b
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-30:
            break
        weights = weights / w_sum

        # Gradient of smoothed max w.r.t. a[i]
        grad_a = np.zeros(n)
        for k in range(2 * n - 1):
            if weights[k] < 1e-8:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            grad_a[j_start:j_end + 1] += weights[k] * 2.0 * a[k - j_end:k - j_start + 1][::-1]

        # Gradient of f = 2n*M/S^2 w.r.t. b[i] = a[i] * df/da[i]
        M_smooth = float(np.dot(weights, conv))
        grad_b = a * ((2.0 * n / (S * S)) * grad_a - (4.0 * n * M_smooth / (S * S * S)))

        b = b - lr * grad_b
        b = np.clip(b, -15, 15)

        a_new = np.exp(b)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        lr = max(0.001, lr * 0.999)

    return best_a, best_obj


def structured_init(n, rng):
    """Generate structured initialization."""
    choice = int(rng.integers(0, 5))
    if choice == 0:
        # Cosine window
        x = np.arange(n) / (n - 1)
        a = np.cos(np.pi * x) ** 2
    elif choice == 1:
        # Two-tone
        a = np.ones(n)
        a[n // 3:2 * n // 3] = 0.5
    elif choice == 2:
        # Periodic with random pattern
        period = int(max(2, rng.integers(3, max(4, n // 10))))
        pattern = rng.random(period) + 0.1
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 3:
        # Gaussian bump
        center = n / 2
        sigma = n / (2 * rng.uniform(1, 3))
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    else:
        a = rng.random(n)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s
