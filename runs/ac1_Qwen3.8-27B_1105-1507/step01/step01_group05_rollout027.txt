import time

def compute_score(f):
    """Fast score computation for a normalized sequence (sum=1)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()

def de_evolve(pop, scores, F=0.8, CR=0.9, max_gens=None, time_budget=0, rng=None):
    """Differential evolution for the population."""
    if rng is None:
        rng = np.random.default_rng()
    NP = len(pop)
    n = len(pop[0])
    best_idx = np.argmin(scores)
    best_score = scores[best_idx]
    best_pop = pop[best_idx].copy()
    
    t0 = time.monotonic()
    for gen in range(10**9):
        if time.monotonic() - t0 > time_budget:
            break
        for i in range(NP):
            # Select three distinct random indices different from i
            r1, r2, r3 = rng.choice(NP, size=3, replace=False)
            while r1 == i or r2 == i or r3 == i:
                r1, r2, r3 = rng.choice(NP, size=3, replace=False)
            
            # Mutation
            v = pop[r1] + F * (pop[r2] - pop[r3])
            v = np.clip(v, 0, None)
            s = v.sum()
            if s < 1e-15:
                continue
            v = v / s
            
            # Crossover (binomial)
            trial = v.copy()
            j_rand = rng.integers(0, n)
            for j in range(n):
                if rng.random() < CR or j == j_rand:
                    trial[j] = v[j]
                else:
                    trial[j] = pop[i][j]
            
            trial = np.clip(trial, 0, None)
            s = trial.sum()
            if s < 1e-15:
                continue
            trial = trial / s
            
            # Selection
            score = compute_score(trial)
            if score < scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_pop = trial.copy()
    
    return best_pop, best_score

"""
Hybrid DE + LP optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses differential evolution for global exploration, LP-based local search for
polishing, and iterated local search for final improvement.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_score(f):
    """Compute 2*n*max(conv(f,f)) for normalized f (sum=1)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()

def normalize(f):
    """Normalize f to sum=1, clip non-negative."""
    f = np.clip(f, 0, None)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s

def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b)."""
    n = len(f)
    # Use FFT-based approach for efficiency
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A

def lp_local_search(f, time_budget, t0=0.5, rng=None):
    """LP-based local search."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    n = len(f)
    f = normalize(f.astype(np.float64))
    best_f = f.copy()
    best_score = compute_score(f)
    t = t0
    
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        g = b / b_sum
        f_new = (1 - t) * f + t * g
        f_new = normalize(f_new)
        new_score = compute_score(f_new)
        
        if new_score < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = new_score
            f = f_new
            stall = 0
        else:
            t *= 0.7
            stall += 1
            if stall > 30:
                # Perturb to escape
                f_pert = f * (1 + 0.1 * rng.normal(0, 1, n))
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    t = t0 * 0.5
                    stall = 0
                elif time.monotonic() - start > time_budget * 0.7:
                    break
    
    return best_f, best_score

def simple_local_search(f, rng, n_iter=5000):
    """Fast local search with random block perturbations."""
    n = len(f)
    f = normalize(f.copy())
    best = f.copy()
    best_score = compute_score(f)
    
    for _ in range(n_iter):
        f_c = f.copy()
        width = max(1, int(rng.integers(1, max(2, n // 16))))
        start = int(rng.integers(0, n - width))
        scale = np.exp(rng.normal(0, 0.5))
        f_c[start:start+width] *= scale
        f_c = normalize(f_c)
        score = compute_score(f_c)
        if score < best_score:
            best = f_c
            best_score = score
            f = f_c
    return best, best_score

def de_phase(pop, scores, F=0.8, CR=0.9, time_budget=0, rng=None):
    """Vectorized differential evolution."""
    if rng is None:
        rng = np.random.default_rng()
    NP = len(pop)
    n = len(pop[0])
    
    best_idx = np.argmin(scores)
    best_score = scores[best_idx]
    best_pop = pop[best_idx].copy()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_budget:
        # Generate all trial individuals
        for i in range(NP):
            # Pick 3 distinct random indices != i
            r1, r2, r3 = rng.choice(NP, size=3, replace=False)
            while i in (r1, r2, r3):
                r1, r2, r3 = rng.choice(NP, size=3, replace=False)
            
            # Mutation
            v = pop[r1] + F * (pop[r2] - pop[r3])
            v = np.clip(v, 0, None)
            s = v.sum()
            if s < 1e-15:
                continue
            v = v / s
            
            # Crossover
            mask = rng.random(n) < CR
            j_rand = rng.integers(0, n)
            mask[j_rand] = True
            trial = np.where(mask, v, pop[i])
            trial = np.clip(trial, 0, None)
            s = trial.sum()
            if s < 1e-15:
                continue
            trial = trial / s
            
            # Selection
            score = compute_score(trial)
            if score < scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_pop = trial.copy()
    
    return best_pop, best_score

def generate_diverse_pop(n, seed=42, parent=None):
    """Generate a diverse initial population."""
    rng = np.random.default_rng(seed)
    pop = []
    
    # Parent
    if parent is not None:
        pop.append(normalize(parent.astype(np.float64)))
    
    # Uniform
    pop.append(np.ones(n) / n)
    
    # Ramps
    pop.append(normalize(np.arange(1, n+1, dtype=float)))
    pop.append(normalize(np.arange(n, 0, -1, dtype=float)))
    pop.append(normalize(np.sqrt(np.arange(1, n+1, dtype=float))))
    
    # Bimodal
    x = np.arange(n)
    pop.append(normalize(np.exp(-((x - n*0.25)**2) / (n*0.05)) + np.exp(-((x - n*0.75)**2) / (n*0.05))))
    
    # Multi-modal
    for k in [3, 4, 5]:
        centers = np.linspace(0.1, 0.9, k) * n
        f = sum(np.exp(-((x - c)**2) / (n*0.03)) for c in centers)
        pop.append(normalize(f))
    
    # Rectified sine
    for freq in [2, 3, 4, 5, 7]:
        f = np.abs(np.sin(2 * np.pi * freq * x / n))
        pop.append(normalize(f))
    
    # Random
    for _ in range(20):
        f = rng.random(n)
        pop.append(normalize(f))
    
    # Structured random (piecewise constant)
    for _ in range(10):
        n_segments = rng.integers(3, 10)
        breaks = np.sort(rng.choice(np.arange(1, n), size=n_segments-1, replace=False))
        segments = np.split(np.arange(n), breaks)
        f = np.zeros(n)
        for seg in segments:
            f[seg] = rng.random()
        pop.append(normalize(f))
    
    # Trim to exactly 100
    while len(pop) < 100:
        f = rng.random(n)
        pop.append(normalize(f))
    if len(pop) > 100:
        pop = pop[:100]
    
    return np.array(pop), np.array([compute_score(p) for p in pop])

Actually, let me think about this more carefully. The LP maximizes sum(b) subject to (f*b)[k] <= M and b >= 0. The solution b tells us how to "spread out" the mass in a way that doesn't increase the convolution peak when combined with f.

The update f_new = (1-t)*f + t*b/h(b) is a convex combination. The question is: what value of t minimizes the new score?

We could do a 1D search over t in [0, 2] (allowing extrapolation) to find the best t. This would be more efficient than the geometric decrease.

def lp_step(f, t_candidates=None):
    """One LP step with line search."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return f, compute_score(f)
    
    b = np.clip(res.x, 0, None)
    b_sum = b.sum()
    if b_sum < 1e-15:
        return f, compute_score(f)
    g = b / b_sum
    
    # Line search: try different t values
    if t_candidates is None:
        t_candidates = np.array([0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0])
    
    best_f = f
    best_score = compute_score(f)
    
    for t in t_candidates:
        f_new = (1-t)*f + t*g
        f_new = normalize(f_new)
        score = compute_score(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new
    
    return best_f, best_score

This is better because it tries multiple step sizes.

Now let me also think about whether there are better global search strategies than DE.

**Simulated Annealing**: 
- Start with a solution
- Propose a random perturbation
- Accept if better, or with probability exp(-Δ/E) if worse
- Decrease E over time

This can be very effective but requires careful tuning of the cooling schedule.

**Tabu Search**:
- Keep a "tabu list" of recent moves
- Propose moves that are not in the tabu list
- Accept the best move even if it's worse (with aspiration criterion)

**Memetic Algorithm** (MA):
- DE + local search on each individual
- This is probably the most effective approach

Let me go with a memetic algorithm: DE + short local search on each individual.

But the local search (LP) is expensive. So I'll use a fast local search (random perturbations) for the DE phase, and reserve the expensive LP for the final polishing.

Here's my final plan:

Phase 1 (40% time): Fast DE with random local search
  - Population of 200

Phase 2 (40% time): LP polish on top 10 candidates

  - Track best

Phase 3 (20% time): ILS
  - From best: perturb + short LP
  - If improved, continue from new best
  - If not, try different perturbation size

"""
Hybrid Differential Evolution + LP local search for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses DE with embedded fast local search for global exploration, then LP-based polishing
with adaptive line search, and iterated local search for final improvement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute 2*n*max(conv(f,f)) for normalized f (sum=1)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def normalize(f):
    """Normalize f to sum=1, clip non-negative."""
    f = np.clip(f, 0, None)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def fast_local_search(f, rng, n_iter=200):
    """Fast local search with random block perturbations."""
    n = len(f)
    f = normalize(f.copy())
    best = f.copy()
    best_score = compute_score(f)
    
    for _ in range(n_iter):
        f_c = f.copy()
        width = max(1, int(rng.integers(1, max(2, n // 8))))
        start = int(rng.integers(0, n - width))
        scale = np.exp(rng.normal(0, 0.4))
        f_c[start:start + width] *= scale
        f_c = normalize(f_c)
        score = compute_score(f_c)
        if score < best_score:
            best = f_c
            best_score = score
            f = f_c
    return best, best_score


def lp_local_search(f, time_budget, rng=None):
    """LP-based local search with adaptive line search."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    n = len(f)
    f = normalize(f.astype(np.float64))
    best_f = f.copy()
    best_score = compute_score(f)
    t = 0.5
    
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        g = b / b_sum
        
        # Adaptive line search: try multiple t values
        t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]
        best_new_f = f.copy()
        best_new_score = best_score
        
        for tc in t_candidates:
            f_new = (1 - tc) * f + tc * g
            f_new = normalize(f_new)
            score = compute_score(f_new)
            if score < best_new_score:
                best_new_score = score
                best_new_f = f_new
        
        if best_new_score < best_score - 1e-12:
            best_f = best_new_f.copy()
            best_score = best_new_score
            f = best_new_f
            stall = 0
            # Keep the successful t or slightly increase
            t = tc * 1.2 if tc < 1.0 else tc
        else:
            t = max(0.05, t * 0.6)
            stall += 1
            if stall > 40:
                # Try to escape with perturbation
                f_pert = f * (1 + 0.2 * rng.normal(0, 1, n))
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    t = 0.5
                    stall = 0
                elif stall > 80:
                    # Larger perturbation
                    f_pert = f * (1 + 0.5 * rng.normal(0, 1, n))
                    f_pert = normalize(f_pert)
                    if compute_score(f_pert) < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = compute_score(f)
                        t = 0.5
                        stall = 0
                    elif time.monotonic() - start > time_budget * 0.8:
                        break
    
    return best_f, best_score


def de_evolve(pop, scores, F=0.8, CR=0.9, time_budget=0, rng=None, local_iters=50):
    """Differential evolution with embedded fast local search."""
    if rng is None:
        rng = np.random.default_rng()
    NP = len(pop)
    n = len(pop[0])
    
    best_idx = np.argmin(scores)
    best_score = scores[best_idx]
    best_pop = pop[best_idx].copy()
    
    t0 = time.monotonic()
    gen = 0
    
    while time.monotonic() - t0 < time_budget:
        for i in range(NP):
            if time.monotonic() - t0 > time_budget:
                    break
            # Pick 3 distinct random indices different from i
            while True:
                r1, r2, r3 = rng.choice(NP, size=3, replace=False)
                if i not in (r1, r2, r3):
                    break
            
            # Mutation
            v = pop[r1] + F * (pop[r2] - pop[r3])
            v = np.clip(v, 0, None)
            s = v.sum()
            if s < 1e-15:
                continue
            v = v / s
            
            # Crossover (binomial, vectorized)
            mask = rng.random(n) < CR
            j_rand = rng.integers(0, n)
            mask[j_rand] = True
            trial = np.where(mask, v, pop[i])
            trial = normalize(trial)
            
            # Embedded local search (short)
            trial, trial_score = fast_local_search(trial, rng, n_iter=local_iters)
            
            # Selection
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_pop = trial.copy()
        
        gen += 1
        if gen % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DE gen={gen}, best={best_score:.6f}, elapsed={elapsed:.1f}s")
    
    return best_pop, best_score, pop, scores


def generate_population(n, seed=42, parent=None, pop_size=200):
    """Generate a diverse initial population."""
    rng = np.random.default_rng(seed)
    pop = []
    
    # Parent
    if parent is not None and len(parent) == n:
        pop.append(normalize(parent.astype(np.float64)))
    
    # Analytical shapes
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    pop.append(np.ones(n) / n)
    
    # Ramps
    pop.append(normalize(np.arange(1, n + 1, dtype=np.float64)))
    pop.append(normalize(np.arange(n, 0, -1, dtype=np.float64)))
    pop.append(normalize(np.sqrt(np.arange(1, n + 1, dtype=np.float64))))
    pop.append(normalize(np.ones(n) - np.abs(x - n/2) / n))
    
    # Bimodal
    pop.append(normalize(np.exp(-((x - n * 0.25) ** 2) / (n * 0.08)) + np.exp(-((x - n * 0.75) ** 2) / (n * 0.08))))
    
    # Trimodal
    pop.append(normalize(np.exp(-((x - n * 0.2) ** 2) / (n * 0.05)) + np.exp(-((x - n * 0.5) ** 2) / (n * 0.05)) + np.exp(-((x - n * 0.8) ** 2) / (n * 0.05))))
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5]:
        pop.append(normalize((1 + x) ** (-p)))
        pop.append(normalize((n - x) ** (-p)))
    
    # Rectified sinusoids
    for freq in [2, 3, 4, 5, 7, 10]:
        for phase in [0, 0.3]:
            f = np.abs(np.sin(2 * np.pi * freq * x / n + phase))
            pop.append(normalize(f))
    
    # Triangle waves
    for freq in [2, 3, 5]:
        f = 1.0 - np.abs(2 * (x * freq / n % 1.0) - 1.0)
        pop.append(normalize(f))
    
    # Piecewise constant (random segments)
    for _ in range(15):
        n_seg = int(rng.integers(3, 12))
        breaks = np.sort(rng.choice(np.arange(1, n), size=n_seg - 1, replace=False))
        segments = np.split(np.arange(n), breaks)
        f = np.zeros(n)
        for seg in segments:
            f[seg] = rng.random()
        pop.append(normalize(f))
    
    # Random smooth (Gaussian mixture)
    for _ in range(20):
        n_gauss = int(rng.integers(1, 6))
        f = np.zeros(n)
        for _ in range(n_gauss):
            center = rng.random() * n
            width = rng.uniform(0.02, 0.2) * n
            amp = rng.uniform(0.5, 2.0)
            f += amp * np.exp(-((x - center) ** 2) / (2 * width ** 2))
        f = np.clip(f, 0, None)
        if f.sum() > 1e-10:
            pop.append(f / f.sum())
    
    # Fill with random to reach pop_size
    while len(pop) < pop_size:
        f = rng.random(n)
        pop.append(normalize(f))
    
    pop = np.array(pop[:pop_size])
    scores = np.array([compute_score(p) for p in pop])
    
    return pop, scores
