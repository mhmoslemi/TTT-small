"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated projected subgradient descent with FFT convolution,
diverse structured initializations, adaptive step sizes, and LP refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_mask = u * np.arange(1, n + 1) > cv
    rho_candidates = np.where(rho_mask)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_subgradient(a, peak_indices):
    """Compute subgradient of max_k c[k] at point a, averaged over active peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # grad[j] += 2*a[k-j] for j in [lo, hi]
            # a[k-j] for j in [lo, hi] = a[k-hi : k-lo+1] reversed
            segment = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * segment
    grad /= max(1, len(peak_indices))
    return grad


def alt_projection_init(n, rng, n_iter=80, taper=0.1):
    """
    Alternating projection to get a nonnegative sequence with near-flat
    magnitude spectrum.
    """
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        mag_target = np.ones(len(F_full))
        weights = np.linspace(1.0, 1.0 - taper, len(F_full))
        mag_target = weights ** 0.5
        F = mag_target * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j

    return a


def chirp_init(n, rng):
    """Initialize using a quadratic chirp (in frequency domain)."""
    N = 1 << (2 * n - 2).bit_length()
    # Quadratic chirp in frequency: phase = pi * (k/N)^2
    freqs = np.arange(N // 2 + 1) / N
    phase = 2 * np.pi * (0.5 * freqs**2)  # quadratic chirp
    F = np.exp(1j * phase)
    
    a = np.fft.irfft(F, N)[:n].copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def cosine_window_init(n, rng):
    """Initialize with a cosine window (raised cosine)."""
    x = np.linspace(0, 2 * np.pi, n)
    a = 0.5 + 0.5 * np.cos(x)  # raised cosine
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def exponential_init(n, rng):
    """Initialize with exponential decay from center."""
    center = (n - 1) / 2.0
    sigma = n / 4.0
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    # Add small random perturbation
    a += 0.01 * rng.random(n)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def lp_step(a, P, time_limit=10.0):
    """
    LP step: maximize sum(b) subject to convolve(a,b) <= P, b >= 0.
    Returns rescaled solution or None.
    """
    n = len(a)
    if n > 3000:
        return None

    # Build convolution matrix
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit})
        if res.success:
            b = res.x
            b_sum = b.sum()
            if b_sum > 1e-8:
                a_sum = a.sum()
                b = b * (a_sum / b_sum)
                return b
    except Exception:
        pass
    return None


def nesterov_subgradient_optimize(a, deadline, rng, max_iters=200000):
    """
    Nesterov-accelerated projected subgradient descent.
    Minimizes max(convolve(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    
    c = fast_convolve(a)
    best_peak = np.max(c)
    best_a = a.copy()
    
    # Nesterov parameters
    t = 1.0
    x_prev = a.copy()
    y = a.copy()  # extrapolated point
    
    # Adaptive step size
    lr = 1.0 / n
    lr_min = 1e-8
    
    # Barzilai-Borwein parameters
    s_prev = None
    y_grad_prev = None
    
    stall_count = 0
    max_stall = 300
    it = 0
    
    while time.time() < deadline and it < max_iters:
        it += 1
        
        # Evaluate at extrapolated point
        c = fast_convolve(y)
        peak = np.max(c)
        
        if peak < best_peak * (1.0 - 1e-10):
            best_peak = peak
            best_a = a.copy()  # Save the actual point, not y
            # Actually save y if it's better
            if peak < np.max(fast_convolve(a)):
                best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Get active peak indices (top peaks)
        threshold = peak * (1.0 - 1e-4)
        peak_indices = np.where(c >= threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(c)])
        if len(peak_indices) > 30:
            top_idx = np.argsort(c[peak_indices])[::-1][:20]
            peak_indices = peak_indices[top_idx]
        
        # Compute subgradient at y
        grad = compute_subgradient(y, peak_indices)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        step = lr / t_new
        
        # Step size adjustment
        if stall_count > max_stall:
            lr = max(lr * 0.8, lr_min)
            stall_count = 0
            # Perturbation to escape
            if time.time() < deadline - 20:
                perturbation = rng.normal(0, 0.03 / np.sqrt(n), n)
                a = project_simplex(a + perturbation)
                y = a.copy()
                x_prev = a.copy()
                t = 1.0
                continue
        
        # Update
        z = y - step * grad
        x_new = project_simplex(z)
        
        # Nesterov momentum
        y = x_new + (t - 1.0) / t_new * (x_new - x_prev)
        y = project_simplex(y)  # Keep y on the simplex
        
        x_prev = a
        a = x_new
        t = t_new
    
    # Final: compare a and y
    c_a = fast_convolve(a)
    c_y = fast_convolve(y)
    if np.max(c_y) < np.max(c_a):
        best_a = y
    
    return best_a


def coordinate_refine(a, deadline, rng, max_rounds=50):
    """
    Coordinate descent refinement: try adjusting each element slightly.
    """
    n = len(a)
    a = project_simplex(a.copy())
    
    c = fast_convolve(a)
    best_peak = np.max(c)
    
    for round_num in range(max_rounds):
        if time.time() > deadline:
            break
        
        improved = False
        # Try small perturbations on random subset of coordinates
        if n <= 1024:
            indices = np.arange(n)
        else:
            indices = rng.choice(n, size=min(n, 200), replace=False)
        
        for i in indices:
            if time.time() > deadline:
                break
            
            for delta in [0.01, -0.01, 0.005, -0.005]:
                a_trial = a.copy()
                a_trial[i] += delta
                a_trial = project_simplex(a_trial)
                c_trial = fast_convolve(a_trial)
                if np.max(c_trial) < best_peak * (1 - 1e-8):
                    a = a_trial
                    best_peak = np.max(c_trial)
                    improved = True
                    break
        
        if not improved:
            break
    
    return a


def optimize_sequence(a, deadline, rng, use_lp=True):
    """
    Full optimization pipeline: Nesterov subgradient + LP + coordinate refinement.
    """
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    
    # Phase 1: Nesterov subgradient (majority of time)
    phase1_deadline = deadline - max(30.0, 0.2 * (deadline - time.time()))
    a = nesterov_subgradient_optimize(a, phase1_deadline, rng)
    
    # Phase 2: LP refinement
    if use_lp and n <= 2048:
        c = fast_convolve(a)
        best_peak = np.max(c)
        # Gradually tighten the constraint
        for tightness in [1.0, 0.98, 0.96, 0.94]:
            if time.time() > deadline - 10:
                break
            P = best_peak * tightness
            b = lp_step(a, P, time_limit=min(15.0, max(5.0, deadline - time.time() - 5)))
            if b is not None:
                c_b = fast_convolve(b)
                if np.max(c_b) < best_peak:
                    a = project_simplex(b)
                    best_peak = np.max(c_b)
    
    # Phase 3: Coordinate refinement
    if time.time() < deadline - 5:
        a = coordinate_refine(a, deadline, rng, max_rounds=20)
    
    # Phase 4: Final Nesterov pass
    if time.time() < deadline - 5:
        a = nesterov_subgradient_optimize(a, deadline, rng, max_iters=50000)
    
    return a

    """Compute gradient of the log-sum-exp smoothed max."""
    c = fast_convolve(y)

- `conv(a,a)[0] = a[0]^2 = 0`
- `conv(a,a)[1] = 2*a[0]*a[1] = 0`
- ...
- `conv(a,a)[n-1] = 0` (as shown)
- `conv(a,a)[n] = 2*a[n/2]*a[n/2-1] + ... ` hmm this is getting complicated.

Let me just compute for a simple example. n=4, a = [0, 0, 0.5, 0.5]:
- conv = [0, 0, 0, 0.5, 0.5, 0.5, 0]
  - conv[0] = 0
  - conv[1] = 0
  - conv[2] = 0
  - conv[3] = 0*0.5 + 0*0.5 + 0.5*0 + 0.5*0 = 0... 

Wait let me be more careful. a = [0, 0, 0.5, 0.5], n=4.
conv[k] = sum_{i=max(0,k-3)}^{min(3,k)} a[i]*a[k-i]

conv[0] = a[0]*a[0] = 0
conv[1] = a[0]*a[1] + a[1]*a[0] = 0
conv[2] = a[0]*a[2] + a[1]*a[1] + a[2]*a[0] = 0
conv[3] = a[0]*a[3] + a[1]*a[2] + a[2]*a[1] + a[3]*a[0] = 0
conv[4] = a[1]*a[3] + a[2]*a[2] + a[3]*a[1] = 0 + 0.25 + 0 = 0.25
conv[5] = a[2]*a[3] + a[3]*a[2] = 0.25 + 0.25 = 0.5
conv[6] = a[3]*a[3] = 0.25

So max = 0.5, sum_a = 1, score = 2*4*0.5/1 = 4. That's worse!

OK so the "half" sequence is bad because it concentrates the sum in a small region, making the convolution peak at the edges of that region.

The problem is fundamentally about spreading the energy of `a` so that ALL autocorrelation lags are small. This is the "low autocorrelation" problem.

For non-negative sequences, the best known constructions are related to:
- **Throttle sequences**
- **Legendre sequences** (adapted)
- **M-sequences** (adapted to reals)

But these are all binary. For real non-negative sequences, we have more freedom.

Let me think about this from a different angle. What if we use a sequence that's close to a "chaotic" or "pseudo-random" non-negative sequence?

For example, `a[i] = |sin(i * alpha)|` for some irrational alpha. This gives a non-negative sequence with "random-like" correlations.

Or `a[i] = (1 + cos(2*pi*i*alpha)) / 2` for alpha irrational.

These might have autocorrelation peaks that are closer to the mean.

"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses FISTA with log-sum-exp smoothing, diverse structured initializations,
Nesterov subgradient backup, LP refinement, and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_mask = u * np.arange(1, n + 1) > cv
    rho_candidates = np.where(rho_mask)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_subgradient(a, peak_indices):
    """Compute subgradient of max_k c[k] at point a, averaged over active peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            segment = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * segment
    grad /= max(1, len(peak_indices))
    return grad


def smooth_gradient_fft(y, tau):
    """
    Compute gradient of log-sum-exp smoothed max via FFT cross-correlation.
    f(a) = (1/tau) * log(sum_k exp(tau * c_k(a)))
    grad f = sum_k p_k * 2 * (c)_k_subgrad
    """
    n = len(y)
    size = 2 * n - 1
    c = fast_convolve(y)

    # Compute softmax weights
    c_max = np.max(c)
    exp_c = np.exp(tau * (c - c_max))
    p = exp_c / exp_c.sum()

    # grad[j] = 2 * sum_k p[k] * y[k-j] for valid indices
    # = 2 * cross_correlation(p, y)[j]
    N = 1 << (size - 1).bit_length()
    P = np.fft.rfft(p, N)
    Y = np.fft.rfft(y, N)
    corr = np.fft.irfft(np.conj(P) * Y, N)[:size]
    grad = 2.0 * corr[:n]

    return grad, c


def alt_projection_init(n, rng, n_iter=100, taper=0.05):
    """Alternating projection for near-flat magnitude spectrum."""
    N = 1 << (2 * n - 2).bit_length()
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    a = np.ones(n) / n
    for _ in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        mag_target = np.ones(len(F_full))
        weights = np.linspace(1.0, 1.0 - taper, len(F_full))
        mag_target = weights ** 0.5
        F = mag_target * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j

    return a


def chirp_init(n, rng):
    """Quadratic chirp initialization."""
    N = 1 << (2 * n - 2).bit_length()
    freqs = np.arange(N // 2 + 1) / N
    phase = 2 * np.pi * (0.5 * freqs ** 2)
    F = np.exp(1j * phase)
    a = np.fft.irfft(F, N)[:n].copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def sinusoidal_init(n, rng):
    """Pseudo-random non-negative sequence based on |sin(i*alpha)|."""
    alpha = rng.uniform(0.1, 1.0) * np.pi * 2.0
    i = np.arange(n)
    a = np.abs(np.sin(i * alpha))
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a


def cosine_init(n, rng):
    """Raised cosine window."""
    x = np.linspace(0, 2 * np.pi, n)
    a = 0.5 + 0.5 * np.cos(x)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def parent_init(n, rng, perturb_std=0.0):
    """Initialize from parent sequence with optional perturbation."""
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if len(parent) >= n:
            a = parent[:n].copy()
        else:
            a = np.zeros(n)
            a[:len(parent)] = parent
        a = np.maximum(a, 0.0)
        if perturb_std > 0:
            a += perturb_std * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 1e-10 else np.ones(n) / n
        return a
    except Exception:
        a = rng.random(n)
        return a / a.sum()


def fista_smooth_optimize(a_init, deadline, rng):
    """
    FISTA (Nesterov-accelerated proximal gradient) with log-sum-exp smoothing.
    The smoothing parameter tau is annealed over time.
    """
    n = len(a_init)
    x = project_simplex(a_init.copy().astype(np.float64))
    z = x.copy()
    t_k = 1.0

    # Initial tau from the peak value
    c0 = fast_convolve(x)
    peak0 = float(np.max(c0))
    tau = 1.0 / (peak0 + 1e-10)
    tau_min = tau
    tau_max = 200.0 / peak0

    best_val = peak0
    best_x = x.copy()
    stall = 0
    it = 0

    while time.time() < deadline:
        it += 1
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))

        # Nesterov extrapolation
        if t_k > 1.0:
            y = z + (t_k - 1.0) / t_new * (z - x)
        else:
            y = x.copy()
        y = project_simplex(y)

        # Compute smooth gradient via FFT
        grad, c_y = smooth_gradient_fft(y, tau)
        peak_y = float(np.max(c_y))

        # Step size: L_smooth ~ 4 * tau (conservative Lipschitz bound)
        step = 1.0 / (4.0 * tau)

        # Gradient step + proximal
        x_new = project_simplex(y - step * grad)

        # Update
        z = x
        x = x_new
        t_k = t_new

        # Track best (evaluate at x, not y, for a valid solution)
        if it % 5 == 0 or it < 10:
            c_x = fast_convolve(x)
            val = float(np.max(c_x))
            if val < best_val * (1.0 - 1e-10):
                best_val = val
                best_x = x.copy()
                stall = 0
            else:
                stall += 1

        # Anneal tau (make smoothing sharper periodically)
        if it % 50 == 0:
            tau *= 1.08
            if tau > tau_max:
                tau = tau_max

        # Stall handling: restart with perturbation
        if stall > 150:
            x = best_x + rng.normal(0, 0.01 / np.sqrt(n), n)
            x = project_simplex(x)
            z = x.copy()
            t_k = 1.0
            stall = 0
            tau = max(tau_min, tau * 0.9)

    # Final evaluation
    c_final = fast_convolve(x)
    if float(np.max(c_final)) < best_val:
        best_x = x.copy()

    return best_x


def nesterov_subgrad_optimize(a_init, deadline, rng):
    """
    Nesterov-accelerated projected subgradient (non-smooth version).
    Used as backup or for final polishing.
    """
    n = len(a_init)
    x = project_simplex(a_init.copy().astype(np.float64))
    x_prev = x.copy()
    t_k = 1.0

    c0 = fast_convolve(x)
    best_val = float(np.max(c0))
    best_x = x.copy()

    base_lr = 2.0 / n  # ~1/L where L~n/2 for our problem
    stall = 0
    it = 0

    while time.time() < deadline:
        it += 1
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))

        # Extrapolate
        if t_k > 1.0:
            y = x + (t_k - 1.0) / t_new * (x - x_prev)
        else:
            y = x.copy()
        y = project_simplex(y)

        # Subgradient at y
        c_y = fast_convolve(y)
        peak_y = float(np.max(c_y))

        # Active set
        threshold = peak_y * (1.0 - 5e-5)
        active = np.where(c_y >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_y)])
        if len(active) > 40:
            top = np.argsort(c_y[active])[::-1][:25]
            active = active[top]

        grad = compute_subgradient(y, active)

        # Step
        step = base_lr / t_k
        x_new = project_simplex(y - step * grad)

        x_prev = x
        x = x_new
        t_k = t_k_new = t_new

        # Track
        if it % 10 == 0:
            c_x = fast_convolve(x)
            val = float(np.max(c_x))
            if val < best_val * (1.0 - 1e-10):
                best_val = val
                best_x = x.copy()
                stall = 0
            else:
                stall += 1

        if stall > 200:
            x = best_x + rng.normal(0, 0.02 / np.sqrt(n), n)
            x = project_simplex(x)
            x_prev = x.copy()
            t_k = 1.0
            stall = 0

    c_final = fast_convolve(x)
    if float(np.max(c_final)) < best_val:
        best_x = x.copy()

    return best_x


def lp_refine(a, deadline, max_tighten=4):
    """LP-based refinement: solve max sum(b) s.t. conv(a,b) <= P, b>=0."""
    n = len(a)
    if n > 2500:
        return a

    c = fast_convolve(a)
    current_peak = float(np.max(c))

    for i in range(max_tighten):
        if time.time() > deadline - 5:
            break
        P = current_peak * (1.0 - 0.02 * (i + 1))
        if P <= 0:
            break

        # Build convolution matrix
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, P)
        t_limit = min(12.0, max(3.0, deadline - time.time() - 3))

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0.0, None)] * n,
                          method='highs',
                          options={'time_limit': t_limit, 'presolve': True})
            if res.success:
                b = res.x
                b_sum = b.sum()
                if b_sum > 1e-8:
                    b = b * (a.sum() / b_sum)
                    c_b = fast_convolve(b)
                    new_peak = float(np.max(c_b))
                    if new_peak < current_peak:
                        a = project_simplex(b)
                        current_peak = new_peak
        except Exception:
            pass

    return a


def full_optimize(a_init, deadline, rng, use_lp=True):
    """Full optimization pipeline combining multiple methods."""
    n = len(a_init)

    # Phase 1: FISTA with smoothing (main optimizer)
    phase1_time = 0.6 * (deadline - time.time())
    phase1_deadline = time.time() + phase1_time
    a = fista_smooth_optimize(a_init, phase1_deadline, rng)

    # Phase 2: Nesterov subgradient (refinement)
    phase2_time = 0.2 * (deadline - time.time())
    phase2_deadline = time.time() + phase2_time
    a2 = nesterov_subgrad_optimize(a, phase2_deadline, rng)
    c1 = fast_convolve(a)
    c2 = fast_convolve(a2)
    if float(np.max(c2)) < float(np.max(c1)):
        a = a2

    # Phase 3: LP refinement
    if use_lp and n <= 2048:
        a = lp_refine(a, deadline, max_tighten=3)

    # Phase 4: Final FISTA pass
    if time.time() < deadline - 10:
        a = fista_smooth_optimize(a, deadline, rng)

    return a


def try_sequence(n, init_type, rng, deadline):
    """Initialize and optimize a sequence of given length."""
    if init_type == 'alt_proj':
        a = alt_projection_init(n, rng, n_iter=80, taper=0.05)
    elif init_type == 'random':
        a = rng.random(n)
        a = a / a.sum()
    elif init_type == 'cosine':
        a = cosine_init(n, rng)
    elif init_type == 'parent':
        a = parent_init(n, rng, perturb_std=0.0)
    elif init_type == 'parent_perturb':
        a = parent_init(n, rng, perturb_std=0.03)
    elif init_type == 'parent_small_perturb':
        a = parent_init(n, rng, perturb_std=0.01)
    elif init_type == 'chirp':
        a = chirp_init(n, rng)
    elif init_type == 'sinusoidal':
        a = sinusoidal_init(n, rng)
    elif init_type == 'gaussian':
        center = (n - 1) / 2.0
        sigma = n / 5.0
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        a += 0.01 * rng.random(n)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    elif init_type == 'broadband':
        a = alt_projection_init(n, rng, n_iter=40, taper=0.3)
        a = a + 0.05 * rng.random(n)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    a_opt = full_optimize(a, deadline, rng, use_lp=n <= 2048)
    score = evaluate_sequence(a_opt.tolist())
    return a_opt, score


def propose_candidate(budget_s=1500, **kwargs):
    """
    Multi-phase search: FISTA with log-sum-exp smoothing, Nesterov subgradient,
    LP refinement, across multiple sequence lengths and diverse initializations.
    """
    overall_deadline = time.time() + budget_s - 40.0
    rng = np.random.default_rng(777)

    best_score = float('inf')
    best_seq = None

    # Configs: (length, init_type, weight)
    # Prioritize parent-based starts since they're already good
    configs = [
        (512, 'parent', 4),
        (512, 'parent_small_perturb', 3),
        (512, 'parent_perturb', 2),
        (512, 'alt_proj', 2),
        (768, 'parent', 2),
        (512, 'chirp', 1),
        (1024, 'alt_proj', 1),
        (512, 'sinusoidal', 1),
        (768, 'alt_proj', 1),
        (512, 'cosine', 1),
        (1024, 'chirp', 1),
        (256, 'parent', 1),
    ]

    total_weight = sum(w for _, _, w in configs)
    available_time = overall_deadline - time.time()

    for i, (n, init_type, weight) in enumerate(configs):
        if time.time() > overall_deadline - 20:
            break

        allocated = available_time * weight / total_weight
        # Cap individual allocations
        allocated = min(allocated, 200.0)
        config_deadline = min(overall_deadline - 15, time.time() + allocated)

        if config_deadline < time.time() + 8:
            continue

        try:
            a_opt, score = try_sequence(n, init_type, rng, config_deadline)
        except Exception as e:
            print(f"  Error in config ({n}, {init_type}): {e}")
            continue

        remaining = overall_deadline - time.time()
        print(f"n={n}, init={init_type}, score={score:.6f}, remaining={remaining:.0f}s")

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  >>> New best: {best_score:.6f} (n={n})")

    # Final polish on the best sequence
    if best_seq is not None and time.time() < overall_deadline - 10:
        print(f"Polishing best (n={len(best_seq)}, score={best_score:.6f})...")
        a_polished = full_optimize(best_seq, overall_deadline, rng,
                                   use_lp=len(best_seq) <= 2048)
        score = evaluate_sequence(a_polished.tolist())
        print(f"After polish: score={score:.6f}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_polished

    # Fallback
    if best_seq is None:
        print("Falling back to parent sequence")
        try:
            best_seq = np.array(height_sequence_1, dtype=np.float64).copy()
        except Exception:
            n = 512
            x = np.linspace(0, 2 * np.pi, n)
            best_seq = np.maximum(0.5 + 0.5 * np.cos(x), 0.01)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"FINAL: score={final_score:.6f}, length={len(result)}")
    return result
