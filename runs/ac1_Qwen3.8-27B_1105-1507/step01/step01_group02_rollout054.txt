```python
"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses smooth gradient descent with log-sum-exp annealing, multiple 
initializations, and peak-shaving heuristics.
"""

import numpy as np
import time

def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result

def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_obj_grad(a, tau):
    """Compute smooth objective and gradient using log-sum-exp."""
    n = len(a)
    c = fast_convolve(a)
    max_c = np.max(c)
    shifted = tau * (c - max_c)
    # Clip for numerical stability
    shifted = np.maximum(shifted, -50)
    exp_shifted = np.exp(shifted)
    log_sum = max_c + np.log(np.sum(exp_shifted)) / tau
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(n)
    # Only process significant peaks
    thresh = 1e-8
    significant = np.where(weights > thresh)[0]
    for k in significant:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += weights[k] * (a[k-hi:k-lo+1][::-1] + a[lo:hi+1])
    
    return log_sum, grad

def gradient_descent_smooth(a, deadline, rng, tau_init=1.0, tau_final=50.0, 
                            n_anneal=2000, lr_init=0.3, lr_decay=0.9999):
    """Smooth gradient descent with annealing."""
    n = len(a)
    a = project_simplex(a.copy())
    
    best_score = np.inf
    best_a = a.copy()
    
    lr = lr_init
    it = 0
    stall = 0
    
    while time.time() < deadline:
        it += 1
        # Anneal temperature
        progress = min(1.0, it / n_anneal)
        tau = tau_init + (tau_final - tau_init) * progress
        
        obj, grad = compute_smooth_obj_grad(a, tau)
        
        # Gradient step
        a_new = a - lr * grad
        a = project_simplex(a_new)
        
        # Track best (using actual objective)
        c = fast_convolve(a)
        actual_obj = np.max(c)
        if actual_obj < best_score * (1 - 1e-10):
            best_score = actual_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Decay learning rate
        lr *= lr_decay
        if lr < 1e-10:
            break
            
        # Periodic perturbation to escape local minima
        if stall > 500 and time.time() < deadline - 5:
            perturb = rng.normal(0, 0.1 / np.sqrt(n), n)
            a = project_simplex(a + perturb)
            lr = lr_init * 0.5
            stall = 0
    
    return best_a

def peak_shaving(a, deadline, rng, n_iters=5000):
    """
    Direct peak shaving: iteratively reduce the maximum autocorrelation value.
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_peak = np.max(fast_convolve(a))
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        c = fast_convolve(a)
        peak_idx = np.argmax(c)
        peak_val = c[peak_idx]
        
        if peak_val < best_peak * (1 - 1e-10):
            best_peak = peak_val
            best_a = a.copy()
        
        # Compute gradient of c[peak_idx]
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        grad = np.zeros(n)
        grad[lo:hi+1] = a[peak_idx-hi:peak_idx-lo+1][::-1] + a[lo:hi+1]
        
        # Find positions with lowest "responsibility" for the peak
        # (these are where we can add mass without increasing the peak)
        # Actually, let's just do a gradient step with small step size
        step = 0.05 / np.max(grad) if np.max(grad) > 0 else 0.05
        a_new = a - step * grad
        a = project_simplex(a_new)
        
        # Also try a "randomized" peak shaving
        if it % 100 == 0:
            # Pick 2 random positions and swap mass
            if n > 2:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j:
                    delta = 0.01 * a[i]
                    a[j] += delta
                    a[i] -= delta

        return np.ones(n) / n
    elif init_type == 'cosine':

        return a / a.sum()
    elif init_type == 'gaussian':

    elif init_type == 'bimodal':

        return a / a.sum()
    elif init_type == 'random':

    elif init_type == 'anti_peak':

        a = np.maximum(a, 0)
        return a / a.sum()
    elif init_type == 'sinc_like':

        a = np.sinc(x * 2)  # sinc with some spread
        a = np.abs(a) + 1e-6  # Ensure non-negative
        return a / a.sum()
    elif init_type == 'trapezoid':

        a = np.clip(x * 4, 0, 1) * np.clip((1-x) * 4, 0, 1)
        a = np.maximum(a, 1e-6)
        return a / a.sum()

        return np.ones(n) / n
```

```python
def hankel_matvec(weights, v):
    """Compute H @ v where H[i,j] = 2 * weights[i+j]."""
    n = len(v)
    # H[i,j] = 2 * weights[i+j]
    # (H @ v)[i] = 2 * sum_j weights[i+j] * v[j]
    # This is like a convolution/cross-correlation
    result = np.zeros(n)
    for i in range(n):
        # sum_j weights[i+j] * v[j] for j=0..n-1
        # weights[i+j] is valid when i+j < len(weights) = 2n-1
        # So j < 2n-1-i
        hi = min(n, 2*n-1-i)
        if hi > 0:
            result[i] = 2.0 * np.dot(weights[i:i+hi], v[:hi])
    return result
```

```python
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    c = fast_convolve(a)
    best_peak = np.max(c)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        c = fast_convolve(a)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]
        
        if peak_val < best_peak * (1 - 1e-12):
            best_peak = peak_val
            best_a = a.copy()
        
        # Find the pair that contributes most to c[peak_idx]
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        # Contributions: a[i] * a[peak_idx - i] for i in [lo, hi]
        i_range = np.arange(lo, hi + 1)

        max_contrib_idx = i_range[np.argmax(contrib)]
        
        # Reduce a[max_contrib_idx] and add to a position that minimizes the increase in peak
        # The position j that adds the least to c[peak_idx] is one where a[peak_idx - j] is small
        # (and j != peak_idx - j to avoid the diagonal)

        # subject to 0 <= j < n, j != max_contrib_idx
        target_options = np.arange(n)

        # Actually, adding mass to j changes c[peak_idx] by 2*a[peak_idx-j]*delta (if j != peak_idx-j)

        # We want to minimize this increase

        #   Add: +delta * (a[peak_idx - j] + a[j]) (approximately, ignoring second order)

        # For j outside [lo, hi], grad_at[j] = 0 (j doesn't contribute to c[peak_idx])
        
        # Find j that minimizes grad_at[j], j != i*
        grad_at_copy = grad_at.copy()
        grad_at_copy[max_contrib_idx] = np.inf
        target_j = int(np.argmin(grad_at_copy))
        
        # Determine step size
        # We can reduce a[i*] by at most a[i*] (to keep non-negativity)
        # and we want the step to actually reduce c[peak_idx]
        delta = 0.1 * a[max_contrib_idx]
        a[max_contrib_idx] -= delta
        a[target_j] += delta
        a = np.maximum(a, 0)
        a = a / a.sum()  # Re-normalize
    
    return best_a
```
