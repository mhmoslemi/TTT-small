Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolve conditional gradient for structured updates (n<=1024)
2. Soft-max smoothed gradient descent with FFT-accelerated convolution (n up to 8192)

4. Multi-scale: solve at small n, upscale, refine at large n
5. Diverse initializations including parent solution interpolation

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_A(a):
    """Build Toeplitz-like constraint matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A

    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A(a)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),

                options={'presolve': True})
    if r.success and r.x.sum() > 1e-10:

    return None

    n = len(a)
    c = fast_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def run_lp_phase(a, max_iters, deadline):
    """LP-based Frank-Wolfe iterations to reduce convolution peak."""
    n = len(a)
    a = np.clip(a.copy(), 1e-6, None)
    a = a / a.sum()
    best = a.copy()
    best_score = evaluate_safe(a)
    no_progress = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()
        # Geometric mixing schedule: start aggressive, decay gently
        t = max(0.02, 0.6 * (1.0 - it / max_iters) ** 0.5)
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new

        score = evaluate_safe(a)
        if score < best_score - 1e-8:
            best_score = score
            best = a.copy()
            no_progress = 0
        else:
            no_progress += 1
            if no_progress > 30:
                break

    return best, best_score


def compute_softgrad(a, tau):
    """Compute soft-max smoothed objective value and gradient."""
    n = len(a)
    c = fast_conv(a)
    c_s = c / tau
    c_max = c_s.max()
    exp_c = np.exp(c_s - c_max)
    P = exp_c / exp_c.sum()
    obj = 2.0 * n * tau * (c_max + np.log(exp_c.sum()))
    # Gradient: 4n * cross_correlation(P, a)[:n]
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    P_pad = np.zeros(N)
    P_pad[:2 * n - 1] = P
    a_pad = np.zeros(N)
    a_pad[:n] = a
    cross = np.fft.irfft(np.fft.rfft(P_pad, N) * np.conj(np.fft.rfft(a_pad, N)), N)
    grad = 4.0 * n * cross[:n]
    return obj, grad


def run_gradient_phase(a, deadline, tau_start=0.05, tau_end=0.002, n_iter=500):
    """Soft-max smoothed gradient descent with momentum."""
    n = len(a)
    a = np.clip(a.copy(), 0.0, None)
    a = a / a.sum()
    best = a.copy()
    best_score = evaluate_safe(a)
    v = np.zeros(n)  # momentum
    lr_base = 1.0 / (4.0 * n)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, n_iter - 1)
        # Quadratic annealing: stay large longer
        tau = tau_start * (tau_end / tau_start) ** (frac ** 1.5)

        obj, grad = compute_softgrad(a, tau)
        lr = lr_base / (1.0 + 0.05 * it)
        # Nesterov-like momentum
        v = 0.9 * v + lr * grad
        a = a - v
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        score = evaluate_safe(a)
        if score < best_score:
            best_score = score
            best = a.copy()

    return best, best_score


def run_subgrad_phase(a, max_iters, deadline):
    """Direct subgradient peak-shaving with decreasing step size."""
    n = len(a)
    a = np.clip(a.copy(), 0.0, None)
    a = a / a.sum()
    best = a.copy()
    best_score = evaluate_safe(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        k_star = int(c.argmax())

        # Subgradient at peak: grad[m] = 2*a[k_star - m] for valid m
        m_lo = max(0, k_star - n + 1)
        m_hi = min(n - 1, k_star)
        grad = np.zeros(n)
        grad[m_lo:m_hi + 1] = 2.0 * a[k_star - m_hi:k_star - m_lo + 1][::-1]

        lr = 0.5 / (it + 1.0)
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        score = evaluate_safe(a)
        if score < best_score:
            best_score = score
            best = a.copy()

    return best, best_score


def upscale_sequence(a, new_n):
    """Upscale sequence via linear interpolation to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    result = np.maximum(np.interp(t_new, t_old, a), 0.0)
    s = result.sum()
    if s > 1e-10:
        result = result / s
    return result


def update_best(a, state):
    """Try to update the best solution tracked in state dict."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 0.01:
        return
    score = evaluate_safe(a)
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best'] = a.copy()
        print(f"  NEW BEST: J={score:.6f} (n={a.size})", flush=True)


def make_starts(rng, n, parent, best):
    """Generate diverse starting points for a given length n."""
    starts = []
    if best is not None:
        starts.append(upscale_sequence(best, n))
    if parent is not None:
        starts.append(upscale_sequence(parent, n))
    starts.append(rng.uniform(0.2, 1.0, n))
    t = np.linspace(0, 1, n, endpoint=False)
    starts.append(1.0 + 0.4 * np.cos(4.0 * np.pi * t))
    starts.append(1.0 + 0.3 * np.cos(6.0 * np.pi * t * t))
    starts.append(1.0 + 0.2 * np.cos(10.0 * np.pi * t))
    starts.append(1.0 + 0.3 * np.sin(2.0 * np.pi * t) * np.cos(4.0 * np.pi * t))
    # Tukey/Hanning window shape
    starts.append(0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5)))
    # "Bump" in the middle
    starts.append(1.0 + 0.5 * np.exp(-4.0 * (t - 0.5) ** 2))
    return starts


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search to minimize J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Uses LP Frank-Wolfe, soft-max gradient descent, and subgradient methods
    across multiple sequence lengths with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    state = {'best': None, 'best_score': float('inf')}

    # Try parent solution
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.clip(p, 0, 1000)
            if p.sum() > 0.01:
                parent = p / p.sum()
                update_best(parent, state)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256.0
        update_best(parent, state)

    print(f"Starting: J={state['best_score']:.6f}, n={state['best'].size}", flush=True)

    # === Phase 1: LP refinement at parent size ===
    n_p = len(parent)
    if n_p <= 1024:
        print(f"\n=== Phase 1: LP at n={n_p} ===", flush=True)
        t0 = time.monotonic()
        a, sc = run_lp_phase(parent, max_iters=200, deadline=deadline)
        update_best(a, state)
        # Gradient polish after LP
        if time.monotonic() < deadline:
            a2, sc2 = run_gradient_phase(a, deadline, tau_start=0.05, tau_end=0.003, n_iter=300)
            update_best(a2, state)
        print(f"  Done: {time.monotonic()-t0:.1f}s, best={state['best_score']:.6f}", flush=True)

    # === Phase 2: LP at n=1024 ===
    if time.monotonic() < deadline:
        n2 = 1024
        print(f"\n=== Phase 2: LP at n={n2} ===", flush=True)
        t0 = time.monotonic()
        a_start = upscale_sequence(state['best'], n2)
        a, sc = run_lp_phase(a_start, max_iters=100, deadline=deadline)
        update_best(a, state)
        # Gradient polish
        if time.monotonic() < deadline:
            a2, sc2 = run_gradient_phase(a, deadline, tau_start=0.04, tau_end=0.002, n_iter=400)
            update_best(a2, state)
            # Subgradient polish
            if time.monotonic() < deadline:
                a3, sc3 = run_subgrad_phase(a2, max_iters=200, deadline=deadline)
                update_best(a3, state)
        print(f"  Done: {time.monotonic()-t0:.1f}s, best={state['best_score']:.6f}", flush=True)

    # === Phase 3: Gradient + Subgradient at larger sizes ===
    for n in [2048, 4096, 8192]:
        if time.monotonic() > deadline:
            break
        print(f"\n=== Phase 3: Gradient at n={n} ===", flush=True)
        t0 = time.monotonic()

        starts = make_starts(rng, n, parent, state['best'])

        for idx, a0 in enumerate(starts):
            if time.monotonic() > deadline:
                break
            a0 = np.clip(a0, 0.01, None)
            a0 = a0 / a0.sum()

            # Soft-max gradient
            a, sc = run_gradient_phase(a0, deadline,
                                       tau_start=0.05 / (1 + 0.1 * idx),
                                       tau_end=0.001,
                                       n_iter=400)
            update_best(a, state)

            # Subgradient follow-up
            if time.monotonic() < deadline:
                a2, sc2 = run_subgrad_phase(a, max_iters=150, deadline=deadline)
                update_best(a2, state)

        print(f"  Done: {time.monotonic()-t0:.1f}s, best={state['best_score']:.6f}", flush=True)

    # === Phase 4: Final polish ===
    if state['best'] is not None and time.monotonic() < deadline:
        n_best = state['best'].size
        print(f"\n=== Phase 4: Final Polish n={n_best} ===", flush=True)
        t0 = time.monotonic()

        # Multiple subgradient restarts from the best
        for attempt in range(3):
            if time.monotonic() > deadline:
                break
            a0 = state['best'].copy()
            if attempt > 0:
                # Add small perturbation
                noise = rng.normal(0, 0.01, a0.size) * a0
                a0 = np.maximum(a0 + noise, 0)
                a0 = a0 / a0.sum()
            a, sc = run_subgrad_phase(a0, max_iters=300, deadline=deadline)
            update_best(a, state)

        # Gradient polish
        if time.monotonic() < deadline:
            a, sc = run_gradient_phase(state['best'], deadline,
                                       tau_start=0.02, tau_end=0.001, n_iter=200)
            update_best(a, state)

        # LP if small enough
        if n_best <= 512 and time.monotonic() < deadline:
            a, sc = run_lp_phase(state['best'], max_iters=80, deadline=deadline)
            update_best(a, state)

        print(f"  Done: {time.monotonic()-t0:.1f}s, best={state['best_score']:.6f}", flush=True)

    # Final verification
    if state['best'] is not None:
        final_list = [float(x) for x in state['best']]
        verified = evaluate_sequence(final_list)
        print(f"\nFINAL VERIFIED: J={verified:.6f}, n={state['best'].size}", flush=True)
        return final_list
    else:
        fallback = [1.0] * 256
        print(f"\nFALLBACK: uniform n=256, J={evaluate_sequence(fallback):.6f}", flush=True)
        return fallback
