Now, j in I_k means 0<=j<n and 0<=k-j<n. Since 0<=j<n is always true (it's our index), this means 0<=k-j<n, i.e., k-n+1 <= j <= k.

k-j in I_k means 0<=k-j<n and 0<=k-(k-j)<n, i.e., 0<=k-j<n and 0<=j<n. Again 0<=j<n is always true, so this means 0<=k-j<n.

So both conditions are the same: 0<=k-j<n.

Therefore: ∂r_k/∂a[j] = 2*a[k-j] if 0<=k-j<n, else 0.

Great, so the gradient is:
∇r_k = 2 * shift(a, k) (with zero padding)

where shift(a, k)[j] = a[k-j] if 0<=k-j<n else 0.

This is essentially a reversed and shifted version of a!

So for the subgradient of J:
∇J[j] = (2n) * (1/|K*|) * sum_{k in K*} 2 * a[k-j] (if 0<=k-j<n)

Where K* is the set of indices where r_k achieves its maximum.

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Direct LP alternating: min t s.t. (a*x)_k <= t, sum(x)=1, x>=0
   (monotonically decreases the convolution peak)
2. Dual LP (max sum) for diversity and escape from local optima
3. L-BFGS-B gradient refinement with FFT-accelerated convolution
4. Multi-length search (512, 1024, 2048) with diverse initializations
5. Subgradient-based diversification for the max-convex objective
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_conv_matrix(g, n):
    """Build matrix A (2n-1 x n) where (A @ x)[k] = (g*x)[k] for k=0..2n-2."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = g[k - jh:k - jl + 1][::-1]
    return A


def lp_min_peak(g, n):
    """
    LP: min t s.t. (g*x)_k <= t for all k, sum(x)=1, x>=0.
    Returns (x, t) or None. This directly minimizes the convolution peak.
    """
    A = build_conv_matrix(g, n)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        x = np.maximum(res.x[:n], 0.0)
        s = x.sum()
        if s > 1e-10:
            x /= s
        return x, float(res.x[-1])
    return None


def lp_max_sum(g, M, n):
    """
    LP: max sum(x) s.t. (g*x)_k <= M for all k, x>=0.
    Returns normalized x or None.
    """
    A = build_conv_matrix(g, n)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        x = np.maximum(res.x, 0.0)
        s = x.sum()
        if s > 1e-10:
            x /= s
            return x
    return None


def subgradient_step(a, lr):
    """One subgradient descent step on J(a) = 2n * max_k (a*a)_k (with sum=1)."""
    n = len(a)
    b = fast_conv(a)
    peak = b.max()
    active = np.where(b >= peak - 1e-12)[0]
    grad = np.zeros(n)
    for k in active:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += 2.0 * a[idx]
    grad /= len(active)
    grad *= 2.0 * n
    # Subtract mean to counteract normalization
    grad -= np.mean(grad)
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    return a_new


def make_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            p = np.clip(parent, 0, None)
            if p.sum() > 0.01:
                inits.append(p / p.sum())
        else:
            t_old = np.linspace(0, 1, len(parent))
            p = np.maximum(np.interp(t, t_old, np.clip(parent, 0, None)), 0)
            if p.sum() > 0.01:
                inits.append(p / p.sum())

    inits.append(np.ones(n) / n)
    inits.append((1.0 + 0.3 * np.cos(2 * np.pi * t)) )
    inits.append((1.0 + 0.5 * np.cos(4 * np.pi * t)))
    inits.append(np.exp(-8.0 * (t - 0.5) ** 2))
    inits.append(np.sin(np.pi * t) ** 2)
    inits.append(1.0 + 0.4 * np.cos(6 * np.pi * t) + 0.2 * np.cos(10 * np.pi * t))
    inits.append(rng.uniform(0.05, 1.0, n))
    inits.append(rng.uniform(0.0, 1.0, n) ** 2)

    # Normalize all
    result = []
    for a in inits:
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.01:
            result.append(a / s)
    return result


def run_lp_alternating(a, n, max_iters, deadline, callback=None):
    """Run LP alternating iterations. Returns best a found."""
    best_a = a.copy()
    best_peak = fast_conv(a).max()
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Primary LP: min peak
        result = lp_min_peak(a, n)
        if result is not None:
            x, t_new = result
            if t_new < best_peak - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            a = x
            if t_new < best_peak:
                best_peak = t_new
                best_a = a.copy()
                if callback:
                    callback(a)
        else:
            break

        # Periodically use dual LP for diversity
        if it % 30 == 15:
            b = fast_conv(a)
            M = b.max() * 1.01
            y = lp_max_sum(a, M, n)
            if y is not None:
                a = 0.5 * a + 0.5 * y
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 0.01:
                    a /= s

        # Early stop if stalled
        if stall_count > 50:
            break

    return best_a, best_peak


def run_gradient_refine(a, n, deadline, callback=None):
    """L-BFGS-B refinement of the objective."""
    a0 = np.clip(a, 0.0, None)
    s = a0.sum()
    if s < 0.01:
        return a

    def _obj(x):
        ax = np.maximum(x, 0.0)
        s_ = ax.sum()
        if s_ < 1e-8:
            return 1e10
        ax = ax / s_
        b = fast_conv(ax)
        return float(b.max())

    res = minimize(_obj, a0, method='L-BFGS-B',
                   bounds=[(0, None)] * n,
                   options={'maxiter': 500, 'max_fun': 20000,
                            'ftol': 1e-16, 'disp': False})
    if res.fun < 1e10:
        if callback:
            callback(res.x)
        return res.x
    return a
