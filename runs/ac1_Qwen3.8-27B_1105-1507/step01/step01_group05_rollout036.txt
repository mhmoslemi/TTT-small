def compute_score(a):

    n = len(a)

def softmax(theta):
    """Convert theta to probability vector via softmax."""
    theta_shifted = theta - np.max(theta)
    a = np.exp(theta_shifted)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_objective_and_grad(theta, tau):
    """
    Compute log-sum-exp smoothed objective and its gradient w.r.t. theta.
    Objective: 2n * (c_max + log(sum(exp(c-c_max)))/tau)
    where c = convolve(a,a), a = softmax(theta), sum(a)=1.
    """
    a = softmax(theta)
    n = len(a)
    c = np.convolve(a, a)
    c_max = np.max(c)
    exp_c = np.exp(c - c_max)
    sum_exp = np.sum(exp_c)

    grad_a = 2.0 * corr[n - 1:n - 1 + n]

            best_score = new_score
            a = a_new
            stall = 0
        else:
            t *= 0.7
            stall += 1
            if stall > 30:
                # Random perturbation to escape
                a_pert = a + rng.normal(0, 0.05, n)
                a_pert = np.clip(a_pert, 0, None)
                if a_pert.sum() > 1e-10:
                    a_pert = a_pert / a_pert.sum()
                    if compute_score(a_pert) < best_score:
                        best_a = a_pert.copy()
                        best_score = compute_score(a_pert)
                        a = a_pert
                t = 0.3
                stall = 0

    return best_a, best_score


def simulated_annealing(a, n_iter=500, time_limit=30.0, rng=None):
    """Simulated annealing with block perturbations."""
    if rng is None:
        rng = np.random.default_rng(1)

    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    current_a = a.copy()
    current_score = best_score
    start = time.monotonic()

    T_init = 0.1 * (best_score - 1.0) + 0.01
    T_min = 0.001
    cooling = (T_min / T_init) ** (1.0 / max(1, n_iter))

    T = T_init
    for it in range(n_iter):
        if time.monotonic() - start > time_limit:
            break

        candidate = current_a.copy()
        # Perturb random blocks
        n_blocks = rng.integers(1, 4)
        for _ in range(n_blocks):
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            pos = int(rng.integers(0, n - width))
            candidate[pos:pos + width] *= np.exp(rng.normal(0, 0.4))
        candidate = np.clip(candidate, 0, None)
        S = candidate.sum()
        if S < 1e-10:
            T *= cooling
            continue
        candidate = candidate / S

        cand_score = compute_score(candidate)
        delta = cand_score - current_score

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_a = candidate
            current_score = cand_score
            if cand_score < best_score:
                best_a = candidate.copy()
                best_score = cand_score

        T *= cooling

    return best_a, best_score


def local_search(a, n_iter=2000, time_limit=30.0, rng=None):
    """Greedy random local search with Gaussian block perturbations."""
    if rng is None:
        rng = np.random.default_rng(2)

    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    start = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - start > time_limit:
            break

        candidate = best_a.copy()
        width = max(1, int(rng.integers(1, max(2, n // 16))))
        pos = int(rng.integers(0, n - width))
        candidate[pos:pos + width] *= np.exp(rng.normal(0, 0.3))
        candidate = np.clip(candidate, 0, None)
        S = candidate.sum()
        if S < 1e-10:
            continue
        candidate = candidate / S

        score = compute_score(candidate)
        if score < best_score:
            best_a = candidate.copy()
            best_score = score

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Combines L-BFGS-B, LP refinement,
    simulated annealing, and local search across diverse initializations
    and sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0

    best_overall = None
    best_score_overall = np.inf

    # Get parent sequence if available
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    # Define initializations: (name, n, initial_array)
    initializations = []

    for n in [256, 384, 512]:
        x = np.arange(n, dtype=np.float64)
        # Uniform with noise
        initializations.append(('uniform', n,
                                np.ones(n) * (1 + 0.1 * rng.normal(0, 1, n))))
        # Gaussian shape
        initializations.append(('gauss', n,
                                np.exp(-0.5 * ((x - n / 2) / (n / 4)) ** 2) + 0.005))
        # Cosine bump
        initializations.append(('cos', n,
                                1.0 + 0.4 * np.cos(2 * np.pi * x / n) + 0.01))
        # Two-lobe
        initializations.append(('tlobe', n,
                                1.0 + 0.5 * np.cos(4 * np.pi * x / n) + 0.1))
        # Random
        initializations.append(('rand', n, rng.random(n) * 0.9 + 0.05))
        # Sqrt ramp (breaks symmetry)
        initializations.append(('sramp', n,
                                np.sqrt(x + 1) * 0.5 + np.sqrt(n - x + 1) * 0.5))

    # Parent
    if parent is not None and 128 <= len(parent) <= 1024:
        initializations.append(('parent', len(parent), parent.copy()))

    # Sort: parent first, then shorter lengths
    initializations.sort(key=lambda x: (0 if x[0] == 'parent' else 1, x[1]))

    n_init = len(initializations)
    print(f"Total initializations: {n_init}")
    print(f"Available time: {budget_s}s, deadline in {budget_s - 20}s")
    print(f"Target: < 1.5027, Previous best: 1.515592")

    idx = 0
    for name, n, a0 in initializations:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        idx += 1
        # Dynamic time allocation: give more time to earlier configs
        fraction = remaining / (n_init - idx + 1)
        t_here = min(fraction, max(30.0, 60.0))
        t_here = min(t_here, remaining - 10)

        a0 = np.clip(a0, 0, None).astype(np.float64)
        S = a0.sum()
        if S < 1e-10:
            continue
        a0 = a0 / S

        print(f"\n[{idx}/{n_init}] {name} n={n} time={t_here:.0f}s")

        a_opt = a0.copy()
        score = compute_score(a0)
        print(f"  Initial: {score:.6f}")

        # Phase 1: L-BFGS with smooth objective (tau=30)
        t1 = max(10.0, t_here * 0.35)
        a_l, s_l = lbfgs_optimize(a_opt, tau=30.0, maxiter=300, time_limit=t1)
        if s_l < score:
            a_opt, score = a_l, s_l
            print(f"  L-BFGS(tau=30): {score:.6f}")

        # Phase 2: L-BFGS with sharper objective (tau=100)
        t2 = max(10.0, t_here * 0.30)
        remaining2 = deadline - time.monotonic()
        t2 = min(t2, remaining2 - 10)
        if t2 > 5:
            a_l2, s_l2 = lbfgs_optimize(a_opt, tau=100.0, maxiter=300, time_limit=t2)
            if s_l2 < score:
                a_opt, score = a_l2, s_l2
                print(f"  L-BFGS(tau=100): {score:.6f}")

        # Phase 3: LP refinement (only for smaller n where LP is fast)
        t3 = max(5.0, t_here * 0.20)
        remaining3 = deadline - time.monotonic()
        t3 = min(t3, remaining3 - 5)
        if t3 > 5 and n <= 512:
            a_lp, s_lp = lp_refine(a_opt, n_iter=150, time_limit=t3, rng=rng)
            if s_lp < score:
                a_opt, score = a_lp, s_lp
                print(f"  LP refine: {score:.6f}")

        # Phase 4: Simulated annealing
        remaining4 = deadline - time.monotonic()
        t4 = min(15.0, remaining4 * 0.25)
        if t4 > 5:
            a_sa, s_sa = simulated_annealing(a_opt, n_iter=500, time_limit=t4, rng=rng)
            if s_sa < score:
                a_opt, score = a_sa, s_sa
                print(f"  SimAnneal: {score:.6f}")

        # Phase 5: Local search polish
        remaining5 = deadline - time.monotonic()
        t5 = min(10.0, remaining5 * 0.3)
        if t5 > 3:
            a_ls, s_ls = local_search(a_opt, n_iter=3000, time_limit=t5, rng=rng)
            if s_ls < score:
                a_opt, score = a_ls, s_ls
                print(f"  LocalSearch: {score:.6f}")

        if score < best_score_overall:
            best_score_overall = score
            best_overall = a_opt.copy()
            print(f"  ** New best: {score:.6f} **")

    # Additional refinement attempts on the best found
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 30:
        n_best = len(best_overall)
        n_attempts = min(15, int(remaining / 15))
        print(f"\n=== Refinement phase: {n_attempts} attempts on n={n_best} ===")

        for attempt in range(n_attempts):
            if time.monotonic() > deadline:
                break
            remaining_a = deadline - time.monotonic() - 5
            if remaining_a < 10:
                break

            # Create perturbed starting point
            perturb_scale = 0.2 + 0.1 * attempt
            a0 = best_overall * (1 + perturb_scale * rng.normal(0, 1, n_best))
            a0 = np.clip(a0, 0, None)
            if a0.sum() < 1e-10:
                continue
            a0 = a0 / a0.sum()

            tau_choice = [30, 50, 100, 200][attempt % 4]
            t_left = min(remaining_a * 0.6, 30.0)
            a_opt, score = lbfgs_optimize(a0, tau=tau_choice, maxiter=200, time_limit=t_left)

            # Quick local search
            if time.monotonic() < deadline - 5:
                a_opt, score = local_search(a_opt, n_iter=1000,
                                            time_limit=min(5.0, deadline - time.monotonic() - 2),
                                            rng=rng)

            if score < best_score_overall:
                best_score_overall = score
                best_overall = a_opt.copy()
                print(f"  Attempt {attempt}: new best {score:.6f}")

    # Final: try one more LP if time allows on the best
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 20 and len(best_overall) <= 512:
        print(f"\nFinal LP polish ({remaining:.0f}s available)")
        a_final, score_final = lp_refine(best_overall, n_iter=100,
                                         time_limit=min(remaining - 5, 120.0), rng=rng)
        if score_final < best_score_overall:
            best_overall = a_final
            best_score_overall = score_final
            print(f"  LP final: {score_final:.6f}")

    # Verify and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        # Ensure sum is sufficient
        if best_overall.sum() < 0.01:
            best_overall = best_overall / best_overall.sum() * 10.0
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL VERIFIED SCORE: {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        print(f"{'='*50}")
        return final_list

    # Fallback
    print("FALLBACK: returning uniform sequence")
    return [1.0] * 256
