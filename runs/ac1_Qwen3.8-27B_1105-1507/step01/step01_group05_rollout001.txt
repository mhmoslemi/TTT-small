Hybrid optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Combines subgradient descent, LP-based direction finding, and local search.
Uses multiple initializations and sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_score_fast(f):
    """Compute score = 2*n*max(conv)/1^2 for a normalized sequence (sum=1)."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M

def subgradient_step(f, step_size, rng):
    """One step of projected subgradient descent."""
    n = len(f)
    conv = np.convolve(f, f)
    k_star = np.argmax(conv)
    # Subgradient: g_j = 2 * f_{k_star - j} (with out-of-bounds = 0)
    g = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            g[j] = 2.0 * f[idx]
    # Update
    f_new = f - step_size * g
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-15:
        return f
    return f_new / s

def subgradient_descent(f, rng, n_iter=5000, initial_step=0.01):
    """Run subgradient descent with adaptive step size."""
    f = f.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    best_f = f.copy()
    best_score = compute_score_fast(f)

            best_score = new_score
            f = f_new
            step = min(step * 1.05, 0.1)

Actually, a uniform sequence has DFT = [n, 0, 0, ..., 0] (all energy at DC), so |DFT|^2 is NOT flat at all. The convolution of a uniform sequence is a triangle, which is the opposite of flat.

So we want a sequence that spreads its energy across frequencies. This means we want a sequence that looks "random" or "noisy" - not smooth.

This is an important insight! The previous LP approach might be getting stuck because it's trying to smooth things out, but we actually want to add "structure" that spreads the frequency content.

Let me reconsider the LP approach:
- We fix f and solve for g that maximizes sum(g) subject to (f*g)_k <= M.
- The solution g will try to put mass where (f*g) has the most "room" (i.e., where (f*g) is smallest).
- This means g will tend to be concentrated where f has small values.
- This creates an oscillatory pattern in the sequence.

So the LP approach should naturally lead to oscillatory patterns, which is what we want!

Let me think about how to make this work better:
1. Start with a sequence that has some oscillation (not uniform)
2. Use the LP to refine the oscillation pattern
3. Use subgradient descent to fine-tune

For initialization, I'll try:
- Uniform + small random noise
- Alternating pattern (1, 0, 1, 0, ...)
- Sinusoidal modulation
- The parent sequence (height_sequence_1)

Let me also think about the sequence length. For the LP to be tractable:
- n = 256: LP has 256 variables, 511 constraints. Very fast.
- n = 512: LP has 512 variables, 1023 constraints. Still fast.
- n = 1024: LP has 1024 variables, 2047 constraints. Moderate.
- n = 2048: LP has 2048 variables, 4095 constraints. Might be slow.

I'll focus on n = 256 to 1024 for the LP, and use subgradient for larger n.

Hybrid optimization combining LP-based direction finding, subgradient descent,
and structured random search for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
"""

from scipy.optimize import linprog
import time

def compute_score(f):
    """Compute score for sum-normalized sequence."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()

def normalize(f):
    """Normalize to sum=1, clip negatives."""
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s

def subgradient_direction(f):
    """Compute subgradient of max(conv) at f."""
    n = len(f)
    conv = np.convolve(f, f)
    k_star = np.argmax(conv)
    # d(conv)_{k_star} / d(f)_j = f[k_star - j] (if in range)
    # Actually it's 2*f[k_star-j] as derived earlier
    # But for the subgradient of max, we use the gradient of the active constraint
    g = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            g[j] = f[idx]
    return g, k_star

def project_and_normalize(f):
    """Project onto {a >= 0} and normalize to sum=1."""
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s

def subgradient_descent(f, rng, n_iter=10000, step_init=0.01, patience=200):
    """Subgradient descent with adaptive step size."""
    f = normalize(f.copy())
    best_f = f.copy()
    best_score = compute_score(f)
    step = step_init
    no_improve = 0
    
    for it in range(n_iter):
        g, k_star = subgradient_direction(f)
        f_new = f - step * g
        f_new = project_and_normalize(f_new)
        score = compute_score(f_new)
        
        if score < best_score:
            best_f = f_new.copy()
            best_score = score
            f = f_new
            no_improve = 0
            step = min(step * 1.02, 0.05)
        else:
            no_improve += 1
            step *= 0.95
            if no_improve > patience:
                break
    
    return best_f, best_score

def lp_solve(f, M, verbose=False):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    n = len(f)
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[k, j] = f[k - j]
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': 30})
    except Exception as e:
        return None
    
    if not res.success:
        return None
    
    g = np.maximum(res.x, 0)
    if g.sum() < 1e-15:
        return None
    return g / g.sum()

def lp_iterate(f, rng, time_budget, verbose=False):
    """LP-based iteration with adaptive step sizes."""
    f = normalize(f.copy())
    best_f = f.copy()
    best_score = compute_score(f)
    
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        conv = np.convolve(f, f)
        M = conv.max()
        
        g = lp_solve(f, M)
        if g is None:
            stall += 1
            if stall > 10:
                break
            continue
        
        # Try multiple step sizes
        improved = False
        for t in [0.5, 0.3, 0.7, 0.2, 0.9]:
            f_new = normalize((1 - t) * f + t * g)
            score = compute_score(f_new)
            if score < best_score:
                best_f = f_new.copy()
                best_score = score
                f = f_new
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            if verbose and stall % 5 == 0:
                print(f"    LP stall={stall}, score={best_score:.6f}")
            if stall > 30:
                # Random restart
                f = normalize(f * (1 + 0.1 * rng.normal(0, 1, len(f))))
                stall = 0
    
    return best_f, best_score

def block_local_search(f, rng, n_iter=5000):
    """Block-based local search."""
    n = len(f)
    f = normalize(f.copy())
    best_f = f.copy()
    best_score = compute_score(f)
    
    for it in range(n_iter):
        f_new = f.copy()
        # Choose random blocks to modify
        n_blocks = rng.integers(1, max(2, n // 64))
        for _ in range(n_blocks):
            width = int(rng.integers(1, max(2, n // 16)))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.5))
        
        f_new = normalize(f_new)
        score = compute_score(f_new)
        
        if score < best_score:
            best_f = f_new.copy()
            best_score = score
            f = f_new
    
    return best_f, best_score
