def gs_step(a, alpha=0.5):
    """One Gerchberg-Saxton step to flatten the autocorrelation."""
    n = len(a)
    # Zero-pad for linear convolution
    N = 1
    while N < 2*n-1:
        N *= 2
    N = 2*N  # extra padding for numerical stability
    
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    
    # Target: flatter magnitude
    # Use |A|^alpha to compress the dynamic range
    target_mag = mag ** alpha  # alpha < 1 makes it flatter
    
    # New spectrum: keep phase, use target magnitude
    phase = A / (mag + 1e-30)
    A_new = target_mag * phase
    
    # Inverse FFT
    a_new = np.fft.irfft(A_new, N)[:n]
    
    # Enforce non-negativity
    a_new = np.maximum(0, a_new)
    
    # Normalize
    s = a_new.sum()
    if s > 1e-300:
        a_new = a_new / s
    else:
        return a.copy()
    
    return a_new

    """One GS step with flattening exponent alpha."""
    n = len(a)
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    # Avoid log(0)
    mag_safe = np.maximum(mag, 1e-30)
    phase = A / mag_safe
    # Flatten: target magnitude = mag^alpha (alpha < 1 makes flatter)
    log_target = alpha * np.log(mag_safe)
    target_mag = np.exp(log_target)
    A_new = target_mag * phase
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(0.0, a_new)
    return a_new

def softmax_gradient_step(a, T, eta, N=None):
    """One soft-max gradient descent step with multiplicative update."""
    n = len(a)
    if N is None:
        N = 1
        while N < 4*n:
            N *= 2
    
    # Compute convolution
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * np.conj(A), N)[:2*n-1]
    # Note: for real signals, |A|^2 = A * conj(A), and IFFT gives circular autocorrelation
    # With enough zero-padding, this approximates linear autocorrelation
    
    # Actually, let me just use np.convolve for correctness
    c = np.convolve(a, a)
    
    # Soft-max weights
    Tc = T * c
    Tc_max = float(Tc.max())
    exp_Tc = np.exp(Tc - Tc_max)
    p = exp_Tc / float(exp_Tc.sum())
    soft_max_val = (Tc_max + float(np.log(exp_Tc.sum()))) / T
    
    # Gradient: 2 * correlate(p, a, 'valid')
    # Using FFT: correlate(p, a) = ifft(fft(p, Np) * conj(fft(a, Np)))
    Np = 1
    while Np < len(p) + n - 1:
        Np *= 2
    P = np.fft.rfft(p, Np)
    Ap = np.fft.rfft(a, Np)
    corr = np.fft.irfft(P * np.conj(Ap), Np)[:n]
    
    grad = 2.0 * corr
    
    # Normalize gradient
    max_grad = float(np.max(np.abs(grad)))
    if max_grad > 1e-30:
        grad = grad / max_grad
    else:
        return a.copy(), soft_max_val, p
    
    # Multiplicative update
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(0.0, a_new)
    a_new = normalize(a_new)
    if a_new is None:
        return a.copy(), soft_max_val, p
    
    return a_new, soft_max_val, p

def topk_multiplicative_step(a, K, eta):
    """Original top-K multiplicative update (from previous code)."""
    n = len(a)
    c = np.convolve(a, a)
    M = float(c.max())
    if M <= 0 or not np.isfinite(M):
        return a.copy()
    
    K = min(K, 2*n-1)
    active = np.argpartition(c, -K)[-K:]
    w = c[active] / M
    
    h = np.zeros(n, dtype=np.float64)
    for idx in range(active.size):
        k = int(active[idx])
        wk = float(w[idx])
        s = max(0, k - (n - 1))
        e = min(n - 1, k)
        if s > e:
            continue
        i = np.arange(s, e + 1)
        j = k - i
        h[i] += wk * a[j]
        h[j] += wk * a[i]
    
    maxh = float(h.max())
    if maxh <= 0 or not np.isfinite(maxh):
        return a.copy()
    
    meanh = float(h.mean())
    fac = eta * (meanh - h) / maxh
    fac = np.clip(fac, -5.0, 5.0)
    a2 = a * np.exp(fac)
    a2 = np.maximum(0.0, a2)
    return normalize(a2)

"""
Search for low-score non-negative step-function coefficients using:
1. FFT-accelerated soft-max gradient with multiplicative (mirror) descent
2. Gerchberg-Saxton spectral flattening as fast pre-optimizer
3. Top-K multiplicative polishing
4. Deterministic annealing (temperature schedule on soft-max)
5. Multi-start with diverse structured initializations
"""

import time
import numpy as np

def _normalize(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return None
    return a / s

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return float('inf')
    s = float(a.sum())
    if s <= 1e-300:
        return float('inf')
    c = np.convolve(a, a)
    m = float(c.max())
    if not np.isfinite(m):
        return float('inf')
    return 2.0 * n * m / (s * s)

def _softmax_grad(a, T):
    """Compute soft-max gradient direction and soft-max value using FFT."""
    n = len(a)
    c = np.convolve(a, a)
    Tc = T * c
    Tc_max = float(Tc.max())
    exp_Tc = np.exp(Tc - Tc_max)
    p = exp_Tc / float(exp_Tc.sum())
    sm_val = (Tc_max + float(np.log(exp_Tc.sum()))) / T

    # Gradient direction = correlate(p, a, 'valid') via FFT
    Np = 1
    while Np < (2 * n - 1) + n - 1:
        Np *= 2
    P = np.fft.rfft(p, Np)
    A = np.fft.rfft(a, Np)
    corr = np.fft.irfft(P * np.conj(A), Np)[:n]

    return corr, sm_val, p

def _softmax_step(a, T, eta):
    """One soft-max multiplicative gradient step."""
    corr, sm_val, p = _softmax_grad(a, T)
    max_g = float(np.max(np.abs(corr)))
    if max_g > 1e-30:
        grad = corr / max_g
    else:
        return a.copy()
    fac = -eta * grad
    fac = np.clip(fac, -4.0, 4.0)
    a2 = a * np.exp(fac)
    a2 = np.maximum(0.0, a2)
    return _normalize(a2)

def _softmax_optimize(a, max_time, seed):
    """Optimize with soft-max gradient, adaptive T and eta."""
    a = _normalize(a)
    if a is None:
        return None, float('inf')
    n = len(a)
    best = a.copy()
    best_sc = _score(best)
    t0 = time.monotonic()
    deadline = t0 + max_time
    rng = np.random.default_rng(seed)

    T = n * 0.5
    T_target = n * 8.0
    eta = 0.4
    stuck = 0

    while time.monotonic() < deadline:
        a2 = _softmax_step(a, T, eta)
        if a2 is None:
            break
        sc = _score(a2)
        if np.isfinite(sc) and sc < best_sc:
            best = a2.copy()
            best_sc = sc
            a = a2
            stuck = 0
            if T < T_target:
                T *= 1.008
            eta = min(eta * 1.002, 1.5)
        else:
            a = a2
            stuck += 1
            if stuck > 15:
                a = best.copy()
                T = n * rng.uniform(0.3, 3.0)
                eta = rng.uniform(0.15, 0.8)
                stuck = 0
            if eta > 0.08:
                eta *= 0.85
            else:
                noise = np.exp(rng.normal(0, 0.12, n))
                a = best * noise
                a = _normalize(a)
                if a is None:
                    a = best.copy()
                eta = 0.4
                T = n * rng.uniform(0.5, 2.0)
                stuck = 0

    return best, best_sc

def _topk_step(a, K, eta):
    n = len(a)
    c = np.convolve(a, a)
    M = float(c.max())
    if M <= 0 or not np.isfinite(M):
        return a.copy()
    K = min(K, 2 * n - 1)
    K = max(K, 1)
    active = np.argpartition(c, -K)[-K:]
    w = c[active] / M
    h = np.zeros(n, dtype=np.float64)
    for idx in range(active.size):
        k = int(active[idx])
        wk = float(w[idx])
        s = max(0, k - (n - 1))
        e = min(n - 1, k)
        if s > e:
            continue
        i = np.arange(s, e + 1)
        j = k - i
        h[i] += wk * a[j]
        h[j] += wk * a[i]
    maxh = float(h.max())
    if maxh <= 0 or not np.isfinite(maxh):
        return a.copy()
    meanh = float(h.mean())
    fac = eta * (meanh - h) / maxh
    fac = np.clip(fac, -5.0, 5.0)
    a2 = a * np.exp(fac)
    a2 = np.maximum(0.0, a2)
    return _normalize(a2)

def _topk_optimize(a, max_time, seed):

    n = len(a)

    stuck = 0

                eta = max(0.1, eta * 0.6)
                K = max(5, min(2*n-1, int(K * rng.uniform(0.5, 2.0))))
                stuck = 0
            elif eta > 0.08:
                eta *= 0.9

    return best, best_sc

def _gs_step_relax(a, alpha, N, t_relax):
    n = len(a)
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    mag_safe = np.maximum(mag, 1e-30)
    phase = A / mag_safe
    target_mag = mag_safe ** alpha
    A_new = target_mag * phase
    a_gs = np.fft.irfft(A_new, N)[:n]
    a_new = (1.0 - t_relax) * a + t_relax * a_gs
    a_new = np.maximum(0.0, a_new)
    return _normalize(a_new)

def _gs_optimize(a, max_time, seed, alpha_start=0.6, alpha_end=0.95):
    a = _normalize(a)
    if a is None:
        return None, float('inf')
    n = len(a)
    N = 1
    while N < 2 * n:
        N *= 2
    best = a.copy()
    best_sc = _score(best)
    t0 = time.monotonic()
    deadline = t0 + max_time
    rng = np.random.default_rng(seed)
    it = 0
    total_iters = max(100, int(max_time * 200))
    t_relax = 1.0

    while time.monotonic() < deadline:
        frac = min(1.0, it / total_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        alpha += rng.normal(0, 0.02)
        alpha = np.clip(alpha, 0.3, 1.0)
        t_relax = max(0.4, 1.0 - 0.6 * frac)

        a2 = _gs_step_relax(a, alpha, N, t_relax)
        if a2 is None:
            break
        sc = _score(a2)
        if np.isfinite(sc) and sc < best_sc:
            best = a2.copy()
            best_sc = sc
        a = a2
        it += 1

        if it % 200 == 0:
            a = best.copy()
            t_relax = 1.0

    return best, best_sc

def _make_starts(n, seed, parent):
    rng = np.random.default_rng(seed)
    t = np.linspace(0, 1, n)
    starts = []

    if parent is not None:
        p = _normalize(parent)
        if p is not None:
            if p.size == n:
                starts.append(p.copy())
            else:
                old_t = np.linspace(0, 1, p.size)
                res = np.interp(t, old_t, p)
                r = _normalize(res)
                if r is not None:
                    starts.append(r)

    starts.append(np.ones(n) / n)
    tri = np.maximum(0, 1 - 2 * np.abs(t - 0.5))
    starts.append(_normalize(tri))
    starts.append(_normalize(0.5 * (1 + np.cos(2 * np.pi * t))))
    starts.append(_normalize(np.exp(-30 * (t - 0.5) ** 2)))
    starts.append(_normalize(np.abs(t - 0.5) + 0.05))
    starts.append(_normalize(np.exp(-80 * (t - 0.25) ** 2) + np.exp(-80 * (t - 0.75) ** 2)))
    starts.append(_normalize(rng.uniform(0.05, 1.0, n)))

    # Symmetric random
    half = n // 2 + 1
    h = rng.uniform(0.1, 1.0, half)
    if n % 2 == 0:
        sym = np.concatenate([h[:n//2], h[:n//2][::-1]])
    else:
        sym = np.concatenate([h[:n//2], h[n//2], h[:n//2][::-1]])
    starts.append(_normalize(sym))

    starts = [s for s in starts if s is not None and np.isfinite(s).all() and s.size == n]
    return starts
