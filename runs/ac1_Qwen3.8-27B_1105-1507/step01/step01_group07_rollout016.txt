import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smoothed_obj_and_grad(a, beta):
    """Compute smoothed objective and gradient.
    a is assumed to have sum=1 and all non-negative.
    """
    n = len(a)
    conv = np.convolve(a, a)
    # Log-sum-exp for numerical stability
    max_conv = np.max(conv)
    log_sum_exp = max_conv + np.log(np.sum(np.exp(beta * (conv - max_conv))))
    obj = (2.0 * n / beta) * log_sum_exp  # since sum(a)=1, obj = 2n * LSE / sum^2 = 2n * LSE
    
    # Compute softmax weights
    p = np.exp(beta * (conv - max_conv))
    p /= np.sum(p)
    
    # Gradient: d(obj)/da[i] = (2n/beta) * sum_k p[k] * (a[k-i] + a[i-k])
    # This is (2n/beta) * ((p cross-correlate a)[i] + (p * a)[i])
    # Using FFT for efficiency
    # p has length 2n-1, a has length n
    # (p * a)[i] = sum_j p[i-j] * a[j], valid for i = 0 to 2n-2
    # We need this for i = 0 to n-1
    
    # Actually, let's just do it directly for simplicity (O(n^2) which is fine for n up to ~500)
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for k in range(2*n-1):
            # a[k-i] is valid if 0 <= k-i < n
            # a[i-k] is valid if 0 <= i-k < n
            if 0 <= k-i < n:
                s += p[k] * a[k-i]
            if 0 <= i-k < n:
                s += p[k] * a[i-k]
        grad[i] = (2.0 * n / beta) * s
    
    return obj, grad
