"""
LP-based iterative search for nonnegative step functions minimizing
2*n*max(convolve(f,f))/(sum(f))^2.

Uses diverse initializations (oscillatory patterns, Gaussians, random-phase IFFT,
parent sequence) with dual LP formulations (max-sum direction and min-peak direction)

for final improvement. Focuses primarily on n=1024 with optional n=2048 extension.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize(x):
    """Normalize a vector to be nonnegative with sum 1."""
    x = np.asarray(x, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s <= 1e-300:
        return None
    return x / s


def score_norm(x, n):
    """Score for normalized sequence (sum=1): 2*n*max(conv(x,x))."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return 2.0 * n * m


def build_template(n):
    """Build sparse matrix structure for linear convolution with a sequence of length n."""
    m = 2 * n - 1
    ks = np.arange(m, dtype=np.int64)
    js = np.maximum(0, ks - n + 1)
    je = np.minimum(n - 1, ks)
    counts = je - js + 1
    total = int(counts.sum())
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    rows = np.repeat(ks, counts)
    cols = (js[rows] + local).astype(np.int32)
    rows = rows.astype(np.int32)
    A = csr_matrix((np.ones(total, dtype=np.float64), (rows, cols)), shape=(m, n))
    col_data = A.indices.copy()
    return A, col_data, m


def solve_maxsum_lp(A, col_data, f, M, n):
    """LP: max sum(g) s.t. f*g <= M, g >= 0."""
    m = A.shape[0]
    A_effects = A.copy()
    A_effects.data[:] = f[col_data]
    c = -np.ones(n, dtype=np.float64)
    b = np.full(m, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A_effects, b_ub=b, bounds=bounds, method="highs")
    return res


def solve_minpeak_lp(A, col_data, f, n):
    """LP: min t s.t. f*g <= t, sum(g)=1, g>=0. Returns (g, t_opt)."""
    m = A.shape[0]
    A_effects = A.copy()
    A_effects.data[:] = f[col_data]
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0
    # A_ub: (f*g)[k] - t <= 0 for all k
    A_ub = A_effects.astype(np.float64).tocsc()
    # Build augmented matrix: [A_effects | -1]
    neg_one = -np.ones((m, 1), dtype=np.float64)
    from scipy.sparse import hstack, csc_matrix
    A_full = hstack([A_effects, csc_matrix(neg_one)], format="csc")
    b_ub = np.zeros(m, dtype=np.float64)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(-np.inf, np.inf)]
    res = linprog(c, A_ub=A_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method="highs")
    if res.success and res.x is not None:
        g = res.x[:n]
        t_opt = res.x[n]
        return g, float(t_opt)
    return None, None

    f = np.exp(-0.5 * ((i - center) / width) ** 2)

    phases = rng.uniform(0, 2 * np.pi, size=n)

def subgradient_polish(n, f, iters, rng):
    """Polish using subgradient of max convolution value."""
    f = normalize(f).copy()
    best_f = f.copy()
    best_score = score_norm(f, n)
    step = 0.1

    for _ in range(iters):
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        k_star = int(np.argmax(conv))

        # Subgradient with respect to f[i]:
        # d/dx_i [conv[k_star]] = f[max(0,k_star-n+1)..min(n-1,k_star)] cross
        # For linear conv: (f*f)[k] = sum_{i: valid} f[i]*f[k-i]
        # d/dx_i = f[k-i] + f[i-k] (when both in range)
        i_range = np.arange(n)
        j1 = k_star - i_range
        j2 = i_range - k_star
        grad = np.zeros(n, dtype=np.float64)
        mask1 = (j1 >= 0) & (j1 < n)
        mask2 = (j2 >= 0) & (j2 < n)
        grad[mask1] += f[j1[mask1]]
        grad[mask2] += f[j2[mask2]]

        # Subgradient step: f -= step * grad, then project
        f_new = f - step * grad
        # Project onto {x >= 0, sum(x) = 1}
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s <= 1e-300:
            f_new = f
        else:
            f_new = f_new / s

        sc = score_norm(f_new, n)
        if sc < best_score:
            best_f = f_new.copy()
            best_score = sc
            f = f_new
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.9

    return best_f, best_score


def coordinate_polish(n, f, max_passes, rng):
    """Polish using random coordinate swaps."""
    f = normalize(f).copy()
    best_f = f.copy()
    best_score = score_norm(f, n)

    for _ in range(max_passes):
        improved = False
        for _ in range(n):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            delta = f[i] * 0.1
            if delta <= 0:
                continue
            f_new = f.copy()
            f_new[i] -= delta
            f_new[j] += delta
            if f_new[i] < 0:
                continue
            sc = score_norm(f_new, n)
            if sc < best_score:
                f = f_new
                best_f = f.copy()
                best_score = sc
                improved = True
        if not improved:
            break

    return best_f, best_score

    total_budget = max(1.0, float(budget_s))
    start_time = time.monotonic()
    deadline = start_time + total_budget - 5.0  # Reserve 5s for finalization

    best_f = None
    best_score = float("inf")
    given_f = None

    # Try to use the given sequence as a starting point
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        given_f = normalize(raw)
        if given_f is not None:
            n_g = given_f.size
            best_f = given_f.copy()
            best_score = score_norm(given_f, n_g)
            print(f"Given sequence: n={n_g}, score={best_score:.6f}", flush=True)
    except Exception:
        pass

    # Primary length
    n = 1024

    if best_f is not None and best_f.size != n:
        # Interpolate to target length
        old = np.linspace(0, 1, best_f.size)
        new = np.linspace(0, 1, n)
        best_f = normalize(np.interp(new, old, best_f))
        best_score = score_norm(best_f, n)
        given_f = best_f.copy()

    # Build template for n=1024
    print(f"\nBuilding template for n={n}...", flush=True)
    A, col_data, m = build_template(n)
    print(f"Template built: {m} constraints, {n} variables", flush=True)

    # Phase 1: Diverse initializations
    print("\n=== Phase 1: Diverse exploration ===", flush=True)
    p1_deadline = start_time + 180.0
    p1_deadline = min(p1_deadline, deadline)

    starts = []

    # Add given/best
    if best_f is not None:
        starts.append(("given", best_f.copy()))

    # Uniform
    starts.append(("uniform", np.ones(n) / n))

    # Oscillatory patterns
    for k_val in [1, 2, 3, 5, 8, 13]:
        for a_val in [0.3, 0.5, 0.7, 0.9]:
            f = gen_oscillatory(n, k_val, a_val, rng)
            if f is not None:
                starts.append((f"osc_k{k_val}_a{a_val}", f))

    # Combinations
    for k1 in [1, 2, 3]:
        for k2 in [5, 7, 11]:
            f = gen_comb(n, k1, k2, 0.5, 0.3, rng)
            if f is not None:
                starts.append((f"comb_{k1}_{k2}", f))

    # Gaussian shapes
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            f = gen_gaussian(n, center, width * n, rng)
            if f is not None:
                starts.append((f"gauss_{center}_{width}", f))

    # Two-peak
    starts.append(("two_peak", gen_two_peak(n, rng)))

    # Random phase
    for _ in range(3):
        f = gen_random_phase(n, rng)
        if f is not None:
            starts.append(("rand_phase", f))

    # Dirichlet
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        f = gen_dirichlet(n, alpha, rng)
        if f is not None:
            starts.append((f"dirichlet_{alpha}", f))

    # Random binary-ish
    for density in [0.3, 0.5, 0.7]:
        f = (rng.random(n) < density).astype(np.float64)
        f = normalize(f)
        if f is not None:
            starts.append((f"binary_{density}", f))

    print(f"Total starts: {len(starts)}", flush=True)

    # Score all starts quickly
    scored_starts = []
    for name, f in starts:
        s = score_norm(f, n)
        scored_starts.append((s, name, f))
    scored_starts.sort(key=lambda x: x[0])

    # Take top 15 for LP optimization
    top_starts = scored_starts[:15]
    p1_time_per = (p1_deadline - time.monotonic()) / max(1, len(top_starts))

    for idx, (s, name, f) in enumerate(top_starts):
        rem = p1_deadline - time.monotonic()
        if rem < 2.0:
            break
        tb = min(p1_time_per * 1.5, rem - 1.0)
        if tb < 1.0:
            tb = 1.0

        print(f"  Start {idx+1}/{len(top_starts)}: {name}, score={s:.6f}, budget={tb:.1f}s", flush=True)
        f_res, s_res = lp_optimize(n, f, A, col_data, m, tb, rng)
        if s_res < best_score:
            best_score = s_res
            best_f = f_res.copy()
            print(f"    NEW BEST: score={best_score:.6f}", flush=True)

    print(f"\nAfter Phase 1: best_score={best_score:.6f}", flush=True)

    # Phase 2: Focused optimization on the best
    print("\n=== Phase 2: Focused optimization ===", flush=True)
    p2_deadline = min(start_time + total_budget - 100.0, deadline)

    # Also try n=2048 if time permits
    n2 = 2048
    do_n2 = (p2_deadline - time.monotonic()) > 300.0

    if do_n2:
        print(f"\n  Also trying n={n2}...", flush=True)
        try:
            A2, col_data2, m2 = build_template(n2)
            # Upsample best to n2
            f_up = None
            if best_f is not None:
                old_pts = np.linspace(0, 1, best_f.size)
                new_pts = np.linspace(0, 1, n2)
                f_up = normalize(np.interp(new_pts, old_pts, best_f))
            if f_up is None:
                f_up = np.ones(n2) / n2

            tb2 = min(200.0, (p2_deadline - time.monotonic()) * 0.4)
            f_res2, s_res2 = lp_optimize(n2, f_up, A2, col_data2, m2, tb2, rng)
            # Compare with current best (need to evaluate at their respective n)
            # The score formula uses n, so compare directly
            if s_res2 < best_score:
                best_score = s_res2
                best_f = f_res2.copy()
                print(f"    NEW BEST at n={n2}: score={best_score:.6f}", flush=True)
            del A2, col_data2
        except MemoryError:
            print("  Memory error for n=2048, skipping", flush=True)
        except Exception as e:
            print(f"  Error for n=2048: {e}", flush=True)

    # Continue LP on best at n=1024
    if best_f.size != n:
        old_pts = np.linspace(0, 1, best_f.size)
        new_pts = np.linspace(0, 1, n)
        f_work = normalize(np.interp(new_pts, old_pts, best_f))
    else:
        f_work = best_f.copy()

    tb_p2 = min(p2_deadline - time.monotonic(), 800.0)
    if tb_p2 > 10:
        print(f"  Continuing LP at n={n}, budget={tb_p2:.1f}s", flush=True)
        f_res, s_res = lp_optimize(n, f_work, A, col_data, m, tb_p2, rng)
        if s_res < best_score:
            best_score = s_res
            best_f = f_res.copy()
            print(f"    NEW BEST: score={best_score:.6f}", flush=True)

    print(f"\nAfter Phase 2: best_score={best_score:.6f}", flush=True)

    # Phase 3: Polishing
    print("\n=== Phase 3: Polishing ===", flush=True)
    p3_deadline = deadline
    p3_time = max(5.0, p3_deadline - time.monotonic())

    # Ensure best_f is at n=1024 for polishing
    if best_f.size != n:
        old_pts = np.linspace(0, 1, best_f.size)
        new_pts = np.linspace(0, 1, n)
        f_work = normalize(np.interp(new_pts, old_pts, best_f))
    else:
        f_work = best_f.copy()

    # Subgradient polish
    sg_iters = max(10, int(p3_time * 2))
    f_sg, s_sg = subgradient_polish(n, f_work, sg_iters, rng)
    if s_sg < best_score:
        best_score = s_sg
        best_f = f_sg.copy()
        f_work = f_sg.copy()
        print(f"  After subgradient: score={best_score:.6f}", flush=True)

    # Coordinate polish
    coord_passes = max(5, int(p3_time))
    f_cd, s_cd = coordinate_polish(n, f_work, coord_passes, rng)
    if s_cd < best_score:
        best_score = s_cd
        best_f = f_cd.copy()
        print(f"  After coordinate: score={best_score:.6f}", flush=True)

    # Final small LP if time remains
    rem = deadline - time.monotonic()
    if rem > 5.0:
        f_fin, s_fin = lp_optimize(n, best_f, A, col_data, m, max(2.0, rem - 2.0), rng)
        if s_fin < best_score:
            best_score = s_fin
            best_f = f_fin.copy()
            print(f"  After final LP: score={best_score:.6f}", flush=True)

    # Final evaluation
    # Try to return at the best length we found
    if best_f is None:
        best_f = np.ones(n) / n

    # Convert to list of floats
    # The evaluator will handle the actual n based on len(sequence)
    final_n = len(best_f)
    seq = [float(x) for x in best_f]

    # Verify with evaluate_sequence
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float("inf")

    if not np.isfinite(final_score):
        # Fallback
        seq = [1.0 / n] * n
        final_score = 2.0

    print(f"\nFINAL: n={len(seq)}, score={final_score:.6f}", flush=True)
    return seq
