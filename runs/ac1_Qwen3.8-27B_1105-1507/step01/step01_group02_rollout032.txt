"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses smooth gradient descent (log-sum-exp approximation of max) with Adam optimizer,
continuation method, and subgradient polishing. Tries multiple sequence lengths
and initializations.

import numpy as np
import time
from scipy.optimize import linprog


def next_power_of_2(n):
    return 1 << (n - 1).bit_length()


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = next_power_of_2(size)
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def compute_smooth_gradient(a, c, t):
    """
    Compute gradient of (1/t)*log(sum_k exp(t*c[k])) with respect to a.
    Returns grad of shape (n,).
    """
    n = len(a)
    # Softmax weights (numerically stable)
    tc = t * c
    tc_shifted = tc - tc.max()
    w = np.exp(tc_shifted)
    w /= w.sum()
    
    # g[i] = sum_k w[k] * a[k-i] (with a[j]=0 for j out of [0,n-1])
    # g = convolve(w, a[::-1])[n-1 : n-1+n]
    size = 2 * n - 1
    N = next_power_of_2(size)
    W = np.fft.rfft(w, N)
    A_rev = np.fft.rfft(a[::-1], N)
    conv = np.fft.irfft(W * A_rev, N)[:size]
    g = conv[n-1:n-1+n]
    
    return 2.0 * g


def compute_subgradient(a, c, threshold_frac=1e-5):
    """Compute subgradient of max(c) at point a, using active peaks."""
    n = len(a)
    peak = np.max(c)
    threshold = peak * (1.0 - threshold_frac)
    peak_indices = np.where(c >= threshold)[0]
    if len(peak_indices) > 40:
        top_idx = np.argsort(c[peak_indices])[::-1][:15]
        peak_indices = peak_indices[top_idx]
    
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= max(1, len(peak_indices))
    return grad


def alt_projection_init(n, rng, n_iter=80):
    """
    Alternating projection to get a nonnegative sequence with near-flat
    magnitude spectrum.
    """
    N = next_power_of_2(2 * n - 2)
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        mag_target = np.ones(len(F_full))
        weights = np.linspace(1.0, 0.7, len(F_full))
        mag_target = weights ** 0.1
        F = mag_target * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j

    return a

"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses smooth gradient descent (log-sum-exp approximation of max) with Adam optimizer,
continuation method, subgradient polishing, and LP-based improvements.
Tries multiple sequence lengths and initializations for diverse search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def next_power_of_2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = next_power_of_2(size)
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    result[result < 0] = 0.0
    return result


def smooth_gradient(a, c, t):
    """
    Compute gradient of (1/t)*log(sum_k exp(t*c[k])) w.r.t. a.
    Uses FFT for efficient cross-correlation.
    """
    n = len(a)
    tc = t * c
    tc_shifted = tc - tc.max()
    w = np.exp(tc_shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w /= wsum

    size = 2 * n - 1
    N = next_power_of_2(size)
    W = np.fft.rfft(w, N)
    A_rev = np.fft.rfft(a[::-1], N)
    conv = np.fft.irfft(W * A_rev, N)[:size]
    g = conv[n - 1:n - 1 + n]
    return 2.0 * g


def subgradient_step(a, c):
    """Compute averaged subgradient of max(c) at a."""
    n = len(a)
    peak = np.max(c)
    threshold = peak * (1.0 - 1e-6)
    peak_indices = np.where(c >= threshold)[0]
    if len(peak_indices) > 25:
        top_idx = np.argsort(c[peak_indices])[::-1][:12]
        peak_indices = peak_indices[top_idx]

    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= max(1, len(peak_indices))
    return grad


def alt_projection_init(n, rng, n_iter=80):
    """Alternating projection for flat-spectrum nonnegative initialization."""
    N = next_power_of_2(2 * n - 2)
    phases = rng.uniform(0.0, 2.0 * np.pi, N // 2 + 1)
    F = np.exp(1j * phases)
    F[0] = 1.0 + 0j

    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(F, N)
        a = a_full[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        F_full = np.fft.rfft(a, N)
        mag_target = np.ones(len(F_full))
        weights = np.linspace(1.0, 0.7, len(F_full))
        mag_target = weights ** 0.1
        F = mag_target * np.exp(1j * np.angle(F_full))
        F[0] = F[0].real + 0j
    return a

            alloc = min(remaining * 0.10, total_time_available / n_configs)
        alloc = max(alloc, 20)

        if time.time() + alloc > overall_deadline - 30:
            alloc = overall_deadline - time.time() - 30
        if alloc < 20:
            continue

        config_deadline = time.time() + alloc
        print(f"\n[{i+1}/{n_configs}] n={n}, init={init_type}, time={alloc:.0f}s")
        try:
            a_opt, score = try_configuration(n, init_type, rng, config_deadline, parent_seq)
        except Exception as e:
            print(f"  Error: {e}")
            continue

        print(f"  => score={score:.6f}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  *** NEW BEST: {best_score:.6f} (n={n}) ***")

    # Phase 2: Intensive polish on best
    if best_seq is not None and time.time() < overall_deadline - 30:
        print(f"\n=== Intensive polish (current best={best_score:.6f}) ===")
        remaining = overall_deadline - time.time()

        # Subgradient polish
        a_p = subgradient_polish(best_seq, time.time() + remaining * 0.5, rng)
        s_p = evaluate_sequence(a_p.tolist())
        print(f"  After subgrad polish: {s_p:.6f}")
        if np.isfinite(s_p) and s_p < best_score:
            best_score = s_p
            best_seq = a_p.copy()

        # Smooth optimize from polished start
        if time.time() < overall_deadline - 20:
            a_s = adam_smooth_optimize(best_seq, overall_deadline, rng,
                                       t_mult=20.0, lr=0.005,
                                       t_growth=1.2, t_growth_interval=500)
            s_s = evaluate_sequence(a_s.tolist())
            print(f"  After smooth re-opt: {s_s:.6f}")
            if np.isfinite(s_s) and s_s < best_score:
                best_score = s_s
                best_seq = a_s.copy()

        # One more subgradient pass
        if time.time() < overall_deadline - 10:
            a_f = subgradient_polish(best_seq, overall_deadline, rng)
            s_f = evaluate_sequence(a_f.tolist())
            print(f"  Final subgrad: {s_f:.6f}")
            if np.isfinite(s_f) and s_f < best_score:
                best_score = s_f
                best_seq = a_f.copy()

    # Fallback
    if best_seq is None:
        print("Fallback to uniform.")
        best_seq = np.ones(256)

    result = [float(x) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"\nFINAL: score={final_score:.6f}, n={len(result)}")
    return result
