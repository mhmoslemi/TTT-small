and adaptive step sizes for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def build_A(a):
    """Build convolution matrix A where A@x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for col in range(n):
        A[col:col+n, col] = a
    return A

def solve_lp(a, M, upper_bound=100.0):
    """Solve LP: max sum(x) s.t. A@x <= M, 0 <= x <= upper_bound."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0.0, upper_bound)]*n, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        return res.x
    return None

def normalize(a):
    s = np.sum(a)
    if s < 1e-10:
        return a
    return a / s

def m_sequence_init(n, rng):
    """Create an m-sequence-like initialization."""
    # Use a simple LFSR to generate a pseudo-random binary sequence
    # Then smooth it slightly
    m = max(2, int(np.log2(n)) )
    # Initialize LFSR
    state = int(rng.integers(1, 2**m))
    seq = []
    for _ in range(n):
        seq.append(state & 1)
        # Simple LFSR step (taps depend on m)
        lsb = state & 1
        state >>= 1
        if lsb:
            state ^= (1 << (m-1))  # Simple tap
    # Map to [0, 1] with some structure
    a = np.array(seq, dtype=np.float64)
    # Add some smoothing
    kernel = np.ones(3) / 3.0
    a = np.convolve(a, kernel, mode='same')
    a = np.clip(a, 0, 1)
    return normalize(a)

    a = a * (1 + 0.5 * window)
    return normalize(a)

        end = min(n, start + burst_width)
        a[start:end] = rng.uniform(0.5, 1.0)
    a += rng.uniform(0, 0.2, n)  # Small background

    LP-based alternating optimization with optional simulated annealing.
    Returns (best_a, best_obj).
    """

    best_obj = compute_obj(best_a)
    t = t_init
    stall_count = 0
    improvement_threshold = 1e-6
    
    for it in range(max_iter):
        s = np.sum(a)

        # Solve LP
