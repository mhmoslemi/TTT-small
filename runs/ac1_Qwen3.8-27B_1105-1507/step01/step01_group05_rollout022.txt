```python
    grad = np.zeros(n)
    for j in range(n):
        if f[j] != 0:
            grad[:2*n-1-j] += f[j] * w[j:j+min(n, 2*n-1-j)]
            # Actually, for grad[i], we need w[i+j] where i in [0,n-1]
            # So grad[i] += f[j] * w[i+j] for i in [0, min(n-1, 2n-2-j)]
            # That is, for i in [0, min(n-1, 2n-2-j)]
    grad *= 4.0 * n
    return obj, grad
```

Hmm, this is O(n^2) but with a large constant. Let me think of a better way.

Actually, the cleanest O(n^2) approach:
```python
# w has length 2n-1
# For i in 0..n-1, grad[i] = 4n * sum_{j=0}^{n-1} f[j] * w[i+j]
# = 4n * (f @ W_row) where W_row = w[0:n], w[1:n+1], ..., w[n-1:2n-1]
# This is 4n * (F @ W) where F = f[None, :] (1 x n) and W is n x n with W[i,j] = w[i+j]
```

The matrix W has a special structure: W[i,j] = w[i+j]. This is a "Toeplitz" matrix (constant along anti-diagonals). I can use `scipy.linalg.toeplitz` or just build it directly.

Actually, the simplest efficient approach:
```python
# Use matrix construction
idx_i, idx_j = np.mgrid[0:n, 0:n]
W = w[idx_i + idx_j]  # W[i,j] = w[i+j]
grad = 4.0 * n * (f @ W)
```

This creates an n×n matrix and does a matrix-vector multiply. For n=1024, that's 1M memory for W and 1M multiply operations. Very fast.

But creating W every call is wasteful. I can precompute it... but w changes each call. So I need to rebuild it.

Alternative: use FFT.
`grad[i] = 4n * sum_j f[j] * w[i+j]`

Let me define `v = f` (length n) and `u = w` (length 2n-1).
I want `g[i] = sum_j v[j] * u[i+j]` for i = 0,...,n-1.

This is the "cross-correlation" of v and u at specific lags. Specifically:
`g[i] = (v ⋆ u)[i]` where (v ⋆ u)[i] = sum_j v[j] * u[i+j]

This is related to convolution: `(v ⋆ u)[i] = (v * u_flipped)[i + n - 1]` where u_flipped = u[::-1] and * is standard convolution.

Let me verify:
`(v * u_flipped)[k] = sum_j v[j] * u_flipped[k-j] = sum_j v[j] * u[2n-2-(k-j)] = sum_j v[j] * u[2n-2-k+j]`

I want `sum_j v[j] * u[i+j]`. So I need `2n-2-k = i`, i.e., `k = 2n-2-i`.

So `g[i] = (v * u_flipped)[2n-2-i]`

The convolution `v * u_flipped` has length n + (2n-1) - 1 = 3n-2.
I need elements at positions 2n-2-i for i=0,...,n-1, which are positions 2n-2, 2n-3, ..., n-1.

So: `g = convolve(f, w[::-1])[n-1:2n-1][::-1]`

Let me verify with a small example:
f = [a, b], n=2
w = [w0, w1, w2] (length 3 = 2n-1)
u_flipped = [w2, w1, w0]
conv = [a*w2, a*w1+b*w2, a*w0+b*w1, b*w0]
g[0] = conv[2] = a*w0 + b*w1 = a*w[0] + b*w[1] ✓ (should be f[0]*w[0] + f[1]*w[1])
g[1] = conv[1] = a*w1 + b*w2 = a*w[1] + b*w[2] ✓ (should be f[0]*w[1] + f[1]*w[2])

So: `g = np.convolve(f, w[::-1], mode='full')[n-1:2*n-1][::-1]`

Let me double-check the indexing:
- conv has length 3n-2 = 4 for n=2: indices 0,1,2,3
- I want indices 2n-2-i for i=0,1: indices 2, 1
- conv[2:0:-1] = [conv[2], conv[1]] = g[0], g[1] ✓

More generally: `g = np.convolve(f, w[::-1], mode='full')[2*n-2 : n-2 : -1]`
For n=2: conv[2:0:-1] = [conv[2], conv[1]] ✓

Let me write this properly:
```python
conv_full = np.convolve(f, w[::-1], mode='full')  # length 3n-2
g = conv_full[n-1:2*n-1][::-1]  # This gives n elements
```

Wait: `conv_full[n-1:2*n-1]` gives elements at indices n-1, n, ..., 2n-2, which is n elements.
Reversed: [conv_full[2n-2], conv_full[2n-3], ..., conv_full[n-1]]

For n=2: conv_full[1:3] = [conv_full[1], conv_full[2]], reversed = [conv_full[2], conv_full[1]] ✓

Great! So:
```python
def soft_obj_grad(f, tau):
    n = len(f)
    conv = np.convolve(f, f)
    c = conv / tau
    cmax = c.max()
    exp_c = np.exp(c - cmax)
    w = exp_c / exp_c.sum()
    lse = cmax + np.log(exp_c.sum())
    obj = 2.0 * n * tau * lse
    
    # Compute gradient using FFT or direct convolution
    conv_full = np.convolve(f, w[::-1], mode='full')
    g = conv_full[n-1:2*n-1][::-1]
    grad = 4.0 * n * g
    return obj, grad
```

This is O(n log n) using FFT-based convolution! Much better.

Let me verify once more with the example:

g should be:

The gradient computation is correct.

4. Return the best sequence

```python
"""
Hybrid smooth-optimization (L-BFGS-B with logsumexp surrogate) + LP-based polishing
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FFT-accelerated gradients, multiple starting points, and adaptive temperature scheduling.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_score_fast(f):
    """Compute 2*n*max(conv(f,f)) assuming sum(f)=1."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def soft_obj_grad(f, tau):
    """Smooth surrogate: 2n*tau*logsumexp(conv(f,f)/tau). Assumes sum(f)=1."""
    n = len(f)
    conv = np.convolve(f, f)
    c = conv / tau
    cmax = c.max()
    exp_c = np.exp(c - cmax)
    w = exp_c / exp_c.sum()
    lse = cmax + np.log(exp_c.sum())
    obj = 2.0 * n * tau * lse

    # grad[i] = 4n * sum_j f[j] * w[i+j]
    conv_full = np.convolve(f, w[::-1], mode='full')
    g = conv_full[n - 1:2 * n - 1][::-1]
    grad = 4.0 * n * g
    return obj, grad


def smooth_optimize(f0, time_budget, verbose=True):
    """L-BFGS-B on smooth surrogate with geometrically decreasing temperature."""
    f = f0.copy()
    n = len(f)
    conv0 = np.convolve(f, f)
    tau_start = conv0.max() * 0.8
    tau_end = conv0.max() * 0.005

    best_f = f.copy()
    best_score = compute_score_fast(f)
    tau = tau_start

    start_time = time.monotonic()
    phase = 0

    while time.monotonic() - start_time < time_budget:
        if time.monotonic() - start_time > time_budget - 2:
            break

        def obj_and_grad(x):
            x = np.clip(x, 0, None)
            S = x.sum()
            if S < 1e-10:
                return 1e10, np.zeros_like(x)
            x = x / S
            return soft_obj_grad(x, tau)

        result = minimize(obj_and_grad, f, method='L-BFGS-B',
                          jac=True, options={'maxiter': 30, 'ftol': 1e-14, 'gtol': 1e-10})

        f_new = np.clip(result.x, 0, None)
        S = f_new.sum()
        if S > 1e-10:
            f_new = f_new / S

        score = compute_score_fast(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            f = f_new

        tau *= 0.65
        if tau < tau_end:
            tau = tau_end

        phase += 1
        if verbose and phase % 3 == 0:
            elapsed = time.monotonic() - start_time
            print(f"      smooth: phase={phase}, tau={tau:.6f}, score={best_score:.6f}, t={elapsed:.1f}s")

    return best_f, best_score


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_polish(f0, time_budget, verbose=True):
    """LP-based active-set polishing."""
    n = len(f0)
    f = f0.astype(np.float64).copy()
    S0 = f.sum()
    if S0 < 1e-10:
        return f, compute_score_fast(f)
    f = f / S0

    best_f = f.copy()
    best_score = compute_score_fast(f)
    t = 0.4
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 10})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        g = b / b_sum
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        S_new = f_new.sum()
        if S_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / S_new

        new_score = compute_score_fast(f_new)

        if new_score < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = new_score
            f = f_new
            stall = 0
            t = min(0.8, t * 1.1)
            if verbose and it % 10 == 0:
                print(f"      LP: it={it}, score={best_score:.6f}, t={t:.3f}")
        else:
            t *= 0.6
            stall += 1
            if stall > 30:
                break

        it += 1

    return best_f, best_score


def make_starting_point(n, start_type, rng):
    """Create a starting point of length n."""
    if start_type == 'uniform':
        f = np.ones(n) * (1 + 0.1 * rng.standard_normal(n))
    elif start_type == 'ramp':
        f = np.linspace(0.1, 2.0, n)
    elif start_type == 'u_shape':
        x = np.linspace(0, 2 * np.pi, n)
        f = 1.0 + 0.8 * np.cos(x) ** 2
    elif start_type == 'arcsine':
        x = np.linspace(0.05, 0.95, n)
        f = 1.0 / np.sqrt(x * (1 - x))
    elif start_type == 'gaussian':
        x = np.linspace(-3, 3, n)
        f = np.exp(-x ** 2 / 4)
    elif start_type == 'cosine':
        x = np.linspace(0, np.pi, n)
        f = 0.5 + 0.5 * np.cos(x) ** 2
    elif start_type == 'sine_env':
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 0.5 + 0.1
    else:
        f = np.ones(n)
    f = np.clip(f, 0.001, None)
    f = f / f.sum()
    return f
```
