    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def build_A(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A

        return r.x
    return None

1. Multi-scale LP mirror descent (n=64 upsampled through 1024)

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_A_matrix(a):
    """Build Toeplitz-like constraint matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_solve(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if r.success and r.x is not None and r.x.sum() > 1e-10:
        return r.x
    return None

    """Compute J from sequence and its convolution (assumes sum(a)=1)."""

        a = np.clip(a, 0.01, None)
        a = a / a.sum()
        T = 0.005
        b_curr = fast_conv(a)
        j_curr = 2 * n * b_curr.max()
        it_count = 0

        while time.monotonic() < deadline - 5:
            # Perturb random subset
            k = max(1, n // 20)
            idx = rng.integers(0, n, size=k)
            a_new = a.copy()
            a_new[idx] *= np.exp(rng.normal(0, 0.05, size=k))
            a_new = np.clip(a_new, 0.0, None)
            sn = a_new.sum()
            if sn < 1e-10:
                continue
            a_new = a_new / sn

            b_new = fast_conv(a_new)
            j_new = 2 * n * b_new.max()

            delta = j_new - j_curr
            if delta < 0 or (T > 1e-6 and rng.random() < np.exp(-delta / T)):
                a = a_new
                b_curr = b_new
                j_curr = j_new
                if j_new < best_score:
                    try_update(a)

            it_count += 1
            T *= 0.998
            if it_count % 500 == 0:
                print(f"  anneal {it_count}: J={j_curr:.6f}, T={T:.6f}", flush=True)

        print(f"  Annealing done: {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # Final evaluation
    if best is not None:
        final_score = float(evaluate_sequence(best.tolist()))
        print(f"\nFINAL VERIFIED: J={final_score:.6f}, n={best.size}", flush=True)
    else:
        best = np.ones(256)
        final_score = float(evaluate_sequence(best.tolist()))
        print(f"\nFALLBACK: J={final_score:.6f}", flush=True)

    return [float(x) for x in best]
