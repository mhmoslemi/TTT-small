```python
import numpy as np
from scipy.optimize import linprog, minimize, linprog
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def build_constraint_matrix(a):
    """Build the (2n-1) x n matrix for the convolution constraint (a*g)[k] = A[k,:] @ g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            A[k, i] = a[k - i]
    return A

    """Solve dual LP: min sum(lambda) s.t. (a*lambda)[j] >= 1 for all j, lambda >= 0.
    Here (a*lambda)[j] = sum_k lambda[k] * a[j-k] for valid k.
    Returns optimal lambda or None."""
```

```python
    """Build (2n-1) x n matrix M where (M @ g)[k] = (a*g)[k]."""
    n = len(a)
    M = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):

        i_hi = min(n - 1, k)
        if i_lo <= i_hi:

def smoothed_gradient(a, epsilon):
    """Compute gradient of smooth approximation to max(C[k]).
    Uses softmax weighting with temperature 1/epsilon."""
    n = len(a)
    C = fast_conv(a)
    # Softmax for numerical stability
    C_shifted = C - C.max()
    weights = np.exp(epsilon * C_shifted)
    weights /= weights.sum()
    # Gradient: grad[m] = 2 * sum_k weights[k] * a[k-m] for valid k-m in [0,n-1]
    # = 2 * cross_correlation(weights, a)[m]
    # np.correlate(weights, a, mode='valid')[m] = sum_j weights[m+j]*a[j]
    grad = 2.0 * np.correlate(weights, a, mode='valid')
    return grad, C, weights
```

Let me also think about what makes the LP approach effective and how to improve it:

The LP approach works as follows:
1. Current sequence a, peak M = max(C)
2. Find g that maximizes sum(g) subject to (a*g)[k] <= M for all k
3. This g is the "best response" - it's the sequence that, when convolved with a, stays within the peak bound
4. Move towards g: a ← (1-t)*a + t*g

The key insight: at the optimum, a = g (the best response of a with respect to itself is a itself). So we're looking for a fixed point of the best-response mapping.

The convergence rate depends on the step size t. Too large t might oscillate; too small t converges slowly. The previous code used a linear decay of t. I'll use an exponential decay or an adaptive scheme.

- Compute g (best response)

J(a) = 2n * max(C) / (sum a)^2

This is scale-invariant. Let sum(a) = 1. Then J = 2n * max(C).

The LP approach minimizes max(C) (equivalently, J/n). The step size in the LP mirror descent controls how fast we converge.

At the optimum, the "active" constraints (where C[k] = max) form a set K*. The optimality condition is that the gradient of max(C) (which is in the convex hull of {grad_k : k in K*}) must be orthogonal to the feasible directions (i.e., the tangent space of {a >= 0, sum(a) = 1}).

This means: for the active set K*, there exist multipliers λ_k >= 0 such that:
sum_k λ_k * grad_k = λ * 1 + μ (for some Lagrange multipliers)

where μ is the component from the non-negativity constraints (μ_i >= 0, μ_i * a_i = 0).

This is a complex condition, but it tells us that at the optimum, the "weighted average" of the active gradients must be (up to the boundary corrections) a constant vector.

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using a multi-phase approach:
1. Smoothed-max gradient descent for fast exploration of diverse initializations
2. LP-based mirror descent (primal LP for best response) with adaptive step sizes
3. LP with line search for accurate convergence
4. L-BFGS-B for final polish
Starts from parent sequence and explores multiple lengths.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix M where (M @ g)[k] = (a*g)[k]."""
    n = len(a)
    M = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            M[k, i_lo:i_hi + 1] = a[k - i_hi:k - i_lo + 1][::-1]
    return M


def lp_best_response(a, M_val):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M_val for all k, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_val)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if res.success and res.x is not None:
            g = res.x
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def smoothed_max_grad(a, epsilon):
    """Compute (gradient, C, weights) for smoothed max of C[k] with softmax temperature."""
    n = len(a)
    C = fast_conv(a)
    C_shifted = C - C.max()
    weights = np.exp(epsilon * C_shifted)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.full(len(C), 1.0 / len(C))
    else:
        weights /= w_sum
    grad = 2.0 * np.correlate(weights, a, mode='valid')
    return grad, C, weights


def run_smooth_descent(a0, deadline, max_iters=5000):
    """Smoothed gradient descent with adaptive epsilon and step size."""
    a = project_simplex(np.array(a0, dtype=np.float64).copy())
    n = len(a)
    best_local = a.copy()
    best_val = float(fast_conv(a).max())

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(max_iters - 1, 1)
        epsilon = 3.0 + 47.0 * frac
        grad, C, weights = smoothed_max_grad(a, epsilon)
        step = min(0.3 / (epsilon * best_val + 1e-10), 5.0 / n)
        a_new = a - step * grad
        a_new = project_simplex(a_new)
        val_new = float(fast_conv(a_new).max())
        if val_new < best_val:
            best_val = val_new
            best_local = a_new.copy()
        a = a_new

    return best_local


def run_lp_descent(a0, deadline, max_iters=500):
    """LP-based mirror descent with adaptive step size (exponential decay)."""
    a = project_simplex(np.array(a0, dtype=np.float64).copy())
    n = len(a)
    best_local = a.copy()
    best_val = float(fast_conv(a).max())

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        C = fast_conv(a)
        M = float(C.max())
        if M < best_val:
            best_val = M
            best_local = a.copy()

        g = lp_best_response(a, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        if float(np.max(np.abs(g - a))) < 1e-7:
            break

        t_mix = max(0.03, 0.35 * np.exp(-0.015 * it))
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = project_simplex(a_new)
        a = a_new

    return best_local


def run_lp_linesearch(a0, deadline, max_iters=80):
    """LP mirror descent with line search on step size."""
    a = project_simplex(np.array(a0, dtype=np.float64).copy())
    n = len(a)
    best_local = a.copy()
    best_val = float(fast_conv(a).max())

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        C = fast_conv(a)
        M = float(C.max())
        if M < best_val:
            best_val = M
            best_local = a.copy()

        g = lp_best_response(a, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        if float(np.max(np.abs(g - a))) < 1e-7:
            break

        # Line search: try several step sizes
        best_t = 0.1
        best_new_val = best_val
        for t_candidate in [0.5, 0.3, 0.2, 0.1, 0.05]:
            a_try = (1.0 - t_candidate) * a + t_candidate * g
            a_try = project_simplex(a_try)

            if val_try < best_new_val:
                best_new_val = val_try
                best_t = t_candidate

        a = project_simplex(a)

    return best_local

def make_initializations(n, rng, best_seq=None):
    """Generate diverse initializations of length n."""
    t_arr = np.linspace(0, 1, n, endpoint=False)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Skewed geometric (mass on left)
    for r in [0.97, 0.985, 0.99, 0.995]:
        a = r ** np.arange(n, dtype=np.float64)
        a /= a.sum()
        inits.append(a)

    # Skewed geometric (mass on right)
    for r in [0.97, 0.99]:
        a = r ** np.arange(n, dtype=np.float64)
        a = a[::-1].copy()
        a /= a.sum()
        inits.append(a)

    # Cosine modulations
    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t_arr)
        a = np.maximum(a, 0.01)
        a /= a.sum()
        inits.append(a)

    # Quadratic / parabolic
    for amp in [0.5, 1.5, 3.0]:
        a = 1.0 + amp * (t_arr - 0.3) ** 2
        a /= a.sum()
        inits.append(a)

    # Skewed parabola
    a = 1.0 + 2.0 * t_arr ** 3
    a /= a.sum()
    inits.append(a)

    # Random
    for _ in range(3):
        a = rng.uniform(0.1, 1.0, n)
        a /= a.sum()
        inits.append(a)

    # From best sequence (interpolated to length n)
    if best_seq is not None and best_seq.size > 1:
        if best_seq.size == n:
            inits.append(best_seq.copy())
        else:
            t_old = np.linspace(0, 1, best_seq.size)
            a_interp = np.maximum(np.interp(t_arr, t_old, best_seq), 0.0)
            a_interp /= a_interp.sum()
            inits.append(a_interp)

    return inits
```
