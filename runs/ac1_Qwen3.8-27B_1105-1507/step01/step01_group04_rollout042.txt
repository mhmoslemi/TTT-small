```python
    """Compute Newton direction for the active constraints."""
    n = len(f)
    # Compute gradient: g[j] = (1/|K|) sum_{k in K} 2*f[k-j] (for valid k-j)
    g = np.zeros(n)
    H_diag = np.zeros(2*n-1)  # h[s] for s = 0, ..., 2n-2
    
    for k in active_set:
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                g[j] += 2.0 * f[kj]
    
    g /= len(active_set)
    
    # Build Hessian (Hankel matrix)
    # H[a,b] = (2/|K|) * count of k in K such that k = a+b
    for k in active_set:
        H_diag[k] += 1.0
    H_diag *= 2.0 / len(active_set)
    
    # Build the full Hankel matrix
    # H[a,b] = H_diag[a+b]
    idx = np.add.outer(np.arange(n), np.arange(n))  # idx[a,b] = a+b
    H = H_diag[idx]
    
    # Solve [H  1; 1'  0] [d; λ] = [-g; 0]
    A = np.zeros((n+1, n+1))
    A[:n, :n] = H
    A[:n, n] = 1.0
    A[n, :n] = 1.0
    rhs = np.zeros(n+1)
    rhs[:n] = -g
    
    try:
        sol = np.linalg.solve(A, rhs)
    except np.linalg.LinAlgError:
        sol = np.linalg.lstsq(A, rhs, rcond=None)[0]
    
    return sol[:n]
```

```python
"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses multi-strategy approach: spectral flattening (GS algorithm), improved mirror descent
with weighted active sets and line search, and log-space Adam optimization.
"""

import numpy as np
import time

def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    """Fast score computation assuming f >= 0, sum(f) > 0.01."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    """Normalize f to sum to 1, clip negatives."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _tapered_init(n, power):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _gaussian_init(n, sigma_frac):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _spectral_flatten(f0, n_iter, flatten_power, deadline):
    """
    GS-style spectral flattening:
    Iteratively flatten the power spectrum while maintaining non-negativity.
    """
    f = _normalize(f0.copy())
    if f is None:
        return f0.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    
    # Target: flatten the spectrum
    # Use a geometric approach: P_new = P_old^alpha where alpha < 1 flattens
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        F = np.fft.rfft(f, 2 * n - 1)
        P = np.abs(F) ** 2
        
        # Flatten: take power < 1 to reduce dynamic range
        P_flat = P ** flatten_power
        # Normalize to preserve energy
        if P_flat.sum() > 0:
            P_flat = P_flat / P_flat.sum() * P.sum()
        
        # Reconstruct with original phase
        F_new = np.sqrt(P_flat) * np.exp(1j * np.angle(F))
        # Make sure F[0] is real and positive (DC)
        F_new[0] = np.abs(F_new[0])
        # Enforce Hermitian symmetry for real signal
        f_new = np.fft.irfft(F_new, 2 * n - 1)[:n]
        
        # Project to non-negative
        f_new = np.maximum(f_new, 0.0)
        f_new = _normalize(f_new)
        if f_new is None:
            break
        f = f_new
        
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
    
    return best_f, best_score

def _improved_mirror_descent(f_init, n_iter, deadline, eta_scale=0.5, threshold_frac=0.7, momentum=0.0):
    """
    Mirror descent with multiplicative updates, weighted active set,
    backtracking line search, and momentum.
    """
    f = _normalize(f_init.copy())
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    
    # Momentum buffer
    log_momentum = np.zeros(n, dtype=np.float64)
    beta = momentum
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 0:
            break
        
        # Weighted active set: weight by how close to peak
        weights = np.zeros(len(h), dtype=np.float64)
        active_mask = h >= threshold_frac * M
        if not np.any(active_mask):
            active_mask[np.argmax(h)] = True
        
        # Soft weighting: weight = (h[k] - threshold) / (M - threshold) for active
        margin = h - (1 - (1 - threshold_frac)) * M
        weights[active_mask] = (h[active_mask] - (1 - (1 - threshold_frac) * M))
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights[active_mask] = 1.0
            w_sum = float(np.sum(active_mask))
        else:
            weights /= w_sum
        
        # Compute weighted pressure
        p = np.zeros(n, dtype=np.float64)
        active_lags = np.where(active_mask)[0]
        for k in active_lags:
            w = weights[k]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                lo = k - j_end
                hi = k - j_start
                p[j_start:j_end + 1] += w * 2.0 * f[lo:hi + 1][::-1]
        
        # Adaptive step size with backtracking
        eta = eta_scale / (M + 1e-15)
        
        # Apply momentum
        log_momentum = beta * log_momentum - eta * p
        log_ratio = np.clip(log_momentum, -20.0, 20.0)
        
        # Try the update
        f_trial = f * np.exp(log_ratio)
        f_trial = _normalize(f_trial)
        if f_trial is None:
            f_trial = np.ones(n) / n
        
        # Line search: try different step sizes
        best_trial = f_trial
        best_trial_score = _score_raw(f_trial)
        
        # Try smaller steps if the full step is bad
        if best_trial_score > best_score:
            for frac in [0.5, 0.25, 0.125]:
                f_small = f * np.exp(frac * log_ratio)
                f_small = _normalize(f_small)
                if f_small is not None:
                    s_small = _score_raw(f_small)
                    if s_small < best_trial_score:
                        best_trial_score = s_small
                        best_trial = f_small
        
        f = best_trial
        score = best_trial_score
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if (it + 1) % 500 == 0:
            print(f"    IMD it={it+1}: best={best_score:.6f}, M={M:.6f}")
    
    return best_f, best_score

def _logspace_adam(f_init, n_iter, lr=0.1, tau=0.05, deadline=None):
    """
    Optimize in log-space using Adam with softmax approximation of max.
    f[j] = exp(g[j]) / sum(exp(g))
    """
    f = _normalize(f_init.copy())
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    
    # Initialize g = log(f) + small offset for numerical stability
    g = np.log(np.maximum(f, 1e-10))
    g -= g.mean()  # center for numerical stability
    
    best_f = f.copy()
    best_score = _score_raw(f)
    
    # Adam state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute f from g (softmax for stability)
        g_max = g.max()
        exp_g = np.exp(g - g_max)
        f = exp_g / exp_g.sum()
        
        # Compute autocorrelation
        h = _fft_convolve_full(f)
        
        # Softmax approximation of max
        logits = h / tau
        logits -= logits.max()
        probs = np.exp(logits)
        probs /= probs.sum()
        
        # "Soft max" value
        soft_max = tau * np.sum(probs * (h / tau))  # = sum(probs * h)
        # Gradient of soft_max w.r.t. f: probs * 2 * (correlation at each lag)
        # For each j, df_score/df[j] = sum_k probs[k] * 2*f[k-j] (weighted)
        
        grad_f = np.zeros(n, dtype=np.float64)
        for k in range(2 * n - 1):
            if probs[k] < 1e-10:
                continue
            w = probs[k]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                lo = k - j_end
                hi = k - j_start
                grad_f[j_start:j_end + 1] += w * 2.0 * f[lo:hi + 1][::-1]
        
        # Gradient w.r.t. g: grad_g[j] = grad_f[j] * f[j] - f[j] * sum(grad_f * f)
        # (because f[j] = exp(g[j])/Z, df[j]/dg[j'] = f[j](delta_jj' - f[j']))
        grad_g = grad_f * f - f * np.dot(grad_f, f)
        
        # Adam update
        t = it + 1
        m = beta1 * m + (1 - beta1) * grad_g
        v = beta2 * v + (1 - beta2) * grad_g ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        g = g - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Track best
        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if (it + 1) % 500 == 0:
            print(f"    Adam it={it+1}: best={best_score:.6f}")
    
    return best_f, best_score

def _conflict_perturbation(f, rng, n_pert=10):
    """Perturb f at positions that conflict with the autocorrelation peak."""
    n = len(f)
    h = _fft_convolve_full(f)
    peak_lag = int(np.argmax(h))
    
    f_pert = f.copy()
    for _ in range(n_pert):
        # Randomly modify a few positions
        idx = rng.choice(n, size=rng.integers(3, 10), replace=False)
        f_pert[idx] *= rng.uniform(0.5, 1.5, size=len(idx))
    
    return _normalize(f_pert)
```
