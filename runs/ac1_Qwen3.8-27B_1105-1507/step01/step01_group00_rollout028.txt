def smooth_obj_grad(a, tau):
    n = len(a)
    conv = np.convolve(a, a)
    # Log-sum-exp
    max_c = np.max(conv)
    shifted = tau * (conv - max_c)
    log_sum = max_c + np.log(np.sum(np.exp(shifted))) / tau
    # Actually: (1/tau) * log(sum exp(tau * conv)) = (1/tau) * (tau*max_c + log(sum exp(tau*(conv-max_c))))
    # = max_c + (1/tau) * log(sum exp(tau*(conv-max_c)))

    # Gradient via cross-correlation
    grad = 2.0 * scipy.signal.correlate(w, a, mode='valid')[:n]
