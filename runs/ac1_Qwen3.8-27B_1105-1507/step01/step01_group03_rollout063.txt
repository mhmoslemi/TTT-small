```python
"""
Search for a non-negative step function minimizing

    2*n*max_k conv(a,a)_k / (sum(a))**2.

The program maintains a best normalized sequence (sum = 1).  It uses:
  * analytic starts, especially coefficients of (1-z)^(-1/2), which make
    the first n convolution coefficients flat;
  * projected fast gradient / momentum iterations on log-weights in a
    softmax simplex parameterization, using FFT convolutions;
  * a linear-programming direction method (maximize sum(g) subject to
    f*g <= M) for local polishing at moderate lengths.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def next_conv_len(n):
    """Smallest FFT length at least 2*n - 1."""
    m = 2 * int(max(1, n)) - 1
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_vector(x):
    """Make a finite nonnegative floating vector of sum 1, or None."""
    try:
        a = np.asarray(x, dtype=np.float64).ravel()
    except Exception:
        return None
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return None
    return a / s


def softmax_from_z(z):
    """Stable softmax vector of z."""
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return np.array([], dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    a = np.exp(np.clip(z, -745.0, 0.0))
    s = float(a.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones_like(z, dtype=np.float64) / float(z.size)
    return a / s


def fft_convolution(a, L):
    """Full nonnegative convolution a*a, length 2*len(a)-1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def fft_convolution2(a, b, L):
    """Full nonnegative convolution a*b, length 2*len(a)-1 (same length)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    if n == 0 or b.size != n:
        return np.array([], dtype=np.float64)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    c = np.fft.irfft(fa * fb, L)
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def fft_convolution_max(a, L=None):
    """max_k conv(a,a)_k for a valid vector."""
    a = normalize_vector(a)
    if a is None:
        return float("inf")
    n = a.size
    if n == 0:
        return float("inf")
    if L is None:
        L = next_conv_len(n)
    c = fft_convolution(a, L)
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    return m


def local_score_fft(a):
    """Score 2*n*max(conv(a,a)) assuming normalized sum 1, using FFT."""
    a = normalize_vector(a)
    if a is None:
        return float("inf")
    n = a.size
    L = next_conv_len(n)
    M = fft_convolution_max(a, L)
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M)


def consider_sequence(a, best_a, best_score, label=""):
    """If a improves the current best, return the new best pair."""
    a = normalize_vector(a)
    if a is None:
        return best_a, best_score
    score = local_score_fft(a)
    if np.isfinite(score) and score < best_score:
        if label:
            print(f"  [{label}] n={a.size} score={score:.6f}", flush=True)
        return a.copy(), score
    return best_a, best_score


def best_sequence_from_starts(starts):
    """Choose the start with the smallest FFT local score."""
    best = None
    best_score = float("inf")
    for s in starts:
        s = normalize_vector(s)
        if s is None:
            continue
        sc = local_score_fft(s)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = s
    if best is None and len(starts) > 0:
        best = normalize_vector(starts[0])
    return best


def exact_score_sequence(seq):
    """Call the provided evaluate_sequence function, if available."""
    try:
        return float(evaluate_sequence(seq))
    except Exception:
        return float("inf")


def sqrt_start(n):
    """
    Coefficients of (1-z)^(-1/2), truncated to n and normalized.

    The infinite square of this series is proportional to 1/(1-z), so the
    first n self-convolution coefficients are exactly flat.
    """
    n = int(max(1, n))
    u = np.ones(n, dtype=np.float64)
    for i in range(n - 1):
        u[i + 1] = u[i] * (2.0 * i + 1.0) / (2.0 * i + 2.0)
    s = float(u.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return u / s


def exponential_start(n, alpha):
    """Simple geometric start exp(-alpha*x)."""
    n = int(max(1, n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x)
    return normalize_vector(a)


def two_mode_start(n, q, alpha_left, alpha_right):
    """Mixture of a left exponential lobe and a right exponential lobe."""
    n = int(max(1, n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    q = min(1.0, max(0.0, float(q)))
    a = q * np.exp(-float(alpha_left) * x) + (1.0 - q) * np.exp(-float(alpha_right) * (1.0 - x))
    return normalize_vector(a)


def power_taper_start(n, p, beta):
    """One-sided power law with a gentle right-end taper."""
    n = int(max(1, n))
    i = np.arange(n, dtype=np.float64) + 0.5
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = i ** (-float(p))
    if abs(float(beta)) > 1e-12:
        a *= (1.0 - x + 1.0 / float(n)) ** float(beta)
    return normalize_vector(a)


def shifted_sqrt_start(n, q, shift_frac):
    """Mixture of a left sqrt-root component and a shifted right copy."""
    n = int(max(1, n))
    shift_frac = min(0.9, max(0.0, float(shift_frac)))
    s = int(round(float(shift_frac) * float(n)))
    nsub = max(1, n - s)
    base = sqrt_start(nsub)
    left = np.zeros(n, dtype=np.float64)
    left[:nsub] = base
    right = np.zeros(n, dtype=np.float64)
    if s < n:
        right[s:] = base
    q = min(1.0, max(0.0, float(q)))
    a = q * left + (1.0 - q) * right
    return normalize_vector(a)


def interpolate_sequence(a, n):
    """Resample a normalized nonnegative sequence to length n."""
    a = normalize_vector(a)
    n = int(max(1, n))
    if a is None:
        return sqrt_start(n)
    m = a.size
    if m == n:
        return a
    if m == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(new, old, a)
    out = normalize_vector(b)
    if out is None:
        out = sqrt_start(n)
    return out


def mix_noisy(base, rng, eps):
    """Mix a base vector with a random Dirichlet vector."""
    f = normalize_vector(base)
    if f is None:
        return None
    eps = min(1.0, max(0.0, float(eps)))
    if eps <= 1e-12:
        return f
    try:
        mix = rng.dirichlet(np.ones(f.size, dtype=np.float64))
    except Exception:
        mix = np.ones(f.size, dtype=np.float64) / float(f.size)
    if eps >= 1.0:
        f = mix
    else:
        f = (1.0 - eps) * f + eps * mix
    return normalize_vector(f)


def add_unique_start(starts, f, tol=1e-7):
    """Append f to starts if it is not numerically too close to existing ones."""
    f = normalize_vector(f)
    if f is None:
        return
    for g in starts:
        try:
            if float(np.linalg.norm(f - g)) < tol:
                return
        except Exception:
            pass
    starts.append(f)


def choose_starts(n, best_a, given, rng, budget):
    """Generate a small diverse set of starts for length n."""
    n = int(max(1, n))
    starts = []
    if best_a is not None:
        add_unique_start(starts, mix_noisy(interpolate_sequence(best_a, n), rng, 0.05))
    add_unique_start(starts, sqrt_start(n))
    if budget > 25.0:
        add_unique_start(starts, power_taper_start(n, 0.5, 0.5))
        add_unique_start(starts, exponential_start(n, 1.5))
    if budget > 60.0:
        add_unique_start(starts, shifted_sqrt_start(n, 0.75, 0.25))
        add_unique_start(starts, two_mode_start(n, 0.7, 1.0, 0.5))
    if budget > 100.0 and given is not None:
        add_unique_start(starts, mix_noisy(interpolate_sequence(given, n), rng, 0.05))
    if budget > 140.0 and n <= 2048:
        try:
            add_unique_start(starts, normalize_vector(rng.dirichlet(np.full(n, 0.35, dtype=np.float64))))
        except Exception:
            pass
    if not starts:
        starts.append(sqrt_start(n))
    return starts


def make_conv_template(n):
    """
    Sparse template for the matrix A with A[k,j] = f_{k-j} when valid.

    The data array is later overwritten by f[edge_offsets].
    """
    n = int(max(1, n))
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = np.asarray(A_off.data, dtype=np.int32).copy()
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(template, f, n, M):
    """Solve max sum(g) subject to f*g <= M, g >= 0."""
    A_template, edge_offsets, c_obj, bounds, m = template
    scale = float(n)
    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= scale
    b = np.full(m, M * scale * scale, dtype=np.float64)
    return linprog(
        c_obj,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )


def optimize_lp(n, f_init, time_budget, rng, template):
    """
    LP-direction local optimizer.

    Returns (best_normalized_A, best_local_score).
    """
    n = int(max(1, n))
    f = normalize_vector(f_init)
    if f is None or f.size != n:
        f = np.ones(n, dtype=np.float64) / float(n)

    L = next_conv_len(n)
    best_f = f.copy()
    best_M = fft_convolution_max(f, L)
    initial_M = best_M
    deadline = time.monotonic() + max(0.0, float(time_budget))
    stall = 0
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        t0 = time.monotonic()

        conv = fft_convolution(f, L)
        M = float(np.max(conv))
        if not np.isfinite(M) or M <= 0.0:
            break

        try:
            res = solve_direction_lp(template, f, n, M)
        except Exception as e:
            print(f"    lp n={n} linprog exception: {e}", flush=True)
            break

        x = getattr(res, "x", None)
        if x is None:
            stall += 1
            if stall > 12:
                break
            f = mix_noisy(best_f, rng, 0.2)
            if f is None:
                f = best_f.copy()
            continue

        x = np.asarray(x, dtype=np.float64).ravel()
        if x.size != n:
            break
        x = np.where(np.isfinite(x), x, 0.0)
        x = np.maximum(x, 0.0)
        s_star = float(x.sum())

        if s_star <= 1.0 + 1e-8:
            stall += 1
            if stall > 12:
                break
            f = mix_noisy(best_f, rng, 0.2)
            if f is None:
                f = best_f.copy()
            continue

        g = x / s_star

        r = fft_convolution2(f, g, L)
        r_max = float(np.max(r))
        if not np.isfinite(r_max) or r_max <= 0.0 or r_max >= M * (1.0 - 1e-10):
            stall += 1
            if stall > 12:
                break
            f = mix_noisy(best_f, rng, 0.2)
            if f is None:
                f = best_f.copy()
            continue

        G = fft_convolution_max(g, L)
        if not np.isfinite(G):
            stall += 1
            if stall > 12:
                break
            f = mix_noisy(best_f, rng, 0.2)
            if f is None:
                f = best_f.copy()
            continue

        # One-dimensional safe step bound:
        # H(t) = (1-t)^2 M + 2(1-t)t r_max + t^2 G <= M.
        a_coef = 2.0 * (r_max - M)
        b_coef = G + M - 2.0 * r_max
        if b_coef > 1e-14:
            t_max = -a_coef / b_coef
        else:
            t_max = 1.0
        if not np.isfinite(t_max) or t_max <= 0.0:
            t_max = 1e-8

        base = min(1.0, 0.75 * t_max)
        if base < 1e-12:
            base = 1e-12

        raw_ts = [
            base,
            0.5 * base,
            0.1 * base,
            0.01 * base,
            min(base, 1e-3),
            min(base, 1e-5),
            min(base, 1e-7),
            1e-12,
        ]
        ts = []
        last = 2.0
        for tv in sorted((float(v) for v in raw_ts), reverse=True):
            if tv <= 1e-14:
                continue
            if tv > 1.0:
                tv = 1.0
            if last - tv < 1e-12:
                continue
            last = tv
            ts.append(tv)
            if len(ts) >= 7:
                break

        best_cand_M = best_M
        best_cand_f = None
        best_t = 0.0

        for tv in ts:
            f_new = (1.0 - tv) * f + tv * g
            f_new = np.maximum(f_new, 0.0)
            ssum = float(f_new.sum())
            if ssum <= 1e-300:
                continue
            f_new /= ssum
            M_new = fft_convolution_max(f_new, L)
            if np.isfinite(M_new) and M_new < best_cand_M - 1e-15:
                best_cand_M = M_new
                best_cand_f = f_new.copy()
                best_t = tv

        if best_cand_f is not None:
            f = best_cand_f
            best_f = best_cand_f.copy()
            best_M = best_cand_M
            stall = 0
            if iteration <= 5 or iteration % 10 == 0 or best_M < initial_M * 0.995:
                print(
                    f"    lp n={n} iter={iteration} score={2.0 * n * best_M:.6f} "
                    f"M={M:.6g} t={best_t:.6f} dt={time.monotonic() - t0:.1f}s",
                    flush=True,
                )
        else:
            stall += 1
            f = mix_noisy(best_f, rng, 0.2)
            if f is None:
                f = best_f.copy()
            if iteration % 5 == 0:
                print(
                    f"    lp n={n} iter={iteration} no improvement, "
                    f"stall={stall}, best={2.0 * n * best_M:.6f}",
                    flush=True,
                )
            if stall > 12:
                break

    return best_f, float(2.0 * n * best_M)


def gradient_polish(a0, n, budget, rng, label=""):
    """
    Momentum/smoothed-max optimizer in log-weight space.

    It returns (best_normalized_A, best_local_score).
    """
    n = int(max(1, n))
    a0 = normalize_vector(a0)
    if a0 is None or a0.size != n:
        a0 = sqrt_start(n)

    L = next_conv_len(n)
    neg_idx = (-np.arange(n, dtype=np.int64)) % L

    z = np.log(a0 + 1e-300)
    z = np.where(np.isfinite(z), z, 0.0)
    z -= float(np.mean(z))

    best_a = a0.copy()
    best_M = fft_convolution_max(a0, L)

    mom = np.zeros(n, dtype=np.float64)
    step_size = 0.08
    tau_list = [0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004]
    tau_idx = 0
    restart = 0
    no_imp = 0
    step_count = 0

    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget))

    while time.monotonic() < deadline:
        step_count += 1
        a = softmax_from_z(z)
        if a.size != n:
            z = np.log(best_a + 1e-300)
            z -= float(np.mean(z))
            mom.fill(0.0)
            continue

        fa = np.fft.rfft(a, L)
        conv = np.fft.irfft(fa * fa, L)[:2 * n - 1]
        conv = np.maximum(conv, 0.0)
        M = float(np.max(conv))

        if np.isfinite(M):
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
                no_imp = 0
                step_size = min(0.25, step_size * 1.03)
                if step_count % 100 == 0 or step_count <= 5:
                    print(
                        f"    grad[{label}] n={n} step={step_count} "
                        f"score={2.0 * n * best_M:.6f} tau={tau_list[tau_idx]:g}",
                        flush=True,
                    )
            else:
                no_imp += 1
        else:
            no_imp += 10
            z = np.log(best_a + 1e-300)
            z -= float(np.mean(z))
            mom.fill(0.0)
            continue

        # If the current point has drifted too far, return to the best known.
        if M > best_M * 1.5 + 1e-300:
            z = np.log(best_a + 1e-300)
            z -= float(np.mean(z))
            mom.fill(0.0)
            no_imp = 0
            step_size = max(0.005, step_size * 0.7)
            step_count -= 1
            continue

        # Stall: reduce smoothing, add noise, shrink step.
        if no_imp > 120 + 20 * restart:
            tau_idx = min(tau_idx + 1, len(tau_list) - 1)
            z = np.log(best_a + 1e-300)
            z -= float(np.mean(z))
            noise = 0.04 + 0.12 * min(1.0, float(restart) / 3.0)
            try:
                z += rng.normal(0.0, noise, n)
            except Exception:
                z = z + 0.01 * rng.standard_normal(n)
            z = np.where(np.isfinite(z), z, 0.0)
            z -= float(np.mean(z))
            mom.fill(0.0)
            step_size = max(0.005, step_size * 0.7)
            restart += 1
            no_imp = 0
            step_count -= 1
            continue

        tau = tau_list[tau_idx]
        scale = best_M * tau
        if scale <= 1e-300 or not np.isfinite(scale):
            scale = 1e-300

        zz = (conv - M) / scale
        zz = np.where(np.isfinite(zz), zz, -50.0)
        zz -= float(np.max(zz))
        zz = np.maximum(zz, -50.0)
        p = np.exp(zz)
        ps = float(p.sum())
        m_len = 2 * n - 1
        if ps <= 1e-300 or not np.isfinite(ps):
            p = np.full(m_len, 1.0 / float(m_len), dtype=np.float64)
        else:
            p = p / ps

        w = np.fft.rfft(p, L)
        h = np.fft.irfft(fa * np.conj(w), L)
        grad = 2.0 * h[neg_idx]
        grad = np.where(np.isfinite(grad), grad, 0.0)

        dot = float(np.dot(a, grad))
        gz = grad - a * dot
        rms = float(np.linalg.norm(gz)) / np.sqrt(float(n))
        if not np.isfinite(rms) or rms <= 1e-16:
            no_imp += 10
            step_count -= 1
            continue

        gz = gz / rms
        mom = 0.9 * mom + 0.1 * gz
        mom_rms = float(np.linalg.norm(mom)) / np.sqrt(float(n))
        if np.isfinite(mom_rms) and mom_rms > 1.0:
            mom = mom / mom_rms
        else:
            mom = np.where(np.isfinite(mom), mom, 0.0)

        z_new = z - step_size * mom
        z_new = np.where(np.isfinite(z_new), z_new, z)
        z = z_new - float(np.mean(z_new))

        if no_imp % 35 == 0 and no_imp > 35:
            step_size = max(0.005, step_size * 0.9)

    return best_a, float(2.0 * n * best_M)


def pick_top_sequences(candidates, k=4):
    """Return up to k sequences with the lowest local FFT scores."""
    candidates = list(candidates)
    top = []
    k = int(max(0, k))
    for cand in candidates:
        cand = normalize_vector(cand)
        if cand is None:
            continue
        sc = local_score_fft(cand)
        if not np.isfinite(sc):
            continue
        if len(top) < k:
            pos = len(top)
            while pos > 0 and top[pos - 1][0] > sc:
                pos -= 1
            top.insert(pos, (sc, cand))
        elif sc < top[-1][0]:
            pos = len(top) - 1
            while pos > 0 and top[pos - 1][0] > sc:
                pos -= 1
            top[pos] = (sc, cand)
            top.pop()
    return [x[1] for x in top]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search entry point.

    Runs a timed multi-stage search and returns the best nonnegative
    sequence found.
    """
    seed = int(seed) if seed is not None else 42
    budget_s = float(budget_s if budget_s is not None else 1500.0)
    rng = np.random.default_rng(seed)

    if budget_s <= 0.0:
        budget_s = 1.0

    reserve = 25.0 if budget_s > 120.0 else max(2.0, 0.1 * budget_s)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, budget_s - reserve)

    best_a = None
    best_score = float("inf")

    given_raw = globals().get("height_sequence_1", None)
    given = normalize_vector(given_raw)
    if given is not None:
        best_a, best_score = consider_sequence(given, best_a, best_score, "given")

    if best_a is None:
        best_a = sqrt_start(1024)
        best_score = local_score_fft(best_a)
        print(f"  [init] n={best_a.size} score={best_score:.6f}", flush=True)

    # ------------------------------------------------------------------
    # Stage 1: moderate FFT-based gradient exploration at several lengths.
    # ------------------------------------------------------------------
    grad_explore_plan = [
        (512, 15.0),
        (1024, 35.0),
        (2048, 60.0),
        (4096, 90.0),
    ]

    for n, alloc in grad_explore_plan:
        rem = deadline - time.monotonic()
        if rem < max(10.0, 0.3 * alloc):
            continue
        b = min(float(alloc), rem - 2.0)
        if b < 8.0:
            continue

        starts = choose_starts(n, best_a, given, rng, b)
        main = best_sequence_from_starts(starts)
        alt = sqrt_start(n)
        if main is not None:
            try:
                if float(np.linalg.norm(main - alt)) < 1e-4:
                    alt = power_taper_start(n, 0.5, 0.75)
            except Exception:
                alt = power_taper_start(n, 0.5, 0.75)

        b1 = min(0.72 * b, rem - 2.0)
        if b1 > 5.0:
            a, sc = gradient_polish(main, n, b1, rng, label=f"exp{n}")
            best_a, best_score = consider_sequence(a, best_a, best_score, f"exp{n}")

        rem = deadline - time.monotonic()
        b2 = min(0.28 * b, rem - 2.0)
        if b2 > 5.0:
            a, sc = gradient_polish(alt, n, b2, rng, label=f"expalt{n}")
            best_a, best_score = consider_sequence(a, best_a, best_score, f"expalt{n}")

    # ------------------------------------------------------------------
    # Stage 2: LP polishing at moderate lengths.
    # ------------------------------------------------------------------
    lp_plan = [
        (1024, 170.0),
        (1536, 110.0),
    ]

    for n, alloc in lp_plan:
        rem = deadline - time.monotonic()
        if rem < max(25.0, 0.3 * alloc):
            continue
        b = min(float(alloc), rem - 3.0)
        if b < 15.0:
            continue

        try:
            template = make_conv_template(n)
        except Exception as e:
            print(f"  lp template failed n={n}: {e}", flush=True)
            continue

        starts = choose_starts(n, best_a, given, rng, b)
        main = best_sequence_from_starts(starts)
        alt = sqrt_start(n)
        if main is not None:
            try:
                if float(np.linalg.norm(main - alt)) < 1e-4:
                    alt = shifted_sqrt_start(n, 0.75, 0.25)
            except Exception:
                alt = shifted_sqrt_start(n, 0.75, 0.25)

        b1 = min(0.65 * b, deadline - time.monotonic() - 2.0)
        if b1 > 8.0:
            a, sc = optimize_lp(n, main, b1, rng, template)
            best_a, best_score = consider_sequence(a, best_a, best_score, f"lp{n}")

        rem = deadline - time.monotonic()
        b2 = min(0.35 * b, rem - 2.0)
        if b2 > 8.0:
            a, sc = optimize_lp(n, alt, b2, rng, template)
            best_a, best_score = consider_sequence(a, best_a, best_score, f"lphalt{n}")

        try:
            del template
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stage 3: long gradient refinement at larger lengths.
    # ------------------------------------------------------------------
    long_grad_plan = [
        (2048, 120.0),
        (4096, 210.0),
        (8192, 300.0),
    ]

    for n, alloc in long_grad_plan:
        rem = deadline - time.monotonic()
        if rem < max(25.0, 0.25 * alloc):
            continue
        b = min(float(alloc), rem - 3.0)
        if b < 15.0:
            continue

        starts = []
        if best_a is not None:
            add_unique_start(starts, mix_noisy(interpolate_sequence(best_a, n), rng, 0.035))
            if n <= 4096:
                add_unique_start(starts, mix_noisy(interpolate_sequence(best_a, n), rng, 0.18))
        add_unique_start(starts, sqrt_start(n))
        if n <= 4096:
            add_unique_start(starts, shifted_sqrt_start(n, 0.75, 0.25))
            add_unique_start(starts, power_taper_start(n, 0.5, 0.7))
        if given is not None and b > 100.0:
            add_unique_start(starts, mix_noisy(interpolate_sequence(given, n), rng, 0.05))

        main = best_sequence_from_starts(starts)
        alt = sqrt_start(n)
        if main is not None:
            try:
                if float(np.linalg.norm(main - alt)) < 0.15:
                    alt = shifted_sqrt_start(n, 0.7, 0.3)
            except Exception:
                alt = shifted_sqrt_start(n, 0.7, 0.3)

        b1 = min(0.7 * b, deadline - time.monotonic() - 2.0)
        if b1 > 10.0:
            a, sc = gradient_polish(main, n, b1, rng, label=f"long{n}")
            best_a, best_score = consider_sequence(a, best_a, best_score, f"long{n}")

        rem = deadline - time.monotonic()
        b2 = min(0.3 * b, rem - 2.0)
        if b2 > 10.0:
            a, sc = gradient_polish(alt, n, b2, rng, label=f"longalt{n}")
            best_a, best_score = consider_sequence(a, best_a, best_score, f"longalt{n}")

    # ------------------------------------------------------------------
    # Stage 4: final short polish of the globally best sequence.
    # ------------------------------------------------------------------
    rem = deadline - time.monotonic()
    if best_a is not None and rem > 30.0:
        b = min(45.0, rem - 2.0)
        a, sc = gradient_polish(best_a, best_a.size, b, rng, label="final")
        best_a, best_score = consider_sequence(a, best_a, best_score, "final")

    # ------------------------------------------------------------------
    # Final exact verification and selection among a few strong candidates.
    # ------------------------------------------------------------------
    candidates = []
    if best_a is not None:
        candidates.append(best_a)
    if given is not None:
        candidates.append(given)

    if best_a is not None:
        for target in (512, 1024, 2048, 4096):
            if target != best_a.size:
                c = interpolate_sequence(best_a, target)
                if c is not None:
                    candidates.append(c)

    top_cands = pick_top_sequences(candidates, k=4)
    if best_a is not None and (not top_cands or top_cands[0] is not best_a):
        top_cands = [best_a] + top_cands[:3]

    best_exact_a = best_a
    best_exact_score = float("inf")

    for a in top_cands:
        a = normalize_vector(a)
        if a is None:
            continue
        seq = [float(x) for x in a]
        sc = exact_score_sequence(seq)
        if np.isfinite(sc) and sc < best_exact_score:
            best_exact_score = sc
            best_exact_a = a.copy()
        if np.isfinite(sc):
            print(f"  exact check: n={a.size} score={sc:.6f}", flush=True)

    if not np.isfinite(best_exact_score) or best_exact_a is None:
        if given is not None:
            best_exact_a = given.copy()
        else:
            best_exact_a = sqrt_start(1024)

    final_seq = [float(x) for x in best_exact_a]

    # Absolute fallback if something went wrong.
    if not final_seq:
        final_seq = [1.0 / 1024.0] * 1024
    else:
        final_check = exact_score_sequence(final_seq)
        if not np.isfinite(final_check):
            fb = sqrt_start(1024)
            final_seq = [float(x) for x in fb]
            final_check = exact_score_sequence(final_seq)

    print(
        f"  FINAL: n={len(final_seq)} local_best={best_score:.6f} "
        f"exact_final={final_check:.6f} elapsed={time.monotonic() - start_time:.1f}s",
        flush=True,
    )

    return final_seq
```
