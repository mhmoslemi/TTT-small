def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1

    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)

def _fft_cross_convolve(a, b):
    """Compute full linear convolution of a and b using FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)

        f_md, s_md = _mirror_descent(f0, n_it, 0.5, 0.7, md_deadline)
        print(f"  MD [{idx+1}/{len(candidates)}] {name}: score={s_md:.8f}", flush=True)
        md_results.append((s_md, name, f_md))

        if s_md < best_score:
            best_score = s_md
            best_f = f_md.copy()
            print(f"  ** New best after MD: {best_score:.8f} from {name} **", flush=True)

    # Sort by score
    md_results.sort(key=lambda x: x[0])

    # Phase 3: LP-based Frank-Wolfe on top candidates
    # Allocate remaining time to top candidates
    remaining = deadline_global - time.monotonic()
    top_k = min(5, len(md_results))
    print(f"\nPhase 3: LP-FW on top {top_k} candidates, remaining={remaining:.1f}s", flush=True)

    # Allocate time: give more time to better candidates
    time_weights = []
    for i in range(top_k):
        time_weights.append(1.0 / (i + 1))
    total_w = sum(time_weights)
    lp_times = [remaining * 0.7 * w / total_w for w in time_weights]

    for r in range(top_k):
        if time.monotonic() > deadline_global - 30:
            break
        s_cur, name, f_cur = md_results[r]
        n = len(f_cur)
        print(f"\n  LP-FW on {name} (n={n}, score={s_cur:.8f}, budget={lp_times[r]:.1f}s)", flush=True)

        budget = min(lp_times[r], deadline_global - time.monotonic() - 10)
        if budget < 5:
            break

        max_iters = max(10, int(budget / 5))  # estimate ~5s per LP iter
        f_lp, s_lp = _lp_frank_wolfe(f_cur, max_iters, budget, verbose=True)

        if s_lp < best_score:
            best_score = s_lp
            best_f = f_lp.copy()
            print(f"  ** New best after LP-FW: {best_score:.8f} from {name} **", flush=True)

        # Continue with the LP result for next rounds
        md_results[r] = (s_lp, name + "_lp", f_lp)

    # Phase 4: Rerun LP-FW on the single best with remaining time
    remaining = deadline_global - time.monotonic()
    if remaining > 20 and len(md_results) > 0:
        s_best, name_best, f_best_cur = md_results[0]
        print(f"\nPhase 4: Final LP-FW on {name_best} (n={len(f_best_cur)}, "
              f"score={s_best:.8f}, budget={remaining - 10:.1f}s)", flush=True)
        max_iters = max(5, int((remaining - 10) / 5))
        f_final, s_final = _lp_frank_wolfe(f_best_cur, max_iters, remaining - 10, verbose=True)
        if s_final < best_score:
            best_score = s_final
            best_f = f_final.copy()
            print(f"  ** New best final: {best_score:.8f} **", flush=True)

    # Phase 5: Try different lengths via upsampling/downsampling the best
    remaining = deadline_global - time.monotonic()
    if remaining > 30:
        print(f"\nPhase 5: Multi-scale attempts, remaining={remaining:.1f}s", flush=True)
        for target_n in [256, 512, 1024]:
            if remaining < 20:
                break
            if len(best_f) == target_n:
                continue
            f_resized = _upsample(best_f, target_n) if target_n > len(best_f) else _downsample(best_f, target_n)
            if f_resized is None:
                continue
            # Quick MD + short LP
            f_md2, s_md2 = _mirror_descent(f_resized, 500, 0.4, 0.8, deadline_global)
            if s_md2 < best_score:
                best_score = s_md2
                best_f = f_md2.copy()
                print(f"  New best after resize to {target_n}: {best_score:.8f}", flush=True)
            remaining = deadline_global - time.monotonic()
            if remaining > 15 and target_n <= 1024:
                f_lp2, s_lp2 = _lp_frank_wolfe(f_md2, max(3, int(remaining / 10)), remaining - 10, verbose=False)
                if s_lp2 < best_score:
                    best_score = s_lp2
                    best_f = f_lp2.copy()
                    print(f"  New best after LP on resize {target_n}: {best_score:.8f}", flush=True)

    # Phase 6: Final perturbation attempts
    remaining = deadline_global - time.monotonic()
    if remaining > 15:
        for pert_frac in [0.02, 0.05, 0.1]:
            if remaining < 10:
                break
            n = len(best_f)
            noise = rng.random(n) * pert_frac
            f_pert = _normalize(best_f + noise)
            if f_pert is None:
                continue
            f_p, s_p = _mirror_descent(f_pert, int(min(1000, remaining * 50)), 0.3, 0.8, deadline_global)
            if s_p < best_score:
                best_score = s_p
                best_f = f_p.copy()
                print(f"  New best after perturb({pert_frac}): {best_score:.8f}", flush=True)
            remaining = deadline_global - time.monotonic()

    # Final verification
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.8f} (n={len(final_list)})", flush=True)

    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Falling back to parent.", flush=True)
        parent_list = [float(max(0.0, min(1000.0, x))) for x in parent]
        parent_score = float(evaluate_sequence(parent_list))
        print(f"  Parent score: {parent_score:.8f}", flush=True)
        return parent_list

    return final_list
