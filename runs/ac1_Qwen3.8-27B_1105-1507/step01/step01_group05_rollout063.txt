def project_simplex(v):

    scaled = np.clip(scaled, -700, 700)
    exp_vals = np.exp(scaled)
    total = exp_vals.sum()
    if total < 1e-300:
        return np.zeros(n), c.max()
    weights = exp_vals / total
    # g[i] = 2 * sum_{j=0}^{n-1} a[j] * weights[i+j]
    g = np.zeros(n)
    for i in range(n):
        g[i] = 2.0 * np.dot(a, weights[i:i + n])
    return g, c.max()


def optimize_mirror_descent(a0, budget, rng, verbose=True):
    """Mirror descent with log-sum-exp continuation."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a0, np.inf
    a = a / S
    
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    start = time.monotonic()
    
    taus = [2.0, 5.0, 10.0, 30.0, 100.0, 300.0]
    
    for tau in taus:
        elapsed = time.monotonic() - start
        if elapsed > budget:
            break
        time_left = budget - elapsed
        if time_left < 3:
            break
        
        lr = 1.0 / (np.sqrt(n) * (tau + 1) ** 0.4)
        
        while time.monotonic() - start < budget:
            if time.monotonic() - start > budget - 2:
                break
            
            g, c_max = gradient_logsumexp(a, tau)
            
            # Mirror descent: a_new[i] = a[i] * exp(-lr * g[i])
            log_update = -lr * g
            log_update = np.clip(log_update, -50, 50)
            a_new = a * np.exp(log_update)
            S_new = a_new.sum()
            if S_new < 1e-30:
                lr *= 0.5
                continue
            a_new = a_new / S_new
            
            score = compute_score_normalized(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            
            a = a_new
            
            # Adaptive: reduce lr if not improving for a while (simple heuristic)
            # Just keep going; the continuation handles it
        
        if verbose:
            print(f"  tau={tau:.0f}, best={best_score:.6f}, elapsed={time.monotonic()-start:.1f}s")
    
    return best_a, best_score


def optimize_subgradient(a0, budget, rng, verbose=True):
    """Exact subgradient method with adaptive step size."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a0, np.inf
    a = a / S
    
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    start = time.monotonic()
    
    lr = 1.0 / np.sqrt(n)
    stall_count = 0
    top_k = 3
    
    it = 0
    while time.monotonic() - start < budget:
        if it > 200000:
            break
        
        c = np.convolve(a, a)
        current_max = c.max()
        
        top_indices = np.argsort(c)[::-1][:top_k]
        
        g = np.zeros(n)
        for k in top_indices:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                g[i] += 2.0 * a[k - i]
        g /= len(top_indices)
        
        a_new = project_simplex(a - lr * g)
        
        score_new = compute_score_normalized(a_new)
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            a = a_new
            stall_count += 1
            if stall_count % 50 == 0:
                lr *= 0.85
                if stall_count > 200:
                    pert = rng.normal(0, lr * 0.3, n)
                    a_pert = a + pert
                    a_pert = np.clip(a_pert, 0, None)
                    if a_pert.sum() > 1e-10:
                        a = a_pert / a_pert.sum()
                    lr = 1.0 / np.sqrt(n)
                    top_k = 3
                    stall_count = 0
        
        it += 1
        if verbose and it % 2000 == 0:
            print(f"    subgrad it={it}, best={best_score:.6f}, lr={lr:.6f}")
    
    return best_a, best_score


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def optimize_lp_direction(a0, budget, rng, verbose=False):
    """LP-based direction method."""
    n = len(a0)
    f = a0.copy().astype(np.float64)
    S0 = f.sum()
    if S0 < 1e-10:
        return a0, np.inf
    f = f / S0
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    t = 0.4
    start = time.monotonic()
    it = 0
    stall_count = 0
    
    while time.monotonic() - start < budget:
        if it > 300:
            break
        if budget - (time.monotonic() - start) < 3:
            break
        
        conv = np.convolve(f, f)
        M = conv.max()
        
        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 3})
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        g = b / b_sum
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        S_new = f_new.sum()
        if S_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / S_new
        
        new_score = compute_score_normalized(f_new)
        
        if new_score < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = new_score
            f = f_new
            stall_count = 0
            if t < 0.5:
                t = min(0.6, t * 1.05)
        else:
            t *= 0.7
            stall_count += 1
            if stall_count > 20:
                f = f + rng.normal(0, 0.02, n)
                f = np.clip(f, 0, None)
                S = f.sum()
                if S > 1e-10:
                    f = f / S
                t = 0.4
                stall_count = 0
                if compute_score_normalized(f) < best_score:
                    best_f = f.copy()
                    best_score = compute_score_normalized(f)
        
        it += 1
    
    return best_f, best_score


def local_polish(a, rng, n_iter=3000):
    """Random local search polish."""
    n = len(a)
    f = a.copy().astype(np.float64)
    S = f.sum()
    f = f / S
    best = f.copy()
    best_score = compute_score_normalized(f)
    
    for _ in range(n_iter):
        f_c = f.copy()
        width = max(1, int(rng.integers(1, max(2, n // 16))))
        start_idx = int(rng.integers(0, n - width))
        scale = np.exp(rng.normal(0, 0.25))
        f_c[start_idx:start_idx + width] *= scale
        f_c = np.clip(f_c, 0, None)
        S_new = f_c.sum()
        if S_new < 1e-10:
            continue
        f_c = f_c / S_new
        score = compute_score_normalized(f_c)
        if score < best_score:
            best = f_c.copy()
            best_score = score
            f = f_c
    
    return best, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search function combining multiple optimization strategies."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0
    
    best_overall = None
    best_score_overall = np.inf
    
    # Get parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None
    
    if parent is not None:
        parent = np.clip(parent, 0, 1000)
        if parent.sum() > 0.01:
            parent = parent / parent.sum()
            pscore = compute_score_normalized(parent)
            print(f"Parent: n={len(parent)}, score={pscore:.6f}")
            best_overall = parent.copy()
            best_score_overall = pscore
        else:
            parent = None
    
    # Phase 1: Mirror descent on parent and a few structured starts
    print("\n=== Phase 1: Mirror Descent ===")
    
    starts = []
    if parent is not None:
        starts.append(("parent", parent))
    
    for n in [256, 512, 1024]:
        # Structured start with asymmetry
        x = np.arange(n, dtype=np.float64)
        f_struct = 1.0 + 0.3 * np.sin(2 * np.pi * x / n) + 0.2 * np.sin(4 * np.pi * x / n)
        f_struct = np.clip(f_struct, 0.05, None)
        f_struct = f_struct / f_struct.sum()
        starts.append((f"struct_{n}", f_struct))
        
        # Trapezoid
        ramp = max(1, n // 6)
        f_trap = np.ones(n)
        f_trap[:ramp] = np.linspace(0.3, 1.0, ramp)
        f_trap[n-ramp:] = np.linspace(1.0, 0.3, ramp)
        f_trap = f_trap / f_trap.sum()
        starts.append((f"trap_{n}", f_trap))
    
    # Allocate time for Phase 1
    phase1_time = min(300.0, (budget_s - 60) * 0.3)
    time_per_start = phase1_time / len(starts)
    
    for name, f0 in starts:
        if time.monotonic() > deadline:
            break
        t_left = deadline - time.monotonic()
        if t_left < 20:
            break
        t_alloc = min(time_per_start, t_left - 15)
        if t_alloc < 10:
            continue
        
        print(f"  MD [{name}] n={len(f0)}, budget={t_alloc:.0f}s")
        f_opt, score = optimize_mirror_descent(f0, t_alloc, rng, verbose=False)
        
        if np.isfinite(score) and score < best_score_overall:
            best_score_overall = score
            best_overall = f_opt.copy()
            print(f"  ** New best: {score:.6f} from MD [{name}] **")
    
    # Phase 2: Subgradient polish on best
    print("\n=== Phase 2: Subgradient Polish ===")
    if best_overall is not None:
        t_left = deadline - time.monotonic()
        t_alloc = min(300.0, t_left * 0.3)
        if t_alloc > 20:
            print(f"  Subgrad on best, n={len(best_overall)}, budget={t_alloc:.0f}s")
            f_opt, score = optimize_subgradient(best_overall, t_alloc, rng, verbose=False)
            if np.isfinite(score) and score < best_score_overall:
                best_score_overall = score
                best_overall = f_opt.copy()
                print(f"  ** New best: {score:.6f} from subgrad **")
    
    # Phase 3: LP direction on multiple starts
    print("\n=== Phase 3: LP Direction ===")
    lp_starts = []
    if best_overall is not None:
        lp_starts.append(("best", best_overall))
    if parent is not None:
        lp_starts.append(("parent", parent))
    # Add a few more diverse starts
    for n in [384, 768, 1536]:
        x = np.arange(n, dtype=np.float64)
        f_rand = np.abs(1.0 + 0.4 * np.cos(3 * np.pi * x / n + rng.uniform(0, 2*np.pi)) + 
                       0.2 * np.sin(7 * np.pi * x / n))
        f_rand = np.clip(f_rand, 0.01, None)
        f_rand = f_rand / f_rand.sum()
        lp_starts.append((f"div_{n}", f_rand))
    
    phase3_time = min(400.0, (deadline - time.monotonic()) * 0.4)
    time_per_lp = max(30.0, phase3_time / len(lp_starts))
    
    for name, f0 in lp_starts:
        if time.monotonic() > deadline:
            break
        t_left = deadline - time.monotonic()
        if t_left < 25:
            break
        t_alloc = min(time_per_lp, t_left - 20)
        if t_alloc < 15:
            continue
        
        print(f"  LP [{name}] n={len(f0)}, budget={t_alloc:.0f}s")
        f_opt, score = optimize_lp_direction(f0, t_alloc, rng, verbose=False)
        
        if np.isfinite(score) and score < best_score_overall:
            best_score_overall = score
            best_overall = f_opt.copy()
            print(f"  ** New best: {score:.6f} from LP [{name}] **")
    
    # Phase 4: Local polish + random restarts
    print("\n=== Phase 4: Polish & Restarts ===")
    if best_overall is not None:
        t_left = deadline - time.monotonic()
        if t_left > 30:
            # Polish
            n_iter_polish = min(10000, int(t_left * 500))
            f_opt, score = local_polish(best_overall, rng, n_iter_polish)
            if np.isfinite(score) and score < best_score_overall:
                best_score_overall = score
                best_overall = f_opt.copy()
                print(f"  ** New best: {score:.6f} from polish **")
        
        # Random restarts from best
        t_left = deadline - time.monotonic()
        n_restarts = min(8, int(t_left / 30))
        print(f"  Running {n_restarts} restarts, n={len(best_overall)}")
        for r in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            t_left_r = deadline - time.monotonic() - 5
            if t_left_r < 15:
                break
            
            # Perturb best
            scale = rng.uniform(0.7, 1.3, len(best_overall))
            f0 = best_overall * scale
            f0 = np.clip(f0, 0, None)
            if f0.sum() < 1e-10:
                continue
            f0 = f0 / f0.sum()
            
            # Quick LP optimization
            f_opt, score = optimize_lp_direction(f0, min(60.0, t_left_r), rng, verbose=False)
            if np.isfinite(score) and score < best_score_overall:
                best_score_overall = score
                best_overall = f_opt.copy()
                print(f"  Restart {r}: new best {score:.6f}")
    
    # Phase 5: Try different lengths with mirror descent
    print("\n=== Phase 5: Length Exploration ===")
    t_left = deadline - time.monotonic()
    if t_left > 60 and best_overall is not None:
        n_best = len(best_overall)
        for n_try in [n_best * 2, max(256, n_best // 2)]:
            if n_try > 4096 or n_try < 64:
                continue
            if time.monotonic() > deadline - 30:
                break
            t_left = deadline - time.monotonic()
            if t_left < 30:
                break
            
            # Resample best to new length
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            f_new = np.interp(x_new, x_old, best_overall)
            f_new = np.clip(f_new, 0, None)
            if f_new.sum() < 1e-10:
                continue
            f_new = f_new / f_new.sum()
            
            t_alloc = min(120.0, t_left - 20)
            if t_alloc < 20:
                continue
            
            print(f"  Trying n={n_try}, budget={t_alloc:.0f}s")
            f_opt, score = optimize_mirror_descent(f_new, t_alloc, rng, verbose=False)
            if np.isfinite(score) and score < best_score_overall:
                best_score_overall = score
                best_overall = f_opt.copy()
                print(f"  ** New best: {score:.6f} at n={n_try} **")
    
    # Final verification
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal verified score: {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        return final_list
    
    # Fallback
    print("Fallback to uniform sequence")
    return [1.0] * 256
