```python
"""
Search for low-score non-negative step-function coefficients using
smoothed L-BFGS optimization with log-sum-exp approximation of the max,
progressive sharpening, and multiplicative polishing.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

def _softmax(x):
    x = np.asarray(x, dtype=np.float64)
    x = x - np.max(x)
    ex = np.exp(x)
    s = ex.sum()
    if s == 0 or not np.isfinite(s):
        return np.full(x.shape, 1.0/len(x))
    return ex / s

def _local_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * n * m / (s * s))

def _cross_corr_fft(p, a):
    """Compute cross[j] = sum_l p[j+l] * a[l] for j=0..n-1 using FFT."""
    n = len(a)
    N = 1
    while N < 3 * n:
        N *= 2
    P = np.fft.rfft(p[::-1], N)
    A = np.fft.rfft(a, N)
    cross_full = np.fft.irfft(P * A, N)[:3*n-2]
    return cross_full[n-1:2*n-1][::-1]

def _smoothed_obj_grad(x, beta):
    """Objective and gradient of 2n * logsumexp(beta * convolve(softmax(x), softmax(x)))."""
    a = _softmax(x)
    n = len(a)
    c = np.convolve(a, a)
    bc = beta * c
    bc_max = float(np.max(bc))
    exp_bc = np.exp(bc - bc_max)
    sum_exp = float(np.sum(exp_bc))
    f = 2.0 * n * (bc_max + np.log(sum_exp))
    p = exp_bc / sum_exp
    
    cross = _cross_corr_fft(p, a)
    pc = float(np.dot(p, c))
    grad = 4.0 * n * beta * a * (cross - pc)
    
    # Numerical stability: replace inf/nan with 0
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return f, grad

def _lbfgs_optimize(x0, beta, maxiter=300, maxfun=5000):
    """Run L-BFGS-B on the smoothed objective."""
    def func(x):
        f, g = _smoothed_obj_grad(x, beta)
        if not np.isfinite(f):
            return 1e10
        return f
    
    def jac(x):
        f, g = _smoothed_obj_grad(x, beta)
        if not np.isfinite(f):
            return np.zeros_like(x)
        return g
    
    res = minimize(func, x0, jac=jac, method='L-BFGS-B',
                   options={'maxiter': maxiter, 'maxfun': maxfun, 'gtol': 1e-8})
    x_opt = res.x
    a = _softmax(x_opt)
    return a

def _random_start(n, rng, kind=0):
    if kind == 0:
        x = rng.normal(0, 1, n)
    elif kind == 1:
        x = rng.normal(0, 1, n) * rng.uniform(0.5, 2.0)
    elif kind == 2:
        x = np.log(rng.uniform(0.1, 1.0, n))
    elif kind == 3:
        w = int(rng.integers(3, max(4, min(20, n+1))))
        x = np.convolve(rng.normal(0, 1, n), np.ones(w)/w, mode='same')
    elif kind == 4:
        x = rng.normal(0, 1, n)
        x = np.convolve(x, np.array([0.3, 0.4, 0.3]), mode='same')
    else:
        x = rng.uniform(-1, 1, n)
    return x

def _symmetrize(a):
    a = 0.5 * (a + a[::-1])
    s = a.sum()
    if s > 0:
        a = a / s
    return a

def _multiplicative_polish(a, max_iter=200, eta=0.3, K=50):
    """Polish using multiplicative updates on active peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s <= 0:
        return a
    a = a / s
    
    best = a.copy()
    bestsc = _local_score(best)
    
    for it in range(max_iter):
        c = np.convolve(a, a)
        M = float(np.max(c))
        if M <= 0:
            break
        
        # Find active peaks
        if K >= c.size:
            active = np.nonzero(c > 0.01 * M)[0]
        else:
            active = np.argpartition(c, -K)[-K:]
        
        # Compute h[i] = sum over active k of (c[k]/M) * a[k-i]
        h = np.zeros(n)
        for k in active:
            w = c[k] / M
            s_start = max(0, k - n + 1)
            s_end = min(n - 1, k)
            for i in range(s_start, s_end + 1):
                j = k - i
                h[i] += w * a[j]
                h[j] += w * a[i]
        
        # Normalize and update
        h_mean = h.mean()
        h_max = h.max()
        if h_max <= 0:
            break
        log_factor = eta * (h_mean - h) / h_max
        log_factor = np.clip(log_factor, -3.0, 3.0)
        a = a * np.exp(log_factor)
        a = np.clip(a, 1e-30, None)
        s = a.sum()
        if s <= 0:
            a = best.copy()
            break
        a = a / s
        
        sc = _local_score(a)
        if sc < bestsc:
            best = a.copy()
            bestsc = sc
    
    return best

def _lp_step(f, M, n, seed=0):
    """LP: maximize sum(g) s.t. (f*g)_k <= M, g >= 0."""
    rng = np.random.default_rng(seed)
    # Build constraint matrix A where A[k, j] = f[k-j] (for valid j)
    rows = 2 * n - 1
    A = np.zeros((rows, n))
    for k in range(rows):
        s = max(0, k - n + 1)
        e = min(n - 1, k)
        for j in range(s, e + 1):
            A[k, j] = f[k - j]
    
    cvec = -np.ones(n) + 1e-9 * rng.uniform(-1, 1, n)
    bvec = np.full(rows, M * (1 + 1e-6))
    bounds = [(0, None)] * n
    
    try:
        res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None
    
    if not res.success:
        return None
    
    g = np.clip(res.x, 0, None)
    s = g.sum()
    if s <= 0:
        return None
    return g / s
```
