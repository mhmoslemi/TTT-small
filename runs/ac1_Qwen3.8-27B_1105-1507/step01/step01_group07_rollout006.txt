```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

Hybrid LP + smoothed gradient optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses alternating LP for global progress and smoothed gradient descent for local refinement.
Tries multiple sequence lengths with time-adaptive allocation.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # We need: sum_j A[k,j] * g[j] = sum_j a[k-j] * g[j] = (a*g)_k
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    return A

def solve_lp(a, M, upper_bound=100.0):
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun)
    return None, None

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def gradient_step(a, beta, lr):
    """One smoothed gradient descent step to minimize max(conv(a,a)) with sum(a)=1."""
    n = len(a)
    conv = np.convolve(a, a)
    M = np.max(conv)
    
    # Softmax weights
    shifted = beta * (conv - M)
    w = np.exp(shifted)
    w /= np.sum(w)
    
    # Compute s_j = sum_k w_k * a[k-j] for j = 0..n-1
    s = np.zeros(n)
    for j in range(n):
        k_start = j
        k_end = min(j + n, len(w))
        if k_start < len(w):
            s[j] = np.dot(w[k_start:k_end], a[:k_end - k_start])
    
    # Gradient of M_smooth = (1/beta) * log(sum(exp(beta*conv_k)))
    # dM_smooth/da_j = (1/beta) * sum_k beta * w_k * 2*a[k-j] = 2 * s_j
    # But we want to minimize 2n*M (with sum=1), so gradient is 2n * 2 * s_j = 4n*s_j
    # Wait, we're minimizing M directly (since sum is fixed at 1).
    # dM/da_j = 2 * s_j
    # We want to decrease M, so step is a - lr * 2 * s
    
    grad = 2.0 * s  # gradient of smoothed M
    
    a_new = a - lr * grad
    a_new = project_simplex(a_new)
    
    return a_new, M

def alternating_lp(a, max_iter, t=0.5, time_limit=None):
    """Run alternating LP iterations."""
    a = a / np.sum(a)
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    start_time = time.monotonic()
    
    stall = 0
    for it in range(max_iter):
        if time_limit is not None and (time.monotonic() - start_time) > time_limit:
            break
        
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        g, g_sum = solve_lp(a, M)
        if g is None:
            break
        
        if g_sum <= 1.0 + 1e-6:
            stall += 1
            if stall > 10:
                # Perturb
                rng = np.random.default_rng(it * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.5, n))
                a = np.clip(a, 0, None)
                s = np.sum(a)
                if s < 1e-10:
                    a = np.ones(n) / n
                    s = 1.0
                a = a / s
                stall = 0
            continue
        stall = 0
        
        g = g / g_sum
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
        
        a = a_new
    
    return best_a, best_obj

def gradient_optimize(a, max_iter=50000, beta=100, lr_init=0.01, time_limit=None):
    """Smoothed gradient descent to minimize max(conv(a,a))."""
    a = a / np.sum(a)
    n = len(a)
    best_a = a.copy()
    best_M = np.max(np.convolve(a, a))
    best_obj = compute_obj(best_a)
    start_time = time.monotonic()
    
    lr = lr_init
    for it in range(max_iter):
        if time_limit is not None and (time.monotonic() - start_time) > time_limit:
            break
        
        a_new, M = gradient_step(a, beta, lr)
        
        ol = compute_obj(a_new)
        if ol < best_obj:
            best_a = a_new.copy()
            best_obj = ol
            best_M = M
            a = a_new
            # Decrease learning rate slowly
            if it % 1000 == 0 and it > 0:
                lr *= 0.8
        else:
            # Try smaller step
            lr *= 0.5
            if lr < 1e-10:
                break
    
    return best_a, best_obj
