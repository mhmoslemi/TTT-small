def gradient_refine(a, max_iter=2000, verbose=False):
    """Gradient-based refinement using subgradient of the max-conv objective."""
    n = len(a)
    a = a.copy()
    a = normalize(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    lr = 0.1
    for it in range(max_iter):
        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))
        
        # Gradient of conv[k_star] w.r.t. a[i]:
        # conv[k_star] = sum_j a[j] * a[k_star - j]
        # d/d(a[i]) = a[k_star - i] + a[i - k_star] (with bounds)
        grad = np.zeros(n)
        for i in range(n):
            # Term 1: a[k_star - i] if 0 <= k_star - i < n
            if 0 <= k_star - i < n:
                grad[i] += a[k_star - i]
            # Term 2: a[i - k_star] if 0 <= i - k_star < n
            if 0 <= i - k_star < n:
                grad[i] += a[i - k_star]

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
            a = a_new
        else:
            lr *= 0.7
            if lr < 1e-8:
                break
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0)
            a_new = normalize(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_a = a_new.copy()
                best_obj = obj_new
                a = a_new
    
    return best_a, best_obj

Minmax LP + gradient-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: Use a minmax LP (directly minimize the convolution peak) instead of
the previous max-sum LP. Add projected gradient refinement and simulated annealing.

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return a
    return a / s

def solve_minmax_lp(a):
    """
    Solve: min t s.t. conv(a,g)_k <= t for k=0,...,2n-2, sum(g)=1, g>=0.
    Returns (g_opt, t_opt) or (None, np.inf) on failure.
    """
    n = len(a)
    
    # Build A_ub: shape (2n-1, n+1)
    # A_ub[k, i] = a[k-i] if 0<=k-i<n else 0, for i=0,...,n-1
    # A_ub[k, n] = -1 (for the -t term)
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            A_ub[k, idx] = a[k - idx]
        A_ub[k, n] = -1.0
    
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})

def lp_optimize(n, a_init, max_iter=500, alpha=0.5, tol=1e-8, verbose=False, deadline=None):
    """
    Run minmax LP-based alternating optimization.
    a is normalized to sum=1.
    """
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    stall_count = 0
    cur_alpha = alpha
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        g, t_opt = solve_minmax_lp(a)
        if g is None:
            break
        
        current_peak = float(np.max(np.convolve(a, a)))
        
        # Check if improvement is possible
        if t_opt >= current_peak * (1.0 - tol):
            stall_count += 1
            if stall_count > 30:
                break
            # Try a small perturbation to escape
            if stall_count % 10 == 0:
                rng = np.random.default_rng(it)
                a = normalize(a * np.exp(rng.normal(0, 0.1, n)))
            continue
        
        stall_count = 0
        
        # Blend with optimal g
        a_new = (1.0 - cur_alpha) * a + cur_alpha * g
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
            a = a_new
            cur_alpha = alpha  # Reset alpha
            if verbose and (it % 50 == 0):
                print(f"    it={it}: obj={best_obj:.6f} peak={current_peak:.8f} t_opt={t_opt:.8f}")
        else:
            # Reduce alpha
            cur_alpha *= 0.5
            if cur_alpha < 1e-4:
                break
            a_new = (1.0 - cur_alpha) * a + cur_alpha * g
            a_new = normalize(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_a = a_new.copy()
                best_obj = obj_new
                a = a_new
    
    return best_a, best_obj

def gradient_refine(a, max_iter=5000, lr_init=0.5, verbose=False, deadline=None):
    """
    Projected subgradient method on the max-conv objective.
    a is normalized to sum=1.
    Minimize max_k conv(a,a)_k.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    lr = lr_init
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))
        current_peak = conv[k_star]
        
        # Gradient of conv[k_star] w.r.t. a
        # grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1)
        grad = np.zeros(n)
        grad[idx] = 2.0 * a[k_star - idx]
        
        # Gradient step
        a_new = a - lr * grad
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
            a = a_new
            # Slightly increase lr if making good progress
            if it % 100 == 0 and it > 0 and verbose:
                print(f"    grad it={it}: obj={best_obj:.6f} peak={current_peak:.8f} k*={k_star}")
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
            # Try again with smaller lr
            a_new = a - lr * grad
            a_new = normalize(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_a = a_new.copy()
                best_obj = obj_new
                a = a_new
    
    return best_a, best_obj

def multi_peak_gradient_refine(a, max_iter=5000, lr_init=0.3, num_peaks=3, verbose=False, deadline=None):
    """
    Gradient method that considers top 'num_peaks' peaks.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    lr = lr_init
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv = np.convolve(a, a)
        # Get top num_peaks positions
        top_idx = np.argsort(conv)[::-1][:num_peaks]
        top_vals = conv[top_idx]
        
        # Weight by value (larger peaks get more weight)
        weights = top_vals / top_vals.sum()
        
        # Combined gradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            i_start = max(0, int(k) - n + 1)
            i_end = min(n - 1, int(k))
            idx = np.arange(i_start, i_end + 1)
            grad[idx] += w * 2.0 * a[int(k) - idx]
        
        # Gradient step
        a_new = a - lr * grad
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
            a = a_new
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    
    return best_a, best_obj

def simulated_annealing(a, max_iter=10000, T_init=0.01, T_end=0.0001, verbose=False, deadline=None):
    """
    Simulated annealing for final polishing.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    T = T_init
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        T = T_init * (T_end / T_init) ** (it / max_iter)
        
        # Random perturbation: modify a random block
        rng = np.random.default_rng(it * 7919 + n)
        width = max(1, n // 50)
        start = int(rng.integers(0, n - width + 1))
        a_new = a.copy()
        scale = np.exp(rng.normal(0, 0.2, width))
        a_new[start:start + width] *= scale
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        delta = obj_new - best_obj
        
        if delta < 0:
            best_a = a_new.copy()
            best_obj = obj_new
            a = a_new
        else:
            # Accept with probability exp(-delta/T)
            if np.exp(-delta / max(T, 1e-10)) > rng.random():
                a = a_new
    
    return best_a, best_obj

Minmax LP + multi-peak gradient + simulated annealing for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses alternating minmax LP (directly minimize
convolution peak) as the main workhorse, with gradient-based and stochastic
refinement for escaping local optima. Tries multiple sequence lengths.

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return a
    return a / s

def solve_minmax_lp(a):
    n = len(a)
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            A_ub[k, idx] = a[k - idx]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    c = np.zeros(n + 1)
    c[n] = 1.0
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status == 0:
        return res.x[:n].copy(), float(res.x[n])
    return None, np.inf
