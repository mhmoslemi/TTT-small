import numpy as np

def _fft_convolve(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score(f):
    """Compute score of a non-negative sequence."""
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    h = _fft_convolve(f)
    M = float(np.max(h))
    return 2.0 * n * M / (S * S)

def _normalize(f):
    """Normalize f to sum to 1."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:

    return f / s

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses multi-start mirror descent with momentum, Successive Zero Insertion (SZI),
LP-based Frank-Wolfe refinement, and diverse structured initializations.

import numpy as np
import time


def _fft_convolve(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _score(f):
    """Compute score of a non-negative sequence (fast, assumes f>=0, sum>0.01)."""
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    h = _fft_convolve(f)
    M = float(np.max(h))
    return 2.0 * n * M / (S * S)


def _normalize(f):
    """Normalize f to sum to 1, clip negatives. Returns None if degenerate."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s


def _pressure(f, active):
    """Compute subgradient of max_k (f*f)_k, averaged over active lags K."""
    n = len(f)
    p = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            idx = k - np.arange(j_lo, j_hi + 1)
            p[j_lo:j_hi + 1] += 2.0 * f[idx]
    p /= float(len(active))
    return p


def _mirror_descent(f_init, n_iter, eta_scale, threshold_frac, momentum=0.7, label=""):
    """Mirror descent with momentum for minimizing 2n*max(conv(f,f)) s.t. sum(f)=1, f>=0."""
    f = _normalize(np.asarray(f_init, dtype=np.float64).copy())
    if f is None:
        return np.asarray(f_init, dtype=np.float64).copy(), float('inf')

    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    p_avg = np.zeros(n, dtype=np.float64)

    for it in range(n_iter):
        h = _fft_convolve(f)
        M = float(np.max(h))
        if M <= 1e-30:
            break

        thr = threshold_frac * M + 1e-15
        active = np.where(h >= thr)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)

        p = _pressure(f, active)
        p_avg = momentum * p_avg + (1.0 - momentum) * p

        eta = eta_scale / (M + 1e-12)
        log_ratio = -eta * p_avg
        log_ratio = np.clip(log_ratio, -22.0, 22.0)

        f = f * np.exp(log_ratio)
        f_new = _normalize(f)
        if f_new is None:
            f = np.ones(n) / n
        else:
            f = f_new

        sc = _score(f)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()

        if (it + 1) % 2000 == 0:
            print(f"    {label} it={it+1}: best={best_score:.6f}, M={M:.2e}, |K|={len(active)}")

    return best_f, best_score


def _szi_step(f, top_k=5):

    f = np.exp(-(j - center1) ** 2 / (2 * sigma ** 2)) + np.exp(-(j - center2) ** 2 / (2 * sigma ** 2))
    s = np.sum(f)

def _lp_frank_wolfe(f_init, max_iters, time_budget):
    """LP-based Frank-Wolfe for the normalized problem (small n only)."""
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix

    f = _normalize(np.asarray(f_init, dtype=np.float64).copy())
    if f is None:
        return np.asarray(f_init, dtype=np.float64).copy(), float('inf')

    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start_time = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break

        h = np.convolve(f, f)
        M = float(np.max(h))
        if M <= 0:
            break

        # Build sparse constraint matrix: C[k, j] = f[k-j] if 0<=k-j<n else 0
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            rows_j = np.arange(j, j + n)
            row_idx.extend(rows_j.tolist())
            col_idx.extend([j] * n)
            vals.extend(f.tolist())

        C = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)

        res = linprog(c_obj, A_ub=C, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')

        if not res.success or res.x is None:
            break

        g = np.maximum(res.x, 0)
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg

        t = 2.0 / (it + 2)
        f_new = (1.0 - t) * f + t * g
        f_n = _normalize(f_new)
        if f_n is None:
            break
        f = f_n

        sc = _score(f)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()

        if (it + 1) % 5 == 0:
            print(f"    LP it={it+1}: score={best_score:.6f}")

    return best_f, best_score


def _random_perturb(f, rng, frac):
    """Add small additive perturbation and renormalize."""
    n = len(f)

    f_p = f + noise
    return _normalize(f_p)

        t0 = time.monotonic()
        f_md, s_md = _mirror_descent(f0, n_it1, 0.5, 0.7, momentum=0.7, label=name)
        elapsed = time.monotonic() - t0
        phase1_results.append((s_md, name, f_md))
        print(f"  P1 [{idx+1}/{len(all_candidates)}] {name}: score={s_md:.6f} ({elapsed:.1f}s)")

        if s_md < best_score:
            best_score = s_md
            best_f = f_md.copy()
            print(f"  ** New best: {best_score:.6f} from {name} **")

    # Sort by score
    phase1_results.sort(key=lambda x: x[0])

    # --- Phase 2: SZI on top candidates ---
    print(f"\n--- Phase 2: SZI ---")
    top_k_szi = min(3, len(phase1_results))

    for r in range(top_k_szi):
        if time.monotonic() > deadline - 200:
            break

        _, szi_name, szi_f = phase1_results[r]
        szi_f = szi_f.copy()
        szi_score = szi_name and phase1_results[r][0]
        szi_n = len(szi_f)
        max_szi_steps = 40
        print(f"  SZI from {szi_name}: n={szi_n}, score={szi_score:.6f}")

        for step in range(max_szi_steps):
            if time.monotonic() > deadline - 150:
                break

            # Try all insertion positions
            t0 = time.monotonic()
            candidates_szi = _szi_step(szi_f, top_k=3)
            szi_time = time.monotonic() - t0

            best_szi_score = candidates_szi[0][0]
            best_szi_f = candidates_szi[0][2].copy()

            if best_szi_score >= szi_score * 1.001:
                # No improvement, try a few random perturbations + re-insert
                improved = False
                for _ in range(3):
                    f_pert = _random_perturb(szi_f, rng, 0.02)
                    if f_pert is None:
                        continue
                    cands2 = _szi_step(f_pert, top_k=1)
                    if cands2[0][0] < szi_score:
                        szi_f = cands2[0][2]
                        szi_score = cands2[0][0]
                        improved = True
                        break
                if not improved:
                    # Try random position insertion
                    pos = rng.integers(0, len(szi_f) + 1)
                    g = np.insert(szi_f, pos, 0.0)
                    sc_g = _score(g)
                    if sc_g < szi_score:
                        szi_f = g
                        szi_score = sc_g
                    else:
                        break
            else:
                szi_f = best_szi_f
                szi_score = best_szi_score
                szi_n = len(szi_f)

            # Quick mirror descent refinement after SZI
            n_ref = int(min(800, max(200, 10.0 * 200)))
            f_ref, s_ref = _mirror_descent(szi_f, n_ref, 0.4, 0.75, momentum=0.75, label=f"szi_{step}")

            if s_ref < szi_score:
                szi_f = f_ref
                szi_score = s_ref

            if szi_score < best_score:
                best_score = szi_score
                best_f = szi_f.copy()
                print(f"  ** SZI new best: {best_score:.6f} (n={len(szi_f)}) **")

        print(f"  SZI {szi_name} done: final score={szi_score:.6f}, n={len(szi_f)}")

    # --- Phase 3: Intensive refinement on overall best ---
    print(f"\n--- Phase 3: Refinement ---")
    print(f"  Current best: score={best_score:.6f}, n={len(best_f)}")

    remaining = deadline - time.monotonic()

    # 3a: Extended mirror descent with varying thresholds
    for tf in [0.6, 0.75, 0.85, 0.92, 0.96]:
        if time.monotonic() > deadline - 30:
            break
        rem = deadline - time.monotonic() - 15
        n_it3 = int(min(5000, max(500, rem * 150)))
        f_t, s_t = _mirror_descent(best_f, n_it3, 0.3, tf, momentum=0.8, label=f"ref_tf{tf}")
        if s_t < best_score:
            best_score = s_t
            best_f = f_t.copy()
            print(f"  ** Refinement new best (tf={tf}): {best_score:.6f} **")

    # 3b: LP Frank-Wolfe if n is small enough
    if len(best_f) <= 256 and (deadline - time.monotonic()) > 30:
        lp_budget = min(80.0, deadline - time.monotonic() - 15)
        f_lp, s_lp = _lp_frank_wolfe(best_f, max_iters=30, time_budget=lp_budget)
        if s_lp < best_score:
            best_score = s_lp
            best_f = f_lp.copy()
            print(f"  ** LP new best: {best_score:.6f} **")

    # 3c: Perturbation restarts
    for pert_frac in [0.01, 0.03, 0.05]:
        if time.monotonic() > deadline - 30:
            break
        f_pert = _random_perturb(best_f, rng, pert_frac)
        if f_pert is None:
            continue
        rem = deadline - time.monotonic() - 15
        n_it4 = int(min(4000, max(400, rem * 100)))
        f_p, s_p = _mirror_descent(f_pert, n_it4, 0.35, 0.8, momentum=0.75, label=f"pert{pert_frac}")
        if s_p < best_score:
            best_score = s_p
            best_f = f_p.copy()
            print(f"  ** Perturb new best ({pert_frac}): {best_score:.6f} **")

    # 3d: Final fine-tuning with very small eta
    if (deadline - time.monotonic()) > 20:
        rem = deadline - time.monotonic() - 10
        n_it5 = int(min(8000, max(1000, rem * 200)))
        f_fin, s_fin = _mirror_descent(best_f, n_it5, 0.15, 0.9, momentum=0.85, label="final")
        if s_fin < best_score:
            best_score = s_fin
            best_f = f_fin.copy()
            print(f"  ** Final tuning new best: {best_score:.6f} **")

    # --- Final verification ---
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.6f} (n={len(final_list)})")

    if not np.isfinite(final_score) or final_score > best_score + 0.001:
        # Fallback to parent
        print("  Falling back to parent.")
        return [float(x) for x in parent]

    return final_list
