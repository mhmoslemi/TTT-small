def normalize_sequence(arr):
    x = np.asarray(arr, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x

def convolve_fft(f, g):
    n = len(f)
    m = len(g)
    size = 2 * max(n, m)  # need at least 2n-1 points
    size = 1 << (size - 1).bit_length()  # round up to power of 2
    F = fft(f, size)
    G = fft(g, size)
    result = ifft(F * G)
    return result[:n+m-1].real

def score_normalized(x, n):
    conv = convolve_fft(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m) or m < 0:
        m = float(np.max(np.convolve(x, x)))
    return float(2.0 * n * m)

def mirror_descent_step(f, t, k_star, n):
    """Compute f_new = normalize(f * exp(-t * subgradient))"""
    # Compute subgradient of (f*f)_{k*} w.r.t. f
    # s_i = 2*f_{k*-i} if valid and 2i != k*, else f_{k*-i} if 2i == k*
    s = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            if 2 * i == k_star:
                s[i] = f[j]
            else:
                s[i] = 2.0 * f[j]
    # Mirror descent step
    f_new = f * np.exp(-t * s)
    f_new = np.maximum(f_new, 1e-300)
    s_new = f_new.sum()
    if s_new <= 1e-300:
        return f.copy()
    f_new /= s_new
    return f_new
