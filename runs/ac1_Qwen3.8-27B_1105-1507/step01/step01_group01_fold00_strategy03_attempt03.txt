def central_difference(func, x, eps=1e-4):
    """Approximates the gradient of func at x using central finite differences."""
    if not isinstance(x, np.ndarray):
        x = np.array(x, dtype=np.float64)
    n = x.size
    grad = np.zeros(n, dtype=np.float64)
    hess = np.zeros((n, n), dtype=np.float64)
    
    # Forward and backward passes
    for i in range(n):
        # Forward difference
        x_forward = x.copy()
        x_forward[i] += eps
        # Backward difference
        x_backward = x.copy()
        x_backward[i] -= eps
        
        # Compute full directional derivative
        f_forward = func(x_forward.tolist())
        f_backward = func(x_backward.tolist())
        grad[i] = (f_forward - f_backward) / (2 * eps)
    
    return grad

def normalize_to_sum1(a):
    """Normalizes an array a to have sum 1."""
    if isinstance(a, list):
        a = np.array(a, dtype=np.float64)
    s = np.sum(a)
    if not np.isfinite(s) or s < 0.0001:
        a = np.ones_like(a) * (np.random.exponential() / (1 + np.random.exponential()))
        s = np.sum(a)
    return a / s if s != 0 else np.ones(len(a))
