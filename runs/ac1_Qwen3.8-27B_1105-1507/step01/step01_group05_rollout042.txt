conv(f, g)[k] = (A @ g)[k] where A is the (2n-1) x n matrix.

This is equivalent to: minimize t subject to (A @ g)[k] <= t for all k, sum(g) = 1, g >= 0, t free.

"""
Hybrid LP + line search optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses a min-max LP (minimize t s.t. conv(f,g)<=t, sum(g)=1, g>=0) to find descent
directions, combined with line search, subgradient fallback, multiple starting
points, and adaptive time allocation across sequence lengths.
"""

import numpy as np

def compute_score(a):
    """Compute 2*n*max(conv)/sum^2 for sequence a (1D array)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)


def build_conv_matrix(f):
    """Build matrix A (shape (2n-1, n)) such that A @ b = np.convolve(f, b)."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(n):
        for j in range(k + 1):
            A[k, j] = f[k - j]
    for k in range(n, m):
        j_start = k - n + 1
        for j in range(j_start, n):
            A[k, j] = f[k - j]
    return A


def solve_lp_direction(f):
    """
    Solve LP: minimize t s.t. A(f)@g <= t, sum(g)=1, g>=0, t>=0.
    Returns (g_normalized, t_opt) or (None, None).
    """
    n = len(f)
    m = 2 * n - 1
    A = build_conv_matrix(f)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10.0})
        if res.success and res.x is not None:
            g = np.clip(res.x[:n], 0, None)
            g_sum = g.sum()
            if g_sum > 1e-15:
                g = g / g_sum
            return g, res.x[n]
    except Exception:
        pass
    return None, None

            print(f"    *** NEW BEST: {score:.6f} ***")

    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 15:
        n_best = len(best_overall)
        print(f"\n=== Final polishing phase ({remaining:.0f}s) ===")

        # Direct optimization on best
        t_direct = min(remaining * 0.4, 300)
        if t_direct > 10:
            f_opt, score = optimize_one(best_overall.copy(), t_direct, rng, verbose=True)
            if np.isfinite(score) and score < best_score_overall:
                best_score_overall = score
                best_overall = f_opt.copy()
                print(f"  Direct polish: {score:.6f}")

        # Random restarts
        remaining2 = deadline - time.monotonic()
        n_restarts = min(8, max(1, int(remaining2 / 25)))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 8:
                break
            scales = [0.02, 0.05, 0.1, 0.15, 0.2, 0.03, 0.08, 0.12]
            sc = scales[r % len(scales)]
            pert = best_overall * (1.0 + sc * rng.normal(0, 1, n_best))
            pert = np.clip(pert, 0, None)
            if pert.sum() < 1e-10:
                continue
            pert = pert / pert.sum()
            t_left = deadline - time.monotonic() - 8
            if t_left < 8:
                break
            t_restart = min(t_left, max(10, remaining2 / (n_restarts - r)))
            f_opt, score = optimize_one(pert, t_restart, rng, verbose=False)
            if np.isfinite(score) and score < best_score_overall:
                best_score_overall = score
                best_overall = f_opt.copy()
                print(f"  Restart {r} (scale={sc}): {score:.6f}")

    # Return best
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    print("Fallback: uniform 256")
    return [1.0] * 256
