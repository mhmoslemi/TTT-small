"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based best response (mirror descent) with adaptive line search
2. Log-sum-exp smoothed mirror descent (multiplicative weight updates) with lambda annealing
3. Subgradient projection method
4. Multiple sequence lengths (512 to 8192) and diverse initializations
Combines structured LP updates with smooth gradient methods for robust convergence.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_A_matrix(a):
    """Build Toeplitz-like constraint matrix A where A[k,j] = a[k-j]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_matrix(a)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                bounds=[(0, None)] * n, method='highs')
    if r.success:
        g = r.x
        if g.sum() > 1e-10:
            return g
    return None


def compute_smoothed_grad(a, lam):
    """Compute gradient of (1/lam)*log(sum(exp(lam*b))) where b=conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    # Log-sum-exp for numerical stability
    bm = b.max()
    shifted = lam * (b - bm)
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    log_sum_exp = bm + (max_shifted + np.log(exp_shifted.sum())) / lam
    obj = log_sum_exp
    # Softmax weights
    p = exp_shifted / exp_shifted.sum()
    # grad[i] = 2 * sum_j p[i+j] * a[j] for j=0..n-1
    # This is cross-correlation: r = convolve(a[::-1], p, 'full')[n-1:2*n-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    F_ar = np.fft.rfft(a[::-1], size)
    F_p = np.fft.rfft(p, size)
    conv = np.fft.irfft(F_ar * F_p, size)[:3 * n - 1]
    r = conv[n - 1:2 * n - 1]
    grad = 2.0 * r
    return obj, grad


def evaluate_J(a):
    """Compute J(a) directly."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * n * b.max() / (s * s))


def renormalize(a):
    """Clip to non-negative and normalize to unit sum."""
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        return None
    return a / s


def make_initializations(n, best_seq, rng):
    """Generate diverse initializations for length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if best_seq is not None and best_seq.size > 0:
        t_old = np.linspace(0, 1, best_seq.size)
        t_new = np.linspace(0, 1, n)
        a_interp = np.maximum(np.interp(t_new, t_old, best_seq), 0.001)
        inits.append(a_interp)

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Cosine shapes (various frequencies)
    for freq in [2, 3, 4, 6, 8]:
        inits.append(1.0 + 0.4 * np.cos(freq * np.pi * t_arr))

    # Sine shapes
    for freq in [2, 4]:
        inits.append(1.0 + 0.5 * np.sin(freq * np.pi * t_arr))

    # Asymmetric ramp
    inits.append(1.0 + 0.6 * t_arr)
    inits.append(1.0 - 0.4 * t_arr)

    # Bell / Gaussian shapes
    for width in [0.2, 0.4, 0.7]:
        inits.append(np.exp(-((t_arr - 0.5) ** 2) / (2 * width ** 2)))

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Chirp
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 5 * t_arr ** 2))

    # Normalize all
    normed = []
    for a in inits:
        a_clip = np.clip(a, 0.001, None)
        s = a_clip.sum()
        if s > 1e-10:
            normed.append(a_clip / s)
    return normed


def lp_mirror_descent(a, deadline, max_iters=200):
    """LP-based mirror descent with adaptive line search."""
    n = len(a)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = renormalize(g)
        if g is None:
            break
        # Adaptive line search
        best_t = 0.1
        best_val = M
        for t in [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
            a_try = (1 - t) * a + t * g
            a_try = renormalize(a_try)
            if a_try is None:
                continue
            b_try = fast_conv(a_try)
            val = float(b_try.max())
            if val < best_val:
                best_val = val
                best_t = t
        a = (1 - best_t) * a + best_t * g
        a = renormalize(a)
        if a is None:
            break
    return a


def smooth_mirror_descent(a, deadline, max_iters_per_lam=80):
    """Log-sum-exp smoothed mirror descent with lambda annealing."""
    n = len(a)
    lam_schedules = [
        [20, 50, 100, 200, 500, 1000, 2000],
        [50, 100, 200, 500, 1000, 2000, 5000],
        [100, 300, 500, 1000, 2000, 5000, 10000],
    ]
    for lam in lam_schedules[0] + lam_schedules[1]:
        if time.monotonic() > deadline:
            break
        obj, grad = compute_smoothed_grad(a, lam)
        # Project gradient to simplex tangent space
        grad_proj = grad - np.mean(grad)
        # Mirror descent (multiplicative weight update)
        step = 1.0 / (lam * n ** 0.5 + 1)
        for gd_iter in range(max_iters_per_lam):
            if time.monotonic() > deadline:
                break
            # Multiplicative update
            a_try = a * np.exp(-step * grad_proj)
            a_try = renormalize(a_try)
            if a_try is None:
                step *= 0.5
                continue
            obj_try, grad_try = compute_smoothed_grad(a_try, lam)
            if obj_try < obj:
                a = a_try
                obj = obj_try
                grad = grad_try
                grad_proj = grad - np.mean(grad)
            else:
                step *= 0.7
            if step < 1e-12:
                break
    return a


def subgradient_method(a, deadline, max_iters=300):
    """Subgradient method for max(b) with simplex projection."""
    n = len(a)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        # Find indices near the maximum
        thresh = M * 0.9
        near_max = np.where(b >= thresh)[0]
        if len(near_max) == 0:
            near_max = np.array([int(np.argmax(b))])

        # Compute subgradient (average of subgradients at near-max points)
        grad = np.zeros(n)
        for k in near_max:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            grad[jl:jh + 1] += 2.0 * a[k - jh:k - jl + 1][::-1]
        grad /= len(near_max)

        # Project to simplex tangent
        grad_proj = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_proj)
        if gnorm < 1e-14:
            break

        # Step size
        eta = 0.5 * n / (it + 10)
        a_try = a - eta * grad_proj
        a_try = renormalize(a_try)
        if a_try is None:
            eta *= 0.5
            a_try = a - eta * grad_proj
            a_try = renormalize(a_try)
            if a_try is None:
                break
        a = a_try
    return a
