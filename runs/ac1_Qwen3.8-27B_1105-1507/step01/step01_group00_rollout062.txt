def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]

        return 1.0 / np.sqrt(1 - tc_clipped ** 2)
    elif name == 'uniform':
        return np.ones(n)
    elif name == 'random':
        if rng is None:
            rng = np.random.default_rng(0)
        return rng.uniform(0.2, 1.0, size=n)
    elif name == 'bimodal':
        return (np.exp(-4 * (tc - 0.5) ** 2) + np.exp(-4 * (tc + 0.5) ** 2))
    elif name == 'step_taper':
        f = np.ones(n)
        edge = max(1, n // 10)
        f[:edge] = np.linspace(0.1, 1.0, edge)
        f[-edge:] = np.linspace(1.0, 0.1, edge)
        return f
    elif name == 'inv_sqrt':
        # 1/sqrt(i*(n-1-i)) with edge handling
        i = np.arange(1, n)
        f_inner = 1.0 / np.sqrt(i * (n - 1 - i))
        f = np.ones(n)
        f[1:-1] = f_inner
        f[0] = f_inner[0] * 0.5
        f[-1] = f_inner[-1] * 0.5
        return f / f.max()
    else:
        return np.ones(n)


def propose_candidate(budget_s=1500, **kwargs):
    """Main search function: find best sequence minimizing the evaluation."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    best_score = float('inf')
    best_seq = None
    rng = np.random.default_rng(12345)

    # --- Load parent ---
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            s = evaluate_sequence(parent.tolist())
            if np.isfinite(s):
                best_score = s
                best_seq = parent.copy()
                print(f"Parent: n={parent.size}, score={s:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # --- Quick evaluation of constant sequence (baseline = 2.0) ---
    const_seq = np.ones(256)
    s = evaluate_sequence(const_seq.tolist())
    print(f"Constant n=256: score={s:.6f}")
    if s < best_score:
        best_score = s
        best_seq = const_seq.copy()

    # --- Parameterized initializations across lengths ---
    lengths = [256, 384, 512, 768, 1024]
    init_names = ['sin0.25', 'sin0.5', 'sin1', 'sin2', 'sin3',
                  'parab', 'parab2', 'gauss0.5', 'gauss1', 'gauss2',
                  'arcsine', 'step_taper', 'inv_sqrt', 'bimodal', 'uniform']

    # Phase 1: Quick scan to find promising (length, init) combinations
    print("\n=== Phase 1: Quick scan ===")
    candidates = []
    for n in lengths:
        if time.monotonic() > deadline - 120:
            break
        for name in init_names:
            f_init = make_initialization(n, name, rng)
            f_init = np.clip(f_init, 0.0, 1000.0)
            s = evaluate_sequence(f_init.tolist())
            if np.isfinite(s):
                candidates.append((s, n, name, f_init.copy()))
                if s < best_score:
                    best_score = s
                    best_seq = f_init.copy()
                    print(f"  NEW BEST: n={n} {name}: {s:.6f}")

    # Sort by initial score
    candidates.sort(key=lambda x: x[0])
    print(f"Scanned {len(candidates)} initializations. Top 5:")
    for s, n, name, _ in candidates[:5]:
        print(f"  n={n} {name}: {s:.6f}")

    # Phase 2: LP-optimize the top candidates
    print("\n=== Phase 2: LP optimization ===")
    top_k = min(8, len(candidates))
    time_per_candidate = (deadline - time.monotonic()) / (top_k + 2)

    for i in range(top_k):
        s_init, n, name, f_init = candidates[i]
        if time.monotonic() > deadline - 30:
            break

        sub_deadline = min(deadline - 30, time.monotonic() + time_per_candidate)
        print(f"  [{i+1}/{top_k}] n={n} {name} (init={s_init:.6f})", end="")

        f_opt, s_opt = lp_optimize(f_init, sub_deadline, max_iters=200,
                                   verbose_prefix=f"[n={n} {name}] ")
        if s_opt < best_score:
            best_score = s_opt
            best_seq = f_opt.copy()
            print(f" -> LP: {s_opt:.6f} ***")
        else:
            print(f" -> LP: {s_opt:.6f}")

        # Subgradient polish
        if time.monotonic() < deadline - 20 and (time.monotonic() - t0) < budget_s * 0.8:
            f_sub, s_sub = subgradient_optimize(f_opt,
                                                min(deadline - 15, time.monotonic() + time_per_candidate * 0.5),
                                                lr_init=0.05, momentum=0.85)
            if s_sub < best_score:
                best_score = s_sub
                best_seq = f_sub.copy()
                print(f"  -> subgrad: {s_sub:.6f} ***")

    # Phase 3: Optimize the parent specifically
    if parent is not None and time.monotonic() < deadline - 60:
        print("\n=== Phase 3: Parent optimization ===")
        sub_deadline = min(deadline - 30, time.monotonic() + 120)
        print(f"  Parent n={len(parent)} (score={best_score:.6f})", end="")
        f_opt, s_opt = lp_optimize(parent, sub_deadline, max_iters=200,
                                   verbose_prefix="[parent] ")
        if s_opt < best_score:
            best_score = s_opt
            best_seq = f_opt.copy()
            print(f" -> LP: {s_opt:.6f} ***")
        else:
            print(f" -> LP: {s_opt:.6f}")

        if time.monotonic() < deadline - 20:
            f_sub, s_sub = subgradient_optimize(f_opt,
                                                min(deadline - 10, time.monotonic() + 60),
                                                lr_init=0.03, momentum=0.9)
            if s_sub < best_score:
                best_score = s_sub
                best_seq = f_sub.copy()
                print(f"  -> subgrad: {s_sub:.6f} ***")

    # Phase 4: Perturbation around best
    if best_seq is not None and time.monotonic() < deadline - 30:
        print("\n=== Phase 4: Perturbation ===")
        sub_deadline = min(deadline - 10, time.monotonic() + 90)
        f_pert, s_pert = perturb_and_optimize(best_seq, sub_deadline, rng, n_perturb=10)
        if s_pert < best_score:
            best_score = s_pert
            best_seq = f_pert.copy()
            print(f"  Perturbation improved to: {s_pert:.6f}")

    # Phase 5: Final refinement
    if best_seq is not None and time.monotonic() < deadline - 5:
        print("\n=== Phase 5: Final refinement ===")
        sub_deadline = deadline
        f_final, s_final = lp_optimize(best_seq, sub_deadline, max_iters=100,
                                       verbose_prefix="[final] ")
        if s_final < best_score:
            best_score = s_final
            best_seq = f_final.copy()
            print(f"  LP: {s_final:.6f} ***")

        if time.monotonic() < deadline - 3:
            f_sub, s_sub = subgradient_optimize(best_seq, deadline,
                                                lr_init=0.01, momentum=0.92)
            if s_sub < best_score:
                best_score = s_sub
                best_seq = f_sub.copy()
                print(f"  subgrad: {s_sub:.6f} ***")

    if best_seq is None:
        return [1.0] * 256

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
