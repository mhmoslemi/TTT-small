Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses accelerated mirror descent with FFT-based convolution, multiple initializations,
and multi-scale refinement strategy.
"""

import numpy as np
import time

def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_normalized(f):
    """Score assuming sum(f)=1, f>=0. Returns 2n*max(conv(f,f))."""
    n = len(f)
    h = _fft_convolve_full(f)
    return 2.0 * n * float(np.max(h))

def _normalize(f):
    """Normalize f to sum=1, clip negatives."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None

def _pressure_from_active(f, active_lags):
    """Compute pressure vector from active lags."""
    n = len(f)
    p = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            # p[j] += 2 * f[k-j] for j in [j_start, j_end]
            # f[k-j] for j in [j_start, j_end] is f[k-j_end] ... f[k-j_start]
            # which is f[k-j_end : k-j_start+1]
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end+1] += 2.0 * f[lo:hi+1][::-1]
    p /= float(len(active_lags))
    return p

def _cosine_init(n, power=2.0):
    j = np.arange(n, dtype=np.float64)
    f = np.sin(np.pi * (j + 0.5) / float(n)) ** power
    f = np.maximum(f, 0.0)
    return _normalize(f)

def _gaussian_init(n, sigma_frac=0.3):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    return _normalize(f)

def _uniform_init(n):
    return np.ones(n, dtype=np.float64) / n

def _bimodal_init(n, gap_frac=0.3):
    """Initialize with two bumps."""
    j = np.arange(n, dtype=np.float64)
    center1 = gap_frac * n
    center2 = n - gap_frac * n
    sigma = 0.1 * n
    f = np.exp(-(j - center1)**2 / (2*sigma*sigma)) + np.exp(-(j - center2)**2 / (2*sigma*sigma))
    return _normalize(f)

def _upsample(f, factor=2):
    """Upsample by inserting zeros."""
    n = len(f)
    new_n = n * factor
    new_f = np.zeros(new_n, dtype=np.float64)
    new_f[::factor] = f
    return _normalize(new_f)

def _interpolate_upsample(f, factor=2):
    """Upsample by linear interpolation."""
    n = len(f)
    new_n = n * factor
    old_x = np.arange(n, dtype=np.float64)
    new_x = np.arange(new_n, dtype=np.float64) / factor
    new_f = np.interp(new_x, old_x, f)
    return _normalize(new_f)

def _softmax_pressure(f, temperature):
    """Compute gradient of smooth max (logsumexp/temperature) approximation using FFT."""
    n = len(f)
    h = _fft_convolve_full(f)

    Th = temperature * h
    Th_shifted = Th - np.max(Th)
    exp_Th = np.exp(Th_shifted)
    Z = np.sum(exp_Th)
    w = temperature * exp_Th / Z  # length 2n-1, sums to temperature

    # Gradient: g[j] = 2 * sum_k w[k] * f[k-j] for valid indices
    # This is 2 * cross_correlation(w, f)[j]
    # Compute via FFT: cross_corr = ifft(fft(w) * conj(fft(f))) padded appropriately
    size = 1
    while size < 4 * n:
        size <<= 1

    W = np.fft.rfft(w, size)
    F = np.fft.rfft(f, size)

    # ifft(conj(W) * F)[j] = sum_k w[k] * f[k-j] (circular, but we padded enough)
    # Actually: ifft(conj(W)*F) gives circular cross-correlation
    # r[m] = sum_k w[k] * f[(k+m) mod size]... let me be careful.
    # numpy: ifft(fft(a) * conj(fft(b))) = cross_correlation(a, b)
    # cross_correlation(a,b)[d] = sum_k a[k+d] * b[k]
    # We want sum_k w[k] * f[k-j] = cross_correlation(f, w)[j]
    # = ifft(fft(f) * conj(fft(w)))[j]

    cross = np.fft.irfft(F * np.conj(W), size)
    g = 2.0 * cross[:n]

    return g, h

            candidates.append((f"gauss_{n}_{int(sf*100)}", _gaussian_init(n, sf)))
    for n in [1024, 2048]:
        for gf in [0.2, 0.3, 0.4]:
            candidates.append((f"bimod_{n}_{int(gf*100)}", _bimodal_init(n, gf)))
    for n in [1024, 2048]:
        fr = _normalize(rng.random(n))
        if fr is not None:
            candidates.append((f"rand_{n}", fr))

    print(f"  {len(candidates)} candidates")
    reserved = 0.70 * (budget_s - 5)
    p1_budget = max(30.0, (deadline - time.monotonic()) - reserved)
    time_per = max(1.0, min(8.0, p1_budget / len(candidates)))

    p1_results = []
    for name, f0 in candidates:
        if time.monotonic() > deadline - reserved:
            break
        n_iter = int(max(100, min(1200, time_per * 200)))
        f_res, s_res = _mirr_descent_adaptive(f0, n_iter, 1.0, 0.7, label=name)
        p1_results.append((s_res, name, f_res))
        if s_res < best_score:
            best_score = s_res
            best_f = f_res.copy()
            print(f"  ** New best: {best_score:.6f} from {name}")

    p1_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: Nesterov refinement
    print("\n=== Phase 2: Nesterov refinement ===")
    top_k = min(4, len(p1_results))
    for rank in range(top_k):
        if time.monotonic() > deadline - 0.30 * (budget_s - 5):
            break
        _, name, f_cur = p1_results[rank]
        n = len(f_cur)
        remaining = deadline - time.monotonic() - 0.25 * (budget_s - 5)
        if remaining < 20:
            break

        for eta_i in [1.0, 2.0]:
            for tf in [0.5, 0.8]:
                if time.monotonic() > deadline - 0.25 * (budget_s - 5):
                    break
                n_it = int(max(2000, min(20000, remaining * 120)))
                f_res, s_res = _mirr_descent_nesterov(
                    f_cur, n_it, eta_i, tf, False, 0, label=f"N_{name}_e{eta_i}_t{tf}")
                if s_res < best_score:
                    best_score = s_res
                    best_f = f_res.copy()
                    print(f"  ** New best (Nesterov): {best_score:.6f}")
                f_cur = f_res

        if time.monotonic() < deadline - 0.25 * (budget_s - 5):
            for tmp in [80.0, 150.0]:
                if time.monotonic() > deadline - 0.25 * (budget_s - 5):
                    break
                n_it = int(max(1000, min(10000, (deadline - time.monotonic() - 0.2 * (budget_s - 5)) * 80)))
                f_res, s_res = _mirr_descent_nesterov(
                    f_cur, n_it, 1.5, 0.7, True, tmp, label=f"S_{name}_T{int(tmp)}")
                if s_res < best_score:
                    best_score = s_res
                    best_f = f_res.copy()
                    print(f"  ** New best (smooth T={tmp}): {best_score:.6f}")
                f_cur = f_res

    # Phase 3: Multi-scale
    print("\n=== Phase 3: Multi-scale ===")
    if best_f is not None and time.monotonic() < deadline - 0.15 * (budget_s - 5):
        current = best_f.copy()
        for target_n in [2048, 4096, 8192]:
            if len(current) >= target_n or time.monotonic() > deadline - 0.12 * (budget_s - 5):
                break
            f_up = _smooth_upsample(current, target_n)
            if f_up is None:
                continue
            n_it = int(max(3000, min(15000, (deadline - time.monotonic() - 0.1 * (budget_s - 5)) * 130)))
            f_res, s_res = _mirr_descent_nesterov(
                f_up, n_it, 1.5, 0.7, False, 0, label=f"upsample_{target_n}")
            print(f"  n={len(f_res)}: score={s_res:.6f} (prev best={best_score:.6f} at n={len(best_f)})")
            if s_res < best_score:
                best_score = s_res
                best_f = f_res.copy()
                current = f_res.copy()
                print(f"  ** New best (multi-scale n={target_n}): {best_score:.6f}")
            else:
                break

    # Phase 4: Perturbation restarts
    print("\n=== Phase 4: Perturbation ===")
    if best_f is not None:
        for att in range(6):
            if time.monotonic() > deadline - 10:
                break
            if att % 3 == 0:
                f_p = _normalize(best_f + rng.random(len(best_f)) * 0.06)
            elif att % 3 == 1:
                f_p = _normalize(best_f * np.abs(1.0 + rng.standard_normal(len(best_f)) * 0.04))
            else:
                f_p = best_f.copy()
                for _ in range(len(best_f) // 8):
                    i, j = rng.integers(0, len(best_f), 2)
                    f_p[i], f_p[j] = f_p[j], f_p[i]
                f_p = _normalize(f_p)
            if f_p is None:
                continue
            n_it = int(max(1000, min(8000, (deadline - time.monotonic() - 5) * 130)))
            f_res, s_res = _mirr_descent_adaptive(f_p, n_it, 0.8, 0.75, label=f"pert{att}")
            if s_res < best_score:
                best_score = s_res
                best_f = f_res.copy()
                print(f"  ** New best (perturb {att}): {best_score:.6f}")

    # Phase 5: Final polish
    print("\n=== Phase 5: Final polish ===")
    if best_f is not None and time.monotonic() < deadline - 3:
        n_it = int(max(2000, min(15000, (deadline - time.monotonic() - 2) * 150)))
        f_res, s_res = _mirr_descent_nesterov(
            best_f, n_it, 0.8, 0.9, False, 0, label="final_polish")
        if s_res < best_score:
            best_score = s_res
            best_f = f_res.copy()
        print(f"  After polish: {best_score:.6f}")

    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified: score={final_score:.6f}, n={len(final_list)}")

    if not np.isfinite(final_score) or final_score > 2.0:
        if parent is not None:
            print("  Falling back to parent.")
            p_score = float(evaluate_sequence([float(x) for x in parent]))
            if np.isfinite(p_score) and p_score < final_score:
                return [float(x) for x in parent]
        return [1.0 / 1024.0] * 1024

    return final_list
