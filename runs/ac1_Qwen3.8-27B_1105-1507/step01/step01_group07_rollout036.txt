            a[i] = 0.0
            
            # Compute conv of a with a[i]=0
            a_prime = a.copy()
            conv_prime = np.convolve(a_prime, a_prime)

            d = np.zeros(2*n-1)

                if 0 <= j < n and j != i:
                    d[k] = a_prime[j]

            eps = np.zeros(2*n-1)
            if 2*i < 2*n-1:

            T = np.sum(a_prime)
            
            # Find x that minimizes 2*n*max(conv_prime + 2*x*d + x^2*eps) / (T+x)^2

        s = n

"""
Coordinate descent for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses exact 1D subproblem solving with dense grid search, multiple diverse
initializations, and adaptive time allocation across sequence lengths.
Includes perturbation-based restarts to escape local optima.

import numpy as np
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def coord_descent(a, max_sweeps=5000, deadline=None, verbose=False, perturb_every=50):
    """
    Coordinate descent with exact 1D subproblem solving.
    a: non-negative array (normalized to sum=1 internally)
    Returns (a_normalized, best_obj).
    """
    n = len(a)
    a = np.maximum(0.0, a.astype(np.float64)).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_obj = compute_obj(a)
    best_a = a.copy()

    no_improve_count = 0
    prev_obj = best_obj

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        any_change = False

        for i in range(n):

            a[i] = 0.0
            conv0 = np.convolve(a, a)
            T = 1.0 - old

            # d[k] = a[k-i] for valid k (a[i] is 0, so d[2i]=0 automatically)
            d = np.zeros(2 * n - 1)
            k_lo = i
            k_hi = i + n - 1
            if k_lo < 2 * n - 1:
                end = min(k_hi, 2 * n - 2)
                d[k_lo:end + 1] = a[:end - k_lo + 1]

            has_eps = True  # 2*i is always in [0, 2n-2] for i in [0, n-1]

            # Find best x >= 0 minimizing F(x) = 2n*max(conv0+2x*d+x^2*eps)/(T+x)^2
            # Use log-spaced candidates

            cands = np.concatenate([
                [0.0, old],
                np.geomspace(max(1e-8, 1.0 / n * 0.001), x_max, 25),
                np.linspace(0.0, min(old * 2.0, x_max), 10),
            ])
            cands = np.unique(cands)

            best_x = old
            best_F = np.inf
            for x in cands:
                denom = T + x
                if denom < 1e-12:
                    continue

                g[2 * i] += x * x
                F = 2.0 * n * np.max(g) / (denom * denom)
                if F < best_F:
                    best_F = F
                    best_x = x

            # Refine around best_x
            if best_x > 1e-10:
                lo = max(0.0, best_x * 0.6)
                hi = max(best_x * 1.4, best_x + 1e-6)
                fine = np.linspace(lo, hi, 15)

                        best_x = x
            elif best_x == 0.0:

                for x in small_cands:

                    F = 2.0 * n * np.max(g) / (denom * denom)
                    if F < best_F:
                        best_F = F
                        best_x = x

def generate_init(n, rng, mode=0):
    """Generate diverse initial sequences of length n, normalized to sum=1."""
    if mode == 0:
        a = rng.random(n) + 0.01
    elif mode == 1:
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0.01, np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    elif mode == 2:
        center = n * rng.uniform(0.3, 0.7)
        sigma = n * rng.uniform(0.15, 0.35)
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2) + 0.01
    elif mode == 3:
        a = (rng.random(n) > 0.3).astype(float)
    elif mode == 4:
        n_bumps = int(rng.integers(2, 6))
        a = np.zeros(n) + 0.01
        for _ in range(n_bumps):
            center = rng.uniform(0, n)
            sigma = rng.uniform(n * 0.05, n * 0.25)
            a += np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    elif mode == 5:
        a = np.sin(np.pi * rng.integers(1, 8) * np.arange(n) / n) ** 2 + 0.05
    elif mode == 6:
        # Two-tone pattern
        p = int(max(2, n // int(rng.integers(5, 20))))
        base = np.tile([1.0, 0.3], p)[:n]
        a = base * (0.8 + 0.2 * rng.random(n))
    elif mode == 7:
        # Power law
        alpha = rng.uniform(-2, 2)
        x = np.arange(1, n + 1)
        a = x ** alpha
        a = np.maximum(0.01, a)
    else:
        a = rng.random(n) + 0.01

    a = np.maximum(0, a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
