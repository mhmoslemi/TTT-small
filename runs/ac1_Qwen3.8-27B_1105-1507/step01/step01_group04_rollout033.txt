We want this to be as flat as possible. This is related to "minimum peak power" sequences or "flat spectrum" design.

For non-negative f, this is a constrained optimization problem in the frequency domain.

One approach: start with a flat spectrum target and project back to the time domain (iterative projection / relaxed FFT algorithm).

The **Relaxed FFT (RFFT)** algorithm:
1. Start with f = some initial non-negative sequence
2. Compute F = DFT(f)
3. Target |F|^2 = constant (flat spectrum)
4. Update: F_new = F * (target / |F|) (or similar)
5. f_new = IDFT(F_new)
6. Project f_new to non-negative: f_new = max(f_new, 0)
7. Repeat

This is similar to the "Gerchberg-Saxton" algorithm used in phase retrieval.

Actually, for our problem, we don't need perfectly flat spectrum. We just need to minimize the peak of the autocorrelation. The RFFT approach might work well here.

Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses gradient-based L-BFGS optimization with softmax parameterization and adaptive
max-smoothing, combined with RFFT-based initialization and multi-start strategy.

def _fft_convolve(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score(f):
    """Compute score for normalized f (sum=1, f>=0)."""
    n = len(f)
    P = _fft_convolve(f)
    M = float(np.max(P))
    return 2.0 * n * M

def _soft_max_weighted(f, alpha):
    """
    Compute soft-max approximation and gradient for max(conv(f,f)).
    Returns (score, gradient_with_respect_to_theta).
    f = softmax(theta) is assumed.
    """
    n = len(f)
    P = _fft_convolve(f)  # length 2n-1
    # Softmax weights
    alpha_P = alpha * P
    alpha_P -= np.max(alpha_P)  # for numerical stability
    exp_aP = np.exp(alpha_P)
    w = exp_aP / np.sum(exp_aP)
    
    # h = sum_k w[k] * P[k]
    h = float(np.dot(w, P))
    
    # g[j] = sum_{m=0}^{n-1} w[j+m] * f[m] for j = 0,...,n-1
    g = np.correlate(w, f, mode='valid')
    
    # Gradient: 4*n*f*(g - h)
    grad_theta = 4.0 * n * f * (g - h)
    
    # Score (soft): 2n * (1/alpha) * log(sum(exp(alpha*P)))
    # = 2n * h (approximately, for large alpha)
    # Actually: soft_max = (1/alpha)*log(sum(exp(alpha*P))) = (1/alpha)*(max(alpha*P) + log(sum(exp(alpha*P - max))))
    # And h = sum w[k]*P[k] is the "expectation" under the softmax distribution
    # For large alpha, h ≈ max(P)
    
    soft_max_val = h  # approximation; exact would be (1/alpha)*log(sum(exp(alpha*P)))
    # Actually let's compute it exactly:
    soft_max_val = (1.0 / alpha) * (np.max(alpha_P) + np.log(np.sum(exp_aP)))
    
    score = 2.0 * n * soft_max_val
    return score, grad_theta

def _softmax(theta):
    """Stable softmax."""
    theta = np.asarray(theta, dtype=np.float64)
    theta_max = np.max(theta)
    f = np.exp(theta - theta_max)
    f /= np.sum(f)
    return f

    f = np.arange(1, n + 1, dtype=np.float64)
    return f / np.sum(f)

    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / n) ** 2
    f = np.maximum(f, 1e-10)
    return f / np.sum(f)

def _init_rfft(n, n_iter=50, rng=None):
    """RFFT (Relaxed FFT) algorithm for flat spectrum design."""
    if rng is None:
        rng = np.random.default_rng()
    f = rng.random(n)
    f = np.maximum(f, 0.01)
    f = f / np.sum(f)
    
    for _ in range(n_iter):
        F = np.fft.fft(f)
        # Target: flat magnitude
        mag = np.abs(F)
        mag = np.maximum(mag, 1e-10)
        # Update phase
        F_new = F * (1.0 / mag)  # makes |F_new| = 1
        # Actually, let's use a more gradual update
        target_mag = 1.0 / np.sqrt(n)  # flat
        F_new = F * (target_mag / mag) ** 0.5  # gradual
        f_new = np.fft.ifft(F_new).real
        f_new = np.maximum(f_new, 0.0)
        s = np.sum(f_new)
        if s < 1e-10:
            break
        f = f_new / s
    
    return f

**Alternative approach: Direct optimization of the peak**

Instead of using softmax smoothing, I can use the subgradient of the max function directly. The subgradient of max(P) with respect to f[i] is:
- If the max is achieved at a unique lag k*, then ∂max/∂f[i] = 2*f[k*-i] (if 0 <= k*-i < n)
- If the max is achieved at multiple lags, the subgradient is the average

"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses gradient-based L-BFGS optimization with softmax parameterization and adaptive
max-smoothing (log-sum-exp), combined with diverse initializations and multi-start
strategy. Includes parametric family search and perturbation-based restarts.
"""

import numpy as np
import time
from scipy.optimize import minimize


def _fft_convolve(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]


def _score_normalized(f):
    """Compute score for f with sum(f)=1, f>=0. Score = 2n*max(conv(f,f))."""
    n = len(f)
    P = _fft_convolve(f)
    M = float(np.max(P))
    return 2.0 * n * M


def _stable_softmax(theta):
    """Numerically stable softmax."""
    theta = np.asarray(theta, dtype=np.float64)
    tmax = np.max(theta)
    e = np.exp(theta - tmax)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones_like(theta) / len(theta)
    return e / s


class _SoftMaxObjective:
    """Computes smoothed objective and gradient for L-BFGS."""

    def __init__(self, alpha):
        self.alpha = alpha

    def __call__(self, theta):
        f = _stable_softmax(theta)
        n = len(f)
        P = _fft_convolve(f)  # length 2n-1
        aP = self.alpha * P
        aP_shifted = aP - np.max(aP)
        exp_aP = np.exp(aP_shifted)
        sum_exp = np.sum(exp_aP)
        w = exp_aP / sum_exp

        # h = sum_k w[k] * P[k]
        h = float(np.dot(w, P))

        # g[j] = sum_m w[j+m] * f[m] for j=0..n-1 (cross-correlation)
        g = np.correlate(w, f, mode='valid')

        # Gradient: 4*n*f*(g - h)
        grad = 4.0 * n * f * (g - h)

        # Soft max value
        soft_max_val = (1.0 / self.alpha) * (np.max(aP_shifted) + np.log(sum_exp))
        score = 2.0 * n * soft_max_val

        return score, grad


def _optimize_single(f_init, alpha_list, max_iter_total, time_budget):
    """
    Optimize starting from f_init using L-BFGS with increasing alpha schedule.
    Returns (best_f, best_score).
    """
    f0 = np.asarray(f_init, dtype=np.float64).copy()
    f0 = np.maximum(f0, 1e-15)
    f0 = f0 / np.sum(f0)
    n = len(f0)

    theta = np.log(f0)
    best_f = f0.copy()
    best_score = _score_normalized(f0)

    start_time = time.monotonic()
    iters_per_stage = max(10, max_iter_total // len(alpha_list))

    for i, alpha in enumerate(alpha_list):
        if time.monotonic() - start_time > time_budget:
            break

        obj = _SoftMaxObjective(alpha)
        try:

        except Exception:
            break

        s_new = _score_normalized(f_new)
        if s_new < best_score:
            best_score = s_new
            best_f = f_new.copy()
            theta = result.x.copy()

    return best_f, best_score
