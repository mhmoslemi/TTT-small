def optimize_soft_subgradient(a, max_time, seed):
    a = normalize_to_sum1(a)
    n = len(a)
    best = a.copy()
    best_score = local_score(best)
    
    t0 = time.monotonic()
    deadline = t0 + max_time
    
    eta = 0.5  # learning rate
    temperature = 0.5  # softmax temperature (fraction of max)
    
    stuck = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() < deadline:
        C = np.convolve(a, a)
        M = float(np.max(C))
        
        # Softmax weights
        w = np.exp(C / (temperature * M))
        w /= np.sum(w)
        
        # Subgradient
        g = 2.0 * np.correlate(w, a, mode='full')[:n]
        g = np.maximum(g, 0)
        
        # Normalize gradient
        g_max = float(np.max(g))
        if g_max > 0:
            g = g / g_max
        
        # Multiplicative update: a_new = a * exp(-eta * (g - mean_g))
        # The mean subtraction ensures the update preserves the sum approximately.
        mean_g = float(np.mean(g))
        a_new = a * np.exp(-eta * (g - mean_g))
        a_new = np.clip(a_new, 1e-10, None)
        a_new = normalize_to_sum1(a_new)
        
        if a_new is not None:
            score = local_score(a_new)
            if score < best_score:
                best = a_new.copy()
                best_score = score
                stuck = 0
            else:
                stuck += 1
                
            a = a_new
        
        # Adapt learning rate
        if stuck > 50:
            eta *= 0.8
            if eta < 0.01:
                # Restart with noise
                noise = np.exp(rng.normal(0, 0.3, n))
                a = best * noise
                a = normalize_to_sum1(a)
                eta = 0.5
                stuck = 0
        
        # Adapt temperature
        temperature = max(0.01, temperature * 0.999)
    
    return best, best_score

"""

Uses soft-subgradient multiplicative updates with adaptive temperature,
LP-assisted direction finding, and multi-start exploration.
Key improvement: efficient gradient computation using the analytic formula
d conv(a,a)[k]/d a[j] = 2*a[k-j], enabling fast softmax-weighted subgradients.

import time

except Exception:

def normalize_to_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    if not np.isfinite(a).all():
        return None
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-300:
        return None
    return a / s


def force_sym(a):
    a = 0.5 * (a + a[::-1])
    return normalize_to_sum1(a)


def local_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    if not np.isfinite(a).all():
        return float('inf')
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * len(a) * m / (s * s))


def compute_soft_subgradient(a, temperature):
    """
    Compute soft subgradient of max(conv(a,a)) at a.
    Uses analytic gradient: dC[k]/da[j] = 2*a[k-j] if 0<=k-j<n.
    g[j] = 2 * sum_m w[j+m] * a[m] where w = softmax(C / (T*max(C))).
    Computed via np.correlate for efficiency.
    """
    n = len(a)
    C = np.convolve(a, a)
    M = float(np.max(C))
    if M <= 0 or not np.isfinite(M):
        return np.zeros(n, dtype=np.float64)

    # Softmax weights (scale by temperature * M for numerical stability)
    scaled = C / (temperature * M)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros(n, dtype=np.float64)
    w /= w_sum

    # g[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    # = 2 * np.correlate(w, a, 'full')[j]
    g = 2.0 * np.correlate(w, a, mode='full')[:n]
    g = np.maximum(g, 0.0)
    return g


def multiplicative_update(a, eta, temperature, force_sym_flag=False, top_k=None):
    """Single multiplicative subgradient step."""
    n = len(a)
    C = np.convolve(a, a)
    M = float(np.max(C))
    if M <= 0 or not np.isfinite(M):
        return a.copy()

    if top_k is not None and top_k < len(C):
        # Hard top-K subgradient
        idx = np.argpartition(C, -top_k)[-top_k:]
        w = np.zeros(len(C), dtype=np.float64)
        w[idx] = 1.0 / float(top_k)
    else:
        # Soft subgradient
        scaled = C / (temperature * M)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = float(np.sum(w))
        if w_sum < 1e-300:
            return a.copy()
        w /= w_sum

    g = 2.0 * np.correlate(w, a, mode='full')[:n]
    g = np.maximum(g, 0.0)

    g_max = float(np.max(g))
    if g_max < 1e-300:
        return a.copy()
    g = g / g_max

    mean_g = float(np.mean(g))
    fac = -eta * (g - mean_g)
    fac = np.clip(fac, -8.0, 8.0)
    a_new = a * np.exp(fac)
    a_new = np.clip(a_new, 1e-15, 1000.0)

    if force_sym_flag:
        a_new = force_sym(a_new)
    else:
        a_new = normalize_to_sum1(a_new)

    if a_new is None:
        return a.copy()
    return a_new


def soft_subgradient_optimize(a, max_time, seed, force_sym_flag=False, eta_init=0.4):
    """Main optimization loop using soft subgradient multiplicative updates."""
    a = normalize_to_sum1(a)
    if force_sym_flag:
        a = force_sym(a)
    if a is None:
        return None, float('inf')

    best = a.copy()
    best_sc = local_score(best)
    n = len(a)

    t0 = time.monotonic()
    deadline = t0 + float(max_time)
    rng = np.random.default_rng(seed)

    eta = float(eta_init)
    temperature = 1.0
    stuck = 0
    it = 0

    while time.monotonic() < deadline:
        a_new = multiplicative_update(a, eta, temperature, force_sym_flag)
        sc = local_score(a_new)

        if np.isfinite(sc) and sc < best_sc:
            best = a_new.copy()
            best_sc = float(sc)
            stuck = 0
            # Slightly increase eta when making progress
            if eta < 0.8:
                eta = min(eta * 1.005, 0.8)
        else:
            stuck += 1
            a = a_new  # still move (stochastic-like)

        # Anneal temperature
        temperature = max(0.01, temperature * 0.9995)

        # Adapt learning rate
        if stuck > 30:
            eta *= 0.85
        if stuck > 150:
            # Reset with perturbation from best
            noise = np.exp(rng.normal(0.0, 0.2, n))
            a = best * noise
            if force_sym_flag:
                a = force_sym(a)
            else:
                a = normalize_to_sum1(a)
            if a is None:
                a = best.copy()
            eta = max(eta_init * 0.5, 0.1)
            temperature = 0.5
            stuck = 0

        it += 1

    return best, best_sc


def lp_step(f, n, force_sym_flag=False, seed=0):
    """LP step: maximize sum(b) s.t. conv(f,b) <= max(conv(f,f)), b >= 0."""
    if not HAVE_SCIPY:
        return None

    f = np.asarray(f, dtype=np.float64)
    c = np.convolve(f, f)
    M = float(np.max(c))
    if M <= 0 or not np.isfinite(M):
        return None

    rng = np.random.default_rng(seed + 777)

    try:
        if force_sym_flag:
            m = (n + 1) // 2
            # Build constraint matrix for symmetric variables
            A = np.zeros((n, m), dtype=np.float64)
            for v in range(m):
                A[v:n, v] += f[:n - v]
                j2 = n - 1 - v
                if j2 != v:
                    A[j2:n, v] += f[:v + 1]
            rows = n
            cols = m
        else:
            m = n
            A = np.zeros((2 * n - 1, n), dtype=np.float64)
            for k in range(2 * n - 1):
                s = max(0, k - (n - 1))
                e = min(k, n - 1)
                if s <= e:
                    j = np.arange(s, e + 1, dtype=np.int64)
                    A[k, j] = f[k - j]
            rows = 2 * n - 1
            cols = n

        cvec = -np.ones(cols, dtype=np.float64)
        cvec += 1e-9 * rng.uniform(-1.0, 1.0, cols)
        bvec = np.full(rows, M * (1.0 + 1e-8), dtype=np.float64)
        bounds = [(0.0, 1.0)] * cols

        res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
    except MemoryError:
        return None
    except Exception:
        return None

    if res is None or not res.success:
        return None

    x = np.asarray(res.x, dtype=np.float64)
    x = np.clip(x, 0.0, 1.0)
    if x.size == 0 or float(np.sum(x)) < 1e-10:
        return None

    if force_sym_flag:
        # Expand symmetric
        if n % 2 == 0:
            g = np.concatenate((x, x[::-1]))
        else:
            g = np.concatenate((x, x[-2::-1]))
        g = force_sym(normalize_to_sum1(g))
    else:
        g = normalize_to_sum1(x)

    if g is None:
        return None
    return g


def lp_optimize(a, max_time, seed, force_sym_flag=False):
    """LP-assisted optimization with line search."""
    f = normalize_to_sum1(a)
    if force_sym_flag:
        f = force_sym(f)

    deadline = t0 + float(max_time)
    ts_steps = (0.03, 0.06, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0)
    stall = 0
    lp_count = 0

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 5.0:
            break

        g = lp_step(best, n, force_sym_flag, seed + lp_count * 131)
        lp_count += 1
        if g is None:
            break

        new_best = best
        new_sc = best_sc

        for t in ts_steps:
            new = (1.0 - t) * best + t * g
            if force_sym_flag:
                new = force_sym(new)
            else:
                new = normalize_to_sum1(new)
            if new is None:
                continue
            sc = local_score(new)
            if np.isfinite(sc) and sc < new_sc:
                new_sc = float(sc)
                new_best = new.copy()

        if new_sc < best_sc - 1e-8:
            best = new_best
            best_sc = new_sc
            stall = 0
        else:
            stall += 1
            if stall > 5:

            # Small multiplicative polish between LP calls

                if time.monotonic() >= deadline:
                    break
                cand = multiplicative_update(best, 0.2, 0.3, force_sym_flag)
                sc = local_score(cand)
                if np.isfinite(sc) and sc < best_sc:
                    best = cand
                    best_sc = float(sc)
                    break

    return best, best_sc


def make_diverse_init(n, seed, kind, parent_arr=None):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed + 100003 * kind)

    if kind == 0:
        a = rng.uniform(0.05, 1.0, n)
    elif kind == 1:
        # Smooth random
        a = rng.normal(0, 1, n)
        w = max(3, n // 16)
        kern = np.ones(w) / w
        a = np.convolve(a, kern, mode='same')
        a = np.clip(a, 0.01, None)
    elif kind == 2:
        # Bimodal
        center = n // 2
        sigma = max(1, n // 8)
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - 0) / sigma) ** 2) + np.exp(-0.5 * ((x - n + 1) / sigma) ** 2)
        a += 0.1 * rng.uniform(0, 1, n)
    elif kind == 3:
        # Multi-peak
        npeaks = rng.integers(3, max(4, n // 64))
        a = np.zeros(n)
        peak_pos = rng.integers(0, n, npeaks)
        widths = rng.integers(max(2, n // 64), max(3, n // 16), npeaks)
        for p, w in zip(peak_pos, widths):
            for i in range(max(0, p - w), min(n, p + w + 1)):
                d = abs(i - p)
                a[i] += np.exp(-0.5 * (d / max(1, w // 3)) ** 2)
        a += 0.01
    elif kind == 4:
        # Parent perturbation
        if parent_arr is not None and parent_arr.size == n:
            a = parent_arr * np.exp(rng.normal(0, 0.3, n))
            a = np.clip(a, 1e-3, None)
        else:
            a = rng.uniform(0.1, 1.0, n)
    elif kind == 5:
        # Oscillatory
        freq = rng.uniform(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        a = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / n + phase)
        a = np.clip(a, 0.01, None)
        a += 0.05 * rng.uniform(0, 1, n)
    else:
        a = rng.beta(2, 2, n)

    a = normalize_to_sum1(a)
    if a is None:
        a = np.ones(n) / n
    return a

        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
        else:
            # Fallback to parent
            if parent_arr is not None:
                fp = normalize_to_sum1(parent_arr)
                if fp is not None:
                    try:
                        off2 = float(evaluate_sequence([float(x) for x in fp]))
                        if np.isfinite(off2):
                            final = fp
                            best_score = off2
                    except Exception:
                        pass
    except Exception:
        pass

    print(f"FINAL: score={best_score:.6f}, length={len(final)}", flush=True)
    return [float(x) for x in np.clip(np.asarray(final, dtype=np.float64), 0.0, 1000.0)]
