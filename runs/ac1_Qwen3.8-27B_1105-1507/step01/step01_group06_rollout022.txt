def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def grad_conv_max_smooth(a, beta):
    """
    Compute M_smooth = (1/beta)*log(sum exp(beta*(a*a)[k]))
    and its gradient w.r.t. a.
    Returns (M_smooth, grad) where grad[j] = dM_smooth/da[j].
    """
    n = len(a)
    b = fast_conv(a)  # length 2n-1
    b_max = b.max()
    shifted = beta * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_b = np.exp(shifted)
    exp_sum = exp_b.sum()
    if exp_sum < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w = exp_b / exp_sum
    M_smooth = b_max + np.log(exp_sum) / beta

    # = 2 * corr(w, a)[j] where corr is cross-correlation
    # corr[j] = sum_k w[k] * a[k-j]
    # = conv(w, a_flip)[n-1+j] where a_flip[m] = a[n-1-m]
    N = 1
    N <<= (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, N)
    a_flip = np.zeros(N)
    a_flip[:n] = a[::-1]
    Af = np.fft.rfft(a_flip, N)
    corr_full = np.fft.irfft(W * Af, N)
    grad = 2.0 * corr_full[n-1:n-1+n]
    
    return M_smooth, grad

    """Objective J(x) and its gradient."""
    n = len(x)

    s = x_pos.sum()

    # (since max(x,0) has subgradient {0} at x=0)
    grad_J[x < 0] = 0.0
    return float(J), grad_J

def build_A_matrix(a):
    """Build the constraint matrix for the LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # (a*g)[k] = sum_j a[k-j] * g[j] for valid j
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            # a[k-j] for j in [jl, jh]
            # k-j ranges from k-jl to k-jh, which is [n-1, 0] in some order
            idx = k - np.arange(jl, jh + 1)
            A[k, jl:jh+1] = a[idx]
    return A

    if res.success and res.fun > -1e-10:

    return None, None

Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. L-BFGS-B with smoothed (log-sum-exp) analytic gradient
2. LP-based best-response (mirror descent) iterations
3. Multiple lengths, diverse initializations, and final polish
The smoothed gradient uses softmax weights over the convolution peak,
providing a C-infinity approximation to the max for stable optimization.

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def smooth_conv_gradient(a, beta):
    """
    Compute M_smooth = (1/beta)*log(sum_k exp(beta*(a*a)[k])) and gradient.
    Returns (M_smooth, grad) where grad[j] = dM_smooth/da[j].
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    exp_sum = exp_b.sum()
    if exp_sum < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w = exp_b / exp_sum
    M_smooth = b_max + np.log(exp_sum) / beta

    # grad[i] = 2 * sum_{m=0}^{n-1} w[m+i] * a[m]
    # = 2 * conv(a, w_flipped)[2n-2-i]
    w_flipped = w[::-1]
    N = 1
    N <<= (3 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    Wf = np.fft.rfft(w_flipped, N)
    conv_full = np.fft.irfft(A * Wf, N)
    grad = 2.0 * conv_full[n-1:2*n-1][::-1]

    return M_smooth, grad


def obj_grad(x, beta):
    """Objective J(x) = 2*n*max(a*a)/sum(a)^2 and its gradient (smoothed)."""
    n = len(x)
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()

        return 1e10, np.zeros(n)
    M_smooth, grad_M = smooth_conv_gradient(x_pos, beta)
    J = 2.0 * n * M_smooth / (s * s)
    grad_J = 2.0 * n * (grad_M * s - 2.0 * M_smooth) / (s**3)
    grad_J[x < 0] = 0.0
    return float(J), grad_J


def build_A_matrix(a):
    """Build constraint matrix A for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            idx = k - np.arange(jl, jh + 1)
            A[k, jl:jh+1] = a[idx]
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.fun < 0:
        return -res.fun, res.x
    return None, None

def resample_sequence(src, new_n):
    """Resample a sequence to a new length via interpolation."""
    if src.size <= 1:
        return np.ones(new_n)
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, new_n)
    out = np.interp(t_new, t_old, src)
    return np.maximum(out, 0.0)

def run_lbfgs(a0, n, beta, maxiter, max_fun):
    """Run L-BFGS-B on the smoothed objective."""
    bounds = [(0.0, 10.0)] * n

    def fun_and_jac(x):
        return obj_grad(x, beta)

    res = minimize(fun_and_jac, a0, jac=True, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'max_fun': max_fun,
                            'ftol': 1e-15, 'gtol': 1e-10, 'disp': False})
    return res


def run_lp_iterations(a, max_iters, deadline, n):
    """Run LP-based mirror descent iterations."""
    current = a.copy()
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(current)
        M = float(b.max())
        val, g = lp_best_response(current, M)
        if g is None or val < 1e-10:
            break
        g = np.clip(g, 0, None)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        t_mix = max(0.02, 0.5 * (1.0 - it / float(max_iters))**0.5)
        a_new = (1.0 - t_mix) * current + t_mix * g
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        current = a_new
    return current
