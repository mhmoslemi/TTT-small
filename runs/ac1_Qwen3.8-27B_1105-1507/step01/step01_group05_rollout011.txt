import numpy as np
from scipy.optimize import linprog
import time

def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    m = len(f)
    A = np.zeros((2 * n - 1, m))
    for k in range(2 * n - 1):
        j_start = max(0, k - m + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A

def compute_score_fast(f):
    """Compute score for normalized sequence (sum=1)."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * n * M

    f = f / f.sum()

"""
Improved LP-based optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses backtracking line search, multiple LP formulations, structured initializations,
coordinate descent polish, and adaptive time allocation across multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def compute_score_fast(f):
    """Compute score = 2*n*max(conv)/sum^2 for a sequence."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    S = f.sum()
    if S < 1e-15:
        return np.inf
    return 2.0 * n * M / (S * S)


def normalize(f):
    """Normalize f to sum=1, handling edge cases."""
    S = f.sum()
    if S < 1e-15:
        return f
    return f / S


def lp_direction(f, M, mode='max_sum'):
    """
    Find LP direction from current f.
    mode='max_sum': maximize sum(b) s.t. conv(f,b) <= M, b >= 0
    mode='min_max': minimize t s.t. conv(f,b) <= t, sum(b)=1, b >= 0
    """
    n = len(f)
    A = build_conv_matrix(f)
    
    if mode == 'max_sum':
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)]*n, method='highs',
                      options={'presolve': True, 'max_iter': 50000})
        if not res.success or res.x is None:
            return None
        b = np.clip(res.x, 0, None)
        if b.sum() < 1e-15:
            return None
        return b / b.sum()
    
    elif mode == 'min_max':
        # min t s.t. A@b <= t*1, sum(b)=1, b>=0
        # Variables: [b (n vars), t (1 var)]
        n_total = n + 1
        c_obj = np.zeros(n_total)
        c_obj[-1] = 1  # minimize t
        
        # A@b - t <= 0  =>  [A, -1] @ [b; t] <= 0
        A_ub = np.zeros((2*n-1, n_total))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2*n-1)
        
        # sum(b) = 1  =>  A_eq @ [b;t] = 1
        A_eq = np.zeros((1, n_total))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'max_iter': 50000})
        if not res.success or res.x is None:
            return None
        b = np.clip(res.x[:n], 0, None)
        if b.sum() < 1e-15:
            return None
        return b / b.sum()
    
    return None

def coordinate_descent(f0, time_budget, rng, verbose=False):
    """

def spectral_iteration(f0, n_iter=100, time_budget=60):
    """
    Iterative spectral shaping: alternate between time-domain (non-negativity + normalize)
    and frequency-domain (flatten magnitude spectrum).
    """
    n = len(f0)
    f = f0.astype(np.float64).copy()
    f = np.clip(f, 0, None)
    f = normalize(f)
    
    best_f = f.copy()
    best_score = compute_score_fast(f)
    
    start = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - start >= time_budget:
            break
        
        # Frequency domain: flatten magnitude
        F = np.fft.rfft(f)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Target magnitude: geometric mean (soft flatten)
        mg_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
        # Blend: keep some of the original structure
        alpha = 0.3
        target_mag = (1 - alpha) * mag + alpha * mg_mean
        # Avoid zeroing out
        target_mag = np.maximum(target_mag, 1e-10)
        
        F_new = target_mag * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n=n)
        
        # Time domain: enforce non-negativity and normalize
        f_new = np.clip(f_new, 0, None)
        f_new = normalize(f_new)
        
        sc = compute_score_fast(f_new)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
        f = f_new
    
    return best_f, best_score

            else:
                f[bstart:bend] = orig_block
                f = normalize(f)
        
        # Also try redistributing within the block
        if len(orig_block) > 1:
            # Move mass from high to low within block
            for _ in range(5):
                i1 = bstart + int(rng.integers(0, bend - bstart))
                i2 = bstart + int(rng.integers(0, bend - bstart))
                if i1 == i2:
                    continue
                delta = 0.1 * f[i1]
                f[i1] -= delta
                f[i2] += delta
                f = np.clip(f, 0, None)
                f = normalize(f)
                sc = compute_score_fast(f)
                if sc < best_score:
                    best_score = sc
                    best_f = f.copy()
                else:
                    f[i1] += delta
                    f[i2] -= delta
                    f = normalize(f)
    
    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Multi-phase approach:
    1. Diverse initializations (structured + random)
    2. LP optimization with backtracking
    3. Spectral iteration
    4. Coordinate descent polish
    5. Block LP optimization
    6. Final restarts from best solution
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0
    
    best_overall = None
    best_score_overall = np.inf
    
    # Try to get parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None
    
    print(f"Budget: {budget_s}s, deadline in {budget_s-15:.0f}s")
    if parent is not None:
        parent_score = compute_score_fast(parent)
        print(f"Parent sequence: n={len(parent)}, score={parent_score:.6f}")
        best_overall = parent.copy()
        best_score_overall = parent_score
    
    # Define initializations for each n
    def get_initializations(n, rng):
        inits = []
        
        # Uniform with perturbation
        f1 = np.ones(n) * (1 + 0.1 * rng.normal(0, 1, n))
        f1 = np.clip(f1, 0.01, None)
        inits.append(('uniform', f1))
        
        # Cosine taper (raised cosine)
        j = np.arange(n)
        f2 = np.cos(np.pi * j / (n - 1)) ** 2
        f2 = np.maximum(f2, 0.01)
        inits.append(('cos_taper', f2))
        
        # Sine-based: 1 + 0.4*cos(2*pi*j/n)
        f3 = 1.0 + 0.4 * np.cos(2 * np.pi * j / n)
        f3 = np.maximum(f3, 0.01)
        inits.append(('sine1', f3))
        
        # Two-peaked: 1 + 0.4*cos(4*pi*j/n)
        f4 = 1.0 + 0.4 * np.cos(4 * np.pi * j / n)
        f4 = np.maximum(f4, 0.01)
        inits.append(('sine2', f4))
        
        # Edge-enhanced: larger at edges
        f5 = 1.0 + 0.5 * np.cos(np.pi * j / (n - 1))
        f5 = np.maximum(f5, 0.01)
        inits.append(('edge', f5))
        
        # Random
        f6 = np.abs(rng.normal(1, 0.5, n))
        f6 = np.maximum(f6, 0.01)
        inits.append(('random', f6))
        
        # Geometric ramp (asymmetric)
        f7 = np.exp(np.linspace(0, 1, n))
        inits.append(('geo_ramp', f7))
        
        # W-shaped: 1 + 0.3*cos(6*pi*j/n)
        f8 = 1.0 + 0.3 * np.cos(6 * np.pi * j / n)
        f8 = np.maximum(f8, 0.01)
        inits.append(('w_shape', f8))
        
        return inits
    
    # Phase 1: Initial exploration with LP optimization
    # Allocate time based on n (larger n gets more time since it can potentially do better)
    n_configs = [256, 512, 1024]
    
    # If parent is available, include its length
    if parent is not None:
        n_parent = len(parent)
        if n_parent not in n_configs:
            if n_parent >= 128 and n_parent <= 2048:
                n_configs.insert(0, n_parent)
    
    print(f"\n=== Phase 1: LP Optimization ===")
    print(f"Testing n values: {n_configs}")
    
    # Time allocation: give more time to larger n (better theoretical bound)
    total_configs = 0
    for n in n_configs:
        total_configs += len(get_initializations(n, rng))
    
    # We'll do a quick pass first to find the best n, then focus there
    # Quick pass: 60 seconds total
    quick_budget = min(120, (budget_s - 60) * 0.1)
    
    quick_results = {}
    for n in n_configs:
        if time.monotonic() > deadline:
            break
        
        inits = get_initializations(n, rng)
        n_inits = len(inits)
        per_init_time = max(5, quick_budget / max(1, n_configs_count_remaining(n, n_configs)))
        
        best_for_n = None
        best_score_n = np.inf
        
        for name, f_init in inits[:4]:  # Quick pass: only 4 inits
            if time.monotonic() > deadline:
                break
            f_init = np.clip(f_init, 0, None)
            f_init = normalize(f_init)
            
            t_left = min(20, deadline - time.monotonic() - 5)
            if t_left < 5:
                break
            
            f_opt, sc = optimize_lp_improved(f_init, t_left, rng, verbose=False)
            if sc < best_score_n:
                best_score_n = sc
                best_for_n = f_opt
        
        if best_for_n is not None:
            quick_results[n] = (best_for_n, best_score_n)
            if best_score_n < best_score_overall:
                best_score_overall = best_score_n
                best_overall = best_for_n.copy()
            print(f"  n={n}: quick best score = {best_score_n:.6f}")
    
    print(f"\nAfter quick pass: best = {best_score_overall:.6f}")
    
    # Phase 2: Focused optimization on the best 2 n values
    print(f"\n=== Phase 2: Focused Optimization ===")
    
    if quick_results:
        sorted_ns = sorted(quick_results.keys(), key=lambda x: quick_results[x][1])
        focus_ns = sorted_ns[:2]  # Focus on top 2
    else:
        focus_ns = [512]
    
    remaining = deadline - time.monotonic()
    per_n_time = remaining / max(1, len(focus_ns)) - 30
    
    for n in focus_ns:
        if time.monotonic() > deadline:
            break
        
        t_left = min(per_n_time, deadline - time.monotonic() - 20)
        if t_left < 30:
            continue
        
        print(f"\n  Focusing on n={n}, time={t_left:.0f}s")
        
        inits = get_initializations(n, rng)
        
        # Start from the best we already have for this n
        if n in quick_results:
            best_start = quick_results[n][0]
            best_start_score = quick_results[n][1]
        else:
            best_start = None
            best_start_score = np.inf
        
        # Try all initializations with more time
        n_inits = len(inits)
        per_init = max(10, t_left / (n_inits + 2))
        
        for idx, (name, f_init) in enumerate(inits):
            if time.monotonic() > deadline - 20:
                break
            
            f_init = np.clip(f_init, 0, None)
            f_init = normalize(f_init)
            
            # If we have a good start, also try from it
            t_here = min(per_init, deadline - time.monotonic() - 15)
            if t_here < 10:
                break
            
            f_opt, sc = optimize_lp_improved(f_init, t_here, rng, verbose=False)
            if sc < best_start_score:
                best_start_score = sc
                best_start = f_opt
                if sc < best_score_overall:
                    best_score_overall = sc
                    best_overall = f_opt.copy()
                    print(f"    {name}: NEW BEST = {sc:.6f}")
        
        # Also try from the best start with more time
        if best_start is not None:
            t_here = max(30, (deadline - time.monotonic() - 20) * 0.4)
            if t_here > 20:
                f_opt, sc = optimize_lp_improved(best_start, t_here, rng, verbose=False)
                if sc < best_score_overall:
                    best_score_overall = sc
                    best_overall = f_opt.copy()
                    print(f"    Refined: NEW BEST = {sc:.6f}")
                
                # Spectral iteration
                if time.monotonic() < deadline - 30:
                    t_spec = max(10, (deadline - time.monotonic() - 30) * 0.2)
                    f_spec, sc_spec = spectral_iteration(f_opt, n_iter=200, time_budget=t_spec)
                    if sc_spec < best_score_overall:
                        best_score_overall = sc_spec
                        best_overall = f_spec.copy()
                        print(f"    Spectral: NEW BEST = {sc_spec:.6f}")
    
    # Phase 3: Coordinate descent polish
    print(f"\n=== Phase 3: Polish ===")
    print(f"Current best: {best_score_overall:.6f}")
    
    if best_overall is not None:
        t_left = deadline - time.monotonic() - 30
        if t_left > 20:
            n_cd = max(1, int(t_left * 0.5))
            f_cd, sc_cd = coordinate_descent(best_overall, time_budget=min(n_cd, 120), rng=rng)
            if sc_cd < best_score_overall:
                best_score_overall = sc_cd
                best_overall = f_cd.copy()
                print(f"  After coordinate descent: {sc_cd:.6f}")
        
        # Block LP
        t_left = deadline - time.monotonic() - 20
        if t_left > 30:
            f_bl, sc_bl = block_lp_optimize(best_overall, time_budget=min(t_left, 120),
                                            rng=rng, block_size=max(8, len(best_overall)//32))
            if sc_bl < best_score_overall:
                best_score_overall = sc_bl
                best_overall = f_bl.copy()
                print(f"  After block LP: {sc_bl:.6f}")
    
    # Phase 4: Final restarts
    print(f"\n=== Phase 4: Final Restarts ===")
    print(f"Current best: {best_score_overall:.6f}")
    
    if best_overall is not None:
        n_best = len(best_overall)
        t_left = deadline - time.monotonic() - 10
        n_restarts = min(8, max(1, int(t_left / 30)))
        
        for r in range(n_restarts):
            t_here = deadline - time.monotonic() - 5
            if t_here < 15:
                break
            
            # Different perturbation strategies
            strategy = r % 4
            if strategy == 0:
                f0 = best_overall * (1 + 0.3 * rng.normal(0, 1, n_best))
            elif strategy == 1:
                f0 = best_overall.copy()
                # Random block swap
                bsize = max(4, n_best // 16)
                i1 = int(rng.integers(0, n_best - bsize))
                i2 = int(rng.integers(0, n_best - bsize))
                f0[i1:i1+bsize], f0[i2:i2+bsize] = f0[i2:i2+bsize].copy(), f0[i1:i1+bsize].copy()
            elif strategy == 2:
                # Add periodic component
                j = np.arange(n_best)
                freq = int(rng.integers(1, 5))
                f0 = best_overall + 0.1 * np.cos(2 * np.pi * freq * j / n_best)
            else:
                # Invert and blend
                f0 = best_overall.max() - best_overall + best_overall.mean()
            
            f0 = np.clip(f0, 0, None)
            f0 = normalize(f0)
            
            t_opt = min(25, deadline - time.monotonic() - 5)
            if t_opt < 10:
                break
            
            f_opt, sc = optimize_lp_improved(f0, t_opt, rng, verbose=False)
            if sc < best_score_overall:
                best_score_overall = sc
                best_overall = f_opt.copy()
                print(f"  Restart {r} (strategy {strategy}): NEW BEST = {sc:.6f}")
    
    # Final verification
    print(f"\n=== Final Result ===")
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = best_overall.tolist()
        final_score = evaluate_sequence(final_list)
        print(f"Final verified score: {final_score:.6f}")
        print(f"Sequence length: {len(final_list)}")
        print(f"Target: 1.5027, Previous: 1.515592")
        print(f"Improvement: {1.515592 - final_score:.6f}")
        return final_list
    
    # Fallback
    print("Fallback: uniform 256")
    return [1.0] * 256
