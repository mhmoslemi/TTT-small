import numpy as np
import time

def subgradient_optimize(a, deadline, lr_init=0.01, momentum=0.9):
    """Subgradient method with Nesterov acceleration."""
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    n = len(a)
    
    # Normalize
    a_sum = np.sum(a)
    if a_sum <= 0:
        return a, float('inf')
    a = a / a_sum
    
    best_a = a.copy()
    best_score = 2 * n * np.max(np.convolve(a, a))  # sum is 1
    
    v = np.zeros(n)  # momentum
    lr = lr_init
    no_improve = 0
    
    it = 0
    while time.monotonic() < deadline:
        conv_a = np.convolve(a, a)
        k_star = int(np.argmax(conv_a))
        max_val = conv_a[k_star]
        score = 2 * n * max_val  # since sum = 1
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute subgradient
        grad = np.zeros(n)
        j_min = max(0, k_star - (n - 1))
        j_max = min(k_star, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] = 2.0 * a[k_star - j]
        
        # Nesterov update
        v = momentum * v + lr * grad
        a_new = a - v
        a_new = np.clip(a_new, 0.0, None)
        a_new_sum = np.sum(a_new)
        if a_new_sum > 0:
            a_new = a_new / a_new_sum
            a = a_new
        else:
            break
        
        # Adaptive lr
        if no_improve > 200:
            lr *= 0.9
            no_improve = 0
            momentum = min(momentum + 0.01, 0.99)
        elif no_improve > 50:
            lr *= 0.95
        
        it += 1
        if it % 1000 == 0:
            print(f"  iter {it}: score={best_score:.6f}, lr={lr:.6f}, momentum={momentum:.3f}")
    
    return best_a, best_score / (np.sum(best_a)**2) * 2 * n  # need to be careful

OK so now the problem makes much more sense. We want to minimize the maximum of the linear autocorrelation, and the maximum can be at any lag.

For a symmetric sequence (a[j] = a[n-1-j]), the center lag n-1 gives sum_j a[j]^2, which is the "energy." For an asymmetric sequence, the center gives a "cross-term" that can be smaller.

This is a key insight! **Asymmetric sequences can have lower peak autocorrelation than symmetric ones!**

For example, a geometric sequence a[j] = r^j (j=0,...,n-1) with r close to 1:
(a*a)[k] = sum_j r^j * r^{k-j} = (number of valid j) * r^k
For k < n: (a*a)[k] = (k+1) * r^k
For k >= n: (a*a)[k] = (2n-1-k) * r^k

The max is at some k that balances the (k+1) factor and the r^k decay. If r is close to 1, the max is near k = n-1 with value approximately n*r^{n-1}.

- At each step, pick one coordinate i

- The objective as a function of a[i] (with a[i] >= 0) is:

For a fixed coordinate i, the contribution to R[k] is:

- Wait, d_k = a[k-i] >= 0, so all d_k >= 0. The max is increasing in a[i], so the minimum is at a[i] = 0?

No wait, that's not right either. If we set a[i] = 0, we might increase the max at other coordinates (because the sequence changes). But for the "one-coordinate-at-a-time" coordinate descent, we fix all other coordinates and optimize just a[i].

If we fix all other coordinates, then R[k] = c_k + a[i] * d_k where:
- c_k = R_without_i[k] (the contribution from all coordinates except i)
- d_k = a[k-i] if 0 <= k-i < n, else 0

Since all a >= 0, we have d_k >= 0 and c_k >= 0. The function max_k (c_k + a[i] * d_k) is:
- Non-decreasing in a[i] (since all d_k >= 0)
- Therefore, the minimum over a[i] >= 0 is at a[i] = 0!

But that's silly - it would set everything to 0. The issue is that we also have the constraint sum(a) = 1 (or we're minimizing the ratio, which is homogeneous).

OK so for the ratio objective: F(a) = 2n * max_k R[k] / (sum a)^2.

If we fix all coordinates except a[i], and let x = a[i], then:
- R[k] = c_k + x * d_k
- sum(a) = S + x where S = sum of all other coordinates

F(x) = 2n * max_k (c_k + x*d_k) / (S+x)^2

This is a 1D optimization problem in x >= 0! We can solve it by finding the minimum of this function.

The function max_k (c_k + x*d_k) is the upper envelope of lines, which is convex and piecewise linear. Dividing by (S+x)^2 (a convex decreasing function for x > 0... well, (S+x)^2 is convex increasing, so 1/(S+x)^2 is convex decreasing).

The ratio of a convex function to a convex function is not necessarily convex or concave. But we can find the minimum by:
1. Finding the "kinks" of the upper envelope (intersection points of the lines)
2. Evaluating F at each kink and at x = 0
3. Taking the minimum

The number of kinks is at most the number of lines = 2n-1, but typically much fewer (the number of "active" lags).

This is a very promising approach! Coordinate descent where each coordinate update is a 1D optimization that can be done analytically.

"""
Coordinate descent + subgradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses exact 1D coordinate minimization (exploiting the structure of linear autocorrelation)
combined with Nesterov-accelerated subgradient method for polishing.
Multi-start with diverse initializations across different sequence lengths.
"""
import numpy as np
import time


def build_d_modeled(i, n, a):
    """Build modeled coefficient array for coordinate i's contribution to R[k]."""
    d = np.zeros(2 * n - 1)
    lo = max(0, i)
    hi = min(2 * n - 2, i + n - 1)
    if lo <= hi:
        d[lo:hi + 1] = a[i - lo:i - lo + (hi - lo + 1)]
    return d


def coord_minimize(a, i, tol=1e-12):
    """Minimize F(x) over x>=0 for coordinate i, fixing others. Returns optimal x."""
    n = len(a)
    x_orig = a[i]
    a[i] = 0.0
    c = np.convolve(a, a)
    S = np.sum(a)
    d = build_d_modeled(i, n, a)
    # R[k] = c[k] + 2*x*d[k]; R[2i] also gets +x^2
    k_self = 2 * i
    dd = 2.0 * d
    dd[k_self] = 0.0  # the x^2 term handled separately
    # g(x) = max_k(c[k] + x*dd[k]), then R_max(x) = max(g(x), c[k_self] + x^2)
    # F(x) = 2n * R_max(x) / (S+x)^2

    # If S == 0, we need x > 0
    if S == 0.0:
        # R_max = max(g(x), c[k_self] + x^2), but c should be all zeros if S=0
        # F(x) = 2n * max(g(x), x^2) / x^2
        # If g(x) = 0 (all d=0), F = 2n. Otherwise...
        # Just return a small value
        return 1.0 / n

    # Find upper bound for x
    x_hi = max(x_orig, 1.0)
    F_0 = 2.0 * n * max(np.max(c + x_orig * dd), c[k_self] + x_orig ** 2) / (S + x_orig) ** 2
    # Expand until F decreases
    for _ in range(20):
        x_try = x_hi * 2.0
        g_val = np.max(c + x_try * dd)
        r_val = max(g_val, c[k_self] + x_try ** 2)
        F_try = 2.0 * n * r_val / (S + x_try) ** 2
        if F_try < F_0:
            x_hi = x_try
            F_0 = F_try
        else:
            break

    # Golden section search on [0, x_hi]
    if x_hi < 1e-15:
        return 0.0

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_lo = 0.0
    a_hi = x_hi

    def F_of_x(x):
        g_val = np.max(c + x * dd)
        r_val = max(g_val, c[k_self] + x * x)
        return 2.0 * n * r_val / (S + x) ** 2

    c_val = a_hi - gr * (a_hi - a_lo)
    d_val = a_lo + gr * (a_hi - a_lo)
    f_c = F_of_x(c_val)
    f_d = F_of_x(d_val)

    for _ in range(200):
        if a_hi - a_lo < tol * max(1.0, abs(c_val), abs(d_val)):
            break
        if f_c < f_d:
            a_hi = d_val
            d_val = c_val
            f_d = f_c
            c_val = a_hi - gr * (a_hi - a_lo)
            f_c = F_of_x(c_val)
        else:
            a_lo = c_val
            c_val = d_val
            f_c = f_d
            d_val = a_lo + gr * (a_hi - a_lo)
            f_d = F_of_x(d_val)

    x_opt = (a_lo + a_hi) / 2.0
    a[i] = x_orig  # restore
    return x_opt


def coord_descent(a, deadline, max_rounds=500, print_interval=50):
    """Cyclic coordinate descent. Returns (best_a, best_score)."""
    a = a.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    no_improve = 0
    rng = np.random.default_rng(99)
    idx_order = np.arange(n)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        # Shuffle order for diversity
        if rnd % 10 == 0:
            rng.shuffle(idx_order)

        improved = False
        for i in idx_order:
            if time.monotonic() > deadline:
                break
            x_new = coord_minimize(a, i)
            if abs(x_new - a[i]) > 1e-12:
                a[i] = x_new
                improved = True

        # Clip
        a = np.clip(a, 0.0, 1000.0)
        if np.sum(a) < 0.01:
            break

        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10:
                break

        if (rnd + 1) % print_interval == 0:
            print(f"    CD round {rnd+1}: score={best_score:.6f}")

    return best_a, best_score


def subgradient_optimize(a, deadline, lr_init=0.02, momentum_init=0.9):
    """Nesterov-accelerated subgradient method. Returns (best_a, best_score)."""
    a = a.copy().astype(np.float64)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return a, float('inf')
    a = a / S  # normalize to sum=1

    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    v = np.zeros(n)
    lr = lr_init
    momentum = momentum_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv_a = np.convolve(a, a)
        k_star = int(np.argmax(conv_a))
        max_val = conv_a[k_star]
        score = 2.0 * n * max_val  # sum=1 so sum^2=1

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Subgradient at k_star
        d = 2.0 * a  # d[k-i] = 2*a[k-i]
        # grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        if lo <= hi:
            grad[lo:hi + 1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]

        # Nesterov momentum update
        v = momentum * v + lr * grad
        a_new = a - v
        a_new = np.clip(a_new, 0.0, None)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a = a_new / S_new

        # Adaptive learning rate and momentum
        if no_improve > 300:
            lr *= 0.9
            momentum = min(momentum + 0.005, 0.995)
            no_improve = 0
        elif no_improve > 100:
            lr *= 0.95
        elif no_improve > 30:
            lr *= 0.98

        it += 1
        if it % 5000 == 0:
            print(f"    SG iter {it}: score={best_score:.6f}, lr={lr:.6f}, mom={momentum:.3f}")

    return best_a, best_score


def multiplicative_update(a, deadline, eta=0.1, max_iter=100000):
    """Multiplicative (proportional) update method. Returns (best_a, best_score)."""
    a = a.copy().astype(np.float64)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return a, float('inf')
    a = a / S

    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        conv_a = np.convolve(a, a)
        k_star = int(np.argmax(conv_a))
        max_val = conv_a[k_star]
        score = 2.0 * n * max_val

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                break

        # Multiplicative update: reduce a[i] proportional to its contribution to R[k_star]
        # Contribution of a[i] to R[k_star] is ~ 2*a[k_star-i] (if valid)
        a_new = a.copy()
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        if lo <= hi:
            # a_new[i] *= (1 - eta * a[k_star-i] / max_a)
            contributions = a[k_star - hi:k_star - lo + 1][::-1]
            max_contrib = np.max(contributions) if np.max(contributions) > 0 else 1.0
            a_new[lo:hi + 1] *= (1.0 - eta * contributions / max_contrib)

        a_new = np.clip(a_new, 0.0, None)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a = a_new / S_new
        it += 1

    return best_a, best_score


def propose_candidate(budget_s=1500, **kwargs):
    """
    Multi-strategy optimizer: coordinate descent (exact 1D), subgradient with
    Nesterov momentum, and multiplicative updates. Multi-start with diverse
    initializations across lengths 256-2048.
    """
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 15.0)

    best_score = float('inf')
    best_sequence = None

    def update_best(a, label=""):
        nonlocal best_score, best_sequence
        a = np.clip(a, 0.0, 1000.0).astype(np.float64)
        if np.sum(a) < 0.01:
            return
        s = evaluate_sequence(a.tolist())
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_sequence = a.copy()
            print(f"  NEW BEST [{label}]: n={len(a)}, score={s:.6f} (t={time.monotonic()-t0:.1f}s)")

    # Try parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)) and np.sum(parent) > 0.01:
            update_best(parent, "parent")
    except (NameError, Exception) as e:
        print(f"  No parent available: {e}")

    rng = np.random.default_rng(777)

    # Generate diverse initializations
    candidates = []

    # Lengths to try
    lengths = [256, 512, 1024, 2048]

    for n in lengths:
        idx = np.arange(n, dtype=np.float64)

        # Uniform
        candidates.append((f"uni_{n}", np.ones(n)))

        # Geometric (several ratios)
        for r in [0.8, 0.9, 0.95, 1.05, 1.1, 1.2]:
            seq = r ** idx
            candidates.append((f"geo_{n}_{r}", seq))

        # Cosine window
        candidates.append((f"cos_{n}", 0.5 + 0.5 * np.cos(np.pi * idx / (n - 1))))

        # Raised cosine (sin^2)
        candidates.append((f"sincos_{n}", np.sin(np.pi * idx / (2.0 * (n - 1))) ** 2 + 0.01))

        # Random
        candidates.append((f"rand_{n}", rng.uniform(0.1, 1.0, n)))

        # Bimodal (two peaks)
        center1, center2 = n * 0.25, n * 0.75
        sigma = n / 8.0
        seq = np.exp(-((idx - center1) / sigma) ** 2) + np.exp(-((idx - center2) / sigma) ** 2)
        candidates.append((f"bi_{n}", seq))

        # Linear ramp
        candidates.append((f"ramp_{n}", 0.1 + 0.9 * idx / (n - 1)))

        # Exponential decay
        candidates.append((f"expd_{n}", np.exp(-3.0 * idx / n)))

    print(f"Total candidates: {len(candidates)}")

    # Sort by length (shorter = faster per iteration)
    candidates.sort(key=lambda x: int(x[0].split('_')[1]) if len(x[0].split('_')) > 1 else 512)

    # Allocate time: give more time to candidates that initially score well
    # First pass: evaluate all quickly
    scored_candidates = []
    for name, a_init in candidates:
        a_init = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
        if np.sum(a_init) < 0.01:
            continue
        s = evaluate_sequence(a_init.tolist())
        if np.isfinite(s):
            scored_candidates.append((name, a_init, s))
        if time.monotonic() > deadline - 60:
            break

    # Sort by score (best first)
    scored_candidates.sort(key=lambda x: x[2])

    print("Top initializations:")
    for name, _, s in scored_candidates[:10]:
        print(f"  {name}: {s:.6f}")

    # Update best from initializations
    for name, a_init, s in scored_candidates:
        if s < best_score:
            best_score = s
            best_sequence = a_init.copy()
            print(f"  Updated from init [{name}]: score={s:.6f}")

    # Second pass: optimize top candidates
    # Allocate time proportional to inverse score (better candidates get more time)
    time_budget = deadline - time.monotonic()
    n_to_optimize = min(len(scored_candidates), 12)

    # Give the parent (if in list) and top scorers more time
    weights = []
    for name, a_init, s in scored_candidates[:n_to_optimize]:
        w = 1.0 / max(s, 1.0)
        if "parent" in name:
            w *= 2.0
        weights.append(w)
    total_w = sum(weights)

    for ci in range(n_to_optimize):
        if time.monotonic() > deadline:
            break
        name, a_init, s_init = scored_candidates[ci]
        time_alloc = max(10.0, time_budget * weights[ci] / total_w)
        sub_deadline = min(deadline, time.monotonic() + time_alloc)

        print(f"\n  Optimizing [{name}] (init={s_init:.6f}, time={time_alloc:.0f}s)")

        # Strategy 1: Coordinate descent
        if time.monotonic() < sub_deadline - 5:
            cd_deadline = min(sub_deadline, time.monotonic() + time_alloc * 0.4)
            a_cd, s_cd = coord_descent(a_init, cd_deadline)
            update_best(a_cd, f"cd_{name}")

            # Strategy 2: Subgradient on CD result
            if time.monotonic() < sub_deadline - 5:
                sg_deadline = min(sub_deadline, time.monotonic() + time_alloc * 0.35)
                a_sg, s_sg = subgradient_optimize(a_cd, sg_deadline)
                update_best(a_sg, f"sg_{name}")

                # Strategy 3: Multiplicative on SG result
                if time.monotonic() < sub_deadline:
                    a_mu, s_mu = multiplicative_update(a_sg, sub_deadline)
                    update_best(a_mu, f"mu_{name}")

            # Also try subgradient directly on init
            if time.monotonic() < sub_deadline - 5:
                sg2_deadline = min(sub_deadline, time.monotonic() + time_alloc * 0.2)
                a_sg2, s_sg2 = subgradient_optimize(a_init, sg2_deadline, lr_init=0.05, momentum_init=0.85)
                update_best(a_sg2, f"sg2_{name}")

    # Final: polish the overall best with all methods
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print(f"\n  Final polish on best (score={best_score:.6f}, n={len(best_sequence)})")

        # Multiple rounds of CD + SG + MU
        for round_idx in range(3):
            if time.monotonic() > deadline - 5:
                break
            sub_dl = min(deadline, time.monotonic() + (deadline - time.monotonic()) / (3 - round_idx))

            a1, s1 = coord_descent(best_sequence, sub_dl)
            update_best(a1, f"final_cd_{round_idx}")

            if time.monotonic() < deadline - 5:
                a2, s2 = subgradient_optimize(best_sequence, min(deadline, time.monotonic() + 60))
                update_best(a2, f"final_sg_{round_idx}")

            if time.monotonic() < deadline - 5:
                a3, s3 = multiplicative_update(best_sequence, min(deadline, time.monotonic() + 60), eta=0.05)
                update_best(a3, f"final_mu_{round_idx}")

    # One last attempt: try extending the best sequence
    if best_sequence is not None and time.monotonic() < deadline - 20:
        n_best = len(best_sequence)
        # Pad with small values
        for pad in [16, 32, 64]:
            if time.monotonic() > deadline - 10:
                break
            a_pad = np.concatenate([best_sequence, np.full(pad, best_sequence[-1] * 0.1)])
            a_pad = np.clip(a_pad, 0.0, 1000.0)
            update_best(a_pad, f"pad{pad}")

            if np.isfinite(evaluate_sequence(a_pad.tolist())):
                a_opt, s_opt = coord_descent(a_pad, min(deadline, time.monotonic() + 30))
                update_best(a_opt, f"pad{pad}_cd")

    # Ensure final answer is valid
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\n  FINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
