```python
import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def normalize_sequence(arr):
    """Normalize to non-negative, sum=1."""
    x = np.asarray(arr, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s <= 1e-300:
        return None
    return x / s

def score_of(p):
    """Score = 2*n*max(convolve(p,p)) for normalized p."""
    n = len(p)
    c = np.convolve(p, p)
    return 2.0 * n * c.max()

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def gradient_smoothed_max(p, lam):
    """Compute (c, grad) where c = convolve(p,p) and grad = grad of smoothed max."""
    c = np.convolve(p, p)
    c_max = c.max()
    w = np.exp(lam * (c - c_max))
    w_sum = w.sum()
    w /= w_sum
    # grad[m] = 2 * sum_j w[m+j] * p[j] = 2 * correlate(w, p, 'valid')[m]
    g = 2.0 * np.correlate(w, p, mode='valid')
    return c, g, c_max

                                  alpha0=1e-3, alpha_decay=1e-4, 
                                  max_iters=None, verbose=False):

    Optimize p to minimize max(convolve(p,p)) using projected gradient
    on the smoothed max objective.
    """
    p = p0.copy()
    n = len(p)
    best_p = p.copy()
    best_score = score_of(p)
    lam = lambda0
    alpha = alpha0
    stall = 0
    start_time = time.monotonic()
    deadline = start_time + time_budget
    iter_count = 0
    
    while time.monotonic() < deadline:
        iter_count += 1
        if max_iters and iter_count >= max_iters:
            break
        
        c, g, c_max = gradient_smoothed_max(p, lam)
```

The Frank-Wolfe method for minimizing f(p) over the simplex:

5. Otherwise: take an "away" step away from e_m

Looking at this problem, I need to minimize `2*n*max(convolve(p,p))/(sum(p))^2` for a non-negative sequence `p`. The previous approach used LP-based direction finding. I'll use a different strategy: **projected gradient descent on a smoothed max objective** with adaptive lambda, heavy-ball momentum, and multiple restarts at different sequence lengths. The key insight is that the gradient of the auto-convolution max can be computed efficiently via cross-correlation, making each iteration O(n) with vectorized numpy operations.

```python
"""
Projected gradient optimization of non-negative step functions minimizing
2*n*max(auto-convolve(p,p))/(sum(p))^2 using smoothed max (log-sum-exp),
adaptive lambda, heavy-ball momentum, backtracking line search, and
multi-length multi-start strategy with periodic LP refinement.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Normalize to non-negative vector with sum 1, or return None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    return x / s


def score_of(p):
    """Score = 2*n*max(convolve(p,p)) assuming sum(p)=1, p>=0."""
    n = p.size
    c = np.convolve(p, p)
    return 2.0 * float(n) * float(c.max())


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_gradient_smoothed(p, lam):
    """
    Compute the smoothed max objective gradient.
    Returns (conv, grad, max_val, smooth_val).
    """
    c = np.convolve(p, p)
    c_max = float(c.max())
    # Softmax weights for numerical stability
    w = np.exp(lam * (c - c_max))
    w_sum = float(w.sum())
    w /= w_sum
    smooth_val = (1.0 / lam) * (log_sum_exp_approx(c, lam) )
    # Gradient: g[m] = 2 * sum_j w[m+j] * p[j] for m=0..n-1
    # This is 2 * correlate(w, p, mode='valid')
    g = 2.0 * np.correlate(w, p, mode='valid')
    return c, g, c_max, smooth_val, w


def log_sum_exp_approx(c, lam):
    """Compute log(sum(exp(lam*c))) with numerical stability."""
    c_max = float(c.max())
    return c_max * lam + float(np.log(np.exp(lam * (c - c_max)).sum()))


def upsample_sequence(p, n):
    """Upsample non-negative sequence to length n via linear interpolation, normalized."""
    m = p.size
    if m == n:
        return p.copy()
    if m == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    old_t = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new_t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new_t, old_t, p)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return out / s


def downsample_sequence(p, n):
    """Downsample non-negative sequence to length n via interpolation, normalized."""
    m = p.size
    if m == n:
        return p.copy()
    if m == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    if n == 1:
        return np.array([1.0])
    old_t = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new_t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new_t, old_t, p)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return out / s


def optimize_projected_gradient(p0, time_budget, lam_start=5.0, lam_end=500.0,
                                 alpha_start=1e-3, momentum=0.9,
                                 verbose=False, verbose_interval=500):

    Main optimization: projected gradient on smoothed max with momentum.
    Returns (best_p, best_score).
    """

    p = p0.copy()

    s = p.sum()
    if s < 1e-300:
        p = np.ones(n) / n
    else:
        p /= s

    c, _, c_max, _, _ = compute_gradient_smoothed(p, lam_start)
    best_score = 2.0 * n * c_max
    best_p = p.copy()

    lam = lam_start
    alpha = alpha_start
    v = np.zeros(n, dtype=np.float64)  # momentum buffer
    stall = 0
    iter_count = 0

    start_time = time.monotonic()
    deadline = start_time + time_budget

    # Log-sum-exp for lambda adaptation
    lam_log_start = np.log(lam_start)
    lam_log_end = np.log(lam_end)

    while time.monotonic() < deadline:
        iter_count += 1

        if time.monotonic() > deadline:
            break

        # Compute gradient
        c, g, c_max, smooth_val, w = compute_gradient_smoothed(p, lam)
        current_score = 2.0 * n * c_max

        # Update best
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_p = p.copy()
            stall = 0
        else:
            stall += 1

                  f"best={best_score:.6f}, lam={lam:.1f}, alpha={alpha:.2e}, "

            p = project_simplex(p_noisy)
            v[:] = 0.0
            alpha = alpha_start
            lam = lam_start
            stall = 0
            continue

            v[:] = 0.0
            alpha = alpha_start * 2.0
            lam = lam_start * 2.0
            stall = 100
            continue

        # Heavy ball momentum
        v = momentum * v + (1.0 - momentum) * g
        direction = -v

        # Backtracking line search
        improved = False
        a = alpha

            a *= 0.5
            p_trial = project_simplex(p + a * direction)
            c_trial = np.convolve(p_trial, p_trial)

                alpha = min(a / 0.5, alpha_start * 10.0)  # don't grow too fast
                improved = True

            # Also try with original alpha

                c_trial = np.convolve(p_trial, p_trial)
                score_trial = 2.0 * n * float(c_trial.max())
                if score_trial < current_score - 1e-12 * max(1.0, current_score):
                    p = p_trial
                    improved = True
                    break

        if not improved:
            alpha *= 0.5
            if alpha < 1e-12:

                alpha = alpha_start
                v[:] = 0.0
                # Small random perturbation
                p = project_simplex(p + 0.001 * np.random.randn(n))

        # Use cosine schedule over the total iterations

def make_conv_template_lp(n):
    """Precompute sparse matrix structure for LP direction finding."""
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int64)
    je = np.minimum(n - 1, k64).astype(np.int64)
    counts = (je - js + 1)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts)
    col_base = js[row]
    col = (col_base + local)
    offsets = (row - col).astype(np.int32)

    A_off = csr_matrix((offsets, (row.astype(np.int32), col.astype(np.int32))),
                       shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def lp_direction_step(p, template, max_time=5.0):
    """
    Try one LP direction step. Returns improved p or None.
    """
    n = p.size
    A_template, edge_offsets, c_obj, bounds, m = template
    t_start = time.monotonic()

    c = np.convolve(p, p)
    M = float(c.max())
    if M <= 0:
        return None

    A = A_template.copy()
    A.data[:] = p[edge_offsets]
    A.data *= float(n)

    b = np.full(m, M * float(n) * float(n), dtype=np.float64)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs",
                      options={"time_limit": max_time, "presolve": True})
    except Exception:
        return None

    if time.monotonic() - t_start > max_time:
        return None

    x = getattr(res, "x", None)
    if x is None:
        return None
    x = np.asarray(x, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s_star = float(x.sum())
    if s_star <= 1e-300:
        return None

    g = x / s_star
    # Check if mixing helps
    best_mix = None
    best_mix_score = score_of(p)
    for t in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
        p_mix = (1.0 - t) * p + t * g
        p_mix = np.maximum(p_mix, 0.0)
        sm = p_mix.sum()
        if sm < 1e-300:
            continue
        p_mix /= sm
        sc = score_of(p_mix)
        if sc < best_mix_score:
            best_mix_score = sc
            best_mix = p_mix

    if best_mix is not None:
        return best_mix
    return None


def optimize_combined(p0, time_budget, use_lp=True):
    """
    Combined optimization: projected gradient with periodic LP steps.
    """
    n = p0.size
    alloc_gp = time_budget * 0.85
    alloc_lp = time_budget * 0.15 if use_lp else 0.0

    # Phase 1: Projected gradient
    p, s = optimize_projected_gradient(p0, alloc_gp,
                                        lam_start=5.0, lam_end=200.0,
                                        alpha_start=1e-3, momentum=0.9,
                                        verbose=True, verbose_interval=1000)

    # Phase 2: LP refinement if time remains
    if use_lp and alloc_lp > 3.0:
        try:
            template = make_conv_template_lp(n)
            rem = time_budget - (time.monotonic() - (time.monotonic() - alloc_lp - alloc_gp))
            # Use remaining time for LP steps
            lp_remaining = time_budget - alloc_gp
            if lp_remaining > 5.0:
                for _ in range(5):
                    if time.monotonic() > time_budget:
                        break
                    p_new = lp_direction_step(p, template, max_time=3.0)
                    if p_new is not None:
                        s_new = score_of(p_new)
                        if s_new < s:
                            p = p_new
                            s = s_new
                            print(f"        LP step improved: {s:.6f}", flush=True)
                    # After LP, do a few more gradient steps
                    rem_time = time_budget - time.monotonic()
                    if rem_time > 2.0:
                        p, s = optimize_projected_gradient(p, min(rem_time, alloc_lp / 5),
                                                            lam_start=50.0, lam_end=500.0,
                                                            alpha_start=5e-4, momentum=0.9,
                                                            verbose=False, verbose_interval=9999999)
            del template
        except Exception as e:
            print(f"        LP phase error: {e}", flush=True)

    return p, s
```
