def build_conv_matrix_sparse(f):

    cols = []

            cols.append(j)

    b_eq = np.zeros(1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
    
    if result.success:
        da = result.x[:n]
        return da
    return None


def lp_cross_corr_step(f, M, top_k=50):
    """
    Alternative LP: find g >= 0 maximizing sum(g) such that conv(f,g)_k <= M for top-k k's.
    Then blend f toward g.
    """
    n = len(f)
    b = np.convolve(f, f)
    top_indices = np.argsort(b)[::-1][:top_k]
    
    # Build sparse matrix for only top-k constraints
    A = build_conv_matrix_sparse(f)
    A_sub = A[top_indices, :]
    
    c = -np.ones(n)
    b_ub = np.full(len(top_indices), M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success and result.x is not None:
        g = result.x
        sum_g = np.sum(g)
        if sum_g > 1e-15:
            g = g * (np.sum(f) / sum_g)
            return g
    return None


def proximal_gradient_step(f, lr, momentum=0.9):
    """
    One step of proximal gradient with momentum.
    Uses subgradient of max_k b[k].
    """
    n = len(f)
    b = np.convolve(f, f)
    max_val = np.max(b)
    
    # Find active peaks (within tolerance of max)
    tol = 0.01 * max_val
    active = np.where(b >= max_val - tol)[0]
    
    # Subgradient: average of gradients of active peaks
    grad = np.zeros(n)
    for k in active:
        g = compute_gradient_for_lag(f, k)
        grad += g
    grad /= len(active)
    
    return grad


def freq_domain_equalize(f, alpha=0.1):
    """
    Frequency-domain equalization: modify the DFT to flatten the autocorrelation.
    """
    n = len(f)
    # Use 2n-length DFT to handle linear convolution
    N = 2 * n
    F = np.fft.rfft(f, N)
    
    # Current autocorrelation (linear)
    b = np.convolve(f, f)
    b_padded = np.zeros(N)
    b_padded[:len(b)] = b
    B = np.fft.rfft(b_padded)
    
    # Target: flat autocorrelation (all b[k] equal)
    target_mean = np.sum(b) / N
    B_target = B.copy()
    # Reduce peaks in frequency domain
    # Where |B| is large (corresponding to time-domain peaks), reduce it
    B_new = F * F.conj()  # = |F|^2, the power spectrum
    # Actually, B = |F|^2 in the frequency domain (circular autocorrelation)
    # For linear autocorrelation, it's approximately |F|^2 for the DFT of length 2n
    
    # Equalization: reduce the peak of |F|^2
    power = np.abs(F) ** 2
    max_power = np.max(power)
    if max_power > 0:
        # Soft clip: reduce peaks
        threshold = 0.8 * max_power
        clip_mask = power > threshold
        reduction = alpha * (power[clip_mask] - threshold)
        phase = F / np.abs(F + 1e-30)
        F_new = F.copy()
        F_new[clip_mask] = phase[clip_mask] * np.sqrt(np.maximum(power[clip_mask] - reduction, 0))
    else:
        F_new = F
    
    f_new = np.fft.irfft(F_new, N)[:n]
    f_new = np.clip(f_new, 0.0, None)
    return f_new


def simulated_annealing(f, deadline, iterations=5000):
    """
    Simulated annealing: randomly perturb positions and accept if objective improves
    or with probability exp(-delta/T).
    """
    n = len(f)
    f_current = f.copy()
    f_best = f.copy()
    score_best = objective_value(f_best)
    score_current = score_best
    
    T = 0.01  # Initial temperature
    T_min = 1e-6
    cooling = 0.999
    
    for it in range(iterations):
        if time.monotonic() > deadline:
            break
        
        # Random perturbation: pick a few positions and adjust
        n_perturb = max(1, n // 64)
        positions = np.random.randint(0, n, size=n_perturb)
        
        f_trial = f_current.copy()
        for pos in positions:
            # Random adjustment
            delta = np.random.normal(0, 0.1 * max(f_current[pos], 0.01))
            f_trial[pos] = max(0, f_current[pos] + delta)
        
        # Renormalize
        s = np.sum(f_trial)
        if s < 1e-15:
            continue
        f_trial = f_trial / s
        
        score_trial = objective_value(f_trial)
        
        if score_trial < score_current:
            f_current = f_trial
            score_current = score_trial
            if score_trial < score_best:
                f_best = f_trial.copy()
                score_best = score_trial
        elif np.random.random() < np.exp(-(score_trial - score_current) / max(T, T_min)):
            f_current = f_trial
            score_current = score_trial
        
        T *= cooling
    
    return f_best, score_best
