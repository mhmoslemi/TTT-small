```python
    if len(active) > max_constraints and max_constraints is not None:
        # Keep the most active constraints
        idx = np.argsort(conv_ff[active])[-max_constraints:]
        active = active[idx]
    
    if len(active) < 3:
        active = np.arange(m)
    
    # Build constraint matrix: A[k, i] = f[active[k] - i] if valid
    # (f*g)[active[k]] = sum_i f[active[k]-i] * g[i] for valid i
    # So A[active_idx, i] = f[active[active_idx] - i] if 0 <= active[active_idx] - i < n else 0
    
    K = len(active)
    A = np.zeros((K, n), dtype=np.float64)
    for ki, k in enumerate(active):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            A[ki, i] = f[k - i]
    
    b = np.full(K, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return res
```

This dense approach is simpler and for the number of active constraints (usually < 200), it's fast enough.

Actually, for n=1024, a 200x1024 dense matrix is about 1.6 MB, and solving an LP with 200 constraints and 1024 variables should be very fast (milliseconds to a few tens of ms with HiGHS).

**Algorithm:**
1. Start with `height_sequence_1` (given) and upsample to target length
2. Run LP-based iterations:
   a. Compute M = max(f*f)
   b. Find active constraints
   c. Solve LP to get direction g
   d. Compute safe step size
   e. Update: f_new = (1-t)*f + t*g_normalized
   f. Repeat until convergence or time budget
3. After LP convergence, do gradient polishing:
   a. Compute gradient of score w.r.t. f (projected onto sum=1, f>=0)
   b. Take small steps in negative gradient direction
4. Try multiple starting points and keep the best

**Gradient computation:**
score = 2*n*max_k b[k] / (sum(f))^2

If we fix sum(f) = 1, then score = 2*n*max_k b[k].
Let k* = argmax b. Then:
d(score)/d(f_i) = 2*n * db[k*]/df_i = 2*n * (f[k*-i] + f[i-k*]) [with zero-padding]

Wait, more carefully: b[k] = sum_j f[j] * f[k-j] (with f[j] = 0 for j<0 or j>=n)
db[k]/df[i] = f[k-i] + f[i-k] (with zero-padding)

So the gradient (before projecting) is:
grad[i] = 2*n * (f[k*-i] + f[i-k*]) for i = 0,...,n-1

To project onto the constraint sum(f) = 1:
grad_proj[i] = grad[i] - mean(grad)

And for the non-negativity constraint, we only move in the negative gradient direction for positive components (or use a projection).

Actually, for a proper line search approach:
- Direction: d = -grad_proj (projected gradient descent direction)
- But we need to maintain f >= 0, so we can use: d[i] = -grad_proj[i] if f[i] > 0, else 0 (rough projection)
- Or better: use the Frank-Wolfe / mirror descent approach

```python
"""
LP-based iterative search with gradient polishing for minimizing
2*n*max(convolve(f,f))/(sum(f))^2 over non-negative sequences.
Uses active-set LP direction finding, aggressive step sizes,
gradient-based refinement, and diverse multi-scale exploration.
"""

import time
import numpy as np
from scipy.optimize import linprog


def normalize_sequence(arr):
    x = np.asarray(arr, dtype=np.float64).ravel()
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    if n <= 0:
        return float('inf')
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * n * m)


def mix_noisy(base, rng, eps):
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    if eps <= 1e-12:
        return f.copy()
    try:
        mix = rng.dirichlet(np.ones(n, dtype=np.float64))
    except Exception:
        mix = np.ones(n, dtype=np.float64) / float(n)
    if eps >= 1.0:
        f = mix
    else:
        f = (1.0 - eps) * f + eps * mix
    return normalize_sequence(f)


def interpolate_sequence(x, n):
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def make_bimodal(n, alpha, rng):
    t = np.linspace(0, 1, n)
    x = np.sin(np.pi * t) ** 2 * (1 - alpha) + alpha * (t**4 + (1 - t)**4) / (2 * n)
    x += np.abs(rng.normal(0, 0.02, n))
    return normalize_sequence(x)


def make_u_shaped(n, alpha, rng):
    t = np.linspace(0, 1, n)
    x = 1.0 + alpha * (2 * t - 1) ** 2
    x += np.abs(rng.normal(0, 0.01, n))
    return normalize_sequence(x)


def gradient_step(f, n, step_size):
    """One projected gradient step to reduce max convolution."""
    conv = np.convolve(f, f)
    k_star = int(np.argmax(conv))
    # Gradient of b[k_star] w.r.t. f[i] = f[k_star - i] + f[i - k_star]
    grad = np.zeros(n, dtype=np.float64)
    for i in range(n):
        v1 = f[k_star - i] if 0 <= k_star - i < n else 0.0
        v2 = f[i - k_star] if 0 <= i - k_star < n else 0.0
        grad[i] = v1 + v2
    # Project onto sum=1 constraint
    grad -= np.mean(grad)
    # Step
    f_new = f - step_size * grad
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s <= 1e-300:
        return f.copy()
    f_new /= s
    return f_new


def solve_lp_direction(f, n, M, max_constr=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    m = 2 * n - 1
    conv_ff = np.convolve(f, f)
    
    # Use all constraints for correctness, but cap if too many
    if max_constr is not None and m > max_constr:
        # Keep the most active constraints
        thr = np.sort(conv_ff)[-max_constr]
        active = np.where(conv_ff >= thr)[0]
    else:
        active = np.arange(m)
    
    K = len(active)
    A = np.zeros((K, n), dtype=np.float64)
    for ki, k in enumerate(active):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            A[ki, i] = f[k - i]
    
    b = np.full(K, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None
    return res


def optimize_lp(n, f_init, time_budget, rng, all_constraints=True):
    """Main LP-based optimizer with gradient polishing."""
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n, dtype=np.float64) / float(n)
    
    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_score = best_score
    deadline = time.monotonic() + max(0.5, float(time_budget))
    stall = 0
    iteration = 0
    max_c = 2 * n - 1 if all_constraints else min(2 * n - 1, 500)
    
    while time.monotonic() < deadline:
        iteration += 1
        t0 = time.monotonic()
        
        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        if not np.isfinite(M) or M <= 0:
            break
        
        res = solve_lp_direction(f, n, M, max_constr=max_c if not all_constraints else None)
        
        if res is None or getattr(res, 'x', None) is None:
            stall += 2
            if stall > 15:
                break
            f = mix_noisy(best_f, rng, 0.3)
            continue
        
        x = np.asarray(res.x, dtype=np.float64).ravel()
        x = np.where(np.isfinite(x), x, 0.0)
        x = np.maximum(x, 0.0)
        s_star = float(x.sum())
        
        if s_star <= 1e-300:
            stall += 1
            if stall > 15:
                break
            f = mix_noisy(best_f, rng, 0.3)
            continue
        
        g = x / s_star
        r = np.convolve(f, g)
        r_max = float(np.max(r))
        
        if not np.isfinite(r_max) or r_max <= 0:
            stall += 2
            if stall > 15:
                break
            f = mix_noisy(best_f, rng, 0.3)
            continue
        
        if r_max >= M * (1.0 - 1e-10):
            # LP didn't find improvement; try gradient step
            for _ in range(3):
                f_grad = gradient_step(f, n, 0.01)
                sc = score_normalized(f_grad, n)
                if sc < best_score:
                    f = f_grad
                    best_f = f.copy()
                    best_score = sc
                    break
                else:
                    break
            stall += 1
            if stall > 15:
                # Try random restart
                f = mix_noisy(best_f, rng, 0.4)
                if stall > 25:
                    break
            continue
        
        G = float(np.max(np.convolve(g, g)))
        if not np.isfinite(G):
            stall += 2
            if stall > 15:
                break
            continue
        
        # Safe step size
        a_coef = 2.0 * (r_max - M)
        b_coef = G + M - 2.0 * r_max
        if b_coef > 1e-14:
            t_max = -a_coef / b_coef
        else:
            t_max = 1.0
        if not np.isfinite(t_max) or t_max <= 0:
            t_max = 1e-8
        t_max = min(t_max, 1.0)
        
        # Try multiple step sizes
        ts = []
        for frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            tv = frac * t_max
            if 0 < tv <= 1.0 and tv not in ts:
                ts.append(tv)
        if not ts:
            ts = [min(t_max, 1e-3), 1e-5]
        
        best_cand_f = None
        best_cand_score = best_score
        best_t = None
        
        for tv in ts:
            f_new = (1.0 - tv) * f + tv * g
            f_new = np.maximum(f_new, 0.0)
            ssum = f_new.sum()
            if ssum <= 1e-300:
                continue
            f_new /= ssum
            sc = score_normalized(f_new, n)
            if np.isfinite(sc) and sc < best_cand_score:
                best_cand_score = sc
                best_cand_f = f_new
                best_t = tv
        
        if best_cand_f is not None:
            f = best_cand_f
            best_f = f.copy()
            best_score = best_cand_score
            stall = 0
            if iteration <= 3 or iteration % 20 == 0:
                print(f"    n={n} iter={iteration} score={best_score:.6f} t={best_t:.4f} "
                      f"elapsed={time.monotonic()-t0:.1f}s", flush=True)
        else:
            stall += 1
            # Gradient polish attempt
            for gs in [0.05, 0.02, 0.005]:
                f_g = gradient_step(best_f, n, gs)
                sc_g = score_normalized(f_g, n)
                if sc_g < best_score * (1 - 1e-6):
                    f = f_g
                    best_f = f.copy()
                    best_score = sc_g
                    stall = 0
                    break
            if stall > 20:
                f = mix_noisy(best_f, rng, 0.35)
                if stall > 35:
                    break
    
    # Final gradient polishing
    for _ in range(50):
        if time.monotonic() > deadline:
            break
        for gs in [0.01, 0.005, 0.001]:
            f_g = gradient_step(best_f, n, gs)
            sc_g = score_normalized(f_g, n)
            if sc_g < best_score * (1 - 1e-7):
                best_f = f_g
                best_score = sc_g
                break
    
    return best_f, float(best_score)
```
