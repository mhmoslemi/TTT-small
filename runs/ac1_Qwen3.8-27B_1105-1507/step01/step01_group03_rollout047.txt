        return np.ones(n, dtype=np.float64) / float(n)
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def mix_noisy(base, rng, eps):
    """Mix a base vector with a random Dirichlet vector for perturbation."""
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    if eps <= 1e-12:
        return f
    try:
        mix = rng.dirichlet(np.ones(n, dtype=np.float64))
    except Exception:
        mix = np.ones(n, dtype=np.float64) / float(n)
    if eps >= 1.0:
        f = mix
    else:
        f = (1.0 - eps) * f + eps * mix
    return normalize_sequence(f)


def optimize_combined(n, f_init, time_budget, rng, template):
    """
    Run combined LP + gradient optimization from one initial vector.
    Returns (best_f, best_score).
    """
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n, dtype=np.float64) / float(n)

    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_score = best_score
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(time_budget))
    stall_count = 0
    iteration = 0
    consecutive_no_improve = 0

    while time.monotonic() < deadline:
        iteration += 1
        t0 = time.monotonic()
        rem = deadline - time.monotonic()
        if rem < 0.5:
            break

        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        if not np.isfinite(M) or M <= 0.0:
            break

        improved = False

        # --- LP Direction ---
        if template is not None and M > 0:
            try:
                res = solve_direction_lp(template, f, n, M)
                x = getattr(res, "x", None)
                if x is not None:
                    x = np.asarray(x, dtype=np.float64).ravel()
                    if x.size == n:
                        x = np.where(np.isfinite(x), x, 0.0)
                        x = np.maximum(x, 0.0)
                        s_star = float(x.sum())
                        if s_star > 1e-300 and s_star / n > 1.0 + 1e-8:
                            g = x / s_star
                            r = np.convolve(f, g)
                            r_max = float(np.max(r))
                            if (np.isfinite(r_max) and 0 < r_max < M * (1.0 - 1e-10)):
                                S_eff = M / r_max
                                if S_eff > 1.0 + 1e-8:
                                    G = float(np.max(np.convolve(g, g)))
                                    if np.isfinite(G) and G >= 0:
                                        a_coef = 2.0 * (r_max - M)
                                        b_coef = G + M - 2.0 * r_max
                                        if b_coef > 1e-14:
                                            t_max = -a_coef / b_coef
                                        else:
                                            t_max = 1.0
                                        if (not np.isfinite(t_max)) or t_max <= 0.0:
                                            t_max = 1e-8
                                        base_t = min(1.0, 0.75 * t_max)
                                        if base_t < 1e-12:
                                            base_t = 1e-12

                                        ts = sorted(set([
                                            base_t, 0.75 * base_t, 0.5 * base_t,
                                            0.25 * base_t, 0.1 * base_t,
                                            0.05 * base_t, 1e-4, 1e-5, 1e-6
                                        ]), reverse=True)

                                        for tv in ts[:8]:
                                            if tv > 1.0:
                                                tv = 1.0
                                            if tv < 1e-14:
                                                continue
                                            f_new = (1.0 - tv) * f + tv * g
                                            f_new = np.maximum(f_new, 0.0)
                                            ssum = float(f_new.sum())
                                            if ssum <= 1e-300:
                                                continue
                                            f_new /= ssum
                                            sc = score_normalized(f_new, n)
                                            if np.isfinite(sc) and sc < best_score - 1e-12:
                                                best_score = sc
                                                best_f = f_new.copy()
                                                f = f_new
                                                improved = True
                                                break
            except Exception:
                pass

        # --- Gradient Step (if LP didn't improve) ---
        if not improved:
            try:
                grad, _ = gradient_direction(f, n)
                gn = np.linalg.norm(grad)
                if gn > 1e-30:
                    grad /= gn
                    # Adaptive learning rates
                    scale = 1.0 / n
                    lrs = [3.0 * scale, scale, 0.5 * scale, 0.2 * scale,
                           0.1 * scale, 0.05 * scale, 1e-4, 5e-5, 1e-5]
                    for lr in lrs:
                        f_new = f - lr * grad
                        f_new = np.maximum(f_new, 0.0)
                        ssum = float(f_new.sum())
                        if ssum <= 1e-300:
                            continue
                        f_new /= ssum
                        sc = score_normalized(f_new, n)
                        if np.isfinite(sc) and sc < best_score - 1e-12:
                            best_score = sc
                            best_f = f_new.copy()
                            f = f_new
                            improved = True
                            break
            except Exception:
                pass

        # --- Coordinate perturbation (last resort) ---
        if not improved:
            # Randomly perturb a few coordinates
            n_perturb = max(1, n // 50)
            f_new = f.copy()
            idx = rng.choice(n, size=n_perturb, replace=False)
            for idx_i in idx:
                f_new[idx_i] = max(0.0, f_new[idx_i] * rng.uniform(0.5, 1.5))
            ssum = float(f_new.sum())
            if ssum > 1e-300:
                f_new /= ssum
                sc = score_normalized(f_new, n)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best_score = sc
                    best_f = f_new.copy()
                    f = f_new
                    improved = True

        if improved:
            stall_count = 0
            consecutive_no_improve = 0
        else:
            stall_count += 1
            consecutive_no_improve += 1

            # Restart strategy after stagnation
            if consecutive_no_improve >= 8:
                if best_score < start_score - 0.01 and consecutive_no_improve >= 20:
                    # Try moderate restart from best
                    f = mix_noisy(best_f, rng, 0.05)
                    if f is None:
                        f = best_f.copy()
                    consecutive_no_improve = 0
                elif consecutive_no_improve >= 15:
                    # Larger restart
                    eps = min(0.4, 0.1 * (consecutive_no_improve - 8) / 10.0)
                    f = mix_noisy(best_f, rng, eps)
                    if f is None:
                        f = best_f.copy()
                    consecutive_no_improve = 0
            if stall_count > 30:
                break

        # Periodic logging
        if iteration % 20 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    n={n} iter={iteration} score={best_score:.6f} "
                  f"M={M:.6g} elapsed={elapsed:.1f}s rem={rem:.1f}s", flush=True)

    return best_f, float(best_score)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a low-scoring nonnegative sequence of length ~1024.
    Combined LP + gradient method with multiple diverse initializations.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    total_budget = max(60.0, float(budget_s))
    start_time = time.monotonic()
    reserve = 20.0
    deadline = start_time + max(1.0, total_budget - reserve)

    best_f = None
    best_score = float("inf")
    given_f = None

    # Load given sequence if available
    if total_budget >= 30.0:
        try:
            raw = np.asarray(height_sequence_1, dtype=np.float64)
            given_f = normalize_sequence(raw)
            if given_f is not None:
                best_f = given_f.copy()
                best_score = score_normalized(given_f, given_f.size)
                print(f"Given sequence loaded: n={given_f.size}, score={best_score:.6f}", flush=True)
        except Exception:
            given_f = None

    if best_f is None:
        best_f = np.ones(64, dtype=np.float64) / 64.0
        best_score = 2.0

    # Define length plan with time allocations
    # Focus heavily on 1024 since that's where our best is
    plan = [
        (128, 15.0),
        (256, 30.0),
        (512, 80.0),
        (1024, 1200.0),
    ]

    templates = {}

    for n, alloc in plan:
        rem = deadline - time.monotonic()
        if rem < 10.0:
            break
        alloc = min(alloc, rem - 5.0)
        if alloc <= 5.0:
            continue

        print(f"\n== n={n}, alloc={alloc:.1f}s, current_best={best_score:.6f}", flush=True)

        # Build template
        try:
            templates[n] = make_conv_template(n)
        except (MemoryError, Exception) as e:
            print(f"  template failed for n={n}: {e}", flush=True)
            continue

        template = templates.get(n)

        # Generate diverse starting points
        starts = []

        # 1. Upsampled best (if different length)
        if best_f is not None and best_f.size != n:
            up = interpolate_sequence(best_f, n)
            if up is not None:
                starts.append(up)
        elif best_f is not None and best_f.size == n:
            starts.append(best_f.copy())

        # 2. Given sequence interpolated
        if given_f is not None and given_f.size != n:
            g_up = interpolate_sequence(given_f, n)
            if g_up is not None:
                starts.append(g_up)

        # 3. Geometric sequences (asymmetric - key insight!)
        for r in [0.95, 0.9, 0.85, 0.8, 0.7]:
            starts.append(geometric_init(n, r))

        # 4. Reversed geometric
        for r in [0.9, 0.85]:
            starts.append(geometric_init(n, r)[::-1].copy())

        # 5. Sine shapes
        starts.append(sine_init(n, power=1.0, center=0.3))
        starts.append(sine_init(n, power=1.5, center=0.4))

        # 6. Ramp sequences
        starts.append(ramp_init(n, p=0.5, direction=1))
        starts.append(ramp_init(n, p=2.0, direction=-1))

        # 7. Bell shapes
        starts.append(bell_init(n, width=0.3, center_ratio=0.3))
        starts.append(bell_init(n, width=0.5, center_ratio=0.5))

        # 8. Uniform with noise
        starts.append(mix_noisy(np.ones(n), rng, 0.3))

        # 9. Dirichlet samples
        starts.append(rng.dirichlet(np.ones(n) * 0.5).copy())
        starts.append(rng.dirichlet(np.ones(n) * 0.2).copy())

        # 10. Noisy version of best
        if best_f is not None:
            for eps in [0.05, 0.15, 0.3]:
                noised = mix_noisy(interpolate_sequence(best_f, n), rng, eps)
                if noised is not None:
                    starts.append(noised)

        # Deduplicate and limit
        unique_starts = []
        seen = []
        for s in starts:
            s = normalize_sequence(s)
            if s is None or s.size != n:
                continue
            is_dup = False
            for prev in seen:
                if np.linalg.norm(s - prev) < 1e-4:
                    is_dup = True
                    break
            if not is_dup:
                unique_starts.append(s)
                seen.append(s)

        # Prioritize: put most promising starts first
        # Score each start and sort
        scored_starts = []
        for s in unique_starts:
            sc = score_normalized(s, n)
            scored_starts.append((sc, s))
        scored_starts.sort(key=lambda x: x[0])

        # Take top starts, up to max 12
        max_starts = min(len(scored_starts), 12)
        final_starts = [s for _, s in scored_starts[:max_starts]]

        # Allocate time: give more time to better starts
        # Use a weighted scheme
        n_starts = len(final_starts)
        if n_starts <= 1:
            weights = [1.0]
        elif n_starts <= 4:
            weights = [0.35, 0.30, 0.20, 0.15]
            weights = weights[:n_starts]
            w_sum = sum(weights)
            weights = [w / w_sum for w in weights]
        elif n_starts <= 8:
            weights = [0.25, 0.18, 0.14, 0.12, 0.10, 0.08, 0.07, 0.06]
            weights = weights[:n_starts]
            w_sum = sum(weights)
            weights = [w / w_sum for w in weights]
        else:
            # Equal-ish with decay
            weights = [0.15 * (0.85 ** i) for i in range(n_starts)]
            w_sum = sum(weights)
            weights = [w / w_sum for w in weights]

        for idx, init in enumerate(final_starts):
            rem = deadline - time.monotonic()
            if rem < 3.0:
                break
            tb = alloc * weights[idx]
            tb = min(tb, rem - 2.0)
            if tb < 2.0:
                continue

            init_score = score_normalized(init, n)
            print(f"  start {idx+1}/{n_starts}: init_score={init_score:.6f}, budget={tb:.1f}s", flush=True)

            f_res, s_res = optimize_combined(n, init, tb, rng, template)

            if np.isfinite(s_res) and s_res < best_score:
                best_score = float(s_res)
                best_f = np.asarray(f_res, dtype=np.float64).copy()
                print(f"  NEW GLOBAL BEST: n={best_f.size}, score={best_score:.6f}", flush=True)

            # Check time
            if time.monotonic() >= deadline - 5.0:
                break

        # Clean up template for memory
        if n in templates and template is not None and n < 1024:
            del templates[n]

    # Final: try a focused optimization at n=1024 from the best found
    if best_f is not None and best_f.size != 1024:
        rem = deadline - time.monotonic()
        if rem > 30.0:
            print(f"\n== Final upscaling to n=1024, rem={rem:.1f}s", flush=True)
            try:
                template_1024 = make_conv_template(1024)
            except Exception:
                template_1024 = None
            up = interpolate_sequence(best_f, 1024)
            if up is not None:
                f_res, s_res = optimize_combined(1024, up, min(rem - 5.0, 200.0), rng, template_1024)
                if np.isfinite(s_res) and s_res < best_score:
                    best_score = float(s_res)
                    best_f = np.asarray(f_res, dtype=np.float64).copy()
                    print(f"  Final upscaling best: n={best_f.size}, score={best_score:.6f}", flush=True)

    # Consolidate: ensure we have a valid sequence
    if best_f is None or best_f.size == 0:
        best_f = np.ones(1024, dtype=np.float64) / 1024.0

    # Final verification
    seq = [float(max(0.0, min(1000.0, x))) for x in best_f]
    final_score = float("inf")
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        pass

    if not np.isfinite(final_score):
        # Fallback to given sequence
        if given_f is not None:
            seq = [float(max(0.0, min(1000.0, x))) for x in given_f]
            try:
                final_score = float(evaluate_sequence(seq))
            except Exception:
                pass

    if not np.isfinite(final_score):
        seq = [1.0 / 64.0] * 64
        try:
            final_score = float(evaluate_sequence(seq))
        except Exception:
            pass

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: n={len(seq)}, score={final_score:.6f}, elapsed={elapsed:.1f}s", flush=True)
    return seq
