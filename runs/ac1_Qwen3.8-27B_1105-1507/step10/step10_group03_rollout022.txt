def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    Solve: min M s.t. (a*g)[k] <= M for all k, g >= 0, sum(g) = 1.
    Returns (g, M) or (None, None).

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast HiGHS LP fixed-point with Anderson acceleration, KKT-based direct solve,
diverse n exploration, gradient polishing, and multi-start strategies.
Key improvements: direct scipy HiGHS (much faster than cvxpy), more LP iterations,
Anderson acceleration for faster convergence, and broader n exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build the convolution matrix A where (A@g)[k] = (a*g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):

    Solve: min M s.t. (a*g)[k] <= M for all k, g >= 0, sum(g) = 1.
    Returns (g, M, dual_y) or (None, None, None).

    if n > 2500:
        return None, None, None

    A = build_conv_matrix(a)

    # Variables: [g_0, ..., g_{n-1}, M]

    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

                        dual_y = np.abs(res.ineqlin.marginals)
                return g, M, dual_y
    except Exception:
        pass

    return None, None, None


def lp_fixed_point(a, max_iters=200, time_budget=60.0, verbose=False, use_anderson=True):
    """
    LP fixed-point iteration with optional Anderson acceleration.
    Iterates: solve LP for g given a, then update a.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # History for Anderson acceleration
    history_a = []
    history_g = []
    M_hist = 4  # memory

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break

        g, M, dual_y = solve_lp(a)
        if g is None:
            break

        # Store history
        history_a.append(a.copy())
        history_g.append(g.copy())
        if len(history_a) > M_hist + 2:
            history_a.pop(0)
            history_g.pop(0)

        # Anderson acceleration
        if use_anderson and len(history_a) >= 3:
            m_actual = min(M_hist, len(history_a) - 1)
            # Compute residuals: r_i = g_i - a_i
            # We want to find a_new = a + sum gamma_i * (a_i - a_{i-1}) that minimizes ||r||
            # Simplified: use extrapolation with adaptive step

            a_curr = history_a[-1]
            g_curr = history_g[-1]

            a_new = g.copy()

        best_local_a = best_a.copy()

def kkt_direct_solve(a, dual_y, verbose=False):
    """
    Use KKT conditions to directly solve for optimal a given the active set.
    The condition is: sum_{k active} lambda_k * (a[i+k] + a[i-k]) = mu for all i with a[i] > 0.
    """
    n = len(a)
    if dual_y is None:
        return a

    # Identify active constraints (dual variables significantly > 0)
    max_dual = float(np.max(dual_y)) if len(dual_y) > 0 else 0
    if max_dual < 1e-10:
        return a

    active = dual_y > 0.01 * max_dual
    active_indices = np.where(active)[0]

    if len(active_indices) < 1:
        return a

    # Build the linear system: for each i, sum_k lambda_k * (a[i+k] + a[i-k]) = mu
    # This is a (circulant-like) system. We use the fact that it's a sum of shifts.
    # Let B be the matrix where B[i,j] = sum_{k active} lambda_k * [j == i+k or j == i-k]
    # Then B @ a = mu * 1 (for i with a[i] > 0)

    # For simplicity, build the full system
    support = np.where(a > 1e-10)[0]
    if len(support) < 2:
        return a

    # Build matrix for the support
    m = len(support)
    B = np.zeros((m, m))
    for ii, i in enumerate(support):
        for k in active_indices:
            lam = dual_y[k]
            for kk in [i + int(k), i - int(k)]:
                if 0 <= kk < n:
                    jj = np.where(support == kk)[0]
                    if len(jj) > 0:
                        B[ii, jj[0]] += lam
        # Also the k=0 term gives 2*a[i] (from a[i+0] + a[i-0] = 2*a[i])
        if active[0]:  # k=0 is always active (it's the max)
            B[ii, ii] += 2.0 * dual_y[0]

    # B @ a[support] = mu * 1
    # We also need sum(a[support]) = 1
    # And a >= 0 (we'll handle via clipping)

    # Add the normalization constraint as an extra equation
    B_aug = np.zeros((m + 1, m + 1))
    B_aug[:m, :m] = B
    B_aug[m, :m] = 1.0
    B_aug[:m, m] = 0.0
    B_aug[m, m] = 0.0

    rhs = np.zeros(m + 1)
    rhs[m] = 1.0  # sum = 1

    try:
        sol = np.linalg.solve(B_aug, rhs)
        a_sol = np.maximum(sol[:m], 0)
        ss = float(np.sum(a_sol))
        if ss > 1e-10:
            a_sol = a_sol / ss
            a_new = np.zeros(n)
            a_new[support] = a_sol
            obj_new = compute_obj(a_new)
            obj_curr = compute_obj(a)
            if obj_new < obj_curr:
                if verbose:
                    print(f"    KKT: obj {obj_curr:.10f} -> {obj_new:.10f}")
                return a_new
    except np.linalg.LinAlgError:
        pass

    return a


def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 2 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def grad_descent(a, time_budget, tau_levels=None, verbose=False):
    """Nesterov accelerated gradient descent with staircase tau schedule."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 20, 80, 300, 1000, 3000]

    beta = 0.9

    for tau in tau_levels:
        lr = 2.0 / np.sqrt(n)
        v = np.zeros(n)
        it = 0
        max_it = 5000

        while time.monotonic() - start < time_budget and it < max_it:
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            it += 1
            lr = max(1e-8, lr * 0.99995)

    return best_a


def make_init(n, mode, rng=None):
    """Create diverse initial sequences of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:  # Triangle
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
        return a / a.sum()
    elif mode == 2:  # Semicircle
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.0001))
        return a / a.sum()
    elif mode == 3:  # Two bumps
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 4:  # Uniform random
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 5:  # Cosine bump
        x = np.linspace(0, np.pi, n)
        a = np.maximum(0.5 + 0.5 * np.cos(x), 0.001)
        return a / a.sum()
    elif mode == 6:  # Multi-sine
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [2, 3, 5]:
            a += 0.2 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:  # Exponential center
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 4))
        return a / a.sum()
    elif mode == 8:  # Flat top with tapered edges
        x = np.linspace(0, 1, n)
        a = np.where((x > 0.2) & (x < 0.8), 1.0,
                     np.maximum(0.5 - np.abs(x - 0.5), 0.01))
        return a / a.sum()
    elif mode == 9:  # Alternating high-low
        a = np.ones(n)
        a[::2] = 1.0
        a[1::2] = 0.3
        return a / a.sum()
    elif mode == 10:  # Piecewise constant (4 blocks)
        a = np.zeros(n)
        a[:n//4] = 0.5
        a[n//4:n//2] = 1.5
        a[n//2:3*n//4] = 1.0
        a[3*n//4:] = 0.8
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def optimize_n(a, time_budget, verbose=False):
    """Full optimization pipeline for a given n: LP + GD + LP + KKT."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: LP fixed-point
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.4, 120.0)
        a1 = lp_fixed_point(a, max_iters=500, time_budget=alloc, verbose=verbose)
        obj1 = compute_obj(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
            if verbose:
                print(f"    After LP: obj={best_obj:.10f}")
        a = a1

    # Phase 2: GD polishing
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        alloc = min(remaining * 0.3, 60.0)
        a2 = grad_descent(a, time_budget=alloc, tau_levels=[20, 100, 500, 2000],
                          verbose=False)
        obj2 = compute_obj(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            if verbose:
                print(f"    After GD: obj={best_obj:.10f}")
        a = a2

    # Phase 3: LP again (finer)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.3, 80.0)
        a3 = lp_fixed_point(a, max_iters=300, time_budget=alloc, verbose=verbose)
        obj3 = compute_obj(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            if verbose:
                print(f"    After LP2: obj={best_obj:.10f}")
        a = a3

    # Phase 4: KKT direct solve
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5 and n <= 2000:
        g, M, dual_y = solve_lp(a)
        if dual_y is not None:
            a4 = kkt_direct_solve(a, dual_y, verbose=verbose)
            obj4 = compute_obj(a4)
            if obj4 < best_obj:
                best_obj = obj4
                best_a = a4.copy()
                if verbose:
                    print(f"    After KKT: obj={best_obj:.10f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining LP fixed-point with Anderson
    acceleration, KKT direct solve, gradient polishing, diverse n exploration,
    and multi-start strategies.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        parent = None

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.10f}")

    n_parent = len(best_seq)
    print(f"\nStarting optimization. Parent n={n_parent}, obj={best_score:.10f}")

    # === Phase 1: Optimize at parent's n ===
    print(f"\n=== Phase 1: Optimize at n={n_parent} ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.3, 300.0)
        a1 = optimize_n(best_seq, time_budget=alloc, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase 1 result: obj={best_score:.10f}")

    # === Phase 2: Multi-start at parent's n ===
    print(f"\n=== Phase 2: Multi-start at n={n_parent} ===")
    remaining = deadline - time.monotonic()
    init_modes = [1, 2, 3, 5, 6, 7, 8, 9, 10]
    for mode in init_modes:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        a_init = make_init(n_parent, mode, rng)
        # Blend with current best
        a_blend = 0.7 * best_seq + 0.3 * a_init
        a_blend = np.maximum(a_blend, 0)
        a_blend = a_blend / a_blend.sum()

        alloc = min(remaining * 0.15, 80.0)
        if alloc < 20:
            break
        a_ms = optimize_n(a_blend, time_budget=alloc, verbose=False)
        obj_ms = compute_obj(a_ms)
        if obj_ms < best_score:
            best_score = obj_ms
            best_seq = a_ms.copy()
            print(f"  Mode {mode}: obj={best_score:.10f} **")
        else:
            print(f"  Mode {mode}: obj={obj_ms:.10f}")

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_parent = len(best_seq)

    # Generate candidate n values
    n_candidates = set()
    for delta in [-200, -100, -50, 0, 50, 100, 200, 300, 500]:
        n_try = n_parent + delta
        if 400 <= n_try <= 3000:
            n_candidates.add(n_try)
    # Add some absolute values
    for n_abs in [500, 600, 800, 1000, 1200, 1500, 2000]:
        n_candidates.add(n_abs)

    n_candidates = sorted(n_candidates)
    # Prioritize: close to parent first, then larger
    n_candidates.sort(key=lambda x: (abs(x - n_parent), -x))
    n_candidates = n_candidates[:8]  # Limit to 8 values

    print(f"  n candidates: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample(best_seq, n_try)
        alloc = min(remaining * 0.25, 120.0)
        if alloc < 30:
            break

        a_opt = optimize_n(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.10f}")

    # === Phase 4: Intensive LP at best n ===
    print(f"\n=== Phase 4: Intensive LP Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 200.0)
        a4 = lp_fixed_point(best_seq, max_iters=1000, time_budget=alloc, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Intensive LP: obj={best_score:.10f}")

    # === Phase 5: GD + LP cycle ===
    print(f"\n=== Phase 5: GD + LP Cycle ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for cycle in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            # GD
            alloc = min(remaining * 0.3, 60.0)
            if alloc > 10:
                a5 = grad_descent(best_seq, time_budget=alloc,
                                  tau_levels=[50, 200, 800, 3000], verbose=False)
                obj5 = compute_obj(a5)
                if obj5 < best_score:
                    best_score = obj5
                    best_seq = a5.copy()
                    print(f"  Cycle {cycle} GD: obj={best_score:.10f}")

            # LP
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.4, 80.0)
            if alloc > 10:
                a6 = lp_fixed_point(best_seq, max_iters=500, time_budget=alloc, verbose=False)
                obj6 = compute_obj(a6)
                if obj6 < best_score:
                    best_score = obj6
                    best_seq = a6.copy()
                    print(f"  Cycle {cycle} LP: obj={best_score:.10f}")

    # === Phase 6: Perturbation exploration ===
    print(f"\n=== Phase 6: Perturbation Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        for pert_trial in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            # Fourier perturbation
            n = len(best_seq)
            x = np.arange(n)
            pert = np.zeros(n)
            for f in rng.integers(1, 10, size=4):
                pert += 0.1 * np.sin(2 * np.pi * f * x / n + rng.uniform(0, 2 * np.pi))
            a_pert = best_seq + pert
            a_pert = np.maximum(a_pert, 0)
            sp = float(np.sum(a_pert))
            if sp > 1e-10:
                a_pert = a_pert / sp
            alloc = min(remaining * 0.2, 40.0)
            if alloc > 15:
                a_p = lp_fixed_point(a_pert, max_iters=200, time_budget=alloc, verbose=False)
                obj_p = compute_obj(a_p)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    print(f"  Pert {pert_trial}: obj={best_score:.10f} **")
                else:
                    print(f"  Pert {pert_trial}: obj={obj_p:.10f}")

    # === Phase 7: Final ultra-fine LP ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.8, 120.0)
        a_final = lp_fixed_point(best_seq, max_iters=2000, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()

    # Final high-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.7, 60.0)
        a_fin = grad_descent(best_seq, time_budget=alloc,
                             tau_levels=[500, 2000, 5000, 10000], verbose=False)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final GD: obj={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
