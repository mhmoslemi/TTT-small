```python
def fw_exact(a, time_budget, verbose=False, max_iters=1000000, eps_frac=0.0, lr_decay=0.99999):
    """
    Frank-Wolfe method with exact subgradient of max(conv(a,a)).
    Each iteration is O(n * |S|) where |S| is the number of near-max indices.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    it = 0
    bad_streak = 0
    t = 0.5  # step size
    
    # Track last moved-to coordinate for away-step
    last_moved = -1
    active_set = []  # recent moved-to coordinates
    
    while time.monotonic() - start < time_budget:
        if it >= max_iters:
            break
        
        b = compute_conv(a)
        b_max = float(np.max(b))
        
        # Find S (set of indices near max)
        if eps_frac > 0:
            threshold = b_max * (1.0 - eps_frac)
            S_idx = np.where(b >= threshold)[0]
        else:
            S_idx = np.where(b >= b_max - 1e-12)[0]
        
        # Compute subgradient g
        g = np.zeros(n)
        for k in S_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(int(k), n - 1)
            if i_lo <= i_hi:
                g[i_lo:i_hi+1] += 2.0 * a[k-i_hi:int(k)-i_lo+1][::-1]
        
        # Find min coordinate
        i_star = int(np.argmin(g))
        
        # FW step
        a_new = (1.0 - t) * a + t * np.zeros(n)
        a_new[i_star] += t
        
        # Optional away-step
        if last_moved >= 0 and a[last_moved] > 1e-8 and np.random.rand() < 0.3:
            a_away = (1.0 - t) * a
            a_away[last_moved] -= t
            if a_away[last_moved] >= -1e-10:
                a_away[last_moved] = max(0, a_away[last_moved])
                a_new_away = a_away + t * np.zeros(n)  # no addition, just removal
                # Actually, away step is: a <- (1-t)*a - t*e_j
                # But we need to clip at 0
                a_away = np.maximum(a_away, 0)
                sa = float(np.sum(a_away))
                if sa > 1e-10:
                    a_away = a_away / sa
                    obj_away = compute_obj(a_away)
                    if obj_away < compute_obj(a_new):
                        a_new = a_away
        
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a = best_a.copy()
            t = 0.1
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            bad_streak = 0
        else:
            bad_streak += 1
        
        a = a_new
        last_moved = i_star
        active_set.append(i_star)
        if len(active_set) > 20:
            active_set.pop(0)
        
        it += 1
        t = max(0.01, t * lr_decay)
        
        if bad_streak > 50000:
            # Restart with perturbation
            a = best_a.copy()
            t = 0.5
            bad_streak = 0
            # Perturb
            perturb_idx = np.random.choice(n, size=min(5, n), replace=False)
            a[perturb_idx] += 0.01
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a
```

This is the cross-correlation of `p_reversed` and `a_rev` at lag `i`:
`= (p_rev cross_corr a_rev)[i]`

In FFT: `ifft(fft(p_rev, size) * conj(fft(a_rev, size)))` ...

```python
def smoothed_grad_logsumexp(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*b_k)) w.r.t. a."""
    n = len(a)
    b = compute_conv(a)
    b_max = float(np.max(b))
    p = np.exp(tau * (b - b_max))
    p = p / p.sum()
    
    # grad_i = 2 * sum_k p[k] * a[k-i] for valid k-i
    g = np.zeros(n)
    for i in range(n):
        # k-i ranges from 0 to n-1, so k ranges from i to i+n-1
        k_lo = i
        k_hi = min(i + n - 1, 2*n - 2)
        if k_lo <= k_hi:
            g[i] = 2.0 * np.dot(p[k_lo:k_hi+1], a[k_hi-i:k_lo-i+1][::-1])
    
    # Hmm, this is O(n^2). For n=800, that's 640K ops per gradient evaluation.
    # With 20K iterations per second target, this is too slow.
    
    # Let me use FFT instead.
    # g[i] = 2 * sum_k p[k] * a[k-i] for 0<=k-i<n
    # = 2 * sum_j a[j] * p[j+i] for j=0..n-1
    # This is the cross-correlation of a and p at positive lags.
    
    size = 1
    while size < 2 * (2*n - 1):
        size <<= 1
    
    A = fft(a, size)
    P = fft(p, size)
    
    # Cross-correlation: corr[i] = sum_j a[j] * p[j+i]
    # = ifft(conj(fft(a)) * fft(p)) ... 
    # Actually: ifft(fft(a, size) * conj(fft(p, size)))[i] = sum_j a[j] * conj(p[j-i])
    # For real p: = sum_j a[j] * p[j-i]
    # I want sum_j a[j] * p[j+i] = sum_j a[j] * p[(j+i)]
    # = sum_j a[j+i] * p[i] ... no that's not right either.
    
    # Let me define: I want C[i] = sum_{j=0}^{n-1} a[j] * p[j+i] for i=0..n-1
    # Note that p has length 2n-1, so p[j+i] is valid for j+i <= 2n-2.
    
    # C[i] = sum_j a[j] * p[j+i]
    # This is the (linear) cross-correlation of a (length n) and p (length 2n-1) at lag i.
    # cross_corr[i] = sum_j a[j] * p[j+i]
    
    # Using FFT: cross_corr = ifft(fft(a, size) * conj(fft(p, size)))... 
    # Let's check: ifft(fft(a) * conj(fft(p)))[n] = sum_k a[k] * p[k-n] (circular, with zero-padding)
    # With zero-padding to size >= n + (2n-1) - 1 = 3n-2, we get linear correlation.
    # ifft(fft(a, size) * conj(fft(p, size)))[n] = sum_k a[k] * p[k-n] for k where both are valid
    # = sum_k a[k] * p[k-n]
    # Let m = k-n: = sum_m a[m+n] * p[m] ... this gives negative lags.
    
    # I want positive lags: sum_j a[j] * p[j+i]
    # = sum_j a[j+i-i] * p[j+i]
    # Let me try: ifft(fft(p, size) * conj(fft(a, size)))[i] = sum_k p[k] * a[k-i]
    # = sum_k p[k] * a[k-i] for valid k, k-i
    # This is exactly what I want! g[i] = 2 * sum_k p[k] * a[k-i]
    
    # So: g = 2 * ifft(fft(p, size) * conj(fft(a, size)))[:n]
    
    result = ifft(fft(p, size) * np.conj(fft(a, size))).real[:n]
    g = 2.0 * result
    
    return g, b_max, float(p[np.argmax(b)] * tau)  # return grad, exact max, and "effective" value
```

Let me verify this FFT formula:
- `fft(p, size)` gives the DFT of `p` (zero-padded to `size`)
- `fft(a, size)` gives the DFT of `a` (zero-padded to `size`)
- `fft(p) * conj(fft(a))` = DFT of the cross-correlation (by the cross-correlation theorem)
- `ifft(fft(p) * conj(fft(a)))` = circular cross-correlation of `p` and `a`
- = `sum_k p[k] * a[(k-i) mod size]` for each `i`
- With sufficient zero-padding (size >= len(p) + len(a) - 1 = 3n-2), this equals the linear cross-correlation
- = `sum_k p[k] * a[k-i]` (with `a[j] = 0` for `j < 0` or `j >= n`)
- = `sum_{k: 0<=k-i<n} p[k] * a[k-i]` ✓

So the formula is:
`g[i] = 2 * ifft(fft(p, size) * conj(fft(a, size)))[:n].real`

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations:
1. Frank-Wolfe with exact subgradient exploiting convolution structure (O(n*log(n)) per iter)
2. Multiplicative Weight Gradient Descent (mirror descent on simplex) with log-sum-exp smoothing
3. Cross-correlation via FFT for efficient smoothed gradient computation
4. LP polishing for local optimality
5. Multi-n exploration and adaptive time allocation
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def subgradient_exact(a, eps_abs=1e-10):
    """
    Compute subgradient of max(conv(a,a)) w.r.t. a using exact argmax set.
    g_i = 2 * sum_{k in S} a[k-i] for valid indices.
    Returns (g, b_max, S_idx).
    """
    n = len(a)
    b = compute_conv(a)
    b_max = float(np.max(b))
    threshold = b_max - eps_abs
    S_idx = np.where(b >= threshold)[0]

    g = np.zeros(n)
    for k in S_idx:
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            g[i_lo:i_hi + 1] += 2.0 * a[k - i_hi:k - i_lo + 1][::-1]

    return g, b_max, S_idx


def smoothed_grad_lse(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*b_k)) w.r.t. a.
    Uses FFT cross-correlation for O(n log n) computation.
    Returns (gradient of length n, exact_max of b).
    """
    n = len(a)
    b = compute_conv(a)
    b_max = float(np.max(b))

    # Compute softmax probabilities
    exp_vals = np.exp(tau * (b - b_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(b))
        p[np.argmax(b)] = 1.0
    else:
        p = exp_vals / sum_exp

    # g[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
    # = 2 * ifft(fft(p, size) * conj(fft(a, size)))[:n]
    size = 1
    while size < 3 * n:
        size <<= 1

    P = fft(p, size)
    A = fft(a, size)
    result = ifft(P * np.conj(A)).real[:n]

    return 2.0 * result, b_max


def fw_method(a, time_budget, verbose=False, eps_abs=1e-10, max_iters=5000000):
    """
    Frank-Wolfe method with exact subgradient.
    Very fast per-iteration: O(n * |S|) for subgradient + O(n log n) for conv.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    it = 0
    bad_streak = 0
    t = 0.5
    last_moved = -1

    while time.monotonic() - start < time_budget:
        if it >= max_iters:
            break

        # Compute subgradient
        g, b_max, S_idx = subgradient_exact(a, eps_abs=eps_abs)

        # FW step: move to vertex with min subgradient
        i_star = int(np.argmin(g))

        # Standard FW update
        a_new = a.copy()
        a_new = (1.0 - t) * a + t * np.zeros(n)
        a_new[i_star] = t

        # Away-step: occasionally remove mass from last moved coordinate
        if last_moved >= 0 and last_moved != i_star and a[last_moved] > 1e-6 and it > 100:
            a_away = a.copy()
            a_away[last_moved] = max(0.0, a[last_moved] - t)
            a_away[i_star] = a[i_star] + t
            a_away = np.maximum(a_away, 0)
            s_away = float(np.sum(a_away))
            if s_away > 1e-10:
                a_away = a_away / s_away
                # Only use away-step if it doesn't increase the max
                b_away = compute_conv(a_away)
                if float(np.max(b_away)) <= b_max + 1e-10:
                    a_new = a_away

        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a = best_a.copy()
            t = max(0.05, t * 0.5)
            it += 1
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            bad_streak = 0
        else:
            bad_streak += 1

        a = a_new
        last_moved = i_star
        it += 1

        # Adaptive step size
        if bad_streak < 1000:
            t = min(0.5, t * 1.001)
        else:
            t = max(0.02, t * 0.9999)

        # Periodic restart to escape local minima
        if bad_streak > 100000:
            a = best_a.copy()
            t = 0.5
            bad_streak = 0
            # Small random perturbation
            n_pert = max(3, n // 20)
            pert_idx = np.random.choice(n, size=n_pert, replace=False)
            a[pert_idx] += np.random.uniform(0, 0.02, size=n_pert)
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and it % 100000 == 0:
            elapsed = time.monotonic() - start
            print(f"    FW it={it}: obj={best_obj:.10f}, t={t:.4f}, bad={bad_streak}, time={elapsed:.0f}s")

    return best_a


def mwg_method(a, time_budget, verbose=False, tau_levels=None, eta_init=None):
    """
    Multiplicative Weight Gradient Descent (mirror descent on simplex).
    a_i <- a_i * exp(-eta * g_i), then normalize.
    Uses log-sum-exp smoothed objective with staircase tau schedule.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 20, 80, 300, 1000, 3000, 10000]

    if eta_init is None:
        eta_init = 5.0

    for tau in tau_levels:
        eta = eta_init / tau
        level_start = time.monotonic()
        max_level_time = 40.0
        it_level = 0
        bad_level = 0

        while time.monotonic() - start < time_budget:
            if time.monotonic() - level_start > max_level_time:
                break
            if it_level >= 200000:
                break

            # Compute smoothed gradient
            g, b_max = smoothed_grad_lse(a, tau)

            # MWG update: a_i <- a_i * exp(-eta * g_i)
            log_a = np.log(np.maximum(a, 1e-300))
            log_a -= eta * g
            # Softmax normalize (equivalent to exp then normalize)
            log_a -= np.max(log_a)
            a_new = np.exp(log_a)
            s_new = float(np.sum(a_new))
            if s_new < 1e-300:
                a_new = a.copy()
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                bad_level = 0
            else:
                bad_level += 1

            a = a_new
            it_level += 1

            # Adaptive eta
            if bad_level > 5000:
                eta *= 0.9
                bad_level = 0

        eta_init = eta * tau / max(1, tau_levels[0])  # carry over scaled
        if verbose:
            elapsed = time.monotonic() - start
            print(f"    MWG tau={tau:6d}: obj={best_obj:.10f}, iters={it_level}, t={elapsed:.0f}s")

    return best_a


def gd_method(a, time_budget, verbose=False, tau_levels=None):
    """
    Nesterov accelerated gradient descent with log-sum-exp smoothing.
    Staircase tau schedule for progressive refinement.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 20, 80, 300, 1000, 3000]

    beta = 0.8
    it = 0
    best_it = 0

    for tau in tau_levels:
        lr = 3.0 / np.sqrt(n)
        v = np.zeros(n)
        it_level = 0
        level_start = time.monotonic()
        max_level_time = 35.0
        bad_level = 0

        while time.monotonic() - start < time_budget:
            if time.monotonic() - level_start > max_level_time:
                break
            if it_level >= 100000:
                break

            # Nesterov lookahead
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            # Compute gradient
            g, _ = smoothed_grad_lse(y, tau)

            # Update
            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it
                bad_level = 0
            else:
                bad_level += 1

            a = a_new
            it += 1
            it_level += 1
            lr = max(1e-8, lr * 0.99997)

            if bad_level > 20000:
                a = best_a.copy()
                v = np.zeros(n)
                lr = 3.0 / np.sqrt(n)
                bad_level = 0

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    GD tau={tau:6d}: obj={best_obj:.10f}, iters={it_level}, t={elapsed:.0f}s")

    return best_a


def lp_polish(a, time_budget, verbose=False, seed=42, max_iters=30):
    """
    LP polishing: solve min M s.t. conv(a,g)<=M, g>=0, sum(g)=1.
    Then blend a with g.
    """
    n = len(a)
    if n > 2500:
        return a

    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    for _ in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Build convolution matrix
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    sg = float(np.sum(g_val))
                    if sg > 1e-10:
                        g_val = g_val / sg
                        solved = True
                        break
            except Exception:
                continue

        if not solved:
            break

        # Try multiple blending factors
        best_t_obj = best_obj
        best_t_val = 0.1
        best_a_local = best_a.copy()

        t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            if verbose:
                print(f"    LP: obj={best_obj:.10f}, t={best_t_val:.3f}")
        else:
            stall += 1
            # Perturb to escape
            pert = 0.01 * np.random.randn(n)
            a = a + pert
            a = np.maximum(a, 0)
            a = a / a.sum()
            if stall > 5:
                break

    return best_a


def make_init(n, mode, rng=None):
    """Create diverse initial sequences of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:  # Triangle
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
        return a / a.sum()
    elif mode == 2:  # Semicircle
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.0001))
        return a / a.sum()
    elif mode == 3:  # Two bumps
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 4:  # Uniform random
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 5:  # Cosine
        x = np.linspace(0, np.pi, n)
        a = np.maximum(0.5 + 0.5 * np.cos(x), 0.001)
        return a / a.sum()
    elif mode == 6:  # Multi-sine
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [2, 3, 5, 7]:
            a += 0.15 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:  # Exponential center
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 4))
        return a / a.sum()
    elif mode == 8:  # Flat top tapered
        x = np.linspace(0, 1, n)
        a = np.where((x > 0.15) & (x < 0.85), 1.0,
                     np.maximum(0.5 - np.abs(x - 0.5), 0.01))
        return a / a.sum()
    elif mode == 9:  # Cosh (Chebyshev-inspired)
        x = np.linspace(0, 1, n)
        beta = 0.3
        a = np.cosh(beta * (2 * x - 1))
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining FW with exact subgradients,
    MWG (mirror descent), smoothed GD, LP polishing, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        parent = None

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.10f}")

    n_current = len(best_seq)

    # === Phase 1: Quick diverse screening ===
    print(f"\n=== Phase 1: Diverse Init Screening ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for mode in [0, 1, 2, 5, 9]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_init = make_init(n_current, mode, rng)
            alloc = min(remaining * 0.03, 10.0)
            if alloc < 5:
                break
            a_screen = gd_method(a_init, time_budget=alloc, tau_levels=[10, 100])
            obj_screen = compute_obj(a_screen)
            if obj_screen < best_score:
                best_score = obj_screen
                best_seq = a_screen.copy()
                print(f"  Init mode {mode}: obj={best_score:.10f} **")
            else:
                print(f"  Init mode {mode}: obj={obj_screen:.10f}")

    # === Phase 2: FW with exact subgradient (fast, many iterations) ===
    print(f"\n=== Phase 2: Frank-Wolfe Exact ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 30:
        a2 = fw_method(best_seq, time_budget=p2_budget, verbose=True, eps_abs=1e-10)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  FW exact: obj={best_score:.10f} **")
        else:
            print(f"  FW exact: obj={obj2:.10f}")

    # === Phase 3: MWG (multiplicative weight GD) ===
    print(f"\n=== Phase 3: MWG Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 180.0)
    if p3_budget > 30:
        a3 = mwg_method(best_seq, time_budget=p3_budget, verbose=True,
                        tau_levels=[10, 50, 200, 800, 3000])
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  MWG: obj={best_score:.10f} **")
        else:
            print(f"  MWG: obj={obj3:.10f}")

    # === Phase 4: Smoothed GD (Nesterov) ===
    print(f"\n=== Phase 4: Nesterov GD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 180.0)
    if p4_budget > 30:
        a4 = gd_method(best_seq, time_budget=p4_budget, verbose=True,
                       tau_levels=[10, 50, 200, 800, 3000])
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  GD: obj={best_score:.10f} **")
        else:
            print(f"  GD: obj={obj4:.10f}")

    # === Phase 5: LP Polishing ===
    print(f"\n=== Phase 5: LP Polishing ===")
    remaining = deadline - time.monotonic()
    n_now = len(best_seq)
    if remaining > 60 and n_now <= 2000:
        p5_budget = min(remaining * 0.15, 180.0)
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = p5_budget / max(1, 3 - rnd)
            a5 = lp_polish(best_seq, time_budget=alloc, verbose=True, seed=100 + rnd * 50)
            obj5 = compute_obj(a5)
            if obj5 < best_score:
                best_score = obj5
                best_seq = a5.copy()
                print(f"  LP rnd{rnd}: obj={best_score:.10f} **")
            else:
                print(f"  LP rnd{rnd}: obj={obj5:.10f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        n_current = len(best_seq)
        n_options = sorted(set([
            n_current - 50, n_current + 50, n_current + 100,
            n_current + 200, n_current + 400,
            600, 800, 1000, 1200, 1500, 2000
        ]))
        n_options = [n for n in n_options if 300 <= n <= 2500 and n != n_current]
        n_options.sort(key=lambda x: abs(x - n_current))
        n_options = n_options[:4]

        print(f"  n candidates: {n_options}")
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break

            a_try = resample(best_seq, n_try)

            # Quick GD
            alloc1 = min(remaining * 0.2, 40.0)
            if alloc1 > 10:
                a_try = gd_method(a_try, time_budget=alloc1, tau_levels=[10, 100, 500])

            # FW for a bit
            remaining = deadline - time.monotonic()
            alloc2 = min(remaining * 0.15, 30.0)
            if alloc2 > 10:
                a_try = fw_method(a_try, time_budget=alloc2, eps_abs=1e-8)

            # LP if small enough
            remaining = deadline - time.monotonic()
            if n_try <= 1200:
                alloc3 = min(remaining * 0.1, 20.0)
                if alloc3 > 10:
                    a_try = lp_polish(a_try, time_budget=alloc3, seed=n_try)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.10f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 7: Combined refinement loop ===
    print(f"\n=== Phase 7: Combined Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        n_ref = len(best_seq)
        for round_idx in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            # FW (short)
            alloc = min(remaining * 0.2, 50.0)
            if alloc > 10:
                a_r = fw_method(best_seq, time_budget=alloc, eps_abs=1e-9)
                obj_r = compute_obj(a_r)
                if obj_r < best_score:
                    best_score = obj_r
                    best_seq = a_r.copy()
                    print(f"  Refine r{round_idx} FW: obj={best_score:.10f} **")

            # GD (short)
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.2, 40.0)
            if alloc > 10:
                a_r = gd_method(best_seq, time_budget=alloc,
                               tau_levels=[50, 300, 1500])
                obj_r = compute_obj(a_r)
                if obj_r < best_score:
                    best_score = obj_r
                    best_seq = a_r.copy()
                    print(f"  Refine r{round_idx} GD: obj={best_score:.10f} **")

            # LP (short)
            remaining = deadline - time.monotonic()
            if len(best_seq) <= 1500:
                alloc = min(remaining * 0.2, 30.0)
                if alloc > 10:
                    a_r = lp_polish(best_seq, time_budget=alloc, seed=500 + round_idx * 30)
                    obj_r = compute_obj(a_r)
                    if obj_r < best_score:
                        best_score = obj_r
                        best_seq = a_r.copy()
                        print(f"  Refine r{round_idx} LP: obj={best_score:.10f} **")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau GD
    if remaining > 30:
        alloc = min(remaining * 0.3, 60.0)
        a_f = gd_method(best_seq, time_budget=alloc,
                       tau_levels=[200, 800, 3000, 8000])
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final GD: obj={best_score:.10f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 20 and len(best_seq) <= 2000:
        alloc = min(remaining * 0.3, 50.0)
        a_f = lp_polish(best_seq, time_budget=alloc, seed=9999)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    # Final FW
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 40.0)
        a_f = fw_method(best_seq, time_budget=alloc, eps_abs=1e-12)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final FW: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
```
