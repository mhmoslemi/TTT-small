"""
Adam-based smooth max optimizer for minimum peak autoconvolution.
Uses adaptive temperature schedule, multiple lengths, and restart strategy.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear autoconvolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def smooth_max_weights(x, tau):
    """Compute softmax weights for smooth max approximation."""
    m = np.max(x)
    shifted = (x - m) / tau
    shifted = np.clip(shifted, -50, 50)
    e = np.exp(shifted)
    w = e / np.sum(e)
    val = m + tau * np.log(np.sum(e) / len(x))
    return val, w

def gradient_from_weights(a, w, n):
    """Compute gradient of smooth_max(conv(a)) w.r.t. a."""
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    w_pad = np.zeros(N)
    w_pad[:2*n-1] = w
    a_pad = np.zeros(N)
    a_pad[:n] = a
    W = np.fft.fft(w_pad)
    A = np.fft.fft(a_pad)
    grad = 2.0 * np.fft.ifft(W * np.conj(A)).real[:n]
    return grad

def adam_optimize(a_init, time_budget, tau_start, tau_end, lr, seed=42):
    """Adam optimizer minimizing smooth max of autoconvolution."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a_init / max(np.sum(a_init), 1e-10)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.6)
        
        conv = compute_conv(a)
        val, w = smooth_max_weights(conv, tau)
        grad = gradient_from_weights(a, w, n)
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        lr = max(1e-6, lr * 0.9998)
    
    return best_a

def resample(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:

    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    new_a = np.interp(x_new, x_old, a)
    return np.maximum(new_a, 0)

"""
Adam-based smooth-max optimizer with multi-length search, subgradient refinement,
and perturbation restarts for minimum peak autoconvolution.
Key innovations: adaptive temperature scheduling, dual parameterization (simplex + log),
systematic length exploration, and Polyak-averaged subgradient polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear autoconvolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def smooth_max_weights(x, tau):
    """Compute smooth max value and softmax weights."""
    m = np.max(x)
    shifted = (x - m) / tau
    shifted = np.clip(shifted, -60, 60)
    e = np.exp(shifted)
    total = np.sum(e)
    w = e / total
    val = m + tau * np.log(total / len(x))
    return val, w


def gradient_from_weights(a, w, n):
    """Compute gradient of smooth_max(conv(a)) w.r.t. a using FFT.
    d(conv[k])/d(a[i]) = 2*a[k-i] for valid indices.
    grad_a[i] = 2 * sum_k w[k] * a[k-i] = 2 * cross_correlation(w, a)[i]
    """
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_pad = np.zeros(N)
    a_pad[:n] = a
    W = np.fft.fft(w_pad)
    A = np.fft.fft(a_pad)
    grad = 2.0 * np.fft.ifft(W * np.conj(A)).real[:n]
    return grad


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_set = np.where(cond)[0]
    if len(rho_set) == 0:
        return np.ones(n) / n
    rho = rho_set[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def adam_optimize(a_init, time_budget, tau_start, tau_end, lr_init, beta1=0.85, beta2=0.999, decay=0.9998):
    """Adam optimizer minimizing smooth max of autoconvolution on simplex."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a_init.copy()
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    eps = 1e-8
    t = 0
    lr = lr_init

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.5)

        conv = compute_conv(a)
        val, w = smooth_max_weights(conv, tau)
        grad = gradient_from_weights(a, w, n)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new
        lr = max(1e-7, lr * decay)

    return best_a


def adam_log_optimize(a_init, time_budget, tau_start, tau_end, lr_init=0.1):
    """Adam optimizer in log-parameterization (a = softmax(x))."""
    n = len(a_init)
    a0 = np.maximum(a_init, 1e-15)
    a0 = a0 / a0.sum()
    x = np.log(a0)

    best_a = a0.copy()
    best_obj = compute_obj(a0)
    start = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    lr = lr_init

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.6)

        # Softmax
        x_shifted = x - np.max(x)
        a = np.exp(x_shifted)
        a = a / a.sum()

        conv = compute_conv(a)
        val, w = smooth_max_weights(conv, tau)
        grad_a = gradient_from_weights(a, w, n)

        # Gradient w.r.t. x: a[j] * (grad_a[j] - dot(a, grad_a))
        correction = np.dot(a, grad_a)
        grad_x = a * (grad_a - correction)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * grad_x ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        lr = max(1e-6, lr * 0.9999)

    return best_a


def subgradient_refine(a_init, time_budget, K=5, decay=0.9995):
    """Subgradient method with Polyak averaging on top-K peaks."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a_init.copy()
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    a_avg = a.copy()
    count = 0
    lr = 5.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        top_vals = conv[top_idx]
        max_val = top_vals[0]
        if max_val < 1e-15:
            break

        # Compute weighted subgradient
        g = np.zeros(n)
        for k in top_idx:
            k = int(k)
            # d(conv[k])/d(a[i]) = 2*a[k-i] for 0<=k-i<n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            a_idx = k - idx
            g[idx] += 2.0 * a[a_idx]

        # Normalize and apply momentum
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        # Polyak averaging
        count += 1
        a_avg = a_avg + (a_new - a_avg) / count
        obj_avg = compute_obj(a_avg)
        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = a_avg.copy()

        a = a_new
        lr = max(1e-10, lr * decay)

    return best_a


def lp_direction(a, time_budget_single=5.0):
    """Solve LP to find direction g minimizing max(a*g conv). Uses cvxpy."""
    try:
        import cvxpy as cp
        n = len(a)
        a = a / a.sum()

        # Build conv matrix: (A @ g)[k] = sum_j a[j]*g[k-j]
        # Actually conv(a, g)[k] = sum_j a[j] * g[k-j]
        # We want: for all k, sum_j a[j]*g[k-j] <= M
        # This is linear in g!
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            # a[j] * g[k-j] contributes to position k
            # For fixed j, k-j ranges over valid g indices: 0 <= k-j < n
            # So k ranges from j to j+n-1
            k_lo = j
            k_hi = min(2 * n - 2, j + n - 1)
            A[k_lo:k_hi + 1, j] = a[j]

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = g_val.sum()
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except Exception:
        pass
    return None


def lp_line_search(a, time_budget, seed=42):
    """LP-based direction finding with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - start < time_budget:
        g = lp_direction(a)
        if g is None:
            break

        best_t = 0
        best_t_obj = best_obj
        for t_val in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t_val) * best_a + t_val * g
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_t_obj - 1e-12:
                best_t_obj = obj
                best_t = t_val

        if best_t > 0:
            best_a = (1 - best_t) * best_a + best_t * g
            best_a = np.maximum(best_a, 0)
            best_a = best_a / best_a.sum()
            best_obj = best_t_obj
            a = best_a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                a = best_a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if time.monotonic() - start > time_budget * 0.8:
                    break

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0.0)


def try_length(a_base, n_target, time_budget, seed=42):
    """Optimize at a specific length. Returns best sequence found."""
    a = resample(a_base, n_target)
    a = a / a.sum()

    best = a.copy()
    best_obj = compute_obj(a)

    # Adam phase 1
    b1 = time_budget * 0.4
    if b1 > 5:
        a1 = adam_optimize(a, b1, tau_start=0.01, tau_end=0.002, lr=0.4 / n_target)
        o1 = compute_obj(a1)
        if o1 < best_obj:
            best_obj = o1
            best = a1.copy()

    # Adam phase 2 (lower temp)
    remaining = time_budget - b1
    b2 = remaining * 0.4
    if b2 > 5:
        a2 = adam_optimize(best, b2, tau_start=0.005, tau_end=0.0005, lr=0.2 / n_target)
        o2 = compute_obj(a2)
        if o2 < best_obj:
            best_obj = o2
            best = a2.copy()

    # Subgradient polish
    remaining = time_budget - b1 - b2
    if remaining > 5:
        a3 = subgradient_refine(best, remaining, K=5)
        o3 = compute_obj(a3)
        if o3 < best_obj:
            best_obj = o3
            best = a3.copy()

    return best


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines Adam smooth-max, LP directions, subgradient refinement,
    multi-length exploration, and perturbation restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    n_target = 800

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        if parent.size > 0 and np.isfinite(parent).all():
            s = parent.sum()
            if s > 0.01:
                best_seq = parent / s
                best_score = compute_obj(best_seq)
                n_target = parent.size
                print(f"Parent (n={n_target}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default (n={n_target}): obj={best_score:.8f}")

    rng = np.random.default_rng(seed)

    # === Phase 1: Adam warmup at parent length ===
    print("\n=== Phase 1: Adam warmup ===")
    remaining = deadline - time.monotonic()
    b1 = min(remaining * 0.15, 150.0)
    if b1 > 20:
        a = adam_optimize(best_seq, b1, tau_start=0.02, tau_end=0.003, lr=0.5 / len(best_seq))
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Phase1 A: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 A: obj={obj:.8f}")

    # === Phase 2: Low-temp Adam + log param ===
    print("\n=== Phase 2: Low-temp + log param ===")
    remaining = deadline - time.monotonic()
    b2 = min(remaining * 0.2, 200.0)
    if b2 > 20:
        # Try both parameterizations
        b2a = b2 * 0.5
        a1 = adam_optimize(best_seq, b2a, tau_start=0.008, tau_end=0.0008, lr=0.3 / len(best_seq))
        o1 = compute_obj(a1)
        if o1 < best_score:
            best_score = o1
            best_seq = a1.copy()
            print(f"  Phase2 simplex: obj={best_score:.8f} **")

        b2b = b2 * 0.5
        a2 = adam_log_optimize(best_seq, b2b, tau_start=0.008, tau_end=0.001, lr_init=0.05)
        o2 = compute_obj(a2)
        if o2 < best_score:
            best_score = o2
            best_seq = a2.copy()
            print(f"  Phase2 log: obj={best_score:.8f} **")

    # === Phase 3: LP direction finding ===
    print("\n=== Phase 3: LP direction ===")
    remaining = deadline - time.monotonic()
    b3 = min(remaining * 0.12, 100.0)
    if b3 > 15:
        a3 = lp_line_search(best_seq, b3, seed=seed)
        o3 = compute_obj(a3)
        if o3 < best_score:
            best_score = o3
            best_seq = a3.copy()
            print(f"  Phase3 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 LP: obj={o3:.8f}")

    # === Phase 4: Multi-length exploration ===
    print("\n=== Phase 4: Multi-length ===")
    n_current = len(best_seq)
    n_candidates = sorted(set([
        max(200, n_current - 300), max(200, n_current - 150),
        n_current + 150, n_current + 300, n_current + 500,
        400, 600, 800, 1000, 1200, 1600, 2000
    ]))
    n_candidates = [n for n in n_candidates if abs(n - n_current) > 80 and 200 <= n <= 2000]

    for n_try in n_candidates[:7]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(remaining * 0.15, 80.0)
        if alloc < 10:
            break
        print(f"  n={n_try} (alloc={alloc:.0f}s, remaining={remaining:.0f}s)")
        a_try = try_length(best_seq, n_try, alloc, seed=seed + n_try)
        obj = compute_obj(a_try)
        if obj < best_score:
            best_score = obj
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj:.8f}")

    # === Phase 5: Perturbation restarts ===
    print("\n=== Phase 5: Perturbations ===")
    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        scale = 0.008 + 0.008 * r
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_p = a_pert.sum()
        if s_p < 1e-10:
            continue
        a_pert = a_pert / s_p

        alloc = min(remaining * 0.12, 50.0)
        if alloc > 8:
            a_pert = adam_optimize(a_pert, alloc, tau_start=0.005, tau_end=0.0005, lr=0.3 / len(a_pert))
            obj = compute_obj(a_pert)
            if obj < best_score:
                best_score = obj
                best_seq = a_pert.copy()
                print(f"  Pert{r} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Pert{r}: obj={obj:.8f}")

    # === Phase 6: Final intensive polish ===
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        b6 = min(remaining * 0.4, 100.0)
        a6 = adam_optimize(best_seq, b6, tau_start=0.003, tau_end=0.0003, lr=0.15 / len(best_seq))
        o6 = compute_obj(a6)
        if o6 < best_score:
            best_score = o6
            best_seq = a6.copy()
            print(f"  Final Adam: obj={best_score:.8f} **")
        else:
            print(f"  Final Adam: obj={o6:.8f}")

    # Subgradient final
    remaining = deadline - time.monotonic()
    if remaining > 15:
        b7 = min(remaining * 0.4, 60.0)
        a7 = subgradient_refine(best_seq, b7, K=3)
        o7 = compute_obj(a7)
        if o7 < best_score:
            best_score = o7
            best_seq = a7.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # LP final
    remaining = deadline - time.monotonic()
    if remaining > 15:
        b8 = min(remaining * 0.4, 40.0)
        a8 = lp_line_search(best_seq, b8, seed=999)
        o8 = compute_obj(a8)
        if o8 < best_score:
            best_score = o8
            best_seq = a8.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # === Final validation ===
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
