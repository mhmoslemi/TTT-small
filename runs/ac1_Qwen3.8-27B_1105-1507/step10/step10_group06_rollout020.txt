    n = len(a)

    # For each active k, gradient is 2*[a_{k-i}] (with bounds)

            j = k - i

                g[i] += a[j]

    a_new = np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based alternating minimization with adaptive line search,
exponentiated gradient polishing, active-set exploitation, and multi-start
diversification with length exploration to escape local minima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_cross_conv(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_lp_best_response(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    import cvxpy as cp
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def line_search_blend(a, g, best_obj, n_evals=12):
    """
    Find optimal blending factor t in (0,1) for (1-t)*a + t*g.
    Uses golden-section search followed by parabolic refinement.
    """
    n = len(a)
    best_t = 0.0
    best_f = best_obj

    # Golden section search on [0.01, 1.0]
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~0.618
    lo, hi = 0.01, 1.0

    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = compute_obj((1 - c) * a + c * g)
    fd = compute_obj((1 - d) * a + d * g)

    for _ in range(n_evals - 2):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = compute_obj((1 - c) * a + c * g)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = compute_obj((1 - d) * a + d * g)

    # Evaluate the best point
    t_candidate = (lo + hi) / 2
    f_candidate = compute_obj((1 - t_candidate) * a + t_candidate * g)

    # Try a few additional points around the minimum
    for t_try in [t_candidate, t_candidate * 0.8, t_candidate * 1.25,
                  t_candidate * 0.5, t_candidate * 2.0, 0.05, 0.1, 0.3, 0.5, 1.0]:
        t_try = min(max(t_try, 0.005), 1.0)
        a_try = (1 - t_try) * a + t_try * g
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        f_try = compute_obj(a_try)
        if f_try < best_f:
            best_f = f_try
            best_t = t_try

    return best_t, best_f

def eg_polish(a, time_budget, tau_init=50.0, tau_max=5000.0, verbose=False):
    """
    Exponentiated gradient (mirror descent on simplex) with adaptive
    smoothness parameter for polishing the solution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = conv.max()

        # Compute smooth approximation of subgradient
        exp_vals = np.exp(tau_init * (conv - max_conv))
        sum_exp = exp_vals.sum()
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient: grad_i = 2 * sum_k p_k * a_{k-i}
        grad = np.zeros(n)
        for i in range(n):
            lo_k = i
            hi_k = min(i + n - 1, 2 * n - 2)
            if lo_k < len(p):
                j_lo = 0
                j_hi = min(n - 1, hi_k - i + n - 1)
                # (a*a)_k grad wrt a_i is a_{k-i} (with bounds)
                # sum_k p_k * a_{k-i} for valid k,i
                indices = np.arange(lo_k, hi_k + 1)
                a_indices = indices - i
                valid = (a_indices >= 0) & (a_indices < n)
                if valid.any():
                    grad[i] = np.sum(p[indices[valid]] * a[a_indices[valid]])
        grad *= 2.0

        # Center gradient for simplex
        grad_centered = grad - np.sum(a * grad)

        # Adaptive step size
        g_norm = np.linalg.norm(grad_centered)
        if g_norm > 1e-15:
            lr = min(1.0 / (g_norm + 0.1), lr * 1.01)
            lr = max(lr, 0.01)

        # Exponentiated gradient update
        a_new = a * np.exp(-lr * grad_centered)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new
            stall += 1
            if stall > 50:
                a = best_a.copy()
                lr *= 0.5
                stall = 0

        # Increase smoothness
        tau_init = min(tau_init * 1.05, tau_max)
        lr *= 0.999

        if verbose and (time.monotonic() - start) % 10 < 0.1:
            print(f"    EG: obj={best_obj:.10f}, tau={tau_init:.1f}, lr={lr:.4f}")

    return best_a

    start = time.monotonic()
    lr = 0.1
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()
        # Active set: lags where conv is within 1% of max
        active = np.where(conv > max_val * 0.999)[0]

        # Compute subgradient: average of gradients over active set
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad *= 2.0 / len(active)

        # Center for simplex
        grad_centered = grad - np.sum(a * grad)

        # Backtracking line search
        a_try = a.copy()
        f_try = best_obj
        lr_try = lr
        improved = False
        for _ in range(20):
            a_try = a * np.exp(-lr_try * grad_centered)
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                lr_try *= 0.5
                continue
            a_try = a_try / s_try
            f_try = compute_obj(a_try)
            if f_try < best_obj - 1e-12:
                improved = True
                break
            lr_try *= 0.5

        if improved:
            best_obj = f_try
            best_a = a_try.copy()
            a = a_try
            lr = lr_try * 1.2
            stall = 0
        else:
            stall += 1
            lr *= 0.7
            if stall > 30:
                break

        if verbose and (time.monotonic() - start) % 5 < 0.1:
            print(f"    SG: obj={best_obj:.10f}, lr={lr:.4f}, |active|={len(active)}")

    return best_a

def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(10, n // 4)
    for k in range(1, n_modes + 1):
        freq = 2 * np.pi * k / n
        amp = strength * rng.uniform(0.3, 1.0) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def random_dirichlet(n, alpha=1.0, rng=None):
    """Generate random Dirichlet-distributed sequence."""
    if rng is None:
        rng = np.random.default_rng()
    return rng.dirichlet(alpha * np.ones(n))

def lp_alternate(a, time_budget, seed=42, verbose=False, max_iters=200):
    """
    LP-based alternating minimization with line search.
    Primary optimization engine.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    stall = 0
    consecutive_no_improve = 0

    while time.monotonic() - start < time_budget and iter_count < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_lp_best_response(a)
        lp_time = time.monotonic() - t0
        iter_count += 1

        if g_lp is None:
            if verbose:
                print(f"    LP failed at iter {iter_count}, breaking.")
            break

        # Line search for optimal blending
        best_t, best_f = line_search_blend(a, g_lp, best_obj, n_evals=10)

        if best_f < best_obj - 1e-13:
            best_obj = best_f
            a_new = (1 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                best_a = a_new.copy()
                a = a_new
            consecutive_no_improve = 0
            if verbose:
                print(f"    LP iter {iter_count}: obj={best_obj:.10f}, "
                      f"t={best_t:.4f}, M={M_lp:.6f}, time={lp_time:.2f}s")
        else:
            consecutive_no_improve += 1
            if verbose:
                print(f"    LP iter {iter_count}: no improvement "
                      f"(best_f={best_f:.10f} >= best_obj={best_obj:.10f})")

            # If stuck, try perturbation
            if consecutive_no_improve >= 5:
                strength = 0.005 + 0.003 * consecutive_no_improve
                a = fourier_perturb(best_a, rng, strength=strength)
                a = a / a.sum()
                consecutive_no_improve = 0
            elif consecutive_no_improve >= 3:
                # Try smaller step toward LP solution
                a_small = (1 - 0.05) * a + 0.05 * g_lp
                a_small = np.maximum(a_small, 0)
                a_small = a_small / a_small.sum()
                f_small = compute_obj(a_small)
                if f_small < best_obj:
                    best_obj = f_small
                    best_a = a_small.copy()
                    a = a_small
            if consecutive_no_improve >= 15:
                break

        # Periodic gradient polish step
        if iter_count % 10 == 0 and time.monotonic() - start < time_budget - 30:
            a_polished = eg_polish(a, time_budget=min(5.0, (time_budget - (time.monotonic() - start)) * 0.1),
                                    tau_init=100.0, tau_max=2000.0, verbose=False)
            obj_polished = compute_obj(a_polished)
            if obj_polished < best_obj:
                best_obj = obj_polished
                best_a = a_polished.copy()
                a = a_polished
                if verbose:
                    print(f"    [polish] obj={best_obj:.10f}")

    if verbose:
        print(f"    LP alternating done: {iter_count} iters, "
              f"best_obj={best_obj:.10f}, time={time.monotonic()-start:.1f}s")

    return best_a

def resample_to_length(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(x_old, a, bc_type='natural')
        a_new = cs(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

def try_single_start(a_init, time_budget, seed, verbose=False):
    """Run full optimization pipeline from a single starting point."""
    n = len(a_init)
    a_init = np.maximum(a_init, 0).astype(np.float64)
    s = a_init.sum()
    if s < 1e-10:
        return None, np.inf
    a_init = a_init / s

    obj_init = compute_obj(a_init)
    best_a = a_init.copy()
    best_obj = obj_init
    start = time.monotonic()

    # Stage 1: LP alternation (70% of budget)
    if time_budget > 30:
        budget_lp = time_budget * 0.70
        a_lp = lp_alternate(a_init, time_budget=budget_lp, seed=seed, verbose=verbose)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Stage 2: Subgradient polish (15% of budget)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        budget_sg = min(remaining * 0.25, time_budget * 0.15)
        a_sg = subgradient_polish(best_a, time_budget=budget_sg, verbose=verbose)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()

    # Stage 3: EG polish (10% of budget)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        budget_eg = min(remaining * 0.3, time_budget * 0.10)
        a_eg = eg_polish(best_a, time_budget=budget_eg, tau_init=200.0, tau_max=10000.0, verbose=verbose)
        obj_eg = compute_obj(a_eg)
        if obj_eg < best_obj:
            best_obj = obj_eg
            best_a = a_eg.copy()

    # Stage 4: Final LP polish (remaining)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        a_final = lp_alternate(best_a, time_budget=remaining * 0.8, seed=seed + 1, verbose=verbose)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()

    return best_a, best_obj

def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-start LP alternation with gradient polishing
    and length exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0  # 30s safety margin

    rng = np.random.default_rng(seed)

    # Load parent sequence if available
    parent_a = None
    n_init = 800
    try:
        a_parent = np.asarray(height_sequence_1, dtype=np.float64)
        a_parent = np.maximum(a_parent, 0)
        if a_parent.size > 10 and a_parent.sum() > 0.01:
            parent_a = a_parent / a_parent.sum()
            n_init = parent_a.size
            print(f"Parent loaded: n={n_init}, obj={compute_obj(parent_a):.10f}")
    except Exception:
        print("No valid parent, using uniform init.")

    if parent_a is None:
        parent_a = np.ones(n_init) / n_init
        print(f"Using uniform init: n={n_init}")

    best_a = parent_a.copy()
    best_obj = compute_obj(parent_a)
    print(f"Initial best: n={len(best_a)}, obj={best_obj:.10f}")

    # === Phase 1: Primary optimization from parent (55% of time) ===
    print(f"\n=== Phase 1: Primary LP optimization (n={n_init}) ===")
    p1_budget = (deadline - time.monotonic()) * 0.55
    p1_budget = max(100, min(p1_budget, 600))
    a_p1, obj_p1 = try_single_start(parent_a, p1_budget, seed=seed, verbose=True)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1.copy()
        print(f"  Phase 1 result: obj={best_obj:.10f} **")
    else:
        print(f"  Phase 1 result: obj={obj_p1:.10f} (no improvement)")

    # === Phase 2: Fourier-perturbed restart (20% of time) ===
    print(f"\n=== Phase 2: Perturbed restart ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = remaining * 0.20
        p2_budget = max(50, min(p2_budget, 250))
        a_pert = fourier_perturb(best_a, rng, strength=0.03)
        a_p2, obj_p2 = try_single_start(a_pert, p2_budget, seed=seed + 100, verbose=True)
        if obj_p2 < best_obj:
            best_obj = obj_p2
            best_a = a_p2.copy()
            print(f"  Phase 2 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 2 result: obj={obj_p2:.10f}")

    # === Phase 3: Length exploration (15% of time) ===
    print(f"\n=== Phase 3: Length exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([n_best - 5, n_best + 5, n_best - 10, n_best + 10,
                            n_best - 20, n_best + 20, n_best - 50, n_best + 50]))
    n_options = [n for n in n_options if 200 <= n <= 2000]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        alloc = min(remaining * 0.2, 120.0)
        if alloc < 40:
            break

        a_try = resample_to_length(best_a, n_try)
        print(f"  Trying n={n_try} (budget={alloc:.0f}s)...")
        a_res, obj_res = try_single_start(a_try, alloc, seed=n_try, verbose=False)
        if obj_res < best_obj:
            best_obj = obj_res
            best_a = a_res.copy()
            print(f"  ** n={n_try}: obj={best_obj:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_res:.10f}")

    # === Phase 4: Random diverse starts (10% of time) ===
    print(f"\n=== Phase 4: Diverse random starts ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_starts = min(3, int(remaining // 80))
        for i in range(n_starts):
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            alloc = min(remaining * 0.3, 100.0)
            if alloc < 30:
                break

            # Different random initializations
            if i == 0:
                a_rand = random_dirichlet(n_best, alpha=0.5, rng=rng)
            elif i == 1:
                # Bump function
                x = np.linspace(0, 1, n_best)
                a_rand = np.exp(-50 * (x - 0.5)**2)
                a_rand = a_rand / a_rand.sum()
            else:
                # Two bumps
                x = np.linspace(0, 1, n_best)
                a_rand = np.exp(-100 * (x - 0.3)**2) + np.exp(-100 * (x - 0.7)**2)
                a_rand = a_rand / a_rand.sum()

            print(f"  Random start {i+1} (n={n_best}, budget={alloc:.0f}s)...")
            a_res, obj_res = try_single_start(a_rand, alloc, seed=seed + 5000 + i, verbose=False)
            if obj_res < best_obj:
                best_obj = obj_res
                best_a = a_res.copy()
                print(f"  ** Random {i+1}: obj={best_obj:.10f} **")
            else:
                print(f"  Random {i+1}: obj={obj_res:.10f}")

    # === Phase 5: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 5: Final intensive polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Short LP pass
        alloc_lp = min(remaining * 0.5, 80.0)
        if alloc_lp > 15:
            a_final = lp_alternate(best_a, time_budget=alloc_lp, seed=seed + 9999, verbose=True)
            obj_final = compute_obj(a_final)
            if obj_final < best_obj:
                best_obj = obj_final
                best_a = a_final.copy()
                print(f"  Final LP: obj={best_obj:.10f} **")

        # Subgradient polish
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc_sg = min(remaining * 0.6, 40.0)
            if alloc_sg > 5:
                a_final = subgradient_polish(best_a, time_budget=alloc_sg, verbose=True)
                obj_final = compute_obj(a_final)
                if obj_final < best_obj:
                    best_obj = obj_final
                    best_a = a_final.copy()
                    print(f"  Final SG: obj={best_obj:.10f} **")

        # EG polish
        remaining = deadline - time.monotonic()
        if remaining > 5:
            alloc_eg = min(remaining * 0.7, 20.0)
            if alloc_eg > 3:
                a_final = eg_polish(best_a, time_budget=alloc_eg, tau_init=500.0, tau_max=20000.0, verbose=True)
                obj_final = compute_obj(a_final)
                if obj_final < best_obj:
                    best_obj = obj_final
                    best_a = a_final.copy()
                    print(f"  Final EG: obj={best_obj:.10f} **")

    # === Final validation ===
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_obj:
        best_obj = val
        print(f"  evaluate_sequence gave better: {val:.10f}")
    elif not np.isfinite(val):
        print(f"  WARNING: evaluate_sequence returned {val}, using fallback")
        result = [1.0] * 100
        best_obj = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
