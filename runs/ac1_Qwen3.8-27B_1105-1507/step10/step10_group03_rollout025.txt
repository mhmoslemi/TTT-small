import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft

    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""

    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a

    A = make_conv_matrix(a)  # Will need to update as a changes

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    n = len(a)

    s = float(np.sum(a))
    if s < 0.01:
        return np.inf

def make_conv_matrix(a):

def lp_solve(a, solver_list=None):

    A = make_conv_matrix(a)

def smoothed_grad(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Returns (gradient of length n, exact_max of conv).
    """
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def gd_polish(a, time_budget, tau_levels=None, verbose=False):
    """Nesterov accelerated gradient descent with calibrated tau schedule."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 15, 50, 150, 400, 1000, 2500, 5000]

    beta = 0.9
    it = 0
    best_it = 0

    for tau in tau_levels:
        lr = 3.0 / np.sqrt(n)
        v = np.zeros(n)
        it_level = 0
        max_iters = max(500, int(200000.0 / n))

        while time.monotonic() - start < time_budget:
            if it_level >= max_iters:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-14:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it
            a = a_new
            it += 1
            it_level += 1
            lr = max(1e-8, lr * 0.99997)

            if it - best_it > max(3000, 5 * max_iters):
                a = best_a.copy()
                v = np.zeros(n)
                lr = 3.0 / np.sqrt(n)
                best_it = it

        if verbose:
            print(f"    GD tau={tau:5d}: obj={best_obj:.10f}, iters={it_level}")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_shape(n, mode, param=1.0):
    """Create parametric shape of length n."""
    x = np.arange(n, dtype=np.float64)
    if mode == 0:  # Uniform
        a = np.ones(n)
    elif mode == 1:  # Sine bump: sin(pi*(k+1)/n)^p
        a = np.sin(np.pi * (x + 1) / n) ** param
        a = np.maximum(a, 1e-6)
    elif mode == 2:  # Cosine: (1+cos(2*pi*k/n))^p
        a = (1.0 + np.cos(2 * np.pi * x / n)) ** param
        a = np.maximum(a, 1e-6)
    elif mode == 3:  # Gaussian center
        center = (n - 1) / 2.0
        width = n / (2.0 * param)
        a = np.exp(-((x - center) ** 2) / (2.0 * width ** 2))
    elif mode == 4:  # Polynomial bump: (1 - ((2k/n)-1)^2)^p
        u = 2.0 * x / (n - 1) - 1.0
        a = np.maximum(1.0 - u ** 2, 0.0) ** param
        a = np.maximum(a, 1e-6)
    elif mode == 5:  # Power law: (k+1)^(-alpha) * (n-k)^(-alpha)
        alpha = 0.3 * param
        a = (x + 1) ** (-alpha) * (n - x) ** (-alpha)
        a = np.maximum(a, 1e-6)
    elif mode == 6:  # Triangle
        mid = n // 2
        a = np.minimum(x + 1, n - x)
        a = np.maximum(a, 1e-3)
    elif mode == 7:  # Semicircle
        u = 2.0 * x / (n - 1) - 1.0
        a = np.sqrt(np.maximum(1.0 - u ** 2, 0.0))
        a = np.maximum(a, 1e-3)
    elif mode == 8:  # Flat spectrum (random phase)
        rng = np.random.default_rng(mode * 1000 + n)
        N = 1
        while N < 2 * n - 1:
            N <<= 1
        # Target: flat power spectrum
        target = np.ones(N) / np.sqrt(N)
        phases = rng.uniform(0, 2 * np.pi, N)
        spec = target * np.exp(1j * phases)
        # Make real and symmetric
        spec[0] = np.abs(spec[0])
        if N % 2 == 0:
            spec[N // 2] = np.abs(spec[N // 2])
        spec = spec + np.conj(spec[::-1])
        a = ifft(spec).real[:n]
        a = np.maximum(a, 1e-6)
    elif mode == 9:  # Bi-modal
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // (4 * param))
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    else:
        a = np.ones(n)

    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def explore_shapes(n, time_per_shape=5.0):
    """Explore parametric shapes and return the best few."""
    best_candidates = []
    configs = [
        (0, 1.0), (1, 0.5), (1, 1.0), (1, 1.5), (1, 2.0), (1, 3.0), (1, 5.0),
        (2, 0.5), (2, 1.0), (2, 2.0), (2, 4.0),
        (3, 0.5), (3, 1.0), (3, 2.0), (3, 3.0),
        (4, 0.5), (4, 1.0), (4, 2.0), (4, 4.0), (4, 8.0),
        (5, 0.5), (5, 1.0), (5, 2.0),
        (6, 1.0), (7, 1.0), (8, 1.0), (9, 1.0), (9, 2.0),
    ]

    for mode, param in configs:
        a = make_shape(n, mode, param)
        obj = compute_obj(a)
        best_candidates.append((obj, a.copy(), mode, param))

    best_candidates.sort(key=lambda x: x[0])
    return best_candidates[:5]


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-n exploration with LP fixed-point, shape parameterization,
    over-relaxed fixed-point iteration, and FFT-based gradient polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        parent = None

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.10f}")

    # Determine which n values to try
    n_parent = len(best_seq)
    n_options = sorted(set([
        n_parent,
        int(n_parent * 1.25),
        int(n_parent * 1.5),
        int(n_parent * 2),
        1000, 1200, 1600, 2000, 2500, 3000, 4000,
    ]))
    n_options = [n for n in n_options if 300 <= n <= 5000]
    # Sort: parent first, then larger ones (more promising for lower objective)
    n_options = sorted(n_options, key=lambda x: (-x if x != n_parent else 0))
    n_options.sort(key=lambda x: (0 if x == n_parent else 1, -x))
    n_options = n_options[:7]  # Limit to 7 sizes

    print(f"n options: {n_options}")

    results = []

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            print(f"Time running out, stopping n exploration. remaining={remaining:.1f}s")
            break

        print(f"\n{'='*60}")
        print(f"Exploring n={n_try} (remaining={remaining:.0f}s)")
        print(f"{'='*60}")

        # Allocate time for this n
        alloc = min(remaining * 0.6, 200.0)
        if n_try > 2000:
            alloc = min(alloc, 150.0)

        # Phase 1: Shape exploration (quick)
        t_shape = min(alloc * 0.3, 30.0)
        if t_shape > 5:
            candidates = explore_shapes(n_try)
            print(f"  Best shapes: obj={candidates[0][0]:.10f} (mode={candidates[0][2]}, param={candidates[0][3]})")

            # Quick GD on top shape
            a_shape = candidates[0][1]
            a_shape = gd_polish(a_shape, time_budget=min(t_shape * 0.5, 10.0),
                                tau_levels=[10, 50, 200])
            obj_shape = compute_obj(a_shape)
            print(f"  Shape+GD: obj={obj_shape:.10f}")

            # Try parent resampled
            a_parent = resample(best_seq, n_try)
            obj_parent = compute_obj(a_parent)
            print(f"  Parent resampled: obj={obj_parent:.10f}")

            # Pick best initialization
            inits = [(obj_shape, a_shape), (obj_parent, a_parent)]
            for obj_c, a_c, m, p in candidates[:3]:
                inits.append((obj_c, a_c))
            inits.sort(key=lambda x: x[0])
            best_init_obj, best_init = inits[0]
            print(f"  Best init for n={n_try}: obj={best_init_obj:.10f}")
        else:
            best_init = resample(best_seq, n_try)
            best_init_obj = compute_obj(best_init)

        if best_score < best_init_obj - 0.01:
            # This n is unlikely to improve, spend less time
            alloc = min(alloc * 0.3, 30.0)
            print(f"  Skipping (init {best_init_obj:.6f} >> best {best_score:.6f})")
            continue

        # Phase 2: LP fixed-point (main workhorse)
        t_lp = alloc * 0.5
        print(f"  Running LP fixed-point (budget={t_lp:.0f}s)")
        a_lp, obj_lp = lp_fixed_point(best_init, time_budget=t_lp, verbose=True,
                                       seed=seed + n_try)
        print(f"  LP result: obj={obj_lp:.10f}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** NEW BEST: n={n_try}, obj={best_score:.10f} **")

        # Phase 3: GD polish on LP result
        remaining = deadline - time.monotonic()
        t_gd = min(remaining * 0.15, 40.0)
        if t_gd > 10:
            a_gd = gd_polish(a_lp, time_budget=t_gd,
                             tau_levels=[20, 100, 400, 1200, 3000])
            obj_gd = compute_obj(a_gd)
            print(f"  GD polish: obj={obj_gd:.10f}")
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
                print(f"  ** NEW BEST: n={n_try}, obj={best_score:.10f} **")

        # Phase 4: Second LP pass on GD result (if improved)
        remaining = deadline - time.monotonic()
        if remaining > 20 and obj_gd < best_init_obj:
            t_lp2 = min(remaining * 0.2, 40.0)
            if t_lp2 > 10:
                a_lp2, obj_lp2 = lp_fixed_point(a_gd, time_budget=t_lp2,
                                                 t_values=[0.8, 1.0, 1.2, 1.0, 0.8, 1.0, 1.1, 0.9],
                                                 verbose=False, seed=seed + n_try + 999)
                print(f"  LP2: obj={obj_lp2:.10f}")
                if obj_lp2 < best_score:
                    best_score = obj_lp2
                    best_seq = a_lp2.copy()
                    print(f"  ** NEW BEST: n={n_try}, obj={best_score:.10f} **")

        results.append((best_score if best_seq is not None else np.inf, n_try))

    # Final intensive polish on the overall best
    print(f"\n{'='*60}")
    print(f"FINAL POLISH on best (n={len(best_seq)}, obj={best_score:.10f})")
    print(f"{'='*60}")
    remaining = deadline - time.monotonic()

    if remaining > 20:
        # High-tau GD
        t_f1 = min(remaining * 0.3, 60.0)
        if t_f1 > 10:
            a_f1 = gd_polish(best_seq, time_budget=t_f1,
                             tau_levels=[50, 200, 800, 2000, 5000, 10000])
            obj_f1 = compute_obj(a_f1)
            print(f"  Final GD: obj={obj_f1:.10f}")
            if obj_f1 < best_score:
                best_score = obj_f1
                best_seq = a_f1.copy()
                print(f"  ** NEW BEST: obj={best_score:.10f} **")

        remaining = deadline - time.monotonic()

        # Final LP
        if remaining > 15 and len(best_seq) <= 3500:
            t_f2 = min(remaining * 0.4, 50.0)
            if t_f2 > 10:
                a_f2, obj_f2 = lp_fixed_point(best_seq, time_budget=t_f2,
                                               t_values=[1.0, 0.8, 1.0, 1.2, 0.9, 1.0, 1.1],
                                               verbose=True, seed=7777)
                print(f"  Final LP: obj={obj_f2:.10f}")
                if obj_f2 < best_score:
                    best_score = obj_f2
                    best_seq = a_f2.copy()
                    print(f"  ** NEW BEST: obj={best_score:.10f} **")

        remaining = deadline - time.monotonic()

        # Ultra-high tau GD
        if remaining > 15:
            t_f3 = min(remaining * 0.4, 40.0)
            if t_f3 > 10:
                a_f3 = gd_polish(best_seq, time_budget=t_f3,
                                 tau_levels=[500, 2000, 5000, 10000, 20000])
                obj_f3 = compute_obj(a_f3)
                print(f"  Ultra GD: obj={obj_f3:.10f}")
                if obj_f3 < best_score:
                    best_score = obj_f3
                    best_seq = a_f3.copy()
                    print(f"  ** NEW BEST: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
