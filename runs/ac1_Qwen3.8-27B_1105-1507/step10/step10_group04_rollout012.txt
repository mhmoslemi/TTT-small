import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    return 2.0 * n * float(np.max(compute_conv(a)))

def smoothed_obj_grad(a, tau):
    """Compute smoothed objective and gradient using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    # Numerical stability
    max_shifted = float(np.max(shifted))
    exp_vals = np.exp(shifted - max_shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    obj = max_c + max_shifted / tau + np.log(sum_exp) / tau
    
    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j] (with bounds)
    # This is 2 * cross-correlation of p and a
    # cross_correlation[p, a][j] = sum_k p[k] * a[k-j]
    # Compute via FFT
    N = 1
    while N < 2*n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    # cross-correlation: ifft(conj(P) * A) gives sum_k conj(p[k]) * a[k-j] = sum_k p[k] * a[k-j] (for real)
    cross = ifft(np.conj(P) * A).real[:n]
    grad = 2.0 * cross
    return obj, grad

def proj_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u / np.arange(1, n+1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def lp_solve(a):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_mat(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val
        except:
            continue
    return None
