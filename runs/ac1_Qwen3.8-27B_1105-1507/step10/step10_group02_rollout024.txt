        it += 1
        if verbose and it % 50 == 0:
            print(f"    LP-GD it{it}: obj={best_obj:.10f}, stall={stall}, t={elapsed:.1f}s")

    return best_a


def coordinate_descent(a, time_budget=300.0, seed=42, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1
        if sweep % 5 == 1:
            conv = compute_conv(a)
            obj_check = float(np.max(conv))
            if obj_check < best_obj:
                best_obj = obj_check
                best_a = a.copy()

        M = float(np.max(conv))
        active = get_active_set(conv, rel_eps=1e-2, max_size=500)
        importance = np.zeros(n)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            importance[lo:hi+1] += conv[k]
        if importance.max() > 0:
            probs = importance / importance.sum()
        else:
            probs = np.ones(n) / n
        order = rng.choice(n, size=n, replace=False, p=probs)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            v = a[i]
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks]
            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]
            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.5, v, v * 1.5, v * 2.0]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            def f_1d(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                return max_val

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + 2.0 * scale
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-8
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)
            for _ in range(40):
                if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            obj_now = float(np.max(conv))
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()

        if verbose and sweep % 20 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, t={time.monotonic()-start:.1f}s")

    return best_a


def polyak_subgradient(a, time_budget=100.0, seed=42, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    best_obj = float(np.max(conv))
    best_a = a.copy()
    lower_bound = 1.0 / n
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    momentum = np.zeros(n)
    beta = 0.9
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        conv = compute_conv(a)
        M = float(np.max(conv))
        if M < best_obj:
            best_obj = M
            best_a = a.copy()

        gap = M - lower_bound
        if gap < 1e-12:
            break
        active, _ = get_active_set(conv, rel_eps=1e-3, max_size=100)
        grad = build_conv_gradient(a, active)
        grad = grad - grad.mean()
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-12:
            break
        lr = 2.0 * gap / (gnorm * gnorm + 1e-12)
        lr = min(lr, 0.5)

        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s_new
        obj_new = float(np.max(compute_conv(a_new)))
        if obj_new < best_obj * 1.01:
            a = a_new
        else:
            a = best_a.copy()
            momentum = np.zeros(n)
            pert_state = rng.normal(0, gap * 0.5, n)
            a = a * np.exp(pert_state)
            a = np.maximum(a, 0)
            a = a / a.sum()

        it += 1
        if verbose and it % 100 == 0:
            print(f"    PGD it{it}: obj={best_obj:.10f}, gap={gap:.6e}, t={elapsed:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [max(1, n // 10), max(1, n // 10)]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2.0 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1.0 - x**2 + 1e-8)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 10, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        a = np.zeros(n)
        a[0] = 1.0
        a[n-1] = 1.0
        a[1:-1] = 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def lp_polish(a, time_budget=60.0, seed=42, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        conv = compute_conv(a)
        M = float(np.max(conv))
        active, _ = get_active_set(conv, rel_eps=1e-4, max_size=150)

        g = cp.Variable(n, nonneg=True)
        Mv = cp.Variable()
        A = np.zeros((len(active), n))
        c_act = conv[active]
        for idx, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                A[idx, i] = a[k - i]
        constraints = [A @ g <= Mv, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(Mv), constraints)
        solved = False
        for solver in [cp.CLARABEL, cp.SCS, cp.MOSEK]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                        solved = True
                        break
            except Exception:
                continue
        if not solved:
            break

        best_t_obj = best_obj
        best_t_a = best_a.copy()
        for t_val in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t_val) * a + t_val * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 5:
                pert = rng.normal(0, 0.01, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if time.monotonic() - start > time_budget * 0.85:
                    break

    return best_a

    return result
