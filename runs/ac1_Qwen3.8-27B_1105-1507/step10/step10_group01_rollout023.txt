Uses a combination of:

- Coordinate descent polishing (exploits piecewise-linear structure)
- LP-based direction finding when stalled
- Multi-n exploration with diverse restarts

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)

        return np.array([])

    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)

def adaptive_smoothed_gradient(a, tau=None, top_k=50):
    """
    Compute gradient of smoothed max conv(a,a) with adaptive temperature.
    Only focuses on top-k convolution values for efficiency and focus.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    if tau is None:
        # Adaptive: use the gap between max and second max
        sorted_conv = np.sort(conv)[::-1]
        if len(sorted_conv) > 1:
            gap = max_conv - sorted_conv[1]
            tau = max(50.0, 10.0 / (gap + 1e-10))
        else:
            tau = 100.0
        tau = min(tau, 5000.0)
    
    # Use top-k values only
    k = min(top_k, len(conv))
    top_idx = np.argpartition(conv, -k)[-k:]
    top_vals = conv[top_idx]
    
    # Log-sum-exp on top-k
    max_top = float(np.max(top_vals))
    exp_vals = np.exp(tau * (top_vals - max_top))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = np.zeros(len(conv))
        p[top_idx] = exp_vals / sum_exp

    N = 1
    while N < 3 * n:

    P = fft(p, N)
    A = fft(a, N)
    grad_full = ifft(P * np.conj(A)).real
    return 2.0 * grad_full[:n], max_conv

        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        t_k = t_new

            lr *= 0.9
            stall = 0

                y = a.copy()
                t_k = 1.0
                lr = base_lr
                best_it = it
        
        a = a_new
        it += 1
        
        if verbose and it % 500 == 0:
            print(f"      FISTA it={it}, obj={best_obj:.8f}, lr={lr:.6f}, tau_top_k={top_k}")
    
    return best_a


def coordinate_descent(a, max_sweeps=20, verbose=False):
    """
    Coordinate descent exploiting the piecewise-linear structure.
    For each coordinate i, the objective (max conv) is piecewise-linear in a[i].
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    conv = compute_conv(a)
    best_obj = compute_obj(a)
    
    for sweep in range(max_sweeps):
        improved = False
        max_improvement = 0.0
        
        for i in range(n):
            old_val = a[i]
            if old_val < 1e-12:
                continue
            
            # Compute slope: d(conv[k])/d(a[i]) = 2*a[k-i] for valid k
            slope = np.zeros(2 * n - 1)
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            slope[k_start:k_end + 1] = 2.0 * a[0:k_end - k_start + 1]
            
            # Find best delta
            current_max = float(np.max(conv))
            best_delta = 0.0
            best_max = current_max
            
            # Generate candidate deltas
            # Heuristic: try to reduce the maximum
            k_max = int(np.argmax(conv))
            if abs(slope[k_max]) > 1e-12:
                # Delta that would reduce conv[k_max]
                suggested = -conv[k_max] / slope[k_max] * 0.3
                suggested = max(suggested, -old_val * 0.9)
                suggested = min(suggested, old_val)  # Don't increase too much
            else:
                suggested = 0.0
            
            candidates = np.array([0.0,
                                   suggested,
                                   suggested * 0.5,
                                   suggested * 0.1,
                                   -old_val * 0.01,
                                   -old_val * 0.1,
                                   -old_val * 0.3,
                                   old_val * 0.1])
            
            for delta in candidates:
                new_val = old_val + delta
                if new_val < 0:
                    continue
                new_conv = conv + delta * slope
                new_max = float(np.max(new_conv))
                if new_max < best_max - 1e-14:
                    best_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_max < current_max - 1e-14:
                a[i] = old_val + best_delta
                conv = conv + best_delta * slope
                improved = True
                improvement = current_max - best_max
                max_improvement = max(max_improvement, improvement)
        
        # Check for convergence
        if not improved:
            break
        
        if verbose and sweep % 5 == 0:
            new_obj = compute_obj(a)
            print(f"      CD sweep {sweep}: obj={new_obj:.8f}, max_impr={max_improvement:.2e}")
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build convolution matrix
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a_norm[diff[valid]]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_line_search_update(a, time_budget=50.0, verbose=False):
    """LP-based direction finding with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)

            break

        best_t = 0.0
        best_obj_try = best_obj

            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)

                a_try = a_try / s_try

                if obj_try < best_obj_try - 1e-13:
                    best_obj_try = obj_try
                    best_t = t

            a_try = (1 - best_t) * a + best_t * g

            if consecutive_fail >= 3:
                # Perturb
                a = a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                consecutive_fail = 0

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 6):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Combines FISTA with adaptive top-k smoothing, exact coordinate descent exploiting
piecewise-linear structure, LP direction finding, and multi-n exploration.
Key difference from previous: coordinate descent as primary polisher,
adaptive top-k gradient focusing, and broader n search with structural priors.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def adaptive_smoothed_gradient(a, tau=None, top_k=30):
    """
    Compute gradient of smoothed max conv(a,a) focusing on top-k values.
    Returns (gradient, max_conv_value).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    if tau is None:
        sorted_conv = np.sort(conv)[::-1]
        if len(sorted_conv) > 1:
            gap = max_conv - sorted_conv[1]
            tau = max(50.0, 8.0 / (gap + 1e-10))
        else:
            tau = 100.0
        tau = min(tau, 5000.0)

    k = min(top_k, len(conv))
    top_idx = np.argpartition(conv, -k)[-k:]
    top_vals = conv[top_idx]

    max_top = float(np.max(top_vals))
    exp_vals = np.exp(tau * (top_vals - max_top))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = np.zeros(len(conv))
        p[top_idx] = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad_full = ifft(P * np.conj(A)).real
    return 2.0 * grad_full[:n], max_conv


def fista_optimize(a, time_budget=100.0, verbose=False, top_k=30, lr_scale=1.0):
    """FISTA with adaptive top-k smoothing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    it = 0
    best_it = 0
    stall = 0
    base_lr = 2.5 / np.sqrt(n) * lr_scale
    lr = base_lr

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.3:
            break

        y_proj = np.maximum(y, 0)
        s_y = float(np.sum(y_proj))
        if s_y < 1e-10:
            y_proj = best_a.copy()
        else:
            y_proj = y_proj / s_y

        grad, max_val = adaptive_smoothed_gradient(y_proj, top_k=top_k)

        a_new = y_proj - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = best_a.copy()

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        t_k = t_new

        if stall > 30:
            lr *= 0.92
            stall = 0
        elif stall > 10:
            lr *= 0.97
            stall = 0

        if it - best_it > 8000:
            a = best_a.copy()
            y = a.copy()
            t_k = 1.0
            lr = base_lr
            best_it = it

        a = a_new
        it += 1

        if verbose and it % 1000 == 0:
            print(f"        FISTA it={it}, obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def coordinate_descent(a, max_sweeps=30, verbose=False):
    """
    Exact coordinate descent: for each coordinate, the max-conv objective
    is piecewise-linear convex. Find the minimizer by checking breakpoints.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    best_obj = compute_obj(a)

    for sweep in range(max_sweeps):
        improved = False
        max_improvement = 0.0

        for i in range(n):
            old_val = a[i]
            if old_val < 1e-14:
                continue

            # slope[k] = d(conv[k])/d(a[i]) = 2*a[k-i] for 0<=k-i<n
            slope = np.zeros(2 * n - 1)
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            slope[k_start:k_end + 1] = 2.0 * a[0:k_end - k_start + 1]

            current_max = float(np.max(conv))

            # Find candidate deltas that could reduce the max
            # Key insight: the optimal delta is where the max transitions
            # between different active constraints
            k_max = int(np.argmax(conv))
            slope_at_max = slope[k_max]

            if abs(slope_at_max) > 1e-12:
                # Delta to zero out the max
                d_zero = -conv[k_max] / slope_at_max
            else:
                d_zero = 0.0

            # Build candidate set: fractions of d_zero and boundaries
            candidates = set()
            candidates.add(0.0)
            candidates.add(max(d_zero * 0.2, -old_val * 0.99))
            candidates.add(max(d_zero * 0.5, -old_val * 0.99))
            candidates.add(max(d_zero * 0.8, -old_val * 0.99))
            candidates.add(max(d_zero, -old_val * 0.99))
            candidates.add(-old_val * 0.01)
            candidates.add(-old_val * 0.05)
            candidates.add(-old_val * 0.15)
            candidates.add(old_val * 0.05)
            candidates.add(old_val * 0.2)

            # Also add delta that equalizes top 2 conv values
            sorted_idx = np.argsort(conv)[::-1]
            if len(sorted_idx) > 1:
                k1, k2 = sorted_idx[0], sorted_idx[1]
                if abs(slope[k1] - slope[k2]) > 1e-12:
                    d_eq = (conv[k2] - conv[k1]) / (slope[k1] - slope[k2])
                    candidates.add(np.clip(d_eq, -old_val * 0.99, old_val * 5.0))

            best_delta = 0.0
            best_max_val = current_max

            for delta in candidates:
                new_val = old_val + delta
                if new_val < 0:
                    continue
                new_conv = conv + delta * slope
                new_max = float(np.max(new_conv))
                if new_max < best_max_val - 1e-14:
                    best_max_val = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_max_val < current_max - 1e-14:
                a[i] = old_val + best_delta
                conv = conv + best_delta * slope
                improved = True
                max_improvement = max(max_improvement, current_max - best_max_val)

        if not improved:
            break

        if verbose and sweep % 10 == 0:
            obj_now = compute_obj(a)
            print(f"      CD sweep {sweep}: obj={obj_now:.8f}, impr={max_improvement:.2e}")

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)

    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a_norm[diff[valid]]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_optimize(a, time_budget=50.0, verbose=False):
    """LP-based direction finding with line search and perturbation on stall."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    consecutive_fail = 0
    rng = np.random.default_rng(77)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        g, M = solve_lp_minpeak(a)
        if g is None:
            break

        best_t = 0.0
        best_obj_try = best_obj

        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj_try - 1e-14:
                    best_obj_try = obj_try
                    best_t = t

        if best_t > 0:
            a_try = (1 - best_t) * a + best_t * g
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            best_a = a_try.copy()
            best_obj = compute_obj(a_try)
            a = a_try
            consecutive_fail = 0
            if verbose:
                print(f"      LP: t={best_t:.3f}, obj={best_obj:.8f}, M={M:.6f}")
        else:
            consecutive_fail += 1
            if consecutive_fail >= 3:
                a = a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                consecutive_fail = 0
            if consecutive_fail >= 8:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.04):
    """Low-frequency Fourier perturbation for diversity."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 7):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def hamming_window_sequence(n, rng, alpha=0.54):
    """Create a Hamming-windowed sequence with random phases."""
    x = np.linspace(0, 1, n)
    hamming = alpha - (1 - alpha) * np.cos(2 * np.pi * x)
    # Add slight random modulation
    mod = 1.0 + 0.1 * rng.standard_normal(n)
    a = hamming * np.maximum(mod, 0.5)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def raised_cosine_sequence(n):
    """Raised cosine (Hann) window sequence."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: FISTA + coordinate descent + LP + multi-n with structural priors.
    Time budget: ~1480s for optimization, 20s for housekeeping.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: FISTA + CD on original n ===
    print(f"\n=== Phase 1: FISTA+CD on n={n_current} ===")
    remaining = deadline - time.monotonic()
    p1_fista = min(remaining * 0.2, 250.0)
    if p1_fista > 20:
        a1 = fista_optimize(best_seq, time_budget=p1_fista, verbose=False, top_k=25)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  FISTA: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    p1_cd = min(remaining * 0.1, 80.0)
    if p1_cd > 10:
        max_sw = int(min(40, p1_cd * 3))
        a1b = coordinate_descent(best_seq, max_sweeps=max_sw, verbose=False)
        obj1b = compute_obj(a1b)
        if obj1b < best_score:
            best_score = obj1b
            best_seq = a1b.copy()
            print(f"  CD: obj={best_score:.8f} **")

    # === Phase 2: LP optimization ===
    print(f"\n=== Phase 2: LP on n={n_current} ===")
    remaining = deadline - time.monotonic()
    p2 = min(remaining * 0.12, 150.0)
    if p2 > 15:
        a2 = lp_optimize(best_seq, time_budget=p2, verbose=False)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  LP: obj={best_score:.8f} **")

    # CD after LP
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a2b = coordinate_descent(best_seq, max_sweeps=20, verbose=False)
        obj2b = compute_obj(a2b)
        if obj2b < best_score:
            best_score = obj2b
            best_seq = a2b.copy()
            print(f"  CD after LP: obj={best_score:.8f} **")

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_candidates = sorted(set([
        n_best - 80, n_best - 40, n_best + 40, n_best + 80,
        n_best + 150, n_best + 250, n_best + 400, n_best + 600,
        700, 850, 950, 1050, 1150, 1300, 1500, 1700
    ]))
    n_candidates = [x for x in n_candidates if 600 <= x <= 1900 and abs(x - n_best) > 25]
    n_candidates = n_candidates[:7]
    print(f"  n candidates: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.18, 50.0)
        if alloc > 8:
            a_try = fista_optimize(a_try, time_budget=alloc, verbose=False, top_k=25)

        remaining = deadline - time.monotonic()
        alloc_cd = min(remaining * 0.05, 20.0)
        if alloc_cd > 3:
            a_try = coordinate_descent(a_try, max_sweeps=15, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Structural priors + restarts ===
    print(f"\n=== Phase 4: Structural Priors & Restarts ===")
    remaining = deadline - time.monotonic()

    restarts = [
        ("fourier", None),
        ("hamming", None),
        ("raised_cos", None),
        ("mirror_blend", None),
        ("smooth_blend", None),
    ]

    n_cur = len(best_seq)

    for name, _ in restarts:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if name == "fourier":
            a_r = fourier_perturb(best_seq, rng, strength=0.05)
        elif name == "hamming":
            a_r = hamming_window_sequence(n_cur, rng)
            # Blend with current to preserve good structure
            a_r = 0.4 * best_seq + 0.6 * a_r
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        elif name == "raised_cos":
            a_r = raised_cosine_sequence(n_cur)
            a_r = 0.3 * best_seq + 0.7 * a_r
            a_r = a_r / a_r.sum()
        elif name == "mirror_blend":
            a_r = 0.6 * best_seq + 0.4 * best_seq[::-1]
            a_r = a_r / a_r.sum()
        else:  # smooth_blend
            ks = max(7, n_cur // 15)
            if ks % 2 == 0:
                ks += 1
            a_sm = uniform_filter1d(best_seq, size=ks)
            a_r = 0.7 * best_seq + 0.3 * a_sm
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

        alloc_f = min(remaining * 0.15, 35.0)
        if alloc_f > 8:
            a_r = fista_optimize(a_r, time_budget=alloc_f, verbose=False, top_k=20)

        remaining = deadline - time.monotonic()
        alloc_c = min(remaining * 0.05, 15.0)
        if alloc_c > 3:
            a_r = coordinate_descent(a_r, max_sweeps=10, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 5: Intensive final optimization ===
    print(f"\n=== Phase 5: Final Intensive Optimization ===")
    remaining = deadline - time.monotonic()

    # FISTA with very focused top-k
    if remaining > 20:
        alloc = min(remaining * 0.25, 120.0)
        a_f = fista_optimize(best_seq, time_budget=alloc, verbose=False, top_k=10, lr_scale=0.7)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  FISTA (top_k=10): obj={best_score:.8f} **")

    # Heavy coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        max_sw = int(min(60, remaining * 2))
        a_c = coordinate_descent(best_seq, max_sweeps=max_sw, verbose=True)
        obj_c = compute_obj(a_c)
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_c.copy()
            print(f"  CD final: obj={best_score:.8f} **")

    # Another FISTA round
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 80.0)
        a_f2 = fista_optimize(best_seq, time_budget=alloc, verbose=False, top_k=15, lr_scale=0.5)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  FISTA final2: obj={best_score:.8f} **")

    # Final CD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_c2 = coordinate_descent(best_seq, max_sweeps=40, verbose=False)
        obj_c2 = compute_obj(a_c2)
        if obj_c2 < best_score:
            best_score = obj_c2
            best_seq = a_c2.copy()
            print(f"  CD final2: obj={best_score:.8f} **")

    # One more FISTA + CD cycle
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 60.0)
        a_f3 = fista_optimize(best_seq, time_budget=alloc, verbose=False, top_k=8, lr_scale=0.4)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  FISTA final3: obj={best_score:.8f} **")

        a_c3 = coordinate_descent(best_seq, max_sweeps=30, verbose=False)
        obj_c3 = compute_obj(a_c3)
        if obj_c3 < best_score:
            best_score = obj_c3
            best_seq = a_c3.copy()
            print(f"  CD final3: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
