def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

def smoothed_grad(a, tau):

    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Returns (gradient of length n, exact_max of conv).

    conv = compute_conv(a)

    A = fft(a, N)

def lp_polish_iteration(a, M_override=None):
    """
    Single LP polishing step: solve min M s.t. conv(a,g)<=M, g>=0, sum(g)=1.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    if n > 3000:
        return None, None

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    M_val = float(M_var.value) if M_var.value is not None else None
                    return g_val / sg, M_val
        except Exception:
            continue

    return None, None

        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [2, 3, 5, 7]:
            a += 0.15 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 4))
        return a / a.sum()
    elif mode == 8:
        x = np.linspace(0, 1, n)
        a = np.where((x > 0.15) & (x < 0.85), 1.0, 0.3)
        return a / a.sum()
    elif mode == 9:
        # Bell-shaped
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2)
        return a / a.sum()
    elif mode == 10:
        # Slightly asymmetric
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.45) ** 2) + 0.5 * np.exp(-3 * (x - 0.65) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def coordinate_descent_step(a, time_budget, max_coords=50):
    """
    Try individual coordinate updates: for each coordinate i, find the optimal
    value of a[i] (holding others fixed) that minimizes the objective.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()

    # Sort indices by magnitude (largest first, they matter more)
    idx_order = np.argsort(-a)[:max_coords]

    for _ in range(3):  # 3 passes
        if time.monotonic() - start > time_budget:
            break
        for i in idx_order:
            if time.monotonic() - start > time_budget:
                break

            # Bisection/search for optimal a[i]
            old_val = a[i]
            a_no_i = a.sum() - old_val

            # The optimal a[i] is somewhere in [0, a_no_i] (keeping sum=1)
            # Binary search for the value that minimizes the objective
            lo, hi = 0.0, a_no_i
            best_local_val = old_val
            best_local_obj = np.inf

            for _ in range(25):
                mid = (lo + hi) / 2
                a[i] = mid
                a[i] = max(0, a[i])
                s_a = float(np.sum(a))
                if s_a < 1e-10:
                    a[i] = old_val
                    break
                a_norm = a / s_a
                obj_try = compute_obj(a_norm)
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_local_val = mid
                # Check if we should go lower or higher
                # Evaluate at mid - delta and mid + delta
                delta = (hi - lo) * 0.1
                a[i] = max(0, mid - delta)
                s1 = float(np.sum(a))
                if s1 > 1e-10:
                    obj_lo = compute_obj(a / s1)
                else:
                    obj_lo = np.inf
                a[i] = min(a_no_i, mid + delta)
                s2 = float(np.sum(a))
                if s2 > 1e-10:
                    obj_hi = compute_obj(a / s2)
                else:
                    obj_hi = np.inf

                if obj_lo < obj_hi:
                    hi = mid
                else:
                    lo = mid

            a[i] = best_local_val
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
                obj_new = compute_obj(a)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
            else:
                a[i] = old_val

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase:
    1. Load parent, quick screening
    2. Aggressive high-tau GD
    3. LP polishing (multiple rounds)
    4. Support refinement
    5. Coordinate descent
    6. Multi-n exploration
    7. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.10f}")

    n_current = len(best_seq)

    def check_time():
        return deadline - time.monotonic()

    # === Phase 1: Quick diverse screening ===
    print(f"\n=== Phase 1: Diverse Screening ===")
    remaining = check_time()
    if remaining > 60:
        for mode in [2, 5, 9, 10, 4]:
            remaining = check_time()
            if remaining < 20:
                break
            a_init = make_init(n_current, mode, rng)
            alloc = min(remaining * 0.08, 25.0)
            if alloc < 5:
                break
            a_s = high_tau_gd(a_init, alloc,
                              tau_levels=[10, 50, 200, 800],
                              verbose=False, lr_scale=2.0, beta=0.88)
            obj_s = compute_obj(a_s)
            if obj_s < best_score:
                best_score = obj_s
                best_seq = a_s.copy()
                print(f"  Mode {mode}: obj={best_score:.10f} **")
            else:
                print(f"  Mode {mode}: obj={obj_s:.10f}")

    # === Phase 2: Main high-tau GD ===
    print(f"\n=== Phase 2: High-Tau GD ===")
    remaining = check_time()
    p2_budget = min(remaining * 0.22, 320.0)
    if p2_budget > 30:
        tau_sched = [5, 20, 60, 200, 600, 2000, 6000, 15000, 30000, 50000]
        a2 = high_tau_gd(best_seq, p2_budget,
                         tau_levels=tau_sched, verbose=True,
                         lr_scale=2.0, beta=0.9)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2: obj={best_score:.10f} **")
        else:
            print(f"  Phase2: obj={obj2:.10f}")

    # === Phase 3: LP Polishing (multiple rounds) ===
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = check_time()
    if remaining > 60 and n_current <= 2500:
        for rnd in range(6):
            remaining = check_time()
            if remaining < 20:
                break
            alloc = min(remaining * 0.12, 60.0)
            if alloc < 10:
                break
            seed_rp = 100 + rnd * 7
            a3 = lp_polish(best_seq, alloc, verbose=True,
                           seed=seed_rp, max_iters=60)
            obj3 = compute_obj(a3)
            if obj3 < best_score:
                best_score = obj3
                best_seq = a3.copy()
                print(f"  LP rnd{rnd}: obj={best_score:.10f} **")
            else:
                print(f"  LP rnd{rnd}: obj={obj3:.10f}")

    # === Phase 4: Support Refinement ===
    print(f"\n=== Phase 4: Support Refinement ===")
    remaining = check_time()
    if remaining > 40:
        alloc = min(remaining * 0.1, 60.0)
        a_sr = support_refine(best_seq, alloc, verbose=True, threshold=1e-5)
        obj_sr = compute_obj(a_sr)
        if obj_sr < best_score:
            best_score = obj_sr
            best_seq = a_sr.copy()
            print(f"  Support: obj={best_score:.10f} **")
        else:
            print(f"  Support: obj={obj_sr:.10f}")

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    remaining = check_time()
    if remaining > 30:
        alloc = min(remaining * 0.08, 50.0)
        a_cd = coordinate_descent_step(best_seq, alloc, max_coords=40)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CoordDesc: obj={best_score:.10f} **")
        else:
            print(f"  CoordDesc: obj={obj_cd:.10f}")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = check_time()
    n_base = len(best_seq)
    n_options = sorted(set([
        n_base - 100, n_base - 50, n_base + 50, n_base + 100,
        n_base + 200, n_base + 400,
        600, 900, 1000, 1200, 1500, 1800
    ]))
    n_options = [n for n in n_options if 400 <= n <= 3000]
    # Prioritize close to current, then larger
    n_options.sort(key=lambda x: abs(x - n_base))
    n_options = n_options[:7]

    print(f"  n candidates: {n_options}")
    for n_try in n_options:
        remaining = check_time()
        if remaining < 50:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample(best_seq, n_try)

        # GD
        alloc1 = min(remaining * 0.15, 45.0)
        if alloc1 > 10:
            a_try = high_tau_gd(a_try, alloc1,
                               tau_levels=[10, 50, 200, 800, 3000],
                               verbose=False, lr_scale=2.0, beta=0.88)

        # LP if small enough
        remaining = check_time()
        if n_try <= 1800:
            alloc2 = min(remaining * 0.08, 30.0)
            if alloc2 > 10:
                a_try = lp_polish(a_try, alloc2, verbose=False,
                                  seed=n_try, max_iters=30)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Intensive Polish ===")
    remaining = check_time()

    # Ultra-high tau GD
    if remaining > 40:
        alloc = min(remaining * 0.2, 80.0)
        a_f1 = high_tau_gd(best_seq, alloc,
                           tau_levels=[200, 1000, 5000, 15000, 30000, 50000],
                           verbose=True, lr_scale=1.5, beta=0.92)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final GD: obj={best_score:.10f} **")

    # Another round of LP
    remaining = check_time()
    if remaining > 25 and len(best_seq) <= 2500:
        alloc = min(remaining * 0.2, 60.0)
        a_f2 = lp_polish(best_seq, alloc, verbose=True, seed=9999, max_iters=50)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    # Another GD after LP
    remaining = check_time()
    if remaining > 30:
        alloc = min(remaining * 0.25, 70.0)
        a_f3 = high_tau_gd(best_seq, alloc,
                           tau_levels=[500, 2000, 8000, 20000, 40000],
                           verbose=False, lr_scale=1.5, beta=0.93)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Post-LP GD: obj={best_score:.10f} **")

    # Final coordinate descent
    remaining = check_time()
    if remaining > 20:
        alloc = min(remaining * 0.1, 30.0)
        a_f4 = coordinate_descent_step(best_seq, alloc, max_coords=30)
        obj_f4 = compute_obj(a_f4)
        if obj_f4 < best_score:
            best_score = obj_f4
            best_seq = a_f4.copy()
            print(f"  Final CoordDesc: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
