"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of: Nesterov-accelerated smoothed gradient descent (mirror + Euclidean),
LP-based block coordinate descent, fast coordinate descent, variance-regularized optimization,
and aggressive multi-n exploration with parametric initializations.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau):
    """Compute smoothed gradient of max conv(a,a) using log-sum-exp smoothing."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:
            continue
    return None, None


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=50.0, tau_end=2000.0,
                               beta=0.9, use_mirror=False, verbose=False):
    """Nesterov accelerated smoothed gradient descent with adaptive tau."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Slower ramp-up of tau for more accurate gradients
        tau = tau_init + (tau_end - tau_init) * (progress ** 0.7)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        # Barzilai-Borwein step size
        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                bb_lr = min(max(bb_lr, 1e-8), 4.0 / np.sqrt(n))
                lr = 0.5 * lr + 0.5 * bb_lr

        v = beta * v - lr * g

        if use_mirror:
            # Mirror descent: multiplicative update
            a_new = y * np.exp(-0.1 * v / max(np.max(np.abs(v)), 1e-10))
            a_new = np.maximum(a_new, 1e-15)
        else:
            a_new = y + v
            a_new = np.maximum(a_new, 0)

        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            beta = min(0.9, beta + 0.01)
            best_it = it

    return best_a


def lp_block_descent(a, time_budget=150.0, verbose=False, seed=42):
    """LP-based block coordinate descent with momentum and line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    prev_lp_sol = None
    momentum_vec = 0.0
    it = 0
    best_it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Momentum extrapolation
        if prev_lp_sol is not None:
            diff = g_lp - prev_lp_sol
            momentum_vec = 0.8 * momentum_vec + 0.2 * np.linalg.norm(diff)
            g_pred = g_lp + 0.4 * momentum_vec * diff
            g_pred = np.maximum(g_pred, 0)
            s_pred = float(np.sum(g_pred))
            if s_pred > 1e-10:
                g_pred = g_pred / s_pred
                obj_pred = compute_obj(g_pred)
                if obj_pred < best_obj + 0.01:
                    g_lp = g_pred

        prev_lp_sol = g_lp.copy()
        it += 1

        # Line search
        t_candidates = np.array([0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0])
        if stall_count > 3:
            t_candidates = np.array([0.1, 0.2, 0.3, 0.5, 1.0, 1.5, 2.0])

        best_t_obj = best_obj
        best_a_local = best_a.copy()

        for t in t_candidates:
            if t <= 1.0:
                a_try = (1.0 - t) * a + t * g_lp
            else:
                a_try = t * g_lp - (t - 1.0) * a
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            best_it = it
            stall_count = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}")
        else:
            stall_count += 1
            if stall_count > 10:
                # Perturb to escape
                strength = min(0.1, 0.01 * stall_count)
                pert = 1.0 + rng.normal(0, strength, n)
                a = best_a * np.maximum(pert, 0)
                a = a / a.sum()
                prev_lp_sol = None
                momentum_vec = 0.0
                stall_count = 5

    return best_a


def coordinate_descent_sweep(a, max_sweeps=5, verbose=False):
    """Fast coordinate descent using quadratic formula per coordinate."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_obj = compute_obj(a)

    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            if a[j] < 1e-15:
                continue
            S = s - a[j]
            if S < 1e-15:
                continue

            # Precompute conv of a with a[j] zeroed
            a_r = a.copy()
            a_r[j] = 0.0
            conv_r = compute_conv(a_r)

            # Precompute cross vector: cross[k] = a[k-j] + a[j-k] (with zeroing at j)
            cross = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                idx1 = k - j
                idx2 = j - k
                if 0 <= idx1 < n and idx1 != j:
                    cross[k] += a[idx1]
                if 0 <= idx2 < n and idx2 != j:
                    cross[k] += a[idx2]

            # Objective as function of t (a_j = t, rest scaled by (1-t)/S):
            # f(t) = max_k [ (1-t)^2/S^2 * conv_r[k] + t(1-t)/S * cross[k] + t^2 * (1 if k==2j else 0) ]
            # Each term is a quadratic in t with non-negative leading coefficient.
            # The max is convex in t. Use golden section search on [0, 1].
            
            def obj_at_t(t):
                alpha = (1.0 - t) ** 2 / (S * S)
                beta = t * (1.0 - t) / S
                gamma = t * t
                vals = alpha * conv_r + beta * cross
                if 2 * j < 2 * n - 1:
                    vals[2 * j] += gamma
                return float(np.max(vals))

            # Golden section search
            gr = (np.sqrt(5) - 1) / 2
            lo, hi = 0.0, 1.0
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc, fd = obj_at_t(c), obj_at_t(d)

            for _ in range(40):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = obj_at_t(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = obj_at_t(d)

            t_opt = (lo + hi) / 2.0
            # Update a
            a_new = a * (1.0 - t_opt) / S
            a_new[j] = t_opt
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    a = a_new
                    best_obj = obj_new
                    improved = True

        if verbose:
            print(f"    Coord sweep {sweep}: obj={best_obj:.10f}, improved={improved}")
        if not improved:
            break

    return a


def variance_regularized_obj(a, lam=0.01):
    """Objective + small variance penalty on the convolution."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    var_c = float(np.var(conv[:n]))  # Only look at the "important" part
    return 2.0 * n * max_c + lam * var_c


def variance_guided_descent(a, time_budget=60.0, lam=0.005, verbose=False):
    """Gradient descent with variance regularization to encourage flat convolution."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_true_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.0 / np.sqrt(n)
    v = np.zeros(n)
    beta = 0.9
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        
        # Gradient of max term (use active set)
        active = np.where(conv >= max_c - 1e-6)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        
        # Gradient of max term: average of gradients over active
        # grad_k[m] = 2*a[k-m] (if valid)
        N = 1
        while N < 3 * n:
            N <<= 1
        A = fft(a, N)
        g_max = np.zeros(n)
        for k in active[:20]:  # Limit to 20 most active
            # Shift a by -k and reverse, take first n
            shifted = np.zeros(N)
            for m in range(n):
                if 0 <= k - m < n:
                    shifted[m] = 2.0 * a[k - m]
            g_max += shifted[:n]
        g_max /= min(len(active), 20)
        
        # Gradient of variance term (approximate)
        # var(conv) ≈ mean(conv^2) - mean(conv)^2
        # d(conv[k])/da[m] = 2*a[k-m]
        # d(mean(conv^2))/da = 2 * convolve(conv, a) sort of...
        # Simplify: just use the max gradient with a small extra push toward uniform
        g_var = np.zeros(n)
        conv_central = conv[n//4:3*n//4] if n > 4 else conv
        g_var[:] = 1.0 / n  # Push toward uniform (opposite of concentration)
        
        g = g_max + lam * 10.0 * g_var / max(np.max(np.abs(g_max)), 1e-10)

        # Nesterov step
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y > 1e-10:
            y = y / s_y
        v = beta * v - lr * g
        a_new = np.maximum(y + v, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_true_obj - 1e-14:
            best_true_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-9, lr * 0.9995)

        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it

    return best_a


def parametric_init(n, mode, rng):
    """Generate parametric initializations."""
    x = np.linspace(0, 1, n)
    if mode == 'uniform':
        a = np.ones(n)
    elif mode == 'gaussian':
        sigma = rng.uniform(0.15, 0.35)
        center = rng.uniform(0.3, 0.7)
        a = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
    elif mode == 'cosine':
        k = rng.integers(1, 4)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + rng.uniform(0.3, 0.8) * np.cos(2 * np.pi * k * x + phase)
    elif mode == 'bimodal':
        c1, c2 = sorted(rng.uniform(0, 1, 2))
        s1, s2 = rng.uniform(0.05, 0.15, 2)
        a = np.exp(-((x - c1) ** 2) / (2 * s1 ** 2)) + np.exp(-((x - c2) ** 2) / (2 * s2 ** 2))
    elif mode == 'trapezoid':
        w = rng.uniform(0.2, 0.8)
        a = np.clip(1.0 - np.abs(x - 0.5) / (w / 2), 0.1, 1.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines Nesterov smoothed GD, LP block descent, coordinate descent,
    variance-regularized optimization, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_default = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={n_current}): obj={obj:.8f}")
            else:
                best_seq = np.ones(n_default) / n_default
                best_score = compute_obj(best_seq)
                n_current = n_default
                print(f"Parent invalid, using default (n={n_default}): obj={best_score:.8f}")
        else:
            best_seq = np.ones(n_default) / n_default
            best_score = compute_obj(best_seq)
            n_current = n_default
            print(f"No valid parent, using default: obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        n_current = n_default
        print(f"Exception loading parent, using default: obj={best_score:.8f}")

    def try_improve(a, method, budget, **kwargs):
        """Try a method and update best if improved."""
        nonlocal best_seq, best_score
        if budget < 10:
            return
        try:
            if method == 'gd':
                a_new = nesterov_smoothed_descent(a, time_budget=budget, **kwargs)
            elif method == 'lp':
                a_new = lp_block_descent(a, time_budget=budget, **kwargs)
            elif method == 'coord':
                a_new = coordinate_descent_sweep(a, **kwargs)
            elif method == 'var':
                a_new = variance_guided_descent(a, time_budget=budget, **kwargs)
            else:
                return
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"    [{method}] obj={best_score:.8f} **")
            else:
                print(f"    [{method}] obj={obj:.8f}")
        except Exception as e:
            print(f"    [{method}] error: {e}")

    # === Phase 1: Quick Nesterov GD for initial improvement ===
    print(f"\n=== Phase 1: Nesterov GD (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.15, 120.0)
    if p1_budget > 30:
        try_improve(best_seq, 'gd', p1_budget, tau_init=50, tau_end=1500, beta=0.9)
        try_improve(best_seq, 'gd', min(remaining * 0.05, 40), tau_init=200, tau_end=3000, beta=0.9, use_mirror=True)

    # === Phase 2: LP block descent (main workhorse) ===
    print(f"\n=== Phase 2: LP Block Descent (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 30:
        try_improve(best_seq, 'lp', p2_budget, verbose=True, seed=42)

    # === Phase 3: Coordinate descent polish ===
    print(f"\n=== Phase 3: Coordinate Descent (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Limit sweeps based on n
        max_sweeps = max(2, min(8, int(20000000 / (n_current * 40))))
        try_improve(best_seq, 'coord', 300, max_sweeps=max_sweeps, verbose=True)

    # === Phase 4: Variance-regularized refinement ===
    print(f"\n=== Phase 4: Variance-regularized (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 60.0)
    if p4_budget > 20:
        try_improve(best_seq, 'var', p4_budget, lam=0.003)

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)

    # Candidate n values
    n_candidates = set()
    for dn in [-200, -100, -50, 0, 50, 100, 150, 200, 300, 400, 500]:
        n_try = n_best + dn
        if 400 <= n_try <= 2500:
            n_candidates.add(n_try)
    for n_abs in [500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600, 2000]:
        if abs(n_abs - n_best) > 30:
            n_candidates.add(n_abs)

    n_candidates = sorted(n_candidates)
    # Limit to 8 most promising (closest to current + spread)
    n_candidates = sorted(n_candidates, key=lambda x: abs(x - n_best))[:8]
    print(f"  Trying n values: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if n_try == n_best:
            continue

        # Initialize by resampling
        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()
        obj_try = compute_obj(a_try)
        print(f"  n={n_try}: resampled obj={obj_try:.8f}")

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try

        # Optimize at this n
        remaining = deadline - time.monotonic()
        alloc1 = min(remaining * 0.12, 80.0)
        if alloc1 > 20:
            try_improve(a_try, 'gd', alloc1, tau_init=100, tau_end=2000)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.08, 60.0)
        if alloc2 > 20:
            try_improve(best_seq if len(best_seq) == n_try else a_try, 'lp', alloc2, seed=n_try)

        remaining = deadline - time.monotonic()
        if remaining > 30:
            max_sweeps = max(2, min(5, int(10000000 / (n_try * 40))))
            try_improve(best_seq if len(best_seq) == n_try else a_try, 'coord', 200,
                        max_sweeps=max_sweeps)

    # === Phase 6: Diversification with parametric inits ===
    print(f"\n=== Phase 6: Diversification ===")
    n_current = len(best_seq)
    remaining = deadline - time.monotonic()

    modes = ['gaussian', 'cosine', 'bimodal', 'trapezoid']
    for mode in modes:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        a_p = parametric_init(n_current, mode, rng)
        alloc = min(remaining * 0.1, 60.0)
        if alloc > 15:
            try_improve(a_p, 'gd', alloc, tau_init=50, tau_end=1000)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                try_improve(best_seq, 'lp', min(remaining * 0.05, 30), seed=hash(mode) % 10000)

    # Also try at a different n with parametric init
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_alt = n_current + rng.integers(-200, 200)
        n_alt = max(400, min(2000, n_alt))
        if n_alt != n_current:
            a_alt = parametric_init(n_alt, 'gaussian', rng)
            try_improve(a_alt, 'gd', min(remaining * 0.1, 50), tau_init=50, tau_end=1000)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                try_improve(best_seq, 'lp', min(remaining * 0.08, 40), seed=9999)

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish (n={len(best_seq)}) ===")
    n_final = len(best_seq)

    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.35, 200.0)
        print(f"  Final LP: budget={alloc:.0f}s")
        try_improve(best_seq, 'lp', alloc, verbose=True, seed=12345)

    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.25, 100.0)
        print(f"  Final GD: budget={alloc:.0f}s")
        try_improve(best_seq, 'gd', alloc, tau_init=500, tau_end=5000, beta=0.95)

    remaining = deadline - time.monotonic()
    if remaining > 30:
        max_sweeps = max(3, min(10, int(30000000 / (n_final * 40))))
        print(f"  Final Coord: {max_sweeps} sweeps")
        try_improve(best_seq, 'coord', 200, max_sweeps=max_sweeps, verbose=True)

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 80.0)
        print(f"  Final LP2: budget={alloc:.0f}s")
        try_improve(best_seq, 'lp', alloc, seed=77777)

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 50.0)
        try_improve(best_seq, 'gd', alloc, tau_init=1000, tau_end=10000, beta=0.97)

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
