Asymmetric LP-polished optimization for step function coefficients.
Key innovation: parametric tilted-U initialization to escape symmetric basin (obj>=2),

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a * a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # A[k, j] = a[k - j] if 0 <= k - j < n, else 0
        # So for column j, non-zero entries are at rows k where 0 <= k-j < n
        # i.e., j <= k <= j+n-1, clipped to [0, m-1]
        row_start = j
        row_end = min(m, j + n)
        vals = a[:row_end - row_start]
        A[row_start:row_end, j] = vals
    return A


def solve_lp(a):
    """Solve LP: min M s.t. A @ g <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.MOSEK]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def subgradient_step(a, lr=0.01):
    """One subgradient step on the normalized simplex."""
    n = len(a)
    a = a / a.sum()
    conv = compute_conv(a)
    k_max = int(np.argmax(conv))
    
    grad = np.zeros(n)
    for i in range(n):
        j1 = k_max - i
        j2 = i - k_max
        if 0 <= j1 < n:
            grad[i] += a[j1]
        if 0 <= j2 < n:
            grad[i] += a[j2]
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s

def resample_sequence(a, n_new):
    """Resample sequence to new length using linear interpolation."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 0.01:
        return None
    return a_new / s

```python
def solve_lp_separation(a, max_iters=10, initial_n_active=10):
    """LP with lazy constraint generation (separation oracle)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A_full = build_conv_matrix(a_norm)
    
    # Start with the most violated constraints
    conv = compute_conv(a_norm)
    active = set(np.argsort(conv)[-initial_n_active:])
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    base_constraints = [cp.sum(g) == 1]
    
    for outer_it in range(max_iters):
        # Build constraints from active set
        constraints = base_constraints + [A_full[k, :] @ g <= M for k in active]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        solved = False
        for solver in [cp.CLARABEL, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        g_val = g_val / s
                        solved = True
                        break
            except Exception:
                continue
        
        if not solved:
            break
        
        # Check for violated constraints
        A_g = A_full @ g_val
        violated = np.where(A_g > M.value + 1e-8)[0]
        
        if len(violated) == 0:
            return g_val, float(M.value)
        
        # Add violated constraints (up to 5 at a time)
        active.update(violated[:5])
    
    # Return the best solution found
    A_g = A_full @ g_val
    return g_val, float(M.value)
```

    """Get row k of the convolution matrix: A[k, :] such that A[k,:] @ g = conv(a,g)[k]."""
