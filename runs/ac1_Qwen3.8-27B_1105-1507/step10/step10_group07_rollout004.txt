        return v

def solve_peak_aware_lp(a, c_offset, M_target=None):
    """
    Solve peak-aware LP:
    min M s.t. (a*g)(k) + c_offset[k] <= M for all k
                sum(g) = 1, g >= 0
    where c_offset is typically (a*a) * alpha for some alpha > 0.
    """
    n = len(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g + c_offset <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_tr_sscp(a, r):
    """
    Trust region SOCP:
    min M s.t. C_old + 2*(a*delta)[k] <= M, ||delta||_2 <= r,
                sum(a+delta) = 1, a+delta >= 0
    """
    n = len(a)
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)
    
    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta
    
    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
        cp.norm(delta, 2) <= r,
    ]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None

def subgradient_polish(a, time_budget=100.0, K=5, verbose=False):
    """Subgradient method focusing on top-K peaks."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 2.0 / np.sqrt(n)
    momentum_decay = 0.75
    v = np.zeros(n)
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        
        # Compute subgradient: average of the top K peak gradients
        g = np.zeros(n)
        for k in top_idx:
            k = int(k)
            # Gradient of (a*a)(k) w.r.t. a is: for each i, a_{k-i} (if valid)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += a[j_start:j_end + 1][::-1]
        
        g = g / K  # Normalize
        
        v = momentum_decay * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.01, 10.0 / np.sqrt(n))
        else:
            stall += 1
            if stall > 100:
                lr *= 0.7
                v *= 0.5
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.998
        
        it += 1
    
    return best_a


def smoothed_gd(a, time_budget=80.0, tau_init=10.0, tau_end=300.0):
    """Smoothed gradient descent with log-sum-exp approximation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 3.0 / np.sqrt(n)
    momentum = 0.85
    v = np.zeros(n)
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        if max_conv < 1e-15:
            break
        
        # Log-sum-exp for smooth max
        shifted = tau * (conv - max_conv)
        exp_vals = np.exp(shifted)
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient via FFT
        N = 1
        while N < 3 * n:
            N <<= 1
        from scipy.fft import fft, ifft
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]
        
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        lr = max(1e-12, lr * 0.999)
        it += 1
    
    return best_a


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def diverse_init(n, rng, mode):
    """Generate diverse initial sequences."""
    if mode == 'uniform':
        return np.ones(n) / n
    elif mode == 'triangular':
        x = np.linspace(0, 1, n)
        a = 1 - np.abs(2 * x - 1)
        return a / a.sum()
    elif mode == 'exponential':
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * x)
        return a / a.sum()
    elif mode == 'bimodal':
        x = np.linspace(0, 1, n)
        a = np.exp(-10 * (x - 0.25)**2) + np.exp(-10 * (x - 0.75)**2)
        return a / a.sum()
    elif mode == 'cosine':
        x = np.linspace(0, 1, n)
        a = (1 + np.cos(2 * np.pi * x)) / 2
        return a / a.sum()
    elif mode == 'random_smooth':
        a = np.exp(rng.normal(0, 0.5, n))
        # Smooth
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=3)
        return a / a.sum()
    else:
        return np.ones(n) / n
