For a random non-negative sequence, the convolution will have its peak at the center (by the central limit theorem-like argument - more terms are added). So the peak is typically at or near the center.

**Key algorithm: "Iterative peak reduction"**

1. Start with a good initial sequence (parent or random)
2. Repeat:
   a. Compute conv = a * a, find peak position k*
   b. The peak conv[k*] = sum_i a[i] * a[k*-i]
   c. To reduce this peak, we want to reduce the "alignment" between a and its shift-reversal
   d. One way: reduce a[i] where a[i] * a[k*-i] is large
   e. Another way: increase a[j] where j doesn't contribute to the peak (or contributes less)
   f. Use a gradient-based step to achieve this balance

def block_coordinate_descent(a, time_budget=200.0, block_size=20, verbose=False, seed=42):
    """
    Block coordinate descent: update blocks of coordinates at once.
    For a block of coordinates, the subproblem is a small QP (minimize max of quadratics).
    We solve it via scipy.optimize.minimize.
    """
    from scipy.optimize import minimize as sp_minimize
    
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    
    while time.monotonic() - start < time_budget:
        # Random permutation of coordinates, process in blocks
        perm = rng.permutation(n)
        improved = False
        
        for block_start in range(0, n, block_size):
            block = perm[block_start:block_start + block_size]
            n_block = len(block)
            
            # Current values of the block
            x0 = a[block].copy()
            
            def block_obj(x):
                """Objective for the block, with other coordinates fixed."""
                a_try = a.copy()
                a_try[block] = np.maximum(x, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    return 1e10
                a_try = a_try / s_try
                conv = compute_conv(a_try)
                return float(np.max(conv))
            
            res = sp_minimize(block_obj, x0, method='Nelder-Mead',
                            options={'maxiter': 50, 'xatol': 1e-6, 'fatol': 1e-8})
            
            if res.fun < best_obj / (2.0 * n) - 1e-12:  # improved (accounting for 2n factor)
                a[block] = np.maximum(res.x, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                improved = True
        
        obj_cur = compute_obj(a)
        if obj_cur < best_obj - 1e-12:
            best_obj = obj_cur
            best_a = a.copy()
        
        if not improved:
            # Try a perturbation
            a = best_a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 0:
                a = a / s_a
        
        it += 1
        if verbose and it % 5 == 0:
            print(f"    BCD it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
    
    return best_a


def peak_shaving(a, time_budget=100.0, verbose=False, seed=42):
    """
    Targeted peak shaving: identify the peak lag, reduce contributions to it,
    and redistribute mass to lags that don't increase other peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_peak = int(np.argmax(conv))
        peak_val = conv[k_peak]
        
        # Find which coordinates contribute most to the peak
        # conv[k_peak] = sum_i a[i] * a[k_peak - i] for valid i
        contrib = np.zeros(n)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        # Also consider the "partner" coordinates
        # Reducing a[i] reduces conv[k_peak] by a[i]*a[k_peak-i] per unit
        # But it also affects other conv[lags]
        
        # Strategy: reduce the top contributors, increase the bottom ones
        top_k = max(1, n // 10)
        bottom_k = max(1, n // 10)
        
        # Sort by contribution to peak
        order = np.argsort(contrib)[::-1]
        reduce_set = order[:top_k]
        increase_set = order[-bottom_k:]
        
        # Amount to transfer
        total_reduce = sum(contrib[i] for i in reduce_set)
        if total_reduce < 1e-10:
            break
        
        # Try different amounts
        best_delta = 0
        best_obj_try = best_obj
        
        for frac in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2]:
            a_try = a.copy()
            for i in reduce_set:
                a_try[i] *= (1 - frac)
            # Redistribute to increase_set proportionally
            mass_freed = sum(a[i] * frac for i in reduce_set)
            if mass_freed > 0:
                inc_total = sum(a_try[i] for i in increase_set)
                if inc_total > 0:
                    for i in increase_set:
                        a_try[i] += mass_freed * a_try[i] / inc_total
                else:
                    a_try[increase_set] += mass_freed / len(increase_set)
            
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_try - 1e-13:
                best_obj_try = obj_try
                best_delta = frac
        
        if best_delta > 0:
            a_try = a.copy()
            for i in reduce_set:
                a_try[i] *= (1 - best_delta)
            mass_freed = sum(a[i] * best_delta for i in reduce_set)
            inc_total = sum(a_try[i] for i in increase_set)
            if inc_total > 0:
                for i in increase_set:
                    a_try[i] += mass_freed * a_try[i] / inc_total
            else:
                a_try[increase_set] += mass_freed / len(increase_set)
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
            a = a_try
            best_obj = best_obj_try
            best_a = a.copy()
        else:
            # Perturb and try again
            a = best_a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 0:
                a = a / s_a
            it += 1
            if it > 100:
                break
        
        it += 1
    
    return best_a

"""
Hybrid optimizer combining exponentiated gradient (mirror descent) with log-sum-exp smoothing,
generalized coordinate descent, peak shaving, subgradient methods, and multi-length exploration
to minimize peak autocorrelation of non-negative step functions.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def exponentiated_gradient(a, time_budget=100.0, tau_init=20.0, tau_end=500.0,
                           lr_init=None, verbose=False, seed=42):
    """
    Exponentiated gradient (mirror descent) with log-sum-exp smoothing.
    Minimizes smoothed max of conv(a,a) subject to a>=0, sum(a)=1.
    Uses multiplicative update: a_i <- a_i * exp(-lr * grad_i) / Z
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if lr_init is None:
        lr_init = 3.0 / np.sqrt(n)
    lr = lr_init
    it = 0
    no_improve = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-3))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)

        conv = compute_conv(a)
        c_max = float(np.max(conv))
        shifted = conv - c_max
        exp_vals = np.exp(tau * shifted)
        p = exp_vals / float(np.sum(exp_vals))

        # Compute gradient: grad_i = 2 * sum_k p[k] * a[k-i] for valid k-i
        # This equals 2 * ifft(fft(p) * conj(fft(a))).real[:n] with proper padding
        N = 1
        while N < 3 * n:
            N <<= 1
        P_ft = fft(p, N)
        A_ft = fft(a, N)
        grad = 2.0 * ifft(P_ft * np.conj(A_ft)).real[:n]

        # Multiplicative update
        a_new = a * np.exp(-lr * grad)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr *= 0.9995

        # Perturb if stuck
        if no_improve > 100:
            a = best_a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 0)
            s_p = a.sum()
            if s_p > 1e-10:
                a = a / s_p
            no_improve = 0
            lr = lr_init * 0.5

        it += 1
        if verbose and it % 100 == 0:
            print(f"    EG it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

    return best_a


def generalized_bcd(a, time_budget=300.0, verbose=False, seed=42):
    """
    Generalized coordinate descent: for each coordinate, try a set of candidate values
    and pick the one that minimizes the objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()

            a_old = a[idx]

            else:

                if s_a < 1e-10:
                    a[idx] = a_old

                if s_a > 1e-10:
                    a = a / s_a

        obj_cur = compute_obj(a)

def peak_shaving(a, time_budget=120.0, verbose=False, seed=42):
    """
    Targeted peak shaving: identify peak lag, reduce top contributors,
    redistribute mass to non-contributing coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_peaks = np.argsort(conv)[::-1][:max(3, n // 50)]

        # Compute contribution to each coordinate from all peak lags
        contrib = np.zeros(n)
        for k in k_peaks:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contrib[i] += a[i] * a[j]

        top_k = max(5, n // 8)
        bot_k = max(5, n // 8)
        order = np.argsort(contrib)[::-1]
        reduce_set = order[:top_k]
        increase_set = order[-bot_k:]

        best_frac = 0
        best_obj_try = best_obj

        for frac in [0.01, 0.02, 0.04, 0.07, 0.1, 0.15, 0.2]:
            a_try = a.copy()
            for i in reduce_set:
                a_try[i] *= (1.0 - frac)
            mass_freed = sum(a[i] * frac for i in reduce_set)
            if mass_freed > 0:
                inc_vals = a_try[increase_set]
                inc_total = inc_vals.sum()
                if inc_total > 1e-15:
                    a_try[increase_set] += mass_freed * inc_vals / inc_total
                else:
                    a_try[increase_set] += mass_freed / len(increase_set)

            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_try - 1e-13:
                best_obj_try = obj_try
                best_frac = frac

        if best_frac > 0:
            a_try = a.copy()
            for i in reduce_set:
                a_try[i] *= (1.0 - best_frac)
            mass_freed = sum(a[i] * best_frac for i in reduce_set)
            inc_vals = a_try[increase_set]
            inc_total = inc_vals.sum()
            if inc_total > 1e-15:
                a_try[increase_set] += mass_freed * inc_vals / inc_total
            else:
                a_try[increase_set] += mass_freed / len(increase_set)
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
            a = a_try
            best_obj = best_obj_try
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            a = best_a * np.exp(rng.normal(0, 0.01 + 0.005 * min(stall, 5), n))
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-10:
                a = a / s_a
            if stall > 50:
                break

        if verbose and stall % 20 == 0:
            print(f"    PS: obj={best_obj:.8f}, stall={stall}, t={time.monotonic()-start:.0f}s")

    return best_a


def subgradient_accelerated(a, time_budget=150.0, K=10, verbose=False, seed=42):
    """
    Accelerated subgradient method (FIST) for minimizing max_k conv_k.
    Uses top-K peak weighting for a smoother subgradient approximation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # FIST parameters
    T = 10000
    y = a.copy()
    v = a.copy()
    it = 0

    for t in range(1, T + 1):
        if time.monotonic() - start >= time_budget:
            break

        # Compute subgradient at y
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:K]
        peak_vals = conv_y[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break

        # Soft weighting of peaks
        weights = np.exp((peak_vals - max_pv) * 5.0 / max(pv for pv in [max_pv, 1e-15]))
        weights /= weights.sum()

        # Compute gradient: g_i = 2 * sum_k w_k * a_y[k-i]
        N_fft = 1
        while N_fft < 3 * n:
            N_fft <<= 1
        # Build weighted peak array
        wp = np.zeros(2 * n - 1)
        for w, k in zip(weights, top_idx):
            wp[int(k)] = w
        P_ft = fft(wp, N_fft)
        Y_ft = fft(y, N_fft)
        g = 2.0 * ifft(P_ft * np.conj(Y_ft)).real[:n]

        # FIST update

        x_new = v - theta_t * g
        x_new = project_simplex(x_new)

        # Update y (interpolation)
        beta_t = (1.0 - theta_t) / (1.0 - alpha_t) if (1.0 - alpha_t) > 1e-10 else 1.0
        y = beta_t * x_new + (1.0 - beta_t) * y
        y = project_simplex(y)

        v = v_new

        # Track best
        obj_x = compute_obj(x_new)
        if obj_x < best_obj - 1e-12:
            best_obj = obj_x
            best_a = x_new.copy()

        it += 1
        if verbose and it % 500 == 0:
            print(f"    SG-A it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

            print(f"  Phase4 PS: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 PS: obj={obj_ps:.8f}")

    # === Phase 5: Accelerated subgradient ===
    print(f"\n=== Phase 5: Accelerated Subgradient ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.10, 120.0)
    if p5_budget > 15:
        a_sga = subgradient_accelerated(best_seq, time_budget=p5_budget,
                                        K=10, verbose=True, seed=300)
        obj_sga = compute_obj(a_sga)
        if obj_sga < best_score:
            best_score = obj_sga
            best_seq = a_sga.copy()
            print(f"  Phase5 SGA: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 SGA: obj={obj_sga:.8f}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 150), max(400, n_current - 50),
        min(2000, n_current + 100), min(2000, n_current + 200),
        min(2000, n_current + 400),
        600, 900, 1000, 1200, 1500, 1800
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_current) > 60]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)

        alloc1 = min(remaining * 0.10, 30.0)
        if alloc1 > 8:
            a_try = smoothed_gd(a_try, time_budget=alloc1,
                                tau_init=10.0, tau_end=250.0, seed=42 + n_try)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.12, 60.0)
        if alloc2 > 10:
            a_try = exponentiated_gradient(a_try, time_budget=alloc2,
                                           tau_init=15.0, tau_end=400.0, seed=100 + n_try)

        remaining = deadline - time.monotonic()
        alloc3 = min(remaining * 0.10, 50.0)
        if alloc3 > 10:
            a_try = generalized_bcd(a_try, time_budget=alloc3, seed=200 + n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Perturbation + refinement ===
    print(f"\n=== Phase 7: Perturbation + Refinement ===")
    remaining = deadline - time.monotonic()

    for r_idx in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        scale = 0.006 + 0.006 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            break
        a_pert = a_pert / s_pert

        alloc = min(remaining * 0.20, 80.0)
        if alloc > 15:
            # Alternate between different methods
            if r_idx % 3 == 0:
                a_pert = generalized_bcd(a_pert, time_budget=alloc, seed=500 + r_idx)
            elif r_idx % 3 == 1:
                a_pert = exponentiated_gradient(a_pert, time_budget=alloc,
                                                tau_init=10.0, tau_end=300.0, seed=600 + r_idx)
            else:
                a_pert = peak_shaving(a_pert, time_budget=alloc, seed=700 + r_idx)

            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 120.0)
        a_final = generalized_bcd(best_seq, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GBCD: obj={best_score:.8f} **")
        else:
            print(f"  Final GBCD: obj={obj_final:.8f}")

    # Final peak shaving
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 60.0)
        a_ps_final = peak_shaving(best_seq, time_budget=alloc, seed=1111)
        obj_ps_final = compute_obj(a_ps_final)
        if obj_ps_final < best_score:
            best_score = obj_ps_final
            best_seq = a_ps_final.copy()
            print(f"  Final PS: obj={best_score:.8f} **")

    # Final smoothed GD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.3, 40.0)
        a_sg_final = smoothed_gd(best_seq, time_budget=alloc,
                                 tau_init=50.0, tau_end=1000.0, seed=2222)
        obj_sg_final = compute_obj(a_sg_final)
        if obj_sg_final < best_score:
            best_score = obj_sg_final
            best_seq = a_sg_final.copy()
            print(f"  Final SGD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
