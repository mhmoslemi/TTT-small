def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a_norm):
    """Build matrix A where (A@g)[k] = conv(a_norm, g)[k]"""
    n = len(a_norm)
    # conv(a,g)[k] = sum_i a[i]*g[k-i] for valid i
    # = sum_j g[j]*a[k-j] for valid j
    # So A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a_norm[idx]
    return A

# Better: vectorized version
def build_conv_matrix_fast(a_norm):
    n = len(a_norm)
    k_idx = np.arange(2*n-1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2*n-1, n))
    A[valid] = a_norm[diff[valid]]
    return A

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_fast(a_norm):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a_norm, g)[k]."""
    n = len(a_norm)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a_norm[diff[valid]]
    return A


def solve_lp_minpeak(a, top_k=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for top_k largest lags (or all), sum(g)=1, g>=0.
    Returns (g, M).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    conv_a = compute_conv(a_norm)
    current_max = float(np.max(conv_a))
    
    if top_k is not None and top_k < 2*n - 1:
        # Only constrain the top_k lags
        k_indices = np.argsort(conv_a)[-top_k:]
        A_full = build_conv_matrix_fast(a_norm)
        A = A_full[k_indices, :]
    else:
        A = build_conv_matrix_fast(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def quadratic_line_search(a, g):
    """
    Quadratic line search: evaluate at 3 points, fit parabola, find minimum.
    Then refine with golden section if needed.
    """
    n = len(a)
    
    # Initial evaluations
    ts = [0.1, 0.3, 0.6, 0.9]
    objs = []
    for t in ts:
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            a_try = a_try / s
            objs.append(compute_obj(a_try))
        else:
            objs.append(np.inf)
    
    best_i = int(np.argmin(objs))
    best_t = ts[best_i]
    best_obj = objs[best_i]
    
    # Parabolic interpolation if possible
    if 0 < best_i < len(ts) - 1:
        t0, t1, t2 = ts[best_i-1], ts[best_i], ts[best_i+1]
        f0, f1, f2 = objs[best_i-1], objs[best_i], objs[best_i+1]
        if f1 < f0 and f1 < f2:
            denom = (f0 - f2) + (f2 - f0) * 0  # simplified
            # Standard parabolic: t* = t1 - 0.5*(t2-t0)*(f0-f1)/(f0-2*f1+f2)
            d = f0 - 2*f1 + f2
            if abs(d) > 1e-15:
                t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / d
                t_par = float(np.clip(t_par, 0.01, 0.99))
                a_par = (1 - t_par) * a + t_par * g
                a_par = np.maximum(a_par, 0)
                s_par = float(np.sum(a_par))
                if s_par > 1e-10:
                    a_par = a_par / s_par
                    f_par = compute_obj(a_par)
                    if f_par < best_obj:
                        best_t = t_par
                        best_obj = f_par
    
    # Try finer search around best
    for t_fine in [best_t - 0.05, best_t - 0.02, best_t + 0.02, best_t + 0.05]:
        if 0 < t_fine < 1:
            a_f = (1 - t_fine) * a + t_fine * g
            a_f = np.maximum(a_f, 0)
            s_f = float(np.sum(a_f))
            if s_f > 1e-10:
                a_f = a_f / s_f
                f_f = compute_obj(a_f)
                if f_f < best_obj:
                    best_t = t_fine
                    best_obj = f_f
    
    return best_t, best_obj


def lp_driven_optimize(a, time_budget=300.0, verbose=True, top_k_frac=0.3):
    """
    LP-driven optimization with adaptive constraint set and quadratic line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    stall_count = 0
    iter_count = 0
    rng = np.random.default_rng(42)
    
    # Adaptive top_k: start small, grow
    top_k = max(50, int(2*n - 1) * top_k_frac)
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break
        
        g, M = solve_lp_minpeak(a, top_k=top_k)
        if g is None:
            break
        
        conv_curr = compute_conv(a)
        max_curr = float(np.max(conv_curr))
        
        # Check if LP can improve
        if M >= max_curr * 0.999:
            # LP says we can't do better than current - try full constraints
            if top_k < 2*n - 1:
                g, M_full = solve_lp_minpeak(a, top_k=None)
                if g is not None:
                    M = M_full
                    if M >= max_curr * 0.999:
                        # Truly at local optimum for this LP
                        stall_count += 1
                        if stall_count >= 3:
                            # Perturb
                            a = a + 0.01 * rng.standard_normal(n)
                            a = np.maximum(a, 0)
                            a = a / a.sum()
                            stall_count = 0
                            continue
                        continue
        
        # Line search
        best_t, obj_at_t = quadratic_line_search(a, g)
        iter_count += 1
        
        if obj_at_t < best_obj - 1e-13:
            a_try = (1 - best_t) * a + best_t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                best_obj = obj_at_t
                best_a = a_try.copy()
                a = a_try
                stall_count = 0
                if verbose and iter_count % 20 == 0:
                    elapsed = time.monotonic() - start
                    print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, time={elapsed:.0f}s")
        else:
            stall_count += 1
            if stall_count >= 5:
                # Try perturb
                a = a + 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0
                if iter_count > 200:
                    break
    
    return best_a


def coordinate_descent(a, time_budget=60.0, verbose=False):
    """
    Coordinate descent: for each coordinate j, find optimal a[j].
    Uses the constraint that max(conv) should decrease.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        improved = False
        for j in range(n):
            if time.monotonic() - start > time_budget:
                break
            
            # Save current value
            old_val = a[j]
            a[j] = 0
            rest_sum = float(np.sum(a))
            rest_a = a.copy()
            
            # Try to find optimal a[j]
            # conv(a+j)*val, a+j)[k] = conv(rest, rest)[k] + val*(rest[k-j] + rest[j-k] if k!=2j else rest[k-j])
            # Actually: conv(a_full, a_full)[k] where a_full = rest + val*e_j
            # = conv(rest, rest)[k] + val*rest[k-j] (if valid) + val*rest[j-k] (if valid, k!=2j) + val^2 (if k=2j)
            
            # Simpler: just try a few values of a[j]
            # a[j] ranges from 0 to rest_sum (to keep sum = 1)
            # Try: current, 0, and a few intermediate values
            
            candidates = [old_val, 0.0, old_val * 0.5, old_val * 1.5, rest_sum * 0.1]
            candidates = [max(0, c) for c in candidates]
            
            best_local = old_val
            best_local_obj = best_obj
            
            for val in candidates:
                a_test = rest_a.copy()
                a_test[j] = val
                s_test = float(np.sum(a_test))
                if s_test < 1e-10:
                    continue
                a_test = a_test / s_test
                obj_test = compute_obj(a_test)
                if obj_test < best_local_obj - 1e-13:
                    best_local_obj = obj_test
                    best_local = val
            
            if best_local != old_val:
                a = rest_a.copy()
                a[j] = best_local
                s = float(np.sum(a))
                a = a / s
                if best_local_obj < best_obj - 1e-13:
                    best_obj = best_local_obj
                    best_a = a.copy()
                    improved = True
        
        if verbose and not improved:
            elapsed = time.monotonic() - start
            print(f"    CD: no improvement, obj={best_obj:.8f}, time={elapsed:.0f}s")
            break
    
    return best_a


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def gd_polish(a, time_budget=100.0, tau_init=50.0, tau_end=1000.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.2 * base_lr), 3.0 * base_lr)
            lr = 0.75 * lr + 0.25 * adapt

        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 5.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

    if verbose:
        print(f"    GD: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def cosine_init(n, p=2.0):
    """Initialize with cosine-based structure."""
    x = np.arange(n)
    a = (1.0 + np.cos(2.0 * np.pi * x / n)) ** p
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    return a


def gaussian_init(n, sigma_frac=0.3):
    """Initialize with Gaussian-like bump."""
    x = np.arange(n)
    center = n / 2.0
    a = np.exp(-0.5 * ((x - center) / (sigma_frac * n)) ** 2)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    return a


def two_peak_init(n, sep=0.25):
    """Initialize with two Gaussian peaks."""
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - n * (0.5 - sep)) / (0.2 * n)) ** 2)
    a += np.exp(-0.5 * ((x - n * (0.5 + sep)) / (0.2 * n)) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    return a


def uniform_init(n):
    return np.ones(n) / n
