        a_segment = a[k_star - j_end : k_star - j_start + 1]
        g[j_start : j_end + 1] = 2.0 * a_segment[::-1]

    e_s = np.zeros(n)
    e_s[s_idx] = 1.0

        best_t = 0.0
        best_obj = compute_obj_normalized(a)
        for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:

            obj_try = compute_obj_normalized(a_try)

                best_obj = obj_try
                best_t = t_val
        a_new = (1 - best_t) * a + best_t * e_s
    else:
        a_new = a  # will be updated by caller

            print(f"      FISTA it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.5f}, t={elapsed:.1f}s")
    
    return best_a


def fw_polish(a, time_budget=100.0, verbose=False, seed=42):
    """Frank-Wolfe polishing with adaptive step sizes and perturbation restarts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0
    
    conv_size = 1
    while conv_size < 2 * n - 1:
        conv_size <<= 1
    
    while time.monotonic() - start < time_budget:
        # Compute subgradient
        fa = rfft(a, conv_size)
        conv = irfft(fa * fa, conv_size)[:2 * n - 1]
        k_star = int(np.argmax(conv))
        
        # g[j] = 2*a[k_star - j] if 0 <= k_star - j < n
        g = np.zeros(n)
        j_start = max(0, k_star - (n - 1))
        j_end = min(n - 1, k_star)
        if j_start <= j_end:
            a_seg = a[k_star - j_end : k_star - j_start + 1]
            g[j_start : j_end + 1] = 2.0 * a_seg[::-1]
        
        s_idx = int(np.argmin(g))
        
        # Try multiple step sizes
        best_t = 0.0
        best_t_obj = best_obj
        step_candidates = [2.0 / (it + 2), 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for t_val in step_candidates:
            t_val = min(t_val, 1.0)
            if t_val <= 1e-10:
                continue
            a_try = (1.0 - t_val) * a + t_val * 0.0
            a_try[s_idx] += t_val
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj - 1e-15:
                best_t_obj = obj_try
                best_t = t_val
        
        if best_t > 1e-10 and best_t_obj < best_obj - 1e-14:
            a = (1.0 - best_t) * a
            a[s_idx] += best_t
            best_obj = best_t_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 30:
                scale = 0.008 + 0.004 * min(no_improve // 30, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                no_improve = 0
                if it > 1000:
                    break
        
        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"      FW it{it}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a
