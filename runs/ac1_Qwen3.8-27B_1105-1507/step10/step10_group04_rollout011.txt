def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    g_val = None
    M_val = None
    dual_weights = None

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.array(g.value).flatten(), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    # Try to extract dual variables for the A@g <= M constraints
                    try:
                        # The dual of A@g <= M gives us weights on each lag
                        dual_arr = np.array(A.value).flatten() if hasattr(A, 'value') else None
                        if dual_arr is not None and dual_arr.size == 2 * n - 1:
                            dual_weights = np.maximum(dual_arr, 0)
                            dw_sum = dual_weights.sum()
                            if dw_sum > 1e-10:
                                dual_weights = dual_weights / dw_sum
                            else:
                                dual_weights = None
                    except Exception:
                        dual_weights = None
                break
        except Exception:
            continue

    return g_val, M_val, dual_weights


def lp_polish(a, time_budget, verbose=False):
    """LP polishing with line-search mixing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5
    iter_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp, _ = solve_lp(a)
        if g_lp is None:
            break

        t_candidates = np.concatenate([
            np.linspace(0.02, 1.0, 25),
            [max(0.01, last_t * 0.7), max(0.01, last_t * 0.9),
             last_t, min(1.0, last_t * 1.1), min(1.0, last_t * 1.3)]
        ])
        t_candidates = np.unique(np.clip(t_candidates, 0.01, 1.0))

        best_t_obj = best_obj
        best_t = last_t
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-13:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t
            stall = 0
            iter_count += 1
            if verbose and iter_count % 5 == 0:
                print(f"    LP iter {iter_count}: obj={best_obj:.8f}, t={best_t:.4f}, t={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 4:
                break

    return best_a


def mirror_descent(a, time_budget, tau_start=100, tau_end=5000, verbose=False):
    """Mirror descent (multiplicative weights) with adaptive tau and line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Adaptive tau: start moderate, increase over time
        tau = tau_start + (tau_end - tau_start) * progress ** 0.6

        # Compute smoothed gradient
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient: grad_j = 2 * sum_k p[k] * a[k-j]
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(a, N)
        g_full = ifft(P * np.conj(A_f)).real
        grad = 2.0 * g_full[:n]

        # Multiplicative update with line search
        # a_new[j] ∝ a[j] * exp(-eta * grad[j])
        # Try multiple eta values
        g_max = float(np.max(grad))
        g_min = float(np.min(grad))
        g_range = g_max - g_min
        if g_range < 1e-15:
            break

        eta_candidates = np.array([0.5 / g_range, 1.0 / g_range, 1.5 / g_range,
                                   2.0 / g_range, 3.0 / g_range, 5.0 / g_range,
                                   1.0 / (g_range * np.sqrt(n / 10.0))])
        eta_candidates = np.clip(eta_candidates, 1e-8, 10.0 / max(g_range, 1e-10))

        best_eta_obj = best_obj
        best_eta = 1.0 / g_range
        best_a_local = best_a.copy()

        for eta in eta_candidates:
            log_update = -eta * (grad - grad.mean())  # center for stability
            log_update = log_update - log_update.max()  # shift for numerical stability
            a_try = a * np.exp(log_update)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            # Ensure non-negativity (should be fine with exp)
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_eta_obj - 1e-14:
                best_eta_obj = obj_try
                best_eta = eta
                best_a_local = a_try.copy()

        if best_eta_obj < best_obj - 1e-14:
            best_obj = best_eta_obj
            best_a = best_a_local.copy()
            a = best_a
            best_it = it
        else:
            # Try smaller steps
            a = a  # keep previous
            stall_check = it - best_it
            if stall_check > 5000:
                a = best_a.copy()
                best_it = it
                it += 10000  # force eventual break

        it += 1

        if verbose and it % 2000 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.0f}s")

        if it > 200000:
            break

    return best_a


def parametric_optimize(n, time_budget, verbose=False):
    """Optimize over parametric family: a[j] = ((j+eps)/(n+eps))^p * ((n-1-j+eps)/(n+eps))^q / Z"""
    best_obj = np.inf
    best_a = None

    # Try a grid of (p, q) values
    p_values = np.concatenate([np.linspace(0.1, 3.0, 20), np.linspace(3.0, 8.0, 10)])
    q_values = np.linspace(0.1, 3.0, 10)

    # First do a coarse grid
    for p in p_values:
        for q in q_values:
            eps = 1e-6
            x = np.arange(n, dtype=np.float64)
            a = ((x + eps) / (n + eps)) ** p * ((n - 1 - x + eps) / (n + eps)) ** q
            s = a.sum()
            if s < 1e-10:
                continue
            a = a / s
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    # Refine locally using scipy
    if time_budget > 5 and best_a is not None:
        from scipy.optimize import minimize

        def param_obj(params):
            p, q = params
            if p < 0.01 or q < 0.01 or p > 20 or q > 20:
                return 1e6
            eps = 1e-6
            x = np.arange(n, dtype=np.float64)
            a = ((x + eps) / (n + eps)) ** p * ((n - 1 - x + eps) / (n + eps)) ** q
            s = a.sum()
            if s < 1e-10:
                return 1e6
            a = a / s
            return compute_obj(a)

        # Find current best p, q from the grid
        best_p, best_q = p_values[0], q_values[0]
        if best_a is not None:
            # Estimate p, q from the best solution
            x = np.arange(n, dtype=np.float64)
            a_pos = np.maximum(best_a, 1e-30)
            log_a = np.log(a_pos)
            log_x = np.log((x + 1) / n)
            log_1mx = np.log((n - x) / n)
            # Linear fit: log_a ≈ p*log_x + q*log_1mx + const
            A_mat = np.column_stack([log_x, log_1mx, np.ones(n)])
            try:
                coeffs, _, _, _ = np.linalg.lstsq(A_mat, log_a, rcond=None)
                best_p = max(0.1, min(10, coeffs[0]))
                best_q = max(0.1, min(10, coeffs[1]))
            except Exception:
                pass

        try:
            res = scipy_minimize(param_obj, [best_p, best_q],
                                 method='Nelder-Mead',
                                 options={'maxiter': 200, 'xatol': 1e-6, 'fatol': 1e-10})
            if res.fun < best_obj:
                p_opt, q_opt = res.x
                eps = 1e-6
                x = np.arange(n, dtype=np.float64)
                best_a = ((x + eps) / (n + eps)) ** p_opt * ((n - 1 - x + eps) / (n + eps)) ** q_opt
                best_a = best_a / best_a.sum()
                best_obj = res.fun
        except Exception:
            pass

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Asymmetric ramp (best for this problem)
        x = np.arange(n, dtype=np.float64)
        a = (x + 1) ** 1.5
        a = a / a.sum()
    elif mode == 3:
        # Stronger ramp
        x = np.arange(n, dtype=np.float64)
        a = (x + 1) ** 2.5
        a = a / a.sum()
    elif mode == 4:
        # Exponential
        a = np.exp(np.arange(n) / (n * 0.5))
        a = a / a.sum()
    elif mode == 5:
        # Double exponential (peaked in middle-left)
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-((x - 0.3 * n) ** 2) / (2 * (0.3 * n) ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Power law with asymmetry
        x = np.arange(n, dtype=np.float64)
        a = ((x + 1) / n) ** 2 * (1 - x / n) ** 0.3
        a = a / a.sum()
    elif mode == 7:
        # Inverse ramp
        x = np.arange(n, dtype=np.float64)
        a = (n - x) ** 1.5
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a


def dual_guided_mirror_descent(a, time_budget, verbose=False):
    """Mirror descent that uses LP dual to guide the smoothing weights."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    lp_interval = 50  # solve LP every 50 iterations

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Every lp_interval iterations, get LP dual for guidance
        if it % lp_interval == 0 and it > 0:
            _, _, dual_w = solve_lp(a)
            if dual_w is not None:
                p = dual_w
            else:
                conv = compute_conv(a)
                max_conv = float(np.max(conv))
                tau = 500 + 5000 * progress
                exp_vals = np.exp(tau * (conv - max_conv))
                sum_exp = float(np.sum(exp_vals))
                if sum_exp < 1e-300:
                    p = np.zeros(len(conv))
                    p[np.argmax(conv)] = 1.0
                else:
                    p = exp_vals / sum_exp
        else:
            conv = compute_conv(a)
            max_conv = float(np.max(conv))
            tau = 200 + 3000 * progress
            exp_vals = np.exp(tau * (conv - max_conv))
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp

        # Compute gradient
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(a, N)
        g_full = ifft(P * np.conj(A_f)).real
        grad = 2.0 * g_full[:n]

        # Multiplicative update
        g_centered = grad - grad.mean()
        g_range = float(np.max(g_centered) - np.min(g_centered))
        if g_range < 1e-15:
            break

        eta = 1.5 / g_range
        log_update = -eta * g_centered
        log_update = log_update - log_update.max()
        a_new = a * np.exp(log_update)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
            else:
                a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Halve the step
            eta *= 0.5
            log_update = -eta * g_centered
            log_update = log_update - log_update.max()
            a_new = a * np.exp(log_update)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
            else:
                a_new = a_new / s_new
                a_new = np.maximum(a_new, 0)
                s_new = float(np.sum(a_new))
                if s_new < 1e-10:
                    a_new = best_a.copy()
                else:
                    a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new

        it += 1

        if verbose and it % 5000 == 0:
            print(f"    DMD it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: parametric init, mirror descent, LP polishing,
    dual-guided refinement, multi-length exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0  # safety margin

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent_loaded = False
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                parent_loaded = True
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass

    # === Phase 0: Parametric Optimization (60s) ===
    print("\n=== Phase 0: Parametric Optimization ===")
    p0_budget = 60.0
    n_init = len(best_seq) if best_seq is not None else 800

    param_a, param_obj = parametric_optimize(n_init, time_budget=p0_budget, verbose=True)
    if param_obj < best_score:
        best_score = param_obj
        best_seq = param_a.copy()
        print(f"  Parametric: obj={best_score:.8f} **")
    else:
        print(f"  Parametric: obj={param_obj:.8f} (no improvement)")

    # Also try parametric at different lengths
    for n_try in [600, 900, 1200, 1500]:
        if time.monotonic() > start_time + p0_budget:
            break
        pa, po = parametric_optimize(n_try, time_budget=10, verbose=False)
        if po < best_score:
            best_score = po
            best_seq = pa.copy()
            print(f"  Parametric n={n_try}: obj={best_score:.8f} **")

    # === Phase 1: Mirror Descent (250s) ===
    print(f"\n=== Phase 1: Mirror Descent ===")
    p1_budget = min(250.0, deadline - time.monotonic() - 60)
    if p1_budget > 30:
        a_md = mirror_descent(best_seq, time_budget=p1_budget,
                              tau_start=100, tau_end=5000, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  MD: obj={best_score:.8f} **")
        else:
            print(f"  MD: obj={obj_md:.8f}")

    # === Phase 2: LP Polishing (300s) ===
    print(f"\n=== Phase 2: LP Polishing ===")
    p2_budget = min(300.0, deadline - time.monotonic() - 60)
    if p2_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p2_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 3: Dual-Guided Mirror Descent (150s) ===
    print(f"\n=== Phase 3: Dual-Guided MD ===")
    p3_budget = min(150.0, deadline - time.monotonic() - 60)
    if p3_budget > 30:
        a_dmd = dual_guided_mirror_descent(best_seq, time_budget=p3_budget, verbose=True)
        obj_dmd = compute_obj(a_dmd)
        if obj_dmd < best_score:
            best_score = obj_dmd
            best_seq = a_dmd.copy()
            print(f"  DMD: obj={best_score:.8f} **")
        else:
            print(f"  DMD: obj={obj_dmd:.8f}")

    # === Phase 4: Another LP Polish (150s) ===
    print(f"\n=== Phase 4: LP Polish Round 2 ===")
    p4_budget = min(150.0, deadline - time.monotonic() - 60)
    if p4_budget > 20:
        a_lp2 = lp_polish(best_seq, time_budget=p4_budget, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP2: obj={best_score:.8f} **")
        else:
            print(f"  LP2: obj={obj_lp2:.8f}")

    # === Phase 5: Multi-n Exploration (200s) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(200.0, remaining * 0.5)
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 200), max(400, n_best - 100),
        n_best + 100, n_best + 200, n_best + 400,
        n_best + 600, n_best + 1000,
        500, 700, 1000, 1500, 2000, 2500, 3000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 4000 and n != n_best]
    n_options = n_options[:10]
    print(f"  Trying n values: {n_options}")

    per_n_budget = p5_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(per_n_budget, remaining - 30)
        if alloc < 30:
            break

        # Resample from best
        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Quick LP polish
        lp_alloc = alloc * 0.5
        if lp_alloc > 15:
            a_try = lp_polish(a_try, time_budget=lp_alloc, verbose=False)

        # Mirror descent
        remaining = deadline - time.monotonic()
        md_alloc = min(alloc * 0.4, remaining - 20)
        if md_alloc > 10:
            a_try = mirror_descent(a_try, time_budget=md_alloc,
                                   tau_start=50, tau_end=2000, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts (80s) ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p6_budget = min(80.0, remaining * 0.2)
    n_current = len(best_seq)
    n_rest = max(1, int(p6_budget / 30))

    for r in range(n_rest):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(30, remaining - 10)
        if alloc < 15:
            break

        mode = r % 8
        a_r = diverse_init(n_current, rng, mode=mode)
        # Blend with best (weighted toward best)
        blend = 0.4 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        # LP polish
        a_r = lp_polish(a_r, time_budget=alloc * 0.7, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish (remaining) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s")

    # LP round
    if remaining > 30:
        alloc = min(remaining * 0.3, 120.0)
        a_final = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # MD round
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 80.0)
        a_final = mirror_descent(best_seq, time_budget=alloc,
                                 tau_start=500, tau_end=8000, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # DMD round
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 60.0)
        a_final = dual_guided_mirror_descent(best_seq, time_budget=alloc, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final DMD: obj={best_score:.8f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 50.0)
        a_final = lp_polish(best_seq, time_budget=alloc, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra-final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(best_seq, 0.0)
    s_final = float(np.sum(best_seq))
    if s_final < 0.01:
        best_seq = np.ones(100)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
