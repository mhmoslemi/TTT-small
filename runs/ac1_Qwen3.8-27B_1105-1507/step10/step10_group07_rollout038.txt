import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)

    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

        for j in range(j_start, j_end + 1):
            A[ii, j] = a[k - j]
    return A


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_tr_sscp_topk(a, r, K=30):
    """Trust region SOCP using only top-K peak constraints for speed."""
    n = len(a)
    a = a / a.sum()
    conv = compute_conv(a)
    top_k = np.argsort(conv)[::-1][:K]

    A_sub = build_conv_matrix_rows(a, top_k)
    C_sub = conv[top_k]

    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta

    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_sub + 2 * (A_sub @ delta) <= M,
        cp.norm(delta, 2) <= r,
    ]

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None


def solve_tr_sscp_full(a, r):
    """Trust region SOCP with all constraints (slower but more accurate)."""
    n = len(a)
    a = a / a.sum()
    conv = compute_conv(a)
    A = build_conv_matrix(a)

    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta

    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        conv + 2 * (A @ delta) <= M,
        cp.norm(delta, 2) <= r,
    ]

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_direction(a, M_target=None):
    """LP: min M s.t. A@g <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def smoothed_gradient_descent(a, time_budget=80.0, tau_init=15.0, tau_end=300.0):
    """Smoothed gradient descent (Nesterov-smoothed max) with simplex projection."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 4.0 / np.sqrt(n)
    momentum = 0.85
    v = np.zeros(n)
    y = a.copy()  # Nesterov momentum position
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        Y = fft(y, N)
        g_full = ifft(P * np.conj(Y)).real
        g = 2.0 * g_full[:n]

        # Nesterov updated
        t_k = 2.0 / (it + 3.0)
        y_new = a - lr * v
        y = y_new
        v = (it / (it + t_k + 1.0)) * v + g

        a_new = project_simplex(y - lr * v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

    return best_a


def tr_sscp_polish(a, time_budget=300.0, r_init=0.25, K_init=15, K_max=50, verbose=False, seed=42):
    """Top-K Trust Region SOCP polishing with adaptive K and radius."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    r = r_init
    K = K_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Adaptive K: increase as we converge
        if no_improve > 5:
            K = min(K_max, K + 5)
        elif no_improve == 0:
            K = max(K_init, K - 2)

        # Use full constraints for small K or when r is very small
        if r < 0.05 and K >= 40:
            delta_dir, M_val = solve_tr_sscp_full(a, r)
        else:
            delta_dir, M_val = solve_tr_sscp_topk(a, r, K=K)

        if delta_dir is None:
            g_lp, M_val = solve_lp_direction(a)
            if g_lp is None:
                break
            delta_dir = g_lp - a

        norm_d = np.linalg.norm(delta_dir)
        if norm_d > r:
            delta_dir = delta_dir * (r / norm_d)

        finite_mask = np.isfinite(delta_dir)
        if not finite_mask.all():
            delta_dir = np.where(finite_mask, delta_dir, 0.0)

        # Line search
        best_alpha = 0
        best_alpha_obj = best_obj
        alphas = [0.03, 0.08, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0, 1.5, 2.0]
        for t_alpha in alphas:
            a_try = a + t_alpha * delta_dir
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj - 1e-15:
                best_alpha_obj = obj_try
                best_alpha = t_alpha

        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-15:
            a = a + best_alpha * delta_dir
            a = np.maximum(a, 0)
            s_n = a.sum()
            if s_n > 1e-10:
                a = a / s_n
            best_obj = best_alpha_obj
            best_a = a.copy()
            no_improve = 0
            r = min(r * 1.08, 1.5)
        else:
            no_improve += 1
            r *= 0.6
            if no_improve >= 8:
                if r < 0.02:
                    # Perturb and restart
                    scale = 0.005 + 0.003 * min(no_improve, 10)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    s_p = a.sum()
                    if s_p > 1e-10:
                        a = a / s_p
                    r = r_init
                    K = K_init
                    no_improve = 0
                    if it > 100:
                        break
                else:
                    r = max(r * 0.7, 0.02)

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    TR it{it}: obj={best_obj:.8f}, r={r:.4f}, K={K}, t={elapsed:.0f}s")

    return best_a


def lp_polish(a, time_budget=200.0, verbose=False, seed=42):
    """LP-based polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        best_t = 0
        best_t_obj = best_obj
        for t_val in [0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]:
            a_try = (1 - t_val) * best_a + t_val * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-15:
                best_t_obj = obj_try
                best_t = t_val

        if best_t > 0 and best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = (1 - best_t) * best_a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            s_ba = best_a.sum()
            if s_ba > 1e-10:
                best_a = best_a / s_ba
            a = best_a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                scale = 0.005 + 0.004 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                no_improve = 0
                if it > 80:
                    break

        it += 1
        if verbose and it % 8 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def subgradient_refine(a, time_budget=60.0, K=10):
    """Subgradient method with top-K peak weighting."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 2.5 / np.sqrt(n)
    momentum = 0.75
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break

        weights = np.exp((peak_vals - max_pv) * 10.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 150:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.998

        it += 1

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def coordinate_refine(a, time_budget=40.0):
    """Simple coordinate refinement: try small multiplicative adjustments."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    factors = [0.5, 0.7, 0.85, 1.15, 1.3, 2.0]

    while time.monotonic() - start < time_budget:
        improved = False
        # Random order of coordinates
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            old_val = a[i]
            if old_val < 1e-12:
                continue
            for f in factors:
                a_try = a.copy()
                a_try[i] = old_val * f
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
        if not improved:
            break

    return best_a

        obj_flp = compute_obj(a_final_lp)
        if obj_flp < best_score:
            best_score = obj_flp
            best_seq = a_final_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final coordinate refinement
    remaining = deadline - time.monotonic()
    if remaining > 8:
        a_cd_final = coordinate_refine(best_seq, time_budget=min(remaining * 0.5, 20.0))
        obj_cd_final = compute_obj(a_cd_final)
        if obj_cd_final < best_score:
            best_score = obj_cd_final
            best_seq = a_cd_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
