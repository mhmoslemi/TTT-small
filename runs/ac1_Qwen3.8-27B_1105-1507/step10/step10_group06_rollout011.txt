def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    """Cross convolution a*b via FFT."""
    n = len(a)

**Alternative: Use projected gradient descent for the best response**

The best response subproblem is:
min M
s.t. (a*g)[k] <= M for all k
     sum(g) = 1
     g >= 0

This is equivalent to:
min max_k (a*g)[k]
s.t. sum(g) = 1, g >= 0

We can solve this with gradient descent on the smoothed objective:
f_tau(g) = (1/tau) * log(sum_k exp(tau * (a*g)[k]))

The gradient is:
grad f_tau(g) = A^T @ (exp(tau * (a*g)) / sum_k exp(tau * (a*g)))

where A is the convolution matrix.

We can compute this efficiently using FFT:
1. Compute u = (a*g) via FFT
2. Compute p = softmax(tau * u)
3. Compute grad = A^T @ p via FFT (cross-correlation of a with p)

Then do projected gradient descent (project onto the simplex).

This should be much faster than LP for each iteration, and we can run many more iterations.

"""

Uses a hybrid approach: fast FFT-accelerated smoothed gradient best-response
for rapid iteration, periodic LP refinement for precision, Nesterov-style

(2) adaptive tau scheduling for the smoothing parameter, (3) dual-space
stalling detection, (4) structured parametric initializations (raised
cosine, Gaussian, sech profiles), and (5) aggressive multi-n exploration
with time-aware allocation.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Cross convolution a*b via FFT. Returns array of length n+m-1."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def best_response_gradient(a, tau, n_iter, g_init=None, verbose=False):
    """
    Find g minimizing max_k (a*g)[k] s.t. g>=0, sum(g)=1
    using smoothed gradient descent with adaptive step size.
    Returns (g, peak_value).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    if g_init is not None:
        g = np.maximum(g_init, 0).copy()
        g = project_simplex(g)
    else:
        g = np.ones(n) / n
    
    # Precompute reversed a for gradient computation
    a_rev = a_norm[::-1]
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    A_rev = rfft(a_rev, N)
    
    # Adaptive tau schedule within the best response computation
    tau_current = tau
    best_g = g.copy()
    best_peak = float(np.max(compute_conv_cross(a_norm, g)))
    
    for it in range(n_iter):
        # Smoothly increase tau
        tau_current = min(tau * (1.0 + 0.05 * it), tau * 5.0)
        
        # Compute (a*g) via FFT
        conv_ag = compute_conv_cross(a_norm, g)
        peak = float(np.max(conv_ag))
        
        if peak < best_peak - 1e-12:
            best_peak = peak
            best_g = g.copy()
        
        # Compute softmax weights
        max_conv = peak
        exp_vals = np.exp(tau_current * (conv_ag - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv_ag))
            p[np.argmax(conv_ag)] = 1.0
        else:
            p = exp_vals / sum_exp
        
        # Gradient: (A^T p)[i] = sum_k p[k] * a[k-i]
        # = conv(p, a_rev)[n-1+i]
        P = rfft(p, N)
        grad_full = irfft(P * A_rev, N)
        grad = grad_full[n - 1:n - 1 + n]
        
        # Learning rate: 1/L where L ~ tau * max(a_norm)^2 * len(valid)
        # Use adaptive step
        lr = 2.0 / (tau_current * (float(np.sum(a_norm ** 2) * n) + 1e-10))
        lr = min(lr, 0.5 / (float(np.linalg.norm(grad)) + 1e-10))
        
        # Gradient step
        g_new = g - lr * (grad - float(np.sum(grad) / n))
        g_new = project_simplex(g_new)
        
        # Check improvement
        conv_new = compute_conv_cross(a_norm, g_new)
        peak_new = float(np.max(conv_new))
        
        if peak_new < peak + 1e-12:
            g = g_new
        else:
            # Backtracking
            lr *= 0.5
            g_new = g - lr * (grad - float(np.sum(grad) / n))
            g_new = project_simplex(g_new)
            g = g_new
    
    conv_final = compute_conv_cross(a_norm, best_g)
    return best_g, float(np.max(conv_final))


def build_conv_matrix_sparse(a_norm):
    """Build sparse convolution matrix A where A[k,j] = a_norm[k-j] if valid."""
    n = len(a_norm)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            idx = k - j
            if idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(a_norm[idx])
    row_arr = np.array(rows)
    col_arr = np.array(cols)
    val_arr = np.array(vals)
    A = csc_matrix((val_arr, (row_arr, col_arr)), shape=(2 * n - 1, n))
    return A


def solve_best_response_lp(a, warm_g=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_sparse(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    if warm_g is not None:
        g.value = np.maximum(warm_g, 0)
        g.value /= g.value.sum()
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def parametric_init(n, profile='raised_cosine', param=1.0):
    """Generate a parametric initial sequence."""

        a = 1.0 / np.cosh(param * t)

        a = np.ones(n)

        a = 1.0 - np.abs(t) ** param
    elif profile == 'w教育是':
        a = np.ones(n)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:

def fourier_perturb(a, rng, strength=0.03, n_modes=8):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp
        
        # Gradient
        P = rfft(p, N)
        grad_full = irfft(P * A_rev, N)
        grad = grad_full[n - 1:n - 1 + n] * 2.0  # factor of 2 for symmetrization
        
        # Adaptive LR
        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            adaptive_lr = 1.0 / (tau * g_norm + 1e-10)
            lr = 0.7 * lr + 0.3 * min(max(adaptive_lr, 0.1 * base_lr), 5.0 * base_lr)
        
        v = beta * v - lr * grad
        a_new = y + v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
            lr *= 0.8
        
        tau = min(tau * 1.03, tau_max)
        lr *= 0.9999
        it += 1

    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def mixed_population(a, b, rng, n_points=5):
    """Generate mixing profiles between a and b."""
    results = []
    for t in np.linspace(0, 1, n_points):
        mixed = (1 - t) * a + t * b
        mixed = np.maximum(mixed, 0)
        s = mixed.sum()
        if s > 1e-10:
            results.append(mixed / s)
        else:
            results.append(a.copy())
    return results


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining gradient best-response, LP refinement,
    parametric initializations, multi-length exploration, and gradient polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0)
        if parent.size == 0 or parent.sum() < 0.01:
            raise ValueError("Invalid parent")
        parent = parent / parent.sum()
        n_parent = parent.size
        print(f"Parent: n={n_parent}, obj={compute_obj(parent):.10f}")
    except Exception:
        n_parent = 800
        parent = np.ones(n_parent) / n_parent
        print(f"Default parent: n={n_parent}")

    best_a = parent.copy()
    best_obj = compute_obj(parent)
    print(f"Initial best: obj={best_obj:.10f}")

    # === Strategy: Try multiple (n, initialization) combinations ===
    
    # Define n values to try (prioritize near parent length)
    n_candidates = []
    n_base = n_parent if 400 <= n_parent <= 2000 else 800
    for offset in [0, -100, 100, -200, 200, -50, 50, -300, 300, 400, -400, 500, -500, 600]:
        n_try = n_base + offset
        if 400 <= n_try <= 2000 and n_try not in n_candidates:
            n_candidates.append(n_try)
    
    # Also add some systematic values
    for n_sys in [500, 700, 900, 1100, 1300, 1500, 1800]:
        if 400 <= n_sys <= 2000 and n_sys not in n_candidates:
            n_candidates.append(n_sys)
    
    n_candidates = sorted(set(n_candidates))
    # Limit to at most 12 values
    n_candidates = n_candidates[:12]
    
    print(f"Will try n values: {n_candidates}")
    
    # Time allocation: give more time to n values closer to parent
    total_opt_time = min(budget_s - 120, 1300)  # reserve time for final polish
    # Weight by proximity to parent
    weights = []
    for n in n_candidates:
        w = 1.0 / (1 + 0.001 * (n - n_parent) ** 2)
        weights.append(w)
    total_w = sum(weights)
    weights = [w / total_w * total_opt_time for w in weights]
    # Cap individual allocations
    weights = [min(w, total_opt_time * 0.35) for w in weights]
    
    # Also try parametric initializations for the best n
    profiles = ['raised_cosine', 'gaussian', 'sech', 'uniform', 'triangle']
    params = [0.5, 1.0, 1.5, 2.0, 3.0]
    
    trials = []  # (n, init_array, time_alloc)
    
    for idx, n in enumerate(n_candidates):
        t_alloc = weights[idx]
        if t_alloc < 30:
            continue
        
        # Parent-based initialization
        if n == n_parent:
            a_init = parent.copy()
        else:
            a_init = resample_to_length(parent, n)
        a_init = a_init / a_init.sum()
        trials.append((n, a_init, t_alloc * 0.5, 'parent'))
        
        # Parametric initializations (only for a few n values)
        if idx < 4:  # only for first 4 n values
            for prof in profiles[:3]:
                for param in params[1:3]:
                    a_p = parametric_init(n, prof, param)
                    trials.append((n, a_p, t_alloc * 0.15, f'{prof}_{param}'))
    
    # Sort trials by allocated time (do longest first)
    trials.sort(key=lambda x: -x[2])
    
    # Execute trials
    trial_count = 0
    for n, a_init, t_alloc, label in trials:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if t_alloc < 30:
            continue
        
        trial_count += 1
        print(f"\n--- Trial {trial_count}: n={n}, init={label}, time={t_alloc:.0f}s ---")
        
        # Run the main optimization
        result = lp_alternate_improved(
            a_init, time_budget=t_alloc,
            seed=seed + trial_count * 100, verbose=(trial_count <= 3)
        )
        
        # Quick gradient polish
        remaining = deadline - time.monotonic()
        if remaining > 30:
            polish_time = min(remaining * 0.15, t_alloc * 0.2, 60.0)
            if polish_time > 10:
                result = gradient_polish(result, time_budget=polish_time,
                                          tau_init=50.0, tau_max=500.0)
        
        obj = compute_obj(result)
        if obj < best_obj:
            best_obj = obj
            best_a = result.copy()
            print(f"  *** NEW BEST: obj={best_obj:.10f} (n={n}, {label}) ***")
        else:
            print(f"  obj={obj:.10f} (no improvement, best={best_obj:.10f})")
        
        # Check time
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break
    
    # === Final refinement on best ===
    print(f"\n=== Final Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        # LP refinement
        lp_time = min(remaining * 0.5, 200.0)
        if lp_time > 20:
            result = lp_alternate_improved(
                best_a, time_budget=lp_time,
                seed=seed + 99999, verbose=False
            )
            obj = compute_obj(result)
            if obj < best_obj:
                best_obj = obj
                best_a = result.copy()
                print(f"  LP refinement: obj={best_obj:.10f} **")
            else:
                print(f"  LP refinement: obj={obj:.10f}")
        
        # Gradient polish
        remaining = deadline - time.monotonic()
        if remaining > 30:
            gp_time = min(remaining * 0.5, 100.0)
            if gp_time > 15:
                result = gradient_polish(best_a, time_budget=gp_time,
                                          tau_init=100.0, tau_max=2000.0)
                obj = compute_obj(result)
                if obj < best_obj:
                    best_obj = obj
                    best_a = result.copy()
                    print(f"  Gradient polish: obj={best_obj:.10f} **")
                else:
                    print(f"  Gradient polish: obj={obj:.10f}")
    
    # === Try a few more n values if time permits ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Additional n Exploration ===")
        n_best = len(best_a)
        extra_ns = [n_best - 20, n_best + 20, n_best + 50, n_best - 50]
        extra_ns = [n for n in extra_ns if 400 <= n <= 2000 and n != n_best]
        
        for n_try in extra_ns[:3]:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            a_try = resample_to_length(best_a, n_try)
            t_alloc = min(remaining * 0.3, 100.0)
            if t_alloc < 30:
                break
            result = lp_alternate_improved(a_try, time_budget=t_alloc * 0.7,
                                            seed=n_try, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                result = gradient_polish(result, time_budget=min(remaining * 0.3, 30.0),
                                          tau_init=50.0, tau_max=500.0)
            obj = compute_obj(result)
            if obj < best_obj:
                best_obj = obj
                best_a = result.copy()
                print(f"  ** n={n_try}: obj={best_obj:.10f} **")
            else:
                print(f"  n={n_try}: obj={obj:.10f}")
    
    # Final validation
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(200)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
