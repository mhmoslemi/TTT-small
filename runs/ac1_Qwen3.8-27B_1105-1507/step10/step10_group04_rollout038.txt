In fictitious play:

- Player A (the "a-player") chooses `a` in the simplex to minimize.
- Player P (the "p-player") chooses `p` in the simplex to maximize.
- The payoff to player P is `sum_k p[k] q_k(a) = a^T Q_p a`.

Fictitious play:
- Maintain `p = average of past "best response" lags`.
- At each step:

  2. Player P's best response: `p* = e_k` where `k = argmax q_k(a*)` (put all mass on the worst lag).

- `p ← softmax of (log(p) - eta * grad_p)` (multiplicative weights)

    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break

        q = compute_conv(a_new)

            best_obj = obj
            best_a = a_new.copy()

        # Blend

        a = a_new
    
    return best_a

        if outside:
            out_arr = np.array(outside, dtype=np.int64)
            min_grad = float(np.min(grad[out_arr]))
            if min_grad >= mu - 1e-8:
                # KKT satisfied
                a_result = np.zeros(n)
                a_result[T_arr] = a_T
                total = a_result.sum()
                if total > 1e-15:
                    a_result /= total
                return a_result, T
            # Add violators
            viol = out_arr[grad[out_arr] < mu - 1e-8]
            # Add up to 20 most violating
            order = np.argsort(grad[out_arr])
            to_add = out_arr[order[:min(20, len(viol))]].tolist()
            T = sorted(set(T) | set(to_add))
        else:
            a_result = np.zeros(n)
            a_result[T_arr] = a_T
            total = a_result.sum()
            if total > 1e-15:
                a_result /= total
            return a_result, T
    
    # Fallback: use last a_T
    if last_a_T is not None:
        a_result = np.zeros(n)
        T_arr = np.array(T, dtype=np.int64)
        a_result[T_arr] = np.maximum(last_a_T, 0)
        total = a_result.sum()
        if total > 1e-15:
            a_result /= total
    return a_result, T


def game_optimize(a, time_budget, gamma_init=20.0, gamma_rate=1.03, gamma_max=1000.0,
                  verbose=True):
    """
    Fictitious play: alternate QP solve and p update.
    """
    n = len(a)
    start = time.monotonic()
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Init p
    q = compute_conv(a)
    M = q.max()
    gamma = gamma_init
    p = np.exp(gamma * (q - M))
    p /= p.sum()
    
    last_support = np.where(a > 1e-10)[0].tolist()
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget - 3:
        elapsed = time.monotonic() - start
        
        # Solve inner QP
        a_new, support = solve_qp_active_set(p, n, init_support=last_support)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            s_new = 1.0
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and it % 30 == 0:
            print(f"    Game it{it}: obj={best_obj:.8f}, |T|={len(support)}, "
                  f"gamma={gamma:.0f}, t={elapsed:.0f}s")
        
        # Update p
        q = compute_conv(a_new)
        M = q.max()
        p = np.exp(gamma * (q - M))
        ps = p.sum()
        if ps < 1e-300:
            p_new = np.zeros(len(q))
            p_new[np.argmax(q)] = 1.0
        else:
            p_new = p / ps
        
        # Blend p for stability
        p = 0.7 * p + 0.3 * p_new
        p /= p.sum()
        
        # Update support
        last_support = [int(i) for i in np.where(a_new > 1e-10)[0]]
        if len(last_support) < 2:
            last_support = [int(i) for i in np.argsort(a_new)[-2:]]
        
        # Increase gamma
        gamma = min(gamma_max, gamma * gamma_rate)
        
        # Restart if stalled too long
        if stall > 50:
            a_new = best_a.copy()
            q = compute_conv(a_new)
            M = q.max()
            gamma = gamma_init
            p = np.exp(gamma * (q - M))
            p /= p.sum()
            last_support = [int(i) for i in np.where(a_new > 1e-10)[0]]
            stall = 0
        
        it += 1
    
    return best_a
