"""
Hybrid optimization: exact coordinate descent + LP polishing + subgradient methods.
Key innovation: 1D exact minimization per coordinate for peak autocorrelation,
combined with LP directional updates and multi-resolution n exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def coord_descent(a, max_rounds=30, time_budget=120.0, verbose=False):
    """
    Coordinate descent with exact 1D minimization per coordinate.
    a should be non-negative; will be normalized to sum=1.
    Minimizes max_k conv(a,a)[k] (equivalent to the normalized objective).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv_a = compute_conv(a)
    best_obj = float(np.max(conv_a))
    best_a = a.copy()
    start = time.monotonic()
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    q_len = 2 * n - 1

    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for m in range(n):
            # Compute coefficient vector c[k] = 2*a[k-m] for valid k
            c = np.zeros(q_len)
            k_start = m
            k_end = min(m + n, q_len)
            c[k_start:k_end] = 2.0 * a[:k_end - m]

            q_pos = 2 * m
            has_quad = (q_pos < q_len)

            # Define 1D objective
            def f_at(delta):
                one_plus_d = 1.0 + delta
                denom = one_plus_d * one_plus_d
                numer = conv_a + c * delta
                if has_quad:
                    numer = numer.copy()
                    numer[q_pos] += delta * delta
                return float(np.max(numer) / denom)

            # Golden section search
            lo = -a[m] + 1e-10
            hi = max(2.0 * a[m] + 0.1, 0.5)

            c_gs = lo + gr * (hi - lo)
            d_gs = lo + (1.0 - gr) * (hi - lo)
            fc = f_at(c_gs)
            fd = f_at(d_gs)

            for _ in range(35):
                if fc < fd:
                    hi = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = lo + gr * (hi - lo)
                    fc = f_at(c_gs)
                else:
                    lo = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = lo + gr * (hi - lo)
                    fd = f_at(d_gs)

            delta_opt = c_gs if fc < fd else d_gs
            obj_opt = min(fc, fd)

            # Also try delta=0 and a few heuristics
            obj_zero = f_at(0.0)
            if obj_zero < obj_opt:
                delta_opt = 0.0
                obj_opt = obj_zero

            if obj_opt < best_obj - 1e-11 and abs(delta_opt) > 1e-12:
                a[m] += delta_opt
                if a[m] < 0:
                    a[m] = 0.0
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                    conv_a = compute_conv(a)
                    actual_obj = float(np.max(conv_a))
                    best_obj = actual_obj
                    best_a = a.copy()
                    improved = True

        if verbose and (rnd % 5 == 0 or rnd == max_rounds - 1):
            print(f"    CD round {rnd}: max_conv={best_obj:.10f}, improved={improved}, t={time.monotonic()-start:.1f}s")

        if not improved:
            break

    return best_a


def subgradient_descent(a, time_budget=30.0):
    """Subgradient descent with momentum on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = float(np.max(compute_conv(a)))
    start = time.monotonic()
    lr = 0.015
    momentum = np.zeros(n)
    beta = 0.6
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))

        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]

        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        lr *= 0.999
        it += 1

        if it - best_it > 3000:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = 0.008
            best_it = it

    return best_a


def line_search_blend(a, g_dir, time_budget=3.0):
    """Golden section line search for optimal blend (1-t)*a + t*g_dir."""
    lo, hi = 0.0, 1.5
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi

    def obj_at(t):
        a_try = (1 - t) * a + t * g_dir
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return float(np.max(compute_conv(a_try / s)))

    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    start = time.monotonic()
    for _ in range(20):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)

    for t_try in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.2, 1.4]:
        obj_try = obj_at(t_try)
        if obj_try < obj_best:
            obj_best = obj_try
            t_best = t_try

    return t_best, obj_best


def lp_polish(a, time_budget=120.0, max_iters=60, verbose=False):
    """LP-based directional polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = float(np.max(compute_conv(a)))
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(777)
    idx_arr = np.arange(n, dtype=np.float64) / max(1, n - 1)

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=min(2.5, remaining * 0.05))

        if t_opt > 0 and obj_opt < best_obj - 1e-12:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = float(np.max(compute_conv(a_new)))
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if verbose and it % 10 == 0:
            print(f"      LP it{it}: obj={best_obj:.10f}, t={t_opt:.4f}, t={time.monotonic()-start:.1f}s")

        if stall >= 3:
            if stall <= 5:
                freq = rng.integers(1, max(2, n // 40))
                phase = rng.uniform(0, 2 * np.pi)
                pert = 0.015 * stall * np.cos(2 * np.pi * freq * idx_arr + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
            elif stall <= 8:
                noise = rng.normal(0, 0.02 * stall, n)
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
            else:
                x = np.linspace(0, 1, n)
                a_div = 1.0 / (0.05 + 3.0 * (2 * x - 1) ** 2)
                a = 0.4 * a_div + 0.6 * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()
            stall = 0

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = 1.0 / (0.05 + 3.0 * (2 * x - 1) ** 2)
    elif mode == 1:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 2:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 3:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 4:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 5:
        x2 = np.linspace(0.01, 0.99, n)
        a = 1.0 / np.sqrt(1 - (2 * x2 - 1) ** 2)
    elif mode == 6:
        a = 1.0 / np.cosh(2.0 * (2 * x - 1) - np.log(2))
    else:
        a = 1.0 / (0.03 + 5.0 * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

    a_try = np.interp(x_new, x_old, a_best)

    if remaining2 > 10:
        a_try = coord_descent(a_try, max_rounds=5, time_budget=min(remaining2 * 0.5, 20.0))

    return a_try
