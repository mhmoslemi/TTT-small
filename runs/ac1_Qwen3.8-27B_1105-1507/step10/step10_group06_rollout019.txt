def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        lr *= 0.9996
        tau = min(tau * 1.035, tau_max)
        it += 1

        if stall > 300:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr
            tau = max(tau_init, tau * 0.4)
            stall = 0

    return best_a


def fourier_perturb(a, rng, strength=0.05):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(15, n // 3)
    for k in range(1, n_modes + 1):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.4, 1.6) / (k * 0.6)
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_phase_perturb(a, rng, strength=0.1):
    """Perturb using random phase shifts in Fourier domain."""
    n = len(a)
    a_hat = fft(a / a.sum(), n)
    for k in range(1, n // 2 + 1):
        angle = rng.uniform(-strength, strength)
        a_hat[k] *= np.exp(1j * angle)
        if k < n - k:
            a_hat[n - k] *= np.exp(-1j * angle)
    a_new = ifft(a_hat).real
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def block_perturb(a, rng, block_size=20, strength=0.1):
    """Perturb random blocks of the sequence."""
    n = len(a)
    a_new = a.copy()
    n_blocks = max(1, n // block_size)
    for _ in range(n_blocks // 4):
        start = rng.integers(0, max(1, n - block_size))
        end = min(start + block_size, n)
        scale = rng.uniform(1.0 - strength, 1.0 + strength)
        a_new[start:end] *= scale
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_alternate(a, time_budget, seed=42, verbose=False):
    """
    LP-based alternating minimization with adaptive blending and stall detection.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    t_blend = 0.4
    stall_count = 0
    total_lp_time = 0.0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_conv_lp(a)
        lp_time = time.monotonic() - t0
        total_lp_time += lp_time

        if g_lp is None:
            break

        iter_count += 1

        candidates_t = [t_blend, t_blend * 0.5, min(t_blend * 1.5, 1.0),
                        0.1, 0.25, 0.5, 0.7, 1.0]
        if iter_count > 30:
            candidates_t.extend([0.05, 0.02, 0.01])

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend

        for t in candidates_t:
            t = min(max(t, 0.001), 1.0)
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            t_blend = max(0.01, best_local_t * 0.93)
            stall_count = 0
            if verbose and iter_count % 25 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.10f} t={t_blend:.3f} "
                      f"M={M_lp:.6f} lp_t={lp_time:.3f}s")
        else:
            stall_count += 1
            if stall_count >= 6:
                method = rng.integers(0, 3)
                if method == 0:
                    a = fourier_perturb(best_a, rng, strength=0.03 + 0.01 * min(stall_count - 5, 5))
                elif method == 1:
                    a = random_phase_perturb(best_a, rng, strength=0.05 + 0.02 * min(stall_count - 5, 5))
                else:
                    a = block_perturb(best_a, rng, block_size=30,
                                      strength=0.05 + 0.02 * min(stall_count - 5, 5))
                t_blend = max(0.05, min(t_blend * 1.4, 0.7))
                stall_count = 0
            elif stall_count >= 2:
                a = best_a + 0.008 * rng.standard_normal(n) * (best_a + 1.0 / n)
                a = np.maximum(a, 0)
                a = a / a.sum()

    if verbose:
        print(f"    LP done: {iter_count} iters, obj={best_obj:.10f}, total_lp={total_lp_time:.1f}s")

    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] / a[0] if a[0] != 0 else 1.0 / new_n)
    a = a / a.sum()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def spectral_init(n, rng, num_modes=8):
    """Create a spectral initial sequence with controlled Fourier content."""
    a = np.ones(n) / n
    for k in range(1, num_modes + 1):
        freq = k * 2.0 * np.pi / n
        amp = rng.uniform(0.02, 0.15) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(freq * np.arange(n) + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: combines LP alternating minimization, Nesterov gradient
    polishing, structured perturbations, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    # Initialize from parent
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0)
        if a.size == 0 or float(a.sum()) < 0.01:
            raise ValueError("Invalid parent")
    except Exception:
        a = np.ones(800) / 800

    best_a = a.copy()
    best_obj = compute_obj(a)
    n_init = len(a)
    print(f"Initial: n={n_init}, obj={best_obj:.10f}")

    rng = np.random.default_rng(seed)

    # === Phase 1: Primary LP Optimization (45% of time) ===
    print("\n=== Phase 1: Primary LP Optimization ===")
    p1_budget = (deadline - time.monotonic()) * 0.45
    a_p1 = lp_alternate(best_a, time_budget=p1_budget, seed=seed, verbose=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1.copy()
        print(f"  Phase 1: obj={best_obj:.10f} **")
    else:
        print(f"  Phase 1: obj={obj_p1:.10f} (no improvement)")

    # === Phase 2: Gradient Polishing (20% of time) ===
    print("\n=== Phase 2: Gradient Polishing ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = min(remaining * 0.2, 200.0)
        a_p2 = nesterov_polish(best_a, time_budget=p2_budget,
                               tau_init=30.0, tau_max=3000.0)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_obj:
            best_obj = obj_p2
            best_a = a_p2.copy()
            print(f"  Phase 2: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 2: obj={obj_p2:.10f}")

    # === Phase 3: Second LP pass with perturbation (20% of time) ===
    print("\n=== Phase 3: Second LP Pass ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        p3_budget = min(remaining * 0.2, 250.0)
        a_pert = fourier_perturb(best_a, rng, strength=0.015)
        a_p3 = lp_alternate(a_pert, time_budget=p3_budget, seed=seed + 1111, verbose=True)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3.copy()
            print(f"  Phase 3: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 3: obj={obj_p3:.10f}")

    # === Phase 4: Multi-length exploration (10% of time) ===
    print("\n=== Phase 4: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    n_options = sorted(set([
        n_best - 50, n_best - 100, n_best + 50, n_best + 100,
        n_best - 200, n_best + 200,
        600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and n != n_best]
    # Sort by distance from current best
    n_options.sort(key=lambda x: abs(x - n_best))

    for n_try in n_options[:3]:
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break

        a_try = resample_to_length(best_a, n_try)
        alloc = min(remaining * 0.25, 150.0)
        if alloc < 40:
            break

        a_try = lp_alternate(a_try, time_budget=alloc * 0.6, seed=n_try, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_try = nesterov_polish(a_try, time_budget=min(remaining * 0.3, 40.0),
                                    tau_init=80.0, tau_max=2000.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: obj={best_obj:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 5: Spectral initialization fresh start (5% of time) ===
    print("\n=== Phase 5: Fresh Spectral Start ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        n_fresh = len(best_a)
        a_fresh = spectral_init(n_fresh, rng, num_modes=10)
        alloc = min(remaining * 0.3, 120.0)
        if alloc > 30:
            a_fresh = lp_alternate(a_fresh, time_budget=alloc, seed=seed + 9999, verbose=False)
            obj_fresh = compute_obj(a_fresh)
            if obj_fresh < best_obj:
                best_obj = obj_fresh
                best_a = a_fresh.copy()
                print(f"  ** Fresh start: obj={best_obj:.10f} **")
            else:
                print(f"  Fresh start: obj={obj_fresh:.10f}")

    # === Phase 6: Final combined refinement ===
    print("\n=== Phase 6: Final Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Alternate between LP and gradient for remaining time
        sub_budget = remaining * 0.5
        a_final = lp_alternate(best_a, time_budget=sub_budget,
                               seed=seed + 4321, verbose=False)
        obj_mid = compute_obj(a_final)
        if obj_mid < best_obj:
            best_obj = obj_mid
            best_a = a_final.copy()

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_final = nesterov_polish(best_a, time_budget=remaining * 0.8,
                                      tau_init=100.0, tau_max=4000.0)
            obj_final = compute_obj(a_final)
            if obj_final < best_obj:
                best_obj = obj_final
                best_a = a_final.copy()
                print(f"  Final: obj={best_obj:.10f} **")

    # Final validation
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_obj:
        best_obj = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
