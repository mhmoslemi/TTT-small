So the formula is: grad[i] = 2 * sum_j a[j] * p[j+i] where p is defined for indices 0 to 2n-2.

This is the cross-correlation of a and p (with p having length 2n-1).

Using FFT: We can compute this as:
- Pad a to length N (N >= 3n-1 to avoid circular convolution issues)
- Pad p to length N
- The cross-correlation (a ⊛ p)[i] = sum_j a[j] * p[i+j]

In the frequency domain, cross-correlation is:
ifft(conj(FFT(a)) * FFT(p)) gives sum_j conj(a[j]) * p[i+j] = sum_j a[j] * p[i+j] for real a.

So: grad = 2 * ifft(conj(fft(a, N)) * fft(p, N)).real[:n]

Now, for the optimization algorithm, I'll use:

1. **FISTA (Fast Iterative Shrinkage-Thresholding Algorithm)**: Accelerated gradient descent with momentum.
2. **Log-sum-exp smoothing**: Approximate the max with a smooth function.
3. **Adaptive tau**: Start with large tau (very smooth), decrease over time.
4. **Simplex projection**: After each gradient step, project onto {x >= 0, sum(x) = 1}.
5. **Multiple initializations**: Ramp, super-ramp, parent, and random.

Let me also think about the LP approach more carefully. The LP formulation is:
minimize M
subject to: conv(a, g)[k] <= M for all k, sum(g) = 1, g >= 0

This finds the direction g that, when moved towards from a, most reduces the peak. But the LP is slow for large n.

A faster alternative: Instead of solving the LP exactly, use a few iterations of the multiplicative weight method (similar to the "blow-up" method):

Initialize g = a (or uniform).
Repeat:
  conv_ag = convolve(a, g)
  Update: g[k] *= exp(-tau * conv_ag[k] / M) for the "active" lags
  Normalize g to sum 1.

This is essentially mirror descent in the space of g, and it's much faster per iteration.

Actually, I think the most effective approach for this problem is:

1. Use FISTA with log-sum-exp smoothing for the main optimization.
2. Use the LP (or a fast approximation) for final polishing.
3. Try multiple initializations including the ramp.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FISTA accelerated gradient descent with adaptive log-sum-exp smoothing,
FFT-based gradient computation, and multi-shape initialization.
Key innovations: ramp initialization exploiting asymmetry, fast FFT gradients,
adaptive smoothing annealing, and multiplicative-weight LP approximation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj_full(a):
    """Full objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    return compute_obj_normalized(a_norm)


def smooth_max_gradient(a, tau):
    """
    Compute gradient of smoothed max of conv(a,a) using log-sum-exp.
    Returns (gradient, conv_values).
    Assumes sum(a) = 1.
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    # Numerically stable log-sum-exp
    shifted = tau * (conv - max_c)
    exps = np.exp(shifted)
    sum_exps = float(np.sum(exps))
    p = exps / sum_exps  # probability distribution over lags
    
    # grad[i] = 2 * sum_j a[j] * p[i+j]
    # This is cross-correlation: (a ⊛ p)[i] * 2
    N = 1
    while N < 4 * n:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    grad = 2.0 * ifft(np.conj(A) * P, N).real[:n]
    
    return grad, conv, float(np.max(conv))


def fista_smooth(a, time_budget=100.0, tau_start=5.0, tau_end=500.0, verbose=False, seed=42):
    """
    FISTA (accelerated gradient descent) with log-sum-exp smoothing.
    a is normalized to sum=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    
    # FISTA parameters
    t = 1.0  # momentum parameter
    x_old = a.copy()  # previous iterate for Nesterov
    x = a.copy()
    
    # Adaptive step size
    step = 2.0 / (tau_start * float(np.max(compute_conv(a))) * 2 * n)
    # Actually, the Lipschitz constant of the gradient of the smoothed objective
    # is related to tau and the norm of a. Let's use a safe step.
    # ||grad|| <= 2 * tau * ||a||_1 * max_p ... let's just use barzilai-borwein or fixed
    
    # The gradient of the smoothed max is a convex combination of gradients of conv[k],
    # each of which has norm at most 2*||a||_2. So the Lipschitz constant is at most 2*||a||_2.
    # For normalized a, ||a||_2 <= 1, so L <= 2.
    # Step size = 1/L = 0.5, but this might be too conservative.
    # Let's use a larger step and project.
    
    step = 0.5  # initial step
    best_it = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Anneal tau: start smooth, end sharp
        tau = tau_start * (tau_end / tau_start) ** progress
        # Or use cosine annealing
        tau = tau_start + (tau_end - tau_start) * (1.0 - np.cos(np.pi * progress * 0.5))
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + (t - 1.0) / t_new * (x - x_old)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Compute gradient
        grad, conv, max_c = smooth_max_gradient(y, tau)
        
        # Gradient step
        x_new = y - step * grad
        x_new = np.maximum(x_new, 0)
        s_new = float(np.sum(x_new))
        if s_new < 1e-10:
            x_new = best_a.copy()
        else:
            x_new = x_new / s_new
        
        # Check improvement
        obj_new = compute_obj_normalized(x_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = x_new.copy()
            best_it = it
            # Backtracking: increase step slightly
            step = min(step * 1.05, 2.0)
        else:
            # Reduce step
            step = max(step * 0.9, 0.01)
        
        x_old = x.copy()
        x = x_new
        t = t_new
        it += 1
        
        if verbose and it % 1000 == 0:
            print(f"    FISTA it{it}: obj={best_obj:.8f}, tau={tau:.1f}, step={step:.4f}")
        
        # Reset if stuck
        if it - best_it > 50000:
            x = best_a.copy()
            x_old = x.copy()
            t = 1.0
            step = 0.5
            best_it = it
            # Perturb
            rng = np.random.default_rng(it)
            x = x * (1 + 0.01 * rng.normal(size=n))
            x = np.maximum(x, 0)
            x /= x.sum()
    
    return best_a


def multiplicative_weight_lp(a, time_budget=50.0, verbose=False, seed=42):
    """
    Fast LP approximation using multiplicative weight updates.
    Solves: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0
    using exponentiated gradient (regret minimization).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    rng = np.random.default_rng(seed)
    
    g = np.ones(n) / n
    best_g = g.copy()
    best_obj = compute_obj_normalized(g)
    start = time.monotonic()
    
    eta = 2.0 / np.sqrt(n)  # learning rate
    
    # Precompute conv(a, g) efficiently
    while time.monotonic() - start < time_budget:
        # conv_a_g = convolve(a_norm, g)
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fa = rfft(a_norm, size)
        fg = rfft(g, size)
        conv_ag = irfft(fa * fg, size)[:2 * n - 1]
        
        max_ag = float(np.max(conv_ag))
        if max_ag < 1e-15:
            break
        
        # Update: g *= exp(-eta * conv_ag_i) where conv_ag_i is the "per-coordinate" penalty
        # The penalty for coordinate i is: sum_k conv_ag[k] * d(conv_ag[k])/d(g[i])
        # d(conv_ag[k])/d(g[i]) = a_padded[k-i]
        # So penalty[i] = sum_k conv_ag[k] * a_padded[k-i] = (a ⊛ conv_ag)[i]
        
        N = 1
        while N < 4 * n:
            N <<= 1
        A = fft(a_norm, N)
        C = fft(conv_ag, N)
        penalty = ifft(np.conj(A) * C, N).real[:n]
        
        # Multiplicative update
        g = g * np.exp(-eta * penalty / max_ag)
        g = np.maximum(g, 1e-30)
        g = g / g.sum()
        
        obj_g = compute_obj_normalized(g)
        if obj_g < best_obj:
            best_obj = obj_g
            best_g = g.copy()
        
        # Adaptive eta
        # Decay eta slowly
        eta *= 0.999
    
    return best_g


def ramp_init(n, power=1.0):
    """Create a ramp-like initialization: a[i] proportional to (i+1)^power."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    a = a / a.sum()
    return a


def super_ramp_init(n, alpha=0.5):
    """Create a super-ramp: a[i] = (i+1)^alpha for 0 < alpha < 1."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** alpha
    a = a / a.sum()
    return a


def bipolar_ramp_init(n):
    """Two-sided ramp: high in middle, low at edges."""
    x = np.linspace(-1, 1, n)
    a = 1.0 - x**2  # parabola, max in middle
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def cosine_taper_init(n):
    """Cosine taper: transitions from 0 to 1."""
    x = np.linspace(0, 1, n)
    a = x * (1 - x)  # parabolic
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    return a


def mixed_init(n, rng, seed=0):
    """Random mixed initialization."""
    x = np.arange(n, dtype=np.float64) / n
    a = 0.3 * x + 0.3 * (1-x) + 0.4 * np.ones(n)/n
    a = a * (1 + 0.1 * rng.normal(size=n))
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key insight: asymmetric (ramp-like) sequences achieve much lower peaks than symmetric ones.
Uses FISTA accelerated gradient descent with FFT-based log-sum-exp smoothing,
adaptive tau annealing, multi-shape initialization, and multi-n exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj_normalized(a):
    """Objective assuming sum(a)=1, a>=0: returns 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def smooth_max_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*conv_k)) w.r.t. a.
    Returns (gradient, conv_array, max_conv_value).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    # Numerically stable softmax
    shifted = tau * (conv - max_c)
    exps = np.exp(shifted)
    sum_exps = float(np.sum(exps))
    if sum_exps < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exps / sum_exps

    # grad[i] = 2 * sum_j a[j] * p[i+j]  (cross-correlation of a and p)
    N = 1
    while N < (2 * n - 1 + n):
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    grad = 2.0 * ifft(np.conj(A) * P, N).real[:n]

    return grad, conv, max_c

        return a
    a = a / s

    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    x_old = a.copy()
    best_it = 0

        # Polynomial annealing of tau

        # Nesterov extrapolation

    best_obj = compute_obj_normalized(g)
    eta = eta_init

        # Compute conv(a, g)
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fa = rfft(a_norm, size)
        fg = rfft(g, size)
        conv_ag = irfft(fa * fg, size)[:2 * n - 1]

        max_ag = float(np.max(conv_ag))
        if max_ag < 1e-15:
            break

        # Penalty for coordinate i: sum_k conv_ag[k] * a_padded[k-i]
        N = 1
        while N < 4 * n:
            N <<= 1
        A = fft(a_norm, N)
        C = fft(conv_ag, N)
        penalty = ifft(np.conj(A) * C, N).real[:n]

        # Multiplicative update
        g = g * np.exp(-eta * penalty / max_ag)
        g = np.maximum(g, 1e-20)
        g = g / g.sum()

        obj_g = compute_obj_normalized(g)
        if obj_g < best_obj:
            best_obj = obj_g
            best_g = g.copy()

        # Decay eta
        eta *= 0.9995

    return best_g


def ramp_init(n, power=1.0):
    """Ramp initialization: a[i] proportional to (i+1)^power."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    a = a / a.sum()
    return a


def asymmetric_init(n, alpha=0.7):
    """Asymmetric initialization blending ramp and uniform."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = alpha * x + (1 - alpha) * 0.5
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    return a


def shape_search(ramp_base, n, time_budget=30.0):
    """Search over shape parameters near the ramp."""
    best_a = ramp_base.copy()
    best_obj = compute_obj_normalized(ramp_base)
    start = time.monotonic()

    for power in [0.8, 1.0, 1.2, 1.5, 0.6]:
        a_try = ramp_init(n, power=power)
        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
        if time.monotonic() - start > time_budget:
            break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)

        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj_p = compute_obj_normalized(parent)
                print(f"Parent (n={parent.size}): obj={obj_p:.8f}")
                best_score = obj_p
                best_seq = parent.copy()
    except Exception:
        pass

    # Critical: try ramp initialization (the key breakthrough)
    n_ramp = 800

        best_score = obj_ramp
        best_seq = ramp.copy()

    # Try several ramp powers
    for p in [0.7, 0.8, 1.2, 1.5]:
        a_try = ramp_init(n_ramp, power=p)
        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  Ramp p={p}: obj={obj_try:.8f} **")

    n_best = len(best_seq)
    print(f"\nBest initial: n={n_best}, obj={best_score:.8f}")

    # === Phase 1: FISTA intensive optimization ===
    print("\n=== Phase 1: FISTA Optimization ===")

    remaining = deadline - time.monotonic()
    mw_budget = min(remaining * 0.12, 100.0)

        a_mw = multiplicative_weight_step(best_seq, num_iters=8000, eta_init=0.8)

        obj_mw2 = compute_obj_normalized(a_mw2)

    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
