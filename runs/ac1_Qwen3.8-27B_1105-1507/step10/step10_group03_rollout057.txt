from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lse_obj_and_grad_sym(a, tau):
    """
    Compute LSE-smoothed objective and gradient for palindromic a.
    a is the full sequence (length n), assumed to be a palindrome.
    Returns (objective_value, gradient of length n).
    The objective is 2*n*LSE(tau*conv(a,a)) (without the 1/tau scaling yet).
    Actually, let's return the scaled version: 2*n*(m + (1/tau)*log(mean(exp(tau*(c-m))))))
    where m = max(c), c = conv(a,a).
    This equals 2*n*(1/tau)*log(sum(exp(tau*c))) - 2*n*tau + ... 
    Let me just return the LSE value and its gradient.
    LSE(tau*c) = m + (1/tau)*log(sum(exp(tau*(c-m))))
    dLSE/da[j] = tau * sum_k p[k] * dc[k]/da[j] where p[k] = exp(tau*(c[k]-m))/sum(exp(...))
    For palindromic a: = 2*tau * (p_flipped * a)[j] (cross-correlation)
    """
    n = len(a)
    c = compute_conv(a)
    m = float(np.max(c))
    shifted = tau * (c - m)
    max_shifted = float(np.max(shifted))
    exp_vals = np.exp(shifted - max_shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    
    lse_val = m + (float(max_shifted) + np.log(sum_exp)) / tau
    
    # Gradient using FFT (valid for palindromic a)
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    grad = 2.0 * tau * cross[:n]
    
    # Full objective includes the 2*n factor
    obj = 2.0 * n * lse_val
    grad_full = 2.0 * n * grad
    
    return obj, grad_full


def lse_obj_and_grad_sym_half(a_half, tau):
    """
    Work with half-sequence for palindromic optimization.
    a_half has length h = ceil(n/2).
    Reconstructs full a and computes obj/grad, then projects grad back to half.
    """
    h = len(a_half)
    n = 2 * h - 1  # odd length palindrome
    # Reconstruct full palindrome
    if n % 2 == 1:
        a = np.empty(n)
        a[:h] = a_half
        a[h:] = a_half[:0:-1]
    else:
        a = np.empty(n)
        a[:h] = a_half
        a[h:] = a_half[:0:-1]
    
    # Actually for even n = 2h: a = [a_half[0], ..., a_half[h-1], a_half[h-1], ..., a_half[0]]
    # For odd n = 2h-1: a = [a_half[0], ..., a_half[h-1], a_half[h-2], ..., a_half[0]]
    # Let me just always use odd n for simplicity.
    
    obj, grad = lse_obj_and_grad_sym(a, tau)

    grad_half = np.empty(h)
    if n % 2 == 1:

    else:
        grad_half[:h] = (grad[:h] + grad[n-1: n-1-h: -1]) / 2.0
    
    return obj, grad_half

    """LSE-smoothed max of conv(a,a) and its gradient (for palindromic a)."""

    Optimize LSE-smoothed objective using L-BFGS-B on the simplex.
    a is assumed to be a palindrome, normalized to sum=1.

    a = a.copy()
    
    start = time.monotonic()

        # x is on the simple grid, we need to project to simplex
        a_try = np.maximum(x, 0)
        a_try = symmetrize(a_try)
        s = float(np.sum(a_try))
        if s < 1e-15:
            return 1e10, np.zeros(n)

def lp_solve(a, solver_list=None):
    """Solve the LP: min M s.t. (a*g)[k] <= M, g >= 0, sum(g) = 1."""
    n = len(a)
    if n > 3000:
        return None, None
    
    # Build convolution matrix A where (A*g)[k] = sum_j a[|k-j|]... 
    # Actually (a*g)[k] = sum_j a[j] * g[k-j] (linear conv)
    # = sum_j a[k-j] * g[j] (by reindexing)
    # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # A has shape (2n-1, n)
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    g_val = g_val / sg
                    return g_val, float(M_var.value)
        except Exception:
            continue
    
    return None, None

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))
