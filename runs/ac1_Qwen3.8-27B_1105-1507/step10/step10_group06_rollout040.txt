Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:

- Subgradient mirror descent (multiplicative weight updates)
- Multi-start with diverse initializations (Fourier modes, windows)
- Adaptive length exploration
The key innovation over the previous approach is the use of multiplicative

blending can get stuck in.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_sum = np.sum(a)
    if a_sum < 1e-10:
        return None, None
    a_norm = a / a_sum
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def subgradient_mirror_descent(a, time_budget, eta_init=0.1, seed=42):
    """
    Subgradient mirror descent with multiplicative weight updates.
    The objective (with sum(a)=1) is f(a) = 2n * max_k (a*a)[k].
    We use the subgradient at the argmax of the convolution.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    eta = eta_init
    patience = 0
    PATIENCE_MAX = 500
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Compute convolution
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Compute subgradient of (a*a)[k_star] w.r.t. a
        # (a*a)[k] = sum_j a[j] * a[k-j] (valid j)
        # d/d a[i] of (a*a)[k] = a[k-i] + a[i - (k - n + 1)] (with bounds)
        # More carefully:
        # (a*a)[k] = sum_{j: max(0,k-n+1) <= j <= min(k, n-1)} a[j]*a[k-j]
        # d/d a[i]: a[k-i] (if max(0,k-n+1) <= k-i <= min(k,n-1)) 
        #           + a[i-(k-n+1)] (if max(0,k-n+1) <= i-(k-n+1) <= min(k,n-1) and k >= n-1... )
        # Actually simpler: by symmetry of the product, d/d a[i] = a[k-i] + a[k-i] if i != k-i
        # No wait. (a*a)[k] = sum_j a[j]*a[k-j].
        # d/d a[i] = sum_j (delta_{j,i}*a[k-j] + a[j]*delta_{k-j,i})
        # = a[k-i] + a[i+k-(k)] ... no
        # = a[k-i] (from the first term, when j=i) + a[k-i] (from the second term, when k-j=i, i.e., j=k-i)
        # Wait, that's only if i != k-i. If i = k/2 (k even), then it's 2*a[i]... no.
        # Let me just be careful:
        # (a*a)[k] = sum_j a_j * a_{k-j}
        # d/d a_i = sum_j [δ_{j,i} * a_{k-j} + a_j * δ_{k-j,i}]
        # = a_{k-i} + a_{k-i} = 2*a_{k-i}  (if both indices are valid)
        # But we need 0 <= k-i < n for the first term, and 0 <= k-i < n for the second.
        # Wait, the second term: k-j = i means j = k-i, and we need 0 <= k-i < n.
        # So both conditions are the same: 0 <= k-i < n.
        # But also i must be in [0, n-1] (which it is since we're computing for each i).
        # So: d/d a_i of (a*a)[k] = 2 * a_{k-i} if 0 <= k-i < n, else 0.
        # Wait, that's not right either. Let me redo.
        # (a*a)[k] = sum_{j=0}^{n-1} a_j * a_{k-j} where a_m = 0 if m < 0 or m >= n.
        # d/d a_i = sum_{j=0}^{n-1} [δ_{j,i} * a_{k-j} + a_j * δ_{k-j,i}]
        # = a_{k-i} + a_{k-i}  (both terms give the same: first when j=i, second when j=k-i)
        # = 2 * a_{k-i} if 0 <= k-i < n
        # But wait, when j=i AND j=k-i (i.e., i = k/2), the two terms are the same, so it's just a_{k-i} = a_i, not 2*a_i.
        # Hmm, no. Let's think of it as: (a*a)[k] = sum_j a_j * b_j where b_j = a_{k-j}.
        # d/d a_i = b_i + a_i = a_{k-i} + a_i... no that's wrong too.
        # OK let me just be very explicit.
        # (a*a)[k] = sum_{j=0}^{n-1} a_j * a_{k-j} (with a_m = 0 for m outside [0,n-1])
        # This is a function of a_0, a_1, ..., a_{n-1}.
        # ∂/∂a_i of this sum = sum_{j=0}^{n-1} [δ_{j,i} * a_{k-j} + a_j * ∂a_{k-j}/∂a_i]
        # = a_{k-i} + sum_{j: k-j=i} a_j
        # = a_{k-i} + a_{k-i}  (the sum has one term: j = k-i, contributing a_{k-i})
        # = 2 * a_{k-i} (if 0 <= k-i < n)
        # EXCEPT when i = k-i, i.e., i = k/2 (k even), in which case:
        # ∂/∂a_i = a_i + a_i = 2*a_i (same formula, just i = k/2)
        # Wait no, when j = i and k-j = i (i.e., j = k-i = i), the two contributions are:
        # first term: δ_{j,i} * a_{k-j} = 1 * a_{k-i} = a_i (since k-i = i)
        # second term: a_j * δ_{k-j,i} = a_i * 1 = a_i
        # Total: 2*a_i. Same formula.
        # So the formula is always: ∂(a*a)[k]/∂a_i = 2*a_{k-i} if 0 <= k-i < n, else 0.
        
        g_sub = np.zeros(n)
        valid = (k_star >= 0) & (k_star < 2*n-1)
        if valid:
            # g_sub[i] = 2*a[k_star - i] if 0 <= k_star - i < n
            # i ranges from 0 to n-1
            # k_star - i ranges from k_star to k_star - (n-1)
            lo = max(0, k_star - (n - 1))
            hi = min(n - 1, k_star)
            for i in range(lo, hi + 1):
                g_sub[i] = 2.0 * a[k_star - i]
        
        # Normalize subgradient
        g_sub_norm = float(np.linalg.norm(g_sub))
        if g_sub_norm < 1e-15:
            break
        
        # Multiplicative weight update
        # a_new[i] = a[i] * exp(-eta * g_sub[i] / g_sub_norm) / Z
        scaled_g = eta * g_sub / g_sub_norm
        # Clip to avoid overflow
        scaled_g = np.clip(scaled_g, -50, 50)
        a_new = a * np.exp(-scaled_g)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        it += 1
        
        # Decay eta
        eta *= 0.999
        eta = max(eta, 0.001)
        
        if patience > PATIENCE_MAX:
            # Reset with perturbation
            a = best_a * (1 + 0.01 * rng.standard_normal(n))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            eta = eta_init * 0.5
            patience = 0
    
    return best_a


def lp_alternate_multiplicative(a, time_budget, seed=42, verbose=True):
    """
    LP alternating minimization with multiplicative (geometric) blending.
    At each step, solve LP for best response g, then blend using geometric mean.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    iter_count = 0
    t_blend = 0.5
    stall_count = 0
    prev_lp_time = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < max(2, prev_lp_time * 2):
            break
        
        t0 = time.monotonic()
        g_lp, M_lp = solve_conv_lp(a)
        prev_lp_time = time.monotonic() - t0
        
        if g_lp is None:
            break
        
        iter_count += 1
        
        # Try multiple blending strategies
        candidates = []
        
        # Geometric blending: a^((1-t)) * g^t
        for t in [t_blend, t_blend*0.5, t_blend*1.5, 0.1, 0.3, 0.5, 0.7, 1.0]:
            t = min(max(t, 0.001), 1.0)
            a_try = a**((1-t)) * g_lp**t
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()
            candidates.append(a_try)
        
        # Additive blending (fallback)
        for t in [t_blend, 0.3, 0.5]:
            t = min(max(t, 0.001), 1.0)
            a_try = (1-t) * a + t * g_lp
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            candidates.append(a_try)
        
        # Find best candidate
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend
        best_local_is_geometric = True
        
        for idx, a_try in enumerate(candidates):
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = min(max([t_blend, t_blend*0.5, t_blend*1.5, 0.1, 0.3, 0.5, 0.7, 1.0][(idx % 8)], 0.001), 1.0)
                best_local_is_geometric = bool(idx < 8)
        
        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall_count = 0
            if verbose and iter_count % 50 == 0:
                print(f"    LP-geo {iter_count}: obj={best_obj:.10f}, t={best_local_t:.4f}, "
                      f"M_lp={M_lp:.6f}, time={prev_lp_time:.3f}s")
        else:
            stall_count += 1
            if stall_count >= 5:
                # Fourier perturbation to escape
                strength = 0.005 + 0.003 * min(stall_count, 10)
                pert = np.zeros(n)
                for k in range(1, 6):
                    freq = k * 2 * np.pi / n
                    amp = strength * rng.uniform(0.5, 1.5) / k
                    phase = rng.uniform(0, 2 * np.pi)
                    pert += amp * np.cos(freq * np.arange(n) + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                t_blend = max(0.1, min(t_blend * 1.3, 0.9))
                stall_count = 0
    
    if verbose:
        print(f"    LP multiplicative: {iter_count} iterations, best_obj={best_obj:.10f}")
    return best_a


def subgradient_polish(a, time_budget, tau_init=50.0, tau_max=3000.0, seed=42):
    """
    Polished subgradient method using log-sum-exp smoothing of the max.
    Uses FFT for fast gradient computation.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    tau = tau_init
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    beta = 0.9
    it = 0
    patience = 0
    
    while time.monotonic() - start < time_budget:
        # Smoothed gradient via log-sum-exp
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp
        
        # Gradient: 2 * (a * p) in the appropriate sense
        # g[i] = 2 * sum_k p[k] * a[k-i] (valid)
        # Use FFT: g = 2 * ifft(fft(a) * conj(fft(p)))
        N = 1
        while N < 3 * n:
            N <<= 1
        A = fft(a, N)
        P = fft(p, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]
        
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        
        # Nesterov step
        y = a + beta * v
        y = np.maximum(y, 1e-10)
        y = y / y.sum()
        
        if g_norm > 1e-15:
            adaptive_lr = 0.5 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.05 * lr), 2.0 * lr)
            lr = 0.9 * lr + 0.1 * adaptive_lr
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 1e-10)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        it += 1
        tau = min(tau * 1.01, tau_max)
        lr *= 0.9997
        beta = min(beta * 1.001, 0.95)
        
        if patience > 1000:
            a = best_a * (1 + 0.005 * rng.standard_normal(n))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            tau = tau_init
            patience = 0
    
    return best_a


def cosine_init(n, rng):
    """Initialize with a cosine (Hanning) window."""
    j = np.arange(n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * j / (n - 1))
    a = a + 0.01 * rng.standard_normal(n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def triangular_init(n, rng):
    """Initialize with a triangular/Bartlett window."""
    j = np.arange(n)
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(j - mid) / (mid + 1e-10)
    a = a + 0.01 * rng.standard_normal(n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def fourier_mode_init(n, mode, phase, rng, base=1.0, amp=0.1):
    """Initialize with a constant plus a single Fourier mode."""
    j = np.arange(n)
    a = base + amp * np.cos(2 * np.pi * mode * j / n + phase)
    a = a + 0.005 * rng.standard_normal(n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines LP alternating with multiplicative blending,
    subgradient mirror descent, and multi-initialization exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0
    
    rng = np.random.default_rng(seed)
    
    # Load parent
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 1e-10)
        if a.size == 0 or a.sum() < 0.01:
            raise ValueError("Invalid parent")
        a = a / a.sum()
        n_init = a.size
        print(f"Parent loaded: n={n_init}, obj={compute_obj(a):.10f}")
    except Exception:
        n_init = 800
        a = np.ones(n_init) / n_init
        print(f"Using default init: n={n_init}, obj={compute_obj(a):.10f}")
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # === Phase 1: LP Alternating with Multiplicative Blending (45% time) ===
    print(f"\n=== Phase 1: LP Alternating (multiplicative) ===")
    p1_budget = (budget_s - 20) * 0.45
    elapsed = time.monotonic() - start_time
    if p1_budget > elapsed:
        a_p1 = lp_alternate_multiplicative(best_a, time_budget=p1_budget - elapsed,
                                            seed=seed, verbose=True)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_obj:
            best_obj = obj_p1
            best_a = a_p1.copy()
            print(f"  Phase 1 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 1 result: obj={obj_p1:.10f} (no improvement)")
    else:
        print("  Phase 1: no time remaining")
    
    # === Phase 2: Subgradient Mirror Descent (15% time) ===
    print(f"\n=== Phase 2: Subgradient Mirror Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = min(remaining * 0.2, (budget_s - 20) * 0.15)
        a_p2 = subgradient_mirror_descent(best_a, time_budget=p2_budget,
                                           eta_init=0.05, seed=seed+1)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_obj:
            best_obj = obj_p2
            best_a = a_p2.copy()
            print(f"  Phase 2 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 2 result: obj={obj_p2:.10f}")
    
    # === Phase 3: Smoothed Subgradient Polish (15% time) ===
    print(f"\n=== Phase 3: Smoothed Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = min(remaining * 0.25, (budget_s - 20) * 0.15)
        a_p3 = subgradient_polish(best_a, time_budget=p3_budget,
                                    tau_init=30.0, tau_max=2000.0, seed=seed+2)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3.copy()
            print(f"  Phase 3 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 3 result: obj={obj_p3:.10f}")
    
    # === Phase 4: Multi-initialization exploration (20% time) ===
    print(f"\n=== Phase 4: Multi-Init Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 120:
        p4_budget = min(remaining * 0.5, (budget_s - 20) * 0.20)
        
        n_best = len(best_a)
        # Try different initializations at the same and nearby lengths
        init_configs = [
            ("cosine", n_best),
            ("triangular", n_best),
            ("fourier_m2", n_best),
            ("fourier_m3", n_best),
            ("cosine", n_best + 50),
            ("fourier_m5", n_best - 50),
        ]
        
        per_init_budget = p4_budget / len(init_configs)
        
        for init_name, n_try in init_configs:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            alloc = min(per_init_budget, remaining - 10)
            if alloc < 30:
                break
            
            print(f"  Trying {init_name} at n={n_try}...")
            try:
                if init_name == "cosine":
                    a_try = cosine_init(n_try, rng)
                elif init_name == "triangular":
                    a_try = triangular_init(n_try, rng)
                elif init_name.startswith("fourier"):
                    parts = init_name.split("_")
                    mode = int(parts[1][1:]) if len(parts) > 1 else 2
                    a_try = fourier_mode_init(n_try, mode, rng.uniform(0, 2*np.pi), rng)
                else:
                    a_try = np.ones(n_try) / n_try
                
                # Quick LP polish
                lp_alloc = alloc * 0.6
                a_try = lp_alternate_multiplicative(a_try, time_budget=lp_alloc,
                                                     seed=n_try, verbose=False)
                # Quick gradient polish
                remaining = deadline - time.monotonic()
                if remaining > 20:
                    gd_alloc = min(alloc * 0.3, remaining - 10)
                    if gd_alloc > 10:
                        a_try = subgradient_polish(a_try, time_budget=gd_alloc,
                                                    tau_init=50.0, tau_max=1000.0, seed=n_try+1)
                
                obj_try = compute_obj(a_try)
                print(f"    {init_name} n={n_try}: obj={obj_try:.10f}")
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    print(f"    ** NEW BEST: obj={best_obj:.10f} **")
            except Exception as e:
                print(f"    Error in {init_name}: {e}")
    
    # === Phase 5: Final LP polish on best (5% time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p5_budget = min(remaining * 0.5, (budget_s - 20) * 0.05)
        a_p5 = lp_alternate_multiplicative(best_a, time_budget=p5_budget,
                                            seed=seed+99999, verbose=False)
        obj_p5 = compute_obj(a_p5)
        if obj_p5 < best_obj:
            best_obj = obj_p5
            best_a = a_p5.copy()
            print(f"  Phase 5 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 5 result: obj={obj_p5:.10f}")
    
    # Final validation
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(200)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val != np.inf and val < best_obj:
        best_obj = val
    elif val == np.inf:
        result = [1.0] * 100
        best_obj = 2.0
    
    elapsed = time.monotonic() - start_time

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating minimization with both primal formulations (min-M and max-sum),
geometric (multiplicative) blending for better exploration of the simplex,
subgradient mirror descent polishing, and multi-initialization search.
Key innovations over prior approach: dual LP formulations, geometric blending,
and adaptive subgradient methods to escape linear-blending local minima.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_min_m(a):
    """
    Solve: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_sum = float(np.sum(a))
    if a_sum < 1e-10:
        return None, None
    a_norm = a / a_sum
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_max_sum(a, M_bound):
    """
    Solve: max sum(g) s.t. (a*g)[k] <= M_bound for all k, g>=0.
    M_bound should be max(a*a). Returns (g_opt_normalized, sum_g) or (None, None).
    """
    n = len(a)
    a_sum = float(np.sum(a))
    if a_sum < 1e-10:
        return None, None
    a_norm = a / a_sum
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M_bound]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, s
        except Exception:
            continue
    return None, None


def mirror_descent_subgrad(a, time_budget, eta_init=0.05, seed=42, verbose=False):
    """
    Subgradient mirror descent with multiplicative weight updates.
    Uses the exact subgradient at the argmax of the convolution.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = eta_init
    patience = 0
    PATIENCE_MAX = 800

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        if lo <= hi:
            g_sub[lo:hi + 1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]

        g_norm = float(np.linalg.norm(g_sub))
        if g_norm < 1e-15:
            break

        # Multiplicative weight update
        scaled = np.clip(eta * g_sub / g_norm, -40, 40)
        a_new = a * np.exp(-scaled)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))

        if obj_new < best_obj - 1e-14:

            patience = 0

            eta = eta_init * 0.4
            patience = 0
            if verbose:

    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    patience = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)

def cosine_init(n, rng):
    """Initialize with a Hanning/cosine window."""
    j = np.arange(n)
    a = 0.5 + 0.5 * np.cos(2.0 * np.pi * j / max(n - 1, 1))
    a = a + 0.01 * rng.standard_normal(n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def triangular_init(n, rng):
    """Initialize with a Bartlett/triangular window."""
    j = np.arange(n)
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(j - mid) / (mid + 1e-10)
    a = a + 0.01 * rng.standard_normal(n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def fourier_mode_init(n, mode, phase, rng, base=1.0, amp=0.15):
    """Initialize with constant + single Fourier mode."""
    j = np.arange(n)
    a = base + amp * np.cos(2.0 * np.pi * mode * j / n + phase)
    a = a + 0.005 * rng.standard_normal(n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0], dtype=np.float64)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating (dual formulations + geometric blend),
    subgradient mirror descent, smoothed gradient polishing, and multi-init exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 25.0

    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 1e-10)
        if a.size == 0 or a.sum() < 0.01:
            raise ValueError("Invalid parent")
        a = a / a.sum()
        n_init = a.size
        obj_init = compute_obj(a)
        print(f"Parent loaded: n={n_init}, obj={obj_init:.10f}")
    except Exception:
        n_init = 800
        a = np.ones(n_init, dtype=np.float64) / n_init
        obj_init = compute_obj(a)
        print(f"Using default init: n={n_init}, obj={obj_init:.10f}")

    best_a = a.copy()
    best_obj = obj_init

    # === Phase 1: LP Alternating (primary, 40% time) ===
    print(f"\n=== Phase 1: LP Alternating ===")
    p1_budget = (budget_s - 25) * 0.40
    elapsed = time.monotonic() - start_time
    if p1_budget > elapsed + 30:
        a_p1 = lp_alternate(best_a, time_budget=p1_budget - elapsed,
                            seed=seed, verbose=True, use_both_lp=True)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_obj:
            best_obj = obj_p1
            best_a = a_p1.copy()
            print(f"  Phase 1 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 1 result: obj={obj_p1:.10f} (no improvement)")
    else:
        print("  Phase 1: insufficient time, skipping")

    # === Phase 2: Mirror Descent (12% time) ===
    print(f"\n=== Phase 2: Subgradient Mirror Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = min(remaining * 0.2, (budget_s - 25) * 0.12)
        a_p2 = mirror_descent_subgrad(best_a, time_budget=p2_budget,
                                       eta_init=0.04, seed=seed + 100)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_obj:
            best_obj = obj_p2
            best_a = a_p2.copy()
            print(f"  Phase 2 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 2 result: obj={obj_p2:.10f}")

    # === Phase 3: Smoothed Gradient Polish (13% time) ===
    print(f"\n=== Phase 3: Smoothed Gradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = min(remaining * 0.22, (budget_s - 25) * 0.13)
        a_p3 = smoothed_gd_polish(best_a, time_budget=p3_budget,
                                   tau_init=25.0, tau_max=2000.0, seed=seed + 200)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3.copy()
            print(f"  Phase 3 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 3 result: obj={obj_p3:.10f}")

    # === Phase 4: Second LP pass with perturbation (15% time) ===
    print(f"\n=== Phase 4: Second LP Pass ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        p4_budget = min(remaining * 0.35, (budget_s - 25) * 0.15)
        # Perturb before second pass
        pert = np.zeros(len(best_a))
        n_cur = len(best_a)
        for k in range(1, 5):
            freq = k * 2.0 * np.pi / n_cur
            amp = 0.006 / k
            phase = rng.uniform(0, 2.0 * np.pi)
            pert += amp * np.cos(freq * np.arange(n_cur) + phase)
        a_pert = best_a * np.exp(pert)
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / a_pert.sum()

        a_p4 = lp_alternate(a_pert, time_budget=p4_budget,
                            seed=seed + 500, verbose=False, use_both_lp=True)
        obj_p4 = compute_obj(a_p4)
        if obj_p4 < best_obj:
            best_obj = obj_p4
            best_a = a_p4.copy()
            print(f"  Phase 4 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 4 result: obj={obj_p4:.10f}")

    # === Phase 5: Multi-initialization exploration (15% time) ===
    print(f"\n=== Phase 5: Multi-Init Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        p5_budget = min(remaining * 0.6, (budget_s - 25) * 0.15)
        n_best = len(best_a)

        init_configs = [
            ("cosine", n_best),
            ("triangular", n_best),
            ("fourier2", n_best),
            ("fourier3", n_best),
            ("cosine", n_best + 40),
            ("fourier5", n_best - 40),
        ]

        per_budget = p5_budget / len(init_configs)

        for init_name, n_try in init_configs:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            alloc = min(per_budget, remaining - 15)
            if alloc < 35:
                break

            print(f"    Trying {init_name} at n={n_try}...", end=" ")
            try:
                if init_name == "cosine":
                    a_try = cosine_init(n_try, rng)
                elif init_name == "triangular":
                    a_try = triangular_init(n_try, rng)
                elif init_name.startswith("fourier"):
                    mode = int(init_name[7:])
                    a_try = fourier_mode_init(n_try, mode, rng.uniform(0, 2 * np.pi), rng)
                else:
                    a_try = np.ones(n_try, dtype=np.float64) / n_try

                # LP polish
                lp_alloc = alloc * 0.55
                a_try = lp_alternate(a_try, time_budget=lp_alloc,
                                     seed=n_try + init_name.__hash__() % 10000,
                                     verbose=False, use_both_lp=False)

                # Gradient polish
                remaining = deadline - time.monotonic()
                if remaining > 25:
                    gd_alloc = min(alloc * 0.3, remaining - 15)
                    if gd_alloc > 15:
                        a_try = smoothed_gd_polish(a_try, time_budget=gd_alloc,
                                                    tau_init=40.0, tau_max=1500.0,
                                                    seed=n_try + 1)

                obj_try = compute_obj(a_try)
                print(f"obj={obj_try:.10f}")
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    print(f"    ** NEW BEST: obj={best_obj:.10f} **")
            except Exception as e:
                print(f"Error: {e}")

    # === Phase 6: Final polish (5% time) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        p6_budget = min(remaining * 0.5, (budget_s - 25) * 0.05)
        a_p6 = lp_alternate(best_a, time_budget=p6_budget,
                            seed=seed + 99999, verbose=False, use_both_lp=True)
        obj_p6 = compute_obj(a_p6)
        if obj_p6 < best_obj:
            best_obj = obj_p6
            best_a = a_p6.copy()
            print(f"  Phase 6 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 6 result: obj={obj_p6:.10f}")

    # Final validation
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(200, dtype=np.float64) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val != np.inf:
        if val < best_obj:
            best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
