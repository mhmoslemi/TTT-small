"""
Mirror Descent + Alternating LP + Block Coordinate Descent optimizer
for minimum peak linear autocorrelation of non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a))) / (s * s)

def compute_conv_and_max(a):
    """Return (convolution, max_value, max_indices)."""
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    # Find all indices within 1% of max
    threshold = max_val * 0.99
    max_idx = np.where(conv >= threshold)[0]
    return conv, max_val, max_idx

def subgradient_topk(a, K=20):
    """
    Compute weighted subgradient of max_k b[k] using top-K peak weighting.
    Returns gradient vector g of length n.
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    
    # Get top K peaks
    top_k_idx = np.argsort(conv)[::-1][:K]
    top_k_vals = conv[top_k_idx]
    
    # Softmax weights over top-K
    tau = max_val * 0.1 if max_val > 0 else 1.0
    weights = np.exp((top_k_vals - max_val) / tau)
    weights /= weights.sum()

    # b[k] = sum_j a[j] * a[k-j]

    # = a[k-i] + a[i] if k-i != i
    # = 2*a[i] if k-i = i (k = 2i)

"""
Mirror Descent + Alternating LP + Length Exploration optimizer
for minimum peak linear autocorrelation of non-negative sequences.
Key differences from previous approach:
1. Primary method is mirror descent (multiplicative weights) with backtracking line search
2. Clean alternating LP (a <-> g) for refinement

4. Adaptive top-K peak weighting
5. More systematic length exploration

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a))) / (s * s)


def subgradient_topk(a, K=20, tau_frac=0.05):
    """
    Compute weighted subgradient of max_k b[k] using top-K peak weighting.
    b[k] = sum_j a[j]*a[k-j], grad b[k][i] = 2*a[k-i] if 0<=k-i<n else 0.
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    if max_val < 1e-15:
        return np.zeros(n)

    K = min(K, len(conv))
    top_k_idx = np.argsort(conv)[::-1][:K]
    top_k_vals = conv[top_k_idx]

    tau = max_val * tau_frac
    weights = np.exp((top_k_vals - max_val) / tau)
    weights /= weights.sum()

    g = np.zeros(n)
    for w, k in zip(weights, top_k_idx):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            a_start = k - i_end
            a_end = k - i_start
            g[i_start:i_end + 1] += 2.0 * w * a[a_start:a_end + 1][::-1]

    return g


def mirror_descent(a, time_budget=300.0, K_init=20, eta_init=0.1,
                   verbose=False, seed=42):
    """
    Mirror descent (multiplicative weights) with backtracking line search.
    a is normalized to sum=1 before starting.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = eta_init
    K = K_init
    it = 0
    stall = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Adapt K: start large, decrease over time
        K_eff = max(5, int(K_init * (1.0 - 0.5 * progress)))
        # Adapt tau: start wide, narrow over time
        tau_frac = 0.02 + 0.06 * (1.0 - progress)

        g = subgradient_topk(a, K=K_eff, tau_frac=tau_frac)

        # Mirror descent update with backtracking
        a_try = None
        obj_try = np.inf
        for trial in range(8):
            eta_trial = eta * (0.5 ** trial)
            # Multiplicative update
            logits = -eta_trial * g
            # Clip to avoid overflow
            logits = np.clip(logits, -50, 50)
            a_cand = a * np.exp(logits)
            a_cand = np.maximum(a_cand, 0)
            s_cand = a_cand.sum()
            if s_cand < 1e-10:
                continue
            a_cand = a_cand / s_cand

            obj_cand = compute_obj(a_cand)
            if obj_cand < best_obj - 1e-14:
                a_try = a_cand
                obj_try = obj_cand
                break

        if a_try is not None:
            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a.copy()
                no_improve = 0
                eta = min(eta * 1.05, 1.0)
            stall = 0
        else:
            # No improvement: try additive update as fallback
            g_norm = np.linalg.norm(g)
            if g_norm > 1e-15:
                lr = eta * 0.5 / g_norm
                a_add = a - lr * g
                a_add = np.maximum(a_add, 0)
                s_add = a_add.sum()
                if s_add > 1e-10:
                    a_add = a_add / s_add
                    obj_add = compute_obj(a_add)
                    if obj_add < best_obj - 1e-14:
                        a = a_add
                        best_obj = obj_add
                        best_a = a.copy()
                        no_improve = 0
                        stall = 0
                    else:
                        eta *= 0.7
                        stall += 1
                else:
                    eta *= 0.7
                    stall += 1
            else:
                eta *= 0.7
                stall += 1

        # Perturbation when stuck
        if stall > 30:
            scale = 0.005 + 0.003 * min(stall // 30, 5)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s_p = a.sum()
            if s_p > 1e-10:
                a = a / s_p
            eta = eta_init * (0.5 + 0.5 * rng.random())
            stall = 0
            no_improve += 1

        if no_improve > 80:
            break

        it += 1
        if verbose and it % 100 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.4f}, K={K_eff}, t={elapsed:.0f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, M_target=None):
    """
    Solve LP: min M s.t. A @ g <= M, sum(g)=1, g>=0.
    This finds g minimizing the peak of cross-correlation a*g.
    """
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def alternating_lp(a, time_budget=200.0, verbose=False, seed=42):
    """
    Alternating LP: alternate between finding g=argmin peak(a*g) and a=argmin peak(g*a).
    With line search to ensure monotone improvement.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        # Find direction: g = argmin peak(a * g)
        g_dir, M_val = solve_lp_direction(a)
        if g_dir is None:
            break

        # Line search: find best t in (0, t_max] such that (1-t)*a + t*g_dir improves
        best_t = 0
        best_t_obj = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * best_a + t * g_dir
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-14:
                best_t_obj = obj_try
                best_t = t

        if best_t > 0 and best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = (1 - best_t) * best_a + best_t * g_dir
            best_a = np.maximum(best_a, 0)
            s_ba = best_a.sum()
            if s_ba > 1e-10:
                best_a = best_a / s_ba
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                # Perturb and retry
                scale = 0.006 + 0.004 * min(no_improve, 6)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                no_improve = 0
                if it > 50:
                    break

        it += 1
        if verbose and it % 5 == 0:
            print(f"    ALT-LP it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

    return best_a


def smoothed_fourier_gd(a, time_budget=100.0, tau_init=10.0, tau_end=300.0):
    """Smoothed gradient descent using log-sum-exp approximation in Fourier domain."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 3.0 / np.sqrt(n)
    momentum = 0.75
    v = np.zeros(n)
    it = 0
    stall = 0

    N = 1
    while N < 3 * n:
        N <<= 1

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break

        # Smoothed max via log-sum-exp
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 0)
        exp_vals = np.exp(shifted)
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient via Fourier
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new
            stall += 1
            if stall > 100:
                lr *= 0.7
                v *= 0.5
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.998

        it += 1

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def structured_init(n, rng, mode=0):
    """Generate structured initial sequences with diverse properties."""
    if mode == 0:
        # Uniform with noise
        a = np.ones(n) * np.exp(rng.normal(0, 0.3, n))
    elif mode == 1:
        # Exponential decay
        a = np.exp(-np.linspace(0, 3, n))
        a += 0.1 * np.exp(rng.normal(0, 0.2, n))
    elif mode == 2:
        # Bimodal (high at edges)
        a = np.exp(-4 * (np.linspace(0, 1, n) - 0.5) ** 2)
        a = np.maximum(a, 0.1)
        a *= np.exp(rng.normal(0, 0.15, n))
    elif mode == 3:
        # Periodic modulation
        freq = rng.integers(2, max(3, n // 20))
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * np.linspace(0, 1, n))
        a = np.maximum(a, 0.05)
        a *= np.exp(rng.normal(0, 0.1, n))
    elif mode == 4:
        # Sparse-ish (many small values, few large)
        a = np.exp(rng.normal(-1, 1, n))
    elif mode == 5:
        # Power law
        a = np.power(np.linspace(1, n, n), -0.5)
        a[:] = 1.0 / a  # invert so edges are larger
        a /= a.max()
        a *= np.exp(rng.normal(0, 0.1, n))
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 1e-6)
    s = a.sum()
    if s > 0:
        a = a / s
    return a

        # Quick mirror descent
        alloc1 = min(remaining * 0.10, 30.0)
        if alloc1 > 8:
            a_try = mirror_descent(a_try, time_budget=alloc1,
                                   K_init=15, eta_init=0.06, seed=n_try)

        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Quick LP
        alloc2 = min(remaining * 0.08, 40.0)
        if alloc2 > 10:
            a_try = alternating_lp(a_try, time_budget=alloc2, seed=n_try + 1)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Structured restarts ===
    print(f"\n=== Phase 6: Structured Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    modes = [1, 3, 4, 5, 2, 0]
    for mode in modes:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Blend structured init with best sequence
        a_struct = structured_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * modes.index(mode)
        a_mix = (1 - blend) * best_seq + blend * a_struct
        a_mix = np.maximum(a_mix, 0)
        s_mix = a_mix.sum()
        if s_mix < 1e-10:
            continue
        a_mix = a_mix / s_mix

        alloc = min(remaining * 0.15, 60.0)
        if alloc > 10:
            a_mix = mirror_descent(a_mix, time_budget=alloc,
                                   K_init=20, eta_init=0.08,
                                   seed=500 + mode)
            obj_mix = compute_obj(a_mix)
            if obj_mix < best_score:
                best_score = obj_mix
                best_seq = a_mix.copy()
                print(f"  Mode{mode}: obj={best_score:.8f} **")
            else:
                print(f"  Mode{mode}: obj={obj_mix:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.4, 100.0)
        a_final = mirror_descent(best_seq, time_budget=alloc,
                                 K_init=8, eta_init=0.02, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final MD: obj={best_score:.8f} **")
        else:
            print(f"  Final MD: obj={obj_final:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 50.0)
        a_final_lp = alternating_lp(best_seq, time_budget=alloc, seed=2000)
        obj_flp = compute_obj(a_final_lp)
        if obj_flp < best_score:
            best_score = obj_flp
            best_seq = a_final_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
