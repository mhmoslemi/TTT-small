```python
from scipy.fft import rfft, irfft, fft, ifft
import time
import math

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def gradient_smoothed(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*c_k)) where c=conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_c = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_c))
    p = exp_c / sum_exp
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n], conv

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
    n = len(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9  # momentum
    v = np.zeros(n)
    
    # Adaptive step size
    eta = 1.0 / (tau_start * n) * 2  # initial step size (slightly aggressive)
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Geometric tau increase
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Nesterov lookahead
        y = a + beta * v
        y = project_simplex(y)
        
        # Gradient
        g, conv = gradient_smoothed(y, tau)

        v = beta * v - eta * g
        a_new = y + v
        a_new = project_simplex(a_new)
        
        # Line search (simple: if obj increases, revert)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:

            no_improve = 0

            no_improve += 1
            if no_improve > 50:

                a_new = best_a.copy()
                eta *= 0.9
                v = np.zeros(n)
                no_improve = 0
        
        a = a_new
        it += 1
        
        # Gradually reduce step size
        eta = max(eta * 0.9999, 1e-10)
        
        if verbose and it % 10000 == 0:
            print(f"  GD it={it}, tau={tau:.1f}, eta={eta:.2e}, obj={best_obj:.8f}")
```

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Combines fast smoothed gradient descent with LP-based direction finding,
golden-section line search, adaptive tau scheduling, and multi-length exploration.
Key innovations: geometric tau ramp, golden-section line search, dual-gap
convergence detection, and systematic perturbation with Fourier structure.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gradient_smoothed(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*c_k)) where c=conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_c = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_c))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_c / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n], conv


def build_conv_matrix_normalized(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g) for normalized a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, solver_list=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    import cvxpy as cp
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_normalized(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]

    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:
            continue
    return None, None


def smoothed_gd_fast(a, time_budget, tau_start, tau_end, beta=0.9, verbose=False):
    """Fast Nesterov smoothed GD with geometric tau."""
    n = len(a)
    a = project_simplex(np.maximum(a, 0))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 2.0 / (tau_start * n)
    v = np.zeros(n)
    it = 0
    no_improve = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start * (tau_end / tau_start) ** progress

        # Nesterov lookahead
        y = a + beta * v
        y = project_simplex(y)

        g, conv = gradient_smoothed(y, tau)

        # Adaptive step: scale with 1/tau
        eta_cur = 2.0 / (tau * n) * (1.0 + 0.5 * math.exp(-it / 500.0))

        v = beta * v - eta_cur * g
        a_new = y + v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                a = best_a.copy()
                v = np.zeros(n)
                eta *= 0.8
                no_improve = 0

        a = a_new
        it += 1

        if verbose and it % 5000 == 0:
            print(f"    GD it={it}, tau={tau:.1f}, obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def lp_iteration_golden(a, time_budget, verbose=False, seed=42):
    """LP iteration with golden-section line search and momentum."""
    import cvxpy as cp
    n = len(a)
    a = project_simplex(np.maximum(a, 0))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    prev_g = None
    momentum_alpha = 0.3
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Momentum extrapolation
        if prev_g is not None:
            diff = g_lp - prev_g
            g_lp = g_lp + momentum_alpha * diff
            g_lp = np.maximum(g_lp, 0)
            s = float(np.sum(g_lp))
            if s > 1e-10:
                g_lp = g_lp / s
        prev_g = g_lp.copy()

        # Golden section search for best t in [0.01, 1.5]
        t_lo, t_hi = 0.01, 1.5
        gr = (math.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618

        best_t = 0.0
        best_t_obj = best_obj

        for _ in range(8):
            t1 = t_hi - gr * (t_hi - t_lo)
            t2 = t_lo + gr * (t_hi - t_lo)

            def eval_t(t):
                a_try = (1.0 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    return np.inf
                a_try = a_try / s_try
                return compute_obj(a_try)

            obj1 = eval_t(t1)
            obj2 = eval_t(t2)

            if obj1 < obj2:
                t_hi = t2
            else:
                t_lo = t1

            if obj1 < best_t_obj:
                best_t_obj = obj1
                best_t = t1
            if obj2 < best_t_obj:
                best_t_obj = obj2
                best_t = t2

        # Apply best step
        if best_t > 0 and best_t_obj < best_obj - 1e-13:
            a_new = (1.0 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / float(np.sum(a_new))
            best_obj = best_t_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10:
                # Perturb and restart
                strength = min(0.1, 0.01 * no_improve)
                pert = 1.0 + rng.normal(0, strength, n)
                a = best_a * np.maximum(pert, 0)
                a = project_simplex(a)
                prev_g = None
                momentum_alpha = 0.1
                no_improve = 0

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            conv_a = compute_conv(a)
            gap = float(np.max(conv_a)) - M_lp
            print(f"    LP it={it}, obj={best_obj:.8f}, gap={gap:.2e}, t={elapsed:.0f}s")

        # Convergence check
        if it % 5 == 0:
            conv_a = compute_conv(a)
            max_conv = float(np.max(conv_a))
            if max_conv - M_lp < 1e-9:
                if verbose:
                    print(f"    LP converged at it={it}, obj={best_obj:.8f}")
                break

    return best_a


def fourier_perturb(a, rng, strength=0.03):
    """Add structured low-frequency perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(5, max(1, n // 10))
    for k in range(1, n_modes + 1):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_to_length(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def hybrid_optimize(a, time_budget, verbose=False, seed=0):
    """Combined optimization: GD warmup then LP polish."""
    n = len(a)
    a = project_simplex(np.maximum(a, 0))
    start = time.monotonic()

    # Phase A: Smoothed GD warmup (40% of budget)
    gd_budget = time_budget * 0.4
    if gd_budget > 10:
        a = smoothed_gd_fast(a, time_budget=gd_budget, tau_start=5.0, tau_end=300.0)

    # Phase B: LP iteration (60% of budget)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a = lp_iteration_golden(a, time_budget=remaining, verbose=verbose, seed=seed)

    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines smoothed GD, LP iteration with golden-section line search,
    multi-length exploration, and diversification restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_current = 800
        best_seq = np.ones(n_current) / n_current
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # === Phase 1: Intensive optimization on current n ===
    print(f"\n=== Phase 1: Intensive optimization (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.3, 400.0)
    if p1_budget > 30:
        a_opt = hybrid_optimize(best_seq, time_budget=p1_budget, verbose=True, seed=42)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_opt:.8f} (no improvement)")

    # === Phase 2: Diversification restarts ===
    print(f"\n=== Phase 2: Diversification ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        if r == 0:
            a_r = fourier_perturb(best_seq, rng, strength=0.03)
        elif r == 1:
            a_r = fourier_perturb(best_seq, rng, strength=0.06)
        elif r == 2:
            # Blend with uniform
            a_r = 0.8 * best_seq + 0.2 * np.ones(n_current) / n_current
            a_r = a_r / a_r.sum()
        elif r == 3:
            # Cosine structure blend
            x = np.linspace(0, 1, n_current)
            a_css = 0.5 + 0.5 * np.cos(2.0 * np.pi * x)
            a_css = np.maximum(a_css, 0.01)
            a_r = 0.7 * best_seq + 0.3 * a_css / a_css.sum()
            a_r = a_r / a_r.sum()
        else:
            # Random multiplicative perturbation
            pert = 1.0 + rng.normal(0, 0.1, n_current)
            a_r = best_seq * np.maximum(pert, 0)
            a_r = project_simplex(a_r)

        alloc = min(remaining * 0.18, 200.0)
        if alloc > 30:
            a_r = hybrid_optimize(a_r, time_budget=alloc, seed=100 + r * 7)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Generate candidate n values
    n_candidates = []
    for dn in [-200, -150, -100, -50, 50, 100, 150, 200, 300, 400, 500]:
        n_try = n_best + dn
        if 300 <= n_try <= 2500:
            n_candidates.append(n_try)
    for n_abs in [600, 700, 800, 900, 1000, 1200, 1400]:
        if n_abs not in n_candidates and abs(n_abs - n_best) > 30:
            n_candidates.append(n_abs)

    n_candidates = sorted(set(n_candidates))
    # Limit to 6 most promising
    n_candidates = n_candidates[:6]
    print(f"  Trying n values: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_to_length(best_seq, n_try)
        a_try = project_simplex(a_try)

        alloc = min(remaining * 0.2, 150.0)
        if alloc > 20:
            a_try = hybrid_optimize(a_try, time_budget=alloc, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 60:
        alloc = min(remaining * 0.5, 250.0)
        a_final = lp_iteration_golden(best_seq, time_budget=alloc, verbose=True, seed=9999)
        obj_final = compute_obj(a_final)

            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    if remaining > 30:
        alloc = min(remaining * 0.4, 100.0)
        a_final = smoothed_gd_fast(best_seq, time_budget=alloc,
                                   tau_start=500.0, tau_end=3000.0)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final high-tau GD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30:

        a_final = lp_iteration_golden(best_seq, time_budget=alloc, verbose=False, seed=7777)

    # Final validation
```
