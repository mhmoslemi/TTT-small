Now, for the optimization:
- We want to minimize f(a) = max_k conv(a,a)_k subject to sum(a) = 1, a >= 0.
- This is a convex optimization problem (f is convex as max of convex functions).
- We can use projected gradient / FISTA.

For FISTA:
- Step size: 1/L where L is the Lipschitz constant of the gradient of f.
- The gradient of f at a is 2 * T_{k*} a where k* = argmax.
- ||grad f(a) - grad f(b)|| <= L * ||a - b||
- The gradient is 2 * T_{k*} a, and ||T_k||_2 = 1 (permutation matrix).
- But k* can change between a and b, so the Lipschitz constant isn't simply 2.
- A safe upper bound: L = 2 * max_k ||T_k||_2 = 2 * 1 = 2. But this might be too conservative.
- Actually, ||2*T_k a - 2*T_m b|| <= 2*||T_k a - T_k b|| + 2*||T_k b - T_m b|| <= 2||a-b|| + 2*||T_k b - T_m b||
- The second term is at most 2*||b|| * (something related to the difference between T_k and T_m).
- For a safer bound, note that ||T_k a||_2 = ||a||_2 for all k (since T_k is orthogonal). So ||grad f(a)||_2 = 2*||a||_2 <= 2 (since ||a||_2 <= ||a||_1 = 1).
- A simple safe step size is 1/2 (using L=2). But we can be more aggressive.

Actually, for FISTA on the simplex, a common approach is to use a backtracking line search or a fixed step size based on the problem structure.

Let me think about what step size to use. The Hessian of the active constraint (k = k*) is 2*T_{k*}, which has eigenvalues ±2. So the curvature is at most 2. A step size of 1/2 should be safe.

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * conv.max()

def simplex_project(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[int(rho)] - 1.0) / (int(rho) + 1.0)
    return np.maximum(v - theta, 0.0)


def fista_optimize(a, time_budget=30.0, step_size=0.3, verbose=False):
    """FISTA for min max_k conv(a,a)_k s.t. sum(a)=1, a>=0."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    conv = compute_conv(a)
    best_obj = float(conv.max())
    start = time.monotonic()
    it = 0
    restart_count = 0
    no_improve = 0

    y = a.copy()
    t_k = 1.0
    a_prev = a.copy()
    lr = step_size

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 0.1:
            break

        conv_y = compute_conv(y)
        k_max = int(np.argmax(conv_y))

        # Subgradient: 2 * y[k_max - i] for valid i
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for i in range(lo, hi + 1):
            grad[i] = 2.0 * y[k_max - i]

        # FISTA step
        x_new = simplex_project(y - lr * grad)

        conv_new = compute_conv(x_new)
        obj_new = float(conv_new.max())
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new
        y = x_new + momentum * (x_new - a_prev)
        y = np.maximum(y, 0.0)
        y_s = float(np.sum(y))
        if y_s < 1e-10:
            y = best_a.copy()
        else:
            y = y / y_s * float(np.sum(x_new))

        a_prev = x_new
        t_k = t_new
        it += 1

        # Adaptive restart
        if no_improve > 3000:
            a_prev = best_a.copy()
            y = best_a.copy()
            t_k = 1.0
            lr = max(0.01, lr * 0.5)
            no_improve = 0
            restart_count += 1

        # Slow decay of learning rate
        lr *= 0.9998

        if verbose and it % 5000 == 0:
            print(f"    FISTA it{it}: obj={best_obj:.8f}, lr={lr:.5f}, restarts={restart_count}")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, upper_bound=None, lower_bound=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0, optional bounds."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if upper_bound is not None:
        constraints.append(g <= upper_bound)
    if lower_bound is not None:
        constraints.append(g >= lower_bound)

    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def parabolic_line_search(a, g, lo, hi, f_lo, f_mid, f_hi):
    """Parabolic interpolation for 1D minimization."""
    if abs(f_hi - f_lo) < 1e-15:
        return (lo + hi) / 2
    denom = 2.0 * ((f_mid - f_lo) / (hi - lo) - (f_hi - f_mid) / (hi - lo))
    denom = (f_lo - 2.0 * f_mid + f_hi)
    if abs(denom) < 1e-15:
        return (lo + hi) / 2
    vertex = lo + 0.5 * (f_lo - f_hi) / (0.5 * (f_lo - 2.0 * f_mid + f_hi) / (hi - lo)) * (hi - lo)
    # Simplified: use standard parabolic formula
    d1 = hi - lo
    d2 = hi - lo
    vertex = lo + 0.5 * d1 * (f_lo - f_hi) / (f_lo - 2.0 * f_mid + f_hi)
    if vertex < lo or vertex > hi:
        vertex = (lo + hi) / 2
    return vertex


def line_search_blend(a, g, time_budget=4.0):
    """Find optimal blend factor t minimizing obj((1-t)*a + t*g)."""
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi

    lo, hi = 0.0, 1.5

    def obj_at(t):
        if t < 0:
            t = 0.0
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    # Sample several points first
    candidates = [0.0, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.2, 1.5]
    best_t = 0.0
    best_f = obj_at(0.0)
    for t_c in candidates:
        f_c = obj_at(t_c)
        if f_c < best_f:
            best_f = f_c
            best_t = t_c

    # Golden section around the best candidate
    center = best_t
    half_width = 0.15
    lo = max(0.0, center - half_width)
    hi = min(1.5, center + half_width)

    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    start = time.monotonic()
    for _ in range(30):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)

    if obj_best < best_f:
        best_t = t_best
        best_f = obj_best

    return best_t, best_f

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = np.exp(3.0 * (2 * x - 1) ** 2)
    else:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def u_shaped_init(n, sharpness=3.0):

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:

    return a / s

def max_conv_gradient(a, n):
    """Gradient of max conv(a,a)_k w.r.t. a (subgradient)."""
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 0.01:
        return np.zeros(n)
    a_norm = a / s
    conv = compute_conv(a_norm)
    k_max = int(np.argmax(conv))
    grad = np.zeros(n)
    lo = max(0, k_max - n + 1)
    hi = min(n - 1, k_max)
    for i in range(lo, hi + 1):
        grad[i] = 2.0 * a_norm[k_max - i]
    return grad / s  # Chain rule for the normalization


def scipy_polish(a, time_budget=60.0, verbose=False):
    """Use scipy SLSQP to polish the solution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Define objective and gradient for SLSQP
    def obj_func(x):
        x = np.maximum(x, 0)
        ss = float(np.sum(x))
        if ss < 0.01:
            return 1e10
        xn = x / ss
        conv = compute_conv(xn)
        return 2.0 * n * float(np.max(conv))

    def grad_func(x):
        x = np.maximum(x, 0)
        ss = float(np.sum(x))
        if ss < 0.01:
            return np.zeros(n)
        xn = x / ss
        conv = compute_conv(xn)
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for i in range(lo, hi + 1):
            grad[i] = 2.0 * xn[k_max - i]
        # Chain rule: d/dx_i [2n * max_conv(x/sum(x))]
        # = 2n * (grad_i * ss - xn[i] * sum(grad) * ss) / ss^2
        # Simplified: just use grad_i / ss (approximate)
        return 2.0 * n * grad / ss

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0, None)] * n

    try:
        result = scipy_minimize(
            obj_func, a, method='SLSQP',
            jac=grad_func,
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-12, 'disp': False}
        )
        if result.success or (result.fun is not None and result.fun < best_obj / (2 * n)):
            a_new = np.maximum(result.x, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    if verbose:
                        print(f"    SLSQP improved: obj={best_obj:.8f}")
    except Exception as e:
        if verbose:
            print(f"    SLSQP error: {e}")

    return best_a


def try_length(a_best, n_try, time_budget, rng):
    """Try a different sequence length by interpolation + optimization."""
    n_old = len(a_best)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        return None, np.inf
    a_try = a_try / s

    elapsed = 0.0

    # Quick FISTA
    t1 = min(15.0, time_budget * 0.2)
    if t1 > 3:
        t0 = time.monotonic()
        a_try = fista_optimize(a_try, time_budget=t1)
        elapsed += time.monotonic() - t0

    # LP polish
    t2 = min(40.0, time_budget * 0.5)
    if t2 > 5:
        t0 = time.monotonic()
        a_try = lp_polish(a_try, time_budget=t2)
        elapsed += time.monotonic() - t0

    obj_try = compute_obj(a_try)
    return a_try, obj_try


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: FISTA warm-up + LP polishing + scipy SLSQP + n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    parent_loaded = False
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                parent_loaded = True
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if not parent_loaded:
        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default U-shaped init (n={n_current}): obj={best_score:.8f}")

    # Try arcsine as well
    a_arc = diverse_init(n_current, rng, mode=3)
    obj_arc = compute_obj(a_arc)
    print(f"Exp init: obj={obj_arc:.8f}")
    if obj_arc < best_score:
        best_score = obj_arc
        best_seq = a_arc.copy()

    # === Phase 1: FISTA warm-up ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 1: FISTA Warm-up ===")
        t0 = time.monotonic()
        a_fista = fista_optimize(best_seq, time_budget=min(40.0, remaining * 0.05),
                                 step_size=0.3, verbose=True)
        obj_fista = compute_obj(a_fista)
        print(f"  FISTA: obj={obj_fista:.8f} (time={time.monotonic()-t0:.1f}s)")
        if obj_fista < best_score:
            best_score = obj_fista
            best_seq = a_fista.copy()
            print(f"  ** Improved to {best_score:.8f}")

    # === Phase 2: Intensive LP polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print("\n=== Phase 2: Main LP Polishing ===")
        alloc = min(remaining * 0.4, 500.0)
        t0 = time.monotonic()
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True, max_iters=150)
        obj_lp = compute_obj(a_lp)
        print(f"  LP: obj={obj_lp:.8f} (time={time.monotonic()-t0:.1f}s)")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** Improved to {best_score:.8f}")

    # === Phase 3: scipy SLSQP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 3: SLSQP Polish ===")
        t0 = time.monotonic()
        a_sqp = scipy_polish(best_seq, time_budget=min(80.0, remaining * 0.15), verbose=True)
        obj_sqp = compute_obj(a_sqp)
        print(f"  SLSQP: obj={obj_sqp:.8f} (time={time.monotonic()-t0:.1f}s)")
        if obj_sqp < best_score:
            best_score = obj_sqp
            best_seq = a_sqp.copy()
            print(f"  ** Improved to {best_score:.8f}")

    # === Phase 4: Second LP round after SLSQP ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 4: LP Round 2 ===")
        alloc = min(remaining * 0.25, 250.0)
        t0 = time.monotonic()
        a_lp2 = lp_polish(best_seq, time_budget=alloc, verbose=True, max_iters=100)
        obj_lp2 = compute_obj(a_lp2)
        print(f"  LP2: obj={obj_lp2:.8f} (time={time.monotonic()-t0:.1f}s)")
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  ** Improved to {best_score:.8f}")

    # === Phase 5: n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print("\n=== Phase 5: n Exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 150, n_best - 100, n_best - 50,
            n_best + 50, n_best + 100, n_best + 200,
            500, 600, 700, 900, 1000, 1200
        ]))
        n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_best) >= 30]
        n_options = n_options[:6]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            alloc = min(80.0, remaining * 0.6)
            a_try, obj_try = try_length(best_seq, n_try, alloc, rng)
            if a_try is not None and obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 6: Diverse Restarts ===")
        n_cur = len(best_seq)
        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            mode = r % 5
            a_r = diverse_init(n_cur, rng, mode=mode)
            blend = 0.3
            a_r = (1.0 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            s = float(np.sum(a_r))
            if s > 0.01:
                a_r = a_r / s

            # FISTA + LP
            t0 = time.monotonic()
            a_r = fista_optimize(a_r, time_budget=min(10.0, remaining * 0.1))
            a_r = lp_polish(a_r, time_budget=min(30.0, remaining * 0.3), max_iters=40)
            obj_r = compute_obj(a_r)
            elapsed_r = time.monotonic() - t0
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} ** (time={elapsed_r:.1f}s)")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f} (time={elapsed_r:.1f}s)")

    # === Phase 7: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 7: Final Polish ===")
        alloc = min(remaining - 10, 150.0)
        if alloc > 20:
            t0 = time.monotonic()
            a_final = lp_polish(best_seq, time_budget=alloc, verbose=True, max_iters=80)
            obj_final = compute_obj(a_final)
            print(f"  Final LP: obj={obj_final:.8f} (time={time.monotonic()-t0:.1f}s)")
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()

    # Very final SLSQP check
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t0 = time.monotonic()
        a_sq = scipy_polish(best_seq, time_budget=min(remaining - 5, 50.0), verbose=True)
        obj_sq = compute_obj(a_sq)
        if obj_sq < best_score:
            best_score = obj_sq
            best_seq = a_sq.copy()
            print(f"  Final SLSQP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
