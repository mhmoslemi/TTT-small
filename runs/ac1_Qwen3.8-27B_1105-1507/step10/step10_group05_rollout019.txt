Uses KKT-Newton refinement combined with LP-based direction finding,
high-tau smoothed gradient polish, and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import fsolve
import time
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a, time_limit=10.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, optimal_M).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:
            continue
    
    return None, None

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, optimal_M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]

                s = float(np.sum(g_val))

            a_at_c = np.maximum((1 - c) * a + c * g, 0)
            a_at_c = a_at_c / a_at_c.sum()
            f_c = compute_obj(a_at_c)

    t_best = (lo + hi) / 2.0
    a_best = np.maximum((1 - t_best) * a + t_best * g, 0)
    a_best = a_best / a_best.sum()
    obj_best = compute_obj(a_best)
    return a_best, t_best, obj_best


def smoothed_gradient_fft(a, tau):
    """Compute gradient of smoothed max of conv(a,a) using FFT. Returns (grad, conv)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def kkt_newton(a_init, active_set, pos_set, max_iter=300, tol=1e-13):
    """
    Solve KKT system via Newton with Levenberg-Marquardt damping.
    Returns (a_sol, max_conv_sol, success).
    """
    n = len(a_init)
    p = len(pos_set)
    K = len(active_set)
    if p == 0 or K == 0:
        return None, None, False

    pos_arr = np.array(pos_set, dtype=np.int64)
    act_arr = np.array(active_set, dtype=np.int64)

    a_p0 = np.array([max(a_init[i], 1e-12) for i in pos_set])
    a_p0 = a_p0 / a_p0.sum()
    conv_init = compute_conv(a_init / np.sum(a_init))
    M0 = float(max(conv_init[k] for k in active_set))
    lam0 = np.ones(K) / K
    mu0 = 1.0 / p

    x = np.concatenate([a_p0, [M0], lam0, [mu0]])
    dim = p + K + 2
    I = np.eye(dim)
    damping = 1e-8

    def residuals(xv):
        a_p = np.maximum(xv[:p], 1e-15)
        Mv = xv[p]
        lamv = xv[p+1:p+1+K]
        muv = xv[p+1+K]

        a_full = np.zeros(n)
        a_full[pos_arr] = a_p
        conv = compute_conv(a_full)

        r = np.zeros(dim)
        r[0] = np.sum(a_p) - 1.0
        for ki in range(K):
            r[1 + ki] = conv[act_arr[ki]] - Mv
        for pi in range(p):
            i = pos_arr[pi]
            val = 0.0
            for ki in range(K):
                j = act_arr[ki] - i
                if 0 <= j < n:
                    val += 2.0 * lamv[ki] * a_full[j]
            r[1 + K + pi] = val - muv
        return r

    def jacobian(xv):
        a_p = np.maximum(xv[:p], 1e-15)
        lamv = xv[p+1:p+1+K]
        a_full = np.zeros(n)
        a_full[pos_arr] = a_p

        J = np.zeros((dim, dim))
        J[0, :p] = 1.0

        for ki in range(K):
            row = 1 + ki
            k_val = act_arr[ki]
            for pi in range(p):
                j = k_val - pos_arr[pi]
                if 0 <= j < n:
                    J[row, pi] = 2.0 * a_full[j]
            J[row, p] = -1.0

        for pi in range(p):
            row = 1 + K + pi
            i_val = pos_arr[pi]
            for ki in range(K):
                k_val = act_arr[ki]
                j = k_val - i_val
                if 0 <= j < n:
                    J[row, p + 1 + ki] = 2.0 * a_full[j]
                    j_idx = np.searchsorted(pos_arr, j)
                    if j_idx < p and pos_arr[j_idx] == j:
                        J[row, j_idx] += 2.0 * lamv[ki]
            J[row, p + 1 + K] = -1.0
        return J

    for it in range(max_iter):
        r = residuals(x)
        r_norm = float(np.max(np.abs(r)))
        if r_norm < tol:
            break

        J = jacobian(x)
        try:
            dx = np.linalg.solve(J + damping * I, -r)
        except np.linalg.LinAlgError:
            dx, _, _, _ = np.linalg.lstsq(J + damping * I, -r, rcond=None)

        alpha = 1.0
        for _ in range(40):
            x_try = x + alpha * dx
            r_try = residuals(x_try)
            if float(np.max(np.abs(r_try))) < r_norm * 0.999:
                break
            alpha *= 0.5

        x = x + alpha * dx
        if alpha >= 0.99:
            damping = max(damping * 0.3, 1e-14)
        else:
            damping = min(damping * 3.0, 1e4)

    r_final = residuals(x)
    if float(np.max(np.abs(r_final))) > 1e-7:
        return None, None, False

    a_p_sol = np.maximum(x[:p], 0.0)
    s = float(np.sum(a_p_sol))
    if s < 1e-10:
        return None, None, False

    a_sol = np.zeros(n)
    a_sol[pos_arr] = a_p_sol / s
    conv_sol = compute_conv(a_sol)
    max_conv = float(np.max(conv_sol))
    if max_conv > x[p] * (1 + 1e-5) + 1e-10:
        return None, None, False
    if np.min(x[p+1:p+1+K]) < -1e-5:
        return None, None, False

    return a_sol, max_conv, True


def kkt_active_set_iteration(a, time_budget=200.0, verbose=False):
    """
    KKT-based active set method: identify active set, solve KKT, update set.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Identify initial active set
    conv_a = compute_conv(a)
    sorted_idx = np.argsort(conv_a)[::-1]
    active_set = sorted_idx[:min(15, 2 * n - 1)].tolist()
    pos_set = list(range(n))

    stall_count = 0
    for outer_it in range(50):
        if time.monotonic() - start > time_budget:
            break

        a_sol, max_conv_sol, success = kkt_newton(a, active_set, pos_set)
        if not success:
            # Try with larger active set
            conv_a = compute_conv(a)
            sorted_idx = np.argsort(conv_a)[::-1]
            active_set = sorted_idx[:min(25, 2 * n - 1)].tolist()
            pos_thresh = float(np.sum(a)) / n * 0.01
            pos_set = [i for i in range(n) if a[i] > pos_thresh]
            if len(pos_set) < n // 4:
                pos_set = list(range(n))

            a_sol, max_conv_sol, success = kkt_newton(a, active_set, pos_set)
            if not success:
                stall_count += 1
                if stall_count > 10:
                    break
                continue

        obj_sol = 2.0 * n * max_conv_sol
        if obj_sol < best_obj - 1e-12:
            best_obj = obj_sol
            best_a = a_sol.copy()
            a = a_sol.copy()
            stall_count = 0
            if verbose and outer_it % 5 == 0:
                print(f"    KKT it{outer_it}: obj={best_obj:.8f}, K={len(active_set)}, P={len(pos_set)}")
        else:
            stall_count += 1

        # Update active set
        conv_new = compute_conv(a)
        max_val = float(np.max(conv_new))
        # Active set: coefficients within 1% of max
        threshold = max_val * 0.995
        new_active = [k for k in range(2 * n - 1) if conv_new[k] >= threshold]
        if len(new_active) == 0:
            new_active = [int(np.argmax(conv_new))]
        # Keep at most 40
        if len(new_active) > 40:
            sorted_new = sorted(new_active, key=lambda k: conv_new[k], reverse=True)
            new_active = sorted_new[:40]

        # Update positive set
        pos_thresh = max(1e-8, float(np.min(a[a > 0])) * 0.1 if np.any(a > 0) else 1e-8)
        new_pos = [i for i in range(n) if a[i] > pos_thresh]
        if len(new_pos) < 10:
            new_pos = list(range(n))

        # Check if sets changed significantly
        if set(new_active) != set(active_set) or set(new_pos) != set(pos_set):
            active_set = new_active
            pos_set = new_pos
        else:
            if stall_count > 15:
                # Nudge with perturbation
                break
            elif stall_count > 5:
                # Try slightly different thresholds
                threshold = max_val * 0.99
                new_active2 = [k for k in range(2 * n - 1) if conv_new[k] >= threshold]
                if len(new_active2) > len(active_set):
                    active_set = new_active2[:min(40, len(new_active2))]

    return best_a


def lp_golden_iteration(a, time_budget=200.0, verbose=False):
    """LP-based iteration with golden-section line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        a_best, t_best, obj_best = golden_section_line_search(a, g_lp)
        if obj_best < best_obj - 1e-12:
            best_obj = obj_best
            best_a = a_best
            a = a_best
            if verbose and it % 5 == 0:
                print(f"    LP-GS it{it}: obj={best_obj:.8f}, t={t_best:.4f}")
        else:
            # Perturbation to escape
            pert = rng.normal(0, 0.03, n)
            a_new = a * np.exp(pert)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            a = a_new

    return best_a


def smoothed_gd_polish(a, time_budget=100.0, tau_start=100.0, tau_end=2000.0):
    """High-tau smoothed gradient descent for final polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    v = np.zeros(n)
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, _ = smoothed_gradient_fft(y, tau)
        v = beta * v - lr * g
        a_new = y + v - np.min(y + v)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        a = a_new
        lr = max(1e-10, lr * 0.99995)
        it += 1

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5

    return best_a


def fejer_init(n):
    """Fejer kernel coefficients (triangular), normalized to sum=1."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def cosine_taper_init(n):
    """Cosine-tapered initialization."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def uniform_init(n):
    return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")

                raise ValueError("sum too small")
    except Exception:
        n_current = 800
        best_seq = fejer_init(n_current)

    return result
