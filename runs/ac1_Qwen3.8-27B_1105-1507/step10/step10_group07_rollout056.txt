"""
Peak-Focused TR-SOCP + LP Hybrid with Multi-Length Search.
Key improvements: (1) Peak-focused direction finding that targets the worst
convolution peak specifically, (2) warm-started SOCP for faster convergence,
(3) diverse structured initializations, (4) adaptive multi-length exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def tr_sscp_step(a, r):
    """
    Single TR-SOCP step. Returns (delta, M_value) or (None, None).
    Solves: min M s.t. c_old + 2*(A_cross @ delta) <= M, ||delta||_2 <= r,
            sum(a+delta) = 1, a+delta >= 0
    """
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)

    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta

    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
    ]

    if r is not None:
        constraints.append(cp.norm(delta, 2) <= r)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                d = np.asarray(delta.value).flatten()
                if np.isfinite(d).all():
                    return d, float(M.value)
        except Exception:
            continue
    return None, None


def peak_focused_lp(a, K=5, r=1.0):
    """
    Peak-focused LP: find direction that reduces the worst peak.
    min conv(a,d)[k*] s.t. conv(a,d)[k] <= 0 for k in top-K,
         ||d||_2 <= r, sum(d) = 0, a+d >= 0
    """
    n = len(a)
    a = a / a.sum()
    c = compute_conv(a)
    # Top K peaks
    top_idx = np.argsort(c)[::-1][:K]
    A_cross = build_conv_matrix(a)
    
    # Only use the top-K rows
    A_sub = A_cross[top_idx, :]
    
    d = cp.Variable(n)
    constraints = [
        A_sub[0] @ d <= 0,  # Primary: reduce worst peak
        A_sub[1:] @ d <= 0,  # Don't increase other major peaks
        cp.sum(d) == 0,
        d >= -a,  # Keep a+d >= 0
        cp.norm(d, 2) <= r,
    ]
    
    prob = cp.Problem(cp.Minimize(A_sub[0] @ d), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if d.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                d_val = np.asarray(d.value).flatten()
                if np.isfinite(d_val).all():
                    return d_val
        except Exception:
            continue
    return None


def lp_direction(a, M_target=None):
    """LP: min M s.t. A @ g <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def gradient_smoothed_descent(a, time_budget=60.0, tau_init=20.0, tau_end=400.0, seed=42):
    """Smoothed gradient descent with increasing temperature (simulated annealing style)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 3.0 / np.sqrt(n)
    momentum = 0.85
    v = np.zeros(n)
    it = 0
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        # Softmax weighting on peaks
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient of weighted convolution
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_freq = fft(a, N)
        g_full = ifft(P * np.conj(A_freq)).real
        g = 2.0 * g_full[:n]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                # Add noise to escape
                a = best_a * np.exp(rng.normal(0, 0.01, n))
                a = project_simplex(a)
                v *= 0.5
                stall = 0

        a = a_new
        lr *= 0.9997
        it += 1

    return best_a


def subgradient_refine(a, time_budget=60.0, K=3, seed=42):
    """Subgradient method focused on top-K peaks with Barzilai-Borwein step."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    momentum = 0.6
    it = 0
    prev_a = a.copy()
    prev_g = None
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        
        # Weight peaks by their value
        weights = peak_vals ** 4  # Emphasize the worst
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        # Barzilai-Borwein step size
        if prev_g is not None:
            s_bb = a - prev_a
            y_bb = g - prev_g
            ss = np.dot(s_bb, s_bb)
            sy = np.dot(s_bb, y_bb)
            if ss > 1e-10 and abs(sy) > 1e-10:
                lr_bb = ss / abs(sy)
                lr = 0.5 * (lr + min(lr_bb, 5.0 / np.sqrt(n)))

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                a = best_a * np.exp(rng.normal(0, 0.015, n))
                a = project_simplex(a)
                v *= 0.3
                lr = 2.0 / np.sqrt(n)
                stall = 0

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        lr = max(1e-6, lr * 0.999)
        it += 1

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def tr_sscp_polish(a, time_budget=200.0, r_init=0.3, verbose=False, seed=42):
    """TR-SOCP polishing with adaptive line search and peak-focused LP hybrid."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    r = r_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    use_peak_lp = 0  # Counter for when to use peak LP

    while time.monotonic() - start < time_budget:
        # Alternate between TR-SOCP and peak-focused LP
        if use_peak_lp % 3 == 0 and r < 0.5:
            # Use peak-focused LP for direction
            delta_dir = peak_focused_lp(a, K=3, r=min(r * 2, 1.0))
        else:
            delta_dir, M_val = tr_sscp_step(a, r)
            if delta_dir is None:
                g_lp, _ = lp_direction(a)
                if g_lp is None:
                    break
                delta_dir = g_lp - a

        use_peak_lp += 1
        norm_d = np.linalg.norm(delta_dir)
        if norm_d < 1e-10:
            no_improve += 1
            continue
        
        # Line search
        best_alpha = 0
        best_alpha_obj = best_obj
        alphas = [0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0, 1.5, 2.0]
        for t_alpha in alphas:
            a_try = a + t_alpha * delta_dir
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj - 1e-15:
                best_alpha_obj = obj_try
                best_alpha = t_alpha

        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-15:
            a = a + best_alpha * delta_dir
            a = np.maximum(a, 0)
            s_n = a.sum()
            if s_n > 1e-10:
                a = a / s_n
            best_obj = best_alpha_obj
            best_a = a.copy()
            no_improve = 0
            r = min(r * 1.15, 2.0)
        else:
            no_improve += 1
            r *= 0.6
            if no_improve >= 5:
                if r < 0.02:
                    a = best_a * np.exp(rng.normal(0, 0.01, n))
                    a = np.maximum(a, 0)
                    s_p = a.sum()
                    if s_p > 1e-10:
                        a = a / s_p
                    r = r_init
                    no_improve = 0
                    if it > 100:
                        break
                else:
                    r = max(r * 0.5, 0.02)

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    TR it{it}: obj={best_obj:.8f}, r={r:.4f}, t={elapsed:.0f}s")

    return best_a


def cosine_window(n):
    """Generate a cosine-windowed sequence."""
    x = np.linspace(0, 1, n)
    return 0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.5))


def gaussian_window(n, sigma_frac=0.3):
    """Generate a Gaussian window."""
    x = np.linspace(-1, 1, n)
    sigma = sigma_frac
    return np.exp(-x**2 / (2 * sigma**2))


def lanczos_window(n, a_param=3):
    """Generate a Lanczos window."""
    x = np.linspace(-1, 1, n)
    M = a_param
    result = np.sinc(x) * np.sinc(x / M)
    result[np.abs(x) >= M] = 0
    return np.maximum(result, 0)

"""
Reduced TR-SOCP + Peak-Focused LP Hybrid with Multi-Length Search.
Key improvements: (1) Reduced TR-SOCP constraining only top-K peaks for speed,
(2) Peak-focused direction finding targeting the worst convolution peak,
(3) Efficient multi-length exploration with resampling,
(4) Diverse structured initializations, (5) Adaptive time management.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_rows(a, indices):
    """Build only specified rows of the convolution matrix A where A@x = conv(a,x)."""
    n = len(a)
    A = np.zeros((len(indices), n))
    for row, k in enumerate(indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start < j_end:
            a_slice = a[k - j_end + 1: k - j_start + 1]
            A[row, j_start:j_end] = a_slice
    return A


def build_conv_matrix(a):
    """Build full matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def tr_sscp_reduced(a, r, K=80):
    """
    Reduced TR-SOCP: min M s.t. c_old[k]+2*(A[k]@d) <= M for top-K k,
    ||d||_2 <= r, sum(a+d)=1, a+d>=0.
    Returns (delta, M_value) or (None, None).
    """
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    top_idx = np.argsort(C_old)[::-1][:min(K, 2 * n - 1)]
    A_sub = build_conv_rows(a, top_idx)
    C_sub = C_old[top_idx]

    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta

    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_sub + 2.0 * (A_sub @ delta) <= M,
    ]
    if r is not None:
        constraints.append(cp.norm(delta, 2) <= r)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                d = np.asarray(delta.value).flatten()
                if np.isfinite(d).all():
                    return d, float(M.value)
        except Exception:
            continue
    return None, None


def peak_focused_socp(a, r, K=5):
    """
    Peak-focused SOCP: find direction reducing the worst peak.
    min A[k*]@d s.t. A[k]@d - A[k*]@d <= 0 for k in top-K, sum(d)=0, a+d>=0, ||d||_2<=r
    """
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    top_idx = np.argsort(C_old)[::-1][:K]
    A_sub = build_conv_rows(a, top_idx)

    d = cp.Variable(n)
    a_new = a + d

    # Objective: reduce worst peak
    obj = A_sub[0] @ d

    constraints = [
        a_new >= 0,
        cp.sum(d) == 0,
        cp.norm(d, 2) <= r,
    ]
    # Other top peaks should not become relatively worse
    for j in range(1, K):
        constraints.append((A_sub[j] - A_sub[0]) @ d <= 0)

    prob = cp.Problem(cp.Minimize(obj), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if d.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                d_val = np.asarray(d.value).flatten()
                if np.isfinite(d_val).all() and np.linalg.norm(d_val) > 1e-12:
                    return d_val
        except Exception:
            continue
    return None


def lp_direction(a, K=100):
    """LP: min M s.t. A[topK]@g <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    top_idx = np.argsort(C_old)[::-1][:min(K, 2 * n - 1)]
    A_sub = build_conv_rows(a, top_idx)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_sub @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def gradient_smoothed_descent(a, time_budget=60.0, tau_init=20.0, tau_end=400.0, seed=42):
    """Smoothed gradient descent with increasing temperature."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 3.0 / np.sqrt(n)
    momentum = 0.85
    v = np.zeros(n)
    it = 0
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_freq = fft(a, N)
        g_full = ifft(P * np.conj(A_freq)).real
        g = 2.0 * g_full[:n]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                a = best_a * np.exp(rng.normal(0, 0.01, n))
                a = project_simplex(a)
                v *= 0.5
                stall = 0

        a = a_new
        lr *= 0.9997
        it += 1

    return best_a


def subgradient_refine(a, time_budget=60.0, K=3, seed=42):
    """Subgradient method with top-K peak weighting and BB step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    momentum = 0.6
    it = 0
    prev_a = a.copy()
    prev_g = None
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]

        weights = peak_vals ** 4
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights = weights / wsum

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        if prev_g is not None:
            s_bb = a - prev_a
            y_bb = g - prev_g
            ss = np.dot(s_bb, s_bb)
            sy = np.dot(s_bb, y_bb)
            if ss > 1e-10 and abs(sy) > 1e-10:
                lr_bb = ss / abs(sy)
                lr = 0.5 * (lr + min(lr_bb, 5.0 / np.sqrt(n)))

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                a = best_a * np.exp(rng.normal(0, 0.015, n))
                a = project_simplex(a)
                v *= 0.3
                lr = 2.0 / np.sqrt(n)
                stall = 0

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        lr = max(1e-6, lr * 0.999)
        it += 1

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def tr_sscp_polish(a, time_budget=200.0, r_init=0.25, verbose=False, seed=42, K=80):
    """TR-SOCP polishing with adaptive line search and peak-focused hybrid."""
    n = len(a)

    s = a.sum()

        return a
    a = a / s

    best_a = a.copy()

        else:

                if g_lp is None:
                    break
                delta_dir = g_lp - a

        norm_d = np.linalg.norm(delta_dir)
        if norm_d < 1e-12:
            no_improve += 1
            if no_improve > 8:
                break
            r *= 0.5
            continue

        # Adaptive line search
        best_alpha = 0
        best_alpha_obj = best_obj
        alphas = [0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0, 1.3, 1.7, 2.2]
        for t_alpha in alphas:
            a_try = a + t_alpha * delta_dir
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj - 1e-15:
                best_alpha_obj = obj_try
                best_alpha = t_alpha
            elif obj_try > best_obj * 1.01:
                break  # Getting too bad, stop early

        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-15:
            a = a + best_alpha * delta_dir
            a = np.maximum(a, 0)
            s_n = a.sum()
            if s_n > 1e-10:
                a = a / s_n
            best_obj = best_alpha_obj
            best_a = a.copy()
            no_improve = 0
            r = min(r * 1.12, 2.5)
        else:
            no_improve += 1
            r *= 0.55
            if no_improve >= 6:
                if r < 0.015:
                    a = best_a * np.exp(rng.normal(0, 0.01, n))
                    a = np.maximum(a, 0)
                    s_p = a.sum()
                    if s_p > 1e-10:
                        a = a / s_p
                    r = r_init
                    no_improve = 0
                    if it > 120:
                        break
                else:
                    r = max(r, 0.015)

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    TR it{it}: obj={best_obj:.8f}, r={r:.4f}, t={elapsed:.0f}s")

    return best_a


def cosine_window(n):
    """Cosine window."""
    x = np.linspace(0, 1, n)
    return 0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.5))


def gaussian_window(n, sigma_frac=0.35):
    """Gaussian window."""
    x = np.linspace(-1, 1, n)
    return np.exp(-x**2 / (2 * sigma_frac**2))

        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_parent}): obj={best_score:.8f}")

    # === Phase 0: Structured initializations ===
    print("\n=== Phase 0: Structured Initializations ===")
    remaining = deadline - time.monotonic()

    n_test = n_parent
    inits = {
        'cosine': cosine_window(n_test),
        'gauss': gaussian_window(n_test, 0.35),
        'gauss_wide': gaussian_window(n_test, 0.5),
        'half_cos': 0.5 + 0.5 * np.cos(np.pi * np.arange(n_test) / (n_test - 1)),
    }

    for name, init in inits.items():
        init = np.maximum(init, 0)
        s = init.sum()
        if s < 1e-10:
            continue
        init = init / s
        obj = compute_obj(init)
        if obj < best_score - 1e-6:
            best_score = obj
            best_seq = init.copy()
            print(f"  {name}: obj={obj:.8f} **")
        else:
            print(f"  {name}: obj={obj:.8f}")

    # === Phase 1: Quick GD warmup ===
    print("\n=== Phase 1: Smoothed GD Warmup ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.05, 50.0)
    if p1_budget > 15:
        a_sgd = gradient_smoothed_descent(best_seq, time_budget=p1_budget,
                                          tau_init=15.0, tau_end=300.0, seed=42)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 GD: obj={obj_sgd:.8f}")

    # === Phase 2: Main TR-SOCP Polish ===
    print("\n=== Phase 2: Main TR-SOCP Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.28, 300.0)
    if p2_budget > 30:
        a_tr = tr_sscp_polish(best_seq, time_budget=p2_budget,
                              r_init=0.25, verbose=True, seed=42, K=100)
        obj_tr = compute_obj(a_tr)
        if obj_tr < best_score:
            best_score = obj_tr
            best_seq = a_tr.copy()
            print(f"  Phase2 TR: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 TR: obj={obj_tr:.8f}")

    # === Phase 3: Subgradient refinement ===
    print("\n=== Phase 3: Subgradient Refine ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.06, 50.0)
    if p3_budget > 10:
        a_sg = subgradient_refine(best_seq, time_budget=p3_budget, K=3, seed=77)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase3 SG: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 SG: obj={obj_sg:.8f}")

    # === Phase 4: Multi-length exploration (focus on longer sequences) ===
    print("\n=== Phase 4: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Try longer sequences first (more room for improvement)
    n_options = sorted(set([
        n_current + 200, n_current + 400, n_current + 600,
        1200, 1500, 1800, 2000
    ]))
    n_options = [n for n in n_options if 600 <= n <= 2500 and n > n_current]
    # Also try one shorter
    n_short = n_current - 200
    if n_short >= 500:
        n_options.append(n_short)
    n_options = sorted(n_options)

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # Quick warmup
        alloc1 = min(remaining * 0.08, 35.0)
        if alloc1 > 10:
            a_try = gradient_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=200.0, seed=n_try)

        # TR-SOCP polish with larger K for longer sequences
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.12, 120.0)
        if alloc2 > 20:
            K_use = min(150, max(80, int(0.05 * (2 * n_try - 1))))
            a_try = tr_sscp_polish(a_try, time_budget=alloc2, r_init=0.2, K=K_use)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Perturbation + TR-SOCP ===
    print("\n=== Phase 5: Perturbation + TR-SOCP ===")
    remaining = deadline - time.monotonic()

    for r_idx in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        scale = 0.004 + 0.004 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            break
        a_pert = a_pert / s_pert

        alloc = min(remaining * 0.15, 80.0)
        if alloc > 20:
            a_pert = tr_sscp_polish(a_pert, time_budget=alloc,
                                    r_init=0.15, seed=400 + r_idx, K=100)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 6: Final intensive TR-SOCP ===
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.35, 150.0)
        a_final = tr_sscp_polish(best_seq, time_budget=alloc,
                                 r_init=0.1, verbose=True, seed=999, K=120)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final TR: obj={best_score:.8f} **")
        else:
            print(f"  Final TR: obj={obj_final:.8f}")

    # === Phase 7: Final subgradient touch-up ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 40.0)
        a_sg_final = subgradient_refine(best_seq, time_budget=alloc, K=2, seed=1234)
        obj_sg_final = compute_obj(a_sg_final)
        if obj_sg_final < best_score:
            best_score = obj_sg_final
            best_seq = a_sg_final.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
