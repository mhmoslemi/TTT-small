import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_gradient(a, tau):
    """
    Compute the gradient of the softmax-smoothed objective w.r.t. a.
    Returns gradient vector of length n.
    f(a) = (1/tau) * log(sum_k exp(tau * conv[k]))
    grad_i f = 2 * sum_k p[k] * a[k-i] (with valid index bounds)
    where p[k] = exp(tau*conv[k]) / sum(exp(tau*conv[j]))

    conv = compute_conv(a)
    # Softmax weights with numerical stability
    conv_shifted = conv - np.max(conv)
    exp_vals = np.exp(tau * conv_shifted)
    p = exp_vals / float(np.sum(exp_vals))
    # Cross-correlation: g[i] = 2 * sum_k p[k] * a[k-i] (valid k-i in [0,n-1])
    # This is equivalent to: for each k, add p[k] * a[max(0,k-n+1):min(n,k+1)] shifted
    # Efficient via FFT: circular cross-correlation

    P = fft(p, N)
    A = fft(a, N)
    # conv[p, a](i) = sum_k p[k] * a[k-i] = cross-correlation
    # In FFT: ifft(fft(p) * conj(fft(a))) gives circular cross-correlation

    g = np.zeros(n)
    for i in range(n):

        # = sum_{k=i}^{i+n-1} p[k] * a[k-i] (for linear, k ranges 0..2n-2)
        # The circular version wraps, so we need to subtract the wrap
        # Actually, let's just compute it directly via the circular result
        pass
    # Better approach: use the circular result and fix the wrap
    # g_circ[i] = sum_j p_circ[j] * a[(j-i) mod N]
    # We want g[i] = sum_{k=0}^{2n-2} p[k] * a[k-i] for 0<=k-i<n
    # = sum_{j=0}^{n-1} p[i+j] * a[j]  (where p is padded to length N)
    # This is a sliding dot product, which is the cross-correlation (p, a)
    # cross_corr(i) = sum_j p[i+j] * a[j]
    # In FFT terms: ifft(conj(fft(p)) * fft(a)) gives correlation(a, p)

    p_padded[:len(p)] = p
    a_padded = np.zeros(N)
    a_padded[:n] = a
    # cross-correlation: (p * a)(i) = sum_j p[i+j] * a[j]
    g_circ = ifft(np.conj(fft(p_padded, N)) * fft(a_padded, N)).real
    # g_circ[i] for i=0..2n-2 gives sum_j p[i+j]*a[j] (with wrap)

    # So g[i] = g_circ[i] for i in 0..n-1
    g = 2.0 * g_circ[:n]
    return g, p, conv

def solve_tr_sscp(a, r, M_target=None):
    """
    Solve trust region SOCP:
    min M
    s.t. C_old[k] + 2*(a*delta)[k] <= M
         ||delta||_2 <= r
         sum(a+delta) = 1
         a+delta >= 0
    """
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)

    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta

    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
    ]

    if r is not None:
        constraints.append(cp.norm(delta, 2) <= r)

    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_direction(a, M_target=None):
    """Solve LP: min M s.t. A@x <= M, sum(x)=1, x>=0."""
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    x = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ x <= M, cp.sum(x) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.maximum(x.value, 0)
                s = float(np.sum(x_val))
                if s > 1e-10:
                    return x_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def tr_sscp_polish(a, time_budget=300.0, r_init=0.3, verbose=False, seed=42):
    """Trust region SOCP polishing with adaptive line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    r = r_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        delta_dir, M_val = solve_tr_sscp(a, r)

        if delta_dir is None:
            g_lp, M_val = solve_lp_direction(a)
            if g_lp is None:
                break
            delta_dir = g_lp - a
        else:
            norm_d = np.linalg.norm(delta_dir)
            if norm_d > r:
                delta_dir = delta_dir * (r / norm_d)

        finite_mask = np.isfinite(delta_dir)
        if not finite_mask.all():
            delta_dir = np.where(finite_mask, delta_dir, 0.0)

        best_alpha = 0
        best_alpha_obj = best_obj
        alphas = [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0, 1.3, 1.7, 2.2]
        for t_alpha in alphas:
            a_try = a + t_alpha * delta_dir
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj - 1e-15:
                best_alpha_obj = obj_try
                best_alpha = t_alpha

        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-15:
            a = a + best_alpha * delta_dir
            a = np.maximum(a, 0)
            s_n = a.sum()
            if s_n > 1e-10:
                a = a / s_n
            best_obj = best_alpha_obj
            best_a = a.copy()
            no_improve = 0
            r = min(r * 1.12, 1.8)
        else:
            no_improve += 1
            r *= 0.65
            if no_improve >= 5:
                if r < 0.03:
                    a = best_a * np.exp(rng.normal(0, 0.015, n))
                    a = np.maximum(a, 0)
                    s_p = a.sum()
                    if s_p > 1e-10:
                        a = a / s_p
                    r = r_init
                    no_improve = 0
                    if it > 80:
                        break
                else:
                    r = max(r * 0.5, 0.03)

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    TR it{it}: obj={best_obj:.8f}, r={r:.4f}, t={elapsed:.0f}s")

    return best_a


def lp_polish(a, time_budget=200.0, verbose=False, seed=42):
    """LP-based polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        best_t = 0
        best_t_obj = best_obj
        for t_val in [0.03, 0.08, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0]:
            a_try = (1 - t_val) * best_a + t_val * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-15:
                best_t_obj = obj_try
                best_t = t_val

        if best_t > 0 and best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = (1 - best_t) * best_a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            s_ba = best_a.sum()
            if s_ba > 1e-10:
                best_a = best_a / s_ba
            a = best_a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                scale = 0.008 + 0.005 * min(no_improve, 8)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                no_improve = 0
                if it > 60:
                    break

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def block_coordinate_refine(a, time_budget=120.0, block_size=None, verbose=False):
    """
    Block coordinate descent: divide sequence into blocks,
    optimize each block via LP while fixing others.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if block_size is None:
        block_size = max(20, n // 10)

    n_blocks = (n + block_size - 1) // block_size
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget and no_improve < n_blocks * 3:
        for b in range(n_blocks):
            i_start = b * block_size
            i_end = min(n, i_start + block_size)
            if i_end - i_start < 3:
                continue

            # Fix all coordinates outside the block, optimize inside
            # Build the QP/LP for the block
            n_b = i_end - i_start
            a_fix = a.copy()

            # The contribution to conv from the block variables
            # conv[k] = sum_{i in block} a[i] * a_fix[k-i] + (fixed contribution)
            # We want to minimize max_k (fixed_conv[k] + sum_{i in block} a[i] * a_fix[k-i])
            # subject to sum(a[block]) = a_fix[block].sum(), a[block] >= 0

            # Compute fixed contribution (block set to 0)
            a_zero = a_fix.copy()
            a_zero[i_start:i_end] = 0
            conv_fixed = compute_conv(a_zero)

            # Cross terms: for each block coordinate i, its contribution to conv[k]
            # is a[i] * a_fix[k-i] for valid k-i
            # Build the matrix for the block
            A_block = np.zeros((2 * n - 1, n_b))
            for bi in range(n_b):
                gi = i_start + bi
                for k in range(2 * n - 1):
                    j = k - gi
                    if 0 <= j < n and not (i_start <= j < i_end):
                        A_block[k, bi] = a_fix[j]

            # Current sum in block
            block_sum = sum(a_fix[i_start:i_end])

            x = cp.Variable(n_b, nonneg=True)
            M = cp.Variable()
            constraints = [
                cp.sum(x) == block_sum,
                conv_fixed + 2.0 * (A_block @ x) <= M,
            ]

            prob = cp.Problem(cp.Minimize(M), constraints)
            solved = False
            for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
                try:
                    prob.solve(solver=solver, verbose=False)
                    if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                        x_val = np.maximum(x.value, 0)
                        s_x = float(np.sum(x_val))
                        if s_x > 1e-15:
                            x_val = x_val * (block_sum / s_x)
                            a_fix[i_start:i_end] = x_val
                            solved = True
                        break
                except Exception:
                    continue

            if solved:
                obj_new = compute_obj(a_fix)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_fix.copy()
                    no_improve = 0
                else:
                    no_improve += 1

            it += 1
            if time.monotonic() - start >= time_budget:
                break

        if verbose and it % n_blocks == 0:
            elapsed = time.monotonic() - start
            print(f"    BCD round{it // n_blocks}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def diverse_initializations(n, rng, num_starts=5):
    """Generate diverse initial sequences."""
    inits = []
    # Uniform
    inits.append(np.ones(n) / n)
    # Triangular (concentrated in middle)
    tri = np.arange(1, n + 1)
    tri = np.concatenate([tri, tri[::-1]])[:n]
    inits.append(tri / tri.sum())
    # Two peaks
    two = np.zeros(n)
    two[n // 4] = 1.0
    two[3 * n // 4] = 1.0
    two += 0.1 * np.ones(n)
    inits.append(two / two.sum())
    # Random with structure (multi-scale)
    for _ in range(num_starts - 4):
        a = np.abs(rng.normal(0, 1, n))
        # Apply multi-scale structure
        for scale in [2, 4, 8]:
            if n % scale == 0:
                blocks = a.reshape(-1, scale)
                block_means = blocks.mean(axis=1, keepdims=True)
                a = (blocks - block_means * 0.5).reshape(-1)
        a = np.abs(a) + 0.01
        inits.append(a / a.sum())
    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Mirror Descent + TR-SOCP + Block Coordinate hybrid.
    Uses entropic mirror descent as the workhorse (fast, O(n log n) per iter),
    TR-SOCP for fine polishing, LP for direction finding, and block
    coordinate descent for local refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Mirror Descent Refined (primary, uses ~35% time) ===
    print(f"\n=== Phase 1: Mirror Descent Refined ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 450.0)
    if p1_budget > 30:
        a_md = mirror_descent_refined(best_seq, time_budget=p1_budget, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Phase1 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 MD: obj={obj_md:.8f}")

    # === Phase 2: TR-SOCP Polish ===
    print(f"\n=== Phase 2: TR-SOCP Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 30:
        a_tr = tr_sscp_polish(best_seq, time_budget=p2_budget, r_init=0.25, verbose=True, seed=42)
        obj_tr = compute_obj(a_tr)
        if obj_tr < best_score:
            best_score = obj_tr
            best_seq = a_tr.copy()
            print(f"  Phase2 TR: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 TR: obj={obj_tr:.8f}")

    # === Phase 3: LP Polish ===
    print(f"\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 120.0)
    if p3_budget > 20:
        a_lp = lp_polish(best_seq, time_budget=p3_budget, verbose=True, seed=77)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase3 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 LP: obj={obj_lp:.8f}")

    # === Phase 4: Block Coordinate Descent ===
    print(f"\n=== Phase 4: Block Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 100.0)
    if p4_budget > 20:
        a_bcd = block_coordinate_refine(best_seq, time_budget=p4_budget, verbose=True)
        obj_bcd = compute_obj(a_bcd)
        if obj_bcd < best_score:
            best_score = obj_bcd
            best_seq = a_bcd.copy()
            print(f"  Phase4 BCD: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 BCD: obj={obj_bcd:.8f}")

    # === Phase 5: Mirror Descent again (on improved sequence) ===
    print(f"\n=== Phase 5: Mirror Descent Round 2 ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.15, 120.0)
    if p5_budget > 20:
        a_md2 = mirror_descent_refined(best_seq, time_budget=p5_budget, verbose=False)
        obj_md2 = compute_obj(a_md2)
        if obj_md2 < best_score:
            best_score = obj_md2
            best_seq = a_md2.copy()
            print(f"  Phase5 MD2: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 MD2: obj={obj_md2:.8f}")

    # === Phase 6: Perturbation + Optimization (multi-start) ===
    print(f"\n=== Phase 6: Multi-Start Perturbation ===")
    remaining = deadline - time.monotonic()

    scales = [0.005, 0.01, 0.015, 0.025, 0.04]
    for r_idx, scale in enumerate(scales):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        # Allocate time: half MD, half TR
        alloc_md = min(remaining * 0.2, 50.0)
        alloc_tr = min(remaining * 0.15, 40.0)

        if alloc_md > 10:
            a_pert = mirror_descent_refined(a_pert, time_budget=alloc_md, verbose=False)
        remaining = deadline - time.monotonic()
        if alloc_tr > 10 and remaining > 20:
            a_pert = tr_sscp_polish(a_pert, time_budget=min(alloc_tr, remaining - 10),
                                    r_init=0.2, seed=200 + r_idx)

        obj_pert = compute_obj(a_pert)
        if obj_pert < best_score:
            best_score = obj_pert
            best_seq = a_pert.copy()
            print(f"  Pert{r_idx} (scale={scale:.4f}): obj={best_score:.8f} **")
        else:
            print(f"  Pert{r_idx} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(500, n_current - 100),
        n_current + 100,
        n_current + 200,
        1000, 1200,
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_current) > 50]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)

        alloc1 = min(remaining * 0.15, 60.0)
        if alloc1 > 15:
            a_try = mirror_descent_refined(a_try, time_budget=alloc1, verbose=False)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.12, 50.0)
        if alloc2 > 15:
            a_try = tr_sscp_polish(a_try, time_budget=alloc2, r_init=0.2)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 100.0)
        a_final = tr_sscp_polish(best_seq, time_budget=alloc, r_init=0.15, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final TR: obj={best_score:.8f} **")
        else:
            print(f"  Final TR: obj={obj_final:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 50.0)
        a_md_final = mirror_descent_refined(best_seq, time_budget=alloc, verbose=False)
        obj_mdf = compute_obj(a_md_final)
        if obj_mdf < best_score:
            best_score = obj_mdf
            best_seq = a_md_final.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # === Phase 9: Final LP squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 30.0)
        a_lp_final = lp_polish(best_seq, time_budget=alloc, seed=3000)
        obj_lpf = compute_obj(a_lp_final)
        if obj_lpf < best_score:
            best_score = obj_lpf
            best_seq = a_lp_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
