"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
import heapq

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A

def solve_lp_minpeak(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None

def solve_lp_weighted_peaks(a, num_peaks=5, verbose=False):

    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    conv_current = compute_conv(a_norm)
    max_val = float(np.max(conv_current))

    weights = weights / weights.sum() * (2 * n - 1)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    Ag = A @ g
    
    # Minimize weighted sum of top peaks + small penalty on max
    obj = cp.sum(weights * Ag) + 0.1 * M
    constraints = [cp.sum(g) == 1]

                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    
    return None

def golden_section_search(f, a, b, tol=1e-12, max_iter=60):
    """Golden section search for minimum of f on [a, b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2.0

def coordinate_descent_priority(a, time_budget=300.0, verbose=True, seed=42):
    """
    Priority-based coordinate descent.
    Uses a priority queue to order coordinates by their impact on active peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1

        # Recompute conv periodically
        if sweep % 5 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

            # Compute priority based on gradient
            max_conv = float(np.max(conv))
            # Active set: elements within 1% of max
            active = np.where(conv > 0.99 * max_conv)[0]
            
            # Priority: coordinates that appear in active convolution elements
            priority = np.zeros(n)
            for k in active:
                j_min = max(0, k - n + 1)
                j_max = min(n - 1, k)
                for j in range(j_min, j_max + 1):
                    other = k - j
                    if 0 <= other < n:
                        priority[j] += conv[k] * a[other]
            
            # Add some randomness to avoid cycling
            priority += rng.uniform(0, 0.01 * np.max(priority) if np.max(priority) > 0 else 0.01, n)
            order = np.argsort(-priority)
        else:
            order = rng.permutation(n)

        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v

            if S_other < 1e-12:
                continue

            # Determine valid k range for coordinate i
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)

            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k_lin.copy() if len(c_k_lin) > 0 else np.array([])

            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

                    best_f = fx

            x_lo = max(0.0, best_x - scale)

def pairwise_exchange_descent(a, time_budget=200.0, verbose=True, seed=42):
    """
    Pairwise mass transfer optimization.
    For pairs (i, j), optimize the amount of mass to transfer.
    This maintains the sum constraint exactly.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 5 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        # Generate random pairs
        n_pairs = min(500, n * 5)
        for _ in range(n_pairs):
            if time.monotonic() - start >= time_budget:
                break

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            vi = a[i]
            vj = a[j]

            # For transferring t from i to j: a[i] -> vi - t, a[j] -> vj + t
            # Valid range: t in [-vj, vi]
            t_lo = -vj
            t_hi = vi

            if t_hi - t_lo < 1e-15:
                continue

            # The effect on conv:
            # conv[k] changes by: -t * (contribution from i) + t * (contribution from j)
            # Specifically:
            # For element i: conv[k] += (vi-t)^2 - vi^2 = -2*vi*t + t^2 (at k=2i)
            #                 conv[k] += 2*(vi-t)*a[k-i] - 2*vi*a[k-i] = -2*t*a[k-i] (at k=i+m for m≠i)
            # For element j: conv[k] += (vj+t)^2 - vj^2 = 2*vj*t + t^2 (at k=2j)
            #                 conv[k] += 2*(vj+t)*a[k-j] - 2*vj*a[k-j] = 2*t*a[k-j] (at k=j+m for m≠j)

            # Build the linear coefficients for each conv element
            # conv_new[k] = conv[k] + linear_coeff[k] * t + quad_coeff[k] * t^2
            linear = np.zeros(2 * n - 1)
            quad = np.zeros(2 * n - 1)

            # Element i contribution
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                m = k - i
                if m == i:
                    # quadratic: -2*vi*t + t^2
                    linear[k] += -2.0 * vi
                    quad[k] += 1.0
                else:
                    # linear: -2*t * a[m]
                    linear[k] += -2.0 * a[m]

            # Element j contribution
            for k in range(max(0, j), min(2 * n - 1, j + n)):
                m = k - j
                if m == j:
                    # quadratic: 2*vj*t + t^2
                    linear[k] += 2.0 * vj
                    quad[k] += 1.0
                else:
                    # linear: 2*t * a[m]
                    linear[k] += 2.0 * a[m]

            # Now find t that minimizes max_k (conv[k] + linear[k]*t + quad[k]*t^2)
            # This is a 1D optimization problem
            
            def f_pair(t, conv=conv, linear=linear, quad=quad, n=n):
                vals = conv + linear * t + quad * t * t
                return 2.0 * n * float(np.max(vals))

            # Golden section on [t_lo, t_hi]
            t_opt = golden_section_search(f_pair, t_lo, t_hi, tol=1e-12, max_iter=35)
            t_opt = np.clip(t_opt, t_lo, t_hi)

            if abs(t_opt) < 1e-15:
                continue

            # Update
            a[i] = vi - t_opt
            a[j] = vj + t_opt

            # Update conv
            conv += linear * t_opt + quad * t_opt * t_opt

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    Pairwise sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a

def lp_polish(a, time_budget=60.0, verbose=False, seed=42):
    """LP-based polishing with multiple strategies."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Strategy 1: standard LP
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is not None:
            best_t_obj = best_obj
            best_a_local = best_a.copy()
            for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_try = (1.0 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_a_local = a_try.copy()
            
            if best_t_obj < best_obj - 1e-12:
                best_obj = best_t_obj
                best_a = best_a_local.copy()
                a = best_a
                stall = 0
            else:
                stall += 1

        # Strategy 2: weighted peak LP
        if remaining > 10:
            g_w = solve_lp_weighted_peaks(a, num_peaks=5)
            if g_w is not None:
                for t in [0.1, 0.3, 0.5]:
                    a_try = (1.0 - t) * a + t * g_w
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = best_a
                        stall = 0

        if stall > 3:
            pert = rng.normal(0, 0.02, n)
            a = a * np.exp(pert)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0

    return best_a

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 8, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Symmetric initialization
        half = n // 2
        a_half = rng.uniform(0.5, 1.5, half)
        a = np.concatenate([a_half, a_half[::-1]])
        if n % 2 == 1:
            a = a[:n]
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def multistage_optimize(a, time_budget=300.0, verbose=True, seed=42):
    """
    Multi-stage optimization combining all methods.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    start = time.monotonic()

    # Stage 1: Priority coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    t1 = min(remaining * 0.4, 150.0)
    if t1 > 20:
        if verbose:
            print(f"      Stage 1: Priority CD (budget={t1:.0f}s)")
        a = coordinate_descent_priority(a, time_budget=t1, verbose=False, seed=seed)

    # Stage 2: Pairwise exchange
    remaining = time_budget - (time.monotonic() - start)
    t2 = min(remaining * 0.3, 100.0)
    if t2 > 20:
        if verbose:
            print(f"      Stage 2: Pairwise exchange (budget={t2:.0f}s)")
        a2 = pairwise_exchange_descent(a, time_budget=t2, verbose=False, seed=seed + 1)
        obj_a = 2.0 * n * float(np.max(compute_conv(a / a.sum())))
        obj_a2 = 2.0 * n * float(np.max(compute_conv(a2 / a2.sum())))
        if obj_a2 < obj_a:
            a = a2

    # Stage 3: LP polish
    remaining = time_budget - (time.monotonic() - start)
    t3 = min(remaining * 0.25, 80.0)
    if t3 > 15:
        if verbose:
            print(f"      Stage 3: LP polish (budget={t3:.0f}s)")
        a3 = lp_polish(a, time_budget=t3, verbose=False, seed=seed + 2)
        obj_a = 2.0 * n * float(np.max(compute_conv(a / a.sum())))
        obj_a3 = 2.0 * n * float(np.max(compute_conv(a3 / a3.sum())))
        if obj_a3 < obj_a:
            a = a3

    # Stage 4: Final CD
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        if verbose:
            print(f"      Stage 4: Final CD (budget={remaining:.0f}s)")
        a = coordinate_descent_priority(a, time_budget=remaining * 0.8, verbose=False, seed=seed + 3)

    return a

"""
Optimizes step function coefficients using priority-based coordinate descent with
pairwise mass transfer, active-set tracking, multi-scale refinement, and LP polishing.
Key innovations:
1. Priority queue ordering based on gradient importance at active peaks
2. Pairwise mass transfer optimization (maintains sum constraint exactly)
3. Active convolution index tracking for focused optimization
4. Multi-scale (coarse-to-fine) optimization with resampling
5. Enhanced LP polishing with weighted peak minimization
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2n*max(conv) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None


def solve_lp_weighted(a):
    """Solve LP minimizing weighted sum of peaks of conv(a,g)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    conv_current = compute_conv(a_norm)
    max_val = float(np.max(conv_current))

    weights = np.exp(50.0 * (conv_current - max_val))
    weights = weights / weights.sum() * (2 * n - 1)

    g = cp.Variable(n, nonneg=True)
    Ag = A @ g

    obj = cp.sum(weights * Ag)
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue

    return None


def _coord_obj(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i):
    """1D objective for coordinate i: minimize 2n*max(conv)/(S+x)^2."""
    if len(a_ki_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom


def _pair_obj(t, conv, linear, quad, n):
    """1D objective for pairwise exchange: minimize 2n*max(conv + linear*t + quad*t^2)."""
    vals = conv + linear * t + quad * t * t
    return 2.0 * n * float(np.max(vals))


def _gs_search_coord(x_lo, x_hi, v, S_other, n, c_k_lin, a_ki_lin, c_2i, max_iter=40):
    """Golden section search for coordinate subproblem."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if x_hi - x_lo < 1e-14:
        return x_lo
    c = x_hi - gr * (x_hi - x_lo)
    d = x_lo + gr * (x_hi - x_lo)
    fc = _coord_obj(c, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
    fd = _coord_obj(d, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
    for _ in range(max_iter):
        if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
            break
        if fc < fd:
            x_hi = d
            d = c
            fd = fc
            c = x_hi - gr * (x_hi - x_lo)
            fc = _coord_obj(c, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
        else:
            x_lo = c
            c = d
            fc = fd
            d = x_lo + gr * (x_hi - x_lo)
            fd = _coord_obj(d, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
    return max(0.0, (x_lo + x_hi) / 2.0)


def _gs_search_pair(x_lo, x_hi, conv, linear, quad, n, max_iter=35):
    """Golden section search for pairwise exchange subproblem."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if x_hi - x_lo < 1e-14:
        return x_lo
    c = x_hi - gr * (x_hi - x_lo)
    d = x_lo + gr * (x_hi - x_lo)
    fc = _pair_obj(c, conv, linear, quad, n)
    fd = _pair_obj(d, conv, linear, quad, n)
    for _ in range(max_iter):
        if x_hi - x_lo < 1e-13 * max(1.0, abs(x_hi), abs(x_lo)):
            break
        if fc < fd:
            x_hi = d
            d = c
            fd = fc
            c = x_hi - gr * (x_hi - x_lo)
            fc = _pair_obj(c, conv, linear, quad, n)
        else:
            x_lo = c
            c = d
            fc = fd
            d = x_lo + gr * (x_hi - x_lo)
            fd = _pair_obj(d, conv, linear, quad, n)
    return (x_lo + x_hi) / 2.0


def coordinate_descent_priority(a, time_budget=300.0, verbose=True, seed=42):
    """
    Priority-based coordinate descent with exact 1D subproblem solving.
    Coordinates are ordered by their impact on active (near-max) convolution elements.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 5 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

            # Compute priority based on active peaks
            max_conv = float(np.max(conv))
            threshold = max_conv * 0.995
            active = np.where(conv >= threshold)[0]

            # Limit active set size for efficiency
            if len(active) > 20:
                idx = np.argsort(-conv[active])[:20]
                active = active[idx]

            priority = np.zeros(n)
            for k in active:
                j_min = max(0, k - n + 1)
                j_max = min(n - 1, k)
                for j in range(j_min, j_max + 1):
                    other = k - j
                    if 0 <= other < n:
                        priority[j] += conv[k] * a[other]

            # Add small random perturbation to break ties
            if np.max(priority) > 0:
                priority += rng.uniform(0, 0.005 * np.max(priority), n)
            else:
                priority += rng.uniform(0, 0.01, n)

            order = np.argsort(-priority)
        else:
            order = rng.permutation(n)

        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v

            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)

            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask].copy()

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Quick scan to find good bracket
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.5, v * 0.75, v, v * 1.25, v * 1.5, v * 2.0,
                          0.5 / n, 1.0 / n, 2.0 / n, 4.0 / n]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))
            candidates = [c for c in candidates if c < 20.0]

            best_x = v
            best_f = _coord_obj(v, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            for x in candidates:
                fx = _coord_obj(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 3.0
            if x_hi < x_lo:
                x_lo, x_hi = 0.0, max(2.0 / n, 0.01)
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-8

            x_opt = _gs_search_coord(x_lo, x_hi, v, S_other, n,
                                      c_k_lin, a_ki_lin, c_2i, max_iter=38)

            if abs(x_opt - v) < 1e-15:
                continue

            # Update a and conv incrementally
            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

        if verbose and sweep % 25 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD-priority sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def pairwise_exchange_descent(a, time_budget=200.0, verbose=True, seed=42):
    """
    Pairwise mass transfer optimization.
    For random pairs (i,j), optimizes the amount of mass to transfer from i to j.
    Maintains sum constraint exactly.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 5 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        # Determine number of pairs based on time
        elapsed = time.monotonic() - start
        n_pairs = min(300, max(50, int(200.0 * (time_budget - elapsed) / time_budget)))

        for _ in range(n_pairs):
            if time.monotonic() - start >= time_budget:
                break

            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue

            vi = a[i]
            vj = a[j]

            # Valid range for t (transfer from i to j): t in [-vj, vi]
            t_lo = -vj
            t_hi = vi

            if t_hi - t_lo < 1e-15:
                continue

            # Build linear and quad coefficients
            linear = np.zeros(2 * n - 1)
            quad = np.zeros(2 * n - 1)

            # Element i: a[i] -> a[i] - t
            for k in range(i, min(2 * n - 1, i + n)):
                m = k - i
                if m == i:
                    linear[k] += -2.0 * vi
                    quad[k] += 1.0
                else:
                    linear[k] += -2.0 * a[m]

            # Element j: a[j] -> a[j] + t
            for k in range(j, min(2 * n - 1, j + n)):
                m = k - j
                if m == j:
                    linear[k] += 2.0 * vj
                    quad[k] += 1.0
                else:
                    linear[k] += 2.0 * a[m]

            # Golden section search
            t_opt = _gs_search_pair(t_lo, t_hi, conv, linear, quad, n, max_iter=30)
            t_opt = np.clip(t_opt, t_lo, t_hi)

            if abs(t_opt) < 1e-15:
                continue

            # Update
            a[i] = vi - t_opt
            a[j] = vj + t_opt
            conv += linear * t_opt + quad * t_opt * t_opt

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    Pairwise sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def lp_polish(a, time_budget=60.0, verbose=False, seed=42):

        return a

    start = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Strategy 1: standard LP
        g_lp, M_lp = solve_lp_minpeak(a)
        improved = False
        if g_lp is not None:

                a_try = (1.0 - t) * a + t * g_lp

                if s_try < 1e-10:

                a_try = a_try / s_try

                    best_a = a_try.copy()
                    a = a_try.copy()
                    stall = 0
                    improved = True
                    break

                    if s_try < 1e-10:

                    a_try = a_try / s_try
                    obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
                    if obj_try < best_obj - 1e-12:

                        best_a = a_try.copy()
                        a = a_try.copy()
                        stall = 0
                        improved = True
                        break

        if not improved:
            stall += 1
            if stall > 4:
                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                sa = float(np.sum(a))
                if sa > 1e-10:
                    a = a / sa
                stall = 0

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [max(1, n // 8), max(1, n // 8)]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2.0 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x ** 2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 8, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2.0 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        half = n // 2
        a_half = rng.uniform(0.5, 1.5, half)
        if n % 2 == 0:
            a = np.concatenate([a_half, a_half[::-1]])
        else:
            a = np.concatenate([a_half, [1.0], a_half[::-1]])
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def multistage_optimize(a, time_budget=300.0, verbose=True, seed=42):
    """
    Multi-stage optimization: priority CD -> pairwise -> LP -> final CD.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    start = time.monotonic()

    # Stage 1: Priority coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    t1 = min(remaining * 0.40, 150.0)
    if t1 > 20:
        if verbose:
            print(f"      Stage 1: Priority CD (budget={t1:.0f}s)")
        a = coordinate_descent_priority(a, time_budget=t1, verbose=False, seed=seed)

    # Stage 2: Pairwise exchange
    remaining = time_budget - (time.monotonic() - start)
    t2 = min(remaining * 0.30, 100.0)
    if t2 > 20:
        if verbose:
            print(f"      Stage 2: Pairwise (budget={t2:.0f}s)")
        a2 = pairwise_exchange_descent(a, time_budget=t2, verbose=False, seed=seed + 1)
        obj_a = compute_obj_normalized(a)
        obj_a2 = compute_obj_normalized(a2)
        if obj_a2 < obj_a:
            a = a2

    # Stage 3: LP polish
    remaining = time_budget - (time.monotonic() - start)
    t3 = min(remaining * 0.20, 80.0)
    if t3 > 15:
        if verbose:
            print(f"      Stage 3: LP (budget={t3:.0f}s)")
        a3 = lp_polish(a, time_budget=t3, verbose=False, seed=seed + 2)
        obj_a = compute_obj_normalized(a)
        obj_a3 = compute_obj_normalized(a3)
        if obj_a3 < obj_a:
            a = a3

    # Stage 4: Final CD refinement
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        if verbose:
            print(f"      Stage 4: Final CD (budget={remaining * 0.8:.0f}s)")
        a = coordinate_descent_priority(a, time_budget=remaining * 0.8,
                                         verbose=False, seed=seed + 3)

    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines priority coordinate descent, pairwise exchange,
    LP polishing, multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = 2.0 * n_default * float(np.max(compute_conv(best_seq)))
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    # === Phase 1: Main multi-stage optimization ===
    print(f"\n=== Phase 1: Multi-stage (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 450.0)
    if p1_budget > 60:
        a_opt = multistage_optimize(best_seq, time_budget=p1_budget,
                                     verbose=True, seed=seed)
        obj_opt = compute_obj_normalized(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_opt:.8f} (no improvement)")

    # === Phase 2: Extended pairwise exchange ===
    print(f"\n=== Phase 2: Pairwise exchange ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 180.0)
    if p2_budget > 30:
        a_pair = pairwise_exchange_descent(best_seq, time_budget=p2_budget,
                                            verbose=True, seed=seed + 100)
        obj_pair = compute_obj_normalized(a_pair)
        if obj_pair < best_score:
            best_score = obj_pair
            best_seq = a_pair.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_pair:.8f}")

    # === Phase 3: LP polishing rounds ===
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 250.0)
    if p3_budget > 30:
        for rnd in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(p3_budget / (6 - rnd), remaining * 0.5)
            if alloc < 12:
                break
            a_lp = lp_polish(best_seq, time_budget=alloc, verbose=False,
                             seed=2000 + rnd * 137)
            obj_lp = compute_obj_normalized(a_lp)
            if obj_lp < best_score - 1e-12:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                # Quick CD after LP if no improvement
                remaining2 = deadline - time.monotonic()
                if remaining2 > 30:
                    a_cd = coordinate_descent_priority(a_lp, time_budget=min(40, remaining2 * 0.15),
                                                       verbose=False, seed=3000 + rnd * 71)
                    obj_cd = compute_obj_normalized(a_cd)
                    if obj_cd < best_score:
                        best_score = obj_cd
                        best_seq = a_cd.copy()
                        print(f"  CD after LP {rnd}: obj={best_score:.8f} **")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best + 100, n_best + 200, n_best + 400, n_best + 600,
        n_best + 800, 1000, 1200, 1500, 2000, 2500
    ]))
    n_options = [n for n in n_options if 500 <= n <= 3000 and abs(n - n_best) > 50]
    n_options = n_options[:5]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break

        a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.20, 100.0)
        if alloc > 30:
            a_try = multistage_optimize(a_try, time_budget=alloc, verbose=False,
                                         seed=n_try)

        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        mode = 1 + r
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.2 + 0.1 * r
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        sr = float(np.sum(a_r))
        if sr > 1e-10:
            a_r = a_r / sr

        alloc = min(remaining * 0.20, 80.0)
        if alloc > 30:
            a_r = multistage_optimize(a_r, time_budget=alloc, verbose=False,
                                       seed=5000 + r * 111)
            obj_r = compute_obj_normalized(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final refinement ===
    print(f"\n=== Phase 6: Final Refinement ===")
    remaining = deadline - time.monotonic()

    # Final priority CD
    if remaining > 30:
        alloc = min(remaining * 0.35, 200.0)
        a_final = coordinate_descent_priority(best_seq, time_budget=alloc,
                                               verbose=False, seed=seed + 9999)
        obj_final = compute_obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # Final pairwise
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.30, 100.0)
        a_fp = pairwise_exchange_descent(best_seq, time_budget=alloc,
                                          verbose=False, seed=seed + 8888)
        obj_fp = compute_obj_normalized(a_fp)
        if obj_fp < best_score:
            best_score = obj_fp
            best_seq = a_fp.copy()
            print(f"  Final pairwise: obj={best_score:.8f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_fl = lp_polish(best_seq, time_budget=min(remaining * 0.5, 50.0),
                         verbose=False, seed=7777)
        obj_fl = compute_obj_normalized(a_fl)
        if obj_fl < best_score:
            best_score = obj_fl
            best_seq = a_fl.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final tiny CD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_fc = coordinate_descent_priority(best_seq, time_budget=min(remaining * 0.8, 30.0),
                                            verbose=False, seed=seed + 4444)
        obj_fc = compute_obj_normalized(a_fc)
        if obj_fc < best_score:
            best_score = obj_fc
            best_seq = a_fc.copy()
            print(f"  Final tiny CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
