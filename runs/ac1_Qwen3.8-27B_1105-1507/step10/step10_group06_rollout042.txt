    n = len(a)

        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf

    return 2.0 * n * float(np.max(conv))

    """Build A matrix such that A @ g = (a*g) at the specified indices."""
    n = len(a)
    if indices is None:

    A = np.zeros((len(indices), n))
    for idx, k in enumerate(indices):

        # So A[k, j] = a[k-j] if 0 <= k-j <= n-1
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):

    return A

    prob = cp.Problem(cp.Minimize(M), constraints)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based best-response alternating with cutting planes, warm starts,
running-average tracking, and multiplicative-weights polishing.
Explores multiple sequence lengths for global optimization.
"""

import numpy as np
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def solve_lp_restricted(a_norm, active_indices, M_init=None):
    """
    Solve LP: min M s.t. (a_norm*g)[k] <= M for k in active_indices,
                 sum(g) = 1, g >= 0.
    Returns (g, M) or (None, None).
    """
    n = len(a_norm)
    if len(active_indices) == 0:
        active_indices = list(range(2 * n - 1))
    active_indices = list(set(active_indices))
    num_k = len(active_indices)
    
    # Build constraint matrix: A[k_idx, j] = a_norm[k - j] if valid
    A = np.zeros((num_k, n))
    for idx, k in enumerate(active_indices):
        # (a_norm * g)[k] = sum_j a_norm[k-j] * g[j]
        # Valid j: max(0, k-n+1) <= j <= min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # j ranges from lo to hi, a_norm[k-j] ranges from a_norm[k-hi] to a_norm[k-lo]
            A[idx, lo:hi+1] = a_norm[k-hi:k-lo+1][::-1]
    
    g = cp.Variable(n, nonneg=True)
    M_val = cp.Variable()
    if M_init is not None:
        M_val = cp.Variable(initial_value=M_init)
    
    constraints = [A @ g <= M_val, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M_val), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M_val.value)
        except Exception:
            continue
    return None, None


def lp_best_response(a, max_cuts=30, verbose=False):
    """
    Find the best response g to a using cutting plane LP.
    a is normalized (sum = 1).
    Returns (g, M) or (None, None).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    # Initialize active set with top indices
    top_count = min(100, 2 * n - 1)
    top_idx = np.argsort(conv)[-top_count:]
    active_indices = list(top_idx)
    M_init = max_conv
    
    g = None
    M = M_init
    
    for _ in range(max_cuts):
        g, M = solve_lp_restricted(a, active_indices, M_init)
        if g is None:
            return None, None
        
        # Check all convolution values
        ag = convolve_ab(a, g)
        max_ag = float(np.max(ag))
        
        if max_ag <= M + 1e-9:
            return g, M
        
        # Add most violated constraint
        k_violated = int(np.argmax(ag))
        if k_violated not in active_indices:
            active_indices.append(k_violated)
        M_init = max(M_init, max_ag)
        
        # Also add a few more near-violated constraints for efficiency
        if len(active_indices) < 2 * n - 1:
            near_violated = np.where((ag > M - 0.01 * M) & (~np.isin(np.arange(2*n-1), np.array(active_indices))))[0]
            for k2 in near_violated[:10]:
                if int(k2) not in active_indices:
                    active_indices.append(int(k2))
    
    return g, M


def subgradient_of_max_conv(a):
    """
    Compute a subgradient of f(a) = max_k (a*a)[k] at a (assumed normalized).
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    # Active set: within 1% of max
    threshold = max_val - 0.01 * max_val
    active = np.where(conv >= threshold)[0]
    
    g = np.zeros(n)
    p = 1.0 / len(active)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += p * a[k-hi:k-lo+1][::-1]
    g *= 2.0
    return g


def mirror_descent_polish(a, time_budget, eta_init=0.5, verbose=False):
    """
    Polish using multiplicative weights (mirror descent) with adaptive step size.
    a is assumed normalized (sum = 1).
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init
    stall = 0
    
    while time.monotonic() - start < time_budget:
        g = subgradient_of_max_conv(a)
        a_new = a * np.exp(-eta * np.clip(g, -20, 20))
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-300:
            eta *= 0.5
            continue
        a_new /= s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.05, 2.0)
        else:
            stall += 1
            eta *= 0.7
            if stall > 50:
                break
    
    return best_a


def lp_alternate(a, time_budget, seed=42, verbose=True):
    """
    LP-based alternating minimization with running average tracking.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    avg_a = a.copy()
    avg_count = 1
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    iter_count = 0
    t_blend = 0.5
    stall = 0
    prev_obj = best_obj
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        t0 = time.monotonic()
        g_lp, M_lp = lp_best_response(a, max_cuts=20, verbose=False)
        lp_time = time.monotonic() - t0
        
        if g_lp is None:
            if verbose:
                print(f"    LP failed at iter {iter_count}, using gradient step")
            # Fallback: use subgradient
            g = subgradient_of_max_conv(a)
            g_lp = a * np.exp(-0.1 * np.clip(g, -10, 10))
            g_lp = np.maximum(g_lp, 0)
            s_g = float(np.sum(g_lp))
            if s_g < 1e-10:
                g_lp = a.copy()
            else:
                g_lp /= s_g
            M_lp = compute_obj(a / np.sum(a))
        
        iter_count += 1
        
        # Try blending
        candidates = [t_blend, t_blend * 0.5, t_blend * 1.5]
        candidates = [t for t in candidates if 0.005 <= t <= 1.0]
        if iter_count > 100:
            candidates.append(0.05)
            candidates.append(0.02)
        
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        
        for t in candidates:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try /= s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
        
        # Update running average
        if best_local_obj < best_obj:
            avg_a = (avg_a * avg_count + best_local_a) / (avg_count + 1)
            avg_count += 1
        
        # Check average
        avg_obj = compute_obj(avg_a)
        
        if best_local_obj < best_obj - 1e-12:
            best_obj = min(best_local_obj, avg_obj)
            if avg_obj < best_local_obj:
                best_a = avg_a.copy()
                a = avg_a.copy()
            else:
                best_a = best_local_a.copy()
                a = best_local_a.copy()
            t_blend = max(0.01, min(t_blend * 1.1, 0.9))
            stall = 0
        else:
            stall += 1
            if stall >= 5:
                # Perturb to potentially escape slow convergence
                a = best_a + 0.01 * rng.standard_normal(n) * (best_a + 0.01/n)
                a = np.maximum(a, 0)
                a = a / float(np.sum(a))
                stall = 0
                t_blend = max(0.2, t_blend * 1.3)
            t_blend = max(0.01, t_blend * 0.95)
        
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    iter {iter_count}: obj={best_obj:.10f}, avg_obj={avg_obj:.10f}, "
                  f"M_lp={M_lp:.6f}, lp_time={lp_time:.2f}s, elapsed={elapsed:.0f}s")
    
    # Final check: compare current, best, and average
    final_candidates = [best_a, avg_a, a]
    for fc in final_candidates:
        fc = np.maximum(fc, 0)
        s_fc = float(np.sum(fc))
        if s_fc < 1e-10:
            continue
        fc = fc / s_fc
        obj_fc = compute_obj(fc)
        if obj_fc < best_obj:
            best_obj = obj_fc
            best_a = fc.copy()
    
    if verbose:
        print(f"    LP alt done: {iter_count} iters, best_obj={best_obj:.10f}")
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based best-response alternating with cutting planes, warm-started solves,
running-average tracking, multiplicative-weights polishing, and multi-length exploration.
"""

import numpy as np
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp_restricted(a_norm, active_indices, M_init=None):
    """
    Solve LP: min M s.t. (a_norm*g)[k] <= M for k in active_indices,
                 sum(g) = 1, g >= 0.
    Returns (g, M) or (None, None).
    """
    n = len(a_norm)
    active_indices = list(set(active_indices))
    if not active_indices:
        active_indices = list(range(2 * n - 1))
    num_k = len(active_indices)

    A = np.zeros((num_k, n))
    for idx, k in enumerate(active_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[idx, lo:hi + 1] = a_norm[k - hi:k - lo + 1][::-1]

    g = cp.Variable(n, nonneg=True)
    if M_init is not None:
        M_val = cp.Variable(initial_value=M_init)
    else:
        M_val = cp.Variable()

    constraints = [A @ g <= M_val, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M_val), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M_val.value)
        except Exception:
            continue
    return None, None


def lp_best_response(a, max_cuts=25):
    """
    Find the best response g to a (normalized, sum=1) using cutting plane LP.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    top_count = min(80, 2 * n - 1)
    top_idx = np.argsort(conv)[-top_count:]
    active_indices = list(top_idx)
    M_init = max_conv

    g = None
    M = M_init

    for _ in range(max_cuts):
        g, M = solve_lp_restricted(a, active_indices, M_init)
        if g is None:
            return None, None

        ag = convolve_ab(a, g)
        max_ag = float(np.max(ag))

        if max_ag <= M + 1e-9:
            return g, M

        k_violated = int(np.argmax(ag))
        if k_violated not in active_indices:
            active_indices.append(k_violated)
        M_init = max(M_init, max_ag)

        if len(active_indices) < 2 * n - 1:
            in_active = np.zeros(2 * n - 1, dtype=bool)
            for k in active_indices:
                in_active[k] = True
            near = np.where((ag > M - 0.02 * max_ag) & (~in_active))[0]
            for k2 in near[:5]:
                if int(k2) not in active_indices:
                    active_indices.append(int(k2))

    return g, M


def subgradient_of_max_conv(a):
    """
    Compute a subgradient of f(a) = max_k (a*a)[k] at normalized a.
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    threshold = max_val - 0.005 * max_val
    active = np.where(conv >= threshold)[0]

    g = np.zeros(n)
    p = 1.0 / len(active)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += p * a[k - hi:k - lo + 1][::-1]
    g *= 2.0
    return g


def mirror_descent_polish(a, time_budget, eta_init=0.3, verbose=False):
    """
    Polish using multiplicative weights (mirror descent) with adaptive step size.
    a is assumed normalized (sum = 1).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 1:
            break
        g = subgradient_of_max_conv(a)
        g_clipped = np.clip(g, -30, 30)
        a_new = a * np.exp(-eta * g_clipped)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-300:
            eta *= 0.5
            continue
        a_new /= s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.03, 3.0)
        else:
            stall += 1
            eta *= 0.8
            if stall > 80:
                break
        it += 1
        if verbose and it % 200 == 0:
            print(f"      MD iter {it}: obj={best_obj:.10f}, eta={eta:.4f}")

    return best_a

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def smoothed_gradient(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of (a*a) using FFT.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = np.fft.fft(p, N)
    A = np.fft.fft(a, N)
    g_full = np.fft.ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def fista_polish(a, time_budget, tau_init=40.0, tau_max=3000.0, verbose=False):
    """

    if s < 1e-10:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    z = a.copy()
    t_k = 1.0
    tau = tau_init
    base_lr = 1.5 / np.sqrt(n)
    lr = base_lr
    it = 0

    while time.monotonic() - start < time_budget:
        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-15:
            adaptive_lr = min(0.8 / (g_norm + 1e-10), 3.0 * base_lr)
            lr = 0.9 * lr + 0.1 * max(adaptive_lr, 0.5 * base_lr)

        y_new = y - lr * g
        y_new = np.maximum(y_new, 0)
        s_y = float(np.sum(y_new))
        if s_y < 1e-10:
            y_new = a.copy()
        else:
            y_new /= s_y

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = y_new + ((t_k - 1.0) / t_new) * (y_new - a)
        z = np.maximum(z, 0)
        s_z = float(np.sum(z))
        if s_z > 1e-10:
            z /= s_z

        a = y_new
        t_k = t_new
        tau = min(tau * 1.015, tau_max)

        obj_new = compute_obj(a)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()

        it += 1

            y = best_a.copy()
            z = best_a.copy()

    # === Phase 1: LP Alternating (main workhorse) ===
