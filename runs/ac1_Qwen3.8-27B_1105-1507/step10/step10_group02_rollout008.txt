import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as sp_minimize
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None

            a_zero[j] = 0.0
            conv_base = compute_conv(a_zero)

            # Linear terms: for position k, linear contribution from x_i is a[k-i] (if valid)
            # and from x_j is a[k-j] (if valid)
            # Quadratic: x_i * x_j at position i+j

            def pair_obj(xi, xj):
                conv_try = conv_base.copy()
                # Add x_i * a[k-i] for valid k
                for k in range(i, min(2*n-1, i+n)):
                    idx = k - i
                    if idx != j:
                        conv_try[k] += xi * a[idx]
                    else:
                        conv_try[k] += xi * xj  # a[i]*a[j] term handled by quadratic
                # Add x_j * a[k-j] for valid k
                for k in range(j, min(2*n-1, j+n)):
                    idx = k - j
                    if idx != i:
                        conv_try[k] += xj * a[idx]
                    # else already handled
                # Add x_i * x_j at position i+j
                if i + j < 2*n-1:
                    conv_try[i+j] += xi * xj
                # Also need x_i*a[i] at position 2i and x_j*a[j] at position 2j
                # Wait, I zeroed a[i] and a[j], so conv_base doesn't have those terms.
                # Let me redo: conv_base has a[i]=0, a[j]=0
                # conv[k] with a[i]=xi, a[j]=xj:
                # = conv_base[k] + xi*(a[k-i] if 0<=k-i<n and k-i!=j else 0)
                #                 + xj*(a[k-j] if 0<=k-j<n and k-j!=i else 0)
                #                 + xi*xj * (1 if k==i+j else 0)
                #                 + xi^2 * (1 if k==2*i else 0)  -- wait no
                # Actually conv[k] = sum_p a[p]*a[k-p]
                # Terms involving a[i]=xi: p=i gives a[i]*a[k-i] = xi*a[k-i]
                #                       p=k-i gives a[k-i]*a[i] = a[k-i]*xi (same term, counted once in sum)
                # Wait, convolution: conv[k] = sum_{p} a[p] * a[k-p] where both indices in [0,n-1]
                # If p=i: a[i]*a[k-i] = xi * a[k-i] (valid if 0<=k-i<=n-1)
                # If p=k-i: a[k-i]*a[i] = a[k-i] * xi (valid if 0<=k-i<=n-1 and k-i != i, i.e., k != 2i)
                # So if k != 2i: contribution is 2*xi*a[k-i]
                # If k == 2i: contribution is xi^2
                # Similarly for j.
                # Cross term: p=i, k-p=j → k=i+j: xi*xj; p=j, k-p=i → k=i+j: xj*xi
                # So at k=i+j: 2*xi*xj (if i!=j)
                # But wait, if i=j, then it's different. We assume i!=j.

                # Let me just compute it directly for clarity (slower but correct)
                # Actually for efficiency, let me precompute the linear coefficient array

                S_new = S_other + xi + xj
                if S_new < 1e-10:
                    return 1e30
                max_conv = float(np.max(conv_try))
                return 2.0 * n * max_conv / (S_new * S_new)

            # This is too slow with the inner loop. Let me precompute.
            # Linear contribution from xi: L_i[k] = 2*a[k-i] if 0<=k-i<n and k-i!=j,
            #                                   = 0 if k-i==j (handled by cross/quad)
            #                                   = a[i] if k==2i → but a[i]=vi, and we want xi^2
            # Actually let me just use a different approach: precompute the full conv
            # as a function of (xi, xj) using the linearity.

            # conv[k] = conv_base[k] + xi * Li[k] + xj * Lj[k] + xi*xj * C[k] + xi^2 * Qi[k] + xj^2 * Qj[k]
            # where Li[k] = a[k-i] if 0<=k-i<n and k-i!=j
            #              + a[k-i] if 0<=k-i<n and k-i != i (i.e., k != 2i) → total 2*a[k-i] if k!=2i
            # This is getting complicated. Let me just use a faster computation.

            # Alternative: compute conv directly for the small number of candidate (xi, xj) pairs
            # using the update formula

            # For efficiency, let me precompute:
            # contrib_i[k] = d conv[k] / d a[i] when a[i] goes from 0 to xi
            # = sum of a[k-i] for valid indices, handling the self-term
            # For k in range:
            #   if 0 <= k-i < n and k-i != i and k-i != j: contrib = 2*a[k-i] (but a[j]=0 in base)
            #   if k-i == i (k=2i): contrib = 2*xi (but this is quadratic, not linear)

            # OK this is getting too complex for the inner loop. Let me use a simpler approach:
            # Just evaluate the objective at a grid of (xi, xj) values using FFT each time.
            # For n=800, each eval is ~800*log(800) ≈ 8000 ops. Grid of 10x10 = 100 evals = 800K ops per pair.
            # With 200 pairs = 160M ops per sweep. Too slow.

            # Better: use the incremental update.
            # Start from conv (with current a). To evaluate at (xi, xj):
            # delta_i = xi - vi, delta_j = xj - vj
            # conv_new[k] = conv[k] + delta_i * (a[k-i] if valid and k-i != j)
            #                            + delta_j * (a[k-j] if valid and k-j != i)
            #                            + delta_i * a_pair[k] ... this is getting messy.

            # Let me just do a simpler version: evaluate at a small grid using the full conv.
            # Limit to 5x5 grid = 25 evals per pair.
            scale_i = max(vi, 0.1 * S / n, 1e-6)
            scale_j = max(vj, 0.1 * S / n, 1e-6)

            best_xi, best_xj = vi, vj
            best_pair_obj = 2.0 * n * float(np.max(conv)) / (S * S)

            # 5x5 grid
            xs_i = np.clip(vi + scale_i * np.array([-1, -0.5, 0, 0.5, 1.0]), 0, S)
            xs_j = np.clip(vj + scale_j * np.array([-1, -0.5, 0, 0.5, 1.0]), 0, S)

            for xi_c in xs_i:
                for xj_c in xs_j:
                    if abs(xi_c - vi) < 1e-12 and abs(xj_c - vj) < 1e-12:
                        continue
                    a_try = a.copy()
                    a_try[i] = xi_c
                    a_try[j] = xj_c
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_norm_try = a_try / s_try
                    conv_try = compute_conv(a_norm_try)
                    obj_try = 2.0 * n * float(np.max(conv_try))
                    if obj_try < best_pair_obj:
                        best_pair_obj = obj_try
                        best_xi, best_xj = xi_c, xj_c

            # Update if improved
            if best_xi != vi or best_xj != vj:
                a[i] = best_xi
                a[j] = best_xj
                # Recompute conv
                S = float(np.sum(a))
                conv = compute_conv(a / S)
                obj_now = 2.0 * n * float(np.max(conv))
                if obj_now < best_obj:
                    best_obj = obj_now
                    best_a = a.copy()

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    PairCD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def single_coordinate_descent(a, time_budget=200.0, verbose=False, seed=42):
    """
    Exact single-coordinate descent with golden section search.
    Optimized version of the previous code's coordinate_descent.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 5 == 1:
            conv = compute_conv(a)
            obj = 2.0 * n * float(np.max(conv))
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks]

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Quick evaluation at a few points
            scale = max(v, S / n, 1e-6)
            candidates = [0.0, v * 0.5, v, v * 1.5, v * 2.0, S / (2 * n)]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            def f_1d(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            best_x = v
            best_f = f_1d(v)
            for x_c in candidates:
                fx = f_1d(x_c)
                if fx < best_f:
                    best_f = fx
                    best_x = x_c

            # Golden section refinement
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(30):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

            obj_try = 2.0 * n * float(np.max(conv)) / (S * S)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a.copy()

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def smoothed_gradient_descent(a, time_budget=100.0, tau_init=30.0, tau_end=300.0, verbose=False, seed=42):
    """Nesterov-accelerated smoothed gradient descent for global exploration."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    from scipy.fft import fft, ifft
    N = 1
    while N < 3 * n:
        N <<= 1

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)

    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = 1.0
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        # Account for normalization (y is already normalized)
        # The gradient of 2n*max(conv)/sum^2 w.r.t. y (with sum=1 constraint)
        # is approximately 2n * (g - mean(g)) for the smooth surrogate
        g = g - np.mean(g)  # project to tangent space of sum=1

        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-15:
            lr = min(max(lr * 0.999 + 0.0005 / (g_norm + 1e-10), 0.1 * 2.0 / np.sqrt(n)),
                     2.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 2000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 10, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Two-bump with random parameters
        c1 = rng.uniform(0.1, 0.4) * n
        c2 = rng.uniform(0.6, 0.9) * n
        w1 = rng.uniform(0.05, 0.15) * n
        w2 = rng.uniform(0.05, 0.15) * n
        x = np.arange(n)
        a = np.exp(-(x - c1)**2 / (2*w1**2)) + np.exp(-(x - c2)**2 / (2*w2**2))
        a = a / a.sum()
    elif mode == 7:
        # Random smooth curve
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.sin(2*np.pi*rng.uniform(1,5)*x + rng.uniform(0, 2*np.pi))
        a += 0.3 * np.sin(2*np.pi*rng.uniform(2,8)*x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining LP line search, pair CD, single CD, smoothed GD,
    and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_default = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj_normalized(parent)
                best_seq = parent.copy()
                n_default = parent.size
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    # === Phase 1: LP with line search (strong global method) ===
    print("\n=== Phase 1: LP with Line Search ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.3, 400.0)
    if p1_budget > 60:
        a_lp = lp_with_linesearch(best_seq, verbose=True, max_iters=15, seed=seed)
        obj_lp = compute_obj_normalized(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP-LS: obj={best_score:.8f} **")
        else:
            print(f"  LP-LS: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Coordinate descent refinement ===
    print("\n=== Phase 2: Single CD Refinement ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 30:
        a_cd = single_coordinate_descent(best_seq, time_budget=p2_budget,
                                         verbose=True, seed=seed)
        obj_cd = compute_obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 3: LP + CD alternating (short bursts) ===
    print("\n=== Phase 3: LP+CD Alternating ===")
    remaining = deadline - time.monotonic()
    for rnd in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Short LP
        lp_budget = min(30.0, remaining * 0.3)
        if lp_budget > 10:
            a_lp = lp_with_linesearch(best_seq, verbose=False, max_iters=3,
                                      seed=100 + rnd)
            obj_lp = compute_obj_normalized(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Alt-LP {rnd}: obj={best_score:.8f} **")

        # Short CD
        cd_budget = min(40.0, remaining * 0.3)
        if cd_budget > 10:
            a_cd = single_coordinate_descent(best_seq, time_budget=cd_budget,
                                             verbose=False, seed=200 + rnd)
            obj_cd = compute_obj_normalized(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Alt-CD {rnd}: obj={best_score:.8f} **")

    # === Phase 4: Pair CD for finer local search ===
    print("\n=== Phase 4: Pair Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 150.0)
    if p4_budget > 30:
        a_pair = pair_coordinate_descent(best_seq, time_budget=p4_budget,
                                         verbose=True, seed=seed + 50)
        obj_pair = compute_obj_normalized(a_pair)
        if obj_pair < best_score:
            best_score = obj_pair
            best_seq = a_pair.copy()
            print(f"  PairCD: obj={best_score:.8f} **")
        else:
            print(f"  PairCD: obj={obj_pair:.8f}")

    # === Phase 5: Multi-n exploration ===
    print("\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 200),
        max(400, n_best - 100),
        n_best + 100,
        n_best + 200,
        n_best + 400,
        600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) > 30]
    n_options = n_options[:6]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick CD on new length
        alloc = min(remaining * 0.2, 50.0)
        if alloc > 15:
            a_try = single_coordinate_descent(a_try, time_budget=alloc,
                                              verbose=False, seed=n_try)

        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts with blending ===
    print("\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = 1 + r
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 60.0)
        if alloc > 15:
            a_r = single_coordinate_descent(a_r, time_budget=alloc,
                                            verbose=False, seed=5000 + r * 100)
            obj_r = compute_obj_normalized(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 7: Smoothed GD for global escape ===
    print("\n=== Phase 7: Smoothed GD Escape ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        # Multiple perturbations
        for pert_idx in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            sigma = 0.02 + 0.01 * pert_idx
            pert = rng.normal(0, sigma, len(best_seq))
            a_pert = best_seq * np.exp(pert)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()

            alloc = min(remaining * 0.25, 60.0)
            a_gd = smoothed_gradient_descent(a_pert, time_budget=alloc,
                                             tau_init=30.0, tau_end=300.0,
                                             verbose=False, seed=9000 + pert_idx)
            obj_gd = compute_obj_normalized(a_gd)
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
                print(f"  GD escape {pert_idx}: obj={best_score:.8f} **")
            else:
                print(f"  GD escape {pert_idx}: obj={obj_gd:.8f}")

    # === Phase 8: Final polish (CD + LP) ===
    print("\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc_cd = min(remaining * 0.4, 120.0)
        if alloc_cd > 15:
            a_final = single_coordinate_descent(best_seq, time_budget=alloc_cd,
                                                verbose=False, seed=seed + 9999)
            obj_final = compute_obj_normalized(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_lp_final = lp_with_linesearch(best_seq, verbose=False, max_iters=5,
                                        seed=7777)
        obj_lp_final = compute_obj_normalized(a_lp_final)
        if obj_lp_final < best_score:
            best_score = obj_lp_final
            best_seq = a_lp_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final CD if time
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_final2 = single_coordinate_descent(best_seq, time_budget=min(remaining * 0.6, 60.0),
                                             verbose=False, seed=seed + 8888)
        obj_final2 = compute_obj_normalized(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
