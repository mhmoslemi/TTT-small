import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    return 2.0 * n * float(np.max(compute_conv(a_n)))


def smooth_grad(a, tau):
    """Compute gradient of smoothed max of conv(a,a) using FFT. a should have sum=1."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    # Gradient: g[j] = tau * 2 * sum_k p[k] * a[k-j] (with a[x]=0 for x outside [0,n-1])
    # = tau * 2 * cross_correlation(a, p)[j]
    N = 1
    while N < 4 * n:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    # cross_correlation(a, p)[j] = sum_k a[k]*p[k+j] = IFFT(FFT(a)*conj(FFT(p)))[j]
    corr = ifft(A * np.conj(P)).real[:n]
    grad = tau * 2.0 * corr
    return grad, conv, p

def kkt_solve(a_init, active_set, max_newton=50, verbose=False):
    """
    Solve KKT conditions for the minimax problem with given active set.
    min max_k conv(a,a)[k] s.t. sum(a)=1, a>=0
    Active set K: indices where conv(a,a)[k] = max.
    """
    n = len(a_init)
    m = len(active_set)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a = a / s

    # Support: all indices with a[j] > threshold
    support_thresh = 1e-12
    support = [j for j in range(n) if a[j] > support_thresh]
    support_idx = np.array(support)
    n_sup = len(support_idx)

    if n_sup < 2:
        return None

    # Initialize multipliers
    lam = np.ones(m) / m
    mu = 0.0
    t_val = float(np.max(compute_conv(a)))

    def compute_g(a_curr, k):
        """Compute gradient of conv(a,a)[k] w.r.t. a."""
        n_c = len(a_curr)
        g = np.zeros(n_c)
        for j in range(n_c):
            idx = k - j
            if 0 <= idx < n_c:
                g[j] += 2.0 * a_curr[idx]
        return g

    def compute_H_row(a_curr, k, j):
        """Compute Hessian row: d2(conv(a,a)[k])/da[j]da[l] = 2*delta(j+l,k)"""
        # H[j,l] = 2 if j+l == k and j,l in [0,n-1], else 0
        l = k - j
        if 0 <= l < n:
            return l, 2.0
        return -1, 0.0

    # Newton iteration
    for ni in range(max_newton):
        # Compute residuals
        conv = compute_conv(a)
        res_conv = np.array([conv[k] - t_val for k in active_set])  # m
        res_sum = float(np.sum(a)) - 1.0  # 1
        # Stationarity: sum_r lam[r] * g_{k_r}[j] + mu = 0 for j in support
        g_stack = np.array([compute_g(a, k) for k in active_set])  # m x n
        stationarity_lhs = g_stack.T @ lam  # n
        res_station = stationarity_lhs[support_idx] + mu  # n_sup
        res_lam = float(np.sum(lam)) - 1.0  # 1

        # Check convergence
        res_norm = max(np.max(np.abs(res_conv)), abs(res_sum),
                       np.max(np.abs(res_station)), abs(res_lam))
        if res_norm < 1e-12:
            break

        # Build Jacobian and solve
        # Variables: a (n), lam (m), mu (1), t (1)
        # Total: n + m + 2
        N_vars = n + m + 2
        N_eqs = m + 1 + n_sup + 1

        if N_vars != N_eqs:
            # In case of reduced support, we need to adjust
            # For now, use full n variables
            N_eqs = N_vars
            # Recompute stationarity for all n
            res_station = stationarity_lhs + mu  # n

        # Jacobian is (N_eqs x N_vars)
        # We'll build it in blocks for efficiency

        # Block (1,1): d(res_conv)/d(a) = G (m x n)
        # G[r, j] = g_{k_r}[j]
        J_11 = g_stack  # m x n

        # Block (1, lam, mu, t): d(res_conv)/d(lam)=0, d/dmu=0, d/dt = -1
        # So last column of first m rows is -1

        # Block (2,1): d(res_sum)/d(a) = ones (1 x n)
        # Block (2, rest): 0

        # Block (3,1): d(res_station)/d(a) = sum_r lam[r] * H_{k_r} (n x n)
        # H_{k_r}[j,l] = 2 if j+l==k_r else 0
        # J_31[j,l] = 2 * sum_r: j+l=k_r lam[r]
        # Build efficiently: for each (j,l) with j+l in active_set, J_31[j,l] = 2*lam[idx]
        J_31 = np.zeros((n, n))
        for r, k in enumerate(active_set):
            if 0 <= k < 2 * n - 1:
                for j in range(n):
                    l = k - j
                    if 0 <= l < n:
                        J_31[j, l] += 2.0 * lam[r]

        # Block (3, lam): d(res_station)/d(lam) = g_stack[support_idx, :] (n x m)
        J_32 = g_stack[support_idx]  # n x m (if full support)
        if N_eqs == N_vars:
            J_32 = g_stack  # n x m

        # Block (3, mu): ones (n x 1)
        # Block (3, t): 0

        # Block (4, lam): ones (1 x m)
        # Block (4, rest): 0

        # Assemble full Jacobian
        J = np.zeros((N_eqs, N_vars))
        # Rows 0..m-1: res_conv
        J[:m, :n] = J_11
        J[:m, n + m + 1] = -1.0  # d/dt
        # Row m: res_sum
        J[m, :n] = 1.0
        # Rows m+1..m+n: res_station
        J[m + 1:m + 1 + n, :n] = J_31
        J[m + 1:m + 1 + n, n:n + m] = J_32
        J[m + 1:m + 1 + n, n + m] = 1.0  # d/dmu
        # Row m+n+1: res_lam
        J[m + n + 1, n:n + m] = 1.0

        # Residual vector
        res = np.zeros(N_eqs)
        res[:m] = res_conv
        res[m] = res_sum
        if N_eqs == N_vars:
            res[m + 1:m + 1 + n] = res_station  # full n
        else:
            res[m + 1:m + 1 + n_sup] = res_station
        res[-1] = res_lam

        # Solve J * dx = -res
        try:
            dx = np.linalg.solve(J, -res)
        except np.linalg.LinAlgError:
            try:
                dx = np.linalg.lstsq(J, -res, rcond=None)[0]
            except Exception:
                return a.copy()

        da = dx[:n]
        dlam = dx[n:n + m]
        dmu = dx[n + m]
        dt = dx[n + m + 1]

        # Damped Newton step with line search
        alpha = 1.0
        for ls in range(20):
            a_trial = a + alpha * da
            lam_trial = lam + alpha * dlam
            mu_trial = mu + alpha * dmu
            t_trial = t_val + alpha * dt

            if np.any(a_trial < -1e-10) or np.any(lam_trial < -1e-10):
                alpha *= 0.5
                continue

            a_trial = np.maximum(a_trial, 0)
            lam_trial = np.maximum(lam_trial, 0)

            conv_trial = compute_conv(a_trial)
            res_c = np.array([conv_trial[k] - t_trial for k in active_set])
            s_sum = float(np.sum(a_trial)) - 1.0
            g_t = np.array([compute_g(a_trial, k) for k in active_set])
            stat = g_t @ lam_trial + mu_trial
            res_l = float(np.sum(lam_trial)) - 1.0
            trial_norm = max(np.max(np.abs(res_c)), abs(s_sum),
                             np.max(np.abs(stat)), abs(res_l))

            if trial_norm < res_norm * (1 - 1e-4 * alpha):
                break
            alpha *= 0.5

        a = np.maximum(a + alpha * da, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            return None
        a = a / s
        lam = np.maximum(lam + alpha * dlam, 0)
        lam_sum = float(np.sum(lam))
        if lam_sum > 0:
            lam = lam / lam_sum
        mu = mu + alpha * dmu
        t_val = t_val + alpha * dt

        if verbose:
            conv_check = compute_conv(a)
            print(f"    KKT it={ni}: res={res_norm:.2e}, alpha={alpha:.4f}, "
                  f"max_conv={np.max(conv_check):.10f}")

    # Final check
    conv_final = compute_conv(a)
    # Verify all non-active conv values are <= t
    for k in range(len(conv_final)):
        if k not in active_set and conv_final[k] > t_val + 1e-10:
            # Active set constraint violated
            return None

    obj = 2.0 * n * float(np.max(conv_final))
    return a


def kkt_multi_active(a_init, n_candidates=8, max_per_set=3, verbose=False):
    """Try multiple active sets and return the best solution."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    # Get top candidate indices
    top_idx = np.argsort(-conv)[:n_candidates]
    top_idx = sorted(set(int(k) for k in top_idx))

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))

    # Try active sets of size 1, 2, 3
    from itertools import combinations
    for size in [1, 2, 3]:
        if size > len(top_idx):
            break
        for combo in combinations(top_idx, size):
            result = kkt_solve(a, list(combo), max_newton=30, verbose=False)
            if result is not None:
                obj = 2.0 * n * float(np.max(compute_conv(result)))
                if obj < best_obj:
                    best_obj = obj
                    best_a = result.copy()
                    if verbose:
                        print(f"    Active set {list(combo)}: obj={obj:.10f} **")

    return best_a


def coordinate_descent_fast(a, time_budget=100.0, verbose=False, seed=42):
    """
    Fast coordinate descent using the exact 1D subproblem formulation.
    Optimized for speed with vectorized operations.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)

            # Linear part: conv[k] += 2*a[i-delta_idx] * (x - v) for k != 2i
            # conv[k] = conv[k] + 2*a[k-i]*(x-v) for k in [i, i+n-1], k != 2i
            # Quadratic part: conv[2i] += x^2 - v^2

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a[ks_lin - i]
            c_k_lin = conv[ks_lin]

            c_2i = float(conv[2 * i]) if 0 <= 2 * i <= 2 * n - 2 else 0.0

            # Quick candidate evaluation
            scale = max(v, S / n, 1e-6)
            candidates = np.array([0.0, v * 0.5, v, v * 1.5, v * 2.0,
                                   S_other * 0.3, S_other * 0.7, S_other])
            candidates = np.clip(candidates, 0, None)

            best_x = v
            best_f = np.inf
            for x in candidates:
                x = float(x)
                if len(ks_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) ** 2
                if denom < 1e-30:
                    continue
                f = 2.0 * n * max_val / denom
                if f < best_f:
                    best_f = f
                    best_x = x

            # Golden section refinement
            x_lo = max(0.0, best_x - scale * 0.5)
            x_hi = best_x + scale

            def f_1d(x):
                if len(ks_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) ** 2
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(30):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14 * max(1.0, v):
                continue

            delta = x_opt - v
            a[i] = x_opt

            # Update conv
            if 0 <= 2 * i <= 2 * n - 2:
                conv[2 * i] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        c1, c2 = n // 4, 3 * n // 4
        w = n // 10
        x = np.arange(n)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like distribution
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x ** 2)
        a = a / a.sum()
    elif mode == 4:
        # Polynomial
        x = np.linspace(0, 1, n)
        a = (1 - x) ** rng.uniform(0.5, 3.0) + (x) ** rng.uniform(0.5, 3.0)
        a = a / a.sum()
    elif mode == 5:
        # Periodic structure
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * rng.integers(2, 6) * x) + \
            0.2 * np.cos(2 * np.pi * rng.integers(6, 12) * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Edge concentrated
        x = np.linspace(0, 1, n)
        a = 1.0 / (x + 0.1) + 1.0 / (1 - x + 0.1)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining Adam optimization, KKT active-set solving,
    coordinate descent, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = 2.0 * n_default * float(np.max(compute_conv(best_seq)))
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    if best_seq is None:
        best_seq = np.ones(500) / 500
        best_score = 2.0

    n_best = len(best_seq)

    # === Phase 1: Adam optimization with smoothed objective ===
    print(f"\n=== Phase 1: Adam SM Optimization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 500.0)
    if p1_budget > 60:
        a_adam = adam_optimize(best_seq, time_budget=p1_budget * 0.7,
                               verbose=True, seed=seed)
        obj_adam = 2.0 * a_adam.size * float(np.max(compute_conv(a_adam / a_adam.sum())))
        if obj_adam < best_score:
            best_score = obj_adam
            best_seq = a_adam.copy()
            print(f"  Adam: obj={best_score:.8f} **")
        else:
            print(f"  Adam: obj={obj_adam:.8f} (no improvement)")

        # Second Adam pass with different params
        remaining = deadline - time.monotonic()
        if remaining > 60:
            a_adam2 = adam_optimize(best_seq, time_budget=min(remaining * 0.3, 150.0),
                                    verbose=False, seed=seed + 100)
            obj_adam2 = 2.0 * a_adam2.size * float(np.max(compute_conv(a_adam2 / a_adam2.sum())))
            if obj_adam2 < best_score:
                best_score = obj_adam2
                best_seq = a_adam2.copy()
                print(f"  Adam2: obj={best_score:.8f} **")

    # === Phase 2: KKT Active Set Solving ===
    print(f"\n=== Phase 2: KKT Active Set ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_kkt = kkt_multi_active(best_seq, n_candidates=10, verbose=True)
        obj_kkt = 2.0 * a_kkt.size * float(np.max(compute_conv(a_kkt / a_kkt.sum())))
        if obj_kkt < best_score:
            best_score = obj_kkt
            best_seq = a_kkt.copy()
            print(f"  KKT: obj={best_score:.8f} **")
        else:
            print(f"  KKT: obj={obj_kkt:.8f}")

        # Try KKT again from slightly perturbed point
        remaining = deadline - time.monotonic()
        if remaining > 30:
            pert = rng.normal(0, 0.005, len(best_seq))
            a_pert = best_seq * np.exp(pert)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            a_kkt2 = kkt_multi_active(a_pert, n_candidates=10, verbose=False)
            obj_kkt2 = 2.0 * a_kkt2.size * float(np.max(compute_conv(a_kkt2 / a_kkt2.sum())))
            if obj_kkt2 < best_score:
                best_score = obj_kkt2
                best_seq = a_kkt2.copy()
                print(f"  KKT2: obj={best_score:.8f} **")

    # === Phase 3: Coordinate Descent Refinement ===
    print(f"\n=== Phase 3: Coordinate Descent ===")

    if p3_budget > 30:

        if obj_cd < best_score:

            print(f"  CD: obj={best_score:.8f} **")

            if obj_kkt3 < best_score:
                best_score = obj_kkt3
                best_seq = a_kkt3.copy()
                print(f"  KKT3: obj={best_score:.8f} **")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")

    n_options = [n for n in n_options if 300 <= n <= 2500 and abs(n - n_best) > 30]

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_try = kkt_multi_active(a_try, n_candidates=6, verbose=False)

        obj_try = 2.0 * a_try.size * float(np.max(compute_conv(a_try / a_try.sum())))
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()

        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 1e-10)
        a_r = a_r / a_r.sum()

        obj_r = 2.0 * a_r.size * float(np.max(compute_conv(a_r / a_r.sum())))
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()

        else:

    # === Phase 6: Final Refinement Loop ===

            if obj_curr < best_score:

        # KKT

            if obj_curr < best_score:

    # Final evaluation

    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
