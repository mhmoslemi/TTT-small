from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):

    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

    if s < 0.01:

    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    Linearized approach: find direction d such that conv(a, a + d) is closer to flat.
    Target: all c_k should be at most target_ratio * current_max + (1-target_ratio) * flat_level.
    Solve linear system with constraints d >= -a, sum(d) = 0.
    """
    n = len(a)
    c = compute_conv(a)
    max_c = float(np.max(c))
    flat_level = 1.0 / (2 * n - 1)
    # Target: cap at a level between flat and current max
    target_level = target_ratio * max_c + (1 - target_ratio) * flat_level

    # We want: conv(a, a + t*d)_k <= target_level for all k
    # Linearize: c_k + t * conv(a, d)_k <= target_level
    # So: conv(a, d)_k <= (target_level - c_k) / t for all k
    # We want the largest t, so minimize max_k (target_level - c_k) / ... 
    # Actually, let's just find d that minimizes max_k (c_k + conv(a,d)_k)
    # subject to d >= -a, sum(d) = 0
    # This is the same LP! But with a different framing.
    # Instead, let's use a simpler approach:
    # Solve: min max_k (conv(a,d)_k) s.t. d >= -a, sum(d) = 0
    # This finds the direction that most reduces the peak.
    
    A = make_conv_matrix(a)
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    # d >= -a means d + a >= 0
    constraints = [A @ d_var <= t_var, d_var >= -a, cp.sum(d_var) == 0]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    solvers = [cp.MOSEK, cp.CLARABEL]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if d_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                d_val = d_var.value
                a_new = a + step_size * d_val
                a_new = np.maximum(a_new, 0)
                s = float(np.sum(a_new))
                if s > 1e-10:
                    a_new = a_new / s
                    return a_new
        except Exception:
            continue
    return a


def spectral_perturb(a, rng, strength=0.3):
    """Perturb in frequency domain with random phases."""
    n = len(a)
    size = 1
    while size < 4 * n:
        size <<= 1
    A = np.fft.fft(a, size)
    # Random phase perturbation, stronger at high frequencies
    freqs = np.fft.fftfreq(size)
    high_freq_mask = np.abs(freqs) > 1.0 / (2 * n)
    phases = rng.uniform(-np.pi, np.pi, size) * strength * high_freq_mask
    A_pert = A * np.exp(1j * phases)
    a_new = np.fft.ifft(A_pert).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_fixed_point(a, max_iters=200, time_budget=60.0, verbose=False, seed=42,
                   stall_patience=30, perturb_every=10):
    """
    LP fixed-point iteration with spectral perturbation to escape local minima.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    last_perturb_obj = best_obj

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        current_max = float(np.max(compute_conv(a)))
        # Check fixed point: M_lp should be close to current_max
        if M_lp > current_max * 1.001:
            # LP found improvement
            stall = 0
        else:
            stall += 1

        # Blend
        t = 0.5
        a_new = (1 - t) * a + t * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0

        a = a_new

        if stall >= stall_patience:
            # Apply perturbation
            a = spectral_perturb(best_a, rng, strength=0.2 + 0.1 * min(stall / 20.0, 1.0))
            a = a / a.sum()
            stall = 0
            if verbose:
                print(f"      perturbed at it={it}, obj={obj_new:.10f}")

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"      it={it}: obj={obj_new:.10f}, best={best_obj:.10f}, t={elapsed:.0f}s")

    return best_a


def smooth_gd(a, time_budget, tau_levels=None, verbose=False):
    """Smoothed gradient descent with log-sum-exp surrogate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 20, 100, 500, 2000, 8000]

    beta = 0.9
    for tau in tau_levels:
        lr = 1.5 / np.sqrt(n)
        v = np.zeros(n)
        it_level = 0
        while time.monotonic() - start < time_budget and it_level < 3000:
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                break
            y = y / sy

            conv = compute_conv(y)
            exact_max = float(np.max(conv))
            exp_vals = np.exp(tau * (conv - exact_max))
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp

            N = 1
            while N < 4 * n:
                N <<= 1
            P = fft(p, N)
            A = fft(y, N)
            cross = ifft(P * np.conj(A)).real
            g = 2.0 * cross[:n]

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            it_level += 1
            lr = max(1e-7, lr * 0.99999)

    return best_a


def make_init(n, mode, rng=None):
    """Create diverse initial sequences of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.0001))
        return a / a.sum()
    elif mode == 3:
        # Super-Gaussian (raised cosine squared)
        x = np.linspace(-1, 1, n)
        a = np.maximum(1 - x ** 2, 0.001) ** 2
        return a / a.sum()
    elif mode == 4:
        # Raised cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        return a / a.sum()
    elif mode == 5:
        # Exponential
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 4))
        return a / a.sum()
    elif mode == 6:
        # Flat top with smooth edges
        x = np.linspace(0, 1, n)
        a = np.where((x > 0.15) & (x < 0.85), 1.0,
                     np.maximum(0.5 - np.abs(x - 0.5), 0.01))
        # Smooth the transitions
        kernel = np.array([0.25, 0.5, 0.25])
        a = np.convolve(a, kernel, mode='same')
        return a / a.sum()
    elif mode == 7:
        # Bimodal
        c1, c2 = n // 3, 2 * n // 3
        w = max(2, n // 10)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 8:
        # Gaussian
        a = np.exp(-0.5 * ((np.arange(n) - n / 2) / (n / 6)) ** 2)
        return a / a.sum()
    else:
        a = rng.uniform(0.1, 1, n)
        return a / a.sum()


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def combined_refine(a, time_budget, verbose=False, seed=42):
    """
    Combined refinement: alternate between LP fixed-point and smooth GD,
    with spectral perturbations to escape local minima.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Alternate between methods
    phase = 0
    no_improve = 0
    max_no_improve = 8

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Decide method based on phase
        if phase % 3 == 0:
            # LP fixed-point
            alloc = min(remaining * 0.4, 40.0)
            a_new = lp_fixed_point(best_a, time_budget=alloc, verbose=verbose,
                                   seed=seed + phase, stall_patience=20)
        elif phase % 3 == 1:
            # Smooth GD
            alloc = min(remaining * 0.4, 40.0)
            a_new = smooth_gd(best_a, time_budget=alloc,
                              tau_levels=[20, 100, 500, 2000], verbose=False)
        else:
            # Linearized target step + LP
            alloc = min(remaining * 0.2, 15.0)
            a_try = best_a.copy()
            # Apply several linearized steps
            for _ in range(3):
                if time.monotonic() - start > time_budget - 5:
                    break
                a_try = linearized_target_step(a_try, target_ratio=0.8, step_size=0.2)
            # Then LP polish
            alloc_lp = min(remaining * 0.2, 20.0)
            a_new = lp_fixed_point(a_try, time_budget=alloc_lp, seed=seed + phase)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose:
                print(f"      phase={phase}: obj={best_obj:.10f} **")
        else:
            no_improve += 1
            if verbose:
                print(f"      phase={phase}: obj={obj_new:.10f} (no improve, {no_improve})")

        if no_improve >= max_no_improve:
            # Reset with perturbation
            best_a = spectral_perturb(best_a, rng, strength=0.3)
            best_a = best_a / best_a.sum()
            no_improve = 0

        phase += 1

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-n exploration with combined LP/GD refinement,
    spectral perturbation, and linearized target steps.

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")

        parent = None

    n_parent = len(best_seq) if best_seq is not None else 800

        700, 750, 800, 850, 900, 950, 1000, 1100, 1200, 1400, 1600
    ]))
    # Filter valid range
    n_candidates = [n for n in n_candidates if 500 <= n <= 2000]
    # Sort by priority: closest to parent first, then by objective potential
    n_candidates.sort(key=lambda x: (abs(x - n_parent), x))
    # Keep at most 8 candidates
    n_candidates = n_candidates[:8]
    print(f"\nN candidates: {n_candidates}")

    # === Phase 1: Quick screening of n values ===
    print(f"\n=== Phase 1: Quick Screening ===")
    for n_try in n_candidates:

        if n_try == n_parent and best_seq is not None:
            a_try = best_seq.copy()
        else:
            a_try = resample(best_seq, n_try) if best_seq is not None else make_init(n_try, 3, rng)

        # Quick GD

        if alloc > 5:

        obj_try = compute_obj(a_try)
        if obj_try < best_score:

            best_seq = a_try.copy()
            print(f"  n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 2: Deep optimization on best n ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.5, 500.0)
    if p2_budget > 60:
        a_deep = combined_refine(best_seq, time_budget=p2_budget, verbose=True, seed=100)
        obj_deep = compute_obj(a_deep)
        if obj_deep < best_score:
            best_score = obj_deep
            best_seq = a_deep.copy()
            print(f"  Deep: obj={best_score:.10f} **")
        else:
            print(f"  Deep: obj={obj_deep:.10f}")

    # === Phase 3: Try other n values with deep optimization ===
    print(f"\n=== Phase 3: Multi-n Deep Search ===")
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample(best_seq, n_try)

        alloc_init = min(remaining * 0.05, 15.0)
        if alloc_init > 5:
            a_try = smooth_gd(a_try, time_budget=alloc_init, tau_levels=[20, 200])

        # Deep refinement
        alloc_deep = min(remaining * 0.25, 80.0)
        if alloc_deep > 20:
            a_try = combined_refine(a_try, time_budget=alloc_deep, verbose=False,
                                    seed=200 + n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
