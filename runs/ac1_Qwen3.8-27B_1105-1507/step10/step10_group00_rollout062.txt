    """Project v onto {x: sum(x)=s, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n+1) > (cssv - s))
    theta = (cssv[rho] - s) / (rho + 1.0)

Enhanced subgradient + LP optimization for step function coefficients.
Uses Nesterov-accelerated projected subgradient with active-set tracking,
coordinate descent refinement, and adaptive length exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def proj_simplex(v, s=1.0):
    """Project v onto {x: sum(x)=s, x>=0} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - s)
    if not np.any(cond):
        theta = max(0, v - s / n)
        return np.maximum(v - s / n, 0)
    rho_val = np.max(np.where(cond)[0])
    theta = (cssv[rho_val] - s) / (rho_val + 1.0)
    return np.maximum(v - theta, 0)


def active_subgradient(a, conv, k_active):
    """
    Compute subgradient of max_{k in k_active} conv(a,a)[k] w.r.t. a.
    For each active peak position k, the gradient of conv(a,a)[k] w.r.t. a[i] is:
      2 * a[k-i] if k-i is in [0, n-1] and k != 2i
      2 * a[i] if k == 2i (and k-i = i)
      0 otherwise
    
    We average over the active set (uniform weights).
    """
    n = len(a)
    grad = np.zeros(n)
    for k in k_active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                if k == 2 * i:
                    grad[i] += 2.0 * a[i]
                else:
                    grad[i] += 2.0 * a[j]
    grad /= len(k_active)
    return grad

def nesterov_subgradient(a, time_budget=60.0, top_k=3, verbose=False):
    """
    Nesterov accelerated projected subgradient descent for min max_k conv(a,a)[k]
    on the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    y = a.copy()  # Nesterov look-ahead point
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    start = time.monotonic()
    t = 1
    beta = 0.5  # momentum parameter
    
    while time.monotonic() - start < time_budget:
        # Compute convolution at y
        conv_y = compute_conv(y)
        obj_y = float(np.max(conv_y))
        
        # Find top-k active peaks
        if top_k > 1:
            k_active = np.argsort(conv_y)[::-1][:top_k]
        else:
            k_active = [int(np.argmax(conv_y))]
        
        # Compute subgradient at y
        grad = np.zeros(n)
        for k in k_active:
            grad += grad_conv_at_k(y, int(k))
        grad /= len(k_active)
        
        # Update Nesterov parameters
        t_new = t + 1
        beta_new = (1 + np.sqrt(1 + 4 * beta**2)) / 2
        
        # Step size: for subgradient method with f having Lipschitz constant L
        # (which is bounded by ~2*max(a)*2*n for our problem), use lr = c / sqrt(t)
        # The "Lipschitz constant" of the subgradient is roughly 2*L where L is the
        # max of the dual norm of the gradient, which is ~ 2*max(a)*n
        # For a on the simplex, max(a) <= 1, so L ~ 2n
        # Standard step: lr = R / (L * sqrt(t)) where R is the diameter of the simplex ~ sqrt(2/n)
        lr = 0.5 / np.sqrt(t)
        
        # Gradient step
        y_new = y - lr * grad
        
        # Nesterov momentum
        y_new = (1 - beta_new / t_new) * y + (beta_new / t_new) * y_new
        
        # Project onto simplex
        y_new = proj_simplex(y_new, 1.0)
        
        # Evaluate
        obj_new = float(np.max(compute_conv(y_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = y_new.copy()
        
        y = y_new
        t = t_new
        beta = beta_new
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    NSGD t={t}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a

"""
Hybrid Nesterov subgradient + LP polishing + coordinate descent optimization
for step function coefficients. Uses active-set tracking, extrapolation,
pairwise coordinate refinement, and adaptive length exploration to minimize
the peak of the full linear auto-convolution.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def proj_simplex(v, s=1.0):
    """Project v onto {x: sum(x)=s, x>=0} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - s)
    if not np.any(cond):
        return np.maximum(v - s / n, 0)
    rho_val = int(np.max(np.where(cond)[0]))
    theta = (cssv[rho_val] - s) / (rho_val + 1.0)
    return np.maximum(v - theta, 0)


def grad_conv_at_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a. O(n)."""
    n = len(a)
    grad = np.zeros(n)
    j = k - np.arange(n)
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]
    return grad


def nesterov_subgrad(a, time_budget=50.0, top_k=3, verbose=False):
    """Nesterov accelerated projected subgradient on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    t = 1
    beta = 0.5
    no_improve = 0

    while time.monotonic() - start < time_budget:
        conv_y = compute_conv(y)
        obj_y = float(np.max(conv_y))

        if top_k > 1:
            k_active = np.argsort(conv_y)[::-1][:top_k]
        else:
            k_active = [int(np.argmax(conv_y))]

        grad = np.zeros(n)
        for k in k_active:
            grad += grad_conv_at_k(y, int(k))
        grad /= len(k_active)

        t_new = t + 1
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        lr = 0.3 / np.sqrt(t)

        y_step = y - lr * grad
        y_new = (1.0 - beta_new / t_new) * y + (beta_new / t_new) * y_step
        y_new = proj_simplex(y_new, 1.0)

        obj_new = float(np.max(compute_conv(y_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = y_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        y = y_new
        t = t_new
        beta = beta_new

        if no_improve > 3000:
            y = best_a.copy()
            beta = 0.5
            t = max(1, t // 4)
            no_improve = 0

        if verbose and t % 2000 == 0:
            print(f"    NSGD t={t}: obj={best_obj:.8f}, elapsed={time.monotonic()-start:.1f}s")

    return best_a


def build_conv_matrix(a, n):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, top_k=50):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses top-k active constraints for speed."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A_full = build_conv_matrix(a_norm, n)

    # Use all constraints for accuracy (the LP is not too large)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_full @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_minpeak_reduced(a, num_constraints=200):

    n = len(a)
    a_norm = a / float(np.sum(a))

    conv_curr = compute_conv(a_norm)
    # Select top positions + spread out positions
    top_pos = np.argsort(conv_curr)[::-1][:num_constraints]
    # Also include positions 0, n-1, 2n-2 (edges and center)

    important.add(0)

    rows = sorted(important)
    A_reduced = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                A_reduced[idx, j] = a_norm[m]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_reduced @ g <= M, cp.sum(g) == 1]

        except Exception:

    return None, None


def line_search_blend(a, g_dir, time_budget=3.0, t_max=2.0):
    """Golden section search for optimal blend factor t in [0, t_max]."""
    lo, hi = 0.0, t_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    def obj_at(t):
        a_try = (1.0 - t) * a + t * g_dir
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    c = lo + gr * (hi - lo)
    d = lo + (1.0 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    start = time.monotonic()
    for _ in range(35):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)

    # Check some additional points
    for t_try in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 1.8, 2.0]:
        if t_try <= t_max:
            ot = obj_at(t_try)
            if ot < obj_best:
                obj_best = ot
                t_best = t_try

    return t_best, obj_best


def lp_polish(a, time_budget=120.0, max_iters=100, verbose=False, use_reduced=False):
    """LP polishing with extrapolation and adaptive restarts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777 + it)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        # Solve LP
        T0 = time.monotonic()
        if use_reduced:
            g_lp, M_lp = solve_lp_minpeak_reduced(a, num_constraints=min(300, 2 * n - 1))
        else:
            g_lp, M_lp = solve_lp_minpeak(a)
        lp_time = time.monotonic() - T0

        if g_lp is None:
            break

        # Line search with extrapolation
        t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=min(3.0, remaining * 0.06), t_max=2.0)

        if t_opt > 0 and obj_opt < best_obj - 1e-15:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 10 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"lp_time={lp_time:.2f}s, elapsed={time.monotonic()-start:.1f}s")

        # Adaptive restart
        if stall >= 3:
            if stall <= 5:
                freq = rng.integers(1, max(2, n // 40))
                phase = rng.uniform(0, 2 * np.pi)
                idx_arr = np.arange(n, dtype=np.float64) / max(1, n - 1)
                pert = 0.03 * stall * np.cos(2 * np.pi * freq * idx_arr + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                ss = float(np.sum(a))
                a = a / ss if ss > 0.01 else best_a.copy()
            elif stall <= 10:
                noise = rng.normal(0, 0.02 * stall, n)
                a = best_a + noise
                a = np.maximum(a, 0)
                ss = float(np.sum(a))
                a = a / ss if ss > 0.01 else best_a.copy()
            else:
                # Blend with a fresh U-shaped init
                x = np.linspace(0, 1, n)
                a_u = 1.0 / (0.1 + 3.0 * (2 * x - 1) ** 2)
                a_u = a_u / a_u.sum()
                a = 0.3 * a_u + 0.7 * best_a
                a = a / a.sum()
            stall = 0

    return best_a


def coordinate_descent(a, time_budget=30.0, verbose=False):
    """Pairwise coordinate descent: for pairs (i,j), find optimal mass transfer."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    # Focus on "important" coordinates: edges and high-value positions
    a_sorted = np.argsort(a)[::-1]
    important = set(list(a_sorted[:50]) + list(a_sorted[-50:]))
    important_list = list(important)

    it = 0
    while time.monotonic() - start < time_budget:
        # Pick a random pair from important coordinates
        if len(important_list) < 2:
            break
        i1 = int(rng.choice(important_list))
        i2 = int(rng.choice(important_list))
        if i1 == i2:
            continue

        c = a[i1] + a[i2]
        if c < 1e-15:
            continue

        # Golden section on a[i1] in [0, c]
        lo, hi = 0.0, c
        gr = (np.sqrt(5.0) - 1.0) / 2.0

        def eval_frac(frac):
            a_try = a.copy()
            a_try[i1] = frac * c
            a_try[i2] = (1.0 - frac) * c
            conv_try = compute_conv(a_try)
            return float(np.max(conv_try))

        p1 = lo + gr * (hi - lo)
        p2 = lo + (1.0 - gr) * (hi - lo)
        f1 = eval_frac(p1)
        f2 = eval_frac(p2)

        for _ in range(25):
            if f1 < f2:
                hi = p2
                p2 = p1
                f2 = f1
                p1 = lo + gr * (hi - lo)
                f1 = eval_frac(p1)
            else:
                lo = p1
                p1 = p2
                f1 = f2
                p2 = lo + gr * (hi - lo)
                f2 = eval_frac(p2)

        best_frac = p1 if f1 < f2 else p2
        best_val = min(f1, f2)

        # Also check current and endpoints
        curr_val = float(np.max(compute_conv(a)))
        f0 = eval_frac(a[i1] / c)  # current
        f_lo = eval_frac(0.0)
        f_hi = eval_frac(1.0)

        candidates = [(best_frac, best_val), (a[i1] / c, f0), (0.0, f_lo), (1.0, f_hi)]
        best_frac, best_val = min(candidates, key=lambda x: x[1])

        a[i1] = best_frac * c
        a[i2] = (1.0 - best_frac) * c
        a = np.maximum(a, 0)
        ss = float(np.sum(a))
        if ss > 0.01:
            a = a / ss

        obj_new = compute_obj(a)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()

        it += 1
        if verbose and it % 100 == 0:
            print(f"    CD it{it}: obj={best_obj:.8f}, elapsed={time.monotonic()-start:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:  # U-shaped
        a = 1.0 / (0.05 + 3.0 * (2 * x - 1) ** 2)
    elif mode == 1:  # Bimodal
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:  # Inverted triangle
        a = 4.0 * np.abs(x - 0.5) + 0.05
    elif mode == 3:  # Random smooth
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7.0
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:  # Exponential edges
        a = np.exp(4.0 * (2 * x - 1) ** 2)
    elif mode == 5:  # High U
        a = 1.0 / (0.01 + 10.0 * (2 * x - 1) ** 2)
    elif mode == 6:  # Cosine
        a = 0.5 + 0.5 * np.cos(np.pi * (2 * x - 1)) ** 4
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def try_length(a_best, n_try, time_budget, rng):
    """Try a different sequence length via interpolation + optimization."""
    n_old = len(a_best)
    if n_try <= 10 or n_try > 5000:
        return None, np.inf
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        return None, np.inf
    a_try = a_try / s

    # Quick subgradient
    t1 = min(10.0, time_budget * 0.2)
    if t1 > 3:
        a_try = nesterov_subgrad(a_try, time_budget=t1, top_k=3)

    # LP polish
    t2 = min(60.0, time_budget * 0.6)
    if t2 > 5:
        a_try = lp_polish(a_try, time_budget=t2, max_iters=30)

    obj_try = compute_obj(a_try)
    return a_try, obj_try
