- We compute conv = conv(a, a)
- p = softmax(tau * (conv - max(conv)))

- We want to minimize max(conv), so we move in the direction -grad

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based minimax polishing, FFT-accelerated high-tau smoothed gradient
descent in log-parameterization, symmetry exploitation, and systematic n exploration.
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
from scipy.optimize import minimize as scipy_minimize
import cvxpy as cp
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes a>=0."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def compute_smoothed_grad(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) w.r.t. a.
    Returns (smoothed_val, grad) where grad[i] = d/d(a[i]) of smoothed max."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # For numerical stability
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    smoothed_val = max_conv + np.log(sum_exp) / tau
    
    # Compute gradient: grad[i] = 2 * sum_k p[k] * a[k-i]
    # Using FFT: grad = 2 * ifft(fft(a,N) * fft(p_rev,N)).real[n-1:2n-1][::-1]
    p_rev = p[::-1]
    N = 1
    while N < 4 * n:
        N <<= 1
    A_f = fft(a, N)
    P_f = fft(p_rev, N)
    conv_ap = ifft(A_f * P_f).real
    grad = 2.0 * conv_ap[n - 1:2 * n - 1][::-1]
    
    return smoothed_val, grad

def compute_smoothed_grad_log(b, tau, support, full_len):
    """Compute smoothed objective and gradient w.r.t. log-parameters b on support."""
    # Compute a from log-params
    b_shifted = b - b.max()
    a_supp = np.exp(b_shifted)
    a_supp = a_supp / a_supp.sum()
    
    a = np.zeros(full_len)
    a[support] = a_supp
    
    smoothed_val, grad_full = compute_smoothed_grad(a, tau)
    
    # Chain rule for log-parameterization
    grad_b = a_supp * (grad_full[support] - np.dot(a_supp, grad_full[support]))
    
    return smoothed_val, grad_b

def lp_solve(a, timeout=30):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:
            continue
    
    return None, None

def lp_polish(a, time_budget, verbose=True):
    """LP polishing with adaptive line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.3
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp, M_lp = lp_solve(a)
        if g_lp is None:
            break
        
        # Adaptive line search
        t_candidates = np.concatenate([
            np.linspace(0.01, 1.0, 15),
            [max(0.01, last_t * 0.5), max(0.01, last_t * 0.8),
             last_t, min(2.0, last_t * 1.2), min(2.0, last_t * 1.5),
             min(2.0, last_t * 2.0), min(2.0, last_t * 3.0)]
        ])
        t_candidates = np.unique(np.clip(t_candidates, 0.005, 2.0))
        
        best_t_obj = best_obj
        best_t = last_t
        best_a_local = best_a.copy()
        
        for t in t_candidates:
            if t <= 1.0:
                a_try = (1.0 - t) * a + t * g_lp
            else:
                # Extrapolation
                a_try = a + t * (g_lp - a)
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:

            a_try = a_try / s_try

            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t
            stall = 0
            if verbose and time.monotonic() - start > 1:
                elapsed = time.monotonic() - start
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.4f}, elapsed={elapsed:.0f}s")
        else:
            stall += 1

def high_tau_gd_log(a, time_budget, tau_start=100, tau_end=50000, verbose=True):
    """High-precision smoothed gradient descent in log-parameterization."""
    full_len = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    # Support: entries above threshold
    threshold = max(1e-8, a.max() * 1e-6)
    support = np.where(a > threshold)[0]
    if len(support) < 2:
        support = np.arange(full_len)
    
    b0 = np.log(a[support] + 1e-30)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Multi-stage tau
    n_stages = 6
    tau_stages = tau_start + (tau_end - tau_start) * (np.arange(n_stages) / max(1, n_stages - 1)) ** 2
    tau_stages = np.maximum(tau_stages, tau_start)
    
    current_b = b0.copy()
    for stage_idx in range(n_stages):
        tau = tau_stages[stage_idx]
        stage_deadline = start + time_budget * (stage_idx + 1) / n_stages
        
        # Use L-BFGS with the smoothed objective
        def obj_grad(b_vec):
            val, grad = compute_smoothed_grad_log(b_vec, tau, support, full_len)
            return val, grad
        
        try:
            res = scipy_minimize(obj_grad, current_b, method='L-BFGS-B',
                                 jac=True, options={'maxiter': 300, 'gtol': 1e-12,
                                                    'ftol': 1e-15, 'maxfun': 1000})
            if res.x is not None:
                b_shifted = res.x - res.x.max()
                a_supp = np.exp(b_shifted)
                a_supp = a_supp / a_supp.sum()
                a_new = np.zeros(full_len)
                a_new[support] = a_supp
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    current_b = res.x.copy()
                    if verbose:
                        print(f"    GD stage{stage_idx}: tau={tau:.0f}, obj={best_obj:.8f}")
        except Exception:
            pass
        
        if time.monotonic() > stage_deadline:
            break
    
    return best_a

        tau = tau_start + (tau_end - tau_start) * progress ** 0.7
        
        # Nesterov
        y = a + beta * v
        y = np.maximum(y, 0)

            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.5
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
        a = a / a.sum()
    elif mode == 5:
        # Symmetric bump
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / (2 * 0.1 ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Two bumps
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) ** 2) / (2 * 0.08 ** 2)) + np.exp(-((x - 0.7) ** 2) / (2 * 0.08 ** 2))
        a = a / a.sum()
    elif mode == 7:
        # Symmetric with dip in middle
        x = np.linspace(0, 1, n)
        a = 1.0 - 0.5 * np.exp(-((x - 0.5) ** 2) / (2 * 0.1 ** 2))
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 8:
        # Cosine window
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a

def perturb(a, rng, strength=0.05):
    """Add random perturbation and renormalize."""
    n = len(a)
    noise = rng.normal(0, strength, n)
    a_new = a + noise * a  # multiplicative noise
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
