def compute_convolution(seq: List[float]) -> np.ndarray:
    """Computes the full linear convolution of a sequence using FFT."""

    if n == 0:
        return np.array([])

        a = np.array(seq, dtype=np.float64)

    # Determine a good FFT size with safe padding

        size *= 2
