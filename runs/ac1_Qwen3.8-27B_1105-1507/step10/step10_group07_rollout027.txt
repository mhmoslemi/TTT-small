"""
Hybrid optimization for minimal peak autocorrelation of non-negative sequences.
Uses smoothed gradient descent with adaptive temperature, multiplicative weight
updates, Nesterov acceleration, multi-length exploration, and LP polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_smoothed_gradient(a, tau):
    """Compute gradient of smoothed objective (1/tau)*log(sum exp(tau*conv_k))."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    # conv_k = sum_i a_i * a_{k-i}
    # d(conv_k)/d(a_j) = a_{k-j} + a_{k-j} = 2*a_{k-j} (with correction for k=2j)

    # = sum_i a_i * sum_k p_k * a_{k-i}

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_gd(a, time_budget=100.0, tau_start=5.0, tau_end=500.0, lr_init=None, seed=42):
    """Smoothed gradient descent with adaptive temperature and Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    if lr_init is None:
        lr_init = 2.0 / np.sqrt(n)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    momentum = 0.85
    v = np.zeros(n)
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        # Smooth temperature increase
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.7)
        
        grad, p = compute_smoothed_gradient(a, tau)
        
        # Nesterov-like momentum
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        lr = max(1e-10, lr * 0.999)
        it += 1
        
    return best_a

def multiplicative_weight_update(a, time_budget=100.0, eta_init=None, tau=50.0, seed=42):
    """
    Multiplicative weight update for minimizing peak autocorrelation.
    a[j] *= exp(-eta * grad[j]), then normalize.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    if eta_init is None:
        eta_init = 1.0
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        # Adjust temperature
        tau_eff = tau * (1.0 + 2.0 * progress)
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau_eff * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Compute gradient
        grad, _ = compute_smoothed_gradient(a, tau_eff)
        
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-300))
        log_a -= eta * grad / max_conv  # Scale gradient by max_conv for stability
        a_new = np.exp(log_a - np.max(log_a))  # Shift for numerical stability
        s_new = a_new.sum()
        if s_new < 1e-30:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        eta *= 0.998
        it += 1
    
    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_direction(a, M_target=None):
    """Solve LP: min M s.t. A @ g <= M, sum(g)=1, g>=0."""
    import cvxpy as cp
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
        cp.norm(delta, 2) <= r,
    ]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

def coordinate_descent(a, time_budget=60.0, seed=42):
    """
    Cyclic coordinate descent for peak autocorrelation minimization.
    For each coordinate j, find the optimal value that minimizes the peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # We'll do a few sweeps
    while time.monotonic() - start < time_budget:
        improved = False
        # Random order for diversity
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            
            # Try a few candidate values for a[j]
            current_val = a[j]
            candidates = [current_val * 0.5, current_val * 0.8, current_val * 1.2,
                         current_val * 1.5, current_val * 0.2, 0.0]
            # Add some absolute values
            mean_val = 1.0 / n
            candidates.extend([mean_val * 0.5, mean_val * 2.0])
            
            best_candidate = a.copy()
            best_local_obj = compute_obj(a)
            
            for c_val in candidates:
                c_val = max(0.0, c_val)
                a_trial = a.copy()
                old_sum = a_trial.sum()
                a_trial[j] = c_val
                new_sum = a_trial.sum()
                if new_sum < 1e-10:
                    continue
                a_trial = a_trial / new_sum
                obj_trial = compute_obj(a_trial)
                if obj_trial < best_local_obj - 1e-12:
                    best_local_obj = obj_trial
                    best_candidate = a_trial.copy()
            
            if np.any(best_candidate != a):
                a = best_candidate
                a = a / a.sum()
                improved = True
                if compute_obj(a) < best_obj - 1e-13:
                    best_obj = compute_obj(a)
                    best_a = a.copy()
        
        if not improved:
            # Perturb to escape
            a = a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
    
    return best_a

aggressive fine-tuning with small-radius TR-SOCP, and complementary MWU method.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(a, tau):
    """
    Compute gradient of smoothed objective (1/tau)*log(sum exp(tau*conv_k)).
    Returns (gradient, softmax_weights).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    # Gradient of sum_k p_k * conv_k w.r.t. a
    # conv_k = sum_i a_i * a_{k-i}
    # d/d(a_j) sum_k p_k * conv_k = sum_k p_k * (2*a_{k-j} - a_{k-j}*[k==2j])
    # = 2 * cross_corr(p, a)[j] - diagonal_correction[j]
    N = 1
    while N < 2 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross_corr = ifft(P * np.conj(A)).real[:n]

    # Diagonal correction: for even k, the middle term is counted once not twice
    correction = np.zeros(n)
    even_ks = np.arange(0, 2 * n - 1, 2)
    for k in even_ks:
        j = k // 2
        if 0 <= j < n:
            correction[j] += p[k] * a[j]

    grad = 2.0 * cross_corr - correction
    return grad, p


def smoothed_gd(a, time_budget=100.0, tau_start=5.0, tau_end=500.0, lr_init=None, seed=42):
    """Smoothed gradient descent with adaptive temperature and momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    if lr_init is None:
        lr_init = 3.0 / np.sqrt(n)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    momentum = 0.85
    v = np.zeros(n)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        # Cosine-like temperature schedule
        tau = tau_start + (tau_end - tau_start) * (1.0 - np.cos(np.pi * progress ** 0.6)) / 2.0

        grad, p = compute_smoothed_gradient(a, tau)

        v = momentum * v + grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        lr = max(1e-12, lr * 0.9995)
        if stall > 200:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-14:
                break
        it += 1

    return best_a


def multiplicative_weight_update(a, time_budget=100.0, tau=40.0, eta_init=None, seed=42):
    """
    Multiplicative weight update: a[j] *= exp(-eta * grad[j] / max_conv), normalize.
    This is a mirror descent method with exponential family mirror.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    if eta_init is None:
        eta_init = 1.5

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau_eff = tau * (1.0 + 3.0 * progress)

        grad, p = compute_smoothed_gradient(a, tau_eff)
        max_conv = float(np.max(compute_conv(a)))
        if max_conv < 1e-15:
            break

        # Multiplicative update with numerical stability
        scaled_grad = grad / max_conv
        log_updates = -eta * scaled_grad
        log_updates -= np.max(log_updates)  # Shift for stability
        weights = np.exp(log_updates)
        a_new = a * weights
        s_new = a_new.sum()
        if s_new < 1e-30:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        eta *= 0.997
        it += 1

    return best_a


def fourier_perturb(a, rng, amount=0.05):
    """Perturb the sequence in the frequency domain (phase perturbation)."""
    n = len(a)
    N = 1
    while N < n:
        N <<= 1
    A = fft(a, N)
    magnitudes = np.abs(A)
    phases = np.angle(A) + rng.uniform(-amount, amount, N)
    A_pert = magnitudes * np.exp(1j * phases)
    a_pert = ifft(A_pert).real[:n]
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-10:
        return a.copy()
    return a_pert / s


def fourier_perturb_magnitude(a, rng, amount=0.1):
    """Perturb the magnitudes in the frequency domain."""
    n = len(a)
    N = 1
    while N < n:
        N <<= 1
    A = fft(a, N)
    magnitudes = np.abs(A) * np.exp(rng.normal(0, amount, N))
    phases = np.angle(A)
    A_pert = magnitudes * np.exp(1j * phases)
    a_pert = ifft(A_pert).real[:n]
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-10:
        return a.copy()
    return a_pert / s


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, M_target=None):
    """Solve LP: min M s.t. A @ g <= M, sum(g)=1, g>=0."""
    import cvxpy as cp
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_tr_sscp(a, r):
    """Trust region SOCP: min M s.t. linearized constraints, ||delta||<=r, a+delta>=0, sum=1."""
    import cvxpy as cp
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)

    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta

    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
        cp.norm(delta, 2) <= r,
    ]

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None


def tr_sscp_polish(a, time_budget=300.0, r_init=0.25, seed=42):
    """Trust region SOCP polishing with adaptive radius and perturbation restarts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    r = r_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        delta_dir, M_val = solve_tr_sscp(a, r)

        if delta_dir is None:
            g_lp, M_val = solve_lp_direction(a)
            if g_lp is None:
                break
            delta_dir = g_lp - a
        else:
            norm_d = np.linalg.norm(delta_dir)
            if norm_d > r:
                delta_dir = delta_dir * (r / norm_d)

        finite_mask = np.isfinite(delta_dir)
        if not finite_mask.all():
            delta_dir = np.where(finite_mask, delta_dir, 0.0)

        best_alpha = 0
        best_alpha_obj = best_obj
        alphas = [0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0, 1.3, 1.7]
        for t_alpha in alphas:
            a_try = a + t_alpha * delta_dir
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj - 1e-15:
                best_alpha_obj = obj_try
                best_alpha = t_alpha

        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-15:
            a = a + best_alpha * delta_dir
            a = np.maximum(a, 0)
            s_n = a.sum()
            if s_n > 1e-10:
                a = a / s_n
            best_obj = best_alpha_obj
            best_a = a.copy()
            no_improve = 0
            r = min(r * 1.08, 1.2)
        else:
            no_improve += 1
            r *= 0.55
            if no_improve >= 7:
                if r < 0.02:
                    # Switch to Fourier perturbation for diversity
                    if rng.random() < 0.5:
                        a = fourier_perturb(best_a, rng, amount=0.03)
                    else:
                        a = best_a * np.exp(rng.normal(0, 0.01, n))
                        a = np.maximum(a, 0)
                        s_p = a.sum()
                        if s_p > 1e-10:
                            a = a / s_p
                    r = r_init
                    no_improve = 0
                    if it > 120:
                        break
                else:
                    r = max(r * 0.6, 0.02)

        it += 1
        if it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    TR it{it}: obj={best_obj:.8f}, r={r:.4f}, t={elapsed:.0f}s")

    return best_a


def lp_polish(a, time_budget=200.0, seed=42):
    """LP-based polishing with line search and adaptive perturbation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        best_t = 0
        best_t_obj = best_obj
        for t_val in [0.03, 0.08, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0]:
            a_try = (1 - t_val) * best_a + t_val * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-15:
                best_t_obj = obj_try
                best_t = t_val

        if best_t > 0 and best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = (1 - best_t) * best_a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            s_ba = best_a.sum()
            if s_ba > 1e-10:
                best_a = best_a / s_ba
            a = best_a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                if rng.random() < 0.4:
                    a = fourier_perturb(best_a, rng, amount=0.02)
                else:
                    scale = 0.008 + 0.004 * min(no_improve, 10)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    s_a = a.sum()
                    if s_a > 1e-10:
                        a = a / s_a
                no_improve = 0
                if it > 100:
                    break

        it += 1

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def subgradient_refine(a, time_budget=80.0, K=12, seed=42):
    """Subgradient method with top-K peak weighting and Nesterov acceleration."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 4.0 / np.sqrt(n)
    momentum = 0.75
    v = np.zeros(n)
    a_prev = a.copy()
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break

        # Soft weighting of top peaks
        weights = np.exp((peak_vals - max_pv) * 10.0 / max_pv)
        weights /= weights.sum()

        # Compute subgradient as weighted sum
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                segment = a[j_start:j_end + 1][::-1]
                g[i_start:i_end + 1] += w * 2.0 * segment
            # Diagonal correction for even k
            if k % 2 == 0:
                j_diag = k // 2
                if 0 <= j_diag < n:
                    g[j_diag] -= w * a[j_diag]

        # Nesterov acceleration
        t = it + 1
        beta = (t - 1) / (t + 2)
        a_nest = a + beta * (a - a_prev)
        a_prev = a.copy()

        v = momentum * v + g
        a_new = a_nest - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        lr = max(1e-12, lr * 0.999)
        if stall > 300:
            lr *= 0.5
            v *= 0.2
            stall = 0
            if lr < 1e-14:
                break
        it += 1

    return best_a

        ("gaussian_tiny", 0.003),
        ("fourier_phase2", 0.02),
        ("gaussian_xl", 0.04),
    ]

    for p_idx, (p_type, p_scale) in enumerate(perturbation_types):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        if p_type == "fourier_phase":
            a_pert = fourier_perturb(best_seq, rng, amount=p_scale)
        elif p_type == "fourier_mag":
            a_pert = fourier_perturb_magnitude(best_seq, rng, amount=p_scale)
        elif p_type == "fourier_phase2":
            a_pert = fourier_perturb(best_seq, rng, amount=p_scale)
        else:
            a_pert = best_seq * np.exp(rng.normal(0, p_scale, len(best_seq)))
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert < 1e-10:
                continue
            a_pert = a_pert / s_pert

        # Quick GD to settle
        alloc1 = min(remaining * 0.12, 25.0)
        if alloc1 > 8:
            a_pert = smoothed_gd(a_pert, time_budget=alloc1, tau_start=10.0, tau_end=300.0)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.22, 80.0)
        if alloc2 > 15:
            a_pert = tr_sscp_polish(a_pert, time_budget=alloc2, r_init=0.15, seed=500 + p_idx)

        obj_pert = compute_obj(a_pert)
        if obj_pert < best_score:
            best_score = obj_pert
            best_seq = a_pert.copy()
            print(f"  Pert{p_idx} ({p_type}, {p_scale}): obj={best_score:.8f} **")
        else:
            print(f"  Pert{p_idx} ({p_type}, {p_scale}): obj={obj_pert:.8f}")

    # === Phase 8: Final intensive fine-tuning ===
    print(f"\n=== Phase 8: Final Fine-tuning ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.45, 130.0)
        a_final = tr_sscp_polish(best_seq, time_budget=alloc, r_init=0.10, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final TR (small r): obj={best_score:.8f} **")
        else:
            print(f"  Final TR (small r): obj={obj_final:.8f}")

    # Final LP polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 40.0)
        a_flp = lp_polish(best_seq, time_budget=alloc, seed=3000)
        obj_flp = compute_obj(a_flp)
        if obj_flp < best_score:
            best_score = obj_flp
            best_seq = a_flp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Ultra-precise final TR with tiny radius
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 30.0)
        a_ultra = tr_sscp_polish(best_seq, time_budget=alloc, r_init=0.05, seed=4000)
        obj_ultra = compute_obj(a_ultra)
        if obj_ultra < best_score:
            best_score = obj_ultra
            best_seq = a_ultra.copy()
            print(f"  Ultra TR: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
