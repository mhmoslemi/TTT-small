import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def subgradient_descent_fftp(a, time_budget=100.0, verbose=False):
    """
    Fast FFT-based subgradient descent with Nesterov momentum and adaptive LR.
    Works on the simplex: a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adaptive parameters
    lr = 0.05
    momentum = np.zeros(n)
    beta = 0.7  # momentum coefficient
    stall_count = 0
    PATIENCE = 2000
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        
        # Subgradient computation
        # d/d a[i] of sum_j a[j]*a[k_max-j]
        # = a[k_max-i] (if 0 <= k_max-i < n) + a[i-k_max] (if 0 <= i-k_max < n)
        grad = np.zeros(n)
        # First term: for each i, if 0 <= k_max - i < n, grad[i] += a[k_max - i]
        valid1 = (k_max - np.arange(n) >= 0) & (k_max - np.arange(n) < n)
        if np.any(valid1):
            grad[valid1] += a[k_max - np.arange(n)[valid1]]
        # Second term: for each i, if 0 <= i - k_max < n, grad[i] += a[i - k_max]
        valid2 = (np.arange(n) - k_max >= 0) & (np.arange(n) - k_max < n)
        if np.any(valid2):
            grad[valid2] += a[np.arange(n)[valid2] - k_max]
        
        # Nesterov momentum update
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)

    if s < 1e-10:

    it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        
        # Subgradient
        grad = np.zeros(n)
        idx = np.arange(n)
        valid1 = (k_max - idx >= 0) & (k_max - idx < n)
        if np.any(valid1):
            grad[valid1] += a[k_max - idx[valid1]]
        valid2 = (idx - k_max >= 0) & (idx - k_max < n)
        if np.any(valid2):
            grad[valid2] += a[idx[valid2] - k_max]
        
        # Mirror descent update (multiplicative)
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-15)  # avoid zeros

            best_a = a_new.copy()
            stall_count = 0

def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return a_new / s

import cvxpy as cp

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

"""
FFT-based adaptive subgradient optimization for step function peak minimization.
Combines Nesterov momentum subgradient, mirror descent, coordinate refinement,
and LP polishing with systematic n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_subgrad(a, k_star):
    """Compute subgradient of conv(a,a)[k_star] w.r.t. a."""
    n = len(a)
    idx = np.arange(n)
    j1 = k_star - idx
    j2 = idx - k_star
    grad = np.zeros(n)
    mask1 = (j1 >= 0) & (j1 < n)
    if np.any(mask1):
        grad[mask1] += a[j1[mask1]]
    mask2 = (j2 >= 0) & (j2 < n)
    if np.any(mask2):
        grad[mask2] += a[j2[mask2]]
    return grad


def subgradient_nesterov(a, time_budget=100.0, verbose=False, lr_init=0.03):
    """
    Nesterov momentum subgradient descent on the simplex.
    O(n log n) per iteration via FFT.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = lr_init
    momentum = np.zeros(n)
    beta = 0.7
    it = 0
    stall = 0
    PATIENCE = 3000
    min_stall_for_reset = 5000

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        grad = compute_subgrad(a, k_max)

        # Nesterov momentum
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        it += 1

        # Adaptive learning rate
        if stall > PATIENCE:
            if stall > min_stall_for_reset:
                # Full restart with perturbation
                rng = np.random.default_rng(it * 7919)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                momentum = np.zeros(n)
                lr = max(0.001, lr_init * 0.5)
                beta = 0.7
                stall = 0
            else:
                # Gentle LR decrease
                lr *= 0.7
                if it % 1000 == 0:
                    beta = 0.5 + 0.3 * (it % 4) / 3.0

        # Gradual LR decay
        lr = max(0.0005, lr_init / (1 + 0.00005 * it))

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - start
            print(f"      SG it={it}: obj={best_obj:.8f}, lr={lr:.6f}, "
                  f"beta={beta:.2f}, stall={stall}, t={elapsed:.1f}s")

    return best_a


def mirror_descent(a, time_budget=50.0, verbose=False, lr_init=0.05):
    """
    Mirror descent (multiplicative weights) on the simplex.
    Better for maintaining positivity.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = lr_init
    it = 0
    stall = 0
    PATIENCE = 5000

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        grad = compute_subgrad(a, k_max)

        # Multiplicative update
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-20)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        it += 1

        if stall > 200:
            lr *= 0.8
            stall = 0

        if stall > PATIENCE:
            rng = np.random.default_rng(it * 104729)
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            lr = lr_init * 0.5
            stall = 0

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - start
            print(f"      MD it={it}: obj={best_obj:.8f}, lr={lr:.6f}, t={elapsed:.1f}s")

    return best_a


def combined_gradient_opt(a, time_budget=100.0, verbose=False):
    """
    Combined gradient optimization: alternates between SGD and MD.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()

    b1 = min(time_budget * 0.4, 40.0)

        if verbose:

    # Phase 2: Mirror descent (better for maintaining structure)

        if verbose:

            best_obj = obj
            best_a = a.copy()
        if verbose:

def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.001, 0.999, n)
    a = 1.0 / np.sqrt(np.maximum(1e-10, 1.0 - (2 * x - 1) ** 2))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
