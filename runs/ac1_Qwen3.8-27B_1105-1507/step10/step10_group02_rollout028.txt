def normalize(a):
    s = float(np.sum(a))
    return a / s if s > 1e-10 else a


def coord_descent(a, time_budget, seed=42, verbose=False):
    """
    Fast coordinate descent with incremental convolution updates and
    priority-based coordinate ordering.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = normalize(a)
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    priority_order = None

    while time.monotonic() - start < time_budget:
        sweep += 1
        elapsed = time.monotonic() - start

        if sweep % 4 == 0:
            conv = compute_conv(a)
            s = float(np.sum(a))
            if s > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (s * s)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if priority_order is None or sweep % 12 == 1:
            max_c = float(np.max(conv))
            # Adaptive threshold: tighter as we converge
            frac = min(1.0, 0.05 + 0.15 * (1.0 - sweep / max(1, sweep + 20)))
            threshold = max_c * (1.0 - frac)
            active = conv > threshold
            affect = np.zeros(n)
            for k in range(len(conv)):
                if active[k]:
                    lo = max(0, k - n + 1)
                    hi = min(n - 1, k)
                    affect[lo:hi + 1] += 1
            priority_order = np.argsort(-affect, kind='stable')

        order = priority_order.copy()
        step = 16
        for i in range(0, n, step):
            chunk = order[i:i + step].copy()
            rng.shuffle(chunk)
            order[i:i + step] = chunk

        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            v = a[i]
            S_o = S - v
            if S_o < 1e-12:
                continue
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks]
            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_lin = a_ki[mask]
            c_lin = c_k[mask]
            k_q = 2 * i
            c_q = float(conv[k_q]) if 0 <= k_q <= 2 * n - 2 else 0.0

            def f1d(x):
                if len(a_lin) > 0:
                    lin = c_lin + 2.0 * a_lin * (x - v)
                    ml = float(np.max(lin))
                else:
                    ml = 0.0
                qv = c_q + x * x - v * v
                mv = max(ml, qv)
                d = S_o + x
                if d < 1e-30:
                    return 1e30
                return 2.0 * n * mv / (d * d)

            scale = max(v, S / n, 1e-6)
            best_x = v
            best_f = f1d(v)
            for cx in [0.0, v * 0.4, v * 1.6, v * 0.1, v * 4.0, S / (2 * n)]:
                cx = max(0.0, float(cx))
                fx = f1d(cx)
                if fx < best_f:
                    best_f = fx
                    best_x = cx

            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + 2.0 * scale
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f1d(c_pt)
            fd = f1d(d_pt)
            for _ in range(30):
                if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-15:
                continue
            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_q <= 2 * n - 2:
                conv[k_q] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_lin
            S += delta
            s_new = float(np.sum(a))
            if s_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (s_new * s_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 20 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={time.monotonic() - start:.1f}s")

    return best_a


def spectral_shaping(a, alpha=0.3, n_iters=3):
    """
    Frequency-domain spectral shaping to reduce autocorrelation peaks.
    Reduces the power spectrum at frequencies contributing most to peaks,
    then projects back to non-negative space.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    a = normalize(a)

    for _ in range(n_iters):
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        if max_c < 1e-15:
            break

        N = 1
        while N < 4 * n:
            N <<= 1

        A = fft(a, N)
        P = np.abs(A) ** 2

        # Compute autocorrelation in frequency domain sense
        c = ifft(P).real

        # Identify peak lags
        peak_lags = np.where(c > 0.9 * c.max())[0]

        # Reduce power at frequencies that contribute to peaks
        # For each peak lag k, the contribution to c[k] is proportional to
        # sum_f |A[f]|^2 * exp(-2j*pi*f*k/N). We reduce |A[f]| where
        # the phase alignment with the peak is strongest.
        phases = np.exp(-2j * np.pi * np.arange(N) * peak_lags[0] / N)
        contrib = np.abs(A * phases) ** 2
        mean_contrib = contrib.mean()

        # Soft threshold: reduce large contributions
        reduction = np.ones(N)
        mask_high = contrib > mean_contrib
        reduction[mask_high] = 1.0 - alpha * (contrib[mask_high] - mean_contrib) / (contrib.max() + 1e-30)
        reduction = np.clip(reduction, 0.3, 1.0)

        A_new = A * np.sqrt(reduction)
        a_new = ifft(A_new).real
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        # Check if this helped
        obj_old = 2.0 * n * float(np.max(compute_conv(a)))
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < obj_old:
            a = a_new
        else:
            # Try with different peak lag
            if len(peak_lags) > 1:
                phases2 = np.exp(-2j * np.pi * np.arange(N) * peak_lags[1] / N)
                contrib2 = np.abs(A * phases2) ** 2
                mean_c2 = contrib2.mean()
                reduction2 = np.ones(N)
                mask_h2 = contrib2 > mean_c2
                reduction2[mask_h2] = 1.0 - alpha * (contrib2[mask_h2] - mean_c2) / (contrib2.max() + 1e-30)
                reduction2 = np.clip(reduction2, 0.3, 1.0)
                A_new2 = A * np.sqrt(reduction2)
                a_new2 = ifft(A_new2).real
                a_new2 = np.maximum(a_new2, 0.0)
                s2 = float(np.sum(a_new2))
                if s2 > 1e-10:
                    a_new2 = a_new2 / s2
                    obj_new2 = 2.0 * n * float(np.max(compute_conv(a_new2)))
                    if obj_new2 < obj_old:
                        a = a_new2

    return a


def gaussian_smooth_step(a, sigma):
    """Apply Gaussian smoothing in the spatial domain."""
    n = len(a)
    x = np.arange(n)
    center = n / 2.0
    kernel = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    # Circular convolution with Gaussian
    N = 1
    while N < 2 * n:
        N <<= 1
    A = fft(a, N)
    K = fft(kernel / kernel.sum(), N)
    a_smooth = ifft(A * K).real
    a_smooth = np.maximum(a_smooth, 0.0)
    s = float(np.sum(a_smooth))
    if s > 1e-10:
        a_smooth = a_smooth / s
    return a_smooth


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a / a.sum()
    elif mode == 3:
        # Kaiser-like (approximate with Bessel)
        x = np.linspace(-1, 1, n)
        beta = 3.0
        a = np.exp(beta * np.sqrt(np.maximum(0, 1 - x ** 2)))
        a = a / a.sum()
    elif mode == 4:
        # Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 / (2 * 0.3 ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Triangular
        a = np.abs(1 - 2 * np.linspace(0, 1, n))
        a = a / a.sum()
    elif mode == 6:
        # Flat top
        x = np.linspace(0, 1, n)
        a = np.where(np.abs(x - 0.5) < 0.3, 1.0, 0.1)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def simulated_annealing(a, time_budget, seed=42, verbose=False):
    """
    Simulated annealing overlay: random smooth perturbations with
    acceptance criterion, followed by short CD refinement.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    a = normalize(a)
    best_a = a.copy()
    best_obj = obj_normalized(a)
    current_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    T_init = 0.02
    T_min = 0.001
    cd_budget_per_accept = min(3.0, time_budget / 50.0)

    it = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < cd_budget_per_accept + 1:
            break

        progress = min(1.0, it / 200.0)
        T = T_init * (T_min / T_init) ** progress

        # Generate smooth perturbation
        pert_scale = T * np.sqrt(n)
        # Use random Fourier features for smooth perturbation
        n_freqs = max(3, n // 20)
        freqs = rng.integers(1, n // 2 + 1, size=n_freqs)
        phases = rng.uniform(0, 2 * np.pi, size=n_freqs)
        amp = rng.uniform(0, 1, size=n_freqs)
        x = np.arange(n)
        pert = np.zeros(n)
        for f, p, a_ in zip(freqs, phases, amp):
            pert += a_ * np.sin(2 * np.pi * f * x / n + p)
        pert *= pert_scale / (np.std(pert) + 1e-10)

        a_trial = a * np.exp(pert)
        a_trial = np.maximum(a_trial, 0)
        s = float(np.sum(a_trial))
        if s < 1e-10:
            a_trial = a.copy()
        else:
            a_trial = a_trial / s

        obj_trial = obj_normalized(a_trial)

        # Accept if better or with probability exp(-delta/T)
        delta = obj_trial - current_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial
            current_obj = obj_trial
            if current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()

                # Short CD refinement after improvement
                if remaining > cd_budget_per_accept * 2:
                    a = coord_descent(a, time_budget=cd_budget_per_accept,
                                      seed=it, verbose=False)
                    obj_after_cd = obj_normalized(a)
                    current_obj = obj_after_cd
                    if obj_after_cd < best_obj:
                        best_obj = obj_after_cd
                        best_a = a.copy()

        it += 1

        if verbose and it % 50 == 0:
            print(f"    SA iter {it}: best={best_obj:.8f}, T={T:.5f}, t={time.monotonic()-start:.1f}s")

    return best_a


def block_descent(a, block_size=2, time_budget=60.0, seed=42):
    """
    Block coordinate descent: optimize groups of coordinates simultaneously
    using a coarse grid search + refinement.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    a = normalize(a)
    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        # Random block
        i0 = rng.integers(0, n - block_size)
        block = list(range(i0, i0 + block_size))

        # Get current values
        v = a[block].copy()
        S_o = float(np.sum(a)) - float(np.sum(v))

        # Try a grid of adjustments
        base_obj = obj_normalized(a)
        best_block_obj = base_obj
        best_vals = v.copy()

        # Scale based on current values
        scale = max(float(np.max(v)), 1.0 / n, 1e-4)

        # 3-level grid around current
        steps = [0.0, -0.5 * scale, -0.25 * scale, 0.25 * scale, 0.5 * scale, scale]
        num_combs = len(steps) ** block_size
        if num_combs > 500:
            # Too many combinations, sample
            for _ in range(100):
                idxs = [rng.integers(0, len(steps)) for _ in range(block_size)]
                trial = v.copy()
                for j, idx in enumerate(idxs):
                    trial[j] = max(0.0, v[j] + steps[idxs[j]])
                a[block] = trial
                obj_t = obj_normalized(a)
                if obj_t < best_block_obj:
                    best_block_obj = obj_t
                    best_vals = trial.copy()
        else:
            # Full grid
            import itertools
            for combo in itertools.product(range(len(steps)), repeat=block_size):
                trial = v.copy()
                for j, idx in enumerate(combo):
                    trial[j] = max(0.0, v[j] + steps[idx])
                a[block] = trial
                obj_t = obj_normalized(a)
                if obj_t < best_block_obj:
                    best_block_obj = obj_t
                    best_vals = trial.copy()

        a[block] = best_vals
        obj_now = obj_normalized(a)
        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines coordinate descent, spectral shaping, block descent,
    simulated annealing, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = obj_normalized(parent)
                best_seq = parent.copy()
                n_parent = len(parent)
                print(f"Parent (n={n_parent}): obj={best_score:.8f}")
            else:
                raise ValueError("sum too small")
    except Exception:
        best_seq = np.ones(n_parent) / n_parent
        best_score = obj_normalized(best_seq)
        print(f"Default init (n={n_parent}): obj={best_score:.8f}")

    # === Phase 1: Quick CD to establish baseline ===
    print(f"\n=== Phase 1: Initial Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.15, 150.0)
    if p1_budget > 10:
        a_cd = coord_descent(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_cd = obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase 1 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 CD: obj={obj_cd:.8f}")

    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current - 100, n_current + 100, n_current + 200,
        n_current + 400, n_current + 600,
        600, 900, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2500]
    # Deduplicate and limit
    seen = set()
    n_options_filt = []
    for no_ in n_options:
        if no_ not in seen and no_ != n_current:
            seen.add(no_)
            n_options_filt.append(no_)
    n_options = n_options_filt[:6]

    p2_total = min(remaining * 0.15, 150.0)
    if n_options and p2_total > 30:
        per_n = p2_total / len(n_options)
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(per_n, remaining * 0.3)
            if alloc < 15:
                break

            a_try = resample_sequence(best_seq, n_try)
            a_try = coord_descent(a_try, time_budget=alloc, verbose=False, seed=n_try)
            obj_try = obj_normalized(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_current = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 3: Main CD with spectral shaping ===
    print(f"\n=== Phase 3: Enhanced CD + Spectral Shaping ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.35, 450.0)

    if p3_budget > 30:
        a_work = best_seq.copy()
        # Interleave CD and spectral shaping
        n_cd_rounds = 5
        cd_per_round = p3_budget * 0.7 / n_cd_rounds
        spec_per_round = p3_budget * 0.1

        for rnd in range(n_cd_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # CD round
            cd_alloc = min(cd_per_round, remaining * 0.4)
            if cd_alloc > 5:
                a_work = coord_descent(a_work, time_budget=cd_alloc,
                                       verbose=False, seed=seed + rnd * 100)
                obj_work = obj_normalized(a_work)
                if obj_work < best_score:
                    best_score = obj_work
                    best_seq = a_work.copy()
                    print(f"  Round {rnd} CD: obj={best_score:.8f} **")

            # Spectral shaping
            remaining = deadline - time.monotonic()
            if remaining > 10:
                spec_alloc = min(spec_per_round, remaining * 0.1)
                if spec_alloc > 3:
                    a_spec = spectral_shaping(a_work, alpha=0.2, n_iters=2)
                    a_spec = coord_descent(a_spec, time_budget=min(10, remaining * 0.05),
                                           verbose=False, seed=seed + 5000 + rnd)
                    obj_spec = obj_normalized(a_spec)
                    if obj_spec < best_score:
                        best_score = obj_spec
                        best_seq = a_spec.copy()
                        a_work = a_spec.copy()
                        print(f"  Round {rnd} Spec: obj={best_score:.8f} **")

    # === Phase 4: Block descent ===
    print(f"\n=== Phase 4: Block Descent ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.1, 80.0)
    if p4_budget > 10:
        a_block = block_descent(best_seq, block_size=2, time_budget=p4_budget,
                                seed=seed + 777)
        obj_block = obj_normalized(a_block)
        if obj_block < best_score:
            best_score = obj_block
            best_seq = a_block.copy()
            print(f"  Block descent: obj={best_score:.8f} **")
        else:
            print(f"  Block descent: obj={obj_block:.8f}")

    # === Phase 5: Simulated Annealing ===
    print(f"\n=== Phase 5: Simulated Annealing ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.15, 150.0)
    if p5_budget > 20:
        a_sa = simulated_annealing(best_seq, time_budget=p5_budget,
                                   seed=seed + 3000, verbose=True)
        obj_sa = obj_normalized(a_sa)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  SA: obj={best_score:.8f} **")
        else:
            print(f"  SA: obj={obj_sa:.8f}")

    # === Phase 6: Gaussian smoothing + CD ===
    print(f"\n=== Phase 6: Smoothing + CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_try = len(best_seq)
        for sigma in [0.1, 0.2, 0.3]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_sm = gaussian_smooth_step(best_seq, sigma * n_try)
            a_sm = coord_descent(a_sm, time_budget=min(20, remaining * 0.2),
                                 verbose=False, seed=seed + 9000)
            obj_sm = obj_normalized(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Smooth sigma={sigma}: obj={best_score:.8f} **")

    # === Phase 7: Final CD polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.6, 200.0)
        a_final = coord_descent(best_seq, time_budget=alloc,
                                verbose=True, seed=seed + 9999)
        obj_final = obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # Final short CD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_last = coord_descent(best_seq, time_budget=min(remaining * 0.5, 50.0),
                               verbose=False, seed=seed + 12345)
        obj_last = obj_normalized(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
