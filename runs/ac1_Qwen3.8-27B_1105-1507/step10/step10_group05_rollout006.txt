"""
Optimizes step function coefficients using fast active-set LP with constraint generation,
HiGHS LP solver, parabolic line search, symmetry exploitation, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time
import math


def conv_same(a, b):
    """Linear convolution of a and b (both same length n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]


def conv_sq(a):
    """Linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    an = a / s
    c = conv_sq(an)
    return 2.0 * n * float(np.max(c))


def build_conv_row(a, k):
    """
    Build the k-th row of the convolution matrix.
    (A g)_k = conv(a, g)[k] = sum_j a[j] * g[k-j]
    where j ranges over valid indices.
    Returns a sparse-ish vector of length n (with zeros where a[j] doesn't contribute).
    """
    n = len(a)
    row = np.zeros(n)
    j_min = max(0, k - n + 1)
    j_max = min(k, n - 1)
    if j_min <= j_max:
        row[j_min:j_max+1] = a[j_min:j_max+1]
    return row


def solve_active_lp(a, active_idx):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for k in active_idx,
    sum(g) = 1, g >= 0.
    Returns (g, M) or None.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    n_act = len(active_idx)
    if n_act == 0:
        return np.ones(n) / n, 0.0
    
    # Build constraint matrix: n_act x n
    # Row i: build_conv_row(a_norm, active_idx[i])
    A_rows = []
    for k in active_idx:
        A_rows.append(build_conv_row(a_norm, k))
    A_act = np.array(A_rows)
    
    # LP: min M
    # Variables: [g_0, ..., g_{n-1}, M] (total n+1)
    # Constraints:
    #   A_act @ g - M <= 0  (n_act constraints)
    #   sum(g) = 1  (1 equality)
    #   g >= 0, M free
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    # Inequality: A_act @ g - M <= 0
    # [A_act | -1] @ [g; M] <= 0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = A_act
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_act)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        g = np.maximum(result.x[:n], 0)
        s = float(np.sum(g))
        if s < 1e-15:
            return None
        g = g / s
        M = float(result.x[n])
        return g, M
    return None


def find_active_set(conv_val, M, tol=0.01):
    """Find indices where conv_val is within tol of M (the max)."""
    return np.where(conv_val >= M - tol)[0]


def line_search_parabolic(a, g, f0, max_evals=10):
    """
    Parabolic line search: find t in [0,1] minimizing f((1-t)*a + t*g).
    Uses quadratic interpolation.
    """
    n = len(a)
    
    def f(t):
        x = (1.0 - t) * a + t * g
        x = np.maximum(x, 0)
        s = float(np.sum(x))
        if s < 1e-10:
            return np.inf
        x = x / s
        return compute_obj(x)
    
    # Initial points
    t_vals = [0.0, 0.3, 0.6, 1.0]
    f_vals = [f0, f(0.3), f(0.6), f(1.0)]
    
    # Find bracket
    best_i = np.argmin(f_vals)
    
    # Try to refine
    for _ in range(max_evals - 4):
        if best_i == 0:
            t1, t2 = 0.0, 0.15
        elif best_i == len(t_vals) - 1:
            t1, t2 = 0.85, 1.0
        else:
            t1 = t_vals[best_i - 1]
            t2 = t_vals[best_i + 1]
        
        # Quadratic interpolation
        if best_i > 0 and best_i < len(t_vals) - 1:
            t_a, t_b, t_c = t_vals[best_i-1], t_vals[best_i], t_vals[best_i+1]
            f_a, f_b, f_c = f_vals[best_i-1], f_vals[best_i], f_vals[best_i+1]
            # Fit quadratic and find minimum
            denom = (f_a - f_c) * (t_b - t_a) + (f_b - f_a) * (t_c - t_b) + (f_c - f_b) * (t_a - t_b)
            # Actually let me just use the standard formula
            # t* = t_b + 0.5 * ((t_b-t_a)^2*(f_b-f_c) - (t_b-t_c)^2*(f_b-f_a)) / ((t_b-t_a)*(f_b-f_c) - (t_b-t_c)*(f_b-f_a))
            num = (t_b - t_a)**2 * (f_b - f_c) - (t_b - t_c)**2 * (f_b - f_a)
            den = (t_b - t_a) * (f_b - f_c) - (t_b - t_c) * (f_b - f_a)
            if abs(den) < 1e-15:
                t_new = 0.5 * (t1 + t2)
            else:
                t_new = t_b + 0.5 * num / den
                t_new = max(0.0, min(1.0, t_new))
        else:
            t_new = 0.5 * (t1 + t2)
        
        f_new = f(t_new)
        t_vals.append(t_new)
        f_vals.append(f_new)
        best_i = np.argmin(f_vals)
    
    t_best = t_vals[best_i]
    f_best = f_vals[best_i]
    x_best = (1.0 - t_best) * a + t_best * g
    x_best = np.maximum(x_best, 0)
    s = float(np.sum(x_best))
    if s > 1e-10:
        x_best = x_best / s
    return x_best, f_best


def active_set_iteration(a, time_budget=300.0, verbose=False, seed=42):

    Main iteration: active-set LP with constraint generation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    it = 0
    best_it = 0
    active_tol = 0.005  # tolerance for active set detection (fraction of max)
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        
        # Compute current convolution
        conv_aa = conv_sq(a)
        M_current = float(np.max(conv_aa))
        
        # Find active set
        tol = M_current * active_tol
        active_idx = np.where(conv_aa >= M_current - tol)[0]
        
        # Solve active LP
        result = solve_active_lp(a, active_idx)
        if result is None:
            break
        g, M_lp = result
        
        # Check if full constraint is satisfied
        conv_ag = conv_same(a, g)
        max_conv_ag = float(np.max(conv_ag))
        
        # If violated, add violated constraints and re-solve
        if max_conv_ag > M_lp + 1e-8:
            # Find violated indices
            violated = np.where(conv_ag > M_lp + 1e-8)[0]
            new_active = np.union1d(active_idx, violated)
            result = solve_active_lp(a, new_active)
            if result is None:
                break
            g, M_lp = result
        
        it += 1
        
        # Line search
        f0 = best_obj
        a_new, obj_new = line_search_parabolic(a, g, f0)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            best_it = it
            if verbose and it % 20 == 0:
                print(f"    ASI it{it}: obj={best_obj:.8f}, |active|={len(active_idx)}, t={it-best_it}")
        else:
            # No improvement - try perturbation
            stalled = it - best_it
            if stalled > 20:
                strength = min(0.1, 0.01 * stalled)
                # Fourier perturbation
                pert = np.zeros(n)
                n_modes = min(3, n // 50)
                for k in range(1, n_modes + 1):
                    freq = 2 * np.pi * k / n
                    amp = strength * rng.uniform(0.5, 1.0) / k
                    phase = rng.uniform(0, 2 * np.pi)
                    pert += amp * np.cos(freq * np.arange(n) + phase)
                a_new = best_a * np.exp(pert)
                a_new = np.maximum(a_new, 0)
                s_new = float(np.sum(a_new))
                if s_new > 1e-10:
                    a = a_new / s_new
                if verbose:
                    print(f"    ASI it{it}: perturbed, strength={strength:.4f}")
            elif stalled > 50:
                break
        
        # Adapt active set tolerance
        if it > 50:
            active_tol = max(0.001, active_tol * 0.95)
    
    return best_a

def mirror_symmetrize(a):
    """Make sequence mirror-symmetric."""
    return (a + a[::-1]) / 2.0


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
