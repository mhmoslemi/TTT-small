def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):

    return A

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_optimize(a, time_budget=150.0, verbose=False, seed=77):

def smoothed_gd(a, time_budget=60.0, tau_init=15.0, tau_end=300.0):
    """Smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 5.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.8)
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]
        
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        lr = max(1e-12, lr * 0.999)
    
    return best_a

def subgradient_optimize(a, time_budget=60.0, K=10):
    """Targeted subgradient with top-K peaks."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        
        if max_pv < 1e-15:
            break
        
        # Weight peaks by exponential
        weights = np.exp((peak_vals - max_pv) * 10.0 / max_pv)
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]
        
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-14:
                    break
    
    return best_a

def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

def diverse_init(n, seed, method='random'):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    
    if method == 'uniform':
        return np.ones(n) / n
    elif method == 'random_smooth':
        x = rng.random(n)
        # Smooth with moving average
        k = max(1, n // 50)
        kernel = np.ones(k) / k
        x = np.convolve(x, kernel, mode='same')
        x = np.maximum(x, 0)
        return x / x.sum()
    elif method == 'bimodal':
        # Two peaks
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = max(3, n // 20)
        x = np.zeros(n)
        for i in range(n):
            x[i] = np.exp(-((i - c1) / w) ** 2) + np.exp(-((i - c2) / w) ** 2)
        return x / x.sum()
    elif method == 'triangular':
        x = np.abs(np.linspace(-1, 1, n))
        x = x ** rng.uniform(0.5, 3.0)
        return x / x.sum()
    elif method == 'alternating':
        x = np.ones(n)
        x[::2] *= rng.uniform(0.3, 0.7)
        return x / x.sum()
    else:
        return np.ones(n) / n

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_tr_sscp(a, r):
    """
    Solve trust region SOCP:
    min M s.t. C_old[k] + 2*(A_cross @ delta)[k] <= M, ||delta||_2 <= r,
               sum(a+delta)=1, a+delta >= 0
    """
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)

    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta

    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
        cp.norm(delta, 2) <= r,
    ]

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                d = np.asarray(delta.value).flatten()
                if np.isfinite(d).all():
                    return d, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. A @ g <= M, sum(g)=1, g>=0.
    The LP gives the "best" target sequence in the sense of minimizing
    max peak of conv(a, g).
    """
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value).flatten(), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, direction, current_obj):
    """Adaptive line search: try geometric sequence of step sizes."""
    n = len(a)
    best_t = 0.0
    best_obj = current_obj

    # Try a range of step sizes
    for t_val in [0.1, 0.2, 0.35, 0.5, 0.75, 1.0, 1.3, 1.7, 2.5, 3.5]:
        a_try = a + t_val * direction
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-15:
            best_obj = obj_try
            best_t = t_val

    return best_t, best_obj


def tr_sscp_polish(a, time_budget=300.0, r_init=0.25, verbose=False, seed=42):
    """
    TR-SOCP polishing with adaptive radius and LP-guided perturbation.
    Key improvement: when stuck, use the LP solution to guide perturbation
    rather than purely random noise.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    r = r_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        delta_dir, M_val = solve_tr_sscp(a, r)

        if delta_dir is None:
            g_lp, M_lp = solve_lp_direction(a)
            if g_lp is None:
                break
            delta_dir = g_lp - a
        else:
            norm_d = np.linalg.norm(delta_dir)
            if norm_d > r * (1 + 1e-6):
                delta_dir = delta_dir * (r / norm_d)

        if not np.isfinite(delta_dir).all():
            delta_dir = np.where(np.isfinite(delta_dir), delta_dir, 0.0)

        best_t, best_t_obj = line_search(a, delta_dir, best_obj)

        if best_t > 0 and best_t_obj < best_obj - 1e-14:
            a = a + best_t * delta_dir
            a = np.maximum(a, 0)
            s_n = a.sum()
            if s_n > 1e-10:
                a = a / s_n
            best_obj = best_t_obj
            best_a = a.copy()
            no_improve = 0
            r = min(r * 1.08, 1.8)
        else:
            no_improve += 1
            r *= 0.7
            if no_improve >= 4:
                if r < 0.025:
                    # LP-guided structured perturbation
                    g_lp, M_lp = solve_lp_direction(a)
                    if g_lp is not None:
                        # Move a bit towards LP solution + random noise
                        t_pert = 0.3 + 0.2 * rng.random()
                        a = (1 - t_pert) * best_a + t_pert * g_lp
                        a = a * np.exp(rng.normal(0, 0.008, n))
                    else:
                        a = best_a * np.exp(rng.normal(0, 0.012, n))
                    a = np.maximum(a, 0)
                    s_p = a.sum()
                    if s_p > 1e-10:
                        a = a / s_p
                    r = r_init
                    no_improve = 0
                    if it > 120:
                        break
                else:
                    r = max(r * 0.5, 0.025)

        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    TR it{it}: obj={best_obj:.8f}, r={r:.4f}, t={elapsed:.0f}s")

    return best_a


def lp_polish(a, time_budget=150.0, verbose=False, seed=77):
    """LP-based polishing with adaptive step and LP-guided restart."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        # Adaptive line search towards LP solution
        best_t = 0.0
        best_t_obj = best_obj
        for t_val in [0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0, 1.3, 1.8]:
            a_try = (1 - t_val) * best_a + t_val * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-15:
                best_t_obj = obj_try
                best_t = t_val

        if best_t > 0 and best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = (1 - best_t) * best_a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            s_ba = best_a.sum()
            if s_ba > 1e-10:
                best_a = best_a / s_ba
            a = best_a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                # LP-guided restart
                g_lp2, _ = solve_lp_direction(best_a)
                if g_lp2 is not None:
                    t_r = 0.5 + 0.3 * rng.random()
                    a = (1 - t_r) * best_a + t_r * g_lp2
                else:
                    a = best_a * np.exp(rng.normal(0, 0.008, n))
                a = a * np.exp(rng.normal(0, 0.005, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                no_improve = 0
                if it > 60:
                    break

        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def smoothed_gd(a, time_budget=60.0, tau_init=15.0, tau_end=400.0):
    """Smoothed gradient descent with increasing temperature."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 4.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr = max(1e-12, lr * 0.9992)

    return best_a


def subgradient_refine(a, time_budget=60.0, K=12):
    """Subgradient method with top-K peak exponential weighting."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break

        # Exponential weighting tilted towards max peak
        weights = np.exp((peak_vals - max_pv) * 12.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 150:
                lr *= 0.5
                v *= 0.2
                stall = 0
                if lr < 1e-14:
                    break

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def diverse_init(n, seed, method):
    """Generate diverse initializations for multi-start."""
    rng = np.random.default_rng(seed)
    if method == 'random_smooth':
        x = rng.random(n)
        k = max(3, n // 30)
        kernel = np.ones(k) / k
        x = np.convolve(x, kernel, mode='same')
        x = np.maximum(x, 1e-10)
        return x / x.sum()
    elif method == 'bimodal':
        c1 = n * rng.uniform(0.15, 0.35)
        c2 = n * rng.uniform(0.65, 0.85)
        w = max(3, n // 15)
        x = np.exp(-((np.arange(n) - c1) / w) ** 2) + np.exp(-((np.arange(n) - c2) / w) ** 2)
        return x / x.sum()
    elif method == 'power_law':
        x = (np.arange(1, n + 1)) ** -rng.uniform(0.3, 1.5)
        return x / x.sum()
    elif method == 'sine_perturbed':
        x = 1.0 + 0.5 * np.sin(2 * np.pi * rng.uniform(2, 8) * np.arange(n) / n)
        x = np.maximum(x, 0.1)
        return x / x.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple optimization strategies:
    1. Diverse initialization screening
    2. Smoothed GD warmup
    3. Blended TR-SOCP + LP optimization (main phase)
    4. Subgradient refinement
    5. Systematic multi-length exploration
    6. Perturbation + re-optimization
    7. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # === Load parent ===
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={n_target}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Diverse init screening ===
    print(f"\n=== Phase 1: Diverse Init Screening ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.05, 50.0)

    if p1_budget > 15:
        methods = ['random_smooth', 'bimodal', 'power_law', 'sine_perturbed']
        for i, m in enumerate(methods):
            if time.monotonic() > deadline - remaining * 0.6:
                break
            a_init = diverse_init(n_target, seed=2000 + i * 137, method=m)
            # Quick GD to evaluate
            a_init = smoothed_gd(a_init, time_budget=max(5, p1_budget / len(methods) - 5),
                                 tau_init=10.0, tau_end=200.0)
            obj_init = compute_obj(a_init)
            tag = " **" if obj_init < best_score else ""
            if obj_init < best_score:
                best_score = obj_init
                best_seq = a_init.copy()
            print(f"  {m}: obj={obj_init:.8f}{tag}")

    # === Phase 2: Smoothed GD warmup on best ===
    print(f"\n=== Phase 2: Smoothed GD Warmup ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.05, 45.0)
    if p2_budget > 10:
        a_sgd = smoothed_gd(best_seq, time_budget=p2_budget, tau_init=12.0, tau_end=350.0)
        obj_sgd = compute_obj(a_sgd)
        tag = " **" if obj_sgd < best_score else ""
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
        print(f"  GD: obj={obj_sgd:.8f}{tag}")

    # === Phase 3: Main blended TR-SOCP + LP optimization ===
    print(f"\n=== Phase 3: Blended TR-SOCP + LP (Main Phase) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.38, 420.0)

    if p3_budget > 40:
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        obj_work = compute_obj(a_work)
        start_p3 = time.monotonic()
        it = 0
        r_adaptive = 0.3

        while time.monotonic() - start_p3 < p3_budget:
            rem = p3_budget - (time.monotonic() - start_p3)
            if rem < 20:
                break

            # Alternate: 2x TR-SOCP then 1x LP
            if it % 3 != 2:
                sub_b = min(rem * 0.25, 70.0)
                a_work = tr_sscp_polish(a_work, time_budget=sub_b,
                                        r_init=r_adaptive, seed=500 + it * 7)
            else:
                sub_b = min(rem * 0.2, 50.0)
                a_work = lp_polish(a_work, time_budget=sub_b,
                                   seed=600 + it * 7)

            obj_work = compute_obj(a_work)
            if obj_work < best_score:
                best_score = obj_work
                best_seq = a_work.copy()
                # Adjust radius for next round
                r_adaptive = max(0.1, min(r_adaptive * 0.95, 0.5))
                print(f"  Blend it{it}: obj={best_score:.8f} **")
            else:
                r_adaptive = min(r_adaptive * 1.05, 0.6)

            it += 1

        print(f"  Phase3 done: {it} sub-iters, obj={obj_work:.8f}")

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient Refine ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.04, 40.0)
    if p4_budget > 8:
        a_sg = subgradient_refine(best_seq, time_budget=p4_budget, K=12)
        obj_sg = compute_obj(a_sg)
        tag = " **" if obj_sg < best_score else ""
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
        print(f"  SG: obj={obj_sg:.8f}{tag}")

    # === Phase 5: Systematic multi-length search ===
    print(f"\n=== Phase 5: Multi-Length Search ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Generate candidate lengths
    n_candidates = set()
    for delta in [-500, -400, -300, -200, -100, 100, 200, 300, 400, 500, 700, 1000]:
        n_try = n_current + delta
        if 300 <= n_try <= 2500 and n_try != n_current:
            n_candidates.add(n_try)
    for n_fixed in [400, 500, 600, 700, 900, 1000, 1200, 1500, 2000]:
        if 300 <= n_fixed <= 2500 and n_fixed != n_current:
            n_candidates.add(n_fixed)

    # Sort by proximity
    n_candidates = sorted(n_candidates, key=lambda x: abs(x - n_current))

    for n_try in n_candidates[:7]:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        print(f"  Trying n={n_try} (rem={remaining:.0f}s)")
        a_try = resample_seq(best_seq, n_try)

        # Quick GD
        alloc1 = min(remaining * 0.06, 20.0)
        if alloc1 > 6:
            a_try = smoothed_gd(a_try, time_budget=alloc1, tau_init=10.0, tau_end=200.0)

        # TR-SOCP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.12, 60.0)
        if alloc2 > 12:
            a_try = tr_sscp_polish(a_try, time_budget=alloc2, r_init=0.25)

        # Quick LP
        remaining = deadline - time.monotonic()
        alloc3 = min(remaining * 0.04, 15.0)
        if alloc3 > 5:
            a_try = lp_polish(a_try, time_budget=alloc3)

        obj_try = compute_obj(a_try)
        tag = " **" if obj_try < best_score else ""
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
        print(f"  n={n_try}: obj={obj_try:.8f}{tag}")

    # === Phase 6: Perturbation + re-optimization ===
    print(f"\n=== Phase 6: Perturbation + Re-optimize ===")
    remaining = deadline - time.monotonic()

    for r_idx in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        # Increasing perturbation scale
        scale = 0.004 + 0.005 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        # Optimize
        alloc = min(remaining * 0.35, 90.0)
        if alloc > 12:
            a_pert = tr_sscp_polish(a_pert, time_budget=alloc * 0.6,
                                    r_init=0.15, seed=4000 + r_idx * 31)
            alloc2 = alloc * 0.4
            if alloc2 > 6:
                a_pert = lp_polish(a_pert, time_budget=alloc2, seed=5000 + r_idx * 31)

            obj_pert = compute_obj(a_pert)
            tag = " **" if obj_pert < best_score else ""
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
            print(f"  Pert{r_idx} (s={scale:.4f}): obj={obj_pert:.8f}{tag}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        final_budget = min(remaining * 0.6, 120.0)
        a_final = best_seq.copy()
        a_final = a_final / a_final.sum()
        obj_final = compute_obj(a_final)
        start_f = time.monotonic()

        while time.monotonic() - start_f < final_budget:
            rem = final_budget - (time.monotonic() - start_f)
            if rem < 12:
                break
            a_final = tr_sscp_polish(a_final, time_budget=min(rem * 0.5, 40.0),
                                     r_init=0.1, seed=9000 + int(time.monotonic()))
            a_final = lp_polish(a_final, time_budget=min(rem * 0.3, 25.0),
                                seed=9500 + int(time.monotonic()))
            obj_check = compute_obj(a_final)
            if obj_check < best_score:
                best_score = obj_check
                best_seq = a_final.copy()
                print(f"  Final: obj={best_score:.8f} **")
                break

    # Final quick GD
    remaining = deadline - time.monotonic()
    if remaining > 12:
        a_last = smoothed_gd(best_seq, time_budget=min(remaining * 0.5, 25.0),
                             tau_init=50.0, tau_end=1000.0)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0)
    result = [float(min(1000.0, x)) for x in best_seq]

    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
