"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses accelerated subgradient descent with FFT-based active-set gradients,
multiplicative weight updates for diversification, and adaptive length exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    p = a / s
    b = compute_conv(p)
    return 2.0 * n * float(np.max(b))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho_idx = rho[cond][-1]
    theta = (cssv[rho_idx - 1] - 1.0) / rho_idx
    return np.maximum(v - theta, 0.0)


def active_set_gradient(p, b, top_k=10, eps=1e-3):
    """
    Compute subgradient of max(conv(p,p)) using active set.
    Returns gradient vector of length n.
    Uses top_k indices and those within eps of max.
    """
    n = len(p)
    max_b = float(np.max(b))
    
    # Active set: indices where b[k] >= max_b - eps
    active = np.where(b >= max_b - max(eps, max_b * 0.01))[0]
    # Also ensure top_k are included
    top_indices = np.argsort(b)[::-1][:top_k]
    active = np.unique(np.concatenate([active, top_indices]))
    
    if len(active) == 0:
        return np.zeros(n)
    
    # Gradient: (2/|A|) * sum_{k in A} p[(k-i) mod N]
    # This is a cross-correlation: (W * p_rev) where W is indicator of active set
    N = 1
    while N < 3 * n:
        N <<= 1
    
    W = np.zeros(N)
    W[active[:N]] = 1.0 / len(active)
    
    # cross-cor: IFFT(conj(FFT(W)) * FFT(p)) gives sum_k W[k] * p[k-i]
    # But we want sum_k W[k] * p[(k-i) mod N]
    # = sum_k W[k] * p_rev[i-k] where p_rev[j] = p[-j mod N]
    # = circular cross-correlation of W and p
    # = IFFT(FFT(W) * conj(FFT(p_padded)))
    
    Fp = fft(p, N)
    FW = fft(W, N)
    # We want g[i] = sum_k W[k] * p[(k-i) mod N]
    # = sum_k W[k] * p_rev[i-k] where p_rev[j] = p[N-j] (for j>0), p_rev[0]=p[0]
    # In Fourier: IFFT(FFT(W) * FFT(p_rev)) = IFFT(FFT(W) * conj(FFT(p)))
    grad = ifft(FW * np.conj(Fp)).real[:n]
    
    return 2.0 * grad

    Accelerated subgradient descent for min 2*n*max(conv(p,p)) s.t. p in simplex.
    Uses Nesterov momentum, adaptive learning rate, and active set gradient.
    """
    n = len(p0)
    p = p0.copy()
    p = project_simplex(p)
    
    best_p = p.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(p)))
    
    start = time.monotonic()
    v = np.zeros(n)  # momentum
    beta = 0.9
    
    # Initial learning rate
    b0 = compute_conv(p)
    max_b0 = float(np.max(b0))
    scale = max_b0 / n  # typical scale of gradient
    lr = 1.0 / (np.sqrt(n) * max(scale, 1e-6))
    
    it = 0
    best_it = 0
    patience = 0
    top_k = min(20, n // 5)
    
    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = p + beta * v
        y = project_simplex(y)
        
        # Compute gradient at y
        b = compute_conv(y)
        max_b = float(np.max(b))
        g = active_set_gradient(y, b, top_k=top_k)
        
        # Scale gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Momentum update
        v = beta * v - lr * g
        p_new = y + v
        p_new = project_simplex(p_new)
        
        # Evaluate
        obj = 2.0 * n * float(np.max(compute_conv(p_new)))
        
        if obj < best_obj:
            best_obj = obj
            best_p = p_new.copy()
            best_it = it
            patience = 0
        else:
            patience += 1
        
        # Adaptive learning rate
        if patience > 50:
            lr *= 0.9
            if patience > 200:

                p = best_p.copy()

def mwm_qp_solve(p0, q, inner_iters=30, lr_init=None):
    """
    Solve inner QP: min p^T M p s.t. p in simplex, M[i,j] = q[i+j].
    Uses projected gradient with FFT-based Hessian-vector products.
    """
    n = len(p0)
    p = p0.copy()
    p = project_simplex(p)
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    Fq = fft(q, N)
    
    if lr_init is None:
        # Estimate Lipschitz constant
        # ||M||_2 <= sum_k q[k] * ||C_k||_2
        # For simplicity, use a heuristic
        lr_init = 1.0 / (2.0 * n * np.max(q) + 1e-8)
    
    for it in range(inner_iters):
        # Compute grad = 2 * M @ p = 2 * IFFT(FFT(q) * conj(FFT(p)))[:n]
        Fp = fft(p, N)
        Mp = ifft(Fq * np.conj(Fp)).real[:n]
        grad = 2.0 * Mp
        
        # Gradient step
        p_new = p - lr_init * grad
        p_new = project_simplex(p_new)
        
        # Backtracking (simple version)
        # obj_new = p_new^T M p_new = sum_k q[k] * conv(p_new, p_new)[k]
        # For efficiency, skip backtracking and just use fixed schedule
        
        p = p_new
        lr_init *= 0.98
    
    return p


def mwm_outer_loop(p0, time_budget=100.0, verbose=False, seed=42):
    """
    Multiplicative Weight Updates outer loop.
    """
    n = len(p0)
    p = p0.copy()
    p = project_simplex(p)
    num_lags = 2 * n - 1
    
    best_p = p.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(p)))
    
    # Initialize q uniformly
    q = np.ones(num_lags) / num_lags
    
    start = time.monotonic()
    it = 0
    eta = 2.0 / np.sqrt(num_lags)  # learning rate
    inner_iters = 30
    
    while time.monotonic() - start < time_budget:
        # Solve inner QP
        p_new = mwm_qp_solve(p, q, inner_iters=inner_iters)
        
        # Compute b = conv(p_new, p_new)
        b = compute_conv(p_new)
        max_b = float(np.max(b))
        obj = 2.0 * n * max_b
        
        if obj < best_obj:
            best_obj = obj
            best_p = p_new.copy()
            if verbose and it % 10 == 0:
                print(f"    MWM it{it}: obj={best_obj:.8f}, max_b={max_b:.8f}, t={time.monotonic()-start:.0f}s")
        
        # Update q: q[k] *= exp(-eta * b[k]), normalize
        # Use log-sum-exp for numerical stability
        log_q = np.log(q + 1e-30) - eta * b
        log_q -= np.max(log_q)
        q = np.exp(log_q)
        q /= np.sum(q)
        
        p = p_new
        it += 1
        
        # Decay eta
        eta *= 0.999
        
        if verbose and it % 50 == 0:
            print(f"    MWM it{it}: obj={best_obj:.8f}, eta={eta:.4f}, t={time.monotonic()-start:.0f}s")
    
    return best_p


def fourier_perturbation(p, rng, strength=0.05):
    """Add low-frequency Fourier modes as structured perturbation."""
    n = len(p)
    pert = np.zeros(n)
    n_modes = min(10, n // 10)
    for k in range(1, n_modes + 1):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    p_new = p * np.exp(pert)
    p_new = np.maximum(p_new, 0)
    s = float(np.sum(p_new))
    if s < 1e-10:
        return p.copy()
    return p_new / s


def random_restart(n, rng):
    """Generate a random initial point on the simplex."""
    a = rng.random(n)
    return a / a.sum()


def cosine_init(n, rng):
    """Initialize with a cosine-based structure."""
    x = np.arange(n) / n
    mode = rng.integers(1, max(2, n // 50))
    p = 0.5 + 0.4 * np.cos(2 * np.pi * mode * x + rng.uniform(0, 2 * np.pi))
    p = np.maximum(p, 0.01)
    return p / p.sum()


def resample_sequence(p, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(p)
    if n == new_n:
        return p.copy()
    if n == 1:
        return np.full(new_n, p[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    p_new = np.interp(x_new, x_old, p)
    p_new = np.maximum(p_new, 0)
    s = float(np.sum(p_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return p_new / s


def lp_polish_step(p, time_budget=30.0, seed=42):
    """
    Single LP-based polish step: solve min max conv(p, g) s.t. g in simplex,
    then blend p toward g.
    """
    n = len(p)
    import cvxpy as cp
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = p[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Try several blend factors
                    best_blend = p.copy()
                    best_obj = compute_obj(p)
                    for t in [0.3, 0.5, 0.7, 1.0]:
                        blend = (1 - t) * p + t * g_val
                        blend = project_simplex(blend)
                        obj = compute_obj(blend)
                        if obj < best_obj:
                            best_obj = obj
                            best_blend = blend
                    return best_blend
        except Exception:
            continue
    
    return p
