"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set LP with dual-informed perturbations, adaptive line search,
and multi-n exploration. Key differences from previous: faster LP via constraint
reduction, smarter perturbations guided by LP dual variables, coordinate
refinement, and broader n exploration.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a, indices):
    """Build convolution matrix rows for given indices. A[i][j] = a[i-j] if valid."""
    n = len(a)
    k_idx = indices
    # For row k: A[k][j] = a[k-j] if 0 <= k-j < n
    # We build a sparse-like representation
    n_constraints = len(indices)
    # Use dense for simplicity (n_constraints x n)
    A = np.zeros((n_constraints, n))
    for i, k in enumerate(indices):
        # valid j: 0 <= j < n and 0 <= k-j < n
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[i, j_start:j_end+1] = a[j_end - (j_end - j_start):0:-1] if False else a[k - j_end:k - j_start]
    return A

    n = len(a)
    k_arr = np.asarray(indices)
    n_c = len(k_arr)
    A = np.zeros((n_c, n))
    for i in range(n_c):
        k = k_arr[i]
        j_start = max(0, int(k) - n + 1)
        j_end = min(n - 1, int(k))
        if j_start <= j_end:
            # A[i, j] = a[k - j] for j in [j_start, j_end]
            # k - j ranges from k - j_end to k - j_start
            A[i, j_start:j_end+1] = a[k - j_end:k - j_start + 1]
    return A


def solve_lp_active(a, top_k=300, time_per_lp=5.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for k in active set, sum(g)=1, g>=0.
    Uses only top_k constraints (those closest to the current max).
    Returns (g, M, active_indices, dual_y) or (None,)*4.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_a = compute_conv(a_norm)
    max_val = float(np.max(conv_a))
    
    # Select top_k constraints: those with highest conv_a values (most likely to be active)
    k_top = top_k
    if k_top >= 2 * n - 1:
        indices = np.arange(2 * n - 1)
    else:
        # Use top k by current conv value (near the max are most constraining)
        sorted_idx = np.argsort(conv_a)[::-1]
        indices = sorted_idx[:k_top]
    
    A = build_conv_matrix_fast(a_norm, indices)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    start_t = time.monotonic()
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if time.monotonic() - start_t > time_per_lp:
                break
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Get dual values
                    dual_y = None
                    try:
                        dual_y = constraints[0].dual_value
                    except:
                        dual_y = None
                    return g_val, float(M.value), indices, dual_y
        except Exception:
            continue
    return None, None, None, None


def line_search_adaptive(a, g, n_evals=40):
    """Adaptive line search: coarse grid then refine around best."""
    n = len(a)
    ts = np.linspace(0.002, 0.998, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = np.maximum((1 - t) * a + t * g, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try / s)
    
    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]
    
    # Parabolic refinement around best
    if 0 < best_idx < n_evals - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.002, 0.998))
            a_par = np.maximum((1 - t_par) * a + t_par * g, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                f_par = compute_obj(a_par / s_par)
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par
    
    return best_t, best_obj


def smoothed_gradient(a, tau):
    """Compute gradient of tau * log-sum-exp(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def gd_polish(a, time_budget=100.0, tau_init=100.0, tau_end=1500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed >= time_budget:
            break
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))
        
        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.1 * base_lr), 5.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adapt
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9995)
        
        if it - best_it > 8000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it
    
    if verbose:
        print(f"    GD: obj={best_obj:.8f}, iters={it}")
    return best_a


def lp_optimize(a, time_budget=300.0, verbose=True, top_k=250):
    """
    Main LP-driven optimization with active set and dual-informed perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    stall_count = 0
    iter_count = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break
        
        # Solve LP with active set
        g, M, active_idx, dual_y = solve_lp_active(a, top_k=top_k, time_per_lp=min(3.0, remaining*0.1))
        if g is None:
            break
        
        conv_curr = compute_conv(a)
        max_curr = float(np.max(conv_curr))
        
        # Line search
        best_t, obj_at_t = line_search_adaptive(a, g, n_evals=35)
        iter_count += 1
        
        improved = False
        if obj_at_t < best_obj - 1e-12:
            a_try = np.maximum((1 - best_t) * a + best_t * g, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                best_obj = obj_at_t
                best_a = a_try.copy()
                a = a_try
                stall_count = 0
                improved = True
                if verbose and (iter_count % 15 == 0 or iter_count <= 3):
                    elapsed = time.monotonic() - start
                    print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, "
                          f"max_conv={max_curr:.6f}, time={elapsed:.0f}s")
        else:
            stall_count += 1
            
            # Try smaller steps
            if stall_count <= 3:
                for t_small in [0.05, 0.02, 0.01, 0.005]:
                    a_small = np.maximum((1 - t_small) * a + t_small * g, 0)
                    s_small = float(np.sum(a_small))
                    if s_small > 1e-10:
                        a_small = a_small / s_small
                        obj_small = compute_obj(a_small)
                        if obj_small < best_obj - 1e-12:
                            best_obj = obj_small
                            best_a = a_small.copy()
                            a = a_small
                            stall_count = 0
                            improved = True
                            if verbose:
                                print(f"    LP it{iter_count}: small step t={t_small}, obj={best_obj:.8f}")
                            break
        
        if not improved:
            if verbose and iter_count % 25 == 0:
                print(f"    LP it{iter_count}: stalled obj={best_obj:.8f}")
            
            if stall_count >= 4:
                # Dual-informed perturbation
                if dual_y is not None:
                    # Focus perturbation on active constraints
                    y = np.zeros(2 * n - 1)
                    for i, idx in enumerate(active_idx):
                        if i < len(dual_y) and dual_y[i] is not None:
                            y[idx] = dual_y[i]
                    y = np.maximum(y, 0)
                    if y.sum() > 0:
                        y = y / y.sum()
                    else:
                        y = None
                else:
                    y = None
                
                # Subgradient-based perturbation
                tau_p = 500.0
                conv_p = compute_conv(a)
                max_p = float(np.max(conv_p))
                exp_p = np.exp(tau_p * (conv_p - max_p))
                p = exp_p / exp_p.sum()
                
                N_fft = 1
                while N_fft < 3 * n:
                    N_fft <<= 1
                P = fft(p, N_fft)
                Af = fft(a, N_fft)
                grad = 2.0 * ifft(P * np.conj(Af)).real[:n]
                
                # Move against gradient
                lr_pert = 0.03 / (float(np.linalg.norm(grad)) + 1e-10)
                a = a - lr_pert * grad
                a = np.maximum(a, 0)
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                else:
                    a = best_a.copy()
                
                # Add random component
                noise_scale = 0.003 + 0.003 * min(stall_count, 10) / 10.0
                a = a + noise_scale * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                
                stall_count = 0
                
                if iter_count > 200:
                    if verbose:
                        print(f"    LP it{iter_count}: terminating after many iterations")
                    break
    
    if verbose:
        print(f"    LP done: obj={best_obj:.8f}, iters={iter_count}, time={time.monotonic()-start:.0f}s")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.04):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 7):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def coordinate_refine(a, time_budget=30.0):
    """Simple coordinate descent: try adjusting each coordinate slightly."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99)
    
    while time.monotonic() - start < time_budget:
        # Try random subset of coordinates
        n_try = min(n, 50)
        idxs = rng.choice(n, size=n_try, replace=False)
        improved = False
        for i in idxs:
            old_val = a[i]
            if old_val < 1e-12:
                # Try adding small value
                for delta in [0.01/n, 0.005/n]:
                    a[i] = delta
                    s_new = float(np.sum(a))
                    if s_new > 1e-10:
                        obj = compute_obj(a / s_new)
                        if obj < best_obj - 1e-13:
                            best_obj = obj
                            best_a = a.copy()
                            improved = True
                            break
            else:
                # Try scaling
                for factor in [0.8, 1.2, 0.5, 1.5]:
                    a[i] = old_val * factor
                    s_new = float(np.sum(a))
                    if s_new > 1e-10:
                        obj = compute_obj(a / s_new)
                        if obj < best_obj - 1e-13:
                            best_obj = obj
                            best_a = a.copy()
                            improved = True
                            break
                    a[i] = old_val
                # Restore if not improved
                a[i] = old_val
            if not improved:
                a[i] = old_val
        if not improved and time.monotonic() - start > time_budget * 0.7:
            # Add noise and continue
            a = best_a + 0.002 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a
