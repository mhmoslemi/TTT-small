def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:

    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_peak(a, k_star):
    """Gradient of (a*a)_{k_star} w.r.t. a. Returns 2*a[k_star-i] where valid."""
    n = len(a)
    grad = np.zeros(n)
    idx = np.arange(n)
    j = k_star - idx
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]
    return grad


def grad_softmax_conv(a, tau, fft_size):
    """Gradient of soft-max convolution objective using FFT correlation."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_c = np.exp((conv - max_c) / tau)
    w = exp_c / float(np.sum(exp_c))
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(np.conj(W) * A, fft_size)
    grad = 2.0 * corr[:n]
    return grad, w

        y = np.maximum(y, 0.0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        grad, _ = grad_softmax_conv(y, tau, fft_size)
        x_new = project_simplex(y - (1.0 / L) * grad)

        obj_new = compute_obj(x_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = x_new.copy()

        x_prev = x
        x = x_new
        t_k = t_new
        it += 1

        if verbose and it % 20000 == 0:
            elapsed = time.monotonic() - start
            print(f"    FISTA it{it}: obj={best_obj:.8f}, tau={tau:.6f}, t={elapsed:.1f}s")

    return best_a


def away_fw(a, time_budget, max_iters=500000, verbose=False):
    """Away-step Frank-Wolfe on the simplex for min max_k (a*a)_k."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    support = set(np.nonzero(a > 1e-10)[0].tolist())
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        g = grad_peak(a, k_star)

        s_idx = int(np.argmin(g))

        if s_idx not in support:
            v = np.zeros(n)
            v[s_idx] = 1.0
            support.add(s_idx)
        else:
            if len(support) > 1:
                W_idx = max(support, key=lambda j: g[j])
                if g[W_idx] > 0.01:
                    v = np.zeros(n)
                    v[W_idx] = 1.0
                    a_away = a - v
                    a_away = np.maximum(a_away, 0.0)
                    s_away = float(np.sum(a_away))
                    if s_away > 1e-10:
                        a_away = a_away / s_away
                        if float(np.dot(g, a_away)) < float(np.dot(g, a)) - 1e-10:
                            support.discard(W_idx)
                            a = a_away
                            obj_away = compute_obj(a)
                            if obj_away < best_obj:
                                best_obj = obj_away
                                best_a = a.copy()
                            it += 1
                            continue
            v = np.zeros(n)
            v[s_idx] = 1.0

        # Line search for step size
        a_s = (1.0 - 0.5) * a + 0.5 * v
        obj_s = compute_obj(a_s)
        if obj_s < best_obj:
            a = a_s
            best_obj = obj_s
            best_a = a.copy()
            stall = 0
        else:
            a_s2 = (1.0 - 0.25) * a + 0.25 * v
            obj_s2 = compute_obj(a_s2)
            if obj_s2 < best_obj:
                a = a_s2
                best_obj = obj_s2
                best_a = a.copy()
                stall = 0
            else:
                a_s3 = (1.0 - 0.1) * a + 0.1 * v
                obj_s3 = compute_obj(a_s3)
                if obj_s3 < best_obj:
                    a = a_s3
                    best_obj = obj_s3
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
                    if stall > 100:
                        break

        it += 1
        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - start
            print(f"    FW it{it}: obj={best_obj:.8f}, |S|={len(support)}, t={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget, max_iters=50, verbose=False):
    """LP-based polishing with line search."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(42)

    it = 0
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Try multiple blend factors
        best_t = 0.0
        best_obj_t = best_obj
        for t_try in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t_try) * a + t_try * g_lp
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t_try

        if best_t > 0 and best_obj_t < best_obj - 1e-14:
            a = (1.0 - best_t) * a + best_t * g_lp
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_obj = compute_obj(a)
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                noise = rng.normal(0, 0.01 * stall, n)
                a = best_a + noise
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                stall = 0
                if stall > 6:
                    break

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={elapsed:.1f}s")

    return best_a


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2.0 * x - 1.0) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.01, 0.99, n)
    a = 1.0 / np.sqrt(np.maximum(0.01, 1.0 - (2.0 * x - 1.0) ** 2))
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2.0 * x - 1.0) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2.0 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2.0 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4.0 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7.0
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5.0 * (2.0 * x - 1.0) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2.0 * x - 1.0)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(3.0 * (2.0 * x - 1.0) - 0.5)
    elif mode == 7:
        a = (2.0 * x - 1.0) ** 6 + 0.02
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def fast_gradient_refine(a, time_budget, verbose=False):
    """Fast projected gradient with adaptive step size and periodic FISTA."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.05
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget and it < 500000:
        conv = compute_conv(a)
        # Use top-3 peaks for smoother gradient
        top_k = min(3, len(conv))
        top_indices = np.argsort(conv)[-top_k:]
        grad = np.zeros(n)
        for k_i in top_indices:
            grad += grad_peak(a, int(k_i))
        grad /= top_k

        a_new = project_simplex(a - lr * grad)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.05, 0.5)
        else:
            stall += 1
            lr *= 0.9
            if lr < 0.001:
                break

        it += 1
        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - start
            print(f"    GradRefine it{it}: obj={best_obj:.8f}, lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase hybrid optimization combining FISTA with soft-max,
    away-step Frank-Wolfe, gradient refinement, and LP polishing.
    Explores multiple sequence lengths and diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        best_seq = u_shaped_init(800, sharpness=3.0)
        best_score = compute_obj(best_seq)
        n_current = 800
        print(f"Default U-shaped init (n={n_current}): obj={best_score:.8f}")

    # === Phase 1: Quick n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print("\n=== Phase 1: n Exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            max(400, n_best - 200), max(400, n_best - 100),
            n_best + 100, n_best + 200, n_best + 400,
            500, 600, 700, 900, 1000, 1100, 1200, 1400
        ]))
        n_options = [n for n in n_options if 400 <= n <= 1600]
        n_options = list(dict.fromkeys(n_options))[:6]
        print(f"  Testing n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            # Interpolate best_seq to new length
            n_old = len(best_seq)
            x_old = np.linspace(0, 1, n_old)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 0.01:
                a_try = a_try / s_try
            obj_try = compute_obj(a_try)

            # Quick FISTA refinement
            alloc = min(15.0, remaining * 0.3)
            if alloc > 5:
                a_try = fista_softmax(a_try, time_budget=alloc, tau_init=0.005, tau_min=0.002)
                obj_try = compute_obj(a_try)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 2: FISTA with soft-max (main workhorse) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 2: FISTA Soft-Max ===")
        n_cur = len(best_seq)
        alloc = min(remaining * 0.35, 400.0)
        a_fista = fista_softmax(best_seq, time_budget=alloc,
                                tau_init=0.008, tau_min=0.0003, verbose=True)
        obj_fista = compute_obj(a_fista)
        if obj_fista < best_score:
            best_score = obj_fista
            best_seq = a_fista.copy()
            print(f"  FISTA: obj={best_score:.8f} **")
        else:
            print(f"  FISTA: obj={obj_fista:.8f}")

    # === Phase 3: Away-step Frank-Wolfe ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print("\n=== Phase 3: Away-step FW ===")
        alloc = min(remaining * 0.2, 200.0)
        a_fw = away_fw(best_seq, time_budget=alloc, max_iters=300000, verbose=True)
        obj_fw = compute_obj(a_fw)
        if obj_fw < best_score:
            best_score = obj_fw
            best_seq = a_fw.copy()
            print(f"  FW: obj={best_score:.8f} **")
        else:
            print(f"  FW: obj={obj_fw:.8f}")

    # === Phase 4: Fast gradient refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 4: Gradient Refinement ===")
        alloc = min(remaining * 0.2, 150.0)
        a_gr = fast_gradient_refine(best_seq, time_budget=alloc, verbose=True)
        obj_gr = compute_obj(a_gr)
        if obj_gr < best_score:
            best_score = obj_gr
            best_seq = a_gr.copy()
            print(f"  GradRefine: obj={best_score:.8f} **")
        else:
            print(f"  GradRefine: obj={obj_gr:.8f}")

    # === Phase 5: LP Polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 5: LP Polishing ===")
        alloc = min(remaining * 0.35, 250.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, max_iters=60, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 6: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 6: Diverse Restarts ===")
        n_cur = len(best_seq)

        for r in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            mode = r % 8
            a_r = diverse_init(n_cur, rng, mode=mode)
            blend = 0.3
            a_r = (1.0 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            # Quick FISTA
            alloc1 = min(20.0, remaining * 0.25)
            if alloc1 > 5:
                a_r = fista_softmax(a_r, time_budget=alloc1, tau_init=0.01, tau_min=0.003)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 7: Final LP Polish ===")
        alloc = min(remaining - 15, 150.0)
        if alloc > 15:
            a_final = lp_polish(best_seq, time_budget=alloc, max_iters=50, verbose=True)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final LP: obj={best_score:.8f} **")
            else:
                print(f"  Final LP: obj={obj_final:.8f}")

    # === Phase 8: One more FISTA pass with low tau ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 8: Final FISTA (low tau) ===")
        alloc = min(remaining - 15, 80.0)
        if alloc > 10:
            a_last = fista_softmax(best_seq, time_budget=alloc,
                                   tau_init=0.002, tau_min=0.0002)
            obj_last = compute_obj(a_last)
            if obj_last < best_score:
                best_score = obj_last
                best_seq = a_last.copy()
                print(f"  Last FISTA: obj={best_score:.8f} **")
            else:
                print(f"  Last FISTA: obj={obj_last:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
