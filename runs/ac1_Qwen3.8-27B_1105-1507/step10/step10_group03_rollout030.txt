import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build the convolution matrix A where (A@g)[k] = sum_j a[k-j]*g[j]."""

    X_sub: (m, n) array of iterates

    if m == 1:

    sa = np.sum(alpha)

            break

        # Add to memory
        X_mem.append(a.copy())
        TX_mem.append(g_lp.copy())

        # Trim memory
        max_mem = memory + 2
        if len(X_mem) > max_mem:
            X_mem = X_mem[-max_mem:]
            TX_mem = TX_mem[-max_mem:]

        # Compute Anderson combination
        if len(X_mem) >= 2:
            m_eff = min(memory, len(X_mem))
            X_sub = np.array(X_mem[-m_eff:])
            TX_sub = np.array(TX_mem[-m_eff:])
            alpha = anderson_coefficients(X_sub, TX_sub)
            a_new = np.zeros(n)
            for i in range(m_eff):
                a_new += alpha[i] * TX_sub[i]
        else:
            a_new = g_lp.copy()

        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            a_new = a.copy()
            sn = 1.0
        a_new = a_new / sn

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            if verbose and it % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP-A it{it}: obj={best_obj:.10f}, lp_time={lp_time:.3f}s, t={elapsed:.1f}s")
        else:
            stall += 1
            # Try blending with best
            if stall <= 6:
                improved = False
                for t in [0.2, 0.4, 0.6, 0.8]:
                    blend = (1 - t) * best_a + t * a_new
                    blend = np.maximum(blend, 0)
                    sb = float(np.sum(blend))
                    if sb > 1e-10:
                        blend = blend / sb
                        ob = compute_obj(blend)
                        if ob < best_obj - 1e-13:
                            best_obj = ob
                            best_a = blend.copy()
                            improved = True
                            break
                if improved:
                    stall = 0

            if stall > max_stall:
                # Perturb to escape local minimum
                scale = 0.005 * (1 + 0.3 * min(stall, 20))
                a = best_a + rng.normal(0, scale, n)
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a < 1e-10:
                    a = best_a.copy()
                else:
                    a = a / s_a
                X_mem = []
                TX_mem = []
                stall = 0
                if verbose:
                    print(f"    LP-A it{it}: perturbed (stall={stall}), obj={best_obj:.10f}")

                if time.monotonic() - start > time_budget * 0.95:
                    break

        a = a_new
        it += 1

    if verbose:
        elapsed = time.monotonic() - start
        print(f"    LP-A done: {it} iters, obj={best_obj:.10f}, t={elapsed:.1f}s")

    return best_a


def smoothed_grad(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Returns (gradient of length n, exact_max of conv).
    """
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = conv - exact_max
    # Clip for numerical stability
    shifted = np.clip(shifted, -50 / tau, 0)
    exp_vals = np.exp(tau * shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def gradient_polish(a, time_budget, tau_levels=None, verbose=False, beta=0.8, lr_scale=2.0):
    """Nesterov accelerated gradient descent with staircase tau."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 20, 80, 300, 1000, 3000]

    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        it_level = 0
        max_iters = 3000

        while time.monotonic() - start < time_budget and it_level < max_iters:
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            it_level += 1
            lr = max(1e-8, lr * 0.99995)

        if verbose:
            print(f"    GD tau={tau}: obj={best_obj:.10f}")

    return best_a


def subgradient_polish(a, time_budget, verbose=False, seed=42):
    """
    Projected subgradient method with Polyak step size.
    Lower bound: 2n/(2n-1) for normalized sequences.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    f_lower = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Running average for subgradient averaging
    a_avg = a.copy()
    count = 0
    momentum = np.zeros(n)
    beta = 0.9

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.1:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        f_current = 2.0 * n * float(conv[k_star])

        # Subgradient: grad f_{k*}(a)[j] = 2*a[k*-j] if valid
        g = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                g[j] = 2.0 * a[idx]

        # Polyak step size
        gap = f_current - f_lower
        g_norm2 = float(np.sum(g * g))
        if g_norm2 < 1e-30:
            break
        eta = gap / g_norm2
        eta = min(eta, 1.0 / np.sqrt(count + 1))  # cap step size

        # Momentum
        momentum = beta * momentum + (1 - beta) * g
        m_norm = float(np.linalg.norm(momentum))
        if m_norm > 1e-15:
            momentum = momentum / m_norm * np.linalg.norm(g)

        a_new = a - eta * momentum
        a_new = project_simplex(a_new)

        # Averaging
        a_avg = (count * a_avg + a_new) / (count + 1)
        count += 1

        obj_new = compute_obj(a_new)
        obj_avg = compute_obj(a_avg)

        if obj_avg < best_obj - 1e-14:
            best_obj = obj_avg
            best_a = a_avg.copy()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

        if verbose and count % 100 == 0:
            print(f"    SubGrad it{count}: obj={best_obj:.10f}")

    return best_a


def peak_targeted_reduction(a, time_budget, verbose=False, seed=42):
    """
    Identify top peaks in autocorrelation and try to reduce them via
    targeted coordinate adjustments.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        conv = compute_conv(best_a)
        # Find top peaks
        top_k = np.argsort(conv)[::-1][:5]
        current_max = float(np.max(conv))

        # For each peak k, the gradient direction to reduce it is -2*a[k-j] (reversed)
        # Try reducing the peak by moving mass from contributors to non-contributors
        improved = False
        for k_peak in top_k:
            if time.monotonic() - start > time_budget - 0.5:
                break
            # Contributors to conv[k_peak] are pairs (i, k_peak-i)
            contributors = []
            for i in range(n):
                j = k_peak - i
                if 0 <= j < n:
                    contributors.append(i)

            if len(contributors) == 0:
                continue

            # Try redistributing: reduce a[i] for large a[i] among contributors,
            # and increase a[j] for small a[j] among non-contributors
            non_contribs = [i for i in range(n) if i not in set(contributors)]
            if not non_contribs:
                continue

            # Pick the largest contributor and smallest non-contributor
            contrib_vals = [(best_a[i], i) for i in contributors]
            contrib_vals.sort(reverse=True)
            non_contrib_vals = [(best_a[i], i) for i in non_contribs]
            non_contrib_vals.sort()

            for _ in range(3):
                if not contrib_vals or not non_contrib_vals:
                    break
                src_val, src_idx = contrib_vals[0]
                dst_val, dst_idx = non_contrib_vals[0]

                if src_val < 1e-8:
                    break

                # Try moving a small amount
                for frac in [0.01, 0.03, 0.05]:
                    move = src_val * frac
                    a_try = best_a.copy()
                    a_try[src_idx] -= move
                    a_try[dst_idx] += move
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break

            if improved:
                break

        it += 1
        if not improved and it > 20:
            # Perturb
            a_try = best_a + rng.normal(0, 0.003, n)
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try
            break

        if verbose and it % 10 == 0:
            print(f"    Peak-Red it{it}: obj={best_obj:.10f}")

    return best_a


def make_init(n, mode, rng=None):
    """Create diverse initial sequences of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.0001))
        return a / a.sum()
    elif mode == 3:
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 4:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, np.pi, n)
        a = np.maximum(0.5 + 0.5 * np.cos(x), 0.001)
        return a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [2, 3, 5]:
            a += 0.2 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 4))
        return a / a.sum()
    elif mode == 8:
        x = np.linspace(0, 1, n)
        a = np.where((x > 0.2) & (x < 0.8), 1.0,
                     np.maximum(0.5 - np.abs(x - 0.5), 0.01))
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s
