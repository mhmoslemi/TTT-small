    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def nesterov_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9, lr_scale=2.0):
    """Nesterov accelerated gradient descent with staircase tau schedule."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 20, 80, 300, 1000, 3000]

    it = 0
    best_it = 0
    stall_patience = 3000

    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0
        max_iters_per_level = 5000

        while time.monotonic() - start < time_budget:
            if tau_iters >= max_iters_per_level:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it

            a = a_new
            it += 1
            tau_iters += 1
            lr = max(1e-7, lr * 0.99999)

            if it - best_it > stall_patience:
                a = best_a.copy()
                v = np.zeros(n)
                lr = lr_scale / np.sqrt(n)
                best_it = it

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    tau={tau:5d}: obj={best_obj:.10f}, iters={tau_iters}, t={elapsed:.0f}s")

    return best_a


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a, g) <= M, g >= 0, sum(g) = 1.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    if n > 3000:
        return None, None

    # Build convolution matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.CBC, cp.GLPK]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue

    return None, None


def structural_mutate(a, rng, strength=0.3):
    """
    Apply a structural mutation to the sequence.
    Different from smooth perturbation - makes discrete structural changes.
    """
    n = len(a)
    m = a.copy()
    mutation_type = rng.integers(0, 5)

    if mutation_type == 0:
        # Block swap: swap two random contiguous blocks
        size1 = rng.integers(max(2, n // 20), max(3, n // 5))
        size2 = rng.integers(max(2, n // 20), max(3, n // 5))
        pos1 = rng.integers(0, max(1, n - size1))
        pos2 = rng.integers(0, max(1, n - size2))
        if pos1 + size1 <= pos2 or pos2 + size2 <= pos1:
            block1 = m[pos1:pos1 + size1].copy()
            block2 = m[pos2:pos2 + size2].copy()
            m[pos1:pos1 + size1] = block2[:size1]
            m[pos2:pos2 + size2] = block1[:size2]
    elif mutation_type == 1:
        # Chunk zeroing: zero out a random chunk and redistribute
        chunk_size = rng.integers(max(5, n // 10), max(10, n // 3))
        start_idx = rng.integers(0, max(1, n - chunk_size))
        removed_mass = float(np.sum(m[start_idx:start_idx + chunk_size]))
        m[start_idx:start_idx + chunk_size] = 0
        m += removed_mass / n
    elif mutation_type == 2:
        # Segment reversal
        seg_len = rng.integers(max(5, n // 8), max(10, n // 2))
        start_idx = rng.integers(0, max(1, n - seg_len))
        m[start_idx:start_idx + seg_len] = m[start_idx:start_idx + seg_len][::-1]
    elif mutation_type == 3:
        # Binary-like perturbation: randomly set some positions to 0
        n_flip = rng.integers(n // 10, n // 3)
        indices = rng.choice(n, size=n_flip, replace=False)
        removed = float(np.sum(m[indices]))
        m[indices] = 0
        m += removed / n
    else:
        # Mirror perturbation: blend with reversed sequence
        t = rng.uniform(0.1, 0.4)
        m = (1 - t) * m + t * m[::-1]

    m = np.maximum(m, 0)
    s = float(np.sum(m))
    if s < 1e-10:
        return a.copy()
    return m / s


def lp_polish_advanced(a, time_budget, verbose=False, seed=42):
    """
    Advanced LP polishing with structural mutations, adaptive line search,
    and diversity injection to escape local optima.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall_count = 0
    max_stall = 8
    last_improvement = 0
    consecutive_no_improve = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # Solve LP
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            # LP failed, try structural mutation and GD
            a = structural_mutate(a, rng)
            a = nesterov_gd(a, min(15.0, remaining * 0.2),
                           tau_levels=[50, 200, 800], verbose=False)
            obj_a = compute_obj(a)
            if obj_a < best_obj:
                best_obj = obj_a
                best_a = a.copy()
            consecutive_no_improve += 1
            if consecutive_no_improve > 5:
                break
            continue

        # Line search over blending factors
        t_candidates = [0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5,
                        0.6, 0.7, 0.8, 0.9, 1.0]
        best_t_obj = best_obj
        best_t_a = a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_t_a.copy()
            stall_count = 0
            consecutive_no_improve = 0
            last_improvement = time.monotonic()
            if verbose:
                print(f"    LP adv: obj={best_obj:.10f}")
        else:
            stall_count += 1
            consecutive_no_improve += 1

            if stall_count <= max_stall:
                # Try structural mutation
                a = structural_mutate(a, rng)
                # Quick GD to settle
                a = nesterov_gd(a, min(10.0, remaining * 0.1),
                               tau_levels=[30, 150], verbose=False)
                obj_a = compute_obj(a)
                if obj_a < best_obj:
                    best_obj = obj_a
                    best_a = a.copy()
                    a = best_a.copy()
                    stall_count = 0
                    consecutive_no_improve = 0
                    if verbose:
                        print(f"    LP+mutate: obj={best_obj:.10f}")
            else:
                # Try more aggressive mutation
                a = structural_mutate(a, rng, strength=0.5)
                a = structural_mutate(a, rng, strength=0.5)
                a = nesterov_gd(a, min(15.0, remaining * 0.15),
                               tau_levels=[20, 100, 500], verbose=False)
                obj_a = compute_obj(a)
                if obj_a < best_obj:
                    best_obj = obj_a
                    best_a = a.copy()
                    a = best_a.copy()
                    stall_count = 0
                    consecutive_no_improve = 0
                    if verbose:
                        print(f"    LP+aggressive: obj={best_obj:.10f}")
                    else:
                        break

        # Periodic GD refinement
        if consecutive_no_improve == 3:
            alloc = min(10.0, remaining * 0.1)
            a_refined = nesterov_gd(best_a, alloc,
                                    tau_levels=[100, 500, 2000], verbose=False)
            obj_ref = compute_obj(a_refined)
            if obj_ref < best_obj:
                best_obj = obj_ref
                best_a = a_refined.copy()
                a = best_a.copy()
                consecutive_no_improve = 0
                if verbose:
                    print(f"    GD refine: obj={best_obj:.10f}")

    return best_a


def subspace_lp_optimize(a, time_budget, verbose=False, seed=42):
    """
    Optimize subsets of coordinates using LP.
    Fix most coordinates, optimize a sliding window.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    window_size = min(100, n // 6)
    step = max(1, window_size // 4)

    improved = True
    while improved and time.monotonic() - start < time_budget:
        improved = False
        for start_idx in range(0, n - window_size + 1, step):
            remaining = time_budget - (time.monotonic() - start)
            if remaining < 5:
                break

            end_idx = start_idx + window_size
            # Fix all except the window
            fixed = a.copy()
            fixed[start_idx:end_idx] = 0

            # For the window, solve: minimize max_k conv(fixed + w, fixed + w)
            # which is complex. Instead, use a simple approach:
            # Solve LP for the window coefficients given the fixed part.
            # conv(fixed+w, fixed+w) = conv(fixed,fixed) + conv(fixed,w) + conv(w,fixed) + conv(w,w)
            # For small window, conv(w,w) is small, so approximate:
            # minimize max_k [conv(fixed,fixed)[k] + conv(fixed,w)[k] + conv(w,fixed)[k]]
            # subject to w >= 0, sum(w) = sum(a[start_idx:end_idx])

            target_sum = float(np.sum(a[start_idx:end_idx]))
            if target_sum < 1e-10:
                continue

            wf = len(fixed)
            ww = window_size
            # Build the effective constraint matrix for the window
            # conv(fixed+w, fixed+w)[k] for the linear part in w:
            # = sum_j (a_fixed[k-j] + a_w[k-j]) * a_w[j] + sum_j a_fixed[k-j]*a_w[j]
            # Linear terms: A_fixed @ w + A_w @ w where A_w is small
            # For simplicity, just do line search on the window

            # Simple approach: try random rescalings of the window
            window = a[start_idx:end_idx].copy()
            best_window = window.copy()
            best_window_obj = best_obj

            for _ in range(20):
                # Perturb the window
                w_new = window * rng.uniform(0.5, 1.5, ww)
                w_new = np.maximum(w_new, 0)
                sw = float(np.sum(w_new))
                if sw < 1e-10:
                    w_new = window.copy()
                    sw = target_sum
                w_new = w_new * (target_sum / sw)

                a_try = a.copy()
                a_try[start_idx:end_idx] = w_new
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)

                if obj_try < best_window_obj:
                    best_window_obj = obj_try
                    best_window = w_new.copy()

            if best_window_obj < best_obj - 1e-14:
                best_obj = best_window_obj
                best_a = a.copy()
                best_a[start_idx:end_idx] = best_window
                a = best_a.copy()
                improved = True
                if verbose:
                    print(f"    Subspace: obj={best_obj:.10f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_structural_init(n, mode, rng=None):
    """Create structural initializations with more contrast than smooth ones."""
    if rng is None:
        rng = np.random.default_rng(0)

    if mode == 0:
        # Uniform
        return np.ones(n) / n
    elif mode == 1:
        # Two top-hats with gap
        k = n // 3
        a = np.zeros(n)
        a[:k] = 1.0
        a[n-k:] = 1.0
        return a / a.sum()
    elif mode == 2:
        # Three top-hats
        k = n // 5
        a = np.zeros(n)
        a[:k] = 1.0
        a[n//3 - k//2:n//3 + k//2] = 1.0
        a[n-k:] = 1.0
        a = np.maximum(a, 0)
        return a / a.sum()
    elif mode == 3:
        # Ramp
        a = np.linspace(0.1, 1.0, n)
        return a / a.sum()
    elif mode == 4:
        # Inverse ramp
        a = np.linspace(1.0, 0.1, n)
        return a / a.sum()
    elif mode == 5:
        # W-shaped (two valleys)
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 6:
        # M-shaped (peak in middle)
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:
        # Random binary (50% ones)
        a = (rng.random(n) > 0.5).astype(float)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 8:
        # Block pattern (alternating high/low)
        block = max(1, n // 10)
        a = np.zeros(n)
        for i in range(0, n, 2 * block):
            a[i:i+block] = 1.0
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 9:
        # Exponential decay from center
        x = np.abs(np.arange(n) - n/2)
        a = np.exp(-x / (n/4))
        return a / a.sum()
    elif mode == 10:
        # Sine-modulated
        x = np.linspace(0, 1, n)
        a = 0.7 + 0.3 * np.sin(6 * np.pi * x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining structural initialization, LP fixed-point
    with structural mutations, subspace optimization, and GD refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent_n = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                parent_n = len(parent)
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent_n}): obj={best_score:.10f}")
    except Exception:
        parent = None

    if best_seq is None:
        best_seq = np.ones(parent_n) / parent_n
        best_score = compute_obj(best_seq)
        print(f"Default init (n={parent_n}): obj={best_score:.10f}")

    # === Phase 1: Structural initialization sweep ===
    print(f"\n=== Phase 1: Structural Init Sweep ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 100.0)

    n_options = sorted(set([parent_n, parent_n - 100, parent_n - 50,
                            parent_n + 50, parent_n + 100, parent_n + 200]))
    n_options = [n for n in n_options if 500 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        per_template = min(p1_budget / max(1, len(n_options) * 6), 8.0)
        for mode in [0, 1, 2, 3, 5, 7, 8]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_init = make_structural_init(n_try, mode, rng)
            alloc = min(per_template, remaining * 0.3)
            if alloc < 3:
                continue
            a_opt = nesterov_gd(a_init, alloc, tau_levels=[10, 50, 200], verbose=False)
            obj_val = compute_obj(a_opt)
            if obj_val < best_score:
                best_score = obj_val
                best_seq = a_opt.copy()
                print(f"  n={n_try}, mode={mode}: obj={best_score:.10f} **")

    print(f"  After Phase 1: best={best_score:.10f} (n={len(best_seq)})")

    # === Phase 2: Intensive LP polishing with structural mutations ===
    print(f"\n=== Phase 2: LP Polishing with Mutations ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.45, 500.0)

    if p2_budget > 30 and len(best_seq) <= 2500:
        # Multiple rounds with different seeds and strategies
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p2_budget / 4.0, remaining * 0.7)
            if alloc < 20:
                break

            a_lp = lp_polish_advanced(best_seq, alloc, verbose=True,
                                      seed=100 + rnd * 7)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.10f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.10f}")

            # GD refinement after LP
            remaining = deadline - time.monotonic()
            if obj_lp < best_score + 0.001 and remaining > 20:
                alloc_gd = min(15.0, remaining * 0.2)
                a_gd = nesterov_gd(a_lp, alloc_gd,
                                   tau_levels=[100, 500, 2000], verbose=False)
                obj_gd = compute_obj(a_gd)
                if obj_gd < best_score:
                    best_score = obj_gd
                    best_seq = a_gd.copy()
                    print(f"  LP+GD round {rnd}: obj={best_score:.10f} **")

    # === Phase 3: Subspace optimization ===
    print(f"\n=== Phase 3: Subspace Optimization ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)

    if p3_budget > 20 and len(best_seq) <= 1500:
        a_sub = subspace_lp_optimize(best_seq, p3_budget, verbose=True, seed=999)
        obj_sub = compute_obj(a_sub)
        if obj_sub < best_score:
            best_score = obj_sub
            best_seq = a_sub.copy()
            print(f"  Subspace: obj={best_score:.10f} **")
        else:
            print(f"  Subspace: obj={obj_sub:.10f}")

    # === Phase 4: Multi-n refinement ===
    print(f"\n=== Phase 4: Multi-n Refinement ===")
    remaining = deadline - time.monotonic()

    n_current = len(best_seq)
    n_tries = sorted(set([n_current - 50, n_current, n_current + 50,
                          n_current + 100, n_current - 100]))
    n_tries = [n for n in n_tries if 600 <= n <= 1500]
    n_tries.sort(key=lambda x: abs(x - n_current))

    for n_try in n_tries[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        if n_try == n_current:
            continue

        a_resampled = resample_sequence(best_seq, n_try)
        # Quick GD
        alloc1 = min(15.0, remaining * 0.2)
        if alloc1 > 5:
            a_resampled = nesterov_gd(a_resampled, alloc1,
                                       tau_levels=[20, 100, 500], verbose=False)
        # Quick LP
        remaining = deadline - time.monotonic()
        if n_try <= 1500:
            alloc2 = min(20.0, remaining * 0.2)
            if alloc2 > 10:
                a_resampled = lp_polish_advanced(a_resampled, alloc2,
                                                  verbose=False, seed=n_try)
        obj_try = compute_obj(a_resampled)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_resampled.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau GD
    if remaining > 40:
        alloc = min(remaining * 0.2, 80.0)
        a_f1 = nesterov_gd(best_seq, alloc,
                           tau_levels=[200, 800, 3000, 6000], verbose=True)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final GD: obj={best_score:.10f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 25 and len(best_seq) <= 2000:
        alloc = min(remaining * 0.3, 60.0)
        a_f2 = lp_polish_advanced(best_seq, alloc, verbose=True, seed=4242)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    # Ultra-high tau GD
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.4, 50.0)
        a_f3 = nesterov_gd(best_seq, alloc,
                           tau_levels=[500, 2000, 8000], verbose=False)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Ultra GD: obj={best_score:.10f} **")

    # One last LP if time permits
    remaining = deadline - time.monotonic()
    if remaining > 30 and len(best_seq) <= 2000:
        alloc = min(remaining * 0.5, 40.0)
        a_f4 = lp_polish_advanced(best_seq, alloc, verbose=True, seed=31337)
        obj_f4 = compute_obj(a_f4)
        if obj_f4 < best_score:
            best_score = obj_f4
            best_seq = a_f4.copy()
            print(f"  Last LP: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
