import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_lp_fast(a):
    """Solve LP: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0.
    Returns (g_opt, M_opt, dual_y) or (None, None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build the convolution matrix A (2n-1 x n)
    # A[k, j] = a_norm[k-j] if 0 <= k-j <= n-1, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        start = j
        end = min(2*n-1, j+n)
        A[start:end, j] = a_norm[:end-j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Get dual variables
                    M_opt = float(M.value)
                    return g_val, M_opt
        except Exception:
            continue
    return None, None

def line_search_blend(a, g, max_iters=50):
    """Find best t in [0,1] that minimizes obj((1-t)*a + t*g)."""
    n = len(a)
    best_t = 0.0
    best_obj = compute_obj(a)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # ~0.618
    a_low, a_high = 0.0, 1.0
    
    def obj_at_t(t):
        a_try = (1-t)*a + t*g
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            return np.inf
        return compute_obj(a_try / s)
    
    c = a_high - gr*(a_high - a_low)
    d = a_low + gr*(a_high - a_low)
    fc = obj_at_t(c)
    fd = obj_at_t(d)
    
    for _ in range(max_iters):
        if fc < fd:
            a_high = d
            d = c
            fd = fc
            c = a_high - gr*(a_high - a_low)
            fc = obj_at_t(c)
        else:
            a_low = c
            c = d
            fc = fd
            d = a_low + gr*(a_high - a_low)
            fd = obj_at_t(d)
        if a_high - a_low < 1e-6:
            break
    
    best_t = (a_high + a_low) / 2
    best_obj = obj_at_t(best_t)
    
    # Also check t=0 and t=1
    if best_obj > obj_at_t(0.0):
        best_t = 0.0
    if best_obj > obj_at_t(1.0):
        best_t = 1.0
    
    a_new = (1-best_t)*a + best_t*g
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy(), 0.0
    return a_new / s, best_obj

But the 1D optimization per coordinate is the key. Let me think about how to do it efficiently.

For coordinate i, with all others fixed:
- Let b_k = conv_k - a[i]*a[k-i] for k in [i, min(2n-2, i+n-1)] (the "base" without a[i]'s contribution)
- Let W = [a[k-i] for k in [i, min(2n-2, i+n-1)]] (the coefficients of x = a[i])
- Let M_rest = max(conv_k for k NOT in [i, i+n-1]) (the max of unaffected positions)

Then for value x:
- New conv at position k (in range): b_k + x * a[k-i]
- New conv at position k (out of range): conv_k (unchanged)
- New max = max(M_rest, max_k_in_range (b_k + x * a[k-i]))
- Objective = 2*n * new_max / (S0 + x)^2

To minimize this, we can note:
- Let f(x) = max(M_rest, max_j (b_{k_j} + x * a[k_j - i]))
- We want to minimize f(x) / (S0 + x)^2

The function f(x) is the max of linear functions, hence convex and piecewise linear.
The function (S0+x)^2 is convex and increasing for x ≥ 0.

The ratio f(x)/(S0+x)^2 is not necessarily convex, but it's a unimodal function in practice (for reasonable inputs). We can minimize it by:
1. Finding the x where f(x) = M_rest (crossover point)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: LP alternating with golden-section line search blending as the main
engine, followed by exact coordinate descent with incremental convolution
updates for fine polishing, and multi-length exploration for diversity.
Key improvements over prior: (1) golden-section line search for optimal blend
factor each iteration, (2) coordinate descent with O(n) incremental conv
updates enabling many more passes, (3) structured diverse initializations,
(4) adaptive time allocation across phases and lengths.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT. Returns array of length len(a)+len(b)-1."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def obj_at_blend(a, g, t):
    """Objective at blend (1-t)*a + t*g, normalized."""
    a_try = (1.0 - t) * a + t * g
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 1e-10:
        return np.inf
    a_try = a_try / s
    return compute_obj(a_try)


def golden_blend(a, g, iters=35):
    """
    Golden section search for best t in [0,1] minimizing obj((1-t)*a+t*g).
    Returns (best_a, best_obj, best_t).
    """
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - inv_phi * (hi - lo)
    d = lo + inv_phi * (hi - lo)
    fc = obj_at_blend(a, g, c)
    fd = obj_at_blend(a, g, d)

    for _ in range(iters):
        if hi - lo < 1e-8:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - inv_phi * (hi - lo)
            fc = obj_at_blend(a, g, c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + inv_phi * (hi - lo)
            fd = obj_at_blend(a, g, d)

    t_best = (lo + hi) / 2.0
    obj_best = obj_at_blend(a, g, t_best)

    # Check endpoints
    obj0 = obj_at_blend(a, g, 0.0)
    obj1 = obj_at_blend(a, g, 1.0)
    if obj0 < obj_best:
        t_best, obj_best = 0.0, obj0
    if obj1 < obj_best:
        t_best, obj_best = 1.0, obj1

    a_new = (1.0 - t_best) * a + t_best * g
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy(), compute_obj(a), t_best
    return a_new / s, obj_best, t_best


def lp_alternate_ls(a, time_budget, seed=42, verbose=True):

    n = len(a)

        return a
    a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    consecutive_no_improve = 0

    while time.monotonic() - start < time_budget:

        a_try, obj_try, t_used = golden_blend(a, g_lp, iters=30)

            best_a = a_try

            stall = 0

            if verbose and iter_count % 30 == 0:
                el = time.monotonic() - start

        else:
            stall += 1

            # Try a smaller perturbation + re-LP
            if stall >= 4:
                strength = 0.005 + 0.003 * min(stall, 8)
                a = best_a * np.exp(strength * rng.standard_normal(n))
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))

                    a = a / s_a

                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a

    if verbose:

            # We need to check which piece is active at each candidate x.

            # Practical approach: generate a few candidates and pick the best.
            candidates = [0.0, old_val]

            # Find approximate crossover
            if M_static > 0:
                for k in range(max(0, i), min(2 * n - 1, i + n)):
                    wi = a[k - i]
                    if wi > 1e-15:
                        x_c = (M_static - conv_base[k]) / wi
                        if x_c > 0:
                            candidates.append(x_c)

            # Optimal for each active piece
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                wi = a[k - i]
                if wi > 1e-15:
                    x_opt = S0 - 2.0 * conv_base[k] / wi
                    if x_opt > 0:
                        candidates.append(x_opt)
                    # Also try x_opt clamped
                    candidates.append(max(0.0, 0.5 * x_opt) if x_opt > 0 else 0.0)

            # Also try geometric mean of neighbors
            if i > 0 and i < n - 1:
                candidates.append(0.5 * (a[i - 1] + a[i + 1]))
            candidates.append(1.0 / n)  # uniform

            # Evaluate candidates
            best_x = old_val
            best_cand_obj = best_obj  # current objective is the baseline

            # We need to compute max_k (conv_base[k] + x*a[k-i] if valid) for each x
            for x in candidates:
                x = max(0.0, x)
                M = M_static
                for k in range(max(0, i), min(2 * n - 1, i + n)):
                    val = conv_base[k] + x * a[k - i]
                    if val > M:
                        M = val
                obj_cand = 2.0 * n * M / (S0 + x) ** 2
                if obj_cand < best_cand_obj:
                    best_cand_obj = obj_cand
                    best_x = x

            if best_x != old_val:
                a[i] = best_x
                improved_any = True

        if improved_any:
            cur_obj = compute_obj(a)
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()

        if not improved_any and pass_count > 1:
            break

    return best_a


def geometric_blend_search(a, g, iters=30):
    """Try both linear and geometric blends, pick the best."""
    # Linear blend
    a_lin, obj_lin, _ = golden_blend(a, g, iters=iters)

    # Geometric blend: a^(1-t) * g^t (element-wise, with epsilon to handle zeros)
    eps = 1e-15
    a_pos = np.maximum(a, eps)
    g_pos = np.maximum(g, eps)
    log_a = np.log(a_pos)
    log_g = np.log(g_pos)

    best_geom_obj = np.inf
    best_geom_a = a_lin

    # Try a few t values for geometric blend
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
        a_try = np.exp((1 - t) * log_a + t * log_g)
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        if obj_try < best_geom_obj:
            best_geom_obj = obj_try
            best_geom_a = a_try

    if best_geom_obj < obj_lin:
        return best_geom_a, best_geom_obj
    return a_lin, obj_lin


def make_initialization(n, seed, n_init_type=0):
    """Create diverse initializations."""
    rng = np.random.default_rng(seed)

    if n_init_type == 0:
        # Uniform
        return np.ones(n) / n
    elif n_init_type == 1:
        # Exponential decay
        a = np.exp(np.linspace(0, -2, n))
        return a / a.sum()
    elif n_init_type == 2:
        # Cosine-shaped
        a = np.cos(np.pi * np.arange(n) / (2 * n)) ** 2
        a = np.maximum(a, 0)
        return a / a.sum()
    elif n_init_type == 3:
        # Random
        a = rng.random(n)
        return a / a.sum()
    elif n_init_type == 4:
        # Two peaks
        a = np.zeros(n)
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.05, 0.15)
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - c1) / w) ** 2) + np.exp(-0.5 * ((x - c2) / w) ** 2)
        a = np.maximum(a, 0)
        return a / a.sum()
    elif n_init_type == 5:
        # Power law with noise
        a = (np.arange(n) + 1) ** (-0.5)
        a = a * np.exp(0.1 * rng.standard_normal(n))
        a = np.maximum(a, 0)
        return a / a.sum()
    else:
        # Asymmetric smooth
        t = np.linspace(0, 1, n)
        a = np.exp(-3 * (t - 0.3) ** 2) + 0.5 * np.exp(-3 * (t - 0.7) ** 2)
        a = a * np.exp(0.05 * rng.standard_normal(n))
        a = np.maximum(a, 0)
        return a / a.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-phase optimization combining LP alternating
    with line search, coordinate descent polishing, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0  # 30s safety margin

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0)
        if parent.size == 0 or parent.sum() < 0.01:
            raise ValueError("Invalid parent")
        parent = parent / parent.sum()
        n_parent = parent.size
        print(f"Parent: n={n_parent}, obj={compute_obj(parent):.10f}")
    except Exception:
        parent = None
        n_parent = 800
        print("No valid parent, using defaults")

    best_a = None
    best_obj = np.inf

    # Try multiple lengths and initializations
    # Focus on lengths near the parent's length, plus some exploration
    if parent is not None:
        n_values = sorted(set([
            n_parent,
            n_parent - 50, n_parent + 50,
            n_parent - 100, n_parent + 100,
            400, 600, 1000, 1200
        ]))
    else:
        n_values = [400, 600, 800, 1000]

    n_values = [n for n in n_values if 200 <= n <= 2000]

    # Allocate time: give more time to promising (near-parent) lengths
    # Phase 1: Quick scan of different n values with short LP runs
    print("\n=== Phase 1: Quick Scan (multi-n, short LP) ===")
    p1_deadline = start_time + budget_s * 0.30
    n_per_scan = max(60, (p1_deadline - time.monotonic()) / len(n_values))

    scan_results = []
    for idx, n in enumerate(n_values):
        remaining = p1_deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(n_per_scan, remaining - 5)
        if alloc < 30:
            break

        # Use parent resampled as init if available
        if parent is not None and abs(n - n_parent) <= 100:
            a_init = resample_sequence(parent, n)
            init_type = 0
        else:
            a_init = make_initialization(n, seed + idx * 100, init_type=idx % 6)
            init_type = idx % 6

        print(f"  Scanning n={n} (init_type={init_type}), budget={alloc:.0f}s...")
        a_scan = lp_alternate_ls(a_init, time_budget=alloc * 0.8, seed=seed + idx, verbose=False)
        obj_scan = compute_obj(a_scan)
        scan_results.append((obj_scan, n, a_scan.copy()))
        print(f"    n={n}: obj={obj_scan:.10f}")

        if obj_scan < best_obj:
            best_obj = obj_scan
            best_a = a_scan.copy()
            print(f"    ** New best: {best_obj:.10f}")

    # Sort scan results by objective
    scan_results.sort(key=lambda x: x[0])
    print(f"\n  Scan results (best first):")
    for obj_s, n_s, _ in scan_results[:5]:
        print(f"    n={n_s}: {obj_s:.10f}")

    # Phase 2: Deep LP optimization on the best 2-3 lengths
    print(f"\n=== Phase 2: Deep LP Optimization ===")
    p2_deadline = start_time + budget_s * 0.60
    top_k = min(3, len(scan_results))
    best_in_phase2 = [obj for obj, _, _ in scan_results[:top_k]]

    for idx in range(top_k):
        remaining = p2_deadline - time.monotonic()
        if remaining < 60:
            break
        alloc = min(remaining * 0.4, 200.0)
        if alloc < 30:
            break

        obj_s, n_s, a_s = scan_results[idx]
        # Start from best found so far if it's for this n, else from scan result
        if best_a is not None and len(best_a) == n_s:
            a_start = best_a.copy()
        else:
            a_start = a_s.copy()

        print(f"  Deep LP n={n_s}, obj_start={compute_obj(a_start):.10f}, budget={alloc:.0f}s")
        a_deep = lp_alternate_ls(a_start, time_budget=alloc, seed=seed + 1000 + idx * 7, verbose=True)
        obj_deep = compute_obj(a_deep)
        print(f"    Result n={n_s}: obj={obj_deep:.10f}")

        if obj_deep < best_obj:
            best_obj = obj_deep
            best_a = a_deep.copy()
            print(f"    ** New best: {best_obj:.10f}")

    # Phase 3: Coordinate descent polish on the best sequence
    print(f"\n=== Phase 3: Coordinate Descent Polish ===")
    p3_deadline = start_time + budget_s * 0.80
    remaining = p3_deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.6, 150.0)
        print(f"  CD polish, budget={alloc:.0f}s, n={len(best_a)}")
        a_cd = fast_coord_descent(best_a, time_budget=alloc, verbose=True)
        obj_cd = compute_obj(a_cd)
        print(f"    Result: obj={obj_cd:.10f}")
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()
            print(f"    ** New best: {best_obj:.10f}")

    # Phase 4: Interleave LP and CD for final refinement
    print(f"\n=== Phase 4: Interleaved LP+CD Final Refinement ===")
    p4_deadline = deadline
    remaining = p4_deadline - time.monotonic()
    if remaining > 30:
        # Alternate between short LP and short CD
        n_inner = 0
        while time.monotonic() < p4_deadline:
            inner_budget = min(60.0, (p4_deadline - time.monotonic()) / 2)
            if inner_budget < 10:
                break

            if n_inner % 2 == 0:
                a_try = lp_alternate_ls(best_a, time_budget=inner_budget,
                                        seed=seed + 5000 + n_inner, verbose=False)
            else:
                a_try = fast_coord_descent(best_a, time_budget=inner_budget, verbose=False)

            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                print(f"  Inner {n_inner}: obj={best_obj:.10f} **")
            else:
                print(f"  Inner {n_inner}: obj={obj_try:.10f} (no improvement)")
            n_inner += 1

    # Phase 5: Try one more diverse initialization and optimize
    print(f"\n=== Phase 5: Diverse Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Try a completely different initialization at the best n
        n_best = len(best_a)
        for init_type in [4, 5, 6, 1]:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            alloc = min(remaining * 0.3, 60.0)
            if alloc < 20:
                break

            a_div = make_initialization(n_best, seed + 9000 + init_type, init_type=init_type)
            # Quick LP to get it in the basin
            a_div = lp_alternate_ls(a_div, time_budget=alloc * 0.5,
                                    seed=seed + 9000 + init_type, verbose=False)
            obj_div = compute_obj(a_div)
            if obj_div < best_obj:
                # Polish it
                remaining = deadline - time.monotonic()
                if remaining > 20:
                    a_div = fast_coord_descent(a_div, time_budget=min(remaining * 0.4, 30.0))
                    obj_div = compute_obj(a_div)
                if obj_div < best_obj:
                    best_obj = obj_div
                    best_a = a_div.copy()
                    print(f"  Diverse init {init_type}: obj={best_obj:.10f} **")
            else:
                print(f"  Diverse init {init_type}: obj={obj_div:.10f}")

    # Final: One last CD polish
    remaining = deadline - time.monotonic()
    if remaining > 15 and best_a is not None:
        a_final = fast_coord_descent(best_a, time_budget=min(remaining * 0.7, 25.0))
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final CD: obj={best_obj:.10f} **")

    # Final validation and output
    if best_a is None:
        best_a = np.ones(100)
        best_a = best_a / best_a.sum()

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    # Ensure sum is large enough
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 100

    val = evaluate_sequence(result)
    if val < np.inf and val < best_obj:
        best_obj = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
