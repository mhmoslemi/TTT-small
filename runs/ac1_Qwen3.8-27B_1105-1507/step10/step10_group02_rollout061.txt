```python
def eigenvector_approach(n, num_active, active_lags, max_iter=100):
    """
    Find non-negative eigenvector of convolution operator defined by active_lags.
    """
    a = np.ones(n) / n
    for _ in range(max_iter):
        # Compute T_lambda @ a
        # (T_lambda a)_j = sum_{k in active_lags} a_{k-j} (with bounds checking)
        new_a = np.zeros(n)
        for k in active_lags:
            # For each j, add a_{k-j} if 0 <= k-j < n
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            # j ranges from lo to hi-1, and k-j ranges from k-hi+1 to k-lo
            # a_{k-j} for j in [lo, hi) is a[k-hi+1 : k-lo+1]
            idx = k - np.arange(lo, hi)
            new_a[lo:hi] += a[idx]
        
        # Normalize
        s = new_a.sum()
        if s < 1e-15:
            break
        a = new_a / s
        
        # Check convergence
        if np.max(np.abs(a - new_a/new_a.sum())) < 1e-10:
            break
    
    return a
```

```python
"""
Hybrid optimizer for step function peak-to-average ratio minimization.
Combines fast coordinate descent (exact 1D subproblems via golden section),
LP-based directional polishing (iterative min-peak LP), subgradient-based
global search with restarts, and systematic multi-n exploration.
Key improvements over previous: faster CD inner loop, more aggressive LP
iteration with adaptive blending, power-iteration-based initializations,
and smarter time allocation.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return float(2.0 * n * np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, pref_solvers=None):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_s = float(np.sum(a))
    if a_s < 1e-15:
        return None, None
    a_norm = a / a_s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if pref_solvers is None:
        pref_solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in pref_solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def fast_coordinate_descent(a, time_budget, seed=42, verbose=False):
    """
    Fast exact coordinate descent. Each coordinate subproblem solved via
    golden section with tight brackets and early termination.
    Updates convolution incrementally for speed.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    no_improve_sweeps = 0

    def f_1d(x, v, S_other, ks_lin, a_ki_lin, c_k_lin, c_2i, n_local):
        """1D objective for coordinate update."""
        if len(ks_lin) > 0:
            lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
            max_lin = float(np.max(lin_vals))
        else:
            max_lin = 0.0
        quad_val = c_2i + x * x - v * v
        max_val = max(max_lin, quad_val)
        denom = (S_other + x) * (S_other + x)
        if denom < 1e-30:
            return 1e30
        return 2.0 * n_local * max_val / denom

    while time.monotonic() - start < time_budget:
        sweep += 1

        # Periodic recompute to avoid drift
        if sweep % 5 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    no_improve_sweeps = 0
                else:
                    no_improve_sweeps += 1
                # Reset if no improvement for a while
                if no_improve_sweeps > 20 and sweep > 10:
                    # Small random perturbation to escape
                    pert = rng.normal(0, 0.005, n)
                    a = a * np.exp(pert)
                    a = np.maximum(a, 0)
                    S = float(np.sum(a))
                    a = a / S
                    conv = compute_conv(a)
                    no_improve_sweeps = 0

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Evaluate at current point
            f_v = f_1d(v, v, S_other, ks_lin, a_ki_lin, c_k_lin, c_2i, n)

            # Quick candidates
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.3, v * 0.7, v * 1.3, v * 2.0,
                          S_other * 0.1, S_other * 0.5]
            candidates = [max(0.0, c) for c in candidates]

            best_x = v
            best_f = f_v
            for x in candidates:
                if x == v:
                    continue
                fx = f_1d(x, v, S_other, ks_lin, a_ki_lin, c_k_lin, c_2i, n)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement
            if best_x != v:
                x_lo = max(0.0, best_x - scale)
                x_hi = best_x + scale
                if x_hi <= x_lo:
                    x_hi = x_lo + 1e-8

                c_pt = x_hi - gr * (x_hi - x_lo)
                d_pt = x_lo + gr * (x_hi - x_lo)
                fc = f_1d(c_pt, v, S_other, ks_lin, a_ki_lin, c_k_lin, c_2i, n)
                fd = f_1d(d_pt, v, S_other, ks_lin, a_ki_lin, c_k_lin, c_2i, n)

                for _ in range(25):
                    if x_hi - x_lo < 1e-12 * max(1.0, abs(x_hi)):
                        break
                    if fc < fd:
                        x_hi = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = x_hi - gr * (x_hi - x_lo)
                        fc = f_1d(c_pt, v, S_other, ks_lin, a_ki_lin, c_k_lin, c_2i, n)
                    else:
                        x_lo = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = x_lo + gr * (x_hi - x_lo)
                        fd = f_1d(d_pt, v, S_other, ks_lin, a_ki_lin, c_k_lin, c_2i, n)

                x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            else:
                x_opt = v

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt

            # Update conv incrementally
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

            # Track best
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

    return best_a

def power_iteration_init(n, p, verbose=False):
    """

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Power law
        p = rng.uniform(0.5, 5.0)
        a = (p + 1) * x ** p
        a = a / a.sum()
    elif mode == 3:
        # Exponential
        rate = rng.uniform(0.5, 5.0)
        a = np.exp(rate * x)
        a = a / a.sum()
    elif mode == 4:
        # Symmetric bump
        center = n // 2
        width = rng.uniform(n * 0.1, n * 0.4)
        a = np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Two bumps
        a = np.exp(-((np.arange(n) - n * 0.25) ** 2) / (2 * (n * 0.1) ** 2)) + \
            np.exp(-((np.arange(n) - n * 0.75) ** 2) / (2 * (n * 0.1) ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Inverse (U-shaped)
        a = 1.0 / (x + 0.1) + 1.0 / (1.0 - x + 0.1)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def subgradient_escape(a, time_budget, seed=42):
    """
    Subgradient-based method: use the subgradient of max(conv) to guide
    updates. The subgradient at a point is supported on the set of
    lags achieving the max.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    alpha = 2.0 / np.sqrt(n)
    beta = 0.7
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))

        # Compute subgradient: probability distribution over active lags
        # Use soft max for smoothness
        tau = 50.0
        exp_vals = np.exp(tau * (conv - max_val))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient of max(conv) w.r.t. a: 2 * (p convolved with a) restricted to [0,n)
        # (a*a)[k] = sum_j a_j * a_{k-j}, d/dx_i (a*a)[k] = a_{k-i} + a_{i-k} (with bounds)
        # Gradient = 2 * sum_k p[k] * (a * delta)(k, i)
        # More precisely: grad_i = 2 * sum_k p[k] * a[k-i] (with bounds)
        grad = np.zeros(n)
        for i in range(n):
            # (d/d a_i)(a*a)[k] = a[k-i] if 0 <= k-i < n
            # sum_k p[k] * a[k-i] for valid k
            lo = max(0, i)
            hi = min(2 * n - 1, i + n)
            valid_ks = np.arange(lo, hi)
            valid_is = valid_ks - i
            grad[i] = 2.0 * float(np.dot(p[valid_ks], a[valid_is]))

        # Nesterov momentum
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = 1.0
        else:
            y = y / s_y

        # Gradient step on y
        lr = min(alpha, 1.0 / (np.linalg.norm(grad) + 1e-10))
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        v = a_new - a
        a = a_new

        obj = 2.0 * n * float(np.max(compute_conv(a)))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        it += 1
        if it % 20000 == 0:
            elapsed = time.monotonic() - start
            if verbose:
                print(f"    SG it{it}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a

                if obj_cd2 < best_score:
                    best_score = obj_cd2
                    best_seq = a_cd2.copy()
                    print(f"  CD after LP {rnd}: obj={best_score:.8f} **")

    # === Phase 3: Subgradient Escape (10%) ===
    print(f"\n=== Phase 3: Subgradient Escape ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 150.0)
    if p3_budget > 20:
        # Perturb and run subgradient
        pert = rng.normal(0, 0.05, len(best_seq))
        a_pert = best_seq * np.exp(pert)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        a_sg = subgradient_escape(a_pert, time_budget=p3_budget, seed=seed + 777)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SG: obj={best_score:.8f} **")
        else:
            print(f"  SG: obj={obj_sg:.8f}")
            # If SG didn't help, do CD on the result anyway
            if obj_sg < best_score + 0.01:
                a_cd3 = fast_coordinate_descent(a_sg, time_budget=min(20, (deadline - time.monotonic()) * 0.1), seed=seed + 888)
                obj_cd3 = compute_obj(a_cd3)
                if obj_cd3 < best_score:
                    best_score = obj_cd3
                    best_seq = a_cd3.copy()
                    print(f"  CD after SG: obj={best_score:.8f} **")

    # === Phase 4: Multi-n Exploration (15%) ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Generate candidate n values around current best
    n_options = set()
    for delta in [-200, -100, -50, 50, 100, 200, 300, 400]:
        n_try = n_best + delta
        if 300 <= n_try <= 2500:
            n_options.add(n_try)
    # Also try some fixed values
    for n_fixed in [400, 500, 600, 700, 900, 1000, 1200, 1500]:
        if 300 <= n_fixed <= 2500 and abs(n_fixed - n_best) > 80:
            n_options.add(n_fixed)

    n_options = sorted(n_options)[:6]  # Limit to 6 options
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.25, 60.0)
        if alloc > 15:
            a_try = fast_coordinate_descent(a_try, time_budget=alloc, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts at best n (10%) ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = 2 + r  # Power law, exponential, bump
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.5 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.3, 60.0)
        if alloc > 15:
            a_r = fast_coordinate_descent(a_r, time_budget=alloc, seed=20000 + r * 200)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        # Final CD
        alloc_cd = min(remaining * 0.5, 100.0)
        a_final = fast_coordinate_descent(best_seq, time_budget=alloc_cd, seed=seed + 99999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_lp = min(remaining * 0.6, 60.0)
        a_lp_final = lp_polish_iterative(best_seq, time_budget=alloc_lp, seed=77777)
        obj_lp_final = compute_obj(a_lp_final)
        if obj_lp_final < best_score:
            best_score = obj_lp_final
            best_seq = a_lp_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
