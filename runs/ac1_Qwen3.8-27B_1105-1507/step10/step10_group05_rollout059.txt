def nesterov_subgradient(a, time_budget=500, active_set_size=20, verbose=True):
    """Nesterov accelerated subgradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9  # momentum
    lr = 0.1 / np.sqrt(n)  # initial learning rate
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead
    
    it = 0
    best_it = 0
    last_update = 0
    
    # PAT averaging
    a_avg = a.copy()
    
    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Compute autocorrelation
        c = compute_conv(y)
        M = float(np.max(c))
        
        # Find active set (top K lags)
        if it - last_update >= 100:  # Update active set periodically
            top_k = min(active_set_size, len(c))
            active_indices = np.argsort(c)[::-1][:top_k]
            # Use weights proportional to how close to max
            eps = 1e-6
            active_mask = c >= M - eps * M
            if np.sum(active_mask) < 3:
                active_mask = np.zeros(len(c), dtype=bool)
                active_mask[active_indices[:5]] = True
            active_set = np.where(active_mask)[0]
            # Weights: uniform over active set (or proportional to c[k])
            weights = np.ones(len(active_set)) / len(active_set)
            last_update = it
        else:
            # Use previous active_set and weights
            pass
        
        # Compute subgradient
        g = compute_subgradient(y, active_set, weights)
        
        # Gradient step with momentum
        v = beta * v - lr * g
        a_new = y + v
        a_new = project_simplex(a_new)
        
        # Update
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        # PAT-style averaging
        a_avg = (a_avg * it / (it + 1)) + (a_new / (it + 1))
        
        a = a_new
        it += 1
        
        # Adaptive learning rate
        lr *= 0.9999
        
        if it % 10000 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"    NSD it{it}: obj={best_obj:.8f}, M={M:.6e}, lr={lr:.2e}, t={elapsed:.0f}s")
        
        # Stall detection
        if it - best_it > 50000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 0.1 / np.sqrt(n)
            best_it = it
    
    return best_a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov accelerated subgradient descent with FFT-based gradient computation,
active set tracking, periodic LP corrections, coordinate descent polishing,
and multi-length exploration. Key innovations: exact subgradient via FFT cross-correlation,
adaptive active set management, and hybrid subgradient+LP strategy.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import math
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_fft(a, active_set, weights, fft_size=None):
    """
    Compute subgradient of max_k (a*a)[k] at a.
    g[j] = 2 * sum_{k in active_set} p_k * a[k-j] (if 0 <= k-j < n)
    Uses FFT cross-correlation for efficiency.
    """
    n = len(a)
    if len(active_set) == 0:
        return np.zeros(n)

    p_full = np.zeros(2 * n - 1)
    p_full[active_set] = weights

    if fft_size is None:
        fft_size = 1
        while fft_size < 3 * n:
            fft_size <<= 1

    A_fft = fft(a, fft_size)
    P_fft = fft(p_full, fft_size)
    g = 2.0 * ifft(np.conj(A_fft) * P_fft).real[:n]

    return g


def get_active_set(c, top_k=10, margin=0.0):
    """Get active set: lags where autocorrelation is near the peak."""
    M = float(np.max(c))
    if margin > 0:
        threshold = M - margin * M
    else:
        threshold = M - 1e-10 * max(M, 1e-15)

    active_mask = c >= threshold
    n_active = np.sum(active_mask)

    if n_active < 2:
        # Use top-k lags
        top_indices = np.argsort(c)[::-1][:top_k]
        active_mask = np.zeros(len(c), dtype=bool)
        active_mask[top_indices] = True
    elif n_active > 50:
        # Limit to top-50
        top_indices = np.argsort(c)[::-1][:50]
        active_mask = np.zeros(len(c), dtype=bool)
        active_mask[top_indices] = True

    active_set = np.where(active_mask)[0]
    # Weights: proportional to how close to peak (with minimum weight)
    active_vals = c[active_set]
    w = active_vals / (active_vals + 1e-15)
    w = w / w.sum()
    return active_set, w


def nesterov_subgradient(a, time_budget=300.0, verbose=False, seed=42,
                          momentum=0.9, lr_scale=1.0, active_update_freq=20,
                          top_k=15, margin=0.001):
    """
    Nesterov accelerated subgradient descent on the simplex.
    Minimizes max_k (a*a)[k] subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    rng = np.random.default_rng(seed)

    # Precompute FFT size
    fft_size = 1
    while fft_size < 3 * n:
        fft_size <<= 1

    beta = momentum
    base_lr = lr_scale / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    y = a.copy()

    # Initial active set
    c = compute_conv(y)
    active_set, weights = get_active_set(c, top_k=top_k, margin=margin)

    it = 0
    best_it = 0
    stall_count = 0
    last_best_obj = best_obj

    # Polyak-averaged solution
    a_avg = a.copy()

    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = a.copy()
        else:
            y = y / s_y

        # Compute autocorrelation
        c = compute_conv(y)
        M = float(np.max(c))

        # Update active set periodically
        if it % active_update_freq == 0:
            active_set, weights = get_active_set(c, top_k=top_k, margin=margin)

        # Compute subgradient via FFT
        g = compute_subgradient_fft(y, active_set, weights, fft_size)

        # Momentum update
        v = beta * v - lr * g
        a_new = y + v
        a_new = project_simplex(a_new)

        # Evaluate
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1

        # Polyak averaging (exponential)
        alpha_avg = 2.0 / (it + 2.0)
        a_avg = (1.0 - alpha_avg) * a_avg + alpha_avg * a_new
        obj_avg = compute_obj(a_avg)
        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = a_avg.copy()
            best_it = it

        a = a_new
        it += 1

        # Adaptive learning rate
        lr = max(base_lr * 0.01, lr * 0.99995)

        # Stall detection and restart
        if stall_count > 100 * active_update_freq:
            # Try restarting with perturbation
            strength = min(0.1, 0.01 * (stall_count // (100 * active_update_freq)))
            a_pert = best_a * np.exp(rng.normal(0, strength, n))
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 1e-10:
                a = a_pert / s_p
            else:
                a = best_a.copy()
            v = np.zeros(n)
            y = a.copy()
            lr = base_lr * 0.5
            stall_count = 0
            # Update active set
            c = compute_conv(y)
            active_set, weights = get_active_set(c, top_k=top_k, margin=margin)

        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - start
            print(f"    NSD it{it}: best_obj={best_obj:.8f}, M={M:.6e}, lr={lr:.2e}, t={elapsed:.0f}s")

    return best_a


def smooth_max_gradient(a, tau, active_set=None, weights=None, fft_size=None):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    c = compute_conv(a)

    if active_set is not None and len(active_set) > 0:
        # Use only active set for efficiency
        c_sub = c[active_set]
        max_c = float(np.max(c_sub))
        exp_vals = np.exp(tau * (c_sub - max_c))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p_sub = np.zeros(len(c_sub))
            p_sub[np.argmax(c_sub)] = 1.0
        else:
            p_sub = exp_vals / sum_exp

        # Build full p vector
        p = np.zeros(len(c))
        p[active_set] = p_sub
    else:
        max_c = float(np.max(c))
        exp_vals = np.exp(tau * (c - max_c))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(c))
            p[np.argmax(c)] = 1.0
        else:
            p = exp_vals / sum_exp

    # Gradient via FFT cross-correlation
    if fft_size is None:
        fft_size = 1
        while fft_size < 3 * n:
            fft_size <<= 1

    A_fft = fft(a, fft_size)
    P_fft = fft(p, fft_size)
    g = 2.0 * ifft(np.conj(A_fft) * P_fft).real[:n]

    return g, c

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = cross_correlation(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] if 0 <= k-j < n
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_direction(a, timeout=10.0):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns optimal g direction or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver, timeout=timeout)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val
        except Exception:
            continue
    return None

def coordinate_descent_polish(a, time_budget=100.0, verbose=False, seed=42):
    """
    Coordinate descent: for each pair (j, m), optimize the 2D update
    (a[j]+delta, a[m]-delta) to minimize the peak autocorrelation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    c = compute_conv(a)
    M = float(np.max(c))

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1:
            break

        # Random order of coordinates
        order = rng.permutation(n)
        improved = False

        for j in order:
            if time.monotonic() - start > time_budget - 1:
                break

            # Choose partner m (cyclic or random)
            m = (j + 1) % n
            if m == j:
                m = (j + 1) % n

            if a[j] < 1e-10 or a[m] < 1e-10:
                continue

            # 1D optimization: vary delta in [-a[j]*0.5, a[m]*0.5]
            # a'[j] = a[j] + delta, a'[m] = a[m] - delta
            # Effect on c[k]: c[k] + 2*delta*(a[k-j] - a[k-m]) (approx, valid indices)

            # Use ternary search on the convex function h(delta) = max_k c'[k]
            delta_lo = -a[j] * 0.8
            delta_hi = a[m] * 0.8

            def h(delta):
                a_try = a.copy()
                a_try[j] += delta
                a_try[m] -= delta
                if a_try[j] < 0 or a_try[m] < 0:
                    return 1e10
                c_try = compute_conv(a_try)
                return float(np.max(c_try))

            # Golden section search
            gr = (np.sqrt(5) - 1) / 2
            a_gs = delta_lo + (1 - gr) * (delta_hi - delta_lo)
            b_gs = delta_lo + gr * (delta_hi - delta_lo)
            fa = h(a_gs)
            fb = h(b_gs)

            for _ in range(30):
                if fa < fb:
                    delta_hi = b_gs
                    b_gs = a_gs
                    fb = fa
                    a_gs = delta_lo + (1 - gr) * (delta_hi - delta_lo)
                    fa = h(a_gs)
                else:
                    delta_lo = a_gs
                    a_gs = b_gs
                    fa = fb
                    b_gs = delta_lo + gr * (delta_hi - delta_lo)
                    fb = h(b_gs)

            delta_opt = (delta_lo + delta_hi) / 2.0
            obj_opt = h(delta_opt)

            if obj_opt < M - 1e-12:
                a[j] += delta_opt
                a[m] -= delta_opt
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s < 1e-10:
                    a = best_a.copy()
                    c = compute_conv(a)
                    M = float(np.max(c))
                    continue
                a = a / s
                c = compute_conv(a)
                M = float(np.max(c))
                if 2.0 * n * M < best_obj - 1e-12:
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    improved = True

        it += 1
        if not improved:
            stall += 1
            if stall > 20:
                break
        else:
            stall = 0

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def asymmetric_init(n, rng, seed_offset=0):

    initializations.append(a4)

    # Random (seeded)
    rng_local = np.random.default_rng(12345 + seed_offset)
    a5 = np.abs(rng_local.standard_normal(n)) + 0.1
    a5 = a5 / a5.sum()
    initializations.append(a5)

    # Ramp up then flat
    ramp_len = n // 3
    a6 = np.ones(n)
    a6[:ramp_len] = np.linspace(0.1, 1.0, ramp_len)
    a6 = a6 / a6.sum()
    initializations.append(a6)

    return initializations


def propose_candidate(seed=42, budget_s=1500):
    """
    Main optimization combining Nesterov subgradient, smoothed gradient descent,
    LP polishing, coordinate descent, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        n_current = 800
        best_seq = np.ones(n_current) / n_current
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # === Phase 1: Try diverse initializations ===
    print(f"\n=== Phase 1: Diverse Initialization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 80.0)
    if p1_budget > 10:
        n_init = n_current
        inits = asymmetric_init(n_init, rng)
        # Also include parent
        inits.append(best_seq.copy())

        per_init_budget = p1_budget / len(inits)
        for i, a_init in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break
            a_try = nesterov_subgradient(a_init, time_budget=per_init_budget,
                                         seed=100 + i, top_k=20)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Init {i}: obj={best_score:.8f} **")
            else:
                print(f"  Init {i}: obj={obj_try:.8f}")

    # === Phase 2: Intensive Nesterov subgradient ===
    print(f"\n=== Phase 2: Intensive NSD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.3, 400.0)
    if p2_budget > 30:
        a_nsd = nesterov_subgradient(best_seq, time_budget=p2_budget,
                                     verbose=True, seed=200,
                                     momentum=0.92, lr_scale=1.2,
                                     active_update_freq=30, top_k=20,
                                     margin=0.002)
        obj_nsd = compute_obj(a_nsd)
        if obj_nsd < best_score:
            best_score = obj_nsd
            best_seq = a_nsd.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_nsd:.8f} (no improvement)")

    # === Phase 3: Accelerated smooth descent ===
    print(f"\n=== Phase 3: Smoothed Gradient Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)
    if p3_budget > 20:
        a_asd = accelerated_smooth_descent(best_seq, time_budget=p3_budget,
                                           verbose=True, tau_init=30.0,
                                           tau_end=3000.0, momentum=0.93,
                                           seed=300)
        obj_asd = compute_obj(a_asd)
        if obj_asd < best_score:
            best_score = obj_asd
            best_seq = a_asd.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj_asd:.8f} (no improvement)")

    # === Phase 4: LP polishing ===
    print(f"\n=== Phase 4: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.2, 250.0)
    if p4_budget > 30:
        a_lp = lp_polished_iteration(best_seq, time_budget=p4_budget,
                                     verbose=True, seed=400)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 4: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4: obj={obj_lp:.8f} (no improvement)")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_candidates = sorted(set([
        n_best - 100, n_best - 50, n_best - 20,
        n_best + 20, n_best + 50, n_best + 100, n_best + 200,
        600, 700, 800, 900, 1000, 1200, 1500
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 2000]
    # Limit to 6 most diverse
    n_candidates = n_candidates[:8]
    print(f"  Trying n: {n_candidates}")

    per_n_budget = 40.0
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try == n_best:
            continue

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        alloc = min(remaining * 0.1, per_n_budget)
        if alloc > 10:
            a_try = nesterov_subgradient(a_try, time_budget=alloc,
                                         seed=n_try, top_k=15)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Intensive optimization at best n ===
    print(f"\n=== Phase 6: Intensive at n={len(best_seq)} ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    # Long NSD
    if remaining > 100:
        alloc = min(remaining * 0.35, 300.0)
        a_final = nesterov_subgradient(best_seq, time_budget=alloc,
                                       verbose=True, seed=500,
                                       momentum=0.94, lr_scale=0.8,
                                       active_update_freq=50, top_k=15,
                                       margin=0.001)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Phase 6 NSD: obj={best_score:.8f} **")

    # Smooth descent
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.25, 150.0)
        a_final = accelerated_smooth_descent(best_seq, time_budget=alloc,
                                             verbose=True, tau_init=100.0,
                                             tau_end=5000.0, momentum=0.95,
                                             seed=600)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Phase 6 ASD: obj={best_score:.8f} **")

    # === Phase 7: Coordinate descent polish ===
    print(f"\n=== Phase 7: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 120.0)
        a_cd = coordinate_descent_polish(best_seq, time_budget=alloc,
                                         verbose=True, seed=700)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase 7 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase 7 CD: obj={obj_cd:.8f}")

    # === Phase 8: Final LP + NSD ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        alloc = min(remaining * 0.3, 100.0)
        a_final = lp_polished_iteration(best_seq, time_budget=alloc,
                                        verbose=True, seed=800)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Phase 8 LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 100.0)
        a_final = nesterov_subgradient(best_seq, time_budget=alloc,
                                       verbose=False, seed=900,
                                       momentum=0.95, lr_scale=0.5,
                                       active_update_freq=100, top_k=10,
                                       margin=0.0005)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Phase 8 NSD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 60.0)
        a_final = coordinate_descent_polish(best_seq, time_budget=alloc,
                                            verbose=False, seed=950)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Phase 8 CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
