```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid approach: fast exponentiated gradient (mirror descent) for bulk
optimization, followed by LP-based minimax polishing with constraint generation,
multi-n exploration, and structured restarts.
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def compute_gradient_fft(a, p):
    """Compute gradient of smoothed max w.r.t. a using FFT.
    grad[j] = 2 * sum_k p[k] * a[k-j] (for valid indices)
    """
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    a_flipped = a[::-1]
    P = fft(p, N)
    A = fft(a_flipped, N)
    conv_result = ifft(P * A).real
    grad = 2.0 * conv_result[n - 1: 2 * n - 1]
    return grad

def exponentiated_gradient(a, time_budget, tau_start=10, tau_end=2000, verbose=True):
    """Fast mirror descent (exponentiated gradient) on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        
        # Adaptive tau
        tau = tau_start + (tau_end - tau_start) * progress ** 0.7
        
        # Compute convolution
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Compute p (softmax)
        if tau * max_conv > 500:
            # Use hard max approximation
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            shifted = tau * (conv - max_conv)
            p = np.exp(shifted)
            p_sum = float(np.sum(p))
            if p_sum < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = p / p_sum
        
        # Compute gradient via FFT
        grad = compute_gradient_fft(a, p)
        
        # Step size: conservative to avoid divergence
        max_grad = float(np.max(np.abs(grad)))
        if max_grad < 1e-15:
            break
        eta = min(2.0 / (tau * max_grad + 1e-10), 1.0 / np.sqrt(n) * 10)
        eta = max(eta, 1e-6)
        
        # Mirror descent update
        log_update = -eta * grad
        # Clip for numerical stability
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 1000 == 0:
                print(f"      EG it{it}: obj={best_obj:.8f}, tau={tau:.1f}, eta={eta:.6f}")
        else:
            stall += 1
            if stall > 5000:
                break
        
        a = a_new
        it += 1
    
    return best_a

def lp_polish(a, time_budget, max_constraints=None, verbose=True):

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5
    
    # Track active constraints
    active_set = None
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        a_norm = a / np.sum(a)
        
        # Determine which constraints to use
        conv_aa = compute_conv(a_norm)
        max_aa = float(np.max(conv_aa))

            # Use all constraints or top-k
            if max_constraints and max_constraints < 2 * n - 1:
                top_k = np.argsort(conv_aa)[::-1][:max_constraints]
                rows = top_k
            else:
                rows = np.arange(2 * n - 1)
            active_set = rows
        else:
            # Check if we need to add more constraints
            # (will be handled after LP solve)
            rows = active_set
        
        # Build LP
        A_full = build_conv_matrix(a_norm)
        A = A_full[rows]
        
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        g_val = None
        M_val = None
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                        M_val = float(M.value)
                    break
            except:
                continue
        
        if g_val is None:

        # Line search
        t_candidates = np.array([0.1, 0.2, 0.3, 0.5, 0.7, 1.0])
        t_candidates = np.concatenate([t_candidates, 
                                        [max(0.05, last_t * 0.5), max(0.05, last_t * 0.8), last_t, min(1.0, last_t * 1.3)]])
        t_candidates = np.unique(np.clip(t_candidates, 0.01, 1.0))
        
        best_t_obj = best_obj
        best_t = last_t
        best_a_local = best_a.copy()
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-14:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t
            stall = 0
            
            # Update active set: find positions where conv(a, g) is close to M
            conv_ag = A_full @ g_val  # This is conv(a_norm, g_val) for all positions
            # Actually we need conv of the new a with g_val
            conv_new = compute_conv(a)
            # Active set: positions where conv is in the top few values
            threshold = max_aa * 0.95  # positions within 5% of max
            active_set = np.where(conv_new >= threshold)[0]
            if len(active_set) > max_constraints and max_constraints:
                top_idx = np.argsort(conv_new)[::-1][:max_constraints]
                active_set = top_idx
            elif len(active_set) < 5:
                active_set = np.argsort(conv_new)[::-1][:max(20, n // 10)]
            
            if verbose and stall == 0 and (time.monotonic() - start) % 5 < 1:
                print(f"      LP: obj={best_obj:.8f}, t={best_t:.3f}, active={len(active_set)}, elapsed={time.monotonic()-start:.0f}s")
        else:

            if stall > 5:

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n, endpoint=True)
    x_new = np.linspace(0, 1, new_n, endpoint=True)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Symmetric bump
        x = np.linspace(-1, 1, n)
        a = np.exp(-x**2 * 2)
        a = a / a.sum()
    elif mode == 3:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        a = a / a.sum()
    elif mode == 4:
        # Asymmetric: high on left
        a = np.exp(-np.arange(n) / (n * 0.5))
        a = a / a.sum()
    elif mode == 5:
        # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3)/0.15)**2) + np.exp(-((x - 0.7)/0.15)**2)
        a = a / a.sum()
    elif mode == 6:
        # Triangle
        a = 1 - np.abs(2 * np.arange(n) / n - 1)
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n))
    return a
