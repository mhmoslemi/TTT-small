import numpy as np
import time
import math
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def cg(a, b):
    """Compute cross-correlation a*b via FFT. Returns array of length len(a)+len(b)-1."""
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = fft(a, size)
    fb = fft(b, size)
    return ifft(fa * fb, size).real[:na+nb-1]

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_grad_obj(a, tau):
    """Compute smoothed gradient and objective for a (assumed normalized)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    # Log-sum-exp smoothing
    shifted = tau * (conv - max_c)
    # Handle overflow
    max_shifted = float(np.max(shifted))
    exp_vals = np.exp(shifted - max_shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    # Gradient of (1/tau) * log(sum(exp(tau*conv))) w.r.t. a
    # = sum_k p[k] * d(conv[k])/da

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of LP-based iteration, adaptive smoothed gradient descent
with high temperature, multi-length exploration, and combinatorial initialization.
Key innovations: adaptive tau scheduling, backtracking line search, simplex
projection, difference set warm starts, and aggressive final polishing.
"""

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """
    Compute gradient of smoothed max of conv(a,a) and the smoothed value.
    a is assumed normalized (sum=1). Returns (grad, smoothed_val).
    The smoothed function is: (1/tau)*log(sum_k exp(tau*conv(k)))
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))

    # Stable log-sum-exp weights
    x = tau * (conv - max_c)
    max_x = float(np.max(x))
    exp_x = np.exp(x - max_x)
    sum_exp = float(np.sum(exp_x))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_x / sum_exp

    smoothed_val = max_c + math.log(sum_exp) / tau

    # Gradient computation via FFT
    # grad[j] = sum_k p[k]*(a[k-j] + a[j-k])
    # = cross_corr(a, p)[j] + (a * p)[j]
    a_flip = a[::-1].copy()
    size = 1
    while size < 3 * n - 1:
        size <<= 1

    Af = fft(a_flip, size)
    Ap = fft(a, size)
    Pf = fft(p, size)

    conv_flip = ifft(Af * Pf, size).real
    conv_direct = ifft(Ap * Pf, size).real

    term1 = conv_flip[n - 1:2 * n - 1]
    term2 = conv_direct[:n]

    grad = term1 + term2
    return grad, smoothed_val


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        if r > 0:
            A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, optimal_M) or (None, None).
    """
    import cvxpy as cp
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:
            continue
    return None, None


def lp_iteration(a, time_budget, verbose=False, seed=42):
    """LP-based iteration with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    it = 0
    stagnant = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        it += 1

        # Line search with golden section
        def obj_at_t(t):
            if t <= 0:
                obj_t = best_obj
                a_t = a.copy()
            else:
                a_t = (1.0 - t) * a + t * g_lp
                a_t = np.maximum(a_t, 0)
                s_t = float(np.sum(a_t))
                if s_t < 1e-10:
                    return np.inf
                a_t = a_t / s_t
                obj_t = compute_obj(a_t)
            return obj_t

        # Golden section search in [0, 1]
        gr = (math.sqrt(5) - 1) / 2
        lo, hi = 0.0, 1.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = obj_at_t(c)
        fd = obj_at_t(d)
        for _ in range(20):
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - gr * (hi - lo)
                fc = obj_at_t(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = obj_at_t(d)

        t_best = (lo + hi) / 2
        obj_best_t = obj_at_t(t_best)

        # Also try extrapolation t > 1
        if obj_best_t < best_obj:
            for t_ext in [1.2, 1.5, 2.0]:
                a_ext = t_ext * g_lp - (t_ext - 1.0) * a
                a_ext = np.maximum(a_ext, 0)
                s_ext = float(np.sum(a_ext))
                if s_ext < 1e-10:
                    continue
                a_ext = a_ext / s_ext
                obj_ext = compute_obj(a_ext)
                if obj_ext < obj_best_t:
                    obj_best_t = obj_ext
                    t_best = t_ext

        if obj_best_t < best_obj - 1e-13:
            a_new = (1.0 - t_best) * a + t_best * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
            else:
                a_new = a_new / s_new
            best_obj = obj_best_t
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t_best:.3f}, M={M_lp:.8f}")
        else:
            stagnant += 1
            if stagnant > 8:
                # Perturb to escape
                strength = min(0.2, 0.03 * stagnant)
                a = best_a * np.exp(rng.normal(0, strength, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                prev_g = None
                stagnant = 0
                if verbose:
                    print(f"    LP it{it}: perturbed (strength={strength:.3f})")

        prev_g = g_lp.copy()

    return best_a


def adaptive_smoothed_gd(a, time_budget, tau_start=50.0, tau_end=5000.0,
                         verbose=False, seed=0):
    """
    Adaptive smoothed gradient descent with Nesterov momentum and
    backtracking line search. Tau increases over time for sharper
    peak targeting.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    beta = 0.9
    v = np.zeros(n)
    base_lr = 2.0 / math.sqrt(n)
    lr = base_lr
    it = 0
    best_it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Cosine annealing for tau
        cos_val = 0.5 * (1 + math.cos(math.pi * progress))
        tau = tau_start + (tau_end - tau_start) * (1 - cos_val)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = a.copy()
        else:
            y = y / s_y

        grad, smooth_val = smoothed_grad(y, tau)

        # Scale gradient: the actual objective gradient is 2*n*grad (for normalized a)
        # But since we're on the simplex, we project the step
        step = lr * grad

        # Backtracking line search
        a_try = a - step
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try > 1e-10:
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
        else:
            obj_try = np.inf

        backtrack = 0
        while obj_try >= best_obj and backtrack < 20:
            lr *= 0.5
            step = lr * grad
            a_try = a - step
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
            else:
                break
            backtrack += 1

        if obj_try < best_obj:
            # Accept
            a_new = a_try
            v = beta * v - (a_new - a)
            a = a_new
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_new.copy()
                best_it = it
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Reject, reset velocity
            v = np.zeros(n)
            no_improve += 1

        it += 1

        # Occasional re-boost
        if no_improve > 500:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * rng.uniform(0.3, 1.0)
            no_improve = 0

        # Slowly decay base learning rate
        lr = max(1e-10, lr * 0.99995)

        if verbose and it % 2000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def difference_set_init(n, rng):
    """Initialize using a difference set structure embedded in length-n sequence."""
    # Use Singer difference sets or known small ones
    # For a (q^2+q+1, q+1, 1) difference set from projective plane of order q
    # We'll create a quasi-difference-set initialization
    a = np.zeros(n)
    # Create a sparse structure with good autocorrelation properties
    # Use a modular arithmetic based construction
    q = max(2, int(math.sqrt(n)) - 1)
    # Singer difference set in Z_{q^2+q+1}
    v = q * q + q + 1
    if v <= n:
        # Generate Singer difference set
        # Elements: {g^{2i} mod (q^2+q+1) : i = 0,...,q} where g is primitive
        # For simplicity, use a heuristic: spread elements with good spacing
        indices = []
        for i in range(q + 1):
            idx = (i * i + i * q) % v
            if idx not in indices:
                indices.append(idx)
        # Pad to get q+1 elements
        while len(indices) < q + 1:
            idx = rng.integers(0, v)
            if idx not in indices:
                indices.append(idx)
        indices = sorted(indices[:q + 1])
        for idx in indices:
            if idx < n:
                a[idx] = 1.0
    else:
        # Fallback: use a geometric progression
        ratio = n / (q + 1)
        for i in range(q + 1):
            idx = int(i * ratio)
            if idx < n:
                a[idx] = 1.0

    # Add some structure: blend with a low-frequency pattern
    if a.sum() < 2:
        a = np.ones(n) / n

    # Blend with uniform for stability
    alpha = 0.3
    a = alpha * a / max(a.sum(), 1e-10) + (1 - alpha) * (np.ones(n) / n)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def fourier_perturbation(a, rng, strength=0.05):
    """Add low-frequency Fourier modes as structured perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(8, max(1, n // 20))
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines LP iteration, adaptive smoothed GD,
    multi-length exploration, and combinatorial initialization.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Initialization ---
    n_current = 800
    inits = []

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                inits.append(('parent', parent, obj))
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        a_init = np.ones(n_current) / n_current
        best_seq = a_init.copy()
        best_score = compute_obj(a_init)
        inits.append(('uniform', a_init, best_score))
        print(f"Default uniform init (n={n_current}): obj={best_score:.8f}")

    # Add difference set init
    a_ds = difference_set_init(n_current, rng)
    obj_ds = compute_obj(a_ds)
    inits.append(('diffset', a_ds, obj_ds))
    print(f"Diffset init: obj={obj_ds:.8f}")

    # Add Fourier-structured init
    x = np.linspace(0, 1, n_current)
    a_four = 0.5 + 0.5 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    a_four = np.maximum(a_four, 0.01)
    a_four = a_four / a_four.sum()
    obj_four = compute_obj(a_four)
    inits.append(('fourier', a_four, obj_four))
    print(f"Fourier init: obj={obj_four:.8f}")

    # --- Phase 1: LP iteration on best initial point ---
    print(f"\n=== Phase 1: LP Iteration ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.20, 250.0)
    if p1_budget > 30:
        start_lp = time.monotonic()
        a_lp = lp_iteration(best_seq, time_budget=p1_budget, verbose=True, seed=42)
        obj_lp = compute_obj(a_lp)
        print(f"  LP done in {time.monotonic()-start_lp:.1f}s: obj={obj_lp:.8f}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** Improved: {best_score:.8f}")

    # --- Phase 2: Adaptive Smoothed GD (main refinement) ---
    print(f"\n=== Phase 2: Adaptive Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 500.0)
    if p2_budget > 30:
        start_gd = time.monotonic()
        a_gd = adaptive_smoothed_gd(best_seq, time_budget=p2_budget,
                                    tau_start=30.0, tau_end=3000.0,
                                    verbose=True, seed=7)
        obj_gd = compute_obj(a_gd)
        print(f"  GD done in {time.monotonic()-start_gd:.1f}s: obj={obj_gd:.8f}")
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  ** Improved: {best_score:.8f}")

    # --- Phase 3: LP polish after GD ---
    print(f"\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)

        a_lp2 = lp_iteration(best_seq, time_budget=p3_budget, verbose=True, seed=100)

        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  ** Improved: {best_score:.8f}")

    # --- Phase 4: Multi-n exploration ---
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_candidates = []
    for dn in [-100, -50, 50, 100, 150, 200, 300, -200, 400, 500]:

            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        elif r == 2:
            # Blend with uniform
            a_r = 0.7 * best_seq + 0.3 * np.ones(n_cur) / n_cur
            a_r = a_r / a_r.sum()
        elif r == 3:

            x = np.linspace(0, 1, n_cur)
            a_struct = 0.5 + 0.5 * np.cos(4 * np.pi * x)
            a_struct = np.maximum(a_struct, 0.01)
            a_r = 0.6 * best_seq + 0.4 * a_struct / a_struct.sum()
            a_r = a_r / a_r.sum()

            a_r = best_seq.copy()

            a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.12, 80.0)

            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30:

        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # One more GD pass
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 80.0)

                                       tau_start=500.0, tau_end=20000.0,

    return result
