def coordinate_descent_step(a, i):
    """Find optimal value of a[i] given all other coordinates fixed."""
    n = len(a)
    x = a[i]
    S = a.sum()
    conv = compute_conv(a)
    
    # Identify k values where conv_k depends on a[i]
    # conv_k = sum_j a[j] * a[k-j]
    # a[i] appears when j=i (if 0<=k-i<n) or k-j=i (if 0<=k-i<n, i.e., j=k-i)
    # So k ranges from i to i+n-1 (for j=i, k-i in [0,n-1] means k in [i, i+n-1])
    # And k ranges from i to i+n-1 (for k-j=i, j=k-i in [0,n-1] means k in [i, i+n-1])
    # So relevant k: [i, min(i+n-1, 2n-2)]
    
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    
    # For each k in [k_start, k_end], compute conv_k(t)
    # and the objective
    def obj_at_t(t):
        S_new = S - x + t
        if S_new < 0.01:
            return np.inf
        max_conv = 0
        for k in range(k_start, k_end + 1):
            # conv_k = conv_k_old + delta
            delta = 0
            if 0 <= k - i < n:
                if k == 2 * i:
                    delta = t*t - x*x
                else:
                    delta = 2*t*a[k-i] - 2*x*a[k-i]
            conv_k_val = conv[k] + delta
            max_conv = max(max_conv, conv_k_val)
        # Also need to check k values outside [k_start, k_end]
        # These don't depend on t, so their max is fixed
        if k_start > 0:
            max_conv = max(max_conv, np.max(conv[:k_start]))
        if k_end < 2*n-2:
            max_conv = max(max_conv, np.max(conv[k_end+1:]))
        return 2.0 * n * max_conv / (S_new * S_new)
    
    # Golden section search
    # t should be in [0, some_upper_bound]
    # Upper bound: maybe 2*x or S (can't be more than total sum)
    lo, hi = 0.0, max(2*x, S)
    # Actually t=0 is valid, and very large t makes the sum large but max_conv also large
    # Let's use [0, 3*max(x, S/n)]
    hi = 3.0 * max(x, S / n)
    
    gr = (np.sqrt(5) - 1) / 2
    a1 = hi - gr * (hi - lo)
    b1 = lo + gr * (hi - lo)
    fa1 = obj_at_t(a1)
    fb1 = obj_at_t(b1)
    
    for _ in range(50):
        if fa1 < fb1:
            hi = b1
            b1 = a1
            fb1 = fa1
            a1 = hi - gr * (hi - lo)
            fa1 = obj_at_t(a1)
        else:
            lo = a1
            a1 = b1
            fa1 = fb1
            b1 = lo + gr * (hi - lo)
            fb1 = obj_at_t(b1)
    
    t_opt = (lo + hi) / 2
    return t_opt, obj_at_t(t_opt)

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

                    update_conv_incremental(conv, a, i, x_old, t_opt)
                    new_max = float(np.max(conv))
                    cur_obj = 2.0 * n * new_max
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
            sweep += 1
            if best_obj < sweep_obj_start - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            if verbose and sweep % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    CD sweep {sweep}: obj={best_obj:.8f}, stall={stall_count}, time={elapsed:.0f}s")
            if stall_count >= 8:
                # Perturb to escape local minimum
                a = best_a.copy()
                tau = 200.0
                conv_p = compute_conv(a)
                max_p = float(np.max(conv_p))
                exp_p = np.exp(tau * (conv_p - max_p))
                p = exp_p / exp_p.sum()
                N_fft = 1
                while N_fft < 3 * n:
                    N_fft <<= 1
                P = fft(p, N_fft)
                Af = fft(a, N_fft)
                grad = 2.0 * ifft(P * np.conj(Af)).real[:n]
                lr_p = 0.01 / (float(np.linalg.norm(grad)) + 1e-10)
                a = a - lr_p * grad
                a = np.maximum(a, 0)
                a = a + 0.002 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                s_new = float(np.sum(a))
                if s_new < 1e-10:
                    a = best_a.copy()
                else:
                    a = a / s_new
                conv = compute_conv(a)
                stall_count = 0
                if time.monotonic() - start > time_budget:
                    break
                if verbose:
                    print(f"    CD sweep {sweep}: perturbed, obj={2.0*n*float(np.max(conv)):.8f}")

    return best_a


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) (sum=1 normalization)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def gd_polish(a, time_budget=100.0, tau_init=100.0, tau_end=3000.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    beta = 0.9
    base_lr = 5.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.2 * base_lr), 3.0 * base_lr)
            lr = 0.75 * lr + 0.25 * adapt

        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 5.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        conv_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(conv_new))
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9997)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.4
            best_it = it

    if verbose:
        print(f"    GD: obj={best_obj:.8f}, iters={it}")
    return best_a


def build_conv_matrix_fast(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_fast(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g):
    """Line search: find best t in (0,1) to minimize obj((1-t)*a + t*g)."""
    n = len(a)
    ts = np.array([0.01, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.7, 0.9])
    objs = np.full(len(ts), np.inf)
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            a_try = a_try / s
            objs[i] = 2.0 * n * float(np.max(compute_conv(a_try)))
    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]
    # Parabolic refinement
    if 0 < best_idx < len(ts) - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.01, 0.99))
            a_par = (1 - t_par) * a + t_par * g
            a_par = np.maximum(a_par, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                a_par = a_par / s_par
                f_par = 2.0 * n * float(np.max(compute_conv(a_par)))
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par
    return best_t, best_obj


def lp_driven_optimize(a, time_budget=200.0, verbose=False):
    """LP-driven optimization loop."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    stall = 0
    it = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - start < time_budget:
        if time_budget - (time.monotonic() - start) < 3.0:
            break
        g, M = solve_lp_minpeak(a)
        if g is None:
            break
        best_t, obj_at_t = line_search(a, g)
        it += 1
        if obj_at_t < best_obj - 1e-12:
            a_try = (1 - best_t) * a + best_t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                best_obj = obj_at_t
                best_a = a_try.copy()
                a = a_try
                stall = 0
                if verbose and it % 5 == 0:
                    print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            stall += 1
            if stall <= 2:
                for t_small in [0.03, 0.01]:
                    a_small = (1 - t_small) * a + t_small * g
                    a_small = np.maximum(a_small, 0)
                    s_small = float(np.sum(a_small))
                    if s_small > 1e-10:
                        a_small = a_small / s_small
                        obj_small = 2.0 * n * float(np.max(compute_conv(a_small)))
                        if obj_small < best_obj - 1e-12:
                            best_obj = obj_small
                            best_a = a_small.copy()
                            a = a_small
                            stall = 0
                            break
            if stall >= 4:
                # Subgradient perturbation
                conv_p = compute_conv(a)
                max_p = float(np.max(conv_p))
                tau_p = 300.0
                exp_p = np.exp(tau_p * (conv_p - max_p))
                p = exp_p / exp_p.sum()
                N_fft = 1
                while N_fft < 3 * n:
                    N_fft <<= 1
                P = fft(p, N_fft)
                Af = fft(a, N_fft)
                grad = 2.0 * ifft(P * np.conj(Af)).real[:n]
                lr_p = 0.015 / (float(np.linalg.norm(grad)) + 1e-10)
                a = a - lr_p * grad
                a = np.maximum(a, 0)
                a = a + 0.003 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                else:
                    a = best_a.copy()
                stall = 0
                if it > 80:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 6):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
