```python
def mirror_descent(a, time_budget, beta_schedule):
    """
    Mirror descent (multiplicative weights) on the simplex.
    Minimizes smooth approximation of max(conv(a,a)).
    """
    n = len(a)
    a = a / np.sum(a)  # normalize to simplex
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    start = time.monotonic()
    eta = 0.1  # step size
    beta = 10.0  # initial smoothing parameter
    
    while time.monotonic() - start < time_budget:
        # Compute softmax weights
        c = compute_conv(a)
        scaled = beta * c
        max_s = np.max(scaled)
        p = np.exp(scaled - max_s)
        p = p / np.sum(p)
        
        # Compute gradient
        g = compute_gradient_direct(a, p)
        
        # Mirror descent update: a_i *= exp(-eta * g_i)
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
            a = a_new
        
        # Adaptive step size and beta
        # Decrease eta slowly, increase beta slowly
        eta *= 0.999
        beta *= 1.001
    
    return best_a
```

```python
import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def best_response(a, n_inner=200, beta=50.0, time_budget=0.05):
    """
    Find g on simplex that minimizes max_k (a*g)_k.
    Uses mirror descent on the smoothed objective.
    """
    n = len(a)
    g = np.ones(n) / n  # start uniform
    
    start = time.monotonic()
    eta = 0.5  # step size for mirror descent
    best_g = g.copy()
    best_val = np.inf
    
    for it in range(n_inner):
        if time.monotonic() - start > time_budget:
            break
            
        # Compute a*g (cross-correlation / convolution)
        ag = compute_conv_cross(a, g)  # length 2n-1
        
        # Softmax
        scaled = beta * ag
        max_s = np.max(scaled)
        p = np.exp(scaled - max_s)
        p_sum = np.sum(p)
        if p_sum < 1e-300:
            break
        p = p / p_sum
        
        # Compute gradient: grad_j = sum_k p_k * a_{k-j} (if 0 ≤ k-j < n)
        # = cross-correlation of p and a
        # grad_j = sum_k p_k * a_{k-j} for 0≤k-j<n
        # = sum_{m=0}^{n-1} p[m+j] * a[m] (where m = k-j, k = m+j)
        # This is the cross-correlation of p and a at lag j
        grad = np.zeros(n)
        for j in range(n):
            # k ranges from j to j+n-1, a_{k-j} = a_0,...,a_{n-1}
            # p[k] for k=j,...,j+n-1
            max_k = min(2*n-1, j+n)
            if j < 2*n-1:
                grad[j] = np.dot(p[j:min(2*n-1, j+n)], a[:min(2*n-1-j, n)])
        
        # Mirror descent update
        g_new = g * np.exp(-eta * grad)
        g_new = np.maximum(g_new, 0)
        s = np.sum(g_new)
        if s > 1e-10:
            g_new = g_new / s
            ag_new = compute_conv_cross(a, g_new)
            val = float(np.max(ag_new))
            if val < best_val:
                best_val = val
                best_g = g_new.copy()
            g = g_new
        
        eta *= 0.999
    
    return best_g
```

```python
def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:

    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def grad_cross(a, p):
    """Compute grad_j = sum_{m=0}^{n-1} a[m] * p[m+j] for j=0,...,n-1 using FFT.
    a: length n, p: length 2n-1"""
    n = len(a)
    L = 1
    while L < 4 * n:
        L <<= 1
    a_rev = a[::-1]
    conv = irfft(rfft(a_rev, L) * rfft(p, L), L)[:3 * n - 2]
    return conv[n - 1:2 * n - 1]

def line_search(a, g, max_evals=30):
    """Golden section search for optimal blend t."""
    # Precompute convolutions
    aa = compute_conv(a)
    ag = compute_conv_cross(a, g)
    gg = compute_conv(g)
    n = len(a)
    
    def obj_at(t):
        vals = (1-t)**2 * aa + 2*t*(1-t) * ag + t**2 * gg
        return 2.0 * n * float(np.max(vals))
    
    # Golden section
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    lo, hi = 0.0, 1.5
    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)

            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

def solve_lp(a):
    """LP polish: min M s.t. (a*g)_k <= M, sum g = 1, g >= 0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Build convolution matrix efficiently
    # (a*g)_k = sum_i a_i * g_{k-i}
    # A[k, j] = a_{k-j} if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        valid_k = range(j, min(j+n, 2*n-1))
        for k in valid_k:
            A[k, j] = a_norm[k-j]
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None
```

def compute_conv(a):
