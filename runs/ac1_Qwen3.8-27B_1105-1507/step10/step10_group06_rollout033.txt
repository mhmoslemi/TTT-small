import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa).real[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_max_conv(a, w):
    """
    Compute gradient of sum_k w[k] * conv(a,a)[k] w.r.t. a.
    w is a weight vector of length 2n-1 (from active set).
    Returns gradient of length n.
    """
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    W = fft(w, N)
    A = fft(a, N)
    cross = ifft(np.conj(A) * W).real
    return 2.0 * cross[:n]

            for t in candidates_t:
                a_try = (1.0 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_local_a = a_try.copy()
            
            if best_local_obj < best_obj - 1e-14:
                best_obj = best_local_obj
                best_a = best_local_a
                a = best_a.copy()
                if verbose and iter_count % 10 == 0:
                    print(f"    LP {iter_count}: obj={best_obj:.10f}")
            else:
                # Mirror descent polish
                md_budget = min(remaining * 0.2, 30.0)
                a = mirror_descent_polish(a, time_budget=md_budget)
                obj_md = compute_obj(a)
                if obj_md < best_obj:
                    best_obj = obj_md
                    best_a = a.copy()
        else:
            # LP failed, use mirror descent
            md_budget = min(remaining * 0.3, 60.0)
            a = mirror_descent_polish(a, time_budget=md_budget)
            obj_md = compute_obj(a)
            if obj_md < best_obj:
                best_obj = obj_md
                best_a = a.copy()
    
    return best_a


def structured_perturb(a, rng, mode=0):
    """Apply structured perturbation to escape local minima."""
    n = len(a)
    if mode == 0:
        # Fourier perturbation
        pert = np.zeros(n)
        for k in range(1, 8):
            freq = k * 2 * np.pi / n
            amp = 0.02 * rng.uniform(0.5, 1.5) / k
            phase = rng.uniform(0, 2 * np.pi)
            pert += amp * np.cos(freq * np.arange(n) + phase)
        a_new = a * np.exp(pert)
    elif mode == 1:
        # Block perturbation
        block_size = n // 8
        blocks = np.zeros(n)
        for i in range(n // block_size):
            blocks[i*block_size:(i+1)*block_size] = rng.uniform(-0.05, 0.05)
        a_new = a * np.exp(blocks)
    else:
        # Random multiplicative perturbation
        a_new = a * np.exp(0.01 * rng.standard_normal(n))
    
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
