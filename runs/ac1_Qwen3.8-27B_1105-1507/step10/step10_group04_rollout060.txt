"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based minimax iteration, coordinate descent polishing, high-precision
smoothed gradient descent, and multi-length exploration.
Key improvements: coordinate descent with incremental conv updates, better
line search, dual-based termination, wider n sweep, and adaptive time allocation.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:
            continue
    return None, None

    beta = 0.85
    v = np.zeros(n)
    it = 0
    best_it = 0

    N = 1
    while N < 3 * n:
        N <<= 1

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.7

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        P = fft(p, N)
        A_f = fft(y, N)
        g_full = ifft(P * np.conj(A_f)).real
        g = 2.0 * g_full[:n]

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        lr = min(1.0 / (g_norm + 1e-10), 3.0 / np.sqrt(n))
        lr = max(lr, 1e-10)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        beta = min(0.95, beta + 0.00005)

        if it - best_it > 50000:
            a = best_a.copy()
            v = np.zeros(n)
            if it > 100000:
                break

        if verbose and it % 2000 == 0:
            print(f"  GD it{it}: obj={best_obj:.10f}, tau={tau:.0f}, time={elapsed:.0f}s")

    return best_a


def log_param_optimize(a, time_budget, tau_min=200, tau_max=8000, verbose=True):
    """Optimize using log-parameterization with L-BFGS."""
    full_len = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    threshold = 1e-8
    support = np.where(a > threshold)[0]
    if len(support) < 3:
        support = np.where(a > a.max() * 1e-10)[0]
    if len(support) < 3:
        return a

    b0 = np.log(np.maximum(a[support], 1e-30))
    b0 = b0 - b0.mean()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau_stages = [tau_min, tau_min * 3, tau_min * 9, tau_min * 27, tau_max]
    current_b = b0.copy()

    N = 1
    while N < 3 * full_len:
        N <<= 1

    for stage_idx, tau in enumerate(tau_stages):
        if time.monotonic() - start > time_budget:
            break

        def obj_grad(b_vec):
            b_shifted = b_vec - b_vec.max()
            a_supp = np.exp(b_shifted)
            a_supp = a_supp / a_supp.sum()
            a_full = np.zeros(full_len)
            a_full[support] = a_supp
            conv = compute_conv(a_full)
            max_conv = float(np.max(conv))
            exp_vals = np.exp(tau * (conv - max_conv))
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp
            obj_val = max_conv + np.log(sum_exp) / tau
            P = fft(p, N)
            A_f = fft(a_full, N)
            g_full = ifft(P * np.conj(A_f)).real
            grad_full = 2.0 * g_full[:full_len]
            grad_b = a_supp * (grad_full[support] - np.sum(a_supp * grad_full[support]))
            return obj_val, grad_b

        try:
            res = scipy_minimize(obj_grad, current_b, method='L-BFGS-B',
                                 jac=True, options={'maxiter': 300, 'gtol': 1e-10,
                                                    'ftol': 1e-14, 'maxfun': 800})
            if res.x is not None:
                b_shifted = res.x - res.x.max()
                a_supp = np.exp(b_shifted)
                a_supp = a_supp / a_supp.sum()
                a_new = np.zeros(full_len)
                a_new[support] = a_supp
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    current_b = res.x.copy()
                    if verbose:
                        print(f"  LogOpt stage{stage_idx}: tau={tau:.0f}, obj={best_obj:.10f}")
        except Exception:
            pass

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 1.5 * (1 - x) ** 0.5
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.25))
        a = a / a.sum()
    elif mode == 5:
        a = np.exp(-np.arange(n) / (n * 0.15)) + 0.5 * np.exp(-np.arange(n) / (n * 0.6))
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x ** 2 * (1 - x) ** 2
        a = a / a.sum()
    elif mode == 7:
        # Asymmetric peak
        center = n * 0.3
        a = np.exp(-((np.arange(n) - center) ** 2) / (2 * (n * 0.2) ** 2))
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP iteration, coordinate descent, high-tau GD,
    log-param optimization, multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = diverse_init(n_init, rng, mode=4)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.10f}")

    n_best = len(best_seq)

    # === Phase 1: LP Iteration (300s) ===
    print("\n=== Phase 1: LP Iteration ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.22, 300.0)
    if p1_budget > 30:
        a_lp = lp_iterate(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 done: obj={best_score:.10f} **")
        else:
            print(f"  Phase1: objective unchanged at {best_score:.10f}")

    # === Phase 2: Coordinate Descent (150s) ===
    print("\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.1, 150.0)
    if p2_budget > 20:
        a_cd = coordinate_descent(best_seq, time_budget=p2_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase2 done: obj={best_score:.10f} **")
        else:
            print(f"  Phase2: objective unchanged at {best_score:.10f}")

    # === Phase 3: High-Tau GD (200s) ===
    print("\n=== Phase 3: High-Tau GD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)
    if p3_budget > 20:
        a_gd = high_tau_gd(best_seq, time_budget=p3_budget,
                           tau_start=100, tau_end=8000, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase3 done: obj={best_score:.10f} **")
        else:
            print(f"  Phase3: objective unchanged at {best_score:.10f}")

    # === Phase 4: Log-Param Optimization (100s) ===
    print("\n=== Phase 4: Log-Param Optimization ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 100.0)
    if p4_budget > 15:
        a_log = log_param_optimize(best_seq, time_budget=p4_budget,
                                   tau_min=200, tau_max=8000, verbose=True)
        obj_log = compute_obj(a_log)
        if obj_log < best_score:
            best_score = obj_log
            best_seq = a_log.copy()
            print(f"  Phase4 done: obj={best_score:.10f} **")
        else:
            print(f"  Phase4: objective unchanged at {best_score:.10f}")

    # === Phase 5: Multi-n Exploration (350s) ===
    print("\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.25, 350.0)
    n_best = len(best_seq)

    n_options = sorted(set([
        max(300, n_best - 200), max(300, n_best - 100),
        n_best + 100, n_best + 200, n_best + 400,
        n_best + 600, n_best + 1000, n_best + 1500,
        600, 900, 1200, 1600, 2000, 2500, 3000
    ]))
    n_options = [n for n in n_options if 400 <= n <= 4000 and n != n_best]
    # Sort by proximity to n_best first, then by size
    n_options.sort(key=lambda x: (abs(x - n_best), x))
    n_options = n_options[:10]
    print(f"  Trying n values: {n_options}")

    per_n_budget = p5_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(per_n_budget, remaining - 30)
        if alloc < 30:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try > 1e-10:
            a_try = a_try / s_try

        # LP polish
        lp_alloc = min(alloc * 0.35, remaining - 20)
        if lp_alloc > 10:
            a_try = lp_iterate(a_try, time_budget=lp_alloc, verbose=False)

        # Coordinate descent
        remaining = deadline - time.monotonic()
        cd_alloc = min(alloc * 0.25, remaining - 15)
        if cd_alloc > 8:
            a_try = coordinate_descent(a_try, time_budget=cd_alloc, verbose=False)

        # GD
        remaining = deadline - time.monotonic()
        gd_alloc = min(alloc * 0.25, remaining - 10)
        if gd_alloc > 8:
            a_try = high_tau_gd(a_try, time_budget=gd_alloc,
                                tau_start=50, tau_end=4000, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 6: Diverse Restarts (150s) ===
    print("\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p6_budget = min(remaining * 0.1, 150.0)
    n_current = len(best_seq)
    n_restarts = max(1, int(p6_budget / 50))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(50, remaining - 15)
        if alloc < 20:
            break

        mode = (r + 2) % 8
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.2 + 0.08 * r
        a_r = np.maximum((1 - blend) * a_r + blend * best_seq, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        half = alloc * 0.5
        a_r = lp_iterate(a_r, time_budget=half, verbose=False)
        remaining = deadline - time.monotonic()
        cd_alloc = min(alloc * 0.3, remaining - 10)
        if cd_alloc > 5:
            a_r = coordinate_descent(a_r, time_budget=cd_alloc, verbose=False)
        remaining = deadline - time.monotonic()
        gd_alloc = min(alloc * 0.2, remaining - 8)
        if gd_alloc > 5:
            a_r = high_tau_gd(a_r, time_budget=gd_alloc, tau_start=50, tau_end=3000, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.10f}")

    # === Phase 7: Final Intensive Polish (remaining time) ===
    print("\n=== Phase 7: Final Intensive Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s")

    # Alternate LP + CD + GD cycles
    cycle = 0
    while remaining > 30:
        cycle += 1
        # LP
        alloc = min(remaining * 0.25, 80.0)
        if alloc > 15:
            a_f = lp_iterate(best_seq, time_budget=alloc, verbose=False)
            obj_f = compute_obj(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Cycle {cycle} LP: obj={best_score:.10f} **")

        # CD
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 60.0)
        if alloc > 10:
            a_f = coordinate_descent(best_seq, time_budget=alloc, verbose=False)
            obj_f = compute_obj(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Cycle {cycle} CD: obj={best_score:.10f} **")

        # GD
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.25, 80.0)
        if alloc > 10:
            a_f = high_tau_gd(best_seq, time_budget=alloc,
                              tau_start=500, tau_end=12000, verbose=False)
            obj_f = compute_obj(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Cycle {cycle} GD: obj={best_score:.10f} **")

        # Log-param
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.15, 40.0)
        if alloc > 8:
            a_f = log_param_optimize(best_seq, time_budget=alloc,
                                     tau_min=500, tau_max=12000, verbose=False)
            obj_f = compute_obj(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Cycle {cycle} Log: obj={best_score:.10f} **")

        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
