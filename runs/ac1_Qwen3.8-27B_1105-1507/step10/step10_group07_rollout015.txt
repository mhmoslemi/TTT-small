"""
Multi-phase optimizer for minimum peak convolution objective.
Uses power-mean smoothed gradient descent with momentum, LP-based direction finding,
and multi-length exploration. Key innovations: FFT-based gradient of power-mean objective,
adaptive power schedule, diverse initialization, and efficient LP polishing with
active-set reduction.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_power_mean(a, p):
    """
    Compute gradient of f_p(a) = (sum_k c[k]^p)^(1/p) where c = conv(a,a).
    Returns (gradient, f_p_value).
    """
    n = len(a)
    c = compute_conv(a)
    # Clip to avoid issues with 0^p
    c_safe = np.maximum(c, 1e-30)
    f_p = float(np.sum(c_safe ** p)) ** (1.0 / p)

    # w[k] = c[k]^(p-1)
    w = c_safe ** (p - 1)

    # Gradient: df_p/da_i = f_p^(1-p) * sum_k w[k] * 2*a[k-i] (for valid k-i)
    # sum_k w[k] * a[k-i] is cross-correlation of w and a
    # We compute: for each i, sum over k of w[k] * a[k-i]
    # This is convolve(w, reverse(a)) evaluated at appropriate indices
    # Actually: sum_k w[k] * a[k-i] where we need 0 <= k-i < n
    # Let's use FFT: this is the cross-correlation of w and a

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    W = rfft(w, size)
    A = rfft(a, size)
    # Cross-correlation: sum_k w[k] * a[k-i]
    # This is ifft(conj(W) * A) or ifft(W * conj(A)) depending on convention
    # sum_k w[k] * a[k-i] = ifft(W * conj(A))[i] ... let me verify
    # Actually: ifft(W[k] * conj(A[k]))[i] = sum_k w[k] * a[k-i] (circular)
    # With zero-padding to size >= 2n-1, this equals linear cross-correlation
    xc = irfft(W * np.conj(A), size)[:n]

    gradient = 2.0 * (f_p ** (1.0 - p)) * xc
    return gradient, f_p


def gradient_descent_power_mean(a, time_budget, p_start=8.0, p_end=200.0,
                                verbose=False, seed=42):
    """
    Gradient descent on power-mean objective with momentum and adaptive step size.
    Optimizes over the probability simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    lr = 1.0 / n * 10.0
    momentum = 0.85
    v = np.zeros(n)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-6))
        p = p_start * (p_end / p_start) ** (progress ** 0.6)

        grad, f_val = compute_gradient_power_mean(a, p)
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        # Adaptive step: if obj increased, reduce lr
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0

def build_conv_matrix_sparse(a):
    """Build the convolution matrix A where A @ x = conv(a, x) using FFT-based approach."""
    # Instead of building explicitly, we'll use it in the LP via the structure
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, M_target=None, top_k=None):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M for all k, sum(g)=1, g>=0.
    Optionally only enforce top_k constraints for speed.
    Returns (g, M_value) or (None, None).
    """
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix_sparse(a)

    if top_k is not None and top_k < 2 * n - 1:
        c_current = compute_conv(a)
        # Keep the top_k indices where c_current is largest
        indices = np.argsort(c_current)[::-1][:top_k]
        indices_sorted = np.sort(indices)
        A = A[indices_sorted, :]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def peak_shaving_refine(a, time_budget, K=16):
    """Subgradient method focusing on top-K peaks with adaptive weighting."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 5.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break

        # Top-K peaks with exponential weighting
        sorted_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[sorted_idx]

        # Temperature-based weighting
        temp = max_conv * 0.3
        weights = np.exp((peak_vals - max_conv) / temp)
        weights /= float(np.sum(weights))

        # Compute subgradient
        g = np.zeros(n)
        for w, k in zip(weights, sorted_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.005, 30.0 / np.sqrt(n))
        else:
            stall += 1
            lr *= 0.99
            if stall > 300:
                lr = 5.0 / np.sqrt(n)
                v *= 0.5
                stall = 0
                if it > 5000:
                    break

        it += 1

    return best_a


def diverse_init(n, seed=0):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear ramp
    x = np.linspace(0, 1, n)
    a = x + 0.01
    inits.append(a / a.sum())

    # Exponential decay
    a = np.exp(-np.linspace(0, 3, n))
    inits.append(a / a.sum())

    # Quadratic (bell-like)
    x = np.linspace(0, 1, n)
    a = np.exp(-10 * (x - 0.5) ** 2)
    inits.append(a / a.sum())

    # Random (Dirichlet)
    a = rng.dirichlet(np.ones(n) * 2.0)
    inits.append(a)

    # Bimodal
    x = np.linspace(0, 1, n)
    a = np.exp(-20 * (x - 0.2) ** 2) + np.exp(-20 * (x - 0.8) ** 2)
    inits.append(a / a.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Multi-phase optimizer combining power-mean gradient descent,
    LP-based polishing, peak shaving, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_parent}): obj={best_score:.8f}")

    # === Phase 1: Gradient descent on parent length ===
    print(f"\n=== Phase 1: Gradient Descent (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.20, 200.0)
    if p1_budget > 20:
        a_gd = gradient_descent_power_mean(best_seq, time_budget=p1_budget,
                                           p_start=8.0, p_end=300.0,
                                           verbose=True, seed=seed)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  GD: obj={best_score:.8f} **")
        else:
            print(f"  GD: obj={obj_gd:.8f}")

    # === Phase 2: Peak shaving ===
    print(f"\n=== Phase 2: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 100.0)
    if p2_budget > 15:
        a_ps = peak_shaving_refine(best_seq, time_budget=p2_budget, K=16)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  PS: obj={best_score:.8f} **")
        else:
            print(f"  PS: obj={obj_ps:.8f}")

    # === Phase 3: LP Polish ===
    print(f"\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.25, 250.0)
    if p3_budget > 25:
        a_lp = lp_polish(best_seq, time_budget=p3_budget, verbose=True, seed=seed + 1)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Try different lengths
    n_options = sorted(set([
        max(400, n_current - 200),
        n_current,
        n_current + 200,
        n_current + 400,
        1000, 1200, 1500, 1800, 2000
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2500]
    # Remove the current length (already optimized)
    n_options_other = [n for n in n_options if abs(n - n_current) > 50]

    for n_try in n_options_other[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)

        # Quick gradient optimization
        alloc1 = min(remaining * 0.15, 60.0)
        if alloc1 > 10:
            a_try = gradient_descent_power_mean(a_try, time_budget=alloc1,
                                                p_start=8.0, p_end=200.0,
                                                seed=seed + n_try)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 80.0)
        if alloc2 > 15:
            a_try = lp_polish(a_try, time_budget=alloc2, seed=seed + n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts on best length ===
    print(f"\n=== Phase 5: Diverse Restarts (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    inits = diverse_init(n_best, seed=seed + 100)
    # Also add perturbed parent
    for scale in [0.02, 0.05, 0.1]:
        a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
        a_pert = np.maximum(a_pert, 0)
        s_p = a_pert.sum()
        if s_p > 1e-10:
            a_pert = a_pert / s_p
            inits.append(a_pert)

    for idx, init in enumerate(inits):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        alloc_grad = min(remaining * 0.12, 40.0)
        alloc_lp = min(remaining * 0.08, 25.0)

        if alloc_grad > 8:
            a_cand = gradient_descent_power_mean(init, time_budget=alloc_grad,
                                                 p_start=8.0, p_end=150.0,
                                                 seed=seed + 200 + idx)
        else:
            a_cand = init.copy()

        remaining = deadline - time.monotonic()
        if remaining > alloc_lp + 5 and alloc_lp > 8:
            a_cand = lp_polish(a_cand, time_budget=alloc_lp, seed=seed + 300 + idx)

        obj_cand = compute_obj(a_cand)
        if obj_cand < best_score:
            best_score = obj_cand
            best_seq = a_cand.copy()
            print(f"  Restart {idx}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {idx}: obj={obj_cand:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Alternate GD and LP
        for round_idx in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            alloc_gd = min(remaining * 0.2, 40.0)
            alloc_lp = min(remaining * 0.2, 40.0)

            if alloc_gd > 10:
                a_f = gradient_descent_power_mean(best_seq, time_budget=alloc_gd,
                                                  p_start=50.0, p_end=500.0,
                                                  seed=seed + 400 + round_idx)
                obj_f = compute_obj(a_f)
                if obj_f < best_score:
                    best_score = obj_f
                    best_seq = a_f.copy()

            remaining = deadline - time.monotonic()
            if remaining > alloc_lp + 5 and alloc_lp > 10:
                a_f2 = lp_polish(best_seq, time_budget=alloc_lp,
                                 seed=seed + 500 + round_idx)
                obj_f2 = compute_obj(a_f2)
                if obj_f2 < best_score:
                    best_score = obj_f2
                    best_seq = a_f2.copy()
                    print(f"  Final round {round_idx}: obj={best_score:.8f} **")

    # === Final peak shaving ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_final_ps = peak_shaving_refine(best_seq, time_budget=min(remaining * 0.3, 30.0), K=20)
        obj_final_ps = compute_obj(a_final_ps)
        if obj_final_ps < best_score:
            best_score = obj_final_ps
            best_seq = a_final_ps.copy()
            print(f"  Final PS: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
