def compute_conv(a):

    fa = rfft(a, size)

"""
Optimizes step function coefficients using Multiplicative Peak Reduction (MPR) with
FFT-based gradient computation, Nesterov-accelerated smoothed gradient descent, and
exact coordinate descent refinement. Key innovations: O(n log n) per-iteration cost
enabling 100k+ iterations, multiplicative updates that preserve non-negativity
inherently, and adaptive smoothing parameter annealing from smooth to sharp.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def objective_from_conv(conv, n):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    return 2.0 * n * float(np.max(conv))


def compute_grad_smoothed(a, p):
    """
    Compute gradient of smoothed max of conv(a,a) w.r.t. a.
    p is the softmax/attention vector (length 2n-1).
    Returns gradient vector of length n.
    g[i] = 2 * sum_j p[i+j]*a[j] - p[2i]*a[i]
    """
    n = len(a)
    h = np.correlate(p, a, mode='valid')
    g = 2.0 * h
    idx = 2 * np.arange(n)
    valid = idx < len(p)
    g[valid] -= p[idx[valid]] * a[valid]
    return g


def softmax_tau(conv, tau):
    """Compute softmax with temperature tau."""
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros_like(conv)
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    return p


def multiplicative_peak_reduction(a, time_budget=400.0, tau_init=10.0, tau_max=500.0,
                                  eta_init=None, verbose=True, seed=42):
    """
    Multiplicative Peak Reduction: a[i] *= exp(-eta * (g[i]-mean(g)))
    with adaptive tau and eta. Preserves non-negativity automatically.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    if eta_init is None:
        eta_init = 2.0 / np.sqrt(n)

    conv = compute_conv(a)
    best_obj = float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    eta = eta_init
    stall_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_max - tau_init) * progress

        conv = compute_conv(a)
        p = softmax_tau(conv, tau)
        g = compute_grad_smoothed(a, p)
        g_proj = g - np.mean(g)

        # Adaptive eta: reduce if oscillating
        if it > 10:
            eta *= 0.9995
        eta = max(eta, 1e-8)

        # Multiplicative update
        update = np.exp(-eta * g_proj)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-30)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        conv_new = compute_conv(a_new)
        obj_new = float(np.max(conv_new))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 500:
                # Reset eta and try again
                eta = eta_init * (1.0 + 0.1 * np.random.default_rng(seed + it).random())
                stall_count = 0

        a = a_new
        it += 1

        if verbose and it % 5000 == 0:
            print(f"    MPR it{it}: obj={objective_from_conv(conv_new, n):.8f}, "
                  f"tau={tau:.0f}, eta={eta:.6f}, t={elapsed:.1f}s")

    return best_a


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=20.0, tau_max=1000.0,
                              verbose=True, seed=42):
    """Nesterov accelerated projected gradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    best_obj = float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()

    beta = 0.9
    v = np.zeros(n)
    lr = 0.5 / np.sqrt(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_max - tau_init) * progress

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient
        conv_y = compute_conv(y)
        p = softmax_tau(conv_y, tau)
        g = compute_grad_smoothed(y, p)
        g = g - np.mean(g)  # Project to tangent space

        # Adaptive step size
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-15:
            max_allowed = float(np.min(y[y > 1e-6])) / (g_norm + 1e-10) * 0.5 if np.any(y > 1e-6) else 0.1
            lr = min(lr * 1.001, max_allowed)
            lr = max(lr, 0.001 / (np.sqrt(n) * (1.0 + tau * 0.1)))
        else:
            lr = 0.5 / np.sqrt(n)

        v = beta * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        conv_new = compute_conv(a_new)
        obj_new = float(np.max(conv_new))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 3000 == 0:
            print(f"    NGD it{it}: obj={objective_from_conv(conv_new, n):.8f}, "
                  f"tau={tau:.0f}, lr={lr:.6f}, t={elapsed:.1f}s")

    return best_a


def coordinate_descent(a, time_budget=300.0, verbose=True, seed=42):
    """Exact coordinate descent with golden section search on 1D subproblems."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    best_obj = float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 5 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks]

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]
            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            def f_1d(x, v=v, S_other=S_other, n=n, c_k_lin=c_k_lin,
                     a_ki_lin=a_ki_lin, c_2i=c_2i):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            scale = max(v, S / n, 1e-6)
            candidates = [0.0, v * 0.5, v, v * 1.5, v * 2.0]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(30):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={2*n*best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2 + 0.1
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)

    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick MPR
        alloc = min(remaining * 0.2, 60.0)
        if alloc > 20:
            a_try = multiplicative_peak_reduction(a_try, time_budget=alloc,
                                                  tau_init=20.0, tau_max=300.0,
                                                  verbose=False, seed=n_try)
            # Quick CD
            cd_alloc = min(remaining * 0.1, 30.0)
            if cd_alloc > 10:
                a_try = coordinate_descent(a_try, time_budget=cd_alloc,
                                           verbose=False, seed=n_try + 500)

        conv_try = compute_conv(a_try)
        obj_try = objective_from_conv(conv_try, a_try.size)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = 3 + r
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.3 + 0.2 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.3, 80.0)
        if alloc > 25:
            a_r = multiplicative_peak_reduction(a_r, time_budget=alloc * 0.6,
                                                tau_init=30.0, tau_max=400.0,
                                                verbose=False, seed=5000 + r * 100)
            a_r = coordinate_descent(a_r, time_budget=alloc * 0.4,
                                     verbose=False, seed=5000 + r * 100 + 50)
            conv_r = compute_conv(a_r)
            obj_r = objective_from_conv(conv_r, a_r.size)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final Coordinate Descent Polish ===
    print(f"\n=== Phase 7: Final CD Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.7, 200.0)
        a_final = coordinate_descent(best_seq, time_budget=alloc,
                                     verbose=False, seed=seed + 9999)
        conv_final = compute_conv(a_final)
        obj_final = objective_from_conv(conv_final, a_final.size)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # Final MPR polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_fmp = multiplicative_peak_reduction(best_seq, time_budget=min(remaining * 0.5, 60.0),
                                              tau_init=100.0, tau_max=800.0,
                                              verbose=False, seed=seed + 8888)
        conv_fmp = compute_conv(a_fmp)
        obj_fmp = objective_from_conv(conv_fmp, a_fmp.size)
        if obj_fmp < best_score:
            best_score = obj_fmp
            best_seq = a_fmp.copy()
            print(f"  Final MPR: obj={best_score:.8f} **")

    # Symmetrize if it helps
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_sym = 0.5 * (best_seq + best_seq[::-1])
        a_sym = np.maximum(a_sym, 0)
        a_sym = a_sym / a_sym.sum()
        conv_sym = compute_conv(a_sym)
        obj_sym = objective_from_conv(conv_sym, a_sym.size)
        if obj_sym < best_score:
            best_score = obj_sym
            best_seq = a_sym.copy()
            print(f"  Symmetrized: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = objective_from_conv(compute_conv(best_seq), 200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
