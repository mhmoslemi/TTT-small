    # Actually: sum_m a[m] * p[m+j] = ifft(fft(a) * fft(p)[::-1])...

    # Pad a to length N: A = [a[0], a[1], ..., a[n-1], 0, ...]
    # We want: for each j, sum_{m=0}^{n-1} a[m] * p[m+j]
    # = circular convolution of a with p at offset j...

    # = ifft(fft(b, N) * conj(fft(c, N)))[-j mod N]
    # But we only need j = 0, ..., n-1

    # = 2 * sum_k p[k] * a[k-j]

    # This equals (p * a_rev)[j + n - 1] where a_rev[m] = a[n-1-m]
    # Because (p * a_rev)[n-1+j] = sum_k p[k] * a_rev[n-1+j-k] = sum_k p[k] * a[k-j]
    # (when 0 <= k-j < n, i.e., a_rev[n-1+j-k] = a[k-j])

def gradient_descent_smoothed(a, time_budget, tau_start=100, tau_end=5000, verbose=False):
    """Gradient descent on smoothed objective with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * (progress ** 0.7)
        
        obj_s, grad = smoothed_gradient(a, tau)
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break
        
        # Step size: use a simple adaptive rule
        # We want to decrease the smoothed objective
        # Use: lr = (obj_s - target) / (g_norm^2) approximately
        # Or just: lr = min(1/g_norm, some_cap)
        lr = min(1.0 / (g_norm * g_norm + 1e-10), 5.0 / n)
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        
        if it - best_it > 5000:
            a = best_a.copy()
            if it > 20000:
                break
        
        if verbose and it % 2000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, g_norm={g_norm:.6f}")
    
    return best_a


def dual_guided_descent(a, time_budget, verbose=False):
    """Use LP dual to guide gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        _, _, dual_y = solve_lp_with_dual(a)
        if dual_y is None or len(dual_y) != 2 * n - 1:
            # Fallback to uniform weights
            dual_y = np.ones(2 * n - 1) / (2 * n - 1)
        
        # Gradient of sum_k y[k] * (a*a)[k]
        # = 2 * sum_k y[k] * a[k-j] for each j
        # = 2 * (y * a_rev)[n-1 + j] (same formula as before)
        a_rev = a[::-1]
        N2 = 1
        while N2 < 3 * n:
            N2 <<= 1
        Y_f = fft(dual_y, N2)
        AR_f = fft(a_rev, N2)
        conv_ya = ifft(Y_f * AR_f, N2).real
        grad = 2.0 * conv_ya[n-1:n-1+n]
        
        # Normalize gradient
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break
        
        # Adaptive step size with backtracking
        lr = 1.0 / (g_norm + 1e-10)
        for _ in range(10):
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                lr *= 0.5
                continue
            a_new /= s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                break
            lr *= 0.5
        
        a_new = np.maximum(a - lr * grad, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new /= s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                break
        else:
            break
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initial sequences."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a /= a.sum()
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
        a /= a.sum()
    elif mode == 3:
        x = np.linspace(0.001, 1, n)
        a = x * (1 - x)
        a = np.maximum(a, 1e-10)
        a /= a.sum()
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
        a /= a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = (1 + np.cos(2 * np.pi * x)) ** 2
        a = np.maximum(a, 1e-10)
        a /= a.sum()
    elif mode == 6:
        a = rng.normal(0, 1, n)
        a = np.abs(a)
        a /= a.sum()
    else:
        a = np.ones(n) / n
    return a
