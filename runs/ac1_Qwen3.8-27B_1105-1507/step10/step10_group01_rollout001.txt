import numpy as np
from scipy.fft import rfft, irfft, fft, ifft, next_fast_len

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = next_fast_len(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smoothed_gradient(a, tau):
    """Gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = next_fast_len(3 * n)
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]

def nesterov_gd(a, time_budget, tau_start, tau_end, verbose=False):
    """Nesterov accelerated gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 1.5

        y = np.maximum(y, 0)

            y = best_a.copy()
        else:
            y = y / s_y

        g_norm = float(np.linalg.norm(g))
        
        if g_norm > 1e-15:
            adapt = base_lr / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.1 * base_lr), 5.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adapt
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)

            v = np.zeros(n)

    # When we change a[i] by delta, conv[k] changes by:
    # For k in [i, i+n-1]: delta * a[k-i] (from j=i term) + if k-i != i: delta * a[k-i] (from k-j=i term)
    # Wait, when j=i: term is a[i]*a[k-i], changes by delta*a[k-i]
    # When k-j=i, i.e., j=k-i: term is a[k-i]*a[i], changes by a[k-i]*delta
    # So if j=i and j=k-i are the same (i.e., 2i=k), the total change is 2*delta*a[i]
    # Otherwise, the total change is delta*a[k-i] + delta*a[k-i] = 2*delta*a[k-i]
    # Wait no. The convolution sum is over j. For a given k, the terms involving a[i] are:
    # - When j=i: a[i]*a[k-i] (valid if 0<=k-i<n)
    # - When j=k-i: a[k-i]*a[i] (valid if 0<=j<n, i.e., 0<=k-i<n)
    # These are the same term when i=k-i, i.e., k=2i.
    # So for k != 2i: change in conv[k] = delta*a[k-i] + a[k-i]*delta = 2*delta*a[k-i]
    # For k = 2i: change in conv[k] = delta*a[i] + a[i]*delta = 2*delta*a[i]
    # (but note a[i] is being changed, so this is a bit different)
    
    # This is getting complicated. Let me just use a simpler approach:
    # Try small perturbations to each coordinate and keep the one that helps most.
    
    # Actually, for efficiency, let's use the gradient-based approach but project onto coordinates.
    # Or better yet, let's use a "block coordinate" approach with blocks of size 1.
    
    # For now, let's use a simpler iterative approach:
    # At each step, pick a random coordinate, try to optimize it.
    
    while time.monotonic() - start < time_budget:
        # Pick a random coordinate
        i = rng.integers(n)
        
        # Current contribution: a[i] appears in conv[k] for k in [i, i+n-1]
        # Try values: current, current*0.5, current*1.5, current+0.1/n, current-0.1/n
        current = a[i]
        candidates = [current, 
                     current * 0.5 if current > 0 else 0,
                     current * 1.5 + 0.001/n,
                     current + 0.05/n,
                     current - 0.05/n if current > 0.05/n else 0]
        candidates = [max(0, c) for c in candidates]
        
        best_c = current
        best_c_obj = best_obj
        
        for c in candidates:
            if abs(c - current) < 1e-15:
                continue
            a[i] = c
            s_tmp = float(np.sum(a))
            if s_tmp > 0.01:
                obj_tmp = compute_obj(a / s_tmp)  # Hmm, this isn't quite right for coordinate descent
            else:
                obj_tmp = np.inf
            if obj_tmp < best_c_obj:
                best_c_obj = obj_tmp
                best_c = c
        a[i] = best_c
        s = float(np.sum(a))
        if s > 0.01:
            a = a / s
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a

def quad_residue_init(n, p):
    """Initialize using quadratic residue pattern modulo p."""
    # QR = {x^2 mod p : x = 0, 1, ..., p-1}
    QR = set()
    for x in range(p):
        QR.add((x * x) % p)
    
    a = np.zeros(n)
    for i in range(n):
        if i % p in QR:
            a[i] = 1.0
        else:
            a[i] = 0.1  # Small background
    
    # Smooth it a bit
    kernel_size = 3
    if n > kernel_size:
        from scipy.ndimage import uniform_filter1d
        a_smooth = uniform_filter1d(a, size=kernel_size, mode='wrap')
        a = 0.7 * a + 0.3 * a_smooth
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    return a

def upsample(a, new_n):
    """Upsample sequence to new length."""
    n = len(a)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
