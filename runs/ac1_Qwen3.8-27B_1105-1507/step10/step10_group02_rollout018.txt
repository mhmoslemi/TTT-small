"""
Optimizes step function coefficients using a hybrid approach:
1. LP-based direction finding with optimal 1D line search (main workhorse)
2. Cutting-plane QP via active set of lags
3. Exact coordinate descent for polishing
4. Multi-n exploration for global diversification

Key innovations over previous approach:
- Optimal line search after LP (instead of fixed blending factors)
- Cutting-plane QP using the min-max structure
- Block coordinate descent for better local exploration
- Adaptive time allocation between phases
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time
from scipy.optimize import minimize as scipy_minimize


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective for normalized sequence (sum=1): 2n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g) element-wise."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, timeout=60.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    start = time.monotonic()
    for solver in solvers:
        if time.monotonic() - start > timeout:
            break
        try:
            prob.solve(solver=solver, timeout=timeout)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def optimal_line_search(caa, cag, cgg, t_lo=0.0, t_hi=1.0, tol=1e-10):
    """
    Find optimal t in [t_lo, t_hi] minimizing max_k [(1-t)^2*caa[k] + 2t(1-t)*cag[k] + t^2*cgg[k]].
    The objective is convex in t (max of quadratics), so golden section works.
    """
    def obj(t):
        val = (1.0 - t) ** 2 * caa + 2.0 * t * (1.0 - t) * cag + t * t * cgg
        return float(np.max(val))

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = t_hi - gr * (t_hi - t_lo)
    b_pt = t_lo + gr * (t_hi - t_lo)
    fa = obj(a_pt)
    fb = obj(b_pt)

    for _ in range(80):
        if t_hi - t_lo < tol:
            break
        if fa < fb:
            t_hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = t_hi - gr * (t_hi - t_lo)
            fa = obj(a_pt)
        else:
            t_lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = t_lo + gr * (t_hi - t_lo)
            fb = obj(b_pt)

    t_opt = (t_lo + t_hi) / 2.0
    return max(t_lo, min(t_hi, t_opt))


def lp_with_line_search(a, time_budget=300.0, verbose=True, seed=42):
    """
    LP direction finding + optimal 1D line search.
    Main workhorse for reducing the peak convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall_count = 0
    iteration = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        iteration += 1
        lp_timeout = min(remaining * 0.3, 45.0)

        g, M_lp = solve_lp_direction(a, timeout=lp_timeout)
        if g is None:
            # Try with perturbation
            pert = rng.normal(0, 0.02, n)
            a_pert = a * np.exp(pert)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            g, M_lp = solve_lp_direction(a_pert, timeout=lp_timeout)
            if g is None:
                break

        caa = compute_conv(a)
        # Need conv(a, g) - compute via building the matrix and multiplying
        A = build_conv_matrix(a)
        cag = A @ g
        cgg = compute_conv(g)

        # Optimal line search
        t_opt = optimal_line_search(caa, cag, cgg, 0.0, 1.0)

        # Also try t slightly beyond t_opt for numerical safety
        candidates = [t_opt, min(1.0, t_opt * 1.1), t_opt * 0.9, 0.5 * t_opt]
        candidates = [max(0.0, min(1.0, c)) for c in candidates]

        best_t = t_opt
        best_blended_obj = float('inf')
        for t in candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_blended_obj:
                best_blended_obj = obj_try
                best_t = t

        if best_blended_obj < best_obj - 1e-12:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_obj = best_blended_obj
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 5:
                # Perturb to escape
                pert = rng.normal(0, 0.05 * min(1.0, stall_count / 10.0), n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0
                if time.monotonic() - start > time_budget * 0.9:
                    break

        if verbose and iteration % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP+LS iter {iteration}: obj={best_obj:.8f}, t={best_t:.4f}, t={elapsed:.1f}s")

    return best_a


def solve_qp_active_set(a, active_lags, max_iter=200):
    """
    Solve min sum_{k in active_lags} w_k * conv(a,a)[k] s.t. a>=0, sum(a)=1.
    Uses projected gradient with adaptive step size.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Build Q matrix: Q[i,j] = sum_{k in active} w_k * delta(k == i+j)
    # Actually: conv(a,a)[k] = sum_i a[i]*a[k-i], so a^T Q a where Q[i,j] has
    # appropriate structure.
    # For efficiency, compute gradient directly.

    weights = np.ones(len(active_lags)) / len(active_lags)

    def obj_and_grad(a_cur):
        a_cur = np.maximum(a_cur, 0)
        s_cur = float(np.sum(a_cur))
        if s_cur < 1e-10:
            return 1e10, np.zeros(n)
        a_n = a_cur / s_cur
        conv = compute_conv(a_n)
        obj = float(np.sum(weights * conv[active_lags]))
        # Gradient: d/d a[i] conv(a,a)[k] = a[k-i] + a[i-k] (for valid indices)
        # For normalized: since we're working with a/sum(a), gradient is more complex.
        # Simplification: work with unnormalized, add constraint separately.
        grad = np.zeros(n)
        for idx, k in enumerate(active_lags):
            w = weights[idx]
            for i in range(n):
                # d/d a[i] of conv(a,a)[k] = a[k-i] + a[i-k] (if valid)
                term = 0.0
                if 0 <= k - i < n:
                    term += a_n[k - i]
                if 0 <= i - k < n:
                    term += a_n[i - k]
                grad[i] += 2.0 * w * term
        return obj, grad

    # Projected gradient with backtracking
    lr = 0.1
    for it in range(max_iter):
        obj_vals, grad = obj_and_grad(a)
        # Projected gradient step
        v = a - lr * grad
        v = np.maximum(v, 0)
        vs = float(np.sum(v))
        if vs < 1e-10:
            v = a
            vs = 1.0
        v = v / vs
        obj_v = obj_and_grad(v)[0]
        if obj_v < obj_vals:
            a = v
            lr = min(lr * 1.05, 2.0 / n)
        else:
            lr = lr * 0.5
            if lr < 1e-8:
                break

    return a


def cutting_plane_qp(a, time_budget=200.0, verbose=False):
    """
    Cutting-plane method for min_a max_k conv(a,a)[k] s.t. a>=0, sum(a)=1.
    Maintains active set of lags, solves weighted QP, updates active set.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    # Initialize active set
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    # Top lags where conv is close to max
    threshold = 0.8 * max_conv
    active_lags = list(np.where(conv >= threshold)[0])
    if not active_lags:
        active_lags = [int(np.argmax(conv))]

    for it in range(50):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # Solve QP with current active set
        a_new = solve_qp_active_set(a, active_lags, max_iter=100)

        # Evaluate
        conv_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(conv_new))
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

        # Update active set
        max_conv_new = float(np.max(conv_new))
        new_active = set(np.where(conv_new >= 0.9 * max_conv_new)[0].tolist())

        # Keep only lags that are still relevant
        if len(new_active) > 20:
            # Keep top 20 by conv value
            top_indices = np.argsort(conv_new)[-20:]
            new_active = set(top_indices.tolist())

        # Merge with old active set (keep some for stability)
        old_relevant = [k for k in active_lags if conv_new[k] >= 0.5 * max_conv_new]
        merged = list(new_active | set(old_relevant[:10]))
        if len(merged) > 30:
            merged = list(new_active)

        if merged == active_lags:
            # Converged
            break
        active_lags = merged

        if verbose and it % 5 == 0:
            print(f"    CP iter {it}: obj={best_obj:.8f}, |active|={len(active_lags)}")

    return best_a


def coordinate_descent(a, time_budget=200.0, verbose=False, seed=42):
    """
    Exact coordinate descent for minimizing 2n*max(conv(a,a))/sum(a)^2.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 3 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks]

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            def f_1d(x, v=v, S_other=S_other, n=n, c_k_lin=c_k_lin,
                     a_ki_lin=a_ki_lin, c_2i=c_2i):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.5, v, v * 1.5, v * 2.0, 1.0 / n]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(30):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def block_coordinate_descent(a, block_size=3, time_budget=100.0, seed=42):
    """
    Block coordinate descent: optimize groups of `block_size` coordinates simultaneously.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    n_blocks = max(1, (n + block_size - 1) // block_size)

    for sweep in range(200):
        if time.monotonic() - start >= time_budget:
            break

        if sweep % 5 == 0:
            conv = compute_conv(a)
            s_check = float(np.sum(a))
            if s_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (s_check * s_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        block_order = rng.permutation(n_blocks)
        S = float(np.sum(a))

        for bi in block_order:
            if time.monotonic() - start >= time_budget:
                break

            i0 = bi * block_size
            i1 = min(n, i0 + block_size)
            B = i1 - i0
            if B <= 0:
                continue

            x_block = a[i0:i1].copy()
            S_other = S - float(np.sum(x_block))

            if S_other < 1e-12:
                continue

            # Build the subproblem: for each k in the relevant range,
            # conv(a,a)[k] as a function of x_block
            # conv[k] = sum_i a[i]*a[k-i]
            # Terms involving x_block elements:
            # - x[j]*a[k-(i0+j)] for fixed a part
            # - a[i0+j]*x[m] for the other direction (same term if j!=m)
            # - x[j]*x[m] for j+m terms

            # Simplified: just do scipy minimize on the B variables
            def sub_obj(x):
                a_try = a.copy()
                a_try[i0:i1] = np.maximum(x, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    return 1e10
                conv_try = compute_conv(a_try)
                return 2.0 * n * float(np.max(conv_try)) / (s_try * s_try)

            # Bounds
            bounds = [(0, 2.0 / B)] * B
            res = scipy_minimize(sub_obj, x_block, method='L-BFGS-B', bounds=bounds,
                                 options={'maxiter': 50, 'ftol': 1e-10})

            if res.fun < best_obj - 1e-14:
                a[i0:i1] = np.maximum(res.x, 0)
                S = float(np.sum(a))

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        # Tapered (raised cosine)
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a + 0.1
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Two-peak with gap
        a = np.zeros(n)
        gap = n // 5
        a[:n//2 - gap] = 1.0
        a[n//2 + gap:] = 1.0
        a = a / a.sum()
    elif mode == 7:
        # Linear ramp
        a = np.linspace(1, 0, n) + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining LP+line search, cutting-plane QP, coordinate descent,
    block coordinate descent, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = 2.0 * n_default * float(np.max(compute_conv(best_seq)))
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 0: Quick diverse restart to check for better start ===
    print(f"\n=== Phase 0: Diverse Start Check ===")
    remaining = deadline - time.monotonic()
    quick_budget = min(remaining * 0.05, 30.0)
    if quick_budget > 10:
        for mode in range(8):
            a_try = diverse_init(n_current, rng, mode=mode)
            # Blend with parent
            blend = 0.5
            a_try = (1 - blend) * a_try + blend * best_seq
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            obj_try = 2.0 * n_current * float(np.max(compute_conv(a_try)))
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Mode {mode}: obj={best_score:.8f} **")
        print(f"  After diverse check: obj={best_score:.8f}")

    # === Phase 1: LP + Optimal Line Search (main workhorse) ===
    print(f"\n=== Phase 1: LP + Optimal Line Search ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.40, 550.0)
    if p1_budget > 60:
        a_lp = lp_with_line_search(best_seq, time_budget=p1_budget,
                                   verbose=True, seed=seed)
        obj_lp = 2.0 * a_lp.size * float(np.max(compute_conv(a_lp / a_lp.sum())))
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP+LS: obj={best_score:.8f} **")
        else:
            print(f"  LP+LS: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Coordinate Descent ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 350.0)
    if p2_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p2_budget,
                                  verbose=True, seed=seed + 1)
        obj_cd = 2.0 * a_cd.size * float(np.max(compute_conv(a_cd / a_cd.sum())))
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 3: Second LP pass (on CD-improved sequence) ===
    print(f"\n=== Phase 3: Second LP Pass ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 250.0)
    if p3_budget > 30:
        a_lp2 = lp_with_line_search(best_seq, time_budget=p3_budget,
                                    verbose=True, seed=seed + 100)
        obj_lp2 = 2.0 * a_lp2.size * float(np.max(compute_conv(a_lp2 / a_lp2.sum())))
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP2: obj={best_score:.8f} **")
        else:
            print(f"  LP2: obj={obj_lp2:.8f}")

        # Coordinate descent after LP
        remaining = deadline - time.monotonic()
        if remaining > 30:
            cd_budget = min(remaining * 0.3, 100.0)
            a_cd2 = coordinate_descent(best_seq, time_budget=cd_budget,
                                       verbose=False, seed=seed + 200)
            obj_cd2 = 2.0 * a_cd2.size * float(np.max(compute_conv(a_cd2 / a_cd2.sum())))
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  CD after LP2: obj={best_score:.8f} **")

    # === Phase 4: Cutting-Plane QP ===
    print(f"\n=== Phase 4: Cutting-Plane QP ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 150.0)
    if p4_budget > 20:
        a_cp = cutting_plane_qp(best_seq, time_budget=p4_budget, verbose=True)
        obj_cp = 2.0 * a_cp.size * float(np.max(compute_conv(a_cp / a_cp.sum())))
        if obj_cp < best_score:
            best_score = obj_cp
            best_seq = a_cp.copy()
            print(f"  CP: obj={best_score:.8f} **")
        else:
            print(f"  CP: obj={obj_cp:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 100), n_best + 100, n_best + 200,
        600, 700, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) > 50]
    n_options = n_options[:4]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick coordinate descent
        alloc = min(remaining * 0.25, 60.0)
        if alloc > 15:
            a_try = coordinate_descent(a_try, time_budget=alloc, verbose=False, seed=n_try)

        obj_try = 2.0 * a_try.size * float(np.max(compute_conv(a_try / a_try.sum())))
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final Polish (CD + LP) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Coordinate descent
        cd_alloc = min(remaining * 0.4, 100.0)
        if cd_alloc > 10:
            a_final = coordinate_descent(best_seq, time_budget=cd_alloc,
                                         verbose=False, seed=seed + 9999)
            obj_final = 2.0 * a_final.size * float(np.max(compute_conv(a_final / a_final.sum())))
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

        # Quick LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            lp_alloc = min(remaining * 0.5, 60.0)
            a_final2 = lp_with_line_search(best_seq, time_budget=lp_alloc,
                                           verbose=False, seed=seed + 7777)
            obj_final2 = 2.0 * a_final2.size * float(np.max(compute_conv(a_final2 / a_final2.sum())))
            if obj_final2 < best_score:
                best_score = obj_final2
                best_seq = a_final2.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # One more CD
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_final3 = coordinate_descent(best_seq, time_budget=min(remaining * 0.6, 40.0),
                                          verbose=False, seed=seed + 12345)
            obj_final3 = 2.0 * a_final3.size * float(np.max(compute_conv(a_final3 / a_final3.sum())))
            if obj_final3 < best_score:
                best_score = obj_final3
                best_seq = a_final3.copy()
                print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    # Ensure all values are valid
    best_seq = np.clip(best_seq, 0, 1000)
    result = [float(x) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
