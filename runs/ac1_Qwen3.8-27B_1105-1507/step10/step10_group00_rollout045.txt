    fa = rfft(a, size)

    n = len(a)
    s = np.sum(a)
    if s < 0.01:

    a_norm = a / s
    conv = compute_conv(a_norm)

        k_star = np.argmax(conv)

    a_new = project_simplex(a_new)
    return a_new, k_star

    n = len(v)

    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)

    if s < 0.01:

    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * np.max(conv)

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def mirror_descent_step(a, g, lr):
    """Mirror descent step with negative entropy mirror."""
    log_a = np.where(a > 1e-15, np.log(a), -30.0)
    new_log = log_a - lr * g
    new_a = np.exp(new_log - np.max(new_log))
    s = np.sum(new_a)
    if s < 1e-15:
        return a.copy()
    return new_a / s

    conv = compute_conv(a)

"""
Optimized subgradient/mirror descent for minimizing max autocorrelation of step functions.
Uses FFT-based convolution, multiplicative weights (mirror descent), multi-peak tracking,
adaptive learning rates, periodic perturbations, and length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_from_conv(conv, n):
    """Compute objective from precomputed convolution (assuming sum(a)=1)."""
    return 2.0 * n * np.max(conv)


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * np.max(conv)


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = np.sum(a)
    return a / s


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.01, 0.99, n)
    a = 1.0 / np.sqrt(np.maximum(1 - (2 * x - 1) ** 2, 1e-6))
    a = np.maximum(a, 0)
    s = np.sum(a)
    return a / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:

        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:

    elif mode == 5:

    elif mode == 6:

    elif mode == 7:
        # Double U
        a = 1.0 / (0.1 + (4 * x - 3) ** 4) + 1.0 / (0.1 + (4 * x - 1) ** 4)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:

    return a / s


def mirror_descent_main(a, time_budget=60.0, lr_init=0.05, verbose=False, rng=None):
    """
    Main mirror descent (multiplicative weights) optimization.
    Minimize max_k (a*a)[k] subject to a >= 0, sum(a) = 1.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = lr_init
    it = 0
    stall_count = 0
    best_it = 0
    peak_history = []
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = np.max(conv)
        obj = 2.0 * n * max_val
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1
        
        # Find peak
        k_star = int(np.argmax(conv))
        
        # Compute subgradient: g[j] = 2*a[k_star - j] for valid j
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo <= hi:
            g[lo:hi+1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]
        
        # Mirror descent step
        # a_new[j] = a[j] * exp(-lr * g[j]) / Z
        # Clip for numerical stability
        exponent = -lr * g
        exponent = np.clip(exponent, -50, 50)
        new_a = a * np.exp(exponent)
        s_new = np.sum(new_a)
        if s_new < 1e-15:
            new_a = a.copy()
        else:
            new_a = new_a / s_new
        
        a = new_a
        it += 1
        
        # Adaptive learning rate

            lr *= 0.9
            stall_count = 0

            a = best_a * np.exp(pert)
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            else:
                a = best_a.copy()
            lr = lr_init * 0.5
            stall_count = 0
            best_it = it
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it{it}: obj={best_obj:.8f}, lr={lr:.6f}, "
                  f"stall={stall_count}, elapsed={elapsed:.1f}s")
    
    return best_a


def subgradient_descent(a, time_budget=30.0, lr_init=0.02, verbose=False, rng=None):
    """Subgradient descent with momentum on the simplex."""
    if rng is None:
        rng = np.random.default_rng(123)
    
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    it = 0
    momentum = np.zeros(n)
    beta = 0.7
    stall_count = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo <= hi:
            g[lo:hi+1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]
        
        # Nesterov momentum update
        momentum = beta * momentum - lr * g
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
