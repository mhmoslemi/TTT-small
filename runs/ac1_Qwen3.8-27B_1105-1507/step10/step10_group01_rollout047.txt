    s.t. (A g)[k] <= M for all k
         sum(g) = 1
         g >= 0
    
    where A is the convolution matrix for a_norm.

    a_norm = a / np.sum(a)
    A = build_conv_matrix_fast(a_norm)  # shape (2n-1, n)

        s = np.sum(g_val)
        if s > 1e-10:

    return None

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: QP-based direction finding (min L2 cross-conv + peak),
spectral diversification, adaptive phased optimization (LP→QP→GD),
improved line search with extrapolation, and multi-length exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_fast(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_fast(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_qp_direction(a, lambda_peak=1.0):
    """
    Solve QP: min ||A_norm @ g||^2 + lambda_peak * M
    s.t. (A_norm @ g)[k] <= M for all k, sum(g)=1, g>=0.
    This balances reducing the peak with spreading the energy.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_fast(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    conv_val = A @ g
    obj = cp.sum_squares(conv_val) + lambda_peak * M
    constraints = [conv_val <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def line_search_enhanced(a, g, n_evals=60):
    """
    Enhanced line search: coarse grid + fine grid + parabolic refinement.
    Also tries extrapolation beyond t=1.
    """
    n = len(a)
    # Extended range: allow slight extrapolation
    ts = np.concatenate([
        np.linspace(0.002, 0.98, n_evals),
        np.array([1.02, 1.05, 1.10])
    ])
    objs = np.full(len(ts), np.inf)
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try)

    best_idx = int(np.argmin(objs))
    best_t = float(ts[best_idx])
    best_obj = objs[best_idx]

    # Parabolic refinement around best point
    if 0 < best_idx < len(ts) - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.002, 1.1))
            a_par = (1 - t_par) * a + t_par * g
            a_par = np.maximum(a_par, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                f_par = compute_obj(a_par)
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par

    # Fine grid around best
    if best_t > 0.01 and best_t < 1.05:
        fine_ts = np.linspace(max(0.002, best_t - 0.05), min(1.1, best_t + 0.05), 20)
        for t_f in fine_ts:
            a_f = (1 - t_f) * a + t_f * g
            a_f = np.maximum(a_f, 0)
            s_f = float(np.sum(a_f))
            if s_f > 1e-10:
                obj_f = compute_obj(a_f)
                if obj_f < best_obj:
                    best_obj = obj_f
                    best_t = float(t_f)

    return best_t, best_obj


def spectral_diversify(a, rng, strength=0.05):
    """
    Diversify using spectral phase manipulation.
    Flattens the power spectrum to reduce autocorrelation peaks.
    """
    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1

    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)

    # Soften the magnitude towards flat (log-domain interpolation)
    log_mag = np.log(mag + 1e-15)
    # Keep first n/4 and last n/4 bins (low freq) mostly intact
    # Soften the rest
    target_log_mag = log_mag.copy()
    for i in range(1, N // 2):
        target_log_mag[i] = 0.7 * log_mag[i] + 0.3 * log_mag[0]
        if i + 1 < N:
            pass  # will handle symmetry
    target_log_mag[N - i] = target_log_mag[i] if i > 0 else target_log_mag[0]

    # Blend with original
    blend = 0.5
    new_mag = np.exp((1 - blend) * log_mag + blend * target_log_mag)

    # Add random phase perturbation
    phase_pert = strength * rng.standard_normal(N)
    new_phase = phase + phase_pert

    a_new = ifft(new_mag * np.exp(1j * new_phase), N).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        return a_new / s
    return a.copy()


def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 6):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    Af = fft(a, N)
    g_full = ifft(P * np.conj(Af)).real
    return 2.0 * g_full[:n]


def subgradient_direction(a):
    """Compute subgradient of max autocorrelation."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    # Find all positions achieving max
    max_positions = np.where(conv >= max_conv - 1e-12)[0]
    p = np.zeros(len(conv))
    for pos in max_positions:
        p[pos] = 1.0 / len(max_positions)

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    Af = fft(a, N)
    g_full = ifft(P * np.conj(Af)).real
    return 2.0 * g_full[:n]


def main_optimize_lp(a, time_budget=300.0, verbose=True):
    """LP-driven optimization with adaptive strategy."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall_count = 0
    iter_count = 0
    consecutive_no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        # Choose solver based on phase
        if consecutive_no_improve < 3:
            g, M = solve_lp_minpeak(a)
        else:
            g = solve_qp_direction(a, lambda_peak=0.5)
            if g is None:
                g, M = solve_lp_minpeak(a)
            M = -1

        if g is None:
            break

        conv_curr = compute_conv(a)
        max_curr = float(np.max(conv_curr))

        best_t, obj_at_t = line_search_enhanced(a, g, n_evals=40)
        iter_count += 1

        if obj_at_t < best_obj - 1e-13:
            a_try = (1 - best_t) * a + best_t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                best_obj = obj_at_t
                best_a = a_try.copy()
                a = a_try
                stall_count = 0
                consecutive_no_improve = 0
                if verbose and (iter_count % 5 == 0 or iter_count == 1):
                    elapsed = time.monotonic() - start
                    print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, "
                          f"max_conv={max_curr:.6f}, time={elapsed:.0f}s")
        else:
            consecutive_no_improve += 1
            stall_count += 1

            if verbose and iter_count % 15 == 0:
                print(f"    LP it{iter_count}: stalled obj={best_obj:.8f}")

            # Try smaller steps
            if consecutive_no_improve <= 3:
                for t_small in [0.05, 0.02, 0.01]:
                    a_small = (1 - t_small) * a + t_small * g
                    a_small = np.maximum(a_small, 0)
                    s_small = float(np.sum(a_small))
                    if s_small > 1e-10:
                        a_small = a_small / s_small
                        obj_small = compute_obj(a_small)
                        if obj_small < best_obj - 1e-13:
                            best_obj = obj_small
                            best_a = a_small.copy()
                            a = a_small
                            stall_count = 0
                            consecutive_no_improve = 0
                            break

            if stall_count >= 4:
                # Diversification: spectral + subgradient
                conv_p = compute_conv(a)
                max_p = float(np.max(conv_p))
                exp_p = np.exp(300.0 * (conv_p - max_p))
                p = exp_p / exp_p.sum()
                N_fft = 1
                while N_fft < 3 * n:
                    N_fft <<= 1
                P = fft(p, N_fft)
                Af = fft(a, N_fft)
                grad = 2.0 * ifft(P * np.conj(Af)).real[:n]

                lr_pert = 0.03 / (float(np.linalg.norm(grad)) + 1e-10)
                a = a - lr_pert * grad
                a = np.maximum(a, 0)
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                else:
                    a = best_a.copy()

                # Spectral diversification
                a = spectral_diversify(a, rng, strength=0.03)
                a = a + 0.003 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()

                stall_count = 0
                consecutive_no_improve = 0

                if iter_count > 120:
                    if verbose:
                        print(f"    LP it{iter_count}: giving up")
                    break

    return best_a


def gd_polish(a, time_budget=100.0, tau_init=100.0, tau_end=1500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.5 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    stagnant = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.2

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.2 * base_lr), 3.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adapt

        if prev_g is not None and it % 5 == 0 and it > 10:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 4.0 / np.sqrt(n))
                lr = 0.6 * lr + 0.4 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stagnant = 0
        else:
            stagnant += 1

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9997)

        if stagnant > 5000:
            rng_local = np.random.default_rng(it)
            a = best_a + 0.01 * rng_local.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = base_lr * 0.5
            stagnant = 0
            best_it = it

    if verbose:
        print(f"    GD: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def two_level_init(n, rng):
    """Create a two-level initialization biased towards good autocorrelation."""
    # Use a chirp-like pattern
    t = np.arange(n) / n
    a = 0.5 + 0.4 * np.cos(2 * np.pi * (t**2 * 2.0 - 0.5))
    a = np.maximum(a, 0.01)
    a = a + 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    return a / a.sum()

            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.6, 80.0)
        a_final2 = gd_polish(best_seq, time_budget=alloc, tau_init=300.0, tau_end=3000.0, verbose=True)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
