    n = len(a)

    n = len(a)

    while size < 2 * n - 1:

    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

    s = float(np.sum(a))

def build_conv_matrix(a):
    """Build sparse-friendly matrix A (2n-1 x n) such that A @ g = conv(a, g).
    A[k, j] = a[k-j] if 0 <= k-j < n, else 0."""
    n = len(a)
    # Use the structure: A is a Hankel-like matrix
    # A[k, j] = a[k-j] for max(0, k-n+1) <= j <= min(k, n-1)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    g_val = g_val / s_val
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:
            continue

    return None, None

    n = len(a)

    s = float(np.sum(a))

        return a.copy()
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.3

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:

        if g_lp is None:

def high_tau_gd(a, time_budget, tau_start=100, tau_end=10000, verbose=True):
    """High-precision smoothed gradient descent with Nesterov acceleration."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a.copy()
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.7

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv_y = compute_conv(y)
        max_conv = float(np.max(conv_y))
        exp_vals = np.exp(tau * (conv_y - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv_y))
            p[np.argmax(conv_y)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient via FFT cross-correlation
        N_fft = 1
        while N_fft < 3 * n:
            N_fft <<= 1
        from scipy.fft import fft as ffft, ifft as fifft
        fa_f = ffft(y, N_fft)
        fp_f = ffft(p, N_fft)
        cross = fifft(np.conj(fa_f) * fp_f)
        grad = 2.0 * cross[:n].real

        # Project gradient to maintain sum=1 constraint
        grad -= np.sum(grad) / n  # Remove mean to stay on the simplex tangent space
        # Actually, for the constraint sum(a)=1, we should project:
        # The feasible directions are those with sum(d) = 0
        # So we subtract the mean of the gradient
        grad_mean = float(np.mean(grad))
        grad -= grad_mean

        # Adaptive step size
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break
        lr = min(1.0 / (g_norm + 1e-10), 1.0 / np.sqrt(n))
        lr = max(lr, 1e-10)

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        beta = min(0.95, beta + 0.00005)

        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it
            if it > 20000:
                break

        if verbose and it % 2000 == 0:
            print(f"  HTGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.0f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.5
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
    elif mode == 5:
        a = np.exp(-np.abs(np.arange(n) - n/2) / (n * 0.2))
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x ** 3 * (1 - x)
    elif mode == 7:
        # Two bumps
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.25) / 0.1) ** 2) + np.exp(-((x - 0.75) / 0.1) ** 2)
    elif mode == 8:
        # Three bumps
        x = np.linspace(0, 1, n)
        a = (np.exp(-((x - 0.2) / 0.08) ** 2) + 
             np.exp(-((x - 0.5) / 0.08) ** 2) + 
             np.exp(-((x - 0.8) / 0.08) ** 2))
    else:
        a = rng.uniform(0.1, 1, n)
    return a / a.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def support_shrink(a, time_budget, verbose=True):
    """Try zeroing out small entries and re-optimizing."""
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a.copy()
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    sorted_vals = np.sort(a)[::-1]
    for frac in [0.3, 0.4, 0.5, 0.6, 0.7, 0.8]:
        if time.monotonic() - start > time_budget:
            break
        k = max(2, int(len(a) * frac))
        threshold = sorted_vals[k - 1]
        support = np.where(a >= threshold)[0]
        if len(support) < 2:
            continue
        a_shrunk = np.zeros(len(a))
        a_shrunk[support] = a[support]
        s_shrunk = float(np.sum(a_shrunk))
        if s_shrunk < 1e-10:
            continue
        a_shrunk = a_shrunk / s_shrunk

        # Quick GD polish
        alloc = min(20, time_budget - 5 - (time.monotonic() - start))
        if alloc > 5:
            a_shrunk = high_tau_gd(a_shrunk, time_budget=alloc,
                                    tau_start=100, tau_end=2000, verbose=False)
        obj_shrunk = compute_obj(a_shrunk)
        if obj_shrunk < best_obj - 1e-12:
            best_obj = obj_shrunk
            best_a = a_shrunk.copy()
            if verbose:
                print(f"  Shrink frac={frac}: obj={best_obj:.8f}, support={len(support)}")

    return best_a
