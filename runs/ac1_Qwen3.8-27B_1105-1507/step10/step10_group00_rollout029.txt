Hybrid LP + smooth optimization for step function peak minimization.
Uses fast LP polishing with precomputed convolution matrices,
convex line search exploiting the quadratic structure of the blend,
log-sum-exp smooth refinement, and adaptive n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

# Precompute helper
def next_pow2(m):
    s = 1
    while s < m:
        s <<= 1
    return s

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = next_pow2(2*n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_conv_cross(a, b):
    """Compute conv(a, b) via FFT."""
    n = len(a)
    m = len(b)
    size = next_pow2(n + m - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n+m-1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Matrix A such that A @ g = conv(a, g) (full, length 2n-1)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except:
            continue
    return None

def convex_line_search(a, g, c_a, c_ag, c_gg):
    """
    Find t* in [0, 1.2] minimizing max_k q_k(t) where
    q_k(t) = (1-t)^2 * c_a[k] + 2t(1-t) * c_ag[k] + t^2 * c_gg[k]
    This is convex in t, so use ternary search.
    """
    def f(t):
        val = (1-t)**2 * c_a + 2*t*(1-t) * c_ag + t**2 * c_gg
        return float(np.max(val))
    
    lo, hi = 0.0, 1.2
    for _ in range(30):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = f(m1)
        f2 = f(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2

def lp_polish(a, time_budget=100.0, max_iters=60, verbose=False):
    """LP polishing with convex line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64) / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        g = solve_lp(a)
        if g is None:
            break
        
        # Precompute convolutions for line search
        c_a = compute_conv(a)
        c_ag = compute_conv_cross(a, g)
        c_gg = compute_conv(g)
        
        t_opt = convex_line_search(a, g, c_a, c_ag, c_gg)
        
        a_new = (1 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s > 0.01:
            a_new = a_new / s
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, elapsed={time.monotonic()-start:.1f}s")
        
        if stall >= 5:
            # Perturb and restart
            rng = np.random.default_rng(it * 7919)
            noise = rng.normal(0, 0.01 * stall, n)
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0
    
    return best_a

"""
Semidefinite-relaxed LP polishing for step function peak autocorrelation minimization.
Key innovations over prior approach:
1. SDP-based LP direction using Burer-Monteiro low-rank factorization
2. Convex line search exploiting quadratic structure of the blend
3. Log-sum-exp smooth refinement phase with adaptive temperature
4. Adaptive length (n) exploration with interpolation + re-optimization
5. Multi-start with diverse structured initializations
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def _next_pow2(m):
    s = 1
    while s < m:
        s <<= 1
    return s


def _compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = _next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def _compute_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = _next_pow2(n + m - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def _compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = _compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def _build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def _solve_lp_direction(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a_norm = a / s
    A = _build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv
        except Exception:
            continue
    return None


def _convex_line_search(c_a, c_ag, c_gg):
    """Find t in [0,1.5] minimizing max_k [(1-t)^2 c_a[k] + 2t(1-t) c_ag[k] + t^2 c_gg[k]]."""
    def f(t):
        val = (1.0 - t) * (1.0 - t) * c_a + 2.0 * t * (1.0 - t) * c_ag + t * t * c_gg
        return float(np.max(val))

    lo, hi = 0.0, 1.5
    for _ in range(35):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = f(m1)
        f2 = f(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


def _lp_polish(a, time_budget, max_iters=80, verbose=False):
    """LP polishing with convex line search and adaptive restarts."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = _compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(42 + n)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        g = _solve_lp_direction(a)
        if g is None:
            break

        c_a = _compute_conv(a)
        c_ag = _compute_conv_cross(a, g)
        c_gg = _compute_conv(g)
        t_opt = _convex_line_search(c_a, c_ag, c_gg)

        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 0.01:
            a_new = a_new / s_new
            obj_new = _compute_obj(a_new)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 10 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, elapsed={time.monotonic()-start:.1f}s")

        if stall >= 4:
            amp = 0.005 * min(stall, 10)
            noise = rng.normal(0, amp, n)
            a = best_a + noise
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 0.01:
                a = a / s
            else:
                a = best_a.copy()
            stall = 0

    return best_a


def _lse_optimize(a, time_budget, verbose=False):
    """Log-sum-exp smooth optimization with adaptive temperature."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = _compute_obj(a)
    start = time.monotonic()
    it = 0
    lr = 0.03
    beta = 20.0

    while time.monotonic() - start < time_budget:
        conv = _compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / np.sum(exp_vals)

        # Gradient computation: grad[i] = beta * sum_k w[k] * (a[k-i] + a[i]) * 1[0<=k-i<n]
        # Vectorized via Toeplitz matrix multiplication
        w_ext = np.zeros(2 * n - 1)
        w_ext[:len(w)] = w
        # term1[i] = sum_j w_ext[i+j] * a[j], j=0..n-1
        # This is a cross-correlation
        col_idx = np.arange(2 * n - 1 - 1, -1, -1)  # reversed indices
        rows = np.tile(col_idx, (n, 1))
        cols = np.tile(np.arange(n), (2 * n - 1, 1))
        cols = cols.T
        # For row i, col j: index into w_ext is i + j
        indices = np.add.outer(np.arange(n), np.arange(n))  # n x n, indices[i,j] = i+j
        valid = indices < (2 * n - 1)
        W_mat = np.where(valid, w_ext[indices], 0.0)  # n x n
        term1 = W_mat @ a

        # term2[i] = sum of w_ext[k] for k in [i, i+n-1]
        cumsum = np.concatenate([[0.0], np.cumsum(w_ext)])
        lo_idx = np.arange(n)
        hi_idx = np.minimum(lo_idx + n, 2 * n - 1)
        term2 = cumsum[hi_idx] - cumsum[lo_idx]

        grad = beta * (term1 + a * term2)
        grad = grad / n

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 0.01:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        obj = _compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            beta = min(beta * 1.02, 300.0)
        else:
            beta = max(beta * 0.98, 10.0)
            lr = max(lr * 0.995, 0.001)

        a = a_new
        it += 1

        if verbose and it % 200 == 0:
            print(f"    LSE it{it}: obj={best_obj:.8f}, beta={beta:.1f}, elapsed={time.monotonic()-start:.1f}s")

    return best_a


def _u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.08 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _arcsine_init(n):
    x = np.linspace(0.02, 0.98, n)
    a = 1.0 / np.sqrt(np.maximum(1e-8, 1 - (2 * x - 1) ** 2))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _diversified_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.1
    elif mode == 1:
        a = np.exp(-((x - 0.25) ** 2) / 0.01) + np.exp(-((x - 0.75) ** 2) / 0.01) + 0.05
    elif mode == 2:
        a = 4.0 * np.abs(x - 0.5) + 0.1
    elif mode == 3:
        a = rng.uniform(0.2, 1.0, n)
        a = np.convolve(a, np.ones(7) / 7, mode='same')
    elif mode == 4:
        a = np.exp(5.0 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(3.0 * (2 * x - 1))
    elif mode == 7:
        a = 1.0 / (0.05 + (2 * x - 1) ** 6)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _subgradient_descent(a, time_budget=30.0):
    """Fast subgradient descent warm-up."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = _compute_obj(a)
    start = time.monotonic()
    lr = 0.015
    momentum = np.zeros(n)
    beta_m = 0.3
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        conv = _compute_conv(a)
        k_max = int(np.argmax(conv))

        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]

        momentum = beta_m * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = _compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        a = a_new
        lr *= 0.9998
        it += 1

        if it - best_it > 3000:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = 0.01
            best_it = it

    return best_a


def _interpolate_to_length(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 0.01:
        return np.ones(n_new) / n_new
    return a_new / s
