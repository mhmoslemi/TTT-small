"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses intensive LP polishing with warm-starting, asymmetric power-law initializations,
adaptive step sizing, multi-n exploration, and spectral gradient refinement.
Key improvements: asymmetric init (exploits conv asymmetry), more LP iterations,
wider n sweep, warm-started LP solves, and dual-gap-based convergence.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_g=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None). Uses warm start if provided."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    objective = cp.Minimize(M)
    prob = cp.Problem(objective, constraints)

    # Try solvers in order of preference
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            kwargs = {}
            if solver == cp.MOSEK:
                kwargs = {'mosek_options': {'MSK_DPAR_INTPNT_CO点心_TOL': 1e-9}}
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:
            continue

    return None, None


def lp_polish(a, time_budget, verbose=True, warm_g=None):
    """LP polishing with adaptive line search and warm starting."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5
    prev_g = warm_g

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = solve_lp(a, warm_g=prev_g)
        if g_lp is None:
            break

        prev_g = g_lp

        # Adaptive line search: try a range of t values
        t_candidates = np.concatenate([
            np.linspace(0.05, 1.0, 15),
            [max(0.02, last_t * 0.7), last_t, min(1.0, last_t * 1.3)]
        ])
        t_candidates = np.unique(np.clip(t_candidates, 0.01, 1.0))

        best_t_obj = best_obj + 1e-10
        best_t = last_t
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.3f}, M_lp={M_lp:.6f}, t={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 5:
                if verbose:
                    print(f"    LP: converged, obj={best_obj:.8f}")
                break

    return best_a


def high_tau_gd(a, time_budget, tau_start=100, tau_end=8000, verbose=False):
    """High-precision smoothed gradient descent with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.4

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(y, N)
        g_full = ifft(P * np.conj(A_f)).real
        g = 2.0 * g_full[:n]

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        lr = min(1.0 / (g_norm + 1e-10), 2.0 / np.sqrt(n))
        lr = max(lr, 1e-8)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        beta = min(0.96, beta + 0.0002)

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it
            if it > 60000:
                break

        if verbose and it % 5000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation with zero-padding."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def init_asymmetric(n, alpha=1.0, mode=0):
    """Generate asymmetric (monotonically decreasing) initializations."""
    x = np.arange(n, dtype=np.float64)
    if mode == 0:
        # Power law decay
        a = (n - x) ** alpha
    elif mode == 1:
        # Exponential decay
        a = np.exp(-alpha * x / n)
    elif mode == 2:
        # Smooth power law
        a = (1.0 + (n - 1 - x) / (n - 1)) ** (alpha + 1)
    elif mode == 3:
        # Two-scale: fast initial decay then slow
        a = np.exp(-alpha * x / (n * 0.3)) + 0.3 * np.exp(-alpha * x / (n * 0.8))
    elif mode == 4:
        # Inverse linear
        a = 1.0 / (1.0 + alpha * x / n)
    elif mode == 5:
        # Cosine power
        a = np.cos(np.pi * x / (2.0 * n)) ** (alpha + 1)
    else:
        a = (n - x) ** alpha
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Intensive LP polishing with asymmetric initialization,
    multi-n exploration, and gradient refinement.

    best_seq = None
    best_score = np.inf

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)

            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1000
        best_seq = init_asymmetric(n_init, alpha=1.5, mode=0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}, power-law): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, obj={best_score:.8f}")

    # === Phase 1: Intensive LP Polishing on current n (350s) ===
    print(f"\n=== Phase 1: LP Polishing (n={n_best}) ===")
    p1_budget = min(350.0, (deadline - start_time) * 0.3)
    if p1_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:

            print(f"  Phase2: obj={obj_gd:.8f}")

    # === Phase 3: Asymmetric Re-initialization + LP (200s) ===
    print(f"\n=== Phase 3: Asymmetric Re-init ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(200.0, remaining * 0.18)
    if p3_budget > 30:
        # Try several asymmetric inits and pick the best after quick LP
        n_cur = len(best_seq)
        best_p3 = best_seq.copy()
        best_p3_score = best_score
        inits_to_try = [
            (init_asymmetric(n_cur, alpha=0.5, mode=0), "pow0.5"),
            (init_asymmetric(n_cur, alpha=1.0, mode=0), "pow1.0"),
            (init_asymmetric(n_cur, alpha=2.0, mode=0), "pow2.0"),
            (init_asymmetric(n_cur, alpha=1.5, mode=1), "exp1.5"),
            (init_asymmetric(n_cur, alpha=2.0, mode=2), "smooth2.0"),

        # Blend with current best to preserve learned structure

            if time.monotonic() > deadline - 30:

            a_blend = 0.4 * a_init + 0.6 * best_seq
            a_blend = np.maximum(a_blend, 0)
