def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A

def solve_lp(a, solver_list=None):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)

    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:

def resample(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-driven iteration with golden-section line search as the main workhorse,
combined with Nesterov GD polishing, multi-n exploration, coordinate descent,
and targeted peak-shaving. Key improvements: tighter LP tolerances, more
golden-section iterations, wider n sweep, Fourier-based perturbation, and
a final coordinate-descent polish phase.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp(a, eps=1e-8):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS]
    for solver in solvers:
        try:
            kwargs = {}
            if solver == cp.MOSEK:
                kwargs = {'MSDK_OPTIMIZATION_TOL': eps}
            elif solver == cp.CLARABEL:
                kwargs = {'tol_gap_abs': eps, 'tol_feas': eps}
            elif solver == cp.SCS:
                kwargs = {'eps': eps}
            prob.solve(solver=solver, **kwargs)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def golden_section_line_search(a, g, tol=1e-7, max_iter=40):
    """Golden section search for optimal t in [0,1] to minimize obj((1-t)*a + t*g)."""
    phi = (np.sqrt(5) - 1) / 2
    lo, hi = 0.001, 0.999
    c = lo + (1 - phi) * (hi - lo)
    d = lo + phi * (hi - lo)
    f_c = compute_obj(np.maximum((1 - c) * a + c * g, 0))
    f_d = compute_obj(np.maximum((1 - d) * a + d * g, 0))

    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if f_c < f_d:
            hi = d
            d = c
            f_d = f_c
            c = lo + (1 - phi) * (hi - lo)
            f_c = compute_obj(np.maximum((1 - c) * a + c * g, 0))
        else:
            lo = c
            c = d
            f_c = f_d
            d = lo + phi * (hi - lo)
            f_d = compute_obj(np.maximum((1 - d) * a + d * g, 0))

    t_best = (lo + hi) / 2.0
    f_best = compute_obj(np.maximum((1 - t_best) * a + t_best * g, 0))
    return t_best, f_best

def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]

    best_it = 0

    while time.monotonic() - start < time_budget:

        tau = tau_init + (tau_end - tau_init) * progress ** 1.5

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g_norm = float(np.linalg.norm(g))

            adapt = min(max(adapt, 0.1 * base_lr), 3.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adapt

        a_new = y + v
        a_new = np.maximum(a_new, 0)

            a_new = best_a.copy()

def coordinate_descent(a, time_budget, verbose=False):
    """
    Coordinate descent: for each coordinate, optimize its value
    given all others fixed. Uses the observation that changing a[i]
    only affects conv[k] for k in [i, i+n-1] (roughly).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    # Precompute: for each i, which conv indices are affected
    # conv[k] = sum_j a[j]*a[k-j]
    # Changing a[i] affects conv[k] for all k where a[i] appears: k = i+j for all j, i.e., k in [0, 2n-2]
    # Actually all conv[k] are affected since a[i]*a[k-i] appears in conv[k] for all valid k

    # Instead, use a smarter approach: try small multiplicative changes
    order = list(range(n))

    sweep = 0
    while time.monotonic() - start < time_budget:
        if time_budget - (time.monotonic() - start) < 1.0:
            break
        # Randomize order each sweep
        rng.shuffle(order)
        sweep += 1
        improved_this_sweep = False

        for i in order:
            if time.monotonic() - start > time_budget - 1.0:
                break
            a_old = a[i]
            if a_old < 1e-15:
                continue

            # Try multiplicative factors
            best_val = a_old
            best_obj_i = best_obj
            for factor in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                a_try = a.copy()
                a_try[i] = a_old * factor
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj_i - 1e-12:
                    best_obj_i = obj_try
                    best_val = a_old * factor

            if best_val != a_old:
                a[i] = best_val
                a = a / a.sum()
                best_obj = best_obj_i
                best_a = a.copy()
                improved_this_sweep = True

        if not improved_this_sweep and sweep > 2:
            # Try random coordinate with larger perturbation
            for _ in range(10):
                i = rng.integers(0, n)
                a[i] *= rng.uniform(0.5, 1.5)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                obj = compute_obj(a)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
            if time.monotonic() - start < time_budget:
                a = best_a.copy()

    if verbose:
        print(f"    CD: obj={best_obj:.8f}, sweeps={sweep}")
    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    a_new = a.copy()
    for k in range(1, 6):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        a_new = a_new * np.exp(amp * np.cos(freq * np.arange(n) + phase))
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def peak_shave(a, time_budget, verbose=False):
    """
    Targeted peak shaving: identify top convolution peaks and try to
    reduce them by adjusting coordinates that contribute to those peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    conv = compute_conv(a)
    max_val = float(np.max(conv))
    # Find indices where conv is close to max
    threshold = max_val * 0.95
    peak_indices = np.where(conv >= threshold)[0]

    if len(peak_indices) == 0:
        return a

    # For each peak index k, the contributing pairs are (i, k-i)
    # Try reducing a[i] or a[k-i] for peaks
    for k in peak_indices[:10]:  # Focus on top peaks
        if time.monotonic() - start > time_budget - 1.0:
            break
        # Find contributing indices
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            if j < 0 or j >= n:
                continue
            # Try reducing a[i]
            for factor in [0.7, 0.8, 0.9]:
                a_try = a.copy()
                a_try[i] *= factor
                # Compensate by increasing a[j] to maintain sum
                diff = a[i] * (1 - factor) / s
                a_try[j] += diff * s
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    conv = compute_conv(a)
                    max_val = float(np.max(conv))
                    threshold = max_val * 0.95
                    peak_indices = np.where(conv >= threshold)[0]
                    break

    return best_a

    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
            else:
                raise Exception("Sum too small")
        else:
            raise Exception("Invalid parent")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Aggressive LP Optimization ===
    print(f"\n=== Phase 1: LP Optimization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.3, 400.0)
    if p1_budget > 30:
        a_lp = lp_optimize(best_seq, p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: GD Polish ===
    print(f"\n=== Phase 2: GD Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.08, 80.0)
    if p2_budget > 10:
        a_gd = gd_optimize(best_seq, p2_budget, tau_init=100.0, tau_end=1500.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_gd:.8f} (no improvement)")

    # === Phase 3: Second LP Round ===
    print(f"\n=== Phase 3: Second LP ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 250.0)
    if p3_budget > 20:
        a_lp2 = lp_optimize(best_seq, p3_budget, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj_lp2:.8f} (no improvement)")

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.1, 100.0)
    if p4_budget > 15:
        a_cd = coordinate_descent(best_seq, p4_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase 4: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4: obj={obj_cd:.8f} (no improvement)")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    n_options = sorted(set([
        n_cur - 100, n_cur - 50, n_cur + 50, n_cur + 100, n_cur + 200, n_cur + 400,
        500, 600, 700, 800, 900, 1000, 1100, 1200, 1300, 1400, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1600 and abs(n - n_cur) > 30]
    n_options = n_options[:6]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # LP optimization
        alloc_lp = min(remaining * 0.2, 60.0)
        if alloc_lp > 15:
            a_try = lp_optimize(a_try, alloc_lp, verbose=False)

        # GD polish
        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.06, 25.0)
        if alloc_gd > 5:
            a_try = gd_optimize(a_try, alloc_gd, tau_init=200.0, tau_end=1000.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_c = len(best_seq)

    # Fourier perturbation restart
    if remaining > 50:
        a_r = fourier_perturb(best_seq, rng, strength=0.04)
        alloc = min(remaining * 0.2, 60.0)
        if alloc > 10:
            a_r = lp_optimize(a_r, alloc, verbose=False)
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.05, 20.0)
        if alloc2 > 5:
            a_r = gd_optimize(a_r, alloc2, tau_init=200.0, tau_end=1000.0)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Fourier restart: obj={best_score:.8f} **")
        else:
            print(f"  Fourier restart: obj={obj_r:.8f}")

    # Smooth blend restart
    remaining = deadline - time.monotonic()
    if remaining > 40:
        ks = max(7, n_c // 10)
        if ks % 2 == 0:
            ks += 1
        a_smooth = uniform_filter1d(best_seq, size=ks)
        a_r = 0.7 * best_seq + 0.3 * a_smooth
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        alloc = min(remaining * 0.2, 50.0)
        if alloc > 10:
            a_r = lp_optimize(a_r, alloc, verbose=False)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Smooth restart: obj={best_score:.8f} **")
        else:
            print(f"  Smooth restart: obj={obj_r:.8f}")

    # Random blend restart
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_r = 0.8 * best_seq + 0.2 * (rng.uniform(0.5, 1.5, n_c) / n_c)
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        alloc = min(remaining * 0.2, 50.0)
        if alloc > 10:
            a_r = lp_optimize(a_r, alloc, verbose=False)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Random restart: obj={best_score:.8f} **")
        else:
            print(f"  Random restart: obj={obj_r:.8f}")

    # === Phase 7: Peak Shaving ===
    print(f"\n=== Phase 7: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.15, 60.0)
        a_ps = peak_shave(best_seq, alloc, verbose=True)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Peak shaving: obj={best_score:.8f} **")
        else:
            print(f"  Peak shaving: obj={obj_ps:.8f}")

    # === Phase 8: Final Intensive Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final LP
    if remaining > 25:
        alloc = min(remaining * 0.4, 150.0)
        a_final = lp_optimize(best_seq, alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final high-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.35, 60.0)
        a_final2 = gd_optimize(best_seq, alloc, tau_init=500.0, tau_end=3000.0, verbose=True)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Final CD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.3, 40.0)
        a_final3 = coordinate_descent(best_seq, alloc, verbose=True)
        obj_final3 = compute_obj(a_final3)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
