def solve_lp(a):
    n = len(a)
    # Build A matrix: A[k, j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    
    # LP: min M s.t. A@g - M <= 0, sum(g) = 1, g >= 0, M >= 0
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1  # minimize M
    
    # Inequality constraints: A@g - M <= 0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1
    b_eq = np.array([1.0])
    
    # Bounds: g >= 0, M >= 0
    bounds = [(0, None)] * (n+1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        g = res.x[:n]
        M = res.x[-1]
        return g, M
    return None, None

"""
Optimizes step function coefficients to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses fast LP-based alternating minimization (scipy HiGHS), FFT-based gradient
polishing with backtracking line search, multi-length exploration, and diverse
structured initializations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy HiGHS solver.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    
    A = build_conv_matrix(a_norm)
    num_rows = 2 * n - 1
    
    # Variables: [g_0, ..., g_{n-1}, M]
    # min M
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A @ g - M <= 0
    A_ub = np.hstack([A, -np.ones((num_rows, 1))])
    b_ub = np.zeros(num_rows)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g))
            if gs < 1e-10:
                return None, None
            return g / gs, float(res.x[-1])
    except Exception:
        pass
    
    return None, None


def smoothed_gradient(a, tau):
    """
    Compute gradient of smoothed max of (a*a) using log-sum-exp.
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p.astype(np.complex128), N)
    A = fft(a.astype(np.complex128), N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def gradient_polish(a, time_budget, tau_init=30.0, tau_max=3000.0):
    """
    Gradient descent with backtracking line search on smoothed objective.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    tau = tau_init
    lr = 1.0 / np.sqrt(n)

    while time.monotonic() - start < time_budget:
        g = smoothed_gradient(a, tau)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        
        # Project gradient onto the tangent space of the simplex
        g_tangent = g - np.mean(g)
        
        # Backtracking line search
        alpha = lr
        a_old = a.copy()
        obj_old = compute_obj(a_old)
        improved = False
        
        for _ in range(10):
            a_try = a_old - alpha * g_tangent
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < obj_old - 1e-12:
                a = a_try
                improved = True
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                break
            alpha *= 0.5
        
        if not improved:
            # Try a smaller base lr
            lr *= 0.7
            if lr < 1e-10:
                break
        
        tau = min(tau * 1.05, tau_max)

    return best_a

            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall_count = 0
            if verbose and iter_count % 50 == 0:
                print(f"    LP alt {iter_count}: obj={best_obj:.10f}, t={best_t:.3f}, "
                      f"lp_time={lp_time:.3f}s")
        else:
            stall_count += 1
            if stall_count >= 5:
                # Perturb to escape
                strength = 0.005 + 0.002 * min(stall_count, 15)
                pert = strength * rng.standard_normal(n)
                a = a * (1 + pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0

    if verbose:
        print(f"    LP alternating: {iter_count} iterations, best_obj={best_obj:.10f}")

    return best_a


def structured_init(n, mode='triangular'):
    """Generate structured initial sequences."""
    if mode == 'uniform':
        return np.ones(n) / n
    elif mode == 'triangular':
        x = np.linspace(0, 1, n)
        a = 1 - np.abs(2 * x - 1)  # triangle peaking at center
        return a / a.sum()
    elif mode == 'bimodal':
        x = np.arange(n)
        center = n // 2
        a = np.exp(-0.5 * ((x - center) / (n * 0.2))**2)
        # Add a second mode
        a += 0.5 * np.exp(-0.5 * ((x - center * 0.3) / (n * 0.1))**2)
        return a / a.sum()
    elif mode == 'cosine':
        x = np.arange(n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 'random_sparse':
        rng = np.random.default_rng(123)
        a = rng.dirichlet(np.ones(n) * 0.5)
        return a
    else:
        return np.ones(n) / n


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s
