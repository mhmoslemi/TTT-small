"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive mirror descent with FFT-accelerated subgradients, LP-based
polishing via scipy linprog, parametric initialization, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_subgrad(p, a):
    """Compute 2 * sum_k p[k] * a[k-j] for j=0..n-1 using FFT.
    p: length 2n-1 distribution over convolution indices.
    a: length n, the sequence.
    Returns gradient of length n.
    """
    n = len(a)
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    h = ifft(np.conj(A) * P).real[:n]
    return 2.0 * h

def mirror_descent_step(a, gradient, lr):
    """Multiplicative (mirror) descent step with entropy regularizer."""
    # a_new[j] = a[j] * exp(-lr * gradient[j])
    # For numerical stability:
    lg = gradient - gradient.max()
    weights = np.exp(-lr * lg)
    a_new = a * weights
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s

def lp_direction(a, use_scipy=True):
    """Solve LP: min M s.t. A(a)@g <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    if n == 0:
        return None, None
    
    a_norm = a / np.sum(a)
    
    # Build the constraint matrix implicitly
    # The LP is:
    # min M
    # s.t. sum_j A[k,j] * g[j] - M <= 0 for k=0..2n-2  (2n-1 constraints)
    #      sum_j g[j] = 1
    #      g[j] >= 0 for j=0..n-1
    #      M is free (but will be >= 0 in practice)
    
    # Variables: [g_0, ..., g_{n-1}, M]  (n+1 variables)
    # A_ub: (2n-1) x (n+1) matrix
    #   Row k: [A[k,0], ..., A[k,n-1], -1]
    # b_ub: zeros(2n-1)
    
    # A_eq: 1 x (n+1) = [1, 1, ..., 1, 0]
    # b_eq: [1]
    
    # c: [0, ..., 0, 1] (minimize M)
    
    # Bounds: g[j] in [0, inf), M in [-inf, inf]
    
    # For scipy linprog, we need to provide A_ub explicitly.
    # For large n, this is a (2n-1) x (n+1) matrix which could be large.
    # For n=800: ~1600 x 801 = ~1.3M elements. Should be OK.
    
    # Build A_ub
    A_ub = np.zeros((2 * n - 1, n + 1))
    for j in range(n):
        A_ub[j:j+n, j] = a_norm
    A_ub[:, n] = -1.0
    
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = g.sum()
            if s > 1e-15:
                g = g / s
            M = res.x[n]
            return g, M
    except Exception:
        pass
    
    return None, None

def lp_polish(a, time_budget, max_iters=30, verbose=False):
    """Iterative LP polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g, M = lp_direction(a)
        if g is None:
            break
        
        # Line search
        best_t = 0.5
        best_t_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-10:
            a_try = (1 - best_t) * a + best_t * g
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            best_obj = best_t_obj
            best_a = a_try
            a = a_try
            if verbose:
                print(f"    LP iter {it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:

    return best_a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses scipy linprog for fast LP polishing, FFT-accelerated mirror descent with
adaptive active sets, top-k subgradient method, multi-n exploration, and
intensive final polishing. Key improvements: faster LP via scipy (vs cvxpy),
mirror descent with entropic regularization, and systematic n sweep.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_subgrad(p, a):
    """Compute gradient 2 * sum_k p[k] * a[k-j] for j=0..n-1 via FFT.
    p: length 2n-1, a: length n."""
    n = len(a)
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    h = ifft(np.conj(A) * P).real[:n]
    return 2.0 * h


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho = np.nonzero(u * idx > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return w / s


def solve_lp(a):
    """Solve LP: min M s.t. A(a_norm)@g <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    if n == 0 or n > 5000:
        return None, None
    a_norm = a / np.sum(a)

    # Variables: [g_0,...,g_{n-1}, M], total n+1
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M

    # A_ub: (2n-1) x (n+1), b_ub = 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = a_norm
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    # A_eq: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(-np.inf, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x[:n], 0.0)
            s = g.sum()
            if s > 1e-15:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_polish(a, time_budget, max_iters=50, verbose=False):
    """LP-based iterative polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g, M = solve_lp(a)
        if g is None:
            break

        # Line search over mixing parameter
        best_t = 0.3
        best_t_obj = best_obj
        t_vals = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
        for t in t_vals:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-12:
            a_try = (1.0 - best_t) * a + best_t * g
            a_try = np.maximum(a_try, 0.0)
            a_try = a_try / a_try.sum()
            best_obj = best_t_obj
            best_a = a_try.copy()
            a = a_try
            stall = 0
            if verbose:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}, "
                      f"elapsed={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 4:

    return best_a

    # Phase B: LP polish
    remaining = time_budget - (time.monotonic() - start)
    alloc_b = min(remaining * 0.35, remaining - 5)
    if alloc_b > 10:
        a_try = lp_polish(best_a, time_budget=alloc_b, max_iters=30, verbose=verbose)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            if verbose:
                print(f"      LP: obj={best_obj:.8f}")

    # Phase C: Subgradient (different perspective)
    remaining = time_budget - (time.monotonic() - start)
    alloc_c = min(remaining * 0.2, remaining - 5)
    if alloc_c > 5:
        a_try = subgrad_opt(best_a, time_budget=alloc_c, verbose=verbose)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            if verbose:
                print(f"      SG: obj={best_obj:.8f}")

    # Phase D: Final LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a_try = lp_polish(best_a, time_budget=min(remaining - 2, 40),
                          max_iters=20, verbose=verbose)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            if verbose:
                print(f"      Final LP: obj={best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-method optimization with n-sweep.
    Combines mirror descent, LP polishing, and subgradient methods.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0  # safety margin

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = diverse_init(800, rng, mode=5)
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800, raised cosine): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting optimization, n={n_best}, obj={best_score:.8f}")

    # === Phase 1: Quick optimization on parent n ===
    print(f"\n=== Phase 1: Optimize parent (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1 = min(remaining * 0.25, 300.0)
    if p1 > 30:
        a_opt = optimize_single(best_seq, time_budget=p1, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 1 improved: obj={best_score:.8f}")
        else:
            print(f"  Phase 1: no improvement ({obj_opt:.8f})")

    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n exploration ===")
    remaining = deadline - time.monotonic()
    p2 = min(remaining * 0.35, 500.0)

    # Generate candidate n values
    n_candidates = set()
    for delta in [-200, -100, -50, 0, 50, 100, 200, 300, 500, 800, 1200]:
        n_try = n_best + delta
        if 200 <= n_try <= 5000:
            n_candidates.add(n_try)
    # Also add some absolute values
    for n_abs in [400, 500, 600, 700, 900, 1000, 1200, 1500, 2000, 2500]:
        if 200 <= n_abs <= 5000:
            n_candidates.add(n_abs)
    n_candidates = sorted([n for n in n_candidates if n != n_best])

    # Limit to top candidates (closest to best_n first, then spread out)
    n_candidates = sorted(n_candidates, key=lambda x: abs(x - n_best))[:12]
    print(f"  n candidates: {n_candidates}")

    per_n_budget = p2 / max(1, len(n_candidates))
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(per_n_budget, remaining - 30)
        if alloc < 30:
            continue

        a_try = resample_seq(best_seq, n_try)
        a_opt = optimize_single(a_try, time_budget=alloc, verbose=False)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 3: Diverse restarts at best n ===
    print(f"\n=== Phase 3: Diverse restarts (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p3 = min(remaining * 0.2, 200.0)
    n_restarts = min(6, max(2, int(p3 / 40)))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(p3 / n_restarts, remaining - 20)
        if alloc < 20:
            break

        mode = 5 + r  # Try various window functions
        a_r = diverse_init(n_best, rng, mode=mode % 11)
        # Blend with current best for warm start
        blend = 0.4
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0.0)
        s_r = a_r.sum()
        if s_r > 1e-10:
            a_r = a_r / s_r

        a_opt = optimize_single(a_r, time_budget=alloc, verbose=False)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode%11}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_try:.8f}")

    # === Phase 4: Intensive final polish ===
    print(f"\n=== Phase 4: Intensive final polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s")

    # Round 1: LP
    if remaining > 30:
        alloc = min(remaining * 0.25, 100.0)
        a_f = lp_polish(best_seq, time_budget=alloc, max_iters=40, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Round 2: Mirror descent with high iterations
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 100.0)
        a_f = mirror_descent_opt(best_seq, time_budget=alloc, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # Round 3: LP again
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 80.0)
        a_f = lp_polish(best_seq, time_budget=alloc, max_iters=30, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Ultra LP: obj={best_score:.8f} **")

    # Round 4: Subgradient
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.2, 50.0)
        a_f = subgrad_opt(best_seq, time_budget=alloc, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Round 5: One more LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 60.0)
        a_f = lp_polish(best_seq, time_budget=alloc, max_iters=25, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
