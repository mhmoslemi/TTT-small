Uses smooth (log-sum-exp) descent with Nesterov acceleration,

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def conv_gradient(a, k):
    """
    Compute gradient of conv(a, k) = sum_i a_i * a_{i+k} with respect to a.
    Returns a vector of length n.
    grad_k[j] = a[j+k] + a[j-k] (with boundary handling)

    d/d a_j = a_{j+k} (if j+k < n) + a_{j-k} (if j-k >= 0)

    n = len(a)

        grad[:n-k] += a[k:]
        # a_{j-k} for valid j: j from k to n-1
        grad[k:] += a[:n-k]
    return grad

    if s < 1e-10:

        progress = min(1.0, elapsed / time_budget)

        A = fft(y, N_fft)

        grad_smooth = 2.0 * g_full[:n]
