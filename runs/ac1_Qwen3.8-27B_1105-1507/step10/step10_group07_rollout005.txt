import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute 2n * max(conv) / (sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * conv.max()

def compute_grad_smoothed(a, p):
    """
    Compute gradient of 2n * (1/tau) * log(sum exp(tau * (a*a))) w.r.t. a.
    a: length n, p: length 2n-1 (softmax weights)
    Returns gradient of same length as a.
    """
    n = len(a)
    # Full convolution of p (len 2n-1) and a (len n), length 3n-2
    size = 1
    while size < 3*n - 1:
        size <<= 1
    P = rfft(p, size)
    A = rfft(a, size)
    conv_full = irfft(P * A, size)[:3*n-2]
    
    # grad[j] = 2n * (conv_full[j] + conv_full[2n-2-j])
    # conv_full[2n-2-j] for j=0..n-1 gives indices 2n-2, 2n-3, ..., n-1
    # = conv_full[n-1:2n-1][::-1]
    grad = 2.0 * n * (conv_full[:n] + conv_full[n-1:2*n-1][::-1])
    return grad

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
