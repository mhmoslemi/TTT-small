"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid approach: subgradient method for coarse optimization, LP best-response
iteration with adaptive step size for fine polishing, cross-n exploration, and
final high-tau smoothed GD. Key improvements over previous: subgradient steps
to escape local minima, backtracking LP step size, and more aggressive iteration.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = conv - exact_max
    max_shifted = np.max(shifted)
    if max_shifted > 500:
        shifted = shifted - max_shifted
        exact_max += max_shifted
    exp_vals = np.exp(tau * shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def subgradient_step(a, lr, top_k=5):
    """
    Take a subgradient step using the top-k max lags of the convolution.
    Returns updated a (normalized) and the subgradient.
    """
    n = len(a)
    conv = compute_conv(a)
    # Get top-k lags
    top_indices = np.argsort(conv)[-top_k:]
    # Compute weighted subgradient
    sg = np.zeros(n)
    for idx in top_indices:
        m = int(idx)
        # sg[j] += 2 * a[m-j] if 0 <= m-j < n
        lo = max(0, m - n + 1)
        hi = min(m, n - 1)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            sg[indices] += 2.0 * a[m - indices]
    a_new = a - lr * sg
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy(), sg
    return a_new / s, sg


def subgradient_phase(a, time_budget, verbose=False):
    """
    Subgradient method with decreasing step size and periodic restart.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    lr = 0.5 / np.sqrt(n)
    best_it = 0
    patience = 300

    while time.monotonic() - start < time_budget:
        if it - best_it > patience:
            # Restart from best with perturbation
            a = best_a.copy()
            rng = np.random.default_rng(it)
            a += 0.01 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            lr = 0.5 / np.sqrt(n)
            best_it = it

        a_new, sg = subgradient_step(a, lr, top_k=3)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            # Slightly increase lr if improving
            lr = min(1.0 / np.sqrt(n), lr * 1.001)
        else:
            lr *= 0.999

        a = a_new
        it += 1
        lr = max(1e-8, lr)

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    Subgrad it={it}, obj={best_obj:.10f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def lp_best_response(a):
    """
    Solve the LP: min M s.t. (a*g)_k <= M for all k, g>=0, sum(g)=1.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_phase(a, time_budget, verbose=False, seed=42):
    """
    LP best-response iteration with adaptive step size and backtracking.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    t = 0.1  # initial blending factor
    stall = 0
    max_stall = 30
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = lp_best_response(a)
        if g_lp is None:
            break

        # Try multiple step sizes
        t_candidates = [t * 0.5, t * 0.7, t, t * 1.3, t * 1.8, min(1.0, t * 2.5)]
        t_candidates = sorted(set([max(0.01, min(1.0, x)) for x in t_candidates]))

        best_t_obj = np.inf
        best_t_val = t
        best_a_local = a.copy()

        for tc in t_candidates:
            a_try = (1.0 - tc) * a + tc * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = tc
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            t = best_t_val
            stall = 0
            if verbose and it % 10 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it={it}, obj={best_obj:.10f}, t={t:.4f}, M_lp={M_lp:.6f}, t_elapsed={elapsed:.0f}s")
        else:
            stall += 1
            # Perturb to escape
            if stall > 5:
                pert = 0.005 * rng.standard_normal(n)
                a = a + pert
                a = np.maximum(a, 0)
                a = a / a.sum()
                t = max(0.03, t * 0.5)
            elif stall > 2:
                # Try a random direction
                direction = rng.standard_normal(n)
                a_mix = a + 0.02 * direction
                a_mix = np.maximum(a_mix, 0)
                sm = float(np.sum(a_mix))
                if sm > 1e-10:
                    a = a_mix / sm
                t = max(0.03, t * 0.7)
            if stall > max_stall:
                break

        it += 1

    return best_a


def high_tau_gd(a, time_budget, verbose=False):
    """
    Smoothed GD with very high tau values for final polishing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau_levels = [200, 500, 1000, 2000, 4000]
    beta = 0.85

    for tau in tau_levels:
        if time.monotonic() - start >= time_budget:
            break
        lr = 1.5 / np.sqrt(n)
        v = np.zeros(n)
        it_level = 0
        best_it = 0
        it_global = 0

        while time.monotonic() - start < time_budget:
            if it_level >= 2000:
                break
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it_global
            a = a_new
            it_level += 1
            it_global += 1
            lr = max(1e-8, lr * 0.99995)

            if it_global - best_it > 1500:
                a = best_a.copy()
                v = np.zeros(n)
                lr = 1.5 / np.sqrt(n)
                best_it = it_global

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    HTGD tau={tau:5d}: obj={best_obj:.10f}, t={elapsed:.0f}s")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    if n >= 4:
        from scipy.interpolate import CubicSpline
        cs = CubicSpline(x_old, a)
        a_new = cs(x_new)
    else:
        a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

    elif mode == 4:

    print(f"\n=== Phase 4: High-Tau GD Polish ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 150.0)
    if p4_budget > 30:
        a4 = high_tau_gd(best_seq, time_budget=p4_budget, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  HTGD: obj={best_score:.10f} **")
        else:
            print(f"  HTGD: obj={obj4:.10f}")

    # === Phase 5: Cross-n exploration ===
    print(f"\n=== Phase 5: Cross-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 100),
        n_current,
        n_current + 100,
        n_current + 200,
        n_current + 400,
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample(best_seq, n_try)

        # Quick GD
        alloc1 = min(remaining * 0.15, 40.0)
        if alloc1 > 15:
            a_try = high_tau_gd(a_try, time_budget=alloc1, verbose=False)

        # LP if feasible
        remaining = deadline - time.monotonic()
        if n_try <= 1500 and remaining > 40:
            alloc2 = min(remaining * 0.15, 50.0)
            if alloc2 > 20:
                a_try = lp_phase(a_try, time_budget=alloc2, verbose=False, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    # Final LP
    if remaining > 30 and n_final <= 2000:
        alloc = min(remaining * 0.3, 100.0)
        a_f1 = lp_phase(best_seq, time_budget=alloc, verbose=True, seed=9999)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    # Final high-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.4, 80.0)
        a_f2 = high_tau_gd(best_seq, time_budget=alloc, verbose=True)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final HTGD: obj={best_score:.10f} **")

    # One more LP if time
    remaining = deadline - time.monotonic()
    if remaining > 20 and n_final <= 2000:
        alloc = min(remaining * 0.5, 60.0)
        a_f3 = lp_phase(best_seq, time_budget=alloc, verbose=False, seed=12345)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Ultra Final LP: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
