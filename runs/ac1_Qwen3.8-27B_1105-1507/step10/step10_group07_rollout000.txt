"""
Multi-stage optimizer for minimizing peak autocorrelation ratio.
Uses smoothed gradient descent, LP polishing, coordinate descent,
and adaptive length exploration with diverse restart strategies.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_descent(a, time_budget=100.0, tau_init=20.0, tau_end=500.0, seed=42):
    """Smoothed gradient descent with adaptive temperature."""

    v = np.zeros(n)
    rng = np.random.default_rng(seed)
    stall = 0

def solve_lp_direction(a, M_target=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    This finds the direction g that minimizes the max autocorrelation
    when convolved with current a.
    """
    n = len(a)
    a = a / a.sum()

    # Build the convolution matrix A such that A @ g = conv(a, g)
    # A has shape (2n-1, n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

        if best_t > 0 and best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = (1 - best_t) * best_a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            s_ba = best_a.sum()
            if s_ba > 1e-10:
                best_a = best_a / s_ba
            a = best_a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                scale = 0.008 + 0.004 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                no_improve = 0
                if it > 80:
                    break

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def coordinate_descent(a, time_budget=150.0, seed=42):
    """
    Coordinate descent: iterate over elements, for each element find the
    optimal value that minimizes the peak autocorrelation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        improved = False
        # Random order for diversity
        order = rng.permutation(n)

        for idx in order:
            if time.monotonic() - start >= time_budget:
                break

            # Try different scales for a[idx]
            base = a[idx]
            if base < 1e-15:
                candidates = [0.001, 0.01, 0.05] / n
            else:
                candidates = [base * f for f in [0.1, 0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]]

            best_val = base
            best_local_obj = best_obj

            for val in candidates:
                a_trial = a.copy()
                a_trial[idx] = max(0, val)
                s_trial = a_trial.sum()
                if s_trial < 1e-10:
                    continue
                a_trial = a_trial / s_trial
                obj_trial = compute_obj(a_trial)
                if obj_trial < best_local_obj - 1e-14:
                    best_local_obj = obj_trial
                    best_val = val

            if best_val != base:
                a[idx] = best_val
                a = a / a.sum()
                if best_local_obj < best_obj - 1e-14:
                    best_obj = best_local_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            # Perturb to escape
            scale = 0.02
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
            if time.monotonic() - start > time_budget * 0.8:
                break

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def random_redirect(a, n_new, seed):
    """Create a diverse random perturbation, possibly with different length."""
    rng = np.random.default_rng(seed)
    if n_new is not None and n_new != len(a):
        a_base = resample(a, n_new)
    else:
        a_base = a.copy()
    n = len(a_base)

    # Mix of different perturbation strategies
    strategy = rng.integers(0, 4)
    if strategy == 0:
        # Gaussian multiplicative
        a_new = a_base * np.exp(rng.normal(0, 0.15, n))
    elif strategy == 1:
        # Uniform additive
        a_new = a_base + rng.uniform(0, 0.5, n) / n
    elif strategy == 2:
        # Exponential distribution
        a_new = rng.exponential(scale=1.0/n, size=n)
        a_new = 0.5 * a_base + 0.5 * a_new
    else:
        # Spike perturbation
        a_new = a_base.copy()
        n_spikes = max(5, n // 50)
        idx = rng.choice(n, size=n_spikes, replace=False)
        a_new[idx] *= rng.uniform(0.1, 3.0, n_spikes)

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new
