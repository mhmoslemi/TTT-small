    """Find max delta such that a[i0] can be increased by delta while keeping all conv[k] <= M."""
    n = len(a)
    conv = compute_conv(a)
    
    # Find max delta
    delta_max = np.inf
    for k in range(2*n - 1):
        # Contribution of a[i0] to conv[k]
        # Term i=i0: valid if max(0,k-n+1) <= i0 <= min(k, n-1)
        # And k-i0 in [0, n-1]
        # The increase per unit delta:
        contrib = 0.0
        # i = i0 term: a[i0] * a[k-i0], derivative w.r.t. a[i0] is a[k-i0]
        if 0 <= k - i0 < n:
            contrib += a[k - i0]
        # i = k-i0 term (different from i0): a[k-i0] * a[i0], derivative w.r.t. a[i0] is a[k-i0]
        if 0 <= k - i0 < n and (k - i0) != i0:
            contrib += a[k - i0]
        # Special case k = 2*i0: derivative of a[i0]^2 is 2*a[i0], handled above as 2*a[k-i0] = 2*a[i0]
        
        if contrib > 0:

            if slack < 0:

from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def coord_ascent_pass(a, M, conv, n):
    """One full pass of coordinate ascent. Returns updated a and conv."""
    for i in range(n):
        # Find max delta for a[i]
        k_low = i
        k_high = min(i + n, 2 * n - 1)
        if k_low >= k_high:
            continue
        idx = np.arange(k_low, k_high) - i  # 0, 1, ..., min(n, 2n-1-i)-i
        contrib = 2.0 * a[idx]
        mask = contrib > 1e-15
        if not np.any(mask):
            continue
        slack = (M - conv[k_low:k_high])[mask]
        c = contrib[mask]
        delta = float(np.min(slack / c))
        delta = max(0.0, delta)
        if delta > 1e-15:
            a[i] += delta
            # Update conv
            conv[k_low:k_high] += delta * contrib  # approximate (ignores delta^2 for k=2i)
    return a, conv

def coord_descent_optimize(a, time_budget=60.0, verbose=False):
    """Optimize via coordinate ascent + normalization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Coordinate ascent
        a_new, conv_new = coord_ascent_pass(a.copy(), M, conv.copy(), n)

        s_new = float(np.sum(a_new))
        if s_new <= s * (1 + 1e-12):

        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        it += 1
        
        if verbose and it % 10 == 0:
            print(f"  CD it{it}: obj={best_obj:.8f}, sum_ratio={s_new/s:.6f}, t={time.monotonic()-start:.1f}s")
    
    return best_a

b_sequence = np.convolve(sequence, sequence)
max_b = max(b_sequence)
sum_a = np.sum(sequence)
return float(2 * n * max_b / (sum_a**2))
```

So it works with the raw sequence (not normalized). The objective is the same: 2n*max_conv/sum^2. Since this is scale-invariant (if we multiply a by c, both max_conv and sum scale, and the ratio is unchanged), we can work with any normalization.

```python
"""
Coordinate descent + LP polish optimization for step function coefficients.
Uses fast coordinate-wise ascent with exact constraint checking,
combined with diverse initializations and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def power_u_init(n, power=4.0, floor=0.05):
    """Power-law U-shape initialization."""
    x = np.linspace(0, 1, n)
    t = (2 * x - 1) ** 2
    a = (t ** power + floor) ** (-0.5)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_u_init(n, alpha, beta):
    """Parameterized U-shape: a_i = (1 + alpha * t_i^beta)^(-1) where t_i = |2i/(n-1) - 1|."""
    x = np.linspace(0, 1, n)
    t = np.abs(2 * x - 1)
    a = 1.0 / (1.0 + alpha * t ** beta)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def coord_ascent(n, a, M, max_rounds=50):
    """
    Coordinate ascent: increase sum(a) while keeping conv(a,a) <= M.
    Returns updated a and the new sum.
    """
    a = a.copy()
    conv = compute_conv(a)
    
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            # Find max delta for a[i]
            k_low = i
            k_high = min(i + n, 2 * n - 1)
            if k_low >= k_high:
                continue
            
            # contrib[k] = increase in conv[k] per unit increase in a[i]
            idx = np.arange(k_low, k_high) - i
            contrib = 2.0 * a[idx]
            
            mask = contrib > 1e-15
            if not np.any(mask):
                continue
            
            slack = (M - conv[k_low:k_high])[mask]
            c = contrib[mask]
            
            # delta_max = min(slack / c)
            delta_max = float(np.min(slack / c))
            delta_max = max(0.0, delta_max)
            
            if delta_max > 1e-12:
                a[i] += delta_max
                # Update conv (first-order approximation)
                conv[k_low:k_high] += delta_max * contrib
                improved = True
        
        if not improved:
            break
    
    return a, conv


def coord_descent_optimize(a, time_budget=30.0, verbose=False):
    """
    Main coordinate descent optimization.
    Iteratively: fix M, maximize sum via coord ascent, normalize, repeat.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Coordinate ascent
        a_new, _ = coord_ascent(n, a, M, max_rounds=20)
        s_new = float(np.sum(a_new))
        
        if s_new > 1.0 + 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                stall_count = 0
            else:
                stall_count += 1
            
            a = a_new
        else:
            stall_count += 1
        
        it += 1
        
        if verbose and it % 10 == 0:
            print(f"  CD it{it}: obj={best_obj:.10f}, s_new={s_new:.6f}, t={time.monotonic()-start:.1f}s")
        
        # If stalled, try perturbation
        if stall_count >= 5:
            if stall_count < 10:
                # Small random perturbation
                a = best_a + np.random.default_rng(it).normal(0, 0.001, n)
                a = np.maximum(a, 0)
                a = a / a.sum()
            else:
                break
            stall_count = 0
    
    return best_a


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Build constraint matrix A such that A @ g = conv(a_norm, g)
    # conv(a_norm, g)[k] = sum_i a_norm[i] * g[k-i]
    # For fixed k, this is sum_i a_norm[i] * g[k-i] = sum_j a_norm[k-j] * g[j]
    # So A[k, j] = a_norm[k-j] if 0 <= k-j < n, else 0
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a_norm[k-j] for valid k
        # k-j in [0, n-1] → k in [j, j+n-1]
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a_norm[:k_end - k_start]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget=60.0, verbose=False, max_iters=50):
    """LP-based polishing: solve LP, line search blend, repeat."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Try several blend factors
        best_t = 0
        best_blend_obj = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t
        
        if best_t > 0 and best_blend_obj < best_obj - 1e-14:
            a = best_a if best_t == 0 else (1 - best_t) * a + best_t * g_lp
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 0.01:
                a = a / s
                best_obj = best_blend_obj
                best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        it += 1
        if verbose and it % 5 == 0:
            print(f"  LP it{it}: obj={best_obj:.10f}, t={best_t:.3f}, M_lp={M_lp:.8f}, elapsed={time.monotonic()-start:.1f}s")
        
        if stall >= 4:
            break
    
    return best_a
