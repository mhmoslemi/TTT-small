```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a multi-phase strategy: fast LP alternating with line search, Nesterov gradient
polishing with adaptive smoothing, and multi-length exploration. The LP solves the
best-response problem at each step, while gradient polishing uses FFT-accelerated
smoothed gradients for fine refinement. Multi-start and symmetric initializations
help navigate the non-convex landscape.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a, row_indices=None):
    """Build matrix A (|row_indices| x n) such that A @ g = conv(a, g) at those indices."""
    n = len(a)
    if row_indices is None:
        row_indices = np.arange(2 * n - 1)
    row_indices = np.asarray(row_indices)
    A = np.zeros((len(row_indices), n))
    for row, k in enumerate(row_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[row, lo:hi + 1] = a[lo:hi + 1]
    return A


def solve_lp_full(a):
    """Solve the full LP for best response. Returns (g_opt, M_opt) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_sparse(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_reduced(a, top_k=30):
    """Solve LP using only the top-k active constraints. Faster approximation."""
    n = len(a)
    a_norm = a / np.sum(a)
    conv = compute_conv(a_norm)
    top_indices = np.argsort(conv)[-top_k:]
    A = build_conv_matrix_sparse(a_norm, top_indices)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS]:
        try:

                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient(a, tau):
    """Compute gradient of smoothed max of (a*a)_k using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def nesterov_gd_polish(a, time_budget, tau_init=20.0, tau_max=3000.0):
    """Nesterov accelerated gradient descent on smoothed objective with simplex projection."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 1.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = project_simplex(y)

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adaptive_lr = 0.3 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.05 * base_lr), 5.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adaptive_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new
        lr *= 0.9995
        tau = min(tau * 1.05, tau_max)
        it += 1

        if patience > 300:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr
            tau = max(tau * 0.5, tau_init)
            beta = 0.8
            patience = 0

    return best_a

                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-13:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            t_current = max(0.05, best_local_t * 0.95)
            stall = 0
            if verbose and it % 50 == 0:
                print(f"    LP line-search {it}: obj={best_obj:.10f}, t={t_current:.4f}")
        else:
            stall += 1
            a = (1 - t_current * 0.5) * a + t_current * 0.5 * g_lp
            a = np.maximum(a, 0)
            a = a / a.sum()

            if stall >= 8:
                # Escalating perturbation
                strength = 0.005 + 0.003 * min(stall, 15)
                noise = rng.standard_normal(n) * strength
                a = project_simplex(best_a + noise)
                t_current = max(0.2, t_current * 1.3)
                stall = 0
            elif stall >= 4:
                a = project_simplex(best_a + 0.003 * rng.standard_normal(n))

        # Ramp down t over time
        if it > 100:
            t_current = max(0.05, t_current * 0.99)

    if verbose:
        print(f"    LP line-search done: {it} iters, best_obj={best_obj:.10f}")
    return best_a


def peak_chasing_refine(a, time_budget, seed=42, verbose=False):
    """
    Refine by specifically targeting the top peaks in the convolution.
    For each peak lag k, find the direction that most reduces (a*a)_k
    while keeping sum(a)=1 and a>=0.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Find top peaks
        top_k = 5
        peak_indices = np.argsort(conv)[-top_k:]

        for k in peak_indices:
            if time.monotonic() - start > time_budget:
                    break
            k = int(k)
            # Direction to reduce (a*a)_k: d/d a_j (a*a)_k = a_{j+k} + a_{j-k}
            # We want to move in the direction that reduces this
            # Grad of (a*a)_k w.r.t. a_j:
            grad_k = np.zeros(n)
            for j in range(n):
                val = 0.0
                if 0 <= j + k < n:
                    val += a[j + k]
                if 0 <= j - k < n:
                    val += a[j - k]
                grad_k[j] = val

            # Move opposite to gradient, project to simplex
            step = 0.01
            a_new = project_simplex(a - step * grad_k / (np.linalg.norm(grad_k) + 1e-10))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new

    if verbose:
        print(f"    Peak chasing: obj={best_obj:.10f}")
    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def generate_initializations(n, parent, rng):
    """Generate diverse initializations for a given length n."""
    inits = []
    x = np.arange(n)

    # 1. Parent resampled
    if parent is not None and len(parent) > 1:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        p = np.interp(x_new, x_old, parent)
        p = np.maximum(p, 0)
        if p.sum() > 0.01:
            inits.append(p / p.sum())

    # 2. Uniform
    inits.append(np.ones(n).copy())

    # 3. Sine wave (fundamental)
    s = np.abs(np.sin(np.pi * x / n)) + 0.05
    inits.append(s)

    # 4. Sine wave (2nd harmonic)
    s = np.abs(np.sin(2 * np.pi * x / n)) + 0.05
    inits.append(s)

    # 5. Two-valued random
    vals = np.where(rng.random(n) < 0.5, 1.0, 0.3)
    inits.append(vals)

    # 6. Symmetric Gaussian bump
    center = n / 2.0
    s = np.exp(-((x - center) / (n * 0.15)) ** 2)
    inits.append(s)

    # 7. Periodic pattern
    period = max(2, n // 7)
    s = np.where(np.arange(n) % period < period // 2, 1.0, 0.4)
    inits.append(s)

    # 8. Random
    r = rng.random(n)
    inits.append(np.maximum(r, 0.01))

    # Normalize all
    normalized = []
    for init in inits:
        init = np.maximum(init, 0)
        s = float(np.sum(init))
        if s > 0.01:
            normalized.append(init / s)
    return normalized


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-length exploration with LP alternating,
    gradient polishing, and peak-chasing refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0)
        if parent.size == 0 or parent.sum() < 0.01:
            parent = None
        else:
            parent = parent / parent.sum()
            print(f"Parent loaded: n={len(parent)}, obj={compute_obj(parent):.10f}")
    except Exception:
        parent = None
        print("No valid parent, using defaults")

    best_a = None
    best_obj = np.inf
    rng = np.random.default_rng(seed)

    # Determine which lengths to try
    if parent is not None:
        n_parent = len(parent)
        n_values = [n_parent, n_parent - 50, n_parent + 50, n_parent - 100,
                    n_parent + 100, n_parent - 150, n_parent + 150,
                    n_parent - 200, n_parent + 200,
                    400, 500, 600, 700, 800, 900, 1000, 1200, 1400, 1600]
    else:
        n_values = [500, 600, 700, 800, 900, 1000, 1200, 1400]

    # Deduplicate and filter
    n_values = sorted(set([n for n in n_values if 200 <= n <= 2000]))
    # Sort by priority: closest to parent first, then by centrality
    if parent is not None:
        n_parent = len(parent)
        n_values.sort(key=lambda n: abs(n - n_parent))
    else:
        n_values.sort(key=lambda n: abs(n - 800))

    print(f"\nLengths to try: {n_values[:10]}")

    # === Phase 1: Multi-length multi-start exploration ===
    print(f"\n=== Phase 1: Multi-length Exploration ===")
    phase1_deadline = start_time + budget_s * 0.55

    for n in n_values:
        remaining = phase1_deadline - time.monotonic()
        if remaining < 40:
            break

        # Time allocation: more time for lengths near the known good one
        if parent is not None:
            time_factor = max(0.3, 1.0 - 0.002 * abs(n - len(parent)))
        else:
            time_factor = max(0.3, 1.0 - 0.002 * abs(n - 800))
        alloc = min(remaining * 0.25, 180.0 * time_factor)
        alloc = max(alloc, 20.0)

        inits = generate_initializations(n, parent, rng)
        per_init_budget = alloc / max(len(inits), 1)
        best_for_n = None
        best_obj_for_n = np.inf

        for idx, init in enumerate(inits):
            rem = phase1_deadline - time.monotonic()
            if rem < 10:
                break
            b = min(per_init_budget, rem * 0.3)
            if b < 5:
                break

            a_res = lp_alternate_with_line_search(
                init, time_budget=b, seed=seed + n * 100 + idx,
                verbose=False, max_iters=min(50, int(b / 1.5)))
            obj = compute_obj(a_res)

            if obj < best_obj_for_n:
                best_obj_for_n = obj
                best_for_n = a_res.copy()

            if obj < best_obj:
                best_obj = obj
                best_a = a_res.copy()
                print(f"  ** n={n}, init={idx}: obj={obj:.10f}")

        # Refine best for this length
        remaining = phase1_deadline - time.monotonic()
        if remaining > 20 and best_for_n is not None:
            refine_budget = min(remaining * 0.3, 80.0)
            a_ref = lp_alternate_with_line_search(
                best_for_n, time_budget=refine_budget, seed=seed + n * 7,
                verbose=False, max_iters=min(80, int(refine_budget / 1.5)))
            obj_ref = compute_obj(a_ref)
            if obj_ref < best_obj_for_n:
                best_obj_for_n = obj_ref
                best_for_n = a_ref.copy()
            if obj_ref < best_obj:
                best_obj = obj_ref
                best_a = best_for_n.copy()
                print(f"  ** n={n} refined: obj={obj_ref:.10f}")

    # === Phase 2: Deep LP alternating on best candidate ===
    print(f"\n=== Phase 2: Deep LP Alternating ===")
    phase2_deadline = start_time + budget_s * 0.75
    remaining = phase2_deadline - time.monotonic()
    if remaining > 30 and best_a is not None:
        deep_budget = min(remaining * 0.8, 300.0)
        print(f"  Budget: {deep_budget:.0f}s, n={len(best_a)}")
        a_deep = lp_alternate_with_line_search(
            best_a, time_budget=deep_budget, seed=seed + 99999,
            verbose=True, max_iters=500)
        obj_deep = compute_obj(a_deep)
        if obj_deep < best_obj:
            best_obj = obj_deep
            best_a = a_deep.copy()
            print(f"  ** Deep LP: obj={best_obj:.10f}")
        else:
            print(f"  Deep LP: obj={obj_deep:.10f} (no improvement)")

    # === Phase 3: Gradient polishing ===
    print(f"\n=== Phase 3: Gradient Polishing ===")
    phase3_deadline = start_time + budget_s * 0.88
    remaining = phase3_deadline - time.monotonic()
    if remaining > 20 and best_a is not None:
        grad_budget = min(remaining * 0.7, 150.0)
        a_grad = nesterov_gd_polish(best_a, time_budget=grad_budget,
                                     tau_init=15.0, tau_max=2000.0)
        obj_grad = compute_obj(a_grad)
        if obj_grad < best_obj:
            best_obj = obj_grad
            best_a = a_grad.copy()
            print(f"  ** Gradient: obj={best_obj:.10f}")
        else:
            print(f"  Gradient: obj={obj_grad:.10f}")

    # === Phase 4: Peak chasing ===
    print(f"\n=== Phase 4: Peak Chasing ===")
    phase4_deadline = start_time + budget_s * 0.94
    remaining = phase4_deadline - time.monotonic()
    if remaining > 15 and best_a is not None:
        peak_budget = min(remaining * 0.8, 60.0)
        a_peak = peak_chasing_refine(best_a, time_budget=peak_budget,
                                      seed=seed + 55555, verbose=True)
        obj_peak = compute_obj(a_peak)
        if obj_peak < best_obj:
            best_obj = obj_peak
            best_a = a_peak.copy()
            print(f"  ** Peak chase: obj={best_obj:.10f}")
        else:
            print(f"  Peak chase: obj={obj_peak:.10f}")

    # === Phase 5: Final short LP + gradient ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15 and best_a is not None:
        lp_budget = min(remaining * 0.5, 60.0)
        if lp_budget > 5:
            a_final = lp_alternate_with_line_search(
                best_a, time_budget=lp_budget, seed=seed + 11111,
                verbose=False, max_iters=30)
            obj_f = compute_obj(a_final)
            if obj_f < best_obj:
                best_obj = obj_f
                best_a = a_final.copy()
                print(f"  ** Final LP: obj={best_obj:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            grad_budget = min(remaining * 0.8, 40.0)
            if grad_budget > 5:
                a_final2 = nesterov_gd_polish(best_a, time_budget=grad_budget,
                                               tau_init=50.0, tau_max=5000.0)
                obj_f2 = compute_obj(a_final2)
                if obj_f2 < best_obj:
                    best_obj = obj_f2
                    best_a = a_final2.copy()
                    print(f"  ** Final Grad: obj={best_obj:.10f}")

    # === Phase 6: Try a perturbed version of the best ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_a is not None:
        print(f"\n=== Phase 6: Perturbation Retry ===")
        a_pert = fourier_perturb(best_a, rng, strength=0.01)
        pert_budget = min(remaining * 0.6, 80.0)
        a_pert_res = lp_alternate_with_line_search(
            a_pert, time_budget=pert_budget, seed=seed + 77777,
            verbose=False, max_iters=40)
        obj_pert = compute_obj(a_pert_res)
        if obj_pert < best_obj:
            best_obj = obj_pert
            best_a = a_pert_res.copy()
            print(f"  ** Perturbation: obj={best_obj:.10f}")
        else:
            print(f"  Perturbation: obj={obj_pert:.10f}")

    # Final validation
    if best_a is None:
        best_a = np.ones(500) / 500

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0
        print("WARNING: evaluate_sequence returned inf, using fallback")

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
```
