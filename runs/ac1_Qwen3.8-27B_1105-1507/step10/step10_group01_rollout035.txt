    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a_norm[diff[valid]]
    return A


def solve_lp(a_norm):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M, status).
    """
    n = len(a_norm)
    A = build_conv_matrix_fast(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    best_g = None
    best_M = np.inf
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.reset()
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    if M_val < best_M:
                        best_M = M_val
                        best_g = g_val
        except Exception:
            continue
    
    if best_g is None:
        return None, None, 'infeasible'
    return best_g, best_M, 'optimal'


def line_search_quadratic(a, g, tol=1e-12):
    """
    Find optimal t in [0,1] to minimize max_k conv((1-t)a + tg, (1-t)a + tg)[k].
    Uses the quadratic structure for precision.
    """
    n = len(a)
    # Precompute conv(a,a), conv(a,g), conv(g,g)
    caa = compute_conv(a)
    cat = compute_conv(a, g) if False else _cross_conv(a, g)
    ctt = compute_conv(g)
    
    # Along the line: f(t) = max_k [(1-t)^2 * caa[k] + 2t(1-t) * cat[k] + t^2 * ctt[k]]
    # = max_k [caa[k] + t*(2*cat[k] - 2*caa[k]) + t^2*(caa[k] - 2*cat[k] + ctt[k])]
    
    def obj_at_t(t):
        val = (1-t)**2 * caa + 2*t*(1-t) * cat + t**2 * ctt
        return float(np.max(val))
    
    # First, coarse grid
    ts = np.linspace(0.0, 1.0, 200)
    objs = np.array([obj_at_t(t) for t in ts])
    best_idx = int(np.argmin(objs))
    
    # Refine with golden section
    if best_idx == 0:
        lo_t, hi_t = 0.0, ts[1]
    elif best_idx == len(ts) - 1:
        lo_t, hi_t = ts[-2], 1.0
    else:
        lo_t, hi_t = ts[best_idx - 1], ts[best_idx + 1]
    
    # Golden section
    phi = (np.sqrt(5) - 1) / 2  # ~0.618
    for _ in range(50):
        t1 = hi_t - phi * (hi_t - lo_t)
        t2 = lo_t + phi * (hi_t - lo_t)
        f1 = obj_at_t(t1)
        f2 = obj_at_t(t2)
        if f1 < f2:
            hi_t = t2
        else:
            lo_t = t1
        if hi_t - lo_t < 1e-10:
            break
    
    best_t = (lo_t + hi_t) / 2
    best_obj = obj_at_t(best_t)
    return best_t, best_obj


def _cross_conv(a, b):

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- LP-driven optimization with precise golden-section line search
- Exact subgradient method with BB step size as fallback
- Multi-length exploration and diverse restarts

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time
import cvxpy as cp


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_norm):
    n = len(a_norm)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a_norm[diff[valid]]
    return A


def solve_lp_direction(a_norm):
    n = len(a_norm)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g):
    """Golden section search for optimal t in [0,1]."""
    caa = compute_conv(a)
    cat = cross_conv(a, g)
    ctt = compute_conv(g)
    
    def f(t):
        val = (1.0-t)**2 * caa + 2.0*t*(1.0-t) * cat + t*t * ctt
        return float(np.max(val))
    
    # Coarse
    ts = np.linspace(0.0, 1.0, 100)
    objs = np.array([f(t) for t in ts])
    bi = int(np.argmin(objs))
    
    if bi == 0:
        lo, hi = 0.0, ts[1]
    elif bi == 99:
        lo, hi = ts[98], 1.0
    else:
        lo, hi = ts[bi-1], ts[bi+1]
    
    # Golden section
    golden = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(60):
        t1 = hi - golden * (hi - lo)
        t2 = lo + golden * (hi - lo)
        if f(t1) < f(t2):
            hi = t2
        else:
            lo = t1
        if hi - lo < 1e-12:
            break
    
    t = (lo + hi) / 2.0
    return t, f(t)


def subgradient_step(a, a_prev, g_prev, base_lr):
    """One subgradient step with BB step size."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    
    # Tolerance for active set
    tol = max(1e-10 * max_val, 1e-15)
    active = np.where(conv >= max_val - tol)[0]
    
    # Limit active set size
    if len(active) > 20:
        top = np.argsort(conv)[-20:]
        active = top
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            grad[lo:hi] += a[k - np.arange(lo, hi)]
    grad *= 2.0 / len(active)
    
    # BB step size
    if g_prev is not None and a_prev is not None:
        s_vec = a - a_prev
        y_vec = grad - g_prev
        ss = float(np.dot(s_vec, s_vec))
        sy = float(np.dot(s_vec, y_vec))
        if ss > 1e-15 and abs(sy) > 1e-12:
            lr = ss / abs(sy)
            lr = min(max(lr, 1e-6), 10.0 * base_lr)
        else:
            lr = base_lr
    else:
        lr = base_lr
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a, grad
    return a_new / s, grad

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def fourier_perturb(a, rng, strength=0.05):
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 6):
        freq = 2.0 * np.pi * k / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

import numpy as np

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_norm):
    n = len(a_norm)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a_norm[diff[valid]]
    return A


def solve_lp_direction(a_norm, g_init=None):
    n = len(a_norm)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    if g_init is not None:
        g.value = g_init
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_golden(a, g, n_coarse=150, n_golden=60):

    cat = cross_conv(a, g)
    ctt = compute_conv(g)

        val = (1.0 - t) * (1.0 - t) * caa + 2.0 * t * (1.0 - t) * cat + t * t * ctt
        return float(np.max(val))

    ts = np.linspace(0.0, 1.0, n_coarse)
    objs = np.empty(n_coarse)
    for i in range(n_coarse):
        objs[i] = f(ts[i])
    bi = int(np.argmin(objs))

    if bi == 0:
        lo, hi = 0.0, ts[1]
    elif bi == n_coarse - 1:
        lo, hi = ts[n_coarse - 2], 1.0
    else:
        lo, hi = ts[bi - 1], ts[bi + 1]

    golden = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(n_golden):
        t1 = hi - golden * (hi - lo)
        t2 = lo + golden * (hi - lo)

        f2 = f(t2)
        if f1 < f2:
            hi = t2
        else:
            lo = t1
        if hi - lo < 1e-13:

def compute_subgradient(a):
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    tol = max(1e-10 * max_val, 1e-15)
    active = np.where(conv >= max_val - tol)[0]
    if len(active) > 30:
        active = np.argsort(conv)[-30:]
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            grad[lo:hi] += a[k - np.arange(lo, hi)]
    grad *= 2.0 / len(active)
    return grad, max_val, len(active)

            print(f"  n={n_try}: obj={obj:.9f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.9f}, time={elapsed:.1f}s ===")
    return result


def fourier_perturb_best_seq(a, rng, strength):
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 6):
        freq = 2.0 * np.pi * k / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def blend_random_best_seq(a, rng, frac):
    n = len(a)
    noise = rng.uniform(0, 1, n) / n
    a_new = (1.0 - frac) * a + frac * noise
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def smooth_perturb_best_seq(a, rng):
    n = len(a)
    ks = max(5, n // 12)
    if ks % 2 == 0:
        ks += 1
    a_smooth = uniform_filter1d(a, size=ks)
    a_new = 0.75 * a + 0.25 * a_smooth
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    return fourier_perturb_best_seq(a_new, rng, 0.04)


def mirror_blend_best_seq(a, rng):
    a_rev = a[::-1].copy()
    blend = 0.5 + 0.3 * rng.uniform(-1, 1)
    a_new = blend * a + (1.0 - blend) * a_rev
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
