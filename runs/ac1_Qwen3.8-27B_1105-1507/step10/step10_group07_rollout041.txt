"""
Coordinate Descent + Smoothed Gradient Optimizer for minimum peak autocorrelation.
Uses exact 1D coordinate minimization with FFT-based convolution evaluation,
combined with smoothed gradient warmup and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
import time
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad_descent(a, time_budget=60.0, tau_init=20.0, tau_end=300.0):
    """Smoothed gradient descent with increasing temperature."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 5.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    it = 0

    N = 1
    while N < 3 * n:
        N <<= 1

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.8)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(np.clip(tau * (conv - max_conv), -50, 50))
        p = exp_vals / float(np.sum(exp_vals))

        P = np.fft.fft(p, N)
        A = np.fft.fft(a, N)
        g_full = np.fft.ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr = max(1e-10, lr * 0.999)
        it += 1

    return best_a


def coordinate_descent(a, time_budget=600.0, seed=42, verbose=False):
    """Coordinate descent with exact 1D minimization per coordinate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    # Precompute for speed
    a_work = a.copy()
    current_sum = s
    
    pass_count = 0
    no_improve_passes = 0
    it = 0

    while time.monotonic() - start < time_budget:
        # Random coordinate order
        order = rng.permutation(n)
        
        pass_improved = False
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            v = a_work[i]
            if v < 1e-15:
                # Try setting to a small positive value
                candidates = [0.0, 1e-6, 1e-5, 5e-5, 1e-4]
            else:
                # Search range around current value
                candidates = np.array([0.0, v*0.3, v*0.6, v*0.8, v*0.9, v, v*1.1, v*1.2, v*1.5, v*2.0, v*3.0])
            
            best_x = v
            best_f = None
            
            for x in candidates:
                if x < 0:
                    x = 0.0
                # Quick evaluation
                old_v = a_work[i]
                a_work[i] = x
                new_sum = current_sum + (x - old_v)
                if new_sum < 1e-10:
                    a_work[i] = old_v
                    continue
                a_norm = a_work / new_sum
                fa = rfft(a_norm, size)
                conv = irfft(fa * fa, size)[:2*n-1]
                f_val = 2.0 * n * float(np.max(conv))
                a_work[i] = old_v
                
                if best_f is None or f_val < best_f - 1e-15:
                    best_f = f_val
                    best_x = x
            
            if best_x != v:
                old_v = a_work[i]
                a_work[i] = best_x
                current_sum += (best_x - old_v)
                # Renormalize
                a_work = a_work / current_sum
                current_sum = 1.0
                pass_improved = True
            
            it += 1
        
        pass_count += 1
        obj_current = compute_obj(a_work)
        if obj_current < best_obj - 1e-12:
            best_obj = obj_current
            best_a = a_work.copy()
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                # Perturb and restart
                scale = 0.005 + 0.003 * no_improve_passes
                a_work = best_a * np.exp(rng.normal(0, scale, n))
                a_work = np.maximum(a_work, 0)
                a_work = a_work / a_work.sum()
                current_sum = 1.0
                no_improve_passes = 0
                if pass_count > 50:
                    break
        
        if verbose and pass_count % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD pass {pass_count}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

- For k in [i, min(i+n-1, 2n-2)]:

The key insight for efficiency: for each coordinate i, the "unaffected max" is constant across all candidate x values. So I compute it once per coordinate, then for each candidate x, I only need to compute max(new_vals) and take the max with the unaffected max.

def coordinate_descent_fast(a, time_budget=600.0, seed=42, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    sum_a = float(a.sum())
    if sum_a < 1e-10:
        return a
    a = a / sum_a
    
    conv = compute_conv(a).copy()
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    pass_count = 0
    no_improve = 0
    a_work = a.copy()
    conv_work = conv.copy()
    sum_work = 1.0
    
    # For efficient max computation, precompute sorted conv indices
    # Actually, let's just use a simple approach:
    # For each coordinate, compute max_unaffected by masking
    
    while time.monotonic() - start < time_budget:
        order = rng.permutation(n)
        improved_count = 0
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            v = a_work[i]
            
            # Affected conv indices
            k_lo = i
            k_hi = min(i + n, 2 * n - 1)
            n_affected = k_hi - k_lo
            
            # j = k - i for each affected k
            j_vals = np.arange(k_lo, k_hi) - i
            is_center = (j_vals == i)
            
            # Other factor: a_work[j] for each affected k
            # For center, the "other" is v itself (same as a_work[i])
            other_factors = a_work[j_vals]  # shape (n_affected,)
            
            # Current conv at affected positions
            conv_aff = conv_work[k_lo:k_hi]  # shape (n_affected,)
            
            # Precompute max of unaffected conv values
            # Use: max_unaffected = max(conv_work) unless the global max is at an affected position
            global_max = float(np.max(conv_work))
            # If the global max is at an affected position, we need the second max
            max_idx = int(np.argmax(conv_work))
            if k_lo <= max_idx < k_hi:
                # Global max is affected, find max of unaffected
                mask = np.ones(2*n-1, dtype=bool)
                mask[k_lo:k_hi] = False
                max_unaffected = float(np.max(conv_work[mask]))
            else:
                max_unaffected = global_max

            if v < 1e-12:
                candidates = [0.0, 1e-6, 2e-6, 5e-6, 1e-5, 5e-5, 1e-4]

                new_sum = sum_work - v + x
                if new_sum < 1e-10:
                    continue

                    # So we need to add an extra (x^2 - v^2) - delta * v = x^2 - v^2 - (x-v)*v = x^2 - v^2 - x*v + v^2 = x^2 - x*v = x*(x-v)

        if current_obj < best_obj - 1e-12:

                    sum_work = 1.0

            print(f"    CD pass {pass_count}: obj={best_obj:.8f}, improved={improved_count}, t={elapsed:.0f}s")
    
    return best_a
