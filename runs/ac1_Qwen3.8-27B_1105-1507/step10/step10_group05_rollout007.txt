import time

def compute_obj(a):

        return np.inf

def solve_lp_fast(a):
    """Solve: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0
    Returns optimal g."""
    n = len(a)
    a_norm = a / a.sum()
    
    # Build constraint matrix: A_ub @ [g; M] <= b_ub
    # conv(a_norm, g)[k] - M <= 0 for k=0..2n-2
    # This is: sum_j A[k,j] * g[j] - M <= 0
    # where A[k,j] = a_norm[k-j] if 0<=k-j<n else 0
    
    # For linprog: c @ x <= d
    # Variables: x = [g_0, ..., g_{n-1}, M]
    # Objective: minimize M => c = [0,...,0, 1]
    # Constraints:
    #   conv_k - M <= 0 => A[k,:] @ g - M <= 0 => [A[k,:], -1] @ x <= 0
    #   sum(g) = 1 => equality
    #   g >= 0 => bounds
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # Inequality constraints: [A, -1] @ x <= 0
    # A is (2n-1) x n
    A_conv = np.zeros((2*n-1, n))
    for j in range(n):
        # a_norm contributes to positions j, j+1, ..., j+n-1 in the convolution
        # conv[k] = sum_j a_norm[k-j] * g[j] (when 0 <= k-j < n)
        # So A[k,j] = a_norm[k-j] if 0 <= k-j < n
        # For fixed j: a_norm[k-j] is nonzero when 0 <= k-j < n, i.e., j <= k < j+n
        start = j
        end = min(2*n-1, j+n)
        A_conv[start:end, j] = a_norm[:end-start]
    
    A_ub = np.hstack([A_conv, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    g = res.x[:n]
    g = np.maximum(g, 0)
    s = g.sum()
    if s < 1e-10:
        return None, None
    g = g / s
    M = res.x[-1]
    return g, M

Let me also think about the line search. The previous approach used a grid of t values. A better approach:
- The objective as a function of t is: obj((1-t)*a + t*g) = 2n * max_k conv((1-t)*a_t + t*g, (1-t)*a_t + t*g)
  where a_t = (1-t)*a + t*g / sum((1-t)*a + t*g) (but since both sum to 1, no rescaling needed)
  
  = 2n * max_k [(1-t)^2 * conv_a[k] + 2t(1-t) * cross[k] + t^2 * conv_g[k]]
  
  where cross[k] = conv(a, g)[k]

This is a max of quadratic functions in t. It's convex in t (max of convex functions). So we can use ternary search or golden section.

Actually, since it's convex, the minimum is where the derivative changes sign. The derivative is:
d/dt obj(t) = 2n * max_k d/dt [quadratic_k(t)] (but the max makes it nonsmooth)

For a smooth approximation, we can use the active set (positions where the quadratic is near the max) and do a quadratic minimization.

"""
Optimizes step function coefficients using accelerated LP iteration with
scipy HiGHS solver, FFT-based mirror descent fine-tuning, and multi-length search.
Key innovations: direct LP formulation (no cvxpy overhead), Nesterov momentum,
adaptive line search, and structured initialization exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix
    A_conv = build_conv_matrix(a_norm)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A_conv, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [('free', 'free')]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0 or res.x is None:
            return None, None
        g = np.maximum(res.x[:n], 0)
        s = g.sum()
        if s < 1e-10:
            return None, None
        g = g / s
        M = float(res.x[-1])
        return g, M
    except Exception:
        return None, None


def lp_accelerated(a, time_budget=100.0, verbose=False, seed=42):
    """
    Accelerated LP iteration with Nesterov momentum and adaptive step.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9  # momentum
    v = np.zeros(n)
    prev_g = None
    it = 0
    stuck_count = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 2:
            break
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-10:
            y = a.copy()
        else:
            y = y / sy
        
        # Solve LP
        g, M = solve_lp_minpeak(y)
        if g is None:
            # Fallback: take the current best
            break
        
        # Compute direction
        d = g - a
        
        # Line search: try different step sizes
        best_t = 1.0
        best_t_obj = compute_obj(a)  # obj at t=0
        
        # Try t = 1.0 first
        a_try = a + 1.0 * d
        a_try = np.maximum(a_try, 0)
        st = a_try.sum()
        if st > 1e-10:
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = 1.0
                best_t_a = a_try.copy()
        
        # If t=1 didn't help, try smaller steps
        if best_t < 1.0:
            for t_c in [0.5, 0.3, 0.1]:
                a_try = a + t_c * d
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st > 1e-10:
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_t_obj:
                        best_t_obj = obj_try
                        best_t = t_c
                        best_t_a = a_try.copy()
        
        # If still no improvement, try extrapolation
        if best_t <= 1.0 and best_t_obj >= best_obj - 1e-14:
            for t_c in [1.5, 2.0]:
                a_try = a + t_c * d
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st > 1e-10:
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_t_obj:
                        best_t_obj = obj_try
                        best_t = t_c
                        best_t_a = a_try.copy()
        
        # Update
        if best_t_obj < best_obj - 1e-14:
            # Nesterov momentum update
            v = beta * v + (g - (a + best_t * d))  # hmm, this isn't quite right
            # Actually, standard Nesterov:
            # x_{k+1} = x_k + d_k
            # v_{k+1} = beta * v_k
            # x_{k+1} is determined by x_k + v_{k+1} + d_k
            # Let me use a simpler formulation:
            a_new = a + best_t * d
            v = beta * v  # decay momentum
            a = a_new
            best_obj = best_t_obj
            best_a = a.copy()
            stuck_count = 0
        else:
            # No improvement, reset momentum and try perturbation
            stuck_count += 1
            if stuck_count > 20:
                # Perturb
                rng = np.random.default_rng(seed + it)
                a = best_a + 0.01 * rng.normal(0, 1, n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                v = np.zeros(n)
                stuck_count = 0
                prev_g = None
            a = best_a.copy()
            v = v * 0.5
        
        prev_g = g.copy()
        it += 1
        
        if verbose and it % 50 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, M={M:.6f}, t={elapsed:.1f}s")
    
    return best_a


def fft_mirror_descent(a, time_budget=60.0, tau_init=50.0, tau_end=3000.0, eta_init=0.5, verbose=False):
    """
    FFT-based multiplicative weights / mirror descent for the smoothed max-conv objective.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    fft_size = 1
    while fft_size < 2 * n:
        fft_size <<= 1
    
    it = 0
    stuck = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 1:
            break
        
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * (1 - np.cos(np.pi * progress)) / 2
        eta = eta_init / (1.0 + 0.001 * it)
        
        # Compute conv(a, a)
        fa = rfft(a, fft_size)
        conv = irfft(fa * fa, fft_size)[:2 * n - 1]
        
        # Softmax weights
        max_c = conv.max()
        p = np.exp(tau * (conv - max_c))
        ps = p.sum()
        if ps < 1e-300:
            p = np.zeros(2 * n - 1)
            p[np.argmax(conv)] = 1.0
        else:
            p = p / ps
        
        # Compute smoothed gradient via FFT
        # grad[j] = 2 * sum_{i} a[i] * p[i+j]
        # = 2 * (B * p)[n-1+j] where B = a[::-1]
        B = a[::-1]
        fb = rfft(B, fft_size)
        fp = rfft(p, fft_size)
        conv_bp = irfft(fb * fp, fft_size)
        grad = 2.0 * conv_bp[n - 1:2 * n - 1]
        
        # Clip gradient to avoid extreme updates
        grad_max = np.abs(grad).max()
        if grad_max > 0:
            grad = grad / grad_max  # normalize to [-1, 1]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 100:
                # Reset with perturbation
                rng = np.random.default_rng(it)
                a = best_a * np.exp(0.02 * rng.normal(0, 1, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stuck = 0
        
        it += 1
        
        if verbose and it % 500 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.1f}s")
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def structured_init(n, kind='uniform', rng=None):
    """Create structured initial sequences."""
    x = np.linspace(0, 1, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'raised_cosine':
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif kind == 'gaussian':
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif kind == 'triangular':
        a = 1 - 2 * np.abs(x - 0.5)
    elif kind == 'bump':
        a = (1 - (2 * x - 1) ** 2) ** 2
    elif kind == 'sinc_like':
        t = 8 * (x - 0.5)
        a = np.sinc(t / (2 * np.pi))  # sin(pi*t)/(pi*t)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s
