"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key insight: asymmetric sequences can beat the symmetric lower bound of 2.
Uses tapered asymmetric initializations, LP fixed-point polishing,
Nesterov subgradient method, and multi-n exploration.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_step(a, a_coeff):
    """
    Compute subgradient of max(conv(a,a)) at point a (normalized sum=1).
    Returns the subgradient vector of length n.
    a_coeff: the normalized sequence (sum=1)
    """
    n = len(a_coeff)
    conv = compute_conv(a_coeff)
    max_val = float(np.max(conv))
    # Find all argmax indices
    tol = 1e-10
    I = np.where(np.abs(conv - max_val) < tol)[0]
    if len(I) == 0:
        I = np.array([np.argmax(conv)])
    
    # Subgradient: average of gradients at argmax points
    # grad[m] = (1/|I|) * sum_{k in I} [a[k-m] if valid + a[m-k] if valid]
    g = np.zeros(n)
    for k in I:
        for m in range(n):
            # a[k-m] valid if 0 <= k-m < n
            if 0 <= k - m < n:
                g[m] += a_coeff[k - m]
            # a[m-k] valid if 0 <= m-k < n  
            if 0 <= m - k < n:
                g[m] += a_coeff[m - k]
    g /= len(I)
    return g, max_val


def subgradient_step_fast(a_coeff):
    """
    Fast subgradient computation using FFT.
    The subgradient at argmax index k is: (a * indicator_k) + (flipped_a * indicator_k)
    where * is convolution.
    """
    n = len(a_coeff)
    conv = compute_conv(a_coeff)
    max_val = float(np.max(conv))
    tol = max(1e-10, 1e-8 * max_val)
    I = np.where(np.abs(conv - max_val) < tol)[0]
    if len(I) == 0:
        I = np.array([np.argmax(conv)])
    
    # For each k in I, the gradient w.r.t. a[m] is:
    # a[k-m] (if 0<=k-m<n) + a[m-k] (if 0<=m-k<n)
    # This is: (a conv e_k)[m] + (a_rev conv e_k)[m] where a_rev[n-1-m] = a[m]
    # Actually it's the cross-correlation: grad[m] = sum_k_in_I (a[k-m]_valid + a[m-k]_valid)
    
    # Use FFT for efficiency
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    # The gradient at a single k is: ifft(fft(a) * fft(e_k)) + ifft(fft(a) * fft(e_{-k}))
    # where e_k is a delta at position k
    # fft(e_k) = exp(-2*pi*i*k/size)
    A = fft(a_coeff, size)
    
    g = np.zeros(n)
    for k in I:
        # Contribution from a[k-m]: this is ifft(A * exp(-2pi i k / size))
        phase1 = np.exp(-2j * np.pi * k / size)
        g1 = ifft(A * phase1).real[:n]
        # Contribution from a[m-k]: this is ifft(A * exp(2pi i k / size))
        phase2 = np.exp(2j * np.pi * k / size)
        g2 = ifft(A * phase2).real[:n]
        g += g1 + g2
    g /= len(I)
    return g, max_val


def nesterov_subgrad(a, time_budget, verbose=False, lr0=None, max_iters=100000):
    """
    Nesterov accelerated subgradient method for minimizing max(conv(a,a))
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if lr0 is None:
        lr0 = 1.0 / np.sqrt(n)
    
    lr = lr0
    beta = 0.5
    v = np.zeros(n)
    it = 0
    best_it = 0
    patience = 5000
    
    while time.monotonic() - start < time_budget:
        if it >= max_iters:
            break
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
            sy = 1.0
        y = y / sy
        
        # Subgradient at y
        g, _ = subgradient_step_fast(y)
        
        # Update
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            lr *= 0.5
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        
        # Decay learning rate
        lr = lr0 / np.sqrt(it + 1)
        beta = (1 + it) / (2 + it)  # Nesterov schedule
        
        # Reset if stuck
        if it - best_it > patience:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it
            patience = int(patience * 0.7)
            if patience < 100:
                break
    
    return best_a


def lp_polish_iteration(a, solver_list=None):
    """
    Single LP step: find g >= 0, sum(g)=1 minimizing max(conv(a,g)).
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    if n > 3000:
        return None, None
    
    a = a / np.sum(a)
    
    # Build convolution matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0
    # This is a Hankel matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    
    return None, None


def lp_fixed_point(a, time_budget, verbose=False, max_iters=100, adaptive_t=True):
    """

            break

        else:

        best_t_obj = np.inf

        best_a_local = best_a.copy()
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp

            a_try = a_try / s_try

                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose:
                print(f"    LP it{it}: obj={best_obj:.10f}, t={best_t_val:.3f}, M={M_lp:.6f}")
        else:

            if stall > 8:

def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def staircase_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.85, lr_scale=2.0):

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()

        return a

        tau_levels = [5, 20, 50, 150, 400, 1000, 3000]
    
    it = 0
    best_it = 0
    stall_patience = 3000
    
    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0
        max_iters_per_level = 3000
        
        while time.monotonic() - start < time_budget:
            if tau_iters >= max_iters_per_level:
                break
            
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy
            
            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break
            
            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))

                best_a = a_new.copy()
                best_it = it
            
            a = a_new
            it += 1
            tau_iters += 1
            lr = max(1e-7, lr * 0.99997)
            
            if it - best_it > stall_patience:
                a = best_a.copy()
                v = np.zeros(n)

def make_tapered_init(n, mode, rng=None):
    """Create tapered asymmetric initial sequences (small at boundaries)."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 1, n)
    
    if mode == 0:
        # Uniform
        return np.ones(n) / n
    elif mode == 1:
        # Triangular taper (symmetric)
        a = 1.0 - 2.0 * np.abs(x - 0.5)
        return np.maximum(a, 0.01) / np.maximum(a, 0.01).sum()
    elif mode == 2:
        # Asymmetric exponential taper (small left, large right)
        a = np.exp(2.0 * x)
        return a / a.sum()
    elif mode == 3:
        # Asymmetric: small both ends, peak off-center
        c = 0.4
        a = np.exp(-4.0 * (x - c) ** 2)
        return a / a.sum()
    elif mode == 4:
        # Cosine taper with asymmetry
        a = np.maximum(np.cos(np.pi * (x - 0.3)), 0.01) ** 2
        return a / a.sum()
    elif mode == 5:
        # Multi-bump asymmetric
        a = np.exp(-3.0 * (x - 0.25) ** 2) + 0.7 * np.exp(-3.0 * (x - 0.6) ** 2)
        return a / a.sum()
    elif mode == 6:
        # Random with taper
        a = rng.uniform(0.1, 1.0, n)
        taper = np.sin(np.pi * x)  # zero at both ends
        a = a * (0.2 + 0.8 * taper)
        return a / a.sum()
    elif mode == 7:
        # Power law taper
        a = x ** 0.5 * (1 - x) ** 0.3
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 8:
        # Comb with taper
        a = 0.3 + 0.7 * np.abs(np.sin(4 * np.pi * x))
        taper = np.sin(np.pi * x)
        a = a * (0.1 + 0.9 * taper)
        return a / a.sum()
    elif mode == 9:
        # Inverse cosine
        a = 1.0 / (1.0 + 5.0 * x * (1 - x))
        return a / a.sum()
    elif mode == 10:
        # Asymmetric Gaussian mix
        a = 0.6 * np.exp(-6.0 * (x - 0.3) ** 2) + 0.4 * np.exp(-4.0 * (x - 0.7) ** 2)
        return a / a.sum()
    elif mode == 11:
        # Sawtooth with taper
        a = (x % 0.2) / 0.2
        taper = np.sin(np.pi * x)
        a = a * (0.05 + 0.95 * taper)
        return a / a.sum()
    elif mode == 12:
        # Step function with taper
        a = np.where(x < 0.5, 1.5, 0.5)
        taper = np.sin(np.pi * x)
        a = a * (0.1 + 0.9 * taper)
        return a / a.sum()
    elif mode == 13:
        # Very concentrated (small support)
        support = int(n * 0.3)
        start = int(n * 0.1)
        a = np.zeros(n)
        a[start:start + support] = 1.0
        taper = np.sin(np.pi * x)
        a = a * (0.01 + 0.99 * taper)
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def perturb_fourier(a, rng, strength=0.05):
    """Perturb a using low-frequency Fourier modes (preserves asymmetry)."""
    n = len(a)
    freqs = rng.integers(1, 8, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.5, 1.0, size=4)
    x = np.arange(n)
    pert = np.zeros(n)
    for f, p, amp in zip(freqs, phases, amps):
        pert += amp * np.sin(2 * np.pi * f * x / n + p)
    a_new = a + strength * pert
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  GD {name}: obj={best_score:.10f} **")
        else:
            print(f"  GD {name}: obj={obj2:.10f}")
    
    # === Phase 3: Nesterov subgradient method ===
    print(f"\n=== Phase 3: Nesterov Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.12, 120.0)
        a_sub = nesterov_subgrad(best_seq, time_budget=alloc, verbose=False)
        obj_sub = compute_obj(a_sub)
        if obj_sub < best_score:
            best_score = obj_sub
            best_seq = a_sub.copy()
            print(f"  Subgrad: obj={best_score:.10f} **")
        else:
            print(f"  Subgrad: obj={obj_sub:.10f}")
    
    # === Phase 4: LP Fixed-Point Polishing (main workhorse) ===
    print(f"\n=== Phase 4: LP Polishing ===")
    remaining = deadline - time.monotonic()
    
    if n_base <= 2500:
        num_lp_rounds = 6
        for rnd in range(num_lp_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(remaining * 0.15, 100.0)
            if alloc < 15:
                break
            
            # Add perturbation every other round to escape local minima
            if rnd > 0 and rnd % 2 == 0:
                a_start = perturb_fourier(best_seq, rng, strength=0.03)
            else:
                a_start = best_seq.copy()
            
            a_lp = lp_fixed_point(a_start, time_budget=alloc, verbose=(rnd == 0),
                                   max_iters=30)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP rnd{rnd}: obj={best_score:.10f} **")
            else:
                print(f"  LP rnd{rnd}: obj={obj_lp:.10f}")
    
    # === Phase 5: GD after LP ===
    print(f"\n=== Phase 5: Post-LP GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.10, 80.0)
        a5 = staircase_gd(best_seq, time_budget=alloc,
                           tau_levels=[50, 200, 800, 2000], verbose=False)
        obj5 = compute_obj(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Post-LP GD: obj={best_score:.10f} **")
        else:
            print(f"  Post-LP GD: obj={obj5:.10f}")
    
    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        n_base, max(400, n_base - 100), n_base + 100, n_base + 200,
        n_base + 400, 600, 800, 1000, 1200, 1500, 1800, 2000
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2500]
    # Prioritize: close to current, then varied
    n_options.sort(key=lambda x: (abs(x - n_base), x))
    n_options = list(dict.fromkeys(n_options))[:5]  # unique, preserve order
    
    print(f"  n candidates: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try == n_base:
            continue
        
        a_try = resample(best_seq, n_try)
        
        # Quick GD
        alloc1 = min(remaining * 0.15, 40.0)
        if alloc1 > 10:
            a_try = staircase_gd(a_try, time_budget=alloc1,
                                  tau_levels=[20, 100, 400], verbose=False)
        
        # LP if reasonable size
        remaining = deadline - time.monotonic()
        if n_try <= 2000 and remaining > 25:
            alloc2 = min(remaining * 0.10, 35.0)
            if alloc2 > 10:
                a_try = lp_fixed_point(a_try, time_budget=alloc2, verbose=False, max_iters=15)
        
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")
    
    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    
    # LP polish
    if remaining > 30 and n_final <= 2500:
        alloc = min(remaining * 0.20, 80.0)
        a_f1 = lp_fixed_point(best_seq, time_budget=alloc, verbose=True, max_iters=30)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final LP: obj={best_score:.10f} **")
        else:
            print(f"  Final LP: obj={obj_f1:.10f}")
    
    # High-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.20, 60.0)
        a_f2 = staircase_gd(best_seq, time_budget=alloc,
                             tau_levels=[200, 800, 2000, 5000], verbose=False)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final GD: obj={best_score:.10f} **")
        else:
            print(f"  Final GD: obj={obj_f2:.10f}")
    
    # One more LP round
    remaining = deadline - time.monotonic()
    if remaining > 20 and n_final <= 2500:
        alloc = min(remaining * 0.25, 50.0)
        a_f3 = lp_fixed_point(best_seq, time_budget=alloc, verbose=True, max_iters=20)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Ultra LP: obj={best_score:.10f} **")
        else:
            print(f"  Ultra LP: obj={obj_f3:.10f}")
    
    # === Phase 8: Perturbation + quick re-optimization ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 8: Perturbation Search ===")
        n_pert = 5
        for pi in range(n_pert):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_p = perturb_fourier(best_seq, rng, strength=0.02 * (pi + 1))
            # Quick optimize
            alloc = min(remaining * 0.15, 30.0)
            if alloc > 8:
                if len(a_p) <= 2000:
                    a_p = lp_fixed_point(a_p, time_budget=alloc * 0.6, verbose=False, max_iters=10)
                    remaining = deadline - time.monotonic()
                    if remaining > 8:
                        a_p = staircase_gd(a_p, time_budget=alloc * 0.4,
                                            tau_levels=[100, 500], verbose=False)
                else:
                    a_p = staircase_gd(a_p, time_budget=alloc,
                                        tau_levels=[50, 200], verbose=False)
            obj_p = compute_obj(a_p)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Pert{pi}: obj={best_score:.10f} **")
            else:
                print(f"  Pert{pi}: obj={obj_p:.10f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    
    return result
