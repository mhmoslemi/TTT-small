Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based smooth gradient descent with simplex projection,
multi-start with diverse initializations, and aggressive tau scheduling.
Key innovation: avoids LP solvers entirely, using only FFT-based operations.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
