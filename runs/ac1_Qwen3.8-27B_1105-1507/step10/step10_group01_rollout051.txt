import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_subset(a, lags):
    """Build convolution matrix for a subset of lags only."""
    n = len(a)
    A = np.zeros((len(lags), n))
    for i, k in enumerate(lags):
        for j in range(n):
            if 0 <= k - j < n:
                A[i, j] = a[k - j]
    return A


def build_conv_matrix_vec(a, lags=None):
    """Vectorized build of convolution matrix for given lags."""
    n = len(a)
    if lags is None:
        lags = np.arange(2 * n - 1)
    else:
        lags = np.asarray(lags)
    
    k_idx = lags[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    
    A = np.zeros((len(lags), n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_adaptive(a, n_constraints=None, verbose=False):
    """
    Solve LP with adaptive constraint selection.
    Only constrain lags near the maximum.
    Returns (g, M, lags_used).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv = compute_conv(a_norm)
    max_val = float(np.max(conv))
    
    # Select top lags (those above some fraction of max)
    if n_constraints is None:
        n_constraints = max(10, min(2 * n - 1, n // 2))
    
    # Use lags with highest convolution values
    top_indices = np.argsort(conv)[-n_constraints:]
    top_indices = np.sort(top_indices)
    
    A = build_conv_matrix_vec(a_norm, top_indices)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value), top_indices
        except Exception:
            continue
    return None, None, top_indices


def line_search_fast(a, g, n_evals=20):
    """Fast line search with parabolic refinement."""
    n = len(a)
    ts = np.linspace(0.01, 0.9, n_evals)
    objs = np.full(n_evals, np.inf)
    
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try)
    
    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]
    
    if 0 < best_idx < n_evals - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.01, 0.9))
            a_par = (1 - t_par) * a + t_par * g
            a_par = np.maximum(a_par, 0)
            if float(np.sum(a_par)) > 1e-10:
                f_par = compute_obj(a_par)
                if f_par < best_obj:
                    best_t, best_obj = t_par, f_par
    
    return best_t, best_obj

def gd_polish_fast(a, time_budget=50.0, tau_init=100.0, tau_end=2000.0):

    prev_g = None
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        conv = compute_conv(y)

        exp_v = np.exp(tau * (conv - max_c))
        sum_e = float(np.sum(exp_v))
        if sum_e < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = exp_v / sum_e
        
        N_fft = 1
        while N_fft < 3 * n:
            N_fft <<= 1
        P = fft(p, N_fft)
        Af = fft(y, N_fft)
        g = 2.0 * ifft(P * np.conj(Af)).real[:n]
        gnorm = float(np.linalg.norm(g))
        
        if gnorm > 1e-15:
            lr = 0.8 * lr + 0.2 * (base_lr / (gnorm / base_lr + 1))
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)
        
        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def fourier_perturb(a, rng, strength=0.05):
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.2) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_vec(a, lags):
    """Vectorized build of convolution matrix for given lags."""
    n = len(a)
    lags = np.asarray(lags)
    k_idx = lags[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((len(lags), n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_adaptive(a, n_constraints=None):

    g = cp.Variable(n, nonneg=True)

    best_it = 0

        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5

        y = np.maximum(y, 0)

        if sum_e < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = exp_v / sum_e

        N_fft = 1
        while N_fft < 3 * n:
            N_fft <<= 1
        P = fft(p, N_fft)
        Af = fft(y, N_fft)
        g = 2.0 * ifft(P * np.conj(Af)).real[:n]
        gnorm = float(np.linalg.norm(g))

        if gnorm > 1e-15:
            lr = 0.85 * lr + 0.15 * (1.0 / (gnorm / base_lr + 0.5))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def fourier_perturb(a, rng, strength=0.04):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 7):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.4, 1.3) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def localize_update(a, conv, step=0.02):
    """
    Targeted local update: reduce positions that contribute most to peak lags.
    This is a fast heuristic that can make small improvements.
    """
    n = len(a)
    max_c = float(np.max(conv))
    # Find the top lag
    peak_lag = int(np.argmax(conv))

    # Compute contribution of each position to the peak lag
    contrib = np.zeros(n)
    for j in range(n):
        c = 0.0
        if 0 <= j - peak_lag < n:
            c += a[j - peak_lag]
        if 0 <= j + peak_lag < n:
            c += a[j + peak_lag]
        contrib[j] = c

    # Reduce high-contribution positions, increase low-contribution
    c_mean = float(np.mean(contrib))
    c_std = float(np.std(contrib)) + 1e-10

    a_new = a.copy()
    for j in range(n):
        if contrib[j] > c_mean + 0.5 * c_std:
            a_new[j] *= (1 - step)
        elif contrib[j] < c_mean - 0.5 * c_std:
            a_new[j] *= (1 + step)

    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a.copy()
    return a_new / s_new

        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_cur = len(best_seq)

    # === Phase 1: Mirror Descent (fast exploration) ===
    print(f"\n=== Phase 1: Mirror Descent (n={n_cur}) ===")
    remaining = deadline - time.monotonic()
    p1 = min(remaining * 0.15, 150.0)
    if p1 > 20:
        a1 = mirror_descent(best_seq, time_budget=p1, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score, best_seq = obj1, a1.copy()
            print(f"  MD done: obj={best_score:.8f} **")
        else:
            print(f"  MD: obj={obj1:.8f} (no improvement)")

    # === Phase 2: Adaptive LP ===
    print(f"\n=== Phase 2: Adaptive LP (n={n_cur}) ===")
    remaining = deadline - time.monotonic()
    p2 = min(remaining * 0.25, 300.0)
    if p2 > 30:
        nc = max(30, n_cur // 3)
        a2 = lp_optimize(best_seq, time_budget=p2, n_constraints=nc, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score, best_seq = obj2, a2.copy()
            print(f"  LP done: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj2:.8f} (no improvement)")

    # === Phase 3: GD Polish ===
    print(f"\n=== Phase 3: GD Polish ===")
    remaining = deadline - time.monotonic()
    p3 = min(remaining * 0.08, 60.0)
    if p3 > 10:
        a3 = gd_polish(best_seq, time_budget=p3, tau_init=50.0, tau_end=1500.0)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score, best_seq = obj3, a3.copy()
            print(f"  GD: obj={best_score:.8f} **")

    # === Phase 4: Localize + MD cycle ===
    print(f"\n=== Phase 4: Localize+MD cycle ===")
    remaining = deadline - time.monotonic()
    p4 = min(remaining * 0.12, 120.0)
    if p4 > 20:
        a4 = best_seq.copy()
        a4_iter_start = time.monotonic()
        best_a4 = a4.copy()
        best_obj4 = compute_obj(a4)
        it4 = 0
        while time.monotonic() - a4_iter_start < p4:
            conv4 = compute_conv(a4)
            a4 = localize_update(a4, conv4, step=0.015)
            # Short MD burst
            a4 = mirror_descent(a4, time_budget=min(5.0, p4 * 0.05), verbose=False)
            obj4 = compute_obj(a4)
            it4 += 1
            if obj4 < best_obj4 - 1e-13:
                best_obj4 = obj4
                best_a4 = a4.copy()
            if it4 % 20 == 0:
                print(f"    L+MD it{it4}: obj={best_obj4:.8f}")
        if best_obj4 < best_score:
            best_score, best_seq = best_obj4, best_a4.copy()
            print(f"  Phase4: obj={best_score:.8f} **")
        else:
            print(f"  Phase4: obj={best_obj4:.8f} (no improvement)")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_now = len(best_seq)
    n_options = sorted(set([
        n_now + 200, n_now + 500, n_now + 1000,
        1200, 1600, 2000, 2500, 3000,
        n_now - 200, n_now - 400, 600
    ]))
    n_options = [n for n in n_options if 400 <= n <= 3500 and abs(n - n_now) > 50]
    n_options = n_options[:5]
    print(f"  Trying n: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample(best_seq, n_try)

        # MD for fast optimization
        alloc_md = min(remaining * 0.2, 50.0)
        if alloc_md > 10:
            a_try = mirror_descent(a_try, time_budget=alloc_md, verbose=False)

        # LP
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.2, 60.0)
        if alloc_lp > 15:
            nc_try = max(50, n_try // 4)
            a_try = lp_optimize(a_try, time_budget=alloc_lp, n_constraints=nc_try, verbose=False)

        # GD
        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.08, 25.0)
        if alloc_gd > 8:
            a_try = gd_polish(a_try, time_budget=alloc_gd, tau_init=50.0, tau_end=800.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score, best_seq = obj_try, a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    n_restarts = 3
    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if r == 0:
            a_r = fourier_perturb(best_seq, rng, strength=0.06)
        elif r == 1:
            # Mirror + blend
            a_mirror = best_seq[::-1].copy()
            a_r = 0.6 * best_seq + 0.4 * a_mirror
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        else:
            # Random blend
            a_r = 0.8 * best_seq + 0.2 * (rng.uniform(0, 1, len(best_seq)) / len(best_seq))
            a_r = a_r / a_r.sum()

        # Optimize with MD + LP
        alloc_md = min(remaining * 0.2, 40.0)
        if alloc_md > 10:
            a_r = mirror_descent(a_r, time_budget=alloc_md, verbose=False)

        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.15, 50.0)
        if alloc_lp > 15:
            nc_r = max(30, len(a_r) // 3)
            a_r = lp_optimize(a_r, time_budget=alloc_lp, n_constraints=nc_r, verbose=False)

        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.06, 20.0)
        if alloc_gd > 5:
            a_r = gd_polish(a_r, time_budget=alloc_gd, tau_init=100.0, tau_end=1000.0)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score, best_seq = obj_r, a_r.copy()
            print(f"  Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    n_final = len(best_seq)

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 200.0)
        nc_fin = max(50, n_final // 3)
        a_fin = lp_optimize(best_seq, time_budget=alloc, n_constraints=nc_fin, verbose=True)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score, best_seq = obj_fin, a_fin.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final MD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 80.0)
        a_fin2 = mirror_descent(best_seq, time_budget=alloc, verbose=True)
        obj_fin2 = compute_obj(a_fin2)
        if obj_fin2 < best_score:
            best_score, best_seq = obj_fin2, a_fin2.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # Final GD with high tau
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.5, 60.0)
        a_fin3 = gd_polish(best_seq, time_budget=alloc, tau_init=300.0, tau_end=5000.0)
        obj_fin3 = compute_obj(a_fin3)
        if obj_fin3 < best_score:
            best_score, best_seq = obj_fin3, a_fin3.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Ensure sum constraint
    if sum(result) < 0.01:
        result = [1.0] * 100
        best_score = 2.0

    val = evaluate_sequence(result)
    if val != np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
