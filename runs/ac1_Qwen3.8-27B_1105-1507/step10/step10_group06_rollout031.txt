Uses fast coordinate descent with incremental convolution updates as the primary

import cvxpy as cp

    if n == 0:
        return np.array([])

    while size < 2 * n - 1:
        size <<= 1

    if n == 0:

        quad_val = b_quad + x * x

            M = quad_val
    if M < 0:
        M = 0.0
    denom = S_other + x
    if denom < 1e-15:
        return 1e30
    return M / (denom * denom)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast coordinate descent with incremental convolution updates as the primary
method, combined with LP-based best-response alternation for global moves,
and multi-length exploration. Coordinate descent exploits the piecewise structure
of the 1D subproblem (linear + one quadratic term) for optimal per-coordinate updates.
"""

import numpy as np
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def eval_1d(const_max, b, d, b_quad, S_other, x):
    """Evaluate M(x)/(S+x)^2 for the coordinate descent subproblem."""
    linear_max = float(np.max(b + d * x))
    M = max(const_max, linear_max)
    if b_quad > -1e30:
        qv = b_quad + x * x
        if qv > M:
            M = qv
    if M < 0:
        M = 0.0
    denom = S_other + x
    if denom < 1e-15:
        return 1e30
    return M / (denom * denom)


def find_optimal_coord(a, conv, i, n):
    """Find optimal value for coordinate i, given all others fixed."""
    ai = a[i]
    S = float(np.sum(a))
    S_other = S - ai

    if S_other < 1e-15:
        return ai

    k_min = max(0, i)
    k_max = min(2 * n - 2, i + n - 1)

    # Unaffected max
    const_max = 0.0
    if k_min > 0:
        v = float(np.max(conv[:k_min]))
        if v > const_max:
            const_max = v
    if k_max < 2 * n - 2:
        v = float(np.max(conv[k_max + 1:]))
        if v > const_max:
            const_max = v

    # Build b and d for affected range
    length = k_max - k_min + 1
    b = np.zeros(length)
    d = np.zeros(length)

    for idx in range(length):
        k = k_min + idx
        ki = k - i
        if k == 2 * i:
            b[idx] = conv[k] - ai * ai
        else:
            b[idx] = conv[k] - 2.0 * ai * a[ki]
            d[idx] = 2.0 * a[ki]

    # Find quadratic term index
    b_quad = -1e30
    if 2 * i >= k_min and 2 * i <= k_max:
        idx_q = 2 * i - k_min
        b_quad = b[idx_q]

    # Generate candidates
    candidates = [0.0, ai]

    for idx in range(length):
        if d[idx] > 1e-15:
            xc = S_other - 2.0 * b[idx] / d[idx]
            if 0.0 <= xc < 1e6:
                candidates.append(xc)

    if b_quad > -1e30 and S_other > 1e-15:
        xc = b_quad / S_other
        if 0.0 <= xc < 1e6:
            candidates.append(xc)

    if S_other > 0:
        candidates.append(0.5 * S_other)
        candidates.append(S_other)
        candidates.append(2.0 * S_other)
        if ai > 0:
            candidates.append(ai * 0.5)
            candidates.append(ai * 2.0)

    # Evaluate all candidates
    best_x = ai
    best_h = eval_1d(const_max, b, d, b_quad, S_other, ai)
    cap = 20.0 * S_other + 1.0

    for xc in candidates:
        if xc < 0:
            xc = 0.0
        if xc > cap:
            xc = cap
        hc = eval_1d(const_max, b, d, b_quad, S_other, xc)
        if hc < best_h:
            best_h = hc
            best_x = xc

    # Local refinement: try points around best_x
    spread = max(0.05 * ai, 0.02 * S_other, 1e-8)
    for scale in [1.0, 0.5, 2.0, 0.25, 4.0]:
        for sgn in [-1.0, 1.0]:
            xt = best_x + sgn * scale * spread
            if xt < 0:
                continue
            if xt > cap:
                continue
            ht = eval_1d(const_max, b, d, b_quad, S_other, xt)
            if ht < best_h - 1e-15:
                best_h = ht
                best_x = xt

    return best_x


def coord_descent(a, n_sweeps, verbose=False, reconv_every=20):
    """Coordinate descent with incremental conv updates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = compute_conv(a)
    S = float(np.sum(a))

    best_obj = 2.0 * n * float(np.max(conv)) / (S * S)
    best_a = a.copy()
    best_S = S

    for sweep in range(n_sweeps):
        for i in range(n):
            new_val = find_optimal_coord(a, conv, i, n)
            if abs(new_val - a[i]) > 1e-15:
                ai = a[i]
                diff = new_val - ai
                k_min = max(0, i)
                k_max = min(2 * n - 2, i + n - 1)

                # Vectorized update for linear terms
                k_range = np.arange(k_min, k_max + 1)
                mask = k_range != 2 * i
                conv[k_range[mask]] += diff * 2.0 * a[k_range[mask] - i]
                # Quadratic term
                conv[2 * i] = conv[2 * i] - ai * ai + new_val * new_val

                a[i] = new_val
                S += diff

        # Periodic reconv to fix drift
        if (sweep + 1) % reconv_every == 0:
            conv = compute_conv(a)
            S = float(np.sum(a))

        obj = 2.0 * n * float(np.max(conv)) / (S * S)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            best_S = S

        if verbose and (sweep < 10 or sweep % 20 == 0):
            print(f"    CD sweep {sweep+1}/{n_sweeps}: obj={obj:.10f}, best={best_obj:.10f}")

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a, time_budget, seed=42, verbose=True):
    """LP-based alternating minimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    t_blend = 0.5
    stall_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = solve_conv_lp(a)
        if g_lp is None:
            break

        iter_count += 1

        candidates_t = [t_blend, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if iter_count > 100:
            candidates_t.extend([0.05, 0.02, 0.01])

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend

        for t in candidates_t:
            t = min(max(t, 0.001), 1.0)
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            t_blend = max(0.01, best_local_t * 0.98)
            stall_count = 0
            if verbose and iter_count % 20 == 0:
                print(f"    LP iter {iter_count}: obj={best_obj:.10f}, t={t_blend:.4f}")
        else:
            stall_count += 1
            if stall_count >= 5:
                # Perturb to escape
                perturb = 0.01 * rng.standard_normal(n)
                a = best_a + perturb * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()
                t_blend = min(0.8, t_blend * 1.3)
                stall_count = 0

    if verbose:
        print(f"    LP alternate: {iter_count} iters, best_obj={best_obj:.10f}")
    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_to_length(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    n_options = [n for n in n_options if 300 <= n <= 2000 and n != n_best]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        a_try = resample_to_length(best_a, n_try)
        alloc = min(remaining * 0.25, 100.0)
        if alloc < 20:
            break

        t0 = time.monotonic()
        _, _ = coord_descent(a_try, n_sweeps=2, verbose=False)
        tps = (time.monotonic() - t0) / 2.0
        n_sw = max(5, int(alloc / max(tps, 0.001)))
        n_sw = min(n_sw, 2000)
        a_try, obj_try = coord_descent(a_try, n_sweeps=n_sw, verbose=False)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: obj={best_obj:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # Phase 6: Final polish
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        t0 = time.monotonic()
        _, _ = coord_descent(best_a, n_sweeps=2, verbose=False)
        tps = (time.monotonic() - t0) / 2.0
        n_sw = max(5, int(remaining * 0.7 / max(tps, 0.001)))
        n_sw = min(n_sw, 5000)
        a_final, obj_final = coord_descent(best_a, n_sweeps=n_sw, verbose=False)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
        print(f"  Final: obj={best_obj:.10f}")

    # Validation
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
