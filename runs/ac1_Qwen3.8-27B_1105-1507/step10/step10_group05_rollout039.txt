def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)

    """One step of smoothed gradient descent."""

            best_t = t

    # Fine refine
    t_lo = max(0.0, best_t - 0.025)
    t_hi = best_t + 0.025
    for t in np.linspace(t_lo, t_hi, 21):
        vals = (1.0 - t) ** 2 * ca + 2.0 * t * (1.0 - t) * cag + t ** 2 * cg
        obj = 2.0 * n * float(np.max(vals))
        if obj < best_val:
            best_val = obj
            best_t = t

    return best_t, best_val


def smooth_gradient_step(a, tau=200.0, lr_scale=0.5):
    """One step of smoothed (log-sum-exp) gradient descent."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    s_exp = float(np.sum(exp_vals))
    if s_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / s_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = np.fft.fft(p, N)
    A = np.fft.fft(a, N)
    grad_full = np.fft.ifft(P * np.conj(A)).real
    grad = 2.0 * grad_full[:n]

    lr = lr_scale / float(np.sqrt(n))
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def optimize_core(a_init, time_budget, seed=42, K=15):
    """Core optimization: active-set LP + exact line search + momentum."""
    start = time.monotonic()
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)

    it = 0
    no_improve = 0
    prev_g = None
    mom_alpha = 0.4
    consecutive_worse = 0

    while time.monotonic() - start < time_budget:
        remaining_check = time_budget - (time.monotonic() - start)
        if remaining_check < 0.1:
            break

        ca = compute_conv(a)

        # Active set: top K positions of conv(a,a)
        top_idx = np.argsort(ca)[::-1][:K]
        active = sorted(top_idx.tolist())

        # Solve LP for direction
        g, M_lp = solve_active_lp(a, active)

        if g is not None:
            # Try momentum extrapolation
            g_try = g
            if prev_g is not None and it > 20:
                g_mom = g + mom_alpha * (g - prev_g)
                g_mom = np.maximum(g_mom, 0)
                s_mom = float(np.sum(g_mom))
                if s_mom > 1e-10:
                    g_mom = g_mom / s_mom
                    cag_m = compute_cross_conv(a, g_mom)
                    cg_m = compute_conv(g_mom)
                    t_m, obj_m = line_search_precomputed(ca, cag_m, cg_m, n)
                    cag_g = compute_cross_conv(a, g)
                    cg_g = compute_conv(g)
                    t_g, obj_g = line_search_precomputed(ca, cag_g, cg_g, n)
                    if obj_m < obj_g:
                        g_try = g_mom
                        t_best, obj_best = t_m, obj_m
                    else:
                        t_best, obj_best = t_g, obj_g
                else:
                    cag_g = compute_cross_conv(a, g)
                    cg_g = compute_conv(g)
                    t_best, obj_best = line_search_precomputed(ca, cag_g, cg_g, n)
            else:
                cag_g = compute_cross_conv(a, g)
                cg_g = compute_conv(g)
                t_best, obj_best = line_search_precomputed(ca, cag_g, cg_g, n)
        else:
            # Fallback: smoothed gradient
            a_sg = smooth_gradient_step(a, tau=150.0)
            t_best, obj_best = 1.0, compute_obj(a_sg)
            g = a_sg

        prev_g = g.copy()
        it += 1

        if obj_best < best_obj - 1e-13:
            a_new = (1.0 - t_best) * a + t_best * g_try
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                best_obj = obj_best
                best_a = a.copy()
                no_improve = 0
                consecutive_worse = 0
            else:
                no_improve += 1
                consecutive_worse += 1
        else:
            no_improve += 1
            consecutive_worse += 1

        # Adaptive momentum
        if no_improve > 10 and mom_alpha > 0.1:
            mom_alpha = max(0.1, mom_alpha * 0.95)
        if no_improve == 0:
            mom_alpha = min(0.6, mom_alpha * 1.02)

        # Perturbation if stuck
        if no_improve > 40:
            strength = min(0.2, 0.003 * no_improve)
            pert = rng.normal(0, strength, n)
            a = best_a * np.exp(pert)
            a = np.maximum(a, 0)
            a = a / float(np.sum(a))
            no_improve = 0
            prev_g = None
            mom_alpha = 0.3

        # Hard reset if very stuck
        if consecutive_worse > 200:
            a = best_a.copy()
            prev_g = None
            mom_alpha = 0.2
            consecutive_worse = 0

        if it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    it={it}: obj={best_obj:.8f}, no_improve={no_improve}, time={elapsed:.0f}s/{time_budget:.0f}s")

    return best_a, best_obj


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization with active-set LP,
    multi-length exploration, diverse restarts, and final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        if parent.size > 0 and np.isfinite(parent).all():
            s = float(np.sum(parent))
            if s > 0.01:
                parent = parent / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_cur = len(best_seq)

    # === Phase 1: Core optimization with K=12 ===
    print(f"\n=== Phase 1: Core LP K=12 (n={n_cur}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.30, 450.0)
    if alloc > 30:
        a_opt, obj_opt = optimize_core(best_seq, alloc, seed=seed, K=12)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_opt:.8f} (no improvement)")

    # === Phase 2: Core optimization with K=20 ===
    print(f"\n=== Phase 2: Core LP K=20 ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.20, 300.0)
    if alloc > 30:
        a_opt, obj_opt = optimize_core(best_seq, alloc, seed=seed + 1, K=20)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_opt:.8f} (no improvement)")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length exploration ===")
    n_best = len(best_seq)
    n_candidates = [n_best - 200, n_best - 100, n_best + 100, n_best + 200,
                    n_best + 400, 500, 700, 900, 1100, 1300]
    n_candidates = [n for n in n_candidates if 300 <= n <= 2500 and abs(n - n_best) > 50]
    n_candidates = sorted(set(n_candidates))[:4]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_seq(best_seq, n_try)
        a_try = a_try / float(np.sum(a_try))

        alloc = min(remaining * 0.15, 120.0)
        if alloc > 25:
            a_opt, obj_opt = optimize_core(a_try, alloc, seed=seed + n_try, K=15)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 4: Diverse restarts ===
    print(f"\n=== Phase 4: Diverse restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    # Build diverse initializations
    inits = []

    # Fourier perturbation
    pert = np.zeros(n_cur)
    for k in range(1, 5):
        amp = rng.uniform(0.05, 0.15) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(2.0 * np.pi * k * np.arange(n_cur) / n_cur + phase)
    a_f = best_seq * np.exp(pert)
    inits.append(a_f / float(np.sum(a_f)))

    # Blend with uniform
    a_u = 0.5 * best_seq + 0.5 * np.ones(n_cur) / n_cur
    inits.append(a_u / float(np.sum(a_u)))

    # Smooth bump (raised cosine)
    x = np.linspace(0, 1, n_cur)
    a_cos = 0.5 + 0.5 * np.cos(2.0 * np.pi * (x - 0.5))
    a_cos = np.maximum(a_cos, 0.01)
    inits.append(a_cos / float(np.sum(a_cos)))

    # Two-peak
    x = np.linspace(0, 1, n_cur)
    a_2p = np.exp(-40 * (x - 0.3) ** 2) + np.exp(-40 * (x - 0.7) ** 2)
    inits.append(a_2p / float(np.sum(a_2p)))

    for i, a_init in enumerate(inits):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(remaining * 0.15, 90.0)
        if alloc > 20:
            a_opt, obj_opt = optimize_core(a_init, alloc, seed=seed + 200 + i * 7, K=15)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Restart {i}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {i}: obj={obj_opt:.8f}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.5, 250.0)
        a_opt, obj_opt = optimize_core(best_seq, alloc, seed=seed + 9999, K=18)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_opt:.8f}")

    # One more round with different K
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.5, 150.0)
        a_opt, obj_opt = optimize_core(best_seq, alloc, seed=seed + 77777, K=8)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final LP K=8: obj={best_score:.8f} **")

    # Convert to list
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
