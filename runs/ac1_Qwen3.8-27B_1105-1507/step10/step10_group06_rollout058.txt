"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:
- Fast LP-based best-response computation with cl
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_best_response(a, solver=cp.CLARABEL):
    """
    Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0.
    Returns g_opt or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build the constraint matrix efficiently
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
        if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(g.value, 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    
    # Fallback to SCS
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
        if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(g.value, 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    
    return None


def smoothed_gradient(a, tau):
    """Compute gradient of smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 4 * n:
        N <<= 1
    P = np.fft.fft(p, N)
    A = np.fft.fft(a, N)
    g_full = np.fft.ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm.shape[0])
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm
    return A


def solve_best_response(a, solver_name=None):
