import time
from scipy.fft import fft, ifft, rfft, irfft
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def cross_corr_fft(z, a):
    """Compute (z cross-correlate a)_j = sum_k z_k * a_{k-j} for j=0..n-1.
    z has length 2n-1, a has length n."""
    n = len(a)
    N = next_pow2(3 * n)
    Z = fft(z, N)
    A = fft(a, N)
    result = ifft(Z * np.conj(A)).real
    return result[:n]

        if time_limit is not None and time.monotonic() - start_time > time_limit:
            break
        
        # Gradient at y
        grad_y = 2.0 * cross_corr_fft(z, y)
        
        # FISTA update: y_new = project(y - grad_y / L)
        step = 1.0 / L
        v = y - step * grad_y
        a_new = simplex_project(v)
        
        # Update y for next iteration
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = best_a.copy()
        else:
            y = y / s_y
        t = t_new
        a = a_new
        
        # Track best
        f_val = float(z @ compute_conv(a))
        if f_val < best_f:
            best_f = f_val
            best_a = a.copy()
    
    return best_a, best_f

def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_conv_lp(a):
    """Solve LP: min M s.t. (a_norm * g)_k <= M, sum(g)=1, g>=0.
    Returns (g_opt, M_opt, z_dual) or (None, None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Extract dual variables for the A@g <= M constraints
                    z_dual = None
                    try:
                        # The dual values are associated with the constraint A@g <= M
                        z_dual = np.maximum(prob.constraints[0].dual.value, 0)
                        z_sum = z_dual.sum()
                        if z_sum > 1e-10:
                            z_dual = z_dual / z_sum
                    except:
                        z_dual = None
                    return g_val, float(M.value), z_dual
        except Exception:
            continue
    return None, None, None

    2. Minimize F_z(a) = sum_k z_k (a*a)_k over simplex using FISTA
    3. Repeat
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(0)

    g_lp, M_lp, z_dual = solve_conv_lp(a)
    lp_count += 1
    if z_dual is None:
        # Fallback: use argmax as z
        conv = compute_conv(a)
        z_dual = np.zeros(2 * n - 1)
        z_dual[np.argmax(conv)] = 1.0
    
    # Main loop
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        # Step 1: FISTA on the quadratic defined by z
        fista_budget = min(5.0, remaining * 0.3)
        fista_start = time.monotonic()
        a_fista, _ = fista_on_quadratic(
            z_dual, a, L=4.0,
            max_iter=20000,
            time_limit=fista_budget,
            start_time=fista_start
        )
        
        obj_fista = compute_obj(a_fista)
        if obj_fista < best_obj - 1e-12:
            best_obj = obj_fista
            best_a = a_fista.copy()
            a = a_fista.copy()
        
        # Step 2: Solve LP to get new z
        lp_start = time.monotonic()
        g_lp, M_lp, z_dual_new = solve_conv_lp(a)
        lp_time = time.monotonic() - lp_start
        lp_count += 1
        
        if z_dual_new is not None:
            z_dual = z_dual_new
        else:
            # Fallback: use softmax of convolution
            conv = compute_conv(a)
            tau = 50.0
            exp_vals = np.exp(tau * (conv - conv.max()))
            z_dual = exp_vals / exp_vals.sum()
        
        # Step 3: Try blending with g_lp
        if g_lp is not None:
            for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
                a_try = (1 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break
        
        if verbose and lp_count % 10 == 1:
            print(f"  DA lp#{lp_count}: obj={best_obj:.10f}, M_lp={M_lp:.6f}, lp_time={lp_time:.2f}s")
    
    return best_a, best_obj
