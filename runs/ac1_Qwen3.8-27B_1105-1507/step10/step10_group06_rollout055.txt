"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvement over previous: uses running average of LP iterates (theoretically 
optimal by minimax theorem), combined with multi-length exploration and 
FFT-based mirror descent for fast refinement.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def mirror_descent_step(a, g, eta):
    """
    One step of mirror descent (exponentiated gradient) on the simplex.
    Given current g and "gradient" from a, update g.
    The gradient here is the smoothed peak direction.
    """
    n = len(a)
    # Compute a*g convolution
    conv = compute_conv_np(a, g)
    max_conv = np.max(conv)
    # Gradient of max(conv) with respect to g is a shifted (the "active" part)
    # For mirror descent, we use the subgradient at the max
    # Actually for exponentiated gradient, we use:
    # g_new[j] = g[j] * exp(-eta * (a shifted by argmax)[j])
    # But this only uses one constraint. Better to use smoothed version.
    
    # Use log-sum-exp smoothing
    tau = 50.0
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = np.sum(exp_vals)
    p = exp_vals / sum_exp  # probability on the "peaks"
    
    # Gradient of the smoothed objective w.r.t. g
    # d/dg_j [sum_k p_k * (a*g)[k]] = sum_k p_k * a[k-j] (valid indices)
    # This is p * a reversed... actually it's a cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A_f = fft(a, N)
    grad_full = ifft(P * np.conj(A_f)).real
    grad = grad_full[:n]
    
    # Exponentiated gradient update
    g_new = g * np.exp(-eta * grad)
    g_new = np.maximum(g_new, 1e-30)
    g_new = g_new / np.sum(g_new)
    return g_new


def compute_conv_np(a, g):
    """Compute convolution a*g via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:2 * n - 1]

def fast_mirror_descent(a, time_budget, verbose=True):

    Fast mirror descent (exponentiated gradient) using FFT for gradient computation.
    Much faster per iteration than LP but converges slower.
    """

    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # We optimize over g (the second player) given fixed a
    g = a.copy()  # start from current
    g_best = g.copy()
    
    eta = 1.0
    tau = 30.0
    it = 0

    # Running average
    g_avg = g.copy()
    
    while time.monotonic() - start < time_budget:
        # Update g using mirror descent against a
        g = mirror_descent_step(a, g, eta)
        g_avg = (g_avg * it + g) / (it + 1)
        it += 1
        
        eta *= 0.999
        tau = min(tau * 1.05, 500.0)

        # Periodically evaluate
        if it % 50 == 0:
            for cand in [g, g_avg]:
                cand = np.maximum(cand, 0)
                s_c = float(np.sum(cand))
                if s_c > 1e-10:
                    cand = cand / s_c
                    obj_c = compute_obj(cand)
                    if obj_c < best_obj:
                        best_obj = obj_c
                        best_a = cand.copy()
                        g_best = cand.copy()

        # Also periodically update a (alternating)
        if it % 100 == 0:
            a_new = 0.5 * a + 0.5 * g_avg
            a_new = np.maximum(a_new, 0)
            s_an = float(np.sum(a_new))
            if s_an > 1e-10:
                a = a_new / s_an

    return best_a

def fourier_perturb(a, rng, strength=0.03):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 5
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: (1) Uses pure best-response LP dynamics with running average
of iterates (theoretically converges to game value by minimax), (2) multi-length

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_sum = float(np.sum(a))
    if a_sum < 1e-15:
        return None, None
    a_norm = a / a_sum
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:

    a = a / s

    best_obj = compute_obj(a)

    beta = 0.9

    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0

            lr = 0.8 * lr + 0.2 * adaptive_lr

        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))

    phases[N // 2 + 1:] = phases[1:N // 2][::-1] if N // 2 > 1 else 0
    spec = np.exp(1j * phases)
    spec[0] = 1.0
    if N % 2 == 0:
        spec[N // 2] = 1.0

    # Inverse FFT
    sig = ifft(spec).real
    sig = sig[:n]
    # Make non-negative
    sig = np.maximum(sig, 0)
    s = float(np.sum(sig))
    if s < 1e-10:
        return np.ones(n) / n
    sig = sig / s

    # Blend with uniform for stability
    a = flatness * sig + (1 - flatness) / n
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def fourier_perturb(a, rng, strength=0.03):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 5
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def lp_best_response_dynamics(a_init, time_budget, seed=42, verbose=True,
                               eval_every=5, use_average=True):
    """
    Pure LP best-response dynamics with running average tracking.
    At each step, solve the LP to get g = best_response(a), set a = g.
    Track the running average of all a iterates and evaluate it periodically.
    The running average converges to the game value (minimax theorem).
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Running sum for average
    a_sum = a.copy()
    a_count = 1

    g_sum = np.zeros(n)
    g_count = 0

    iter_count = 0
    no_improve = 0
    lp_fail_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_conv_lp(a)
        lp_time = time.monotonic() - t0

        if g_lp is None:
            lp_fail_count += 1
            if lp_fail_count > 10:
                break
            continue

        lp_fail_count = 0
        iter_count += 1

        # Update running averages
        a_count += 1
        a_sum = a_sum + g_lp  # a_{t} = g_{t-1}, so new a is g_lp

        g_count += 1
        g_sum = g_sum + g_lp

        # Set next a to the LP solution (pure best response)
        a = g_lp.copy()

        # Periodically evaluate the running average
        if use_average and iter_count % eval_every == 0:
            a_avg = a_sum / a_count
            a_avg = np.maximum(a_avg, 0)
            s_avg = float(np.sum(a_avg))
            if s_avg > 1e-10:
                a_avg = a_avg / s_avg
                obj_avg = compute_obj(a_avg)
                if obj_avg < best_obj:
                    best_obj = obj_avg
                    best_a = a_avg.copy()
                    no_improve = 0
                else:
                    no_improve += 1

            # Also check g average
            g_avg = g_sum / g_count
            g_avg = np.maximum(g_avg, 0)
            s_g = float(np.sum(g_avg))
            if s_g > 1e-10:
                g_avg = g_avg / s_g
                obj_g = compute_obj(g_avg)
                if obj_g < best_obj:
                    best_obj = obj_g
                    best_a = g_avg.copy()
                    no_improve = 0

        # Also evaluate current iterate
        if iter_count % (eval_every * 2) == 0:
            obj_cur = compute_obj(a)
            if obj_cur < best_obj:
                best_obj = obj_cur
                best_a = a.copy()
                no_improve = 0

        # If stuck for too long, perturb
        if no_improve >= eval_every * 4:
            strength = 0.005 + 0.003 * min(no_improve // (eval_every * 4), 10)
            a = fourier_perturb(a, rng, strength=strength)
            # Also perturb the running sum to "reset" the average partially
            a_sum = a_sum * 0.5 + a * a_count * 0.5
            no_improve = 0

        if verbose and iter_count % 30 == 0:
            print(f"    BR-dyn {iter_count}: best={best_obj:.10f}, "
                  f"M_lp={M_lp:.6f}, t={lp_time:.2f}s")

    # Final evaluation of the average
    if use_average and a_count > 0:
        a_avg = a_sum / a_count
        a_avg = np.maximum(a_avg, 0)
        s_avg = float(np.sum(a_avg))
        if s_avg > 1e-10:
            a_avg = a_avg / s_avg
            obj_avg = compute_obj(a_avg)
            if obj_avg < best_obj:
                best_obj = obj_avg
                best_a = a_avg.copy()
                if verbose:
                    print(f"    Final avg: {best_obj:.10f} **")

    if verbose:
        print(f"    BR-dyn done: {iter_count} iters, best={best_obj:.10f}")

    return best_a


def lp_blend_method(a_init, time_budget, seed=42, verbose=True):
    """
    LP alternating with adaptive blending (similar to previous SOTA but improved).
    Uses multiple t candidates and evaluates running average.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Running average
    a_avg_sum = a.copy()
    a_avg_count = 1

    iter_count = 0
    no_improve = 0
    t_blend = 0.4

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_conv_lp(a)
        lp_time = time.monotonic() - t0

        if g_lp is None:
            break

        iter_count += 1

        # Try multiple candidates
        candidates = []
        # Pure LP solution
        candidates.append(g_lp)
        # Blended
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, t_blend]:
            candidates.append((1 - t) * a + t * g_lp)
        # Running average
        a_avg = a_avg_sum / a_avg_count
        candidates.append(a_avg)
        # Blend current with average
        candidates.append(0.5 * a + 0.5 * a_avg)

        best_local = best_obj
        best_local_a = best_a.copy()

        for cand in candidates:
            cand = np.maximum(cand, 0)
            s_c = float(np.sum(cand))
            if s_c < 1e-10:
                continue
            cand = cand / s_c
            obj_c = compute_obj(cand)
            if obj_c < best_local:
                best_local = obj_c
                best_local_a = cand.copy()

        if best_local < best_obj - 1e-15:
            best_obj = best_local
            best_a = best_local_a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update running average with the new best
        a_avg_count += 1
        a_avg_sum = a_avg_sum + best_local_a

        # Update a for next LP solve
        if no_improve > 15:
            a = fourier_perturb(best_a, rng, strength=0.01)
            no_improve = 0
        else:
            a = 0.6 * a + 0.4 * g_lp
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a

        if verbose and iter_count % 30 == 0:
            print(f"    LP-blend {iter_count}: best={best_obj:.10f}, "
                  f"M_lp={M_lp:.6f}, t={lp_time:.2f}s")

    if verbose:
        print(f"    LP-blend done: {iter_count} iters, best={best_obj:.10f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines LP best-response dynamics with running average,
    LP blending, multi-length exploration, and gradient polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0

    # Load parent
    try:
        a_parent = np.asarray(height_sequence_1, dtype=np.float64)
        a_parent = np.maximum(a_parent, 0)
        if a_parent.size == 0 or a_parent.sum() < 0.01:
            raise ValueError("Invalid parent")
        a_parent = a_parent / a_parent.sum()
        n_parent = a_parent.size
        obj_parent = compute_obj(a_parent)
        print(f"Parent: n={n_parent}, obj={obj_parent:.10f}")
    except Exception:
        n_parent = 800
        a_parent = np.ones(n_parent) / n_parent
        obj_parent = compute_obj(a_parent)
        print(f"Default init: n={n_parent}, obj={obj_parent:.10f}")

    best_a = a_parent.copy()
    best_obj = obj_parent
    rng = np.random.default_rng(seed)

    # === Phase 1: LP Best-Response Dynamics with Running Average ===
    print(f"\n=== Phase 1: Best-Response Dynamics (n={n_parent}) ===")
    p1_deadline = start_time + budget_s * 0.35
    p1_budget = max(60, p1_deadline - time.monotonic())
    a_p1 = lp_best_response_dynamics(best_a, time_budget=p1_budget,
                                      seed=seed, verbose=True,
                                      eval_every=5, use_average=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1.copy()
        print(f"  Phase 1: obj={best_obj:.10f} **")
    else:
        print(f"  Phase 1: obj={obj_p1:.10f}")

    # === Phase 2: LP Blend Method ===
    print(f"\n=== Phase 2: LP Blend Method ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        p2_budget = min(350, remaining * 0.35)
        a_p2 = lp_blend_method(best_a, time_budget=p2_budget,
                                seed=seed + 100, verbose=True)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_obj:
            best_obj = obj_p2
            best_a = a_p2.copy()
            print(f"  Phase 2: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 2: obj={obj_p2:.10f}")

    # === Phase 3: Nesterov GD Polish ===
    print(f"\n=== Phase 3: Nesterov GD Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = min(120, remaining * 0.2)
        a_p3 = nesterov_gd_polish(best_a, time_budget=p3_budget,
                                   tau_init=20.0, tau_max=1500.0)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3.copy()
            print(f"  Phase 3: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 3: obj={obj_p3:.10f}")

    # === Phase 4: Second Best-Response pass from perturbed start ===
    print(f"\n=== Phase 4: Second BR-Dynamics Pass ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        p4_budget = min(300, remaining * 0.3)
        a_pert = fourier_perturb(best_a, rng, strength=0.006)
        a_p4 = lp_best_response_dynamics(a_pert, time_budget=p4_budget,
                                          seed=seed + 500, verbose=True,
                                          eval_every=5, use_average=True)
        obj_p4 = compute_obj(a_p4)
        if obj_p4 < best_obj:
            best_obj = obj_p4
            best_a = a_p4.copy()
            print(f"  Phase 4: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 4: obj={obj_p4:.10f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    # Generate candidate lengths
    n_candidates = set()
    for delta in [-50, -30, -20, -10, 10, 20, 30, 50, 75, 100]:
        n_try = n_best + delta
        if 400 <= n_try <= 1200:
            n_candidates.add(n_try)
    # Also try some specific "nice" lengths
    for n_try in [500, 600, 700, 800, 900, 1000, 1100]:
        if 400 <= n_try <= 1200:
            n_candidates.add(n_try)

    # Sort by distance to current best
    n_sorted = sorted(n_candidates, key=lambda x: abs(x - n_best))

    explored = 0
    for n_try in n_sorted:
        remaining = deadline - time.monotonic()
        if remaining < 50 or explored >= 5:
            break

        a_try = resample_to_length(best_a, n_try)
        alloc = min(remaining * 0.2, 120.0)
        if alloc < 20:
            break

        explored += 1
        print(f"  Trying n={n_try} (budget={alloc:.0f}s)...")
        a_try = lp_best_response_dynamics(a_try, time_budget=alloc,
                                           seed=n_try, verbose=False,
                                           eval_every=10, use_average=True)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: obj={best_obj:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 6: Final LP polish on best found ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Final LP Polish ===")
        alloc = min(remaining * 0.4, 80.0)
        a_final = lp_blend_method(best_a, time_budget=alloc,
                                   seed=seed + 9999, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final LP: obj={best_obj:.10f} **")
        else:
            print(f"  Final LP: obj={obj_final:.10f}")

    # === Phase 7: Ultra-final gradient polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 30.0)
        a_ultra = nesterov_gd_polish(best_a, time_budget=alloc,
                                      tau_init=100.0, tau_max=3000.0)
        obj_ultra = compute_obj(a_ultra)
        if obj_ultra < best_obj:
            best_obj = obj_ultra
            best_a = a_ultra.copy()
            print(f"  Ultra polish: obj={best_obj:.10f} **")

    # Final validation
    if best_a is None:
        best_a = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
