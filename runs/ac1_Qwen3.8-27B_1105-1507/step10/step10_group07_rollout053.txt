"""
Multi-strategy optimizer for minimum peak autocorrelation of non-negative sequences.
Uses smoothed gradient descent, mirror descent, coordinate descent, and simulated annealing
with diverse initializations and length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_smoothed_obj_and_grad(a, tau, top_k=None):
    """
    Compute smoothed objective and gradient using log-sum-exp or top-k softmax.
    Returns (smoothed_obj, gradient_of_smoothed_max).
    The smoothed max approximates max(conv) using softmax weighting.
    """
    n = len(a)
    conv = compute_conv(a)
    if top_k is not None and top_k < len(conv):
        # Use top-k peaks with softmax weighting
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        top_vals = conv[top_idx]
        max_val = top_vals.max()
        weights = np.exp(tau * (top_vals - max_val))
        weights /= weights.sum()
        # Smoothed max ≈ max_val + log(sum(exp(tau*(top_vals - max_val))))/tau
        # But for gradient purposes, use weighted average
        smoothed_max = float(np.dot(weights, top_vals))
        # Gradient: sum over top peaks of w_k * db[k]/da
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            # db[k]/da[i] = 2*a[k-i] if 0<=k-i<n and k!=2i, or 2*a[i] if k=2i
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if 0 <= j < n:
                    if i == j:
                        grad[i] += w * 2.0 * a[i]
                    else:
                        grad[i] += w * 2.0 * a[j]
        return smoothed_max, grad
    else:
        max_val = conv.max()
        weights = np.exp(tau * (conv - max_val))
        weights /= weights.sum()
        smoothed_max = float(np.dot(weights, conv))
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = weights[k]
            if wk < 1e-12:
                continue
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if 0 <= j < n:
                    if i == j:
                        grad[i] += wk * 2.0 * a[i]
                    else:
                        grad[i] += wk * 2.0 * a[j]
        return smoothed_max, grad

from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a / s)))

def compute_gradient(a, tau, top_k):
    """Compute gradient of smoothed max autocorrelation w.r.t. a (on simplex)."""
    n = len(a)
    conv = compute_conv(a)
    L = len(conv)  # 2n-1
    
    if top_k < L:
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        top_vals = conv[top_idx]
    else:
        top_idx = np.arange(L)
        top_vals = conv
    
    max_val = top_vals.max()
    log_weights = tau * (top_vals - max_val)
    log_weights -= log_weights.max()  # numerical stability
    weights = np.exp(log_weights)
    weights /= weights.sum()
    
    smoothed_max = float(np.dot(weights, top_vals))
    
    # Build weight vector of length 2n-1
    w = np.zeros(2*n-1)
    w[top_idx] = weights
    
    # Gradient via cross-correlation: g[i] = 2 * sum_j w[i+j] * a[j]
    N = 1
    while N < 3*n:
        N <<= 1
    W = fft(w, N)
    A = fft(a, N)
    g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    return smoothed_max, g

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_gd(a, time_budget, tau_init, tau_end, lr_init, top_k=20, seed=42):
    """Smoothed gradient descent with Adam optimizer."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    lr = lr_init
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        
        _, g = compute_gradient(a, tau, top_k)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new

            no_improve = 0

            no_improve = 0

    x_old = np.linspace(0, 1, n)

    return np.maximum(np.interp(x_new, x_old, a), 0)

"""
FFT-accelerated smoothed gradient optimizer for minimum peak autocorrelation
of non-negative step functions. Uses Adam optimizer, adaptive top-k peaks,
mirror descent, structured perturbations, and multi-length exploration.
Key innovation: O(n log n) gradient via FFT cross-correlation, enabling
many more iterations within the time budget.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_gradient_fft(a, tau, top_k):
    """
    Compute gradient of smoothed max autocorrelation using FFT cross-correlation.
    Returns (smoothed_max_value, gradient_vector_of_length_n).
    """
    n = len(a)
    conv = compute_conv(a)
    L = len(conv)  # 2n-1

    if top_k < L:
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        top_vals = conv[top_idx]
    else:
        top_idx = np.arange(L)
        top_vals = conv.copy()

    max_val = float(top_vals.max())
    log_w = tau * (top_vals - max_val)
    log_w -= log_w.max()
    weights = np.exp(log_w)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones_like(top_vals) / len(top_vals)
    else:
        weights /= wsum

    smoothed_max = float(np.dot(weights, top_vals))

    # Build full weight vector
    w = np.zeros(2 * n - 1)
    w[top_idx] = weights

    # Gradient: g[i] = 2 * sum_j w[i+j] * a[j] (cross-correlation)
    N = 1
    while N < 3 * n:
        N <<= 1
    W = fft(w, N)
    A = fft(a, N)
    g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    return smoothed_max, g


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def adam_smoothed_gd(a, time_budget, tau_init, tau_end, lr_init,
                     top_k_init, top_k_end, seed=42):
    """
    Smoothed gradient descent with Adam optimizer and adaptive top-k.
    Uses FFT-based gradient for O(n log n) per iteration.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    lr = lr_init
    beta1, beta2, eps = 0.75, 0.95, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0
    max_no_improve = 60

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1.0))

        # Adaptive temperature and top-k
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        top_k = max(3, int(top_k_init + (top_k_end - top_k_init) * progress))
        top_k = min(top_k, 2 * n - 1)

        _, g = compute_gradient_fft(a, tau, top_k)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr *= 0.9995

        if no_improve >= max_no_improve:
            # Structured perturbation: multiply by smooth random function
            freqs = rng.integers(1, 8, size=3)
            phases = rng.uniform(0, 2 * np.pi, size=3)
            amplitudes = rng.uniform(0.005, 0.03, size=3)
            x = np.arange(n) / n
            pert = np.zeros(n)
            for f, ph, amp in zip(freqs, phases, amplitudes):
                pert += amp * np.sin(2 * np.pi * f * x + ph)
            a = a * np.exp(pert)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
            else:
                a = best_a.copy()
            no_improve = 0
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = lr_init * 0.7

    return best_a


def mirror_descent_opt(a, time_budget, eta_init, top_k=10, seed=42):
    """Mirror descent with multiplicative weight updates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    eta = eta_init
    no_improve = 0

    while time.monotonic() - start < time_budget:
        _, g = compute_gradient_fft(a, 1.0, top_k)

        # Multiplicative update
        scale = np.exp(-eta * g)
        a_new = a * scale
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            eta *= 0.93

        if no_improve >= 40:
            # Perturb
            a = best_a * np.exp(rng.normal(0, 0.012, n))
            a = np.maximum(a, 0)
            s = a.sum()
            a = a / s if s > 1e-10 else np.ones(n) / n
            no_improve = 0
            eta = eta_init

    return best_a


def coordinate_polish(a, time_budget, n_sweeps=5, seed=42):
    """
    Coordinate-wise line search: for each coordinate, try increasing/decreasing
    and redistribute to a random partner. Accept if objective improves.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for sweep in range(n_sweeps):
        if time.monotonic() - start >= time_budget:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            if a[i] < 1e-12:
                continue
            # Try moving fraction of a[i] to a random j
            j = rng.integers(0, n)
            if j == i:
                continue
            for frac in [0.1, 0.3, 0.5]:
                delta = frac * a[i]
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break
    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def structured_init(n, mode, seed=0):
    """Create diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = 1.0 - 2.0 * np.abs(x - 0.5)
    elif mode == 2:
        a = np.exp(-2.0 * (x - 0.5) ** 2) + 0.1
    elif mode == 3:
        a = np.exp(-3.0 * x) + np.exp(-3.0 * (1.0 - x)) + 0.1
    elif mode == 4:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x)
    elif mode == 5:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 5 * x) * np.sin(2 * np.pi * 2 * x)
    elif mode == 6:
        a = rng.exponential(1.0, n)
    else:
        a = 1.0 + 0.2 * np.sin(2 * np.pi * rng.integers(1, 10) * x)
    a = np.maximum(a, 0.01)
    return a / a.sum()

    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        budget = min(remaining * 0.35, 100.0)
        a_f = adam_smoothed_gd(best_seq, budget,
                               tau_init=2.0, tau_end=50.0,
                               lr_init=0.2 / np.sqrt(len(best_seq)),
                               top_k_init=8, top_k_end=40,
                               seed=999)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score, best_seq = obj_f, a_f
            print(f"  Final Adam: obj={best_score:.8f} **")
        else:
            print(f"  Final Adam: obj={obj_f:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        budget = min(remaining * 0.25, 50.0)
        a_m = mirror_descent_opt(best_seq, budget,
                                 eta_init=0.05, top_k=5, seed=777)
        obj_m = compute_obj(a_m)
        if obj_m < best_score:
            best_score, best_seq = obj_m, a_m
            print(f"  Final Mirror: obj={best_score:.8f} **")

    # === Final coordinate polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_c = coordinate_polish(best_seq, min(remaining * 0.3, 20.0),
                                n_sweeps=2, seed=1234)
        obj_c = compute_obj(a_c)
        if obj_c < best_score:
            best_score, best_seq = obj_c, a_c
            print(f"  Final Coord: obj={best_score:.8f} **")

    # --- Final validation ---
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
