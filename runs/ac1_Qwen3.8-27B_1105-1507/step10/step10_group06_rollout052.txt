"""
Multi-phase optimization for step function coefficients minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternating minimization with warm starts, Nesterov gradient polishing with
progressive log-sum-exp sharpening, peak-targeted pairwise exchanges, and multi-length exploration.
Key improvements over previous: faster LP via warm starts, adaptive line search,
progressive tau sharpening for gradient, and peak-directed local search.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    # Try warm start
    if warm_start is not None:
        try:
            g.value = np.clip(warm_start, 0, None)
            gs = float(np.sum(g.value))
            if gs > 0:
                g.value = g.value / gs
        except Exception:
            pass

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of (a*a)_k using FFT.
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Numerically stable softmax
    exp_vals = np.exp(shifted - shifted.max())
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def project_simplex(v):
    """Project vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """
    Nesterov accelerated gradient descent on smoothed objective with progressive tau.

    """
    n = len(a)

        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 1.5 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0
    no_improve_count = 0
    obj_history = []

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        # Nesterov lookahead
        y = a + beta * v
        y = project_simplex(y)

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        # Adaptive learning rate
        if g_norm > 1e-15:
            adaptive_lr = 0.3 / (g_norm + 1e-10)

        for t in candidates_t:
            t = min(max(t, 0.001), 1.0)
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-15:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            best_warm = best_a.copy()
            t_blend = max(0.002, best_local_t * 0.97)
            stall_count = 0
            if verbose and iter_count % 50 == 0:
                print(f"    LP alt {iter_count}: obj={best_obj:.10f}, t={t_blend:.4f}, M={M_lp:.8f}")
        else:
            stall_count += 1
            if stall_count >= 8:
                # Structured perturbation
                strength = 0.005 + 0.002 * min(stall_count, 20)
                # Fourier perturbation
                pert = np.zeros(n)
                for k in range(1, 8):
                    freq = 2.0 * np.pi * k / n
                    amp = strength / k
                    phase = rng.uniform(0, 2 * np.pi)
                    pert += amp * np.cos(freq * np.arange(n) + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                t_blend = max(0.1, min(0.6, t_blend * 1.5))
                stall_count = 0
            elif stall_count >= 4:
                # Small random nudge
                a = best_a + 0.003 * rng.standard_normal(n) * np.maximum(best_a, 0.001)
                a = np.maximum(a, 0)
                a = a / a.sum()

    if verbose:
        print(f"    LP alternating done: {iter_count} iterations, best_obj={best_obj:.10f}")

    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_initializations(n, rng):
    """Generate diverse initializations for different shapes."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window (high center)
    j = np.arange(n)
    c = np.cos(np.pi * (j / (n - 1) - 0.5)) ** 2
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())

    # Inverse cosine (high edges)
    s = np.sin(np.pi * (j / (n - 1) - 0.5)) ** 2
    s = np.maximum(s, 0.01)
    inits.append(s / s.sum())

    # Two peaks
    peak1 = np.exp(-((j - 0.25 * n) ** 2) / (2 * (0.1 * n) ** 2))
    peak2 = np.exp(-((j - 0.75 * n) ** 2) / (2 * (0.1 * n) ** 2))
    two = peak1 + peak2
    two = np.maximum(two, 0.01)
    inits.append(two / two.sum())

    # Random
    r = rng.dirichlet(np.ones(n))
    inits.append(r)

    # Smooth random (low-pass filtered)
    r2 = rng.standard_normal(n)
    # Simple moving average
    kernel = np.ones(10) / 10
    from scipy.signal import convolve as sig_convolve
    r2_smooth = sig_convolve(r2, kernel, mode='same')
    r2_smooth = np.maximum(r2_smooth, 0)
    if r2_smooth.sum() > 0:
        inits.append(r2_smooth / r2_smooth.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-phase optimization combining LP alternating,
    gradient polish, peak-targeted search, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0  # Leave 20s for safety

    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0)
        if parent.size == 0 or parent.sum() < 0.01:
            raise ValueError("Invalid parent")
        parent = parent / parent.sum()
        n_init = parent.size
        print(f"Parent loaded: n={n_init}, obj={compute_obj(parent):.10f}")
    except Exception:
        n_init = 800
        parent = np.ones(n_init) / n_init
        print(f"Using default init: n={n_init}")

    best_a = parent.copy()
    best_obj = compute_obj(parent)
    print(f"Initial best: obj={best_obj:.10f}")

    # === Phase 1: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternating (n={n_init}) ===")
    p1_deadline = start_time + budget_s * 0.55
    p1_budget = max(60, p1_deadline - time.monotonic())
    a_p1 = lp_alternate(best_a, time_budget=p1_budget, seed=seed, verbose=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1.copy()
        print(f"  Phase 1 improved: obj={best_obj:.10f} **")
    else:
        print(f"  Phase 1: obj={obj_p1:.10f} (no improvement)")

    # === Phase 2: Try alternative initialization with LP ===
    print(f"\n=== Phase 2: Alternative Init ===")
    remaining = deadline - time.monotonic()
    if remaining > 120:
        p2_budget = min(remaining * 0.2, 200.0)
        # Try a perturbed version
        a_alt = best_a * np.exp(0.02 * rng.standard_normal(n_init))
        a_alt = np.maximum(a_alt, 0)
        a_alt = a_alt / a_alt.sum()
        a_p2 = lp_alternate(a_alt, time_budget=p2_budget, seed=seed + 77, verbose=False)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_obj:
            best_obj = obj_p2
            best_a = a_p2.copy()
            print(f"  Phase 2 improved: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 2: obj={obj_p2:.10f}")

    # === Phase 3: Nesterov Gradient Polish ===
    print(f"\n=== Phase 3: Gradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        p3_budget = min(remaining * 0.2, 250.0)
        a_p3 = nesterov_polish(best_a, time_budget=p3_budget,
                               tau_init=30.0, tau_max=8000.0, verbose=True)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3.copy()
            print(f"  Phase 3 improved: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 3: obj={obj_p3:.10f}")

    # === Phase 4: Peak-Targeted Search ===
    print(f"\n=== Phase 4: Peak-Targeted Search ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p4_budget = min(remaining * 0.15, 150.0)
        a_p4 = peak_targeted_search(best_a, time_budget=p4_budget,
                                     rng=np.random.default_rng(seed + 123), verbose=True)
        obj_p4 = compute_obj(a_p4)
        if obj_p4 < best_obj:
            best_obj = obj_p4
            best_a = a_p4.copy()
            print(f"  Phase 4 improved: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 4: obj={obj_p4:.10f}")

    # === Phase 5: Final LP Refinement ===
    print(f"\n=== Phase 5: Final LP Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p5_budget = min(remaining * 0.25, 200.0)
        a_p5 = lp_alternate(best_a, time_budget=p5_budget, seed=seed + 5555, verbose=False)
        obj_p5 = compute_obj(a_p5)
        if obj_p5 < best_obj:
            best_obj = obj_p5
            best_a = a_p5.copy()
            print(f"  Phase 5 improved: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 5: obj={obj_p5:.10f}")

    # === Phase 6: Gradient Polish Again ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p6_budget = min(remaining * 0.15, 100.0)
        a_p6 = nesterov_polish(best_a, time_budget=p6_budget,
                                tau_init=200.0, tau_max=15000.0, verbose=False)
        obj_p6 = compute_obj(a_p6)
        if obj_p6 < best_obj:
            best_obj = obj_p6
            best_a = a_p6.copy()
            print(f"  Phase 6 (GD) improved: obj={best_obj:.10f} **")

    # === Phase 7: Multi-length Exploration ===
    print(f"\n=== Phase 7: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    # Try nearby lengths
    n_options = sorted(set([
        n_best - 50, n_best + 50,
        n_best - 100, n_best + 100,
        600, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and n != n_best]
    n_options = n_options[:3]  # Limit to 3 to save time

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_to_length(best_a, n_try)
        alloc = min(remaining * 0.2, 100.0)
        if alloc < 20:
            break

        print(f"  Trying n={n_try} (budget={alloc:.0f}s)...")
        a_try = lp_alternate(a_try, time_budget=alloc * 0.7, seed=n_try, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_try = nesterov_polish(a_try, time_budget=min(remaining * 0.3, 30.0),
                                     tau_init=100.0, tau_max=5000.0, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_obj:.10f} ** IMPROVED!")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 8: Ultra-final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 8: Ultra-final Polish ({remaining:.0f}s remaining) ===")
        alloc = min(remaining * 0.5, 60.0)
        a_final = lp_alternate(best_a, time_budget=alloc, seed=seed + 99999, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final LP: obj={best_obj:.10f} **")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_final2 = nesterov_polish(best_a, time_budget=min(remaining * 0.4, 20.0),
                                        tau_init=500.0, tau_max=20000.0, verbose=False)
            obj_final2 = compute_obj(a_final2)
            if obj_final2 < best_obj:
                best_obj = obj_final2
                best_a = a_final2.copy()
                print(f"  Final GD: obj={best_obj:.10f} **")

    # Final validation
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_obj:
        best_obj = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
