"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP fixed-point iteration with line search as the main workhorse,
complemented by subgradient descent, spectral perturbation, and multi-n exploration.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max

def lp_solve(a, solver_list=None):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    if n > 3000:
        return None, None
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    
    for solver in solver_list:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver, verbose=False)
            t1 = time.monotonic()
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value), (t1-t0)
        except Exception:
            continue
    
    return None, None, None

def subgradient_step(a, lr):
    """One step of projected subgradient descent."""
    n = len(a)
    conv = compute_conv(a)
    max_val = np.max(conv)
    # Find all indices achieving max (within tolerance)
    tol = 1e-10 * max_val
    K_star = np.where(conv >= max_val - tol)[0]
    
    # Compute subgradient (average over argmax set)
    g = np.zeros(n)
    for k in K_star:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                g[j] += 2.0 * a[idx]
    g /= len(K_star)
    
    # Update and project
    a_new = a - lr * g
    a_new = project_simplex(a_new)
    return a_new, g

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

    start = time.monotonic()
    
    lr = 1.0 / n
    it = 0

    while time.monotonic() - start < time_budget:
        a_new, g = subgradient_step(a, lr)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-14:

            best_a = a_new.copy()
            last_improve = it
            # Increase LR slightly on improvement
            lr = min(lr * 1.01, 1.0 / np.sqrt(n))
        else:
            # Decrease LR on stagnation
            lr = lr * 0.999
        
        a = a_new
        it += 1

            # Reset
            a = best_a.copy()
            lr = 1.0 / n
            last_improve = it
    
    if verbose:
        print(f"    Subgrad: best={best_obj:.10f}, iters={it}")
    
    return best_a

Combines LP fixed-point iteration with adaptive line search (main workhorse),
projected subgradient descent, smoothed gradient descent with staircase tau,

from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def lp_solve(a, solver_list=None):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0. Returns (g, M, time) or (None, None, None)."""
    n = len(a)
    if n > 2500:
        return None, None, None

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]

    for solver in solver_list:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver, verbose=False)
            t1 = time.monotonic()
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0.0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value), (t1 - t0)
        except Exception:
            continue

    return None, None, None


def subgradient_step(a, lr):
    """One step of projected subgradient descent on max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    tol = max(1e-12, 1e-10 * max_val)
    K_star = np.where(conv >= max_val - tol)[0]

    g = np.zeros(n)
    for k in K_star:
        for j in range(n):
            idx = int(k) - j
            if 0 <= idx < n:
                g[j] += 2.0 * a[idx]
    g /= float(len(K_star))

    a_new = a - lr * g
    a_new = project_simplex(a_new)
    return a_new


def staircase_gd(a, time_budget, tau_levels=None, verbose=False):
    """Nesterov accelerated GD with staircase tau schedule."""
    n = len(a)
    a = np.maximum(a, 0.0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [10, 50, 200, 800, 2000]

    beta = 0.9
    lr = 2.0 / np.sqrt(n)

    for tau in tau_levels:
        v = np.zeros(n)
        it_level = 0

        while time.monotonic() - start < time_budget:
            if it_level >= 3000:
                break

            y = a + beta * v
            y = np.maximum(y, 0.0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0.0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            it_level += 1
            lr = max(1e-8, lr * 0.99995)

        if verbose:
            print(f"    tau={tau:5d}: best={best_obj:.10f}, iters={it_level}")

    return best_a


def lp_fixed_point(a, time_budget, verbose=False, max_rounds=500, seed=42):
    """LP fixed-point iteration with adaptive line search and perturbation."""
    n = len(a)
    a = np.maximum(a, 0.0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    t_current = 0.3
    stall_count = 0
    max_stall = 40
    total_perturbs = 0
    max_perturbs = 30

    for rnd in range(max_rounds):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        g_opt, M_val, solve_time = lp_solve(a)
        if g_opt is None:
            break

        # Line search: try decreasing t values
        best_t = None
        best_obj_try = best_obj
        best_a_try = best_a.copy()

        t_candidates = [1.0, 0.7, 0.5, round(t_current, 3), 0.3, 0.2, 0.1, 0.05]
        t_candidates = sorted(set(t_candidates), reverse=True)

        for t in t_candidates:
            if t <= 0 or t > 1.0:
                continue
            a_try = (1.0 - t) * a + t * g_opt
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_t = t
                best_a_try = a_try.copy()

        if best_t is not None and best_obj_try < best_obj - 1e-15:
            best_obj = best_obj_try
            best_a = best_a_try.copy()
            a = best_a
            t_current = best_t
            stall_count = 0
            if verbose and (rnd < 20 or rnd % 20 == 0):
                print(f"    LP-fp it{rnd:3d}: obj={best_obj:.10f}, t={best_t:.3f}, M={M_val:.6f}, solve={solve_time:.3f}s")
        else:
            stall_count += 1
            if stall_count >= max_stall and total_perturbs < max_perturbs:
                total_perturbs += 1
                # Spectral perturbation
                freqs = rng.integers(1, 10, size=rng.integers(2, 5))
                phases = rng.uniform(0, 2 * np.pi, size=len(freqs))
                x = np.arange(n)
                pert = np.zeros(n)
                for f, p in zip(freqs, phases):
                    pert += 0.05 * np.sin(2.0 * np.pi * f * x / n + p)
                a_pert = a + pert
                a_pert = np.maximum(a_pert, 0.0)
                s_pert = float(np.sum(a_pert))
                if s_pert > 1e-10:
                    a = a_pert / s_pert
                stall_count = 0
            elif stall_count > max_stall * 3:
                break

    if verbose:
        print(f"    LP-fp done: obj={best_obj:.10f}, iters={rnd+1}, perturbs={total_perturbs}")

    return best_a


def subgradient_descent(a, time_budget, verbose=False, seed=42):
    """Projected subgradient descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0.0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    lr = 2.0 / n
    it = 0
    patience = 800
    last_improve = 0

    while time.monotonic() - start < time_budget:
        a_new = subgradient_step(a, lr)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            last_improve = it
            lr = min(lr * 1.005, 2.0 / np.sqrt(n))
        else:
            lr = lr * 0.998

        a = a_new
        it += 1

        if it - last_improve > patience:
            a = best_a.copy()
            lr = 2.0 / n
            last_improve = it
            # Small perturbation
            noise = rng.normal(0, 0.005, n)
            a = a + noise
            a = np.maximum(a, 0.0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a

    if verbose:
        print(f"    Subgrad: best={best_obj:.10f}, iters={it}")

    return best_a


def make_init(n, mode, rng=None):
    """Create diverse initial sequences of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(0, 1, n)
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.001)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1.0 - x ** 2, 0.0001))
        return a / a.sum()
    elif mode == 3:
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 4:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, np.pi, n)
        a = np.maximum(0.5 + 0.5 * np.cos(x), 0.001)
        return a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [2, 3, 5, 7]:
            a += 0.3 * np.sin(2.0 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:
        a = np.exp(-np.abs(np.arange(n) - n / 2.0) / (n / 4.0))
        return a / a.sum()
    elif mode == 8:
        x = np.linspace(0, 1, n)
        a = np.where((x > 0.15) & (x < 0.85), 1.0,
                     np.maximum(0.3 - np.abs(x - 0.5), 0.01))
        return a / a.sum()
    elif mode == 9:
        x = np.linspace(0.01, 0.99, n)
        a = 1.0 / np.sqrt(x * (1.0 - x))
        return a / a.sum()
    elif mode == 10:
        x = np.linspace(0.01, 0.99, n)
        a = x * (1.0 - x)
        return a / a.sum()
    elif mode == 11:
        # Multi-scale: combination of different frequency sines
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [3, 5, 8, 13]:
            a += 0.4 * np.abs(np.sin(2.0 * np.pi * f * x + rng.uniform(0, 2 * np.pi)))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 12:
        # Arcsine with offset
        x = np.linspace(0.05, 0.95, n)
        a = 1.0 / (np.sqrt(x * (1.0 - x)) + 0.1)
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP fixed-point with line search as workhorse,
    complemented by subgradient descent, smoothed GD, spectral perturbation,
    and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        parent = None

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.10f}")

    n_current = len(best_seq)
    print(f"Starting: n={n_current}, obj={best_score:.10f}, budget={budget_s}s")

    # === Phase 1: Diverse initialization screening (45s) ===
    print(f"\n=== Phase 1: Diverse Init Screening ===")
    init_modes = [0, 2, 3, 6, 8, 9, 10, 11, 12]
    for mode in init_modes:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break
        a_init = make_init(n_current, mode, rng)
        alloc = min(remaining * 0.04, 12.0)
        if alloc < 4:
            break
        a_screen = staircase_gd(a_init, time_budget=alloc,
                                 tau_levels=[20, 100, 500], verbose=False)
        obj_screen = compute_obj(a_screen)
        if obj_screen < best_score:
            best_score = obj_screen
            best_seq = a_screen.copy()
            print(f"  Mode {mode:2d}: obj={best_score:.10f} **")
        else:
            print(f"  Mode {mode:2d}: obj={obj_screen:.10f}")

    print(f"  After screening: best={best_score:.10f}")

    # === Phase 2: LP Fixed-Point - main workhorse (40% of remaining) ===
    print(f"\n=== Phase 2: LP Fixed-Point (Main) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 500.0)
    if p2_budget > 50 and n_current <= 2000:
        a2 = lp_fixed_point(best_seq, time_budget=p2_budget, verbose=True, seed=42)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  LP-FP: obj={best_score:.10f} **")
        else:
            print(f"  LP-FP: obj={obj2:.10f}")

    # === Phase 3: Subgradient refinement (10% of remaining) ===
    print(f"\n=== Phase 3: Subgradient Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.10, 120.0)
    if p3_budget > 20:
        a3 = subgradient_descent(best_seq, time_budget=p3_budget, verbose=True, seed=123)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Subgrad: obj={best_score:.10f} **")
        else:
            print(f"  Subgrad: obj={obj3:.10f}")

    # === Phase 4: Multi-n exploration (20% of remaining) ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 200),
        n_current,
        n_current + 100,
        n_current + 200,
        n_current + 400,
        600, 800, 1000, 1200, 1500, 2000
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000]
    n_options.sort(key=lambda x: abs(x - n_current))
    n_options = n_options[:6]

    print(f"  n candidates: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample(best_seq, n_try)

        # LP polish
        if n_try <= 1500:
            alloc_lp = min(remaining * 0.25, 50.0)
            if alloc_lp > 15:
                a_try = lp_fixed_point(a_try, time_budget=alloc_lp, verbose=False, seed=n_try)

        # GD finish
        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.15, 25.0)
        if alloc_gd > 10:
            a_try = staircase_gd(a_try, time_budget=alloc_gd,
                                  tau_levels=[50, 200, 800], verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 5: Second LP round (20% of remaining) ===

        # Use different seed for different perturbation pattern

        obj5 = compute_obj(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  LP2: obj={best_score:.10f} **")
        else:
            print(f"  LP2: obj={obj5:.10f}")

    # === Phase 6: Final GD + subgradient combo (15% of remaining) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    p6_budget = min(remaining * 0.3, 120.0)
    if p6_budget > 20:
        # High-tau GD
        alloc1 = p6_budget * 0.5
        a6 = staircase_gd(best_seq, time_budget=alloc1,
                           tau_levels=[100, 500, 1500, 3000], verbose=False)
        obj6 = compute_obj(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Final GD: obj={best_score:.10f} **")
        else:
            print(f"  Final GD: obj={obj6:.10f}")

        # Subgradient
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.3, p6_budget * 0.5)
        if alloc2 > 10:
            a6b = subgradient_descent(best_seq, time_budget=alloc2, verbose=False, seed=777)
            obj6b = compute_obj(a6b)
            if obj6b < best_score:
                best_score = obj6b
                best_seq = a6b.copy()
                print(f"  Final Subgrad: obj={best_score:.10f} **")
            else:
                print(f"  Final Subgrad: obj={obj6b:.10f}")

    # === Phase 7: Final LP (remaining) ===
    print(f"\n=== Phase 7: Final LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 30 and len(best_seq) <= 2000:
        a7 = lp_fixed_point(best_seq, time_budget=remaining * 0.85, verbose=True, seed=3141)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7

            print(f"  Final LP: obj={best_score:.10f} **")
        else:
            print(f"  Final LP: obj={obj7:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
