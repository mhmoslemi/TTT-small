def compute_conv(a):

def build_conv_matrix(a):

def solve_lp(a, time_limit=10):

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, time_limit=5.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None


def parametric_seq(n, coeffs):
    """Generate sequence from cosine coefficients.
    coeffs: flat array [f1, r1, f2, r2, ...] where a[i] = 1 + sum r_m*cos(2*pi*f_m*i/n)
    """
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(0, len(coeffs) - 1, 2):
        f = int(coeffs[k])
        r = coeffs[k + 1]
        if f > 0 and abs(r) > 1e-6:
            a += r * np.cos(2.0 * np.pi * f * i / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def make_symmetric(a):
    """Make sequence symmetric: a[i] = a[n-1-i]."""
    return (a + a[::-1]) / 2.0


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                print(f"  Parametric n={n_try}: obj={best_score:.8f} **")
            # Quick LP polish on best parametric
            if best_for_n_obj < 1.6:
                alloc = min(15.0, (deadline - time.monotonic()) * 0.02)
                if alloc > 5:
                    a_pol = lp_iterate(best_for_n, time_budget=alloc, verbose=False)
                    obj_pol = compute_obj(a_pol)
                    if obj_pol < best_score:
                        best_score = obj_pol
                        best_seq = a_pol.copy()
                        n_best = n_try
                        print(f"  After LP n={n_try}: obj={best_score:.8f} **")

    # === Phase 2: LP Fictitious Play (300s) ===
    print(f"\n=== Phase 2: LP Fictitious Play (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(300.0, remaining * 0.28)
    a_lp = lp_iterate(best_seq, time_budget=p2_budget, verbose=True)
    obj_lp = compute_obj(a_lp)
    if obj_lp < best_score:
        best_score = obj_lp
        best_seq = a_lp.copy()
        print(f"  Phase2 LP: obj={best_score:.8f} **")
    else:
        print(f"  Phase2 LP: obj={obj_lp:.8f}")

    # === Phase 3: Adam Log-Param Optimization (180s) ===
    print(f"\n=== Phase 3: Adam Log-Param ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(180.0, remaining * 0.2)
    a_adam = adam_log_optimize(best_seq, time_budget=p3_budget,
                                tau_start=200, tau_end=10000, verbose=True)
    obj_adam = compute_obj(a_adam)
    if obj_adam < best_score:
        best_score = obj_adam
        best_seq = a_adam.copy()
        print(f"  Phase3 Adam: obj={best_score:.8f} **")
    else:
        print(f"  Phase3 Adam: obj={obj_adam:.8f}")

    # === Phase 4: Nesterov GD (120s) ===
    print(f"\n=== Phase 4: Nesterov GD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(120.0, remaining * 0.15)
    a_nest = nesterov_gd(best_seq, time_budget=p4_budget,
                          tau_start=300, tau_end=12000, verbose=True)
    obj_nest = compute_obj(a_nest)
    if obj_nest < best_score:
        best_score = obj_nest
        best_seq = a_nest.copy()
        print(f"  Phase4 Nesterov: obj={best_score:.8f} **")
    else:
        print(f"  Phase4 Nesterov: obj={obj_nest:.8f}")

    # === Phase 5: Multi-n Exploration (350s) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(350.0, remaining * 0.35)
    n_current = len(best_seq)

    n_options = sorted(set([
        max(300, n_current - 300), n_current - 150, n_current - 50,
        n_current + 50, n_current + 150, n_current + 300,
        n_current + 600, n_current + 1000, n_current + 2000,
        400, 700, 1000, 1500, 2500, 3500, 4500
    ]))
    n_options = [n for n in n_options if 300 <= n <= 5000 and abs(n - n_current) > 30]
    n_options = n_options[:12]
    print(f"  Trying n: {n_options}")

    per_n = p5_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(per_n, remaining - 30)
        if alloc < 25:
            break

        a_try = resample_seq(best_seq, n_try)

        # LP
        lp_alloc = alloc * 0.45
        a_try = lp_iterate(a_try, time_budget=lp_alloc, verbose=False)

        # Adam
        remaining = deadline - time.monotonic()
        adam_alloc = min(alloc * 0.35, remaining - 20)
        if adam_alloc > 10:
            a_try = adam_log_optimize(a_try, time_budget=adam_alloc,
                                       tau_start=100, tau_end=5000, verbose=False)

        # Nesterov
        remaining = deadline - time.monotonic()
        nest_alloc = min(alloc * 0.2, remaining - 15)
        if nest_alloc > 8:
            a_try = nesterov_gd(a_try, time_budget=nest_alloc,
                                 tau_start=200, tau_end=8000, verbose=False)

        # Symmetrize and check
        a_sym = make_symmetric(a_try)
        a_sym = a_sym / a_sym.sum()
        obj_sym = compute_obj(a_sym)
        obj_try = compute_obj(a_try)
        if obj_sym < obj_try:
            a_try = a_sym
            obj_try = obj_sym

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Support Shrink + Diverse Restarts (150s) ===
    print(f"\n=== Phase 6: Shrink + Restarts ===")
    remaining = deadline - time.monotonic()
    p6_budget = min(150.0, remaining * 0.18)

    # Support shrink
    if p6_budget > 20:
        a_shrunk = support_shrink(best_seq, time_budget=p6_budget * 0.3, verbose=True)
        obj_shrunk = compute_obj(a_shrunk)
        if obj_shrunk < best_score:
            best_score = obj_shrunk
            best_seq = a_shrunk.copy()
            print(f"  Shrink: obj={best_score:.8f} **")

    # Diverse restarts
    remaining = deadline - time.monotonic()
    n_rest_budget = min(p6_budget * 0.7, remaining - 30)
    n_cur = len(best_seq)
    n_restarts = max(3, int(n_rest_budget / 30))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break
        alloc = min(30, remaining - 15)
        if alloc < 15:
            break

        mode = r % 8
        a_r = diverse_init(n_cur, rng, mode=mode)
        # Blend with current best
        blend = 0.2 + 0.1 * (r % 5)
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        a_r = lp_iterate(a_r, time_budget=alloc * 0.6, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_r = adam_log_optimize(a_r, time_budget=min(alloc * 0.4, remaining - 10),
                                     tau_start=100, tau_end=3000, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s")

    # Round 1: LP
    if remaining > 40:
        alloc = min(remaining * 0.25, 100.0)
        a_f = lp_iterate(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f
            print(f"  Final LP: {best_score:.8f} **")

    # Round 2: Adam with high tau
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 80.0)
        a_f = adam_log_optimize(best_seq, time_budget=alloc,
                                 tau_start=500, tau_end=15000, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f
            print(f"  Final Adam: {best_score:.8f} **")

    # Round 3: Nesterov
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.25, 60.0)
        a_f = nesterov_gd(best_seq, time_budget=alloc,
                           tau_start=500, tau_end=15000, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f
            print(f"  Final Nesterov: {best_score:.8f} **")

    # Round 4: Support shrink
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 40.0)
        a_f = support_shrink(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f
            print(f"  Final Shrink: {best_score:.8f} **")

    # Round 5: One more LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 50.0)
        a_f = lp_iterate(best_seq, time_budget=alloc, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f
            print(f"  Ultra LP: {best_score:.8f} **")

    # Final Adam
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 30.0)
        a_f = adam_log_optimize(best_seq, time_budget=alloc,
                                 tau_start=1000, tau_end=20000, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f
            print(f"  Ultra Adam: {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    print(f"\n=== FINAL: n={len(result)}, internal={best_score:.8f}, evaluated={val:.8f}, "
          f"time={time.monotonic()-start_time:.1f}s ===")

    if val < np.inf:
        return result
    else:
        return [1.0] * 100
