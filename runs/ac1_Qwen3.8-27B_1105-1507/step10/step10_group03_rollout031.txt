"""
Convolution Sculpting optimizer: minimizes 2*n*max(conv(a,a))/(sum(a))^2
by iteratively reducing peaks in the self-convolution using FFT-based
cross-correlation gradients, adaptive target levels, momentum, and
frequency-domain initialization. Combined with LP polishing and multi-n search.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft, rfft, irfft
from scipy.signal import fftconvolve


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return float(2.0 * n * np.max(conv))


def next_pow2(x):
    """Next power of 2 >= x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def sculpt_optimize(a, time_budget, verbose=True, seed=42):
    """
    Main sculpting optimizer: iteratively reduces convolution peaks.
    Uses FFT-based cross-correlation gradients with adaptive target.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    L = 2 * n - 1
    N_fft = next_pow2(3 * n + 2)  # for cross-correlation
    N_conv = next_pow2(L)  # for convolution

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Parameters
    lr = 0.5
    momentum = 0.9
    v = np.zeros(n)
    target_ratio = 1.0  # target = target_ratio * current_max
    flat_level = 1.0 / L

    it = 0
    stagnant = 0
    best_stagnant_reset = 0

    while time.monotonic() - start < time_budget:
        # Compute convolution
        fa = rfft(a, N_conv)
        c = irfft(fa * fa, N_conv)[:L]
        c_max = float(np.max(c))

        # Adaptive target: between flat level and current max
        target = max(flat_level * 1.05, c_max * target_ratio)

        # Compute excess (smooth version)
        excess = np.maximum(c - target, 0.0)

        if np.sum(excess) < 1e-15:
            # No excess: try lowering target
            target_ratio = max(0.85, target_ratio * 0.98)
            stagnant += 1
        else:
            # Compute gradient via cross-correlation
            # grad[i] = sum_k excess[k] * a[k-i] = (a * excess_corr)[i]
            A_fft = fft(a, N_fft)
            E_fft = fft(excess, N_fft)
            grad = ifft(np.conj(A_fft) * E_fft).real[:n]

            # Also add a small variance-reducing component
            # Push toward making c more uniform
            c_mean = float(np.sum(c)) / L
            c_dev = c - c_mean
            # Gradient of sum(c_dev^2) w.r.t. a
            CD_fft = fft(c_dev, N_fft)
            grad_var = ifft(np.conj(A_fft) * CD_fft).real[:n]
            grad += 0.1 * grad_var

            # Clip gradient to avoid huge steps
            g_norm = float(np.linalg.norm(grad))
            if g_norm > 1.0:
                grad = grad / g_norm

            # Momentum update
            v = momentum * v - lr * grad
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
                it += 1
                continue
            a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                stagnant = 0
                if verbose and it % 2000 == 0:
                    print(f"    sculpt it={it}: obj={best_obj:.10f}, cmax={best_obj/(2*n):.8f}")
            else:
                stagnant += 1

            a = a_new
            it += 1

            # Learning rate decay
            lr = max(1e-5, lr * 0.99999)

            # Gradually tighten target
            if c_max > flat_level * 2:
                target_ratio = max(0.9, target_ratio * 0.999)

        # Restart logic
        if stagnant > 3000:
            if stagnant > 12000:
                # Full restart with perturbation
                a = best_a + 0.05 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                v = np.zeros(n)
                lr = 0.5
                target_ratio = 1.0
            else:
                # Softer restart
                a = best_a + 0.02 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                v = np.zeros(n) * 0.5
                lr = max(lr, 0.1)
            stagnant = 0
            it += 1

    if verbose:
        print(f"    sculpt done: it={it}, best_obj={best_obj:.10f}")
    return best_a


def freq_domain_init(n, rng, num_attempts=20):
    """
    Generate initializations with flat-spectrum property.
    Set |A(f)|^2 to be as flat as possible, randomize phase, IFFT, project.
    """
    L = 2 * n - 1
    N = next_pow2(L)

    best_a = np.ones(n) / n
    best_obj = compute_obj(best_a)

    for _ in range(num_attempts):
        # Target power spectrum: flat except DC
        # |A(0)|^2 = 1 (since sum(a)=1), others flat
        # Total power = N * sum(a_i^2) >= N/n (by Parseval + Cauchy-Schwarz)
        # So non-DC power >= N/n - 1
        min_non_dc_power = max(N / n - 1.0, 0.1)
        per_bin_power = min_non_dc_power / (N - 1)

        # Build target power spectrum
        power = np.full(N, per_bin_power)
        power[0] = 1.0  # DC

        # Random phase
        phases = rng.uniform(0, 2 * np.pi, N)
        # Enforce Hermitian symmetry for real output
        phases[0] = 0
        if N % 2 == 0:
            phases[N // 2] = 0
        for k in range(1, N // 2 + 1):
            if k < N - k:
                phases[N - k] = -phases[k]

        A_target = np.sqrt(power) * np.exp(1j * phases)
        # Enforce Hermitian symmetry
        A_target[0] = np.sqrt(power[0])
        if N % 2 == 0:
            A_target[N // 2] = np.sqrt(power[N // 2])
        for k in range(1, N // 2 + 1):
            A_target[N - k] = np.conj(A_target[k])

        a_candidate = np.real(ifft(A_target))[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = float(np.sum(a_candidate))
        if s < 0.01:
            continue
        a_candidate = a_candidate / s
        obj = compute_obj(a_candidate)
        if obj < best_obj:
            best_obj = obj
            best_a = a_candidate.copy()

    return best_a


def lp_polish_iteration(a, solver_list=None):
    """Single LP polishing step."""
    n = len(a)
    if n > 2000:
        return None, None

    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_mat[j:j + n, j] = a

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A_mat @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget, verbose=False, seed=42, max_iters=30):
    """Iterative LP polishing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    for _ in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        g_lp, M_lp = lp_polish_iteration(a)
        if g_lp is None:
            break

        t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        best_t_obj = best_obj
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            if verbose:
                print(f"    LP: obj={best_obj:.10f}")
        else:
            stall += 1
            a = a + 0.01 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            if stall > 4:
                break

    return best_a


def local_search(a, time_budget, verbose=False, seed=42):
    """Random local search with targeted peak reduction."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    L = 2 * n - 1
    N_conv = next_pow2(L)

    while time.monotonic() - start < time_budget:
        # Compute current convolution
        fa = rfft(a, N_conv)
        c = irfft(fa * fa, N_conv)[:L]
        c_max = float(np.max(c))
        peak_indices = np.where(c > c_max * 0.95)[0]

        # Strategy: reduce a at positions that contribute to peak lags
        # For peak at lag k, the contribution comes from a[j]*a[k-j]
        # Reducing a[j] where j is in [0, n-1] and k-j is in [0, n-1] helps
        for _ in range(50):
            if time.monotonic() - start > time_budget:
                break

            # Pick a random peak lag
            k = int(rng.choice(peak_indices)) if len(peak_indices) > 0 else n - 1

            # Find indices j where both j and k-j are valid
            valid_j = np.arange(max(0, k - n + 1), min(n, k + 1))
            if len(valid_j) == 0:
                continue

            # Reduce a at these positions (weighted by contribution)
            contributions = a[valid_j] * a[k - valid_j]
            total_contrib = float(np.sum(contributions))
            if total_contrib < 1e-15:
                continue

            # Randomly pick which j to reduce
            probs = contributions / total_contrib
            j_pick = int(rng.choice(valid_j, p=probs))

            # Reduce a[j_pick] slightly
            a_try = a.copy()
            reduction = rng.uniform(0.001, 0.01) * a[j_pick]
            a_try[j_pick] -= reduction
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)

            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                # Recompute peaks periodically
                if rng.random() < 0.1:
                    fa = rfft(a, N_conv)
                    c = irfft(fa * fa, N_conv)[:L]
                    c_max = float(np.max(c))
                    peak_indices = np.where(c > c_max * 0.95)[0]

    if verbose:
        print(f"    local_search: obj={best_obj:.10f}")
    return best_a


def make_init(n, mode, rng=None):
    """Create diverse initial sequences."""
    if rng is None:
        rng = np.random.default_rng(0)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.0001))
        return a / a.sum()
    elif mode == 3:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 4:
        # One-sided concentration
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * x)
        return a / a.sum()
    elif mode == 5:
        # Two-sided (high at both ends)
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.5) ** 2 / 0.01)
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 6:
        # Sine-based
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.sin(4 * np.pi * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:
        # Geometric
        r = 1 - 1.0 / n
        a = r ** np.arange(n)
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
