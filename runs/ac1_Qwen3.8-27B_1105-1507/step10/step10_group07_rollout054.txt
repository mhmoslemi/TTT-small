    n = len(a_init)
    a = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    constraints = [cp.sum(a) == 1]
    
    for k in active_indices:
        Q_k = np.zeros((n, n))
        for i in range(max(0, k-n+1), min(n, k+1)):
            j = k - i
            if 0 <= j < n:
                Q_k[i, j] = 1.0
        # Make symmetric (it already is since i+j=k iff j+i=k)
        constraints.append(a @ Q_k @ a <= M)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve(solver=cp.MOSEK, verbose=False)
    
    if a.value is not None:
        return a.value, M.value
    return None, None

import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear auto-convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_conv_gradient(a, k):
    """Compute gradient of (a*a)[k] with respect to a.
    (a*a)[k] = sum_i a[i]*a[k-i], grad_j = a[k-j] (with bounds).
    """
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        i = k - j
        if 0 <= i < n:
            g[j] = a[i]
    return g


def active_set_tr_sscp(a, time_budget=100.0, K_init=15, r_init=0.25,
                       verbose=False, seed=42):
    """
    Trust region SOCP with active-set constraint selection.
    Only constrains the top K peaks, updating the active set adaptively.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    r = r_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        # Compute current convolution and identify active peaks
        conv_cur = compute_conv(a)
        max_conv = float(np.max(conv_cur))
        
        # Identify top K active peaks
        sort_idx = np.argsort(conv_cur)[::-1]
        # Use peaks that are within 95% of max
        threshold = max_conv * 0.95
        active_idx = [k for k in sort_idx[:min(K_init, len(sort_idx))] if conv_cur[k] >= threshold]
        if len(active_idx) == 0:
            active_idx = [sort_idx[0]]
        active_idx = active_idx[:K_init]

        # Build SOCP: minimize M
        # s.t. conv_cur[k] + 2*(a*delta)[k] <= M for k in active_idx
        #      ||delta||_2 <= r
        #      sum(a+delta) = 1
        #      a+delta >= 0

        delta = cp.Variable(n)
        M = cp.Variable()
        a_new = a + delta

        constraints = [
            cp.sum(a_new) == 1,
            a_new >= 0,
            cp.norm(delta, 2) <= r,
        ]

        for k in active_idx:
            # (a*delta)[k] = sum_j a[j]*delta[k-j]
            # Build the linear form
            coeff = np.zeros(n)
            for j in range(n):
                i = k - j
                if 0 <= i < n:
                    coeff[j] = a[i]
            # Constraint: conv_cur[k] + 2 * coeff @ delta <= M
            constraints.append(conv_cur[k] + 2.0 * (coeff @ delta) <= M)

        prob = cp.Problem(cp.Minimize(M), constraints)

        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    delta_val = np.asarray(delta.value).flatten()
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            # Fallback: steepest descent on the peak
            k_peak = int(np.argmax(conv_cur))
            grad = compute_conv_gradient(a, k_peak)
            # Project gradient to tangent space (sum=0)
            grad = grad - np.mean(grad)
            delta_val = -grad / (np.linalg.norm(grad) + 1e-15) * r

        # Ensure feasible direction
        norm_d = np.linalg.norm(delta_val)
        if norm_d > r:
            delta_val = delta_val * (r / (norm_d + 1e-15))

        # Line search
        best_alpha = 0.0
        best_alpha_obj = best_obj
        # Adaptive alpha set
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]
        for t_alpha in alphas:
            a_try = a + t_alpha * delta_val
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj - 1e-14:
                best_alpha_obj = obj_try
                best_alpha = t_alpha

        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-14:
            a = a + best_alpha * delta_val
            a = np.maximum(a, 0)
            s_n = a.sum()
            if s_n > 1e-10:
                a = a / s_n
            best_obj = best_alpha_obj
            best_a = a.copy()
            no_improve = 0
            r = min(r * 1.1, 1.5)
        else:
            no_improve += 1
            r *= 0.6
            if no_improve >= 4:
                if r < 0.05:
                    # Restart with perturbation
                    scale = 0.005 + 0.003 * min(no_improve, 10)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    s_p = a.sum()
                    if s_p > 1e-10:
                        a = a / s_p
                    r = r_init
                    no_improve = 0
                    if it > 100:
                        break
                else:
                    r = max(r * 0.5, 0.05)

        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    AS-TR it{it}: obj={best_obj:.8f}, r={r:.4f}, K={len(active_idx)}, t={elapsed:.0f}s")

    return best_a


def spectral_init(n, shape='raised_cosine', scale=1.0, rng=None):
    """Create a sequence with shaped spectrum for good initial peak structure."""
    if rng is None:
        rng = np.random.default_rng(0)

    if shape == 'uniform':
        a = np.ones(n)
    elif shape == 'raised_cosine':
        a = np.sin(np.pi * np.arange(n) / n) ** 2
    elif shape == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif shape == 'gaussian':
        x = np.arange(n) / n - 0.5
        a = np.exp(-x ** 2 / (0.05 * scale))
    elif shape == 'exponential':
        x = np.arange(n) / n
        a = np.exp(-x * 2.0 * scale)
    elif shape == 'bell':
        x = np.arange(n) / n - 0.5
        a = np.exp(-x ** 2 / (2 * 0.15 ** 2 * scale))
    elif shape == 'double_bell':
        x = np.arange(n) / n - 0.5
        a = np.exp(-(x - 0.2) ** 2 / 0.02) + np.exp(-(x + 0.2) ** 2 / 0.02)
    elif shape == 'spectral_flat':
        # Try to make |FFT(a)|^2 as flat as possible
        a = np.ones(n)
        # Add small random perturbation in frequency domain
        N = 1
        while N < 2 * n - 1:
            N <<= 1
        A = rfft(a, N)
        A[1:N // 2] *= (1 + 0.1 * rng.standard_normal(N // 2 - 1))
        A[N // 2] *= (1 + 0.05 * rng.standard_normal())
        a = irfft(A, N)[:n]
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a


def structured_perturb(a, rng, mode='sinusoidal'):
    """Apply a smooth structured perturbation."""
    n = len(a)
    if mode == 'sinusoidal':
        freq = rng.uniform(1, min(8, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, 0.08)
        pert = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif mode == 'polynomial':
        deg = rng.integers(2, 6)
        x = np.linspace(-1, 1, n)
        coeffs = rng.normal(0, 0.02, deg + 1)
        pert = np.polyval(coeffs, x)
    elif mode == 'multi_sinusoidal':
        pert = np.zeros(n)
        for _ in range(rng.integers(2, 5)):
            freq = rng.uniform(1, min(6, n // 6))
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.005, 0.04)
            pert += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif mode == 'envelope':
        # Perturb the envelope
        x = np.linspace(0, 1, n)
        envelope = 1 + rng.uniform(-0.1, 0.1) * (x - 0.5) ** 2 + \
                   rng.uniform(-0.05, 0.05) * np.sin(np.pi * x)
        pert = (envelope - 1) * a * n  # Relative perturbation
    else:
        pert = rng.normal(0, 0.01, n)

    return a * np.exp(pert)


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def smoothed_gd_refine(a, time_budget=60.0, tau_init=10.0, tau_end=200.0):
    """Smoothed gradient descent refinement."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 3.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr = max(1e-12, lr * 0.9997)

    return best_a


def pairwise_swap_refine(a, time_budget=60.0):
    """Try pairwise swaps to reduce peak auto-convolution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    while time.monotonic() - start < time_budget:
        # Pick a random pair and try to optimize their values
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Try transferring mass from i to j and vice versa
        current_val = best_obj
        found = False
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
            # Move frac of a[i] to a[j]
            a_try = best_a.copy()
            transfer = a_try[i] * frac
            a_try[i] -= transfer
            a_try[j] += transfer
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < current_val - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                current_val = obj_try
                found = True
            # Also try reverse
            a_try2 = best_a.copy()
            transfer2 = a_try2[j] * frac
            a_try2[j] -= transfer2
            a_try2[i] += transfer2
            a_try2 = np.maximum(a_try2, 0)
            s_try2 = a_try2.sum()
            if s_try2 < 1e-10:
                continue
            a_try2 = a_try2 / s_try2
            obj_try2 = compute_obj(a_try2)
            if obj_try2 < current_val - 1e-14:
                best_obj = obj_try2
                best_a = a_try2.copy()
                current_val = obj_try2
                found = True

        if not found and rng.random() < 0.01:
            # Occasionally do a larger perturbation
            scale = rng.uniform(0.01, 0.05)
            a_try = best_a * np.exp(rng.normal(0, scale, n))
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()

    return best_a

    deadline = start_time + float(budget_s) - 25.0

            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                n_parent = parent.size
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={n_parent}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = spectral_init(n_parent, 'raised_cosine')
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_parent}): obj={best_score:.8f}")

    # === Phase 1: Active-set TR-SOCP on parent ===
    print(f"\n=== Phase 1: Active-Set TR-SOCP (parent) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 200.0)
    if p1_budget > 30:
        a_opt = active_set_tr_sscp(best_seq, time_budget=p1_budget,
                                   K_init=20, r_init=0.25, verbose=True, seed=42)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase1: obj={best_score:.8f} **")

            print(f"  Phase1: obj={obj_opt:.8f} (no improvement)")

    # === Phase 2: Spectral initialization + optimize ===
    print(f"\n=== Phase 2: Spectral Initialization ===")
    remaining = deadline - time.monotonic()
    shapes = ['raised_cosine', 'hann', 'bell', 'double_bell', 'gaussian']
    n_try = len(best_seq)

    for shape in shapes[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 40:

        a_init = spectral_init(n_try, shape, rng=rng)

                                        tau_init=8.0, tau_end=150.0)
        # TR-SOCP polish
        remaining = deadline - time.monotonic()
        alloc_tr = min(remaining * 0.12, 60.0)
        if alloc_tr > 10:
            a_init = active_set_tr_sscp(a_init, time_budget=alloc_tr,
                                        K_init=15, r_init=0.3, seed=100 + hash(shape) % 1000)

        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Shape '{shape}': obj={best_score:.8f} **")
        else:
            print(f"  Shape '{shape}': obj={obj_init:.8f}")

    # === Phase 3: Structured perturbation + optimize ===
    print(f"\n=== Phase 3: Structured Perturbation ===")
    remaining = deadline - time.monotonic()
    modes = ['sinusoidal', 'polynomial', 'multi_sinusoidal', 'envelope']

    for mode in modes:

    # === Phase 5: Intensive multi-start ===

        else:
            a_seed = structured_perturb(best_seq, rng, mode='multi_sinusoidal')
            a_seed = a_seed * np.exp(rng.normal(0, scale * 0.5, len(best_seq)))

        a_seed = np.maximum(a_seed, 0)
        s_seed = a_seed.sum()
        if s_seed < 1e-10:
            continue
        a_seed = a_seed / s_seed

        alloc = min(remaining * 0.2, 90.0)
        if alloc > 12:
            a_seed = active_set_tr_sscp(a_seed, time_budget=alloc,
                                        K_init=20, r_init=0.2, seed=1000 + restart * 7)

            if obj_seed < best_score:

                best_seq = a_seed.copy()
                print(f"  Restart{restart}: obj={best_score:.8f} **")
            else:
                print(f"  Restart{restart}: obj={obj_seed:.8f}")

    # === Phase 6: Final intensive polishing ===
    print(f"\n=== Phase 6: Final Intensive Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Long TR-SOCP run
        alloc = min(remaining * 0.5, 120.0)
        a_final = active_set_tr_sscp(best_seq, time_budget=alloc,
                                     K_init=25, r_init=0.15, verbose=True, seed=9999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final TR: obj={best_score:.8f} **")
        else:
            print(f"  Final TR: obj={obj_final:.8f}")

    # Pairwise swap refinement
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 40.0)
        a_swap = pairwise_swap_refine(best_seq, time_budget=alloc)
        obj_swap = compute_obj(a_swap)
        if obj_swap < best_score:
            best_score = obj_swap
            best_seq = a_swap.copy()
            print(f"  Swap refine: obj={best_score:.8f} **")
        else:
            print(f"  Swap refine: obj={obj_swap:.8f}")

    # === Phase 7: One more TR-SOCP pass on best ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 50.0)
