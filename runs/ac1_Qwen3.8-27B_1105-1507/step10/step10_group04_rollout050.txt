"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based smoothed gradient descent with Nesterov acceleration, adaptive tau,
LP-based polishing, coordinate descent, and multi-length exploration.
Key improvements: much faster inner loop via FFT gradients (O(n log n) per step),
enabling 100x more iterations than LP-only approaches.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def next_pow2(x):
    """Next power of 2 >= x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def compute_conv_fft(a):
    """Linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    N = next_pow2(2 * n - 1)
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_grad_smoothed(a, tau):
    """
    Compute gradient of 2n * (1/tau) * log(sum exp(tau*(a*a)[k])) w.r.t. a.
    Assumes a is normalized (sum=1).
    Returns (obj_value, gradient_vector).
    """
    n = len(a)
    conv = compute_conv_fft(a)
    max_conv = float(np.max(conv))
    
    # Compute softmax weights p
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # Smoothed objective value
    obj_val = 2.0 * n * (max_conv + np.log(sum_exp) / tau)
    
    # Gradient: 4n * (p cross-correlate a)[j]
    # (p ⊛ a)[j] = sum_k p[k] * a[k-j] for j=0,...,n-1
    N = next_pow2(3 * n)  # need N >= len(p) + n - 1 = (2n-1) + n - 1 = 3n-2
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real[:n]
    
    grad = 4.0 * n * cross
    
    # Project onto tangent space of simplex (subtract mean)
    grad = grad - np.mean(grad)
    
    return obj_val, grad

    FFT-based smoothed gradient descent with Nesterov acceleration
    and backtracking line search.
    """

    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0

        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        tau = tau_start + (tau_end - tau_start) * progress * progress  # quadratic schedule

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        obj_smooth, grad = compute_grad_smoothed(y, tau)

        g_norm = float(np.linalg.norm(grad))

def solve_lp(a, tol=1e-10):
    """
    Solve LP: min M s.t. (a_norm * g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build convolution matrix A (2n-1) x n where A[k,j] = a_norm[k-j] if valid
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k,j] = a_norm[k-j] for k-j in [0, n-1]
        # k ranges from j to j+n-1
        A[j:j+n, j] = a_norm
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver, **({'MSDK_PARAMS': {'MSDK_IPAR_INTPNT_CO_OPNORM': 0}} if solver == cp.MOSEK else {}))
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M_var.value)
        except Exception:
            continue
    
    return None, None


def lp_polish(a, time_budget, verbose=True):
    """LP-based alternating polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Solve LP for g given a
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        
        # Line search over mixing parameter
        t_candidates = np.concatenate([
            np.linspace(0.05, 1.0, 15),
            [max(0.05, last_t * 0.5), max(0.05, last_t * 0.7),
             last_t, min(1.0, last_t * 1.2), min(1.0, last_t * 1.5)]
        ])
        t_candidates = np.unique(np.clip(t_candidates, 0.01, 1.0))
        
        best_t_obj = best_obj
        best_t = last_t
        best_a_local = best_a.copy()
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-13:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t
            stall = 0
        else:
            stall += 1
            if stall > 3:
                break
    
    return best_a


def coordinate_descent(a, n_passes=50, verbose=False):
    """
    Coordinate descent: try perturbing each coordinate and accept if improved.
    Very fast for final polishing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for p in range(n_passes):
        improved = False
        for j in range(n):
            # Try a few perturbations
            base = a[j]
            if base < 1e-15:
                # Try adding a small value

                    a_try = a.copy()
                    a_try[j] = delta
                    a_try = a_try / a_try.sum()
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break
            else:
                # Try scaling
                for factor in [0.5, 0.8, 1.2, 1.5, 2.0]:
                    a_try = a.copy()
                    a_try[j] = base * factor
                    s_try = float(np.sum(a_try))
