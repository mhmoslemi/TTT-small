def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # Scale to target_sum
    v = v * (target_sum / max(v.sum(), 1e-15))
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - target_sum)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Specifically, suppose the peak is at lag k*. The contribution to (a*a)_{k*} is sum_j a_j * a_{j+k*}. We want to reduce this sum. One way is to reduce the product a_j * a_{j+k*} for the j where this product is large.

This suggests a "multiplicative" update: `a_j <- a_j * exp(-eta * w_j)` where w_j is the "responsibility" of a_j for the peaks.

This is similar to the "power method" idea in spectral algorithms.

Let me implement a version of this:

```

        # Responsibility of a_j: sum_k p_k * (a_{j+k} + a_{j-k})
        # This is the gradient of the smoothed max
        grad = 2 * ifft(fft(p, N) * conj(fft(a, N))).real[:n]
        # Multiplicative update
        a = a * exp(-lr * grad)
        a = np.maximum(a, 0)
        a = a / a.sum()
```

"""
Multi-scale smoothed optimization + active-set trust region + multiplicative deflation
for minimum peak autocorrelation of non-negative sequences.
Key ideas: progressive multi-scale warmup, adaptive log-sum-exp temperature with
Nesterov momentum, multiplicative (log-domain) gradient updates, improved TR-SOCP,
LP polish, and aggressive multi-length exploration.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if v.sum() < 1e-15:
        return np.full(n, target_sum / n)
    v = v * (target_sum / v.sum())
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - target_sum)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def smooth_gradient(a, tau):
    """Compute gradient of (1/tau)*log(sum_k exp(tau*(a*a)_k)) w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    exp_vals = np.exp(shifted)
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 2 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def nesterov_smoothed_gd(a, time_budget=200.0, tau_init=5.0, tau_end=400.0,
                          lr_scale=4.0, seed=42, verbose=False):
    """Nesterov-accelerated smoothed gradient descent with adaptive tau."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_scale / np.sqrt(n)
    v = np.zeros(n)
    momentum = 0.85
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        grad = smooth_gradient(a, tau)
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr = max(1e-12, lr * 0.9997)
        it += 1

        if verbose and it % 40 == 0:
            print(f"    NSGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

    return best_a


def multiplicative_gd(a, time_budget=150.0, tau_init=10.0, tau_end=300.0,
                      lr_scale=0.5, seed=42, verbose=False):
    """Multiplicative (log-domain) gradient descent. Naturally maintains a >= 0."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_scale / np.sqrt(n)
    it = 0
    log_a = np.log(a)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        a_curr = np.exp(log_a)
        a_curr = a_curr / a_curr.sum()

        conv = compute_conv(a_curr)
        max_c = float(np.max(conv))
        shifted = tau * (conv - max_c)
        exp_vals = np.exp(shifted)
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 2 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a_curr, N)
        g_full = ifft(P * np.conj(A)).real
        grad = 2.0 * g_full[:n]

        log_a = log_a - lr * grad
        log_a = np.maximum(log_a, -30.0)
        lr = max(1e-8, lr * 0.9995)

        a_curr = np.exp(log_a)
        a_curr = a_curr / a_curr.sum()

        obj_new = compute_obj(a_curr)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_curr.copy()

        it += 1
        if verbose and it % 40 == 0:
            print(f"    MGDD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

    return best_a


def solve_tr_sscp(a, r, verbose=False):
    """Trust region SOCP: min M s.t. C_old[k] + 2*(a*delta)[k] <= M, ||delta||<=r, sum(a+delta)=1, a+delta>=0."""
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)

    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta

    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
        cp.norm(delta, 2) <= r,
    ]

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_direction(a):
    """Solve LP: min M s.t. a*g[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def tr_sscp_polish(a, time_budget=300.0, r_init=0.3, verbose=False, seed=42):
    """Trust region SOCP polishing with adaptive line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    r = r_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        delta_dir, M_val = solve_tr_sscp(a, r)

        if delta_dir is None:
            g_lp, M_val = solve_lp_direction(a)
            if g_lp is None:
                break
            delta_dir = g_lp - a
        else:
            norm_d = np.linalg.norm(delta_dir)
            if norm_d > r:
                delta_dir = delta_dir * (r / norm_d)

        finite_mask = np.isfinite(delta_dir)
        if not finite_mask.all():
            delta_dir = np.where(finite_mask, delta_dir, 0.0)

        best_alpha = 0
        best_alpha_obj = best_obj
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]
        for t_alpha in alphas:
            a_try = a + t_alpha * delta_dir
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj - 1e-15:
                best_alpha_obj = obj_try
                best_alpha = t_alpha

        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-15:
            a = a + best_alpha * delta_dir
            a = np.maximum(a, 0)
            s_n = a.sum()
            if s_n > 1e-10:
                a = a / s_n
            best_obj = best_alpha_obj
            best_a = a.copy()
            no_improve = 0
            r = min(r * 1.15, 2.0)
        else:
            no_improve += 1
            r *= 0.6
            if no_improve >= 5:
                if r < 0.04:
                    a = best_a * np.exp(rng.normal(0, 0.015, n))
                    a = np.maximum(a, 0)
                    s_p = a.sum()
                    if s_p > 1e-10:
                        a = a / s_p
                    r = r_init
                    no_improve = 0
                    if it > 100:
                        break
                else:
                    r = max(r * 0.5, 0.04)

        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    TR it{it}: obj={best_obj:.8f}, r={r:.4f}, t={elapsed:.0f}s")

    return best_a


def lp_polish(a, time_budget=200.0, verbose=False, seed=42):
    """LP-based polishing with line search and adaptive perturbation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        best_t = 0
        best_t_obj = best_obj
        for t_val in [0.03, 0.07, 0.12, 0.2, 0.3, 0.45, 0.6, 0.8, 1.0]:
            a_try = (1 - t_val) * best_a + t_val * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-15:
                best_t_obj = obj_try
                best_t = t_val

        if best_t > 0 and best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = (1 - best_t) * best_a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            s_ba = best_a.sum()
            if s_ba > 1e-10:
                best_a = best_a / s_ba
            a = best_a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                scale = 0.006 + 0.004 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                no_improve = 0
                if it > 80:
                    break

        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def peak_balancing(a, time_budget=100.0, K=5, verbose=False):
    """
    Peak balancing: identify top-K peaks, solve LP to equalize them.
    min max_{k in active} (a*g)_k s.t. sum(g)=1, g>=0
    Then blend toward g.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        # Find active peaks
        threshold = max_c * 0.95
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            break
        # Limit to K most important
        if len(active) > K:
            active = active[np.argsort(conv[active])[-K:]]

        # Solve LP: min M s.t. (a*g)[k] <= M for k in active, sum(g)=1, g>=0
        A = build_conv_matrix(a)
        A_active = A[active, :]

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A_active @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        g_val = None
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        g_val = g_val / gs
                    break
            except Exception:
                continue

        if g_val is None:
            break

        # Line search blend
        best_t = 0
        best_t_obj = best_obj
        for t_val in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            a_try = (1 - t_val) * best_a + t_val * g_val
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-15:
                best_t_obj = obj_try
                best_t = t_val

        if best_t > 0:
            best_obj = best_t_obj
            best_a = (1 - best_t) * best_a + best_t * g_val
            best_a = np.maximum(best_a, 0)
            s_ba = best_a.sum()
            if s_ba > 1e-10:
                best_a = best_a / s_ba
            a = best_a.copy()
        else:
            # Perturb
            a = best_a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-10:
                a = a / s_a

        it += 1
        if verbose and it % 10 == 0:
            print(f"    PB it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

    return best_a

    print(f"\n=== Phase 5: LP Polish ===")

    p5_budget = min(remaining * 0.12, 120.0)
    if p5_budget > 20:
        a_lp = lp_polish(best_seq, time_budget=p5_budget,
                          verbose=True, seed=77)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase5 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 LP: obj={obj_lp:.8f}")

    # === Phase 6: Peak Balancing ===
    print(f"\n=== Phase 6: Peak Balancing ===")
    remaining = deadline - time.monotonic()
    p6_budget = min(remaining * 0.06, 60.0)
    if p6_budget > 10:
        a_pb = peak_balancing(best_seq, time_budget=p6_budget, K=5, verbose=True)
        obj_pb = compute_obj(a_pb)
        if obj_pb < best_score:
            best_score = obj_pb
            best_seq = a_pb.copy()
            print(f"  Phase6 PB: obj={best_score:.8f} **")
        else:
            print(f"  Phase6 PB: obj={obj_pb:.8f}")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600
    ]))
    n_options = [n for n in n_options if abs(n - n_current) > 60 and 300 <= n <= 2000]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)

        alloc1 = min(remaining * 0.1, 30.0)
        if alloc1 > 8:
            a_try = nesterov_smoothed_gd(a_try, time_budget=alloc1,
                                          tau_init=5.0, tau_end=200.0, lr_scale=4.0)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.12, 60.0)
        if alloc2 > 15:
            a_try = tr_sscp_polish(a_try, time_budget=alloc2, r_init=0.25)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Perturbation + re-optimization ===
    print(f"\n=== Phase 8: Perturbation + TR-SOCP ===")
    remaining = deadline - time.monotonic()

    for r_idx in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        scale = 0.005 + 0.006 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            break
        a_pert = a_pert / s_pert

        alloc = min(remaining * 0.2, 80.0)
        if alloc > 15:
            a_pert = tr_sscp_polish(a_pert, time_budget=alloc,
                                     r_init=0.2, seed=300 + r_idx)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 9: Final intensive TR-SOCP + LP ===
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 120.0)
        a_final = tr_sscp_polish(best_seq, time_budget=alloc,
                                  r_init=0.15, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final TR: obj={best_score:.8f} **")
        else:
            print(f"  Final TR: obj={obj_final:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 50.0)
        a_final_lp = lp_polish(best_seq, time_budget=alloc, seed=2000)
        obj_flp = compute_obj(a_final_lp)
        if obj_flp < best_score:
            best_score = obj_flp
            best_seq = a_final_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
