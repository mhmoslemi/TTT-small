def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa, size).real[:2*n-1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    c = compute_conv(a_norm)
    return 2.0 * n * float(np.max(c))

def build_conv_matrix(a):
    """Build matrix A such that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    # conv(a, g)[k] = sum_j a[k-j] * g[j] for valid j
    # = sum_j a[k-j] * g[j] where 0 <= k-j < n and 0 <= j < n
    k_idx = np.arange(2*n-1)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            if 0 <= k-j < n:
                A[k, j] = a[k-j]
    return A
