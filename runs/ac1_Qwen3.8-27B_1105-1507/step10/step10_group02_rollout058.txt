import numpy as np

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a, time_limit=30.0):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def diverse_init(n, rng, mode=0):

        a = a / a.sum()
    elif mode == 5:

        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Triangular
        x = np.arange(n)
        mid = n / 2.0

    else:
        a = np.ones(n) / n
    return a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def block_cd(a, time_budget=100.0, block_size=2, seed=42):

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective for normalized (sum=1) sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_limit=60.0):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def coord_desc(a, time_budget=100.0, seed=42):

    Exact coordinate descent. Each coordinate optimized via golden section

            c_k = conv[ks]

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]

            k_quad = 2 * i

                    return 1e30

            c_pt = x_hi - gr * (x_hi - x_lo)

            fd = f_1d(d_pt)

            for _ in range(25):

                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

        best_t = 0.0

        # Fine grid around previous best
        t_values = [0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.25, 0.3,
                    0.4, 0.5, 0.6, 0.7, 0.8, 1.0]

        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
                best_t = t

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                # Perturb to escape
                pert_size = 0.01 * (1.0 + 0.3 * stall)
                pert = rng.normal(0, pert_size, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if time.monotonic() - start > time_budget * 0.75:
                    break

        if iter_count % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP iter {iter_count}: obj={best_obj:.8f}, t={elapsed:.1f}s, best_t={best_t:.2f}")

    return best_a


def lp_cd_alternate(a, time_budget=200.0, seed=42):
    """
    Alternate between LP polish and coordinate descent.
    Each cycle: LP step + short CD refinement.
    This helps CD escape local minima that LP creates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    cycle = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 20:
            break
        cycle += 1

        # LP step
        lp_budget = min(remaining * 0.4, 40.0)
        if lp_budget > 10:
            a_lp = lp_polish_aggressive(a, time_budget=lp_budget, seed=seed + cycle * 100)
            obj_lp = 2.0 * n * float(np.max(compute_conv(a_lp / a_lp.sum())))
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                a = a_lp
            else:
                a = a_lp  # still use the LP result as starting point for CD

        # CD step
        cd_budget = min(remaining * 0.4, 40.0)
        if cd_budget > 10:
            a_cd = coord_desc(a, time_budget=cd_budget, seed=seed + 1000 + cycle * 100)
            obj_cd = 2.0 * n * float(np.max(compute_conv(a_cd / a_cd.sum())))
            if obj_cd < best_obj:
                best_obj = obj_cd
                best_a = a_cd.copy()
                a = a_cd

        if cycle % 3 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP-CD cycle {cycle}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.05, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [max(2, n // 12), max(2, n // 12)]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2.0 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x ** 2 + 0.05)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 12, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.uniform(0.1, 0.4, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1.0, n)
        alpha = rng.uniform(-1.5, 1.5)
        a = x ** alpha
        a = a / a.sum()
    elif mode == 6:
        x = np.arange(n)
        mid = n / 2.0
        a = 1.0 - 0.7 * np.abs(x - mid) / mid
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Periodic structure
        period = max(2, n // rng.integers(3, 8))
        pattern = rng.uniform(0.3, 1.0, period)
        a = np.tile(pattern, n // period + 1)[:n]
        a = a / a.sum()
    elif mode == 8:
        # Smooth random
        a = rng.uniform(0, 1, n)
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: aggressive LP polishing + coordinate descent alternation
    + multi-n exploration + diverse restarts with simulated annealing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = 2.0 * n_default * float(np.max(compute_conv(best_seq)))
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Initial CD convergence ===
    print(f"\n=== Phase 1: Initial Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 300.0)
    if p1_budget > 30:
        a_cd = coord_desc(best_seq, time_budget=p1_budget, seed=seed)
        obj_cd = 2.0 * a_cd.size * float(np.max(compute_conv(a_cd / a_cd.sum())))
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 2: Aggressive LP-CD Alternation (main workhorse) ===
    print(f"\n=== Phase 2: LP-CD Alternation ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.4, 500.0)
    if p2_budget > 50:
        a_alt = lp_cd_alternate(best_seq, time_budget=p2_budget, seed=seed + 100)
        obj_alt = 2.0 * a_alt.size * float(np.max(compute_conv(a_alt / a_alt.sum())))
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"  LP-CD alt: obj={best_score:.8f} **")
        else:
            print(f"  LP-CD alt: obj={obj_alt:.8f}")

    # === Phase 3: Pure LP polishing ===
    print(f"\n=== Phase 3: Pure LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 250.0)
    if p3_budget > 30:
        n_rounds = 4
        for rnd in range(n_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(p3_budget / (n_rounds - rnd), remaining * 0.5)
            if alloc < 10:
                break
            a_lp = lp_polish_aggressive(best_seq, time_budget=alloc, seed=2000 + rnd * 50)
            obj_lp = 2.0 * a_lp.size * float(np.max(compute_conv(a_lp / a_lp.sum())))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                # Quick CD after stagnation
                if remaining > 30:
                    a_cd = coord_desc(a_lp, time_budget=min(20, remaining * 0.2),
                                      seed=3000 + rnd * 50)
                    obj_cd = 2.0 * a_cd.size * float(np.max(compute_conv(a_cd / a_cd.sum())))
                    if obj_cd < best_score:
                        best_score = obj_cd
                        best_seq = a_cd.copy()
                        print(f"  CD after LP {rnd}: obj={best_score:.8f} **")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = []
    deltas = [-300, -200, -100, 100, 200, 300, 500]
    for d in deltas:
        n_try = n_current + d
        if 200 <= n_try <= 2500:
            n_options.append(n_try)
    # Also add some absolute values
    n_options.extend([400, 600, 800, 1000, 1200, 1600])
    n_options = sorted(set(n_options))
    n_options = [n for n in n_options if 200 <= n <= 2500 and abs(n - n_current) > 30]
    n_options = n_options[:8]

    print(f"  n_current={n_current}, trying: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)

        # CD on new length
        alloc_cd = min(remaining * 0.15, 50.0)
        if alloc_cd > 15:
            a_try = coord_desc(a_try, time_budget=alloc_cd, seed=n_try * 7)

        obj_try = 2.0 * a_try.size * float(np.max(compute_conv(a_try / a_try.sum())))

        # If promising, do LP too
        if obj_try < best_score + 0.01 and remaining > 60:
            alloc_lp = min(remaining * 0.2, 50.0)
            if alloc_lp > 15:
                a_try = lp_polish_aggressive(a_try, time_budget=alloc_lp, seed=n_try * 13)
                obj_try = 2.0 * a_try.size * float(np.max(compute_conv(a_try / a_try.sum())))

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts with Annealing ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_use = len(best_seq)

    num_restarts = 8
    for r in range(num_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = (r + 1) % 9
        a_r = diverse_init(n_use, rng, mode=mode)
        blend = 0.15 + 0.08 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # Simulated annealing: if blend is small, accept worse solutions
        alloc_cd = min(remaining * 0.12, 45.0)
        alloc_lp = min(remaining * 0.08, 30.0)

        if alloc_cd > 10:
            a_r = coord_desc(a_r, time_budget=alloc_cd, seed=6000 + r * 200)

        obj_r = 2.0 * a_r.size * float(np.max(compute_conv(a_r / a_r.sum())))

        if alloc_lp > 10 and remaining > 60:
            a_r = lp_polish_aggressive(a_r, time_budget=alloc_lp, seed=7000 + r * 200)
            obj_r = 2.0 * a_r.size * float(np.max(compute_conv(a_r / a_r.sum())))

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}, blend={blend:.2f}): obj={best_score:.8f} **")
        else:
            # Annealing: accept with small probability if close
            if obj_r < best_score * 1.002 and rng.random() < 0.1:
                best_seq = a_r.copy()
                best_score = obj_r
                print(f"  Restart {r}: accepted via annealing: obj={best_score:.8f}")
            else:
                print(f"  Restart {r} (mode={mode}, blend={blend:.2f}): obj={obj_r:.8f}")

    # === Phase 6: Final Refinement ===
    print(f"\n=== Phase 6: Final Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Final CD
        alloc = min(remaining * 0.35, 150.0)
        if alloc > 20:
            a_final = coord_desc(best_seq, time_budget=alloc, seed=seed + 9999)
            obj_final = 2.0 * a_final.size * float(np.max(compute_conv(a_final / a_final.sum())))
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

        # Final LP
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.35, 80.0)
        if alloc > 10:
            a_lp = lp_polish_aggressive(best_seq, time_budget=alloc, seed=8888)
            obj_lp = 2.0 * a_lp.size * float(np.max(compute_conv(a_lp / a_lp.sum())))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation with actual evaluator
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
