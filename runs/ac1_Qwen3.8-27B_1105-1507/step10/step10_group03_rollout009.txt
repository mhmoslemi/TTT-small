import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Returns (gradient of length n, exact_max of conv).
    """
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build the convolution matrix A where (A @ g)_k = (a*g)_k."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        start = j
        end = min(start + n, m)
        length = end - start
        A[start:end, j] = a[:length]
    return A


def solve_lp(a, lower_bound=0.0, solver_list=None):
    """
    Solve LP: min M s.t. (a*g)_k <= M for all k, g >= lower_bound, sum(g) = 1.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    if n > 5000:
        return None, None

    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]

    A = build_conv_matrix(a)

    g_var = cp.Variable(n, nonneg=(lower_bound > 0))
    if lower_bound > 0:
        constraints = [A @ g_var <= cp.Variable(),
                       g_var >= lower_bound,
                       cp.sum(g_var) == 1]
        M_var = constraints[0][1]
    else:
        M_var = cp.Variable()
        constraints = [A @ g_var <= M_var,
                       cp.sum(g_var) == 1]

    prob = cp.Problem(cp.Minimize(M_var), constraints)

    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0.0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    g_val = g_val / sg
                    # Recompute M from actual solution
                    conv_check = compute_conv(a)
                    # Actually compute (a * g) properly
                    n_a = len(a)
                    n_g = len(g_val)
                    ag_size = 1
                    while ag_size < n_a + n_g - 1:
                        ag_size <<= 1
                    fa = rfft(a, ag_size)
                    fg = rfft(g_val, ag_size)
                    ag = irfft(fa * fg, ag_size)[:n_a + n_g - 1]
                    M_val = float(np.max(ag))
                    return g_val, M_val
        except Exception:
            continue

    return None, None


def lp_fixed_point(a, time_budget, verbose=True, seed=42, max_iters=500,
                   lower_bound=0.0, adaptive=True):
    """
    LP fixed-point iteration with adaptive blending.
    The key insight: solve LP to find best response g, then blend.
    Use adaptive step size based on improvement ratio.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    last_t = 1.0
    stall_count = 0
    consec_no_improve = 0

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        # Solve LP
        g_lp, M_lp = solve_lp(a, lower_bound=lower_bound)
        if g_lp is None:
            break

        # Current peak
        conv_a = compute_conv(a)
        cur_peak = float(np.max(conv_a))

        # Adaptive blending parameter
        if adaptive and cur_peak > 1e-15:
            ratio = M_lp / cur_peak
            # If LP can significantly improve, take a bigger step
            if ratio < 0.8:
                t_candidate = min(1.0, 0.8)
            elif ratio < 0.95:
                t_candidate = min(1.0, 0.5)
            else:
                t_candidate = min(1.0, 0.2)
            # Damp if stalling
            if stall_count > 5:
                t_candidate *= 0.5
        else:
            t_candidate = last_t

        # Line search over a few blending parameters
        t_values = sorted(set([
            t_candidate * 0.5, t_candidate * 0.75, t_candidate,
            t_candidate * 1.25, t_candidate * 1.5,
            min(1.0, t_candidate * 2.0),
            0.1, 0.3, 0.5
        ]))
        t_values = [t for t in t_values if 0.01 <= t <= 1.0]

        best_t_obj = best_obj
        best_t_val = t_candidate
        best_a_local = best_a.copy()

        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            conv_try = compute_conv(a_try)
            obj_try = 2.0 * n * float(np.max(conv_try))
            if obj_try < best_t_obj - 1e-14:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall_count = 0
            consec_no_improve = 0
            if verbose and (it < 20 or it % 20 == 0):
                elapsed = time.monotonic() - start
                print(f"    LP it{it}: obj={best_obj:.10f}, t={best_t_val:.3f}, "
                      f"M_lp={M_lp:.6f}, peak={cur_peak:.6f}, t_elapsed={elapsed:.0f}s")
        else:
            stall_count += 1
            consec_no_improve += 1
            # Try perturbation to escape
            if stall_count % 3 == 0:
                pert_scale = 0.01 * (1 + stall_count * 0.1)
                a_pert = a + pert_scale * (rng.random(n) - 0.5)
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                # Only accept perturbation if it doesn't make things much worse
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj + 0.01:
                    a = a_pert

            if stall_count > 30:
                break

    if verbose:
        print(f"    LP done: obj={best_obj:.10f}, iters={it}, time={time.monotonic()-start:.0f}s")
    return best_a


def staircase_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9, lr_scale=2.0):
    """
    Nesterov accelerated gradient descent with staircase tau schedule.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 15, 40, 100, 250, 600, 1500, 3000]

    it = 0
    best_it = 0
    stall_patience = 3000

    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0
        max_iters_per_level = 3000

        while time.monotonic() - start < time_budget:
            if tau_iters >= max_iters_per_level:
                break

            y = a + beta * v
            y = project_simplex(y)

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = project_simplex(a_new)

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it

            a = a_new
            it += 1
            tau_iters += 1
            lr = max(1e-7, lr * 0.99999)

            if it - best_it > stall_patience:
                a = best_a.copy()
                v = np.zeros(n)
                lr = lr_scale / np.sqrt(n)
                best_it = it

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    GD tau={tau:5d}: obj={best_obj:.10f}, iters={tau_iters}, t={elapsed:.0f}s")

    return best_a


def subgradient_descent(a, time_budget, verbose=False, seed=42):
    """
    Subgradient method with heavy ball acceleration.
    Uses the exact subgradient (directional) of max(conv(a,a)).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    lr = 1.0 / np.sqrt(n)
    beta = 0.8
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        if it % 1000 == 0 and verbose:
            print(f"    SubGrad it{it}: obj={best_obj:.10f}, t={time.monotonic()-start:.0f}s")

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Find all lags that achieve the max (within tolerance)
        tol = max_val * 1e-8 + 1e-15
        active = np.where(conv >= max_val - tol)[0]

        # Compute subgradient: average gradient over active set
        # grad of (a*a)_k w.r.t. a_j is a_{k-j} + a_{j+k-j}... 
        # Actually (a*a)_k = sum_j a_j * a_{k-j}
        # d/dx (x*x) type: if we write (a*a)_k as quadratic form, 
        # grad component for a_j is: a_{k-j} if valid + a_{k-j} if valid (symmetric)
        # More carefully: (a*a)_k = sum_{j: 0<=j<n, 0<=k-j<n} a_j * a_{k-j}
        # d(a*a)_k / da_j = a_{k-j} (if 0<=k-j<n) + a_{j+k-j}... 
        # Actually for the quadratic form (a*a)_k = a^T T_k a where T_k is a 
        # symmetric banded matrix, the gradient is 2*T_k*a.
        # (T_k a)_j = a_{k-j} if 0<=k-j<n else 0 (this is the cross-correlation of a with itself shifted)
        # So grad_k = 2 * shift(a, k) where shift(a, k)_j = a_{k-j} if valid
        
        # For efficiency, compute gradient as: for each active k, add 2*a_shifted
        # where a_shifted[j] = a[k-j] if 0<=k-j<n
        
        g = np.zeros(n)
        for k in active:
            # (a*a)_k = sum_j a_j * a_{k-j} (where indices are valid)
            # gradient w.r.t. a_j: a_{k-j} for valid k-j, plus a_{k-j'} for the "other" term
            # Since (a*a)_k is symmetric in the two copies of a, grad = 2 * (a *_k) where
            # (a *_k)_j = a_{|k-j|}... no that's not right for linear conv.
            # 
            # Let me just compute directly:
            # (a*a)_k = sum over j of a[j] * a[k-j] where 0<=k-j<n
            # d/d a[j] = a[k-j] if 0<=k-j<n
            # But also, a[j] appears in (a*a)_k when k-j = j, i.e., j = k/2 (if k even)
            # No wait, (a*a)_k = sum_j a[j] * a[k-j]. The term a[j] appears with coefficient a[k-j].
            # The term a[k-j] also appears (when we substitute j' = k-j) with coefficient a[j].
            # So the full gradient w.r.t. a[j] is:
            #   a[k-j] (from the j term) + a[k-j] (from the j'=k-j term, which gives a[k-j']=a[j])
            # Wait, let me be more careful.
            # (a*a)_k = sum_{j=0}^{n-1} a[j] * a[k-j] * I(0<=k-j<=n-1)
            # Let m = k-j, so j = k-m. The sum becomes sum_m a[k-m] * a[m] * I(0<=k-m<=n-1)
            # This is the same sum (just reindexed). So (a*a)_k is indeed a quadratic form.
            # grad_{a[j]} (a*a)_k = a[k-j] (if 0<=k-j<n) + a[j-k+n-1]... 
            # 
            # Simpler: grad_{a[j]} (a*a)_k = a[k-j] if 0<=k-j<n, PLUS a[k-j'] where j' satisfies k-j'=j, i.e., j'=k-j.
            # Wait that's the same thing. Let me just do it directly:
            # 
            # (a*a)_k = sum_{j} a_j * b_j where b_j = a_{k-j} (shifted version of a)
            # d/d a_j (sum_i a_i * b_i) = b_j + a_j * db_j/d a_j
            # db_j/d a_j = 0 (b_j = a_{k-j} doesn't depend on a_j unless k-j=j)
            # So grad = b = shift of a
            # But wait, (a*a)_k = a^T D_k a where D_k is a diagonal/permutation matrix
            # Actually (a*a)_k = sum_j a_j * a_{k-j}. This is NOT a standard quadratic form a^T Q a
            # unless we're careful. Let's write it as:
            # (a*a)_k = sum_j a_j * a_{k-j}
            # This IS a quadratic form: a^T P_k a where P_k[j,m] = 1 if m = k-j, i.e., j+m=k
            # P_k is a permutation matrix (anti-diagonal slice). Since P_k is symmetric (j+m=k 
            # iff m+j=k), the gradient is 2*P_k*a, i.e., 2*(a reversed and shifted).
            # (P_k a)_j = a_{k-j} if 0<=k-j<n
            # So grad_{a_j} (a*a)_k = 2 * a_{k-j} if 0<=k-j<n, else 0.
            # Wait, but if k-j = j (i.e., j = k/2), then we're differentiating a_j^2, giving 2*a_j.
            # And 2*a_{k-j} = 2*a_j in that case. So it's consistent.
            #
            # So: grad_k[j] = 2*a[k-j] if 0 <= k-j < n, else 0

            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]

        # Normalize subgradient to unit norm for step size control
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Heavy ball step
        v = beta * v - lr * g / g_norm
        a_new = a + v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1
        # Decay learning rate
        lr = max(1e-6, lr * 0.9999)

        # Restart if stuck
        if it % 2000 == 0 and obj_new > best_obj * 1.01:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 1.0 / np.sqrt(n)

    return best_a


def make_init(n, mode, rng=None):
    """Create diverse initial sequences of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:  # Triangle
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
    elif mode == 2:  # Semicircle
        xx = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - xx ** 2, 0.0001))
    elif mode == 3:  # Two bumps
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
    elif mode == 4:  # Uniform random
        a = rng.uniform(0.1, 1, n)
    elif mode == 5:  # Cosine bump
        a = np.maximum(0.5 + 0.5 * np.cos(np.pi * x * 2), 0.001)
    elif mode == 6:  # Multi-sine
        a = np.ones(n)
        for f in [2, 3, 5, 7]:
            a += 0.15 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
    elif mode == 7:  # Exponential center
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 4))
    elif mode == 8:  # Flat top with tapered edges
        a = np.where((x > 0.15) & (x < 0.85), 1.0,
                     np.maximum(0.6 - 2 * np.abs(x - 0.5), 0.01))
    elif mode == 9:  # Asymmetric
        a = np.exp(-((np.arange(n) - n * 0.35) ** 2) / (2 * (n / 5) ** 2))
        a += 0.5 * np.exp(-((np.arange(n) - n * 0.65) ** 2) / (2 * (n / 6) ** 2))
    elif mode == 10:  # Bump with oscillation
        a = np.exp(-((np.arange(n) - n / 2) ** 2) / (2 * (n / 3) ** 2))
        a *= (1 + 0.2 * np.sin(2 * np.pi * 3 * x))
        a = np.maximum(a, 0.001)
    elif mode == 11:  # Parabola (inverted)
        a = np.maximum(1 - (2 * x - 1) ** 2, 0.001)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0001)
    return a / a.sum()


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP fixed-point as primary engine, with subgradient method,
    smoothed GD, multi-n exploration, and final polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        parent = None

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.10f}")

    n_current = len(best_seq)

    def time_remaining():
        return deadline - time.monotonic()

    # === Phase 1: Quick diverse initialization screening ===
    print(f"\n=== Phase 1: Diverse Init Screening ===")
    init_modes = [0, 1, 2, 7, 9, 10, 11]
    for mode in init_modes:
        if time_remaining() < 15:
            break
        a_init = make_init(n_current, mode, rng)
        alloc = min(time_remaining() * 0.03, 8.0)
        if alloc < 2:
            break
        a_screen = staircase_gd(a_init, time_budget=alloc,
                                tau_levels=[10, 50, 200], verbose=False)
        obj_screen = compute_obj(a_screen)
        if obj_screen < best_score:
            best_score = obj_screen
            best_seq = a_screen.copy()
            print(f"  Init mode {mode}: obj={best_score:.10f} **")
        else:
            print(f"  Init mode {mode}: obj={obj_screen:.10f}")

    print(f"  After screening: best={best_score:.10f}")

    # === Phase 2: LP Fixed-Point (primary workhorse) ===
    print(f"\n=== Phase 2: LP Fixed-Point Iteration ===")
    p2_budget = min(time_remaining() * 0.35, 500.0)
    if p2_budget > 20 and n_current <= 3000:
        a2 = lp_fixed_point(best_seq, time_budget=p2_budget, verbose=True,
                           seed=100, max_iters=1000, lower_bound=0.0)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 LP: obj={best_score:.10f} **")
        else:
            print(f"  Phase2 LP: obj={obj2:.10f}")

    # === Phase 3: Subgradient Method ===
    print(f"\n=== Phase 3: Subgradient Method ===")
    remaining = time_remaining()
    p3_budget = min(remaining * 0.12, 150.0)
    if p3_budget > 10:
        a3 = subgradient_descent(best_seq, time_budget=p3_budget, verbose=True, seed=200)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  SubGrad: obj={best_score:.10f} **")
        else:
            print(f"  SubGrad: obj={obj3:.10f}")

    # === Phase 4: GD Polishing ===
    print(f"\n=== Phase 4: Smoothed GD Polishing ===")
    remaining = time_remaining()
    p4_budget = min(remaining * 0.12, 150.0)
    if p4_budget > 10:
        a4 = staircase_gd(best_seq, time_budget=p4_budget,
                          tau_levels=[20, 80, 300, 1000, 3000], verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  GD: obj={best_score:.10f} **")
        else:
            print(f"  GD: obj={obj4:.10f}")

    # === Phase 5: Second LP round (on polished result) ===
    print(f"\n=== Phase 5: LP Round 2 ===")
    remaining = time_remaining()
    p5_budget = min(remaining * 0.15, 200.0)
    if p5_budget > 15 and len(best_seq) <= 3000:
        a5 = lp_fixed_point(best_seq, time_budget=p5_budget, verbose=True,
                           seed=300, max_iters=500, lower_bound=0.0, adaptive=True)
        obj5 = compute_obj(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  LP2: obj={best_score:.10f} **")
        else:
            print(f"  LP2: obj={obj5:.10f}")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = time_remaining()
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current, max(400, n_current - 200), n_current + 100,
        n_current + 200, n_current + 400, n_current + 800,
        500, 700, 900, 1100, 1500, 2000, 2500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 3500 and n != n_current]
    # Prioritize: close to current, then slightly larger
    n_options.sort(key=lambda x: (abs(x - n_current), x))
    n_options = n_options[:5]

    print(f"  n candidates: {n_options}")
    for n_try in n_options:
        remaining = time_remaining()
        if remaining < 60:
            break

        a_try = resample(best_seq, n_try)

        # Quick GD
        alloc1 = min(remaining * 0.2, 30.0)
        if alloc1 > 10:
            a_try = staircase_gd(a_try, time_budget=alloc1,
                                  tau_levels=[10, 50, 200], verbose=False)

        # LP if feasible
        remaining = time_remaining()
        if n_try <= 2000:
            alloc2 = min(remaining * 0.15, 40.0)
            if alloc2 > 10:
                a_try = lp_fixed_point(a_try, time_budget=alloc2, verbose=False,
                                       seed=n_try, max_iters=100)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = time_remaining()

    # Final LP
    if remaining > 30 and len(best_seq) <= 3000:
        alloc = min(remaining * 0.3, 100.0)
        a_f1 = lp_fixed_point(best_seq, time_budget=alloc, verbose=True,
                             seed=999, max_iters=200, lower_bound=0.0)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    # Final GD at high tau
    remaining = time_remaining()
    if remaining > 20:
        alloc = min(remaining * 0.25, 60.0)
        a_f2 = staircase_gd(best_seq, time_budget=alloc,
                             tau_levels=[200, 800, 2000, 5000], verbose=False)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final GD: obj={best_score:.10f} **")

    # Subgradient at the end
    remaining = time_remaining()
    if remaining > 20:
        alloc = min(remaining * 0.3, 50.0)
        a_f3 = subgradient_descent(best_seq, time_budget=alloc, verbose=False, seed=777)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final SubGrad: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
