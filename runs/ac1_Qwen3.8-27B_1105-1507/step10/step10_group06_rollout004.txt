import numpy as np

from scipy.fft import rfft, irfft, fft, ifft


def compute_conv_fft(a, b=None):
    """Full linear convolution a*b via FFT. If b is None, compute a*a."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(conv))

def coordinate_descent_polish(a, n_sweeps=10, time_budget=None):
    """
    Coordinate descent: for each coordinate, try optimal value given others fixed.
    Uses 1D ternary search for the optimal value.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_obj = compute_obj(a)
    start = time.monotonic()
    if time_budget is None:
        time_budget = 10.0
    
    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                    break
            
            # Try a few values for a[i]
            old_val = a[i]
            a[i] = 0.0
            s_others = float(np.sum(a))
            
            if s_others < 1e-10:
                a[i] = old_val
                continue
            
            # The optimal a[i] balances max(conv) vs sum
            # Try: 0, old_val, and some values around old_val
            candidates = [0.0, old_val]
            # Also try values that would make sum = 1
            if s_others > 0:
                candidates.append(1.0 - s_others)  # Makes total sum = 1
                candidates.append(old_val * 0.5)
                candidates.append(old_val * 1.5)
                candidates.append(0.01 / max(n, 1))
            
            best_local = a.copy()
            best_local_obj = best_obj
            a[i] = old_val  # Restore first
            
            for val in candidates:
                if val < 0:
                    val = 0.0
                a[i] = val
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    a[i] = 1.0
                    s_new = float(np.sum(a))
                if s_new < 0.01:
                    continue
                obj = compute_obj(a)
                if obj < best_local_obj:
                    best_local_obj = obj
                    best_local = a.copy()
            
            if best_local_obj < best_obj - 1e-14:
                a = best_local.copy()
                best_obj = best_local_obj
                improved = True
        
        if not improved:
            break
    
    return a


def simulated_annealing_moves(a, n_moves=10000, time_budget=30.0, rng=None):
    """
    Simulated annealing: pick random pairs, move mass between them.
    """
    if rng is None:
        rng = np.random.default_rng()
    
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    T = 1.0  # Temperature
    
    for move in range(n_moves):
        if time.monotonic() - start > time_budget:
            break
        
        # Pick two random coordinates
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Move mass from i to j (or vice versa)
        delta = rng.uniform(0.01, 0.1) * a[i]
        
        a_new = a.copy()
        a_new[i] = max(0, a[i] - delta)
        a_new[j] = a[j] + delta
        
        s_new = float(np.sum(a_new))
        if s_new < 0.01:
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        
        # Accept if better or with probability exp((-delta_obj)/T)
        delta_obj = obj_new - best_obj
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
            a = a_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
        
        T *= 0.999
    
    return best_a


def mwu_alternate(a, time_budget=60.0, n_inner_iters=50000, verbose=True):
    """

    Alternates between finding best g for current a, then updating a.
    """

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    t_blend = 0.3
    outer_iter = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)

        t0 = time.monotonic()
        g, val_g = mwu_solve_inner(a, n_iter=n_inner_iters, eta=1.0, tau_smooth=20.0)
        inner_time = time.monotonic() - t0
        
        outer_iter += 1
        
        # Try multiple blending factors
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        
        for t in [t_blend, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:

            a_try = a_try / s_try

            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t
        
        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            if verbose and outer_iter % 10 == 0:
                print(f"    MWU alt {outer_iter}: obj={best_obj:.10f}, t={best_local_t:.3f}, "
                      f"inner_val={val_g:.6f}, time={inner_time:.2f}s")
        else:
            # No improvement, perturb

def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fast_gradient_polish(a, time_budget=30.0, tau_init=50.0, tau_max=2000.0):
    """
    Fast gradient descent with Nesterov acceleration on the smoothed objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    beta = 0.9
    lr = 1.0 / np.sqrt(n)
    v = np.zeros(n)
    tau = tau_init
    it = 0
    
    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Compute smoothed gradient
        fa = rfft(y, size)
        conv = irfft(fa * fa, size)[:2*n-1]
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(2*n-1)
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp
        
        # Gradient: d/dy_j max_k conv[k] ≈ sum_k p[k] * 2*y[k-j] (with bounds)
        fp = rfft(p, size)
        grad_full = irfft(np.conj(fa) * fp, size)
        grad = 2.0 * grad_full[:n]
        
        # Update
        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            adaptive_lr = 0.5 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.1 * lr), 3.0 * lr)
            lr = 0.9 * lr + 0.1 * adaptive_lr
        
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        tau = min(tau * 1.05, tau_max)
        it += 1
    
    return best_a
