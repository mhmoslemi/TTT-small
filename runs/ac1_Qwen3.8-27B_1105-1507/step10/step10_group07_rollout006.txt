"""
Multi-strategy optimizer for minimum peak convolution ratio.
Combines Nesterov-accelerated smoothed gradient descent with adaptive temperature,
progressive subgradient refinement, LP-based escape moves, convolution-matching
gradient, and multi-start with diverse initializations and length exploration.
Key innovations over previous: Nesterov momentum improves convergence speed,
progress-based temperature adapts to optimization progress, convolution-matching
gradient provides a complementary flattening force, and diverse initializations
help escape local minima.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, tau):
    """
    Compute smoothed gradient of the objective.
    Uses exponential weighting of convolution peaks.
    Returns gradient vector of length n.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    if max_conv < 1e-15:
        return np.zeros(n)

    exp_vals = np.exp(tau * (conv - max_conv) / max_conv)
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def conv_matching_gradient(a, c_target=None, alpha=0.3):
    """
    Compute gradient based on convolution L2 matching.
    Encourages the convolution to be flat.
    """
    n = len(a)
    conv = compute_conv(a)
    s = float(np.sum(a))

    if c_target is None:
        c_target = np.full(len(conv), s * s / (2 * n - 1))

    error = conv - c_target
    N = 1
    while N < 3 * n:
        N <<= 1
    E = fft(error, N)
    A = fft(a, N)
    g_full = ifft(E * np.conj(A)).real
    return alpha * 2.0 * g_full[:n]

def adaptive_subgradient(a, time_budget, K_start=4, K_max=30, seed=42):
    """
    Subgradient method with adaptively increasing K (number of top peaks used).
    Uses Polyak-type step size with running average.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    lr = 5.0 / np.sqrt(n)
    momentum = 0.75
    v = np.zeros(n)
    K = K_start
    stall = 0
    it = 0
    min_obj = best_obj
    max_obj = best_obj

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 2:
            break

        # Adapt K: increase as we make progress
        progress_ratio = min((it + 1) / 200.0, 1.0)
        K = int(K_start + (K_max - K_start) * progress_ratio)

        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break

        # Exponential weighting within top-K
        temp = max_pv / max(1.0, np.std(peak_vals) + 1e-15)
        temp = max(3.0, min(temp, 15.0))
        weights = np.exp((peak_vals - max_pv) * temp / max_pv)
        weights /= weights.sum()

        # Compute subgradient
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            min_obj = min(min_obj, obj_new)
            max_obj = max(max_obj, obj_new)
            lr = min(lr * 1.01, 8.0 / np.sqrt(n))
            stall = 0
        else:
            a = a_new
            stall += 1
            if stall > 150:
                lr *= 0.7
                v *= 0.5
                stall = 0
                if lr < 1e-13:
                    # Restart
                    a = best_a * np.exp(rng.normal(0, 0.01, n))
                    a = np.maximum(a, 0)
                    sa = a.sum()
                    if sa > 1e-10:
                        a = a / sa
                    lr = 5.0 / np.sqrt(n)
                    v = np.zeros(n)
                    if it > 500:
                        break
            else:
                lr *= 0.996

        it += 1

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        lo = j
        hi = min(m, j + n)
        length = hi - lo
        if length > 0:
            A[lo:hi, j] = a[:length]
    return A


def lp_escape_move(a, time_budget=30.0, M_target=None, seed=42):
    """
    LP-based escape: find g that minimizes max conv(a,g).
    Use line search to blend with current a.
    """
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    try:
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
        if M_target is not None:
            constraints.append(M_var <= M_target)
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(np.asarray(g_var.value).flatten(), 0)
                    sg = float(np.sum(g_val))
                    if sg > 1e-10:
                        g_val = g_val / sg
                        break
            except Exception:
                continue
        else:
            return None
    except Exception:
        return None

    # Line search
    best_t = 0
    best_obj = compute_obj(a)
    for t_val in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        a_try = (1 - t_val) * a + t_val * g_val
        a_try = np.maximum(a_try, 0)
        st = a_try.sum()
        if st < 1e-10:
            continue
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t_val

    if best_t > 0:
        result = (1 - best_t) * a + best_t * g_val
        result = np.maximum(result, 0)
        sr = result.sum()
        if sr > 1e-10:
            result = result / sr
        return result
    return None


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def diverse_init(n, seed):
    """Generate a diverse random initialization."""
    rng = np.random.default_rng(seed)
    # Dirichlet with concentrated alpha for structure
    alpha = 0.5 + rng.random() * 2.0
    v = rng.dirichlet(alpha * np.ones(n) / n * n)
    # Mix with a structured pattern
    pattern = np.ones(n)
    # Add some periodic structure
    period = int(n / (3 + rng.integers(0, 4)))
    for i in range(n):
        pattern[i] = 1.0 + 0.5 * np.sin(2 * np.pi * i / period)
    pattern = np.maximum(pattern, 0.1)
    blended = 0.5 * v + 0.5 * pattern / pattern.sum()
    return blended / blended.sum()

            a_init = np.ones(n)
            block = max(3, n // 40)
            for i in range(n):
                a_init[i] = 1.0 + 0.4 * np.sin(2 * np.pi * i / (block * 2))
            a_init = np.maximum(a_init, 0.01)
            a_init = a_init * np.exp(rng.normal(0, 0.02, n))
            a_init = a_init / a_init.sum()

        alloc_total = min(time_left() * 0.12, 80.0)
        if alloc_total < 20:
            break

        # Quick GD
        alloc_gd = alloc_total * 0.7
        a_ms = nesterov_smoothed_gd(a_init, time_budget=alloc_gd,
                                    tau_init=5.0, tau_max=300.0,
                                    combine_alpha=0.2, seed=seed * 100 + si)
        # Quick SG
        if time_left() > 15:
            alloc_sg = min(time_left() * 0.04, alloc_total * 0.3)
            a_ms = adaptive_subgradient(a_ms, time_budget=alloc_sg,
                                        K_start=3, K_max=15, seed=seed * 100 + si + 50)

        obj_ms = compute_obj(a_ms)
        print(f"  Start{si}: obj={obj_ms:.8f}" + (" ***" if obj_ms < best_score else ""))
        if obj_ms < best_score:
            best_score = obj_ms
            best_seq = a_ms.copy()

    # === Phase 6: Length Exploration ===
    print("\n=== Phase 6: Length Exploration ===")
    n_current = len(best_seq)
    n_options = [n_current + 100, n_current + 200, n_current - 100,
                 n_current + 50, n_current - 50]
    n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_current) > 30]
    n_options = sorted(set(n_options))[:3]

    for n_try in n_options:
        if time_left() < 40:
            break

        print(f"  Trying n={n_try} (remaining={time_left():.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        alloc = min(time_left() * 0.10, 60.0)
        if alloc > 15:
            a_try = nesterov_smoothed_gd(a_try, time_budget=alloc,
                                         tau_init=5.0, tau_max=250.0,
                                         combine_alpha=0.15, seed=seed + 500 + n_try)
            obj_try = compute_obj(a_try)
            print(f"  n={n_try}: obj={obj_try:.8f}" + (" ***" if obj_try < best_score else ""))
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()

    # === Phase 7: Final Intensive Polish ===
    print("\n=== Phase 7: Final Polish ===")
    if time_left() > 30:
        alloc = min(time_left() * 0.35, 150.0)
        a_final = nesterov_smoothed_gd(best_seq, time_budget=alloc,
                                       tau_init=50.0, tau_max=600.0,
                                       combine_alpha=0.05, seed=seed + 999)
        obj_final = compute_obj(a_final)
        print(f"  Final GD: obj={obj_final:.8f}" + (" ***" if obj_final < best_score else ""))
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()

    # Final subgradient squeeze
    if time_left() > 20:
        alloc = min(time_left() * 0.25, 60.0)
        a_sq = adaptive_subgradient(best_seq, time_budget=alloc,
                                    K_start=6, K_max=30, seed=seed + 1111)
        obj_sq = compute_obj(a_sq)
        print(f"  Final SG: obj={obj_sq:.8f}" + (" ***" if obj_sq < best_score else ""))
        if obj_sq < best_score:
            best_score = obj_sq
            best_seq = a_sq.copy()

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
