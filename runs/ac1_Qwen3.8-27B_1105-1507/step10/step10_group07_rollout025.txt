def compute_conv(a):
    """Full convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]

def compute_obj(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a_n = a / s
    c = compute_conv(a_n)
    return 2.0 * n * float(np.max(c))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    while N < 2 * n - 1:

    return irfft(fa * fa, N)[:2 * n - 1]

    n = len(a)
    if n == 0:

    s = float(np.sum(a))
    if s < 0.01:

    a_n = a / s

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)

    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
FISTA + LP + Coordinate Descent hybrid for minimum peak convolution.
Key improvements over previous approach:
1. Pure numpy/FFT-based gradients (no CVXPY) for 50-100x faster per-iteration
2. FISTA acceleration with adaptive logsumexp temperature
3. scipy HiGHS LP for direction finding (much faster than CVXPY overhead)
4. Random coordinate descent for final micro-optimization
5. Aggressive multi-length exploration with structured resampling
6. Simulated annealing restarts to escape local minima
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    c = compute_conv(a_n)
    return 2.0 * n * float(np.max(c))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} via Euclidean projection."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def resample_1d(a, new_n):
    """Resample sequence to new length via linear interpolation, then clip & normalize."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    new_vals = np.interp(x_new, x_old, a)
    new_vals = np.maximum(new_vals, 0)
    s = new_vals.sum()
    if s > 1e-15:
        new_vals = new_vals / s
    else:
        new_vals = np.ones(new_n) / new_n
    return new_vals


def solve_lp_direction(a, use_max_bound=None):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_value) or (None, None).
    """
    n = len(a)
    a = a / a.sum()

    # Build A_ub: (2n-1) x (n+1) for constraints conv(a,g)[k] - M <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            indices = np.arange(i_start, i_end + 1)
            A_ub[k, indices] = a[k - indices]
        A_ub[k, n] = -1.0

    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Objective: minimize M
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Bounds
    bounds = [(0, None)] * n + [(None, None)]
    if use_max_bound is not None:
        bounds[n] = (None, use_max_bound)

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success and result.x is not None:
            g = np.maximum(result.x[:n], 0)
            s = g.sum()
            if s > 1e-10:
                g = g / s
                return g, float(result.x[n])
    except Exception:
        pass
    return None, None


def fista_smoothed(a, time_budget, tau_init=30.0, tau_end=800.0, seed=42):
    """
    FISTA with logsumexp smoothing of the max convolution.
    Adaptive temperature ramp for smooth-to-sharp transition.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    t_ista = 1.0
    step_size = 4.0 / np.sqrt(n)
    no_improve = 0
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if time.monotonic() - start >= time_budget:
            break
        progress = min(1.0, elapsed / max(time_budget, 1e-6))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        # Compute smoothed gradient
        c = compute_conv(a)
        max_c = np.max(c)
        shifted = tau * (c - max_c)
        max_shifted = np.max(shifted)
        exp_vals = np.exp(shifted - max_shifted)
        p = exp_vals / np.sum(exp_vals)

        # Gradient via FFT cross-correlation
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(a, N)
        grad = ifft(np.conj(P) * A_f).real[:n]

        # FISTA step
        a_new = project_simplex(a - step_size * grad)

        # Momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_ista * t_ista)) / 2.0
        a_y = a_new + ((t_ista - 1.0) / t_new) * (a_new - a)
        a_y = np.maximum(a_y, 0)
        s_y = a_y.sum()
        if s_y > 1e-15:
            a_y = a_y / s_y
        else:
            a_y = a_new.copy()

        a = a_new
        t_ista = t_new

        # Evaluate and track best
        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                step_size *= 0.7
                t_ista = 1.0
                no_improve = 0
                if step_size < 1e-6:
                    break

        step_size *= 0.9998
        it += 1

        if it % 50 == 0:
            print(f"    FISTA it{it}: obj={best_obj:.8f}, tau={tau:.1f}, step={step_size:.6f}")

    return best_a


def lp_line_search(a, g, max_t=1.5):
    """Line search along direction (1-t)*a + t*g. Returns best a and t."""
    best_obj = compute_obj(a)
    best_t = 0.0
    best_a = a.copy()
    t_vals = np.linspace(0.05, max_t, 20)
    for t in t_vals:
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_t = t
            best_a = a_try.copy()
    return best_a, best_t, best_obj

                best_delta = delta
