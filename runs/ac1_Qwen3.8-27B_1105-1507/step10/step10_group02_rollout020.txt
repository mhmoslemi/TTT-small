def compute_conv(a):

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""

def compute_subgradient_smoothed(a, conv, tau):
    """
    Compute smoothed subgradient of max(conv(a,a)) using softmax approximation.
    grad of conv[k] w.r.t. a[j] = 2*a[k-j] if 0<=k-j<n else 0.
    """
    n = len(a)
    conv_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - conv_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    Ar = fft(a, N)
    g_full = ifft(P * Ar).real
    g = 2.0 * g_full[:n]
    return g


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(1.0 - u / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """1D objective for coordinate subproblem."""
    if len(a_ki_lin) > 0:

    return 2.0 * n * max_val / denom
