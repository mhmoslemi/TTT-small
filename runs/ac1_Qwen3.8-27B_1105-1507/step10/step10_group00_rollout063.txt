def resample(a, n_new):
    """Resample sequence to new length using interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return a_new / s

"""
Hybrid FISTA + LP optimization for step function coefficient search.
Uses FFT-accelerated FISTA with adaptive log-sum-exp smoothing for fast initial
convergence, followed by intensive LP polishing with thorough line search,
n-exploration, and diverse restarts. Key improvements over prior state:
- FFT-based gradient for O(n log n) FISTA iterations
- Thorough line search with adaptive t range
- Multi-temp schedule for log-sum-exp
- Smaller-n warmup before large-n polish
- More aggressive restart strategy with spectral perturbations

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build sparse-friendly matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, timeout=8.0):
    """Solve LP: min M s.t. conv(a, g) <= M for all k, sum(g) = 1, g >= 0."""
    n = len(a)
    a_s = float(np.sum(a))
    if a_s < 1e-10:
        return None, None
    a_norm = a / a_s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    t0 = time.monotonic()
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        if time.monotonic() - t0 > timeout:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def fista_optimize(a, time_budget=60.0, tau_start=300.0, tau_end=40.0, verbose=False):
    """
    FISTA with log-sum-exp smooth approximation of max(conv(a,a)).
    Uses FFT for both conv and gradient computation (O(n log n) per iteration).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    y = a.copy()
    tk = 1.0
    it = 0
    best_it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / max(1.0, time_budget))
        tau = tau_start * (tau_end / tau_start) ** progress

        # Compute conv and weights
        fa = rfft(y, fft_size)
        conv = irfft(fa * fa, fft_size)[:2 * n - 1]
        conv_max = np.max(conv)
        scaled = tau * (conv - conv_max)
        scaled = np.clip(scaled, -500, 500)
        exp_scaled = np.exp(scaled)
        w_sum = np.sum(exp_scaled)
        weights = exp_scaled / w_sum

        # Gradient via FFT cross-correlation
        fw = rfft(weights, fft_size)
        fy = rfft(y, fft_size)
        cross = irfft(fw * np.conj(fy), fft_size)
        grad = 2.0 * cross[:n]

        # Step size
        step = 1.0 / (tau + 1.0)

        x_new = y - step * grad
        x_new = np.maximum(x_new, 0)
        s_new = float(np.sum(x_new))
        if s_new < 1e-10:
            x_new = y.copy()
            s_new = 1.0
        x_new = x_new / s_new

        # FISTA momentum update
        tk_new = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        y_new = x_new + ((tk - 1.0) / tk_new) * (x_new - a)
        y_new = np.maximum(y_new, 0)
        s_y = float(np.sum(y_new))
        if s_y > 1e-10:
            y_new = y_new / s_y
        tk = tk_new

        # Evaluate
        obj = compute_obj(x_new)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = x_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1

        a = x_new
        y = y_new
        it += 1

        # Restart if long stall
        if stall > 300:
            a = best_a.copy()
            y = best_a.copy()
            tk = 1.0
            stall = 0

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"      FISTA it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.1f}s")

    return best_a


def subgradient_fast(a, time_budget=30.0, verbose=False):
    """
    Fast subgradient method using FFT. Uses the peak of conv(a,a) as subgradient.
    With adaptive step size and momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    momentum = np.zeros(n)
    beta = 0.7
    lr = 0.01
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        # Compute conv
        fa = rfft(a, fft_size)
        conv = irfft(fa * fa, fft_size)[:2 * n - 1]
        k_max = int(np.argmax(conv))

        # Subgradient: d/dx[m] conv[k_max] = 2*x[k_max-m] if 0<=k_max-m<n
        j_range = np.arange(max(0, k_max - (n - 1)), min(n, k_max + 1))
        grad = np.zeros(n)
        valid_j = k_max - j_range
        valid_mask = (valid_j >= 0) & (valid_j < n)
        grad[j_range[valid_mask]] = 2.0 * a[valid_j[valid_mask]]

        # Update with momentum
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            momentum = np.zeros(n)
            lr = lr * 0.5
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        lr *= 0.9997
        it += 1

        if it - best_it > 2000:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = 0.005
            beta = 0.5
            best_it = it

        if verbose and it % 3000 == 0:
            print(f"      SG it{it}: obj={best_obj:.8f}, lr={lr:.5f}, t={time.monotonic()-start:.1f}s")

    return best_a

            a_try = np.maximum(a_try, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations with various shapes."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=np.float64)
    t = 2.0 * x - 1.0  # in [-1, 1]

    if mode == 0:  # Lorentzian U
        a = 1.0 / (0.05 + 4.0 * t ** 2)
    elif mode == 1:  # Power U
        a = np.abs(t) ** 3 + 0.01
    elif mode == 2:  # Bimodal
        a = np.exp(-((idx - 0.2 * n) ** 2) / (2.0 * (0.1 * n) ** 2)) + \
            np.exp(-((idx - 0.8 * n) ** 2) / (2.0 * (0.1 * n) ** 2)) + 0.005
    elif mode == 3:  # Sech U
        a = 1.0 / np.cosh(2.5 * t - 0.5)
    elif mode == 4:  # Inverted cosine
        a = 1.0 + 0.8 * t ** 2
    elif mode == 5:  # Exponential edges
        a = np.exp(2.5 * t ** 2) + 0.01
    elif mode == 6:  # Smooth step
        a = 1.0 / (1.0 + np.exp(-8.0 * t)) + 1.0 / (1.0 + np.exp(8.0 * t)) + 0.01
    elif mode == 7:  # Arcsine-like
        a = 1.0 / np.sqrt(0.1 + t ** 2)
    elif mode == 8:  # Power law
        a = np.abs(t) ** 0.7 + 0.005
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_seq(a, n_new):
    """Resample a sequence to a new length via linear interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    if n_old < 2 or n_new < 2:
        return np.ones(n_new) / n_new
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return a_new / s

                a_r = fista_optimize(a_r, time_budget=min(20.0, remaining * 0.2),
                                     tau_start=200.0, tau_end=40.0)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_r = lp_polish(a_r, time_budget=min(30.0, remaining * 0.35),
                                verbose=False, max_iters=20)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 7: Final Polish ===")
        alloc = min(remaining - 10, 150.0)
        if alloc > 20:
            # FISTA warm
            a_f = fista_optimize(best_seq, time_budget=alloc * 0.3, tau_start=100.0, tau_end=20.0)
            obj_f = compute_obj(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()

            # LP finish
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_l = lp_polish(best_seq, time_budget=min(remaining - 5, alloc * 0.6),
                                verbose=True, max_iters=50)
                obj_l = compute_obj(a_l)
                if obj_l < best_score:
                    best_score = obj_l
                    best_seq = a_l.copy()
                    print(f"  Final: obj={best_score:.8f} **")

    # Extra: one more LP round if time permits
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_extra = lp_polish(best_seq, time_budget=min(remaining - 5, 60.0), verbose=False, max_iters=25)
        obj_extra = compute_obj(a_extra)
        if obj_extra < best_score:
            best_score = obj_extra
            best_seq = a_extra.copy()
            print(f"  Extra: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
