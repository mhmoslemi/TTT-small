    rows = np.arange(2*n-1)[:, None]  # (2n-1, 1)
    cols = np.arange(n)[None, :]  # (1, n)

    return A

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based alternating minimization with GLOP solver, adaptive line search,
structured multi-start initialization, Nesterov gradient polishing, and
multi-length exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft

# ============================================================
# Core computation functions
# ============================================================

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a*a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    rows = np.arange(2 * n - 1)[:, None]
    cols = np.arange(n)[None, :]
    idx = rows - cols
    valid = (idx >= 0) & (idx < n)
    return np.where(valid, a_norm[np.clip(idx, 0, n - 1)], 0.0)

# ============================================================
# LP Solver
# ============================================================

def solve_best_response(a, solver_priority=None):
    """
    Solve: min M s.t. (a_norm * g)_k <= M, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_priority is None:
        solver_priority = [cp.GLOP, cp.CBC, cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    
    for solver in solver_priority:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def smoothed_gradient(a, tau):
    """Gradient of tau * log(sum(exp(tau * (a*a)_k))) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]

def make_uniform(n):
    return np.ones(n) / n

def make_u_shaped(n, strength=1.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + strength * x**2
    return a / a.sum()

def make_bell(n, width=0.3):
    x = np.linspace(-1, 1, n)
    a = np.exp(-x**2 / (2 * width**2)) + 0.01
    return a / a.sum()

def make_cosine(n):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x / 2)**2 + 0.05
    return a / a.sum()

def make_two_peak(n, separation=0.5):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - (1-separation)/2)**2) / 0.02) + np.exp(-((x - (1+separation)/2)**2) / 0.02) + 0.01
    return a / a.sum()

def random_init(n, rng, alpha=1.0):
    """Random initialization with power-law structure."""
    x = np.linspace(0, 1, n)
    a = rng.gamma(alpha, 1.0, size=n)
    a = a + 0.01
    return a / a.sum()
