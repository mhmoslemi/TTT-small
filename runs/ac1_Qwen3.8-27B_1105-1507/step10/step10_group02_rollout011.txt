"""
Optimizes step function coefficients using binomial-coefficient initialization,
active-set Newton refinement, aggressive LP polishing, and exact coordinate descent.
Key improvements over previous: principled initialization from flat-convolution theory,
direct active-set solving, and more aggressive LP iterations.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None


def binomial_init(n):
    """Initialize with a[k] = binom(2k,k)/4^k, the flat-convolution sequence."""
    a = np.zeros(n)
    for k in range(n):
        a[k] = math.comb(2*k, k) / (4.0 ** k)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """Exact coordinate descent with golden section 1D sub-solves."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 3 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v

            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)

            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(c_k[(ks == 2 * i)[0]]) if (ks == 2 * i).any() else 0.0

            # Define 1D objective as a helper
            def f_1d(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            scale = max(v, S / n, 1e-6)
            candidates = [0.0, v * 0.5, v, v * 1.5, v * 2.0,
                          S_other * 0.3, S_other * 0.7, S_other]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(40):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            obj_now = 2.0 * n * float(np.max(conv)) / (S * S)
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, now={obj_now:.8f}, t={elapsed:.1f}s")

    return best_a


def lp_polish_aggressive(a, time_budget=120.0, verbose=False, seed=42):
    """Aggressive LP-based polishing with many iterations and perturbations."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall_count = 0
    max_stall = 5

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Try multiple blending factors
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > max_stall:
                # Perturb to escape
                pert_size = 0.02 * (1 + stall_count * 0.5)
                pert = rng.normal(0, pert_size, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0

    return best_a


def active_set_newton(a, max_iters=50, tol=1e-10):
    """
    Try to solve the active-set system directly using Newton's method.
    At the optimum, conv(a,a)[k] = M for active positions k.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    for it in range(max_iters):
        conv = compute_conv(a)
        M = float(np.max(conv))
        obj = 2.0 * n * M

        # Identify active set (within tolerance)
        tol_active = M * 0.01 + 1e-12
        active = np.where(conv >= M - tol_active)[0]
        # Keep only the most active
        n_active = min(len(active), n - 1)
        if n_active < 2:
            break
        active = active[np.argsort(conv[active])[::-1][:n_active]]

        # Build the system: conv[k] = M for k in active, sum(a) = 1
        # Unknowns: a (n vars) + M (1 var) = n+1 unknowns
        # Equations: n_active + 1
        # If n_active = n-1: n equations in n+1 unknowns -> underdetermined
        # Fix M and solve for a, or use a regularization

        # Compute Jacobian
        # J[k][i] = d(conv[k])/da[i]
        J = np.zeros((n_active + 1, n + 1))
        for idx, k in enumerate(active):
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    if 2 * i == k:
                        J[idx, i] = a[ki]
                    else:
                        J[idx, i] = 2.0 * a[ki]
            J[idx, n] = -1.0  # d/dM of (conv[k] - M) = -1

        # Last row: sum(a) = 1
        J[n_active, :n] = 1.0
        J[n_active, n] = 0.0

        # Residual
        res = np.zeros(n_active + 1)
        for idx, k in enumerate(active):
            res[idx] = conv[k] - M
        res[n_active] = float(np.sum(a)) - 1.0

        # Solve J @ delta = -res
        try:
            # Use least squares if underdetermined
            if n_active + 1 <= n + 1:
                delta, _, _, _ = np.linalg.lstsq(J, -res, rcond=None)
            else:
                delta = np.linalg.solve(J, -res)
        except np.linalg.LinAlgError:
            break

        # Line search
        alpha = 1.0
        for _ in range(30):
            a_new = a + alpha * delta[:n]
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                alpha *= 0.5
                continue
            a_new = a_new / s_new
            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < obj:
                break
            alpha *= 0.5

        if alpha < 1e-8:
            break

        a = a + alpha * delta[:n]
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            break
        a = a / s

        obj_now = 2.0 * n * float(np.max(compute_conv(a)))
        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()

        if abs(obj_now - obj) < tol:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return binomial_init(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** rng.uniform(0.5, 3.0)
        a = a / a.sum()
    elif mode == 3:
        # Decaying exponential
        rate = rng.uniform(1, 5)
        x = np.arange(n)
        a = np.exp(-rate * x / n)
        a = a / a.sum()
    elif mode == 4:
        # Power law decay
        alpha = rng.uniform(0.5, 2.5)
        x = np.arange(1, n + 1)
        a = x ** (-alpha)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a
