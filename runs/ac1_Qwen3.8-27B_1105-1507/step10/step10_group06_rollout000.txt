import numpy as np
import cvxpy as cp

from scipy.fft import rfft, irfft, fft, ifft


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_fft(a)
    return 2.0 * n * float(np.max(conv))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm
    return A


def solve_lp_warm(a_norm, g_warm=None):
    """
    Solve LP: min M s.t. (a_norm*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a_norm)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    # Add warm start objective to guide solver
    if g_warm is not None:
        constraints.append(cp.sum((g - g_warm)**2) <= 0.5)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.GLOP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)

    prob2 = cp.Problem(cp.Minimize(M), [A @ g <= M, cp.sum(g) == 1])

def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of (a*a)_k.
    Uses FFT for efficiency. Returns gradient of shape (n,).
    """
    n = len(a)
    conv = compute_conv_fft(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def multiplicative_weight_update(a, p, eta):
    """
    Multiplicative weight (mirror descent) update on the simplex.
    a_j <- a_j * exp(-eta * (p*a)_j) / Z
    where p is the weight distribution on convolution peaks.
    """
    n = len(a)
    # Compute (p * a)_j for j = 0..n-1
    N = 1
    while N < 2 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * A).real[:n]
    
    # Multiplicative update
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_new = log_a - eta * cross
    # Shift to avoid overflow
    log_a_new -= np.max(log_a_new)
    a_new = np.exp(log_a_new)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s


def fista_smoothed(a, time_budget, tau_init=50.0, tau_max=5000.0):
    """
    FISTA (Fast Iterative Shrinkage-Thresholding Algorithm) on the smoothed
    objective with simplex projection.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    z = a.copy()
    t_k = 1.0
    tau = tau_init

    while time.monotonic() - start < time_budget:
        g = smoothed_gradient_fft(z, tau)
        
        # Gradient step
        lr = 1.0 / (float(np.linalg.norm(g)) + 1e-10)
        lr = min(lr, 0.5 / np.sqrt(n))
        x = z - lr * g
        
        # Project onto simplex {x >= 0, sum(x) = 1}
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx < 1e-10:
            x = best_a.copy()
        else:
            x = x / sx

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = x + ((t_k - 1.0) / t_new) * (x - a)
        z = np.maximum(z, 0)
        sz = float(np.sum(z))
        if sz < 1e-10:
            z = x.copy()
        else:
            z = z / sz
        a = x
        t_k = t_new

        obj_new = compute_obj_normalized(a)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()

        tau = min(tau * 1.05, tau_max)

        # Periodic restart to avoid FISTA oscillation
        if int(t_k) % 200 == 0:
            t_k = 1.0
            z = a.copy()

    return best_a

            # Compute peak weights
            conv = compute_conv_fft(a)
            max_conv = float(np.max(conv))
            tau_mw = 100.0
            p = np.exp(tau_mw * (conv - max_conv))
            p = p / p.sum()
            
            a_mw = multiplicative_weight_update(a, p, eta=0.02)
            obj_mw = compute_obj_normalized(a_mw)
            if obj_mw < best_obj:
                best_obj = obj_mw
                best_a = a_mw.copy()
                a = a_mw.copy()
            else:
                a = best_a.copy()
            continue

        # Solve LP
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp_warm(a, g_prev)
        lp_time = time.monotonic() - t0

        if g_lp is None:
            break
        g_prev = g_lp

        # Adaptive blending with line search
        candidates_t = [t_blend, t_blend * 0.5, 0.3, 0.5, 0.7, 1.0]
        if iter_count > 50:
            candidates_t.extend([0.1, 0.2, 0.05])

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend

        for t in candidates_t:
            t = min(max(t, 0.005), 1.0)
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            t_blend = max(0.05, best_local_t * 0.95)
            stall_count = 0
            if verbose and iter_count % 50 == 0:
                print(f"    LP+MW {iter_count}: obj={best_obj:.10f}, t={t_blend:.4f}, "
                      f"lp_time={lp_time:.3f}s")
        else:
            stall_count += 1
            if stall_count >= 2:
                # Escalating perturbation
                strength = 0.005 + 0.003 * min(stall_count, 8)
                # Fourier perturbation
                pert = np.zeros(n)
                for k in range(1, 6):
                    freq = 2 * np.pi * k / n
                    amp = strength * rng.uniform(0.5, 1.5) / k
                    phase = rng.uniform(0, 2 * np.pi)
                    pert += amp * np.cos(freq * np.arange(n) + phase)
                a = best_a * np.exp(pert)
                a = normalize(a)
                t_blend = min(t_blend * 1.3, 0.8)
                stall_count = 0
            elif stall_count == 1:
                a = best_a + 0.003 * rng.standard_normal(n) * best_a
                a = normalize(a)

        # Every 100 iterations, try a full MW step
        if iter_count % 100 == 0:
            conv = compute_conv_fft(best_a)
            max_conv = float(np.max(conv))
            p = np.exp(200.0 * (conv - max_conv))
            p = p / p.sum()
            a_test = multiplicative_weight_update(best_a, p, eta=0.01)
            obj_test = compute_obj_normalized(a_test)
            if obj_test < best_obj:
                best_obj = obj_test
                best_a = a_test.copy()

    if verbose:
        print(f"    LP+MW enhanced: {iter_count} iters, best_obj={best_obj:.10f}")
    return best_a


def anti_peak_refinement(a, time_budget, seed=42):
    """
    Direct anti-peak refinement: identify peak locations and
    use targeted coordinate updates to reduce them.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        conv = compute_conv_fft(a)
        max_conv = float(np.max(conv))
        peak_indices = np.where(conv > max_conv * 0.95)[0]
        
        # For each peak, compute which pairs contribute
        # Gradient direction: reduce a_j where a_{k-j} is large for peak k
        grad = np.zeros(n)
        for k in peak_indices[:5]:  # Focus on top 5 peaks
            k = int(k)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += a[idx]
        
        # Normalize gradient
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break
        
        # Damped update: move opposite to gradient, projected to simplex
        for eta in [0.01, 0.05, 0.1, 0.2, 0.5]:
            a_try = a - eta * grad / g_norm
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break
        else:
            # No improvement, try perturbation
            a = best_a + 0.002 * rng.standard_normal(n) * best_a
            a = normalize(a)
            if time.monotonic() - start > time_budget * 0.8:
                a = fista_smoothed(best_a, time_budget=0.05, tau_init=100.0)
                obj_check = compute_obj_normalized(a)
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a.copy()

    return best_a


def coordinate_descent_refine(a, time_budget):
    """
    Coordinate-wise refinement: for each position, find the optimal value
    given all others fixed (1D minimization of piecewise-linear function).
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    # Precompute: for coordinate j, changing a_j from v to v+delta
    # affects conv[k] for k in [j, j+n-1] by delta * a_{k-j}
    # We want to minimize max_k (conv_base[k] + delta * a_{k-j})
    # subject to the constraint sum(a) = 1 (so other coords adjust)
    
    # Simplified: just do gradient-based coordinate updates
    order = np.arange(n)
    np.random.default_rng(42).shuffle(order)
    
    while time.monotonic() - start < time_budget:
        improved = False
        for j in order:
            j = int(j)
            # Try small perturbations of a[j]
            base = a[j]
            if base < 1e-8:
                continue
            for frac in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                delta = (frac - 1.0) * base
                a[j] = base + delta
                # Redistribute: subtract delta/n from all others
                a[:j] -= delta / (n - 1) if n > 1 else 0
                a[j+1:] -= delta / (n - 1) if n > 1 else 0
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s < 1e-10:
                    a[j] = base
                    continue
                a = a / s
                obj_try = compute_obj_normalized(a)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    # Revert
                    a = best_a.copy()
            
        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def random_reseed(a, rng, strength=0.1):
    """Random perturbation to escape local minima."""
    n = len(a)
    # Random multiplicative perturbation
    pert = rng.uniform(1 - strength, 1 + strength, size=n)
    a_new = a * pert
    return normalize(a_new)
