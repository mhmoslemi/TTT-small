from scipy.optimize import minimize as scipy_minimize

def compute_conv_fft(a):
    """Fast convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a) = 1."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:

    a = a / s
    conv = compute_conv_fft(a)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_lp(a, prev_g=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP/MM (Majorization-Minimization) with warm starts, high-precision
smoothed gradient descent with aggressive tau scheduling, support optimization,
combinatorial initialization, and multi-length exploration.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import time


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, inf)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:
            continue

    return None, np.inf


def lp_mm_optimize(a, time_budget, verbose=True, max_iter=10000):
    """LP/MM optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()

            break

def smoothed_grad_obj(a, tau):
    """Compute smoothed objective value and gradient."""
    n = len(a)
    a = a / float(np.sum(a))
    conv = compute_conv_fft(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    obj_val = max_conv + np.log(sum_exp) / tau

    # Gradient via FFT
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A_f = fft(a, N)
    g_full = ifft(P * np.conj(A_f)).real
    grad = 2.0 * g_full[:n] - 2.0 * np.sum(g_full[:n] * a) * a  # center for simplex constraint

    return obj_val, grad

    n = len(a)

        return a

        # Nesterov look-ahead

        y = np.maximum(y, 0)

        exp_vals = np.exp(tau * (conv - max_conv))

        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:

def diverse_init(n, rng, mode=0):

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def log_param_optimize(a, time_budget, tau_min=200, tau_max=10000, verbose=False):

    best_a = a.copy()

    start = time.monotonic()

    tau_stages = [tau_min, tau_min * 3, tau_min * 10, tau_min * 30, tau_max]
    current_b = b0.copy()

    for tau in tau_stages:
        if time.monotonic() - start >= time_budget:
            break

        def obj_grad(b_vec):

            a_supp = np.exp(b_shifted)

            a_full = np.zeros(full_len)
            a_full[support] = a_supp

            conv = compute_conv_fft(a_full)

            # Chain rule for log-param with softmax normalization
            grad_b = a_supp * (grad_full[support] - np.sum(a_supp * grad_full[support]))

            return obj_val, grad_b

        try:

            if res.x is not None:

        except Exception:
            pass

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP/MM (Majorization-Minimization) with warm starts, high-precision
smoothed gradient descent with aggressive tau scheduling, support optimization,
combinatorial initialization, and multi-length exploration.
Key improvements: extended LP/MM runtime, better line search, dual-guided
perturbations, combinatorial inits, and aggressive multi-n sweep.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import time


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, inf)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:
            continue

    return None, np.inf


def lp_mm_optimize(a, time_budget, verbose=True, max_iter=50000):
    """LP/MM optimization with adaptive line search and perturbation escapes."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5
    it = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget and it < max_iter:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        it += 1

        # Adaptive line search
        t_candidates = np.linspace(0.1, 1.0, 12)
        t_candidates = np.concatenate([t_candidates,
                                        [max(0.05, last_t * 0.5),
                                         max(0.05, last_t * 0.7),
                                         last_t,
                                         min(1.0, last_t * 1.3)]])
        t_candidates = np.unique(np.clip(t_candidates, 0.05, 1.0))

        best_t_obj = best_obj
        best_t = 0
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-13:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = max(best_t, 0.1)
            stall = 0
            if verbose and it % 100 == 0:
                print(f"    LP/MM it{it}: obj={best_obj:.8f}, t={last_t:.3f}, t_elapsed={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall == 3:
                # Perturbation to escape local optimum
                pert = rng.dirichlet(np.ones(n) * 10)
                a_pert = 0.95 * a + 0.05 * pert
                a_pert = a_pert / float(np.sum(a_pert))
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj + 0.01:
                    a = a_pert
                    stall = 0
            elif stall > 8:
                break

    return best_a


def high_tau_gd(a, time_budget, tau_start=500, tau_end=30000, verbose=True):
    """High-precision smoothed gradient descent with Nesterov acceleration."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.4

        # Nesterov look-ahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv_fft(y)
        max_conv = float(np.max(conv))
        # Numerical safety for exp
        scaled = tau * (conv - max_conv)
        scaled = np.maximum(scaled, -50)
        exp_vals = np.exp(scaled)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(y, N)
        g_full = ifft(P * np.conj(A_f)).real
        g = 2.0 * g_full[:n]

        # Project onto tangent space (sum preservation)
        g = g - np.sum(g * y) * y

        # Adaptive step size
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        lr = min(1.0 / (g_norm + 1e-10), 3.0 / np.sqrt(n))
        lr = max(lr, 1e-9)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        beta = min(0.98, beta + 0.00003)

        if verbose and it % 3000 == 0:
            print(f"    HTGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.0f}s")

        if it - best_it > 10000 and it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            beta = 0.9
            best_it = it
            if it > 30000:
                break

    return best_a


def support_optimize(a, time_budget, verbose=True):
    """Try different support sizes and re-optimize on each."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    sorted_vals = np.sort(a)[::-1]
    for frac in [0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.85, 0.9]:
        if time.monotonic() - start > time_budget - 5:
            break
        k = max(3, int(n * frac))
        if k >= n:
            continue
        threshold = sorted_vals[k - 1]
        support = np.where(a >= threshold)[0]
        if len(support) < 3:
            continue

        a_shrunk = np.zeros(n)
        a_shrunk[support] = a[support]
        a_shrunk = a_shrunk / float(np.sum(a_shrunk))

        alloc = min(25, time_budget - (time.monotonic() - start) - 5)
        if alloc > 8:
            a_shrunk = lp_mm_optimize(a_shrunk, time_budget=alloc, verbose=False, max_iter=200)

        obj_shrunk = compute_obj(a_shrunk)
        if obj_shrunk < best_obj - 1e-12:
            best_obj = obj_shrunk
            best_a = a_shrunk.copy()
            if verbose:
                print(f"    Support frac={frac}: obj={best_obj:.8f}, support={len(support)}")

    return best_a


def combinatorial_init(n, rng):
    """Initialize using combinatorial structures for good sumset properties."""
    a = np.zeros(n)
    k = max(10, int(np.sqrt(2 * n) * 3))
    k = min(k, n - 1)

    best_a = None
    best_score = np.inf

    # Try several modular arithmetic patterns
    for trial in range(20):
        mod_base = n + trial * 7
        positions = set()
        for i in range(2 * k):
            pos = (i * i * (3 + trial) + i * (7 + trial * 2) + trial * 11) % mod_base
            if pos < n:
                positions.add(pos)
            pos2 = (i * (i + 1) // 2 + trial * 5) % mod_base
            if pos2 < n:
                positions.add(pos2)

        if len(positions) < k // 3:
            continue

        a_try = np.zeros(n)
        for pos in positions:
            a_try[pos] = 1.0
        a_try = a_try / float(np.sum(a_try))
        score = compute_obj(a_try)
        if score < best_score:
            best_score = score
            best_a = a_try.copy()

    if best_a is not None:
        return best_a

    # Fallback: random sparse
    positions = rng.choice(n, size=min(k, n), replace=False)
    a = np.zeros(n)
    a[positions] = 1.0
    return a / float(np.sum(a))


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # W-shaped
        x = np.linspace(0, 1, n)
        a = x**2 * (1 - x)**2 + 0.01
        a = a / a.sum()
    elif mode == 3:
        # U-shaped
        x = np.linspace(0, 1, n)
        a = (x - 0.5)**2 + 0.05
        a = a / a.sum()
    elif mode == 4:
        # Comb
        a = np.zeros(n)
        a[::3] = 1.0
        a = a / a.sum()
    elif mode == 5:
        # Two clusters
        a = np.zeros(n)
        a[:n//4] = 1.0
        a[3*n//4:] = 1.0
        a = a / a.sum()
    elif mode == 6:
        # Cosine
        x = np.linspace(0, np.pi, n)
        a = np.cos(x)**2 + 0.01
        a = a / a.sum()
    elif mode == 7:
        # Ramp
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 8:
        # Sparse random
        k = max(5, n // 10)
        a = np.zeros(n)
        positions = rng.choice(n, size=k, replace=False)
        a[positions] = rng.uniform(0.5, 1.5, k)
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a
