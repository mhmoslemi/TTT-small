"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses aggressive LP-based Frank-Wolfe, support pruning, multi-n exploration,
and high-tau smoothed gradient descent.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def convolve_ab(a, b):
    """Compute linear convolution of a and b via FFT."""
    n_a, n_b = len(a), len(b)
    size = 1
    while size < n_a + n_b - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n_a + n_b - 1]


def lp_solve(a, solver_list=None):
    """
    Solve the LP: minimize M s.t. (a*g)_k <= M for all k, g >= 0, sum(g) = 1.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    if n > 3000:
        return None, None
    
    # Build convolution matrix: A[k,j] gives (a*g)_k = sum_j a[k-j]*g[j]
    # A is (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.reshape(A, (2*n-1, n)) @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    
    return None, None


def fast_lp_solve(a, solver_list=None):
    """
    Faster LP solve using the structure. For moderate n, builds the matrix explicitly.
    Uses a more efficient formulation.
    """
    n = len(a)
    if n > 3000:
        return None, None
    
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # This is a Toeplitz matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        A[j:j + length, j] = a[:length]
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Use the fact that max_k (A@g)_k <= M
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g_var.value).flatten(), 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    
    return None, None


def lp_frank_wolfe(a, time_budget, verbose=False, max_iters=200, seed=42):
    """
    LP-based Frank-Wolfe (conditional gradient) method.
    At each iteration:
    1. Solve LP to get g (minimizes max_k (a_current * g)_k)
    2. Take step towards g with diminishing step size
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    
    stall_count = 0
    stall_threshold = 15
    
    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        
        # Solve LP
        g_lp, M_lp = fast_lp_solve(a, solver_list)
        if g_lp is None:
            # Fallback: use smoothed gradient step
            break
        
        # Compute step size: diminishing
        # t_k = 2 / (k + 2) gives guaranteed convergence
        # But we can be more aggressive initially
        if it < 10:
            t_step = 0.8  # Aggressive initial steps
        elif it < 30:
            t_step = 0.5
        elif it < 60:
            t_step = 0.3
        else:
            t_step = 2.0 / (it + 2)
        
        # Try multiple step sizes and pick the best
        t_candidates = [t_step, min(1.0, t_step * 1.5), max(0.05, t_step * 0.5)]
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
        
        if best_local_obj < best_obj - 1e-15:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a
            stall_count = 0
            if verbose and it % 10 == 0:
                print(f"    FW it{it:3d}: obj={best_obj:.10f}, M_lp={M_lp:.6f}")
        else:
            stall_count += 1
            # Perturb to escape
            pert = rng.normal(0, 0.01, n)
            a = a + pert
            a = np.maximum(a, 0)
            a = a / a.sum()
            
            if stall_count > stall_threshold:
                if verbose:
                    print(f"    FW stalled at it{it}, obj={best_obj:.10f}")
                break
    
    return best_a


def smoothed_grad(a, tau):
    """Compute gradient of smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 2 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def gd_adam(a, time_budget, verbose=False, tau_levels=None, lr=0.01, max_iters=50000):
    """
    Adam optimizer with Nesterov on the smoothed objective.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if tau_levels is None:
        tau_levels = [5, 20, 50, 100, 200, 500, 1000, 2000, 5000]
    
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    for tau in tau_levels:
        it_level = 0
        while time.monotonic() - start < time_budget:
            if it_level >= max(100, max_iters // len(tau_levels)):
                break
            t += 1
            it_level += 1
            
            # Compute gradient
            g, _ = smoothed_grad(a, tau)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * g ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
            else:
                a_new = a_new / s_new
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            
            a = a_new
    
    return best_a


def gd_simple(a, time_budget, verbose=False, tau_levels=None, lr_scale=2.0):
    """
    Simple projected gradient descent with Nesterov.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if tau_levels is None:
        tau_levels = [10, 50, 150, 400, 1000, 2500, 5000, 10000]
    
    beta = 0.9
    v = np.zeros(n)
    it = 0
    
    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        it_level = 0
        max_per_level = 3000
        
        while time.monotonic() - start < time_budget:
            if it_level >= max_per_level:
                break
            it_level += 1
            it += 1
            
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = a.copy()
            y = y / sy
            
            g, _ = smoothed_grad(y, tau)
            
            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
            else:
                a_new = a_new / s_new
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
            
            a = a_new
            lr = max(1e-8, lr * 0.99999)
    
    return best_a


def support_prune(a, threshold=1e-6, time_budget=30, verbose=False):
    """
    Try to improve by zeroing out small elements and re-optimizing.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a_norm = a / s
    best_a = a_norm.copy()
    best_obj = compute_obj(a_norm)
    
    # Sort by value
    sorted_idx = np.argsort(a_norm)
    
    # Try zeroing out progressively more small elements
    for frac in [0.02, 0.05, 0.10, 0.15, 0.20]:
        k = int(n * frac)
        if k == 0:
            continue
        a_try = a_norm.copy()
        a_try[sorted_idx[:k]] = 0
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        
        # Quick GD to see if it helps
        a_opt = gd_simple(a_try, time_budget=min(10, time_budget), tau_levels=[50, 200, 800])
        obj_opt = compute_obj(a_opt)
        
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    Pruned {frac*100:.0f}%: obj={best_obj:.10f}")
    
    return best_a


def resample_a(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)

    s = float(np.sum(a_new))
    if s < 1e-10:

    return a_new / s


def make_init(n, mode, rng=None):
    """Create diverse initial sequences."""
    if rng is None:
        rng = np.random.default_rng(0)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x**2, 0.0001))
        return a / a.sum()
    elif mode == 3:
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 4:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, np.pi, n)
        a = np.maximum(0.5 + 0.5 * np.cos(x), 0.001)
        return a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [2, 3, 5, 7]:
            a += 0.15 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 6))
        return a / a.sum()
    elif mode == 8:
        x = np.linspace(0, 1, n)
        a = np.where((x > 0.15) & (x < 0.85), 1.0,
                     np.maximum(0.6 - np.abs(x - 0.5) * 2, 0.01))
        return a / a.sum()
    elif mode == 9:
        # Difference set inspired: try to make autocorrelation as flat as possible
        a = np.ones(n) / n
        # Add small random perturbation
        a += rng.normal(0, 0.1, n) / n
        a = np.maximum(a, 0)
        return a / a.sum()
    elif mode == 10:
        # Gaussian window
        x = np.arange(n)
        sigma = n / 4
        a = np.exp(-0.5 * ((x - n/2) / sigma) ** 2)
        return a / a.sum()
    elif mode == 11:
        # Power law decay from center
        x = np.abs(np.arange(n) - n / 2)
        a = 1.0 / (x + 1) ** 0.5
        return a / a.sum()
    else:
        return np.ones(n) / n

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses aggressive LP-based Frank-Wolfe with line search, composite gradient steps,
support optimization, multi-n exploration, and high-tau smoothed GD polishing.
Key improvements over previous: better step size via line search, composite steps,
more diverse restarts, and adaptive support pruning.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def convolve_ab(a, b):
    """Compute linear convolution of a and b via FFT."""
    n_a, n_b = len(a), len(b)
    size = 1
    while size < n_a + n_b - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n_a + n_b - 1]


def build_conv_matrix(a):
    """Build the convolution matrix A where (A@g)_k = (a*g)_k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        A[j:j + length, j] = a[:length]
    return A


def lp_solve(a, solver_list=None):
    """
    Solve LP: minimize M s.t. (a*g)_k <= M for all k, g >= 0, sum(g) = 1.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    if n > 3000:
        return None, None

    A = build_conv_matrix(a)

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64).flatten(), 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue

    return None, None


def smoothed_grad(a, tau):
    """Compute gradient of smoothed max of conv(a,a). Returns (grad, exact_max)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 2 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def lp_frank_wolfe_v2(a, time_budget, verbose=False, max_iters=300, seed=42):
    """
    Enhanced LP-based Frank-Wolfe with:
    - Line search for step size
    - Composite steps (LP direction + gradient direction)
    - Adaptive perturbation on stall
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]

    stall_count = 0
    stall_threshold = 20
    last_improve_it = 0
    composite_weight = 0.0  # Start with pure LP, increase if stalling

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 5:
            break

        # Solve LP
        g_lp, M_lp = lp_solve(a, solver_list)
        if g_lp is None:
            break

        # Compute gradient direction (for composite step)
        g_grad, _ = smoothed_grad(a, tau=200)

        # Line search: try multiple step sizes
        base_t = 2.0 / (it + 2.0)
        if it < 5:
            base_t = 0.9
        elif it < 15:
            base_t = 0.7
        elif it < 40:
            base_t = 0.4
        elif it < 80:
            base_t = 0.25

        t_candidates = [base_t, min(1.0, base_t * 1.5), max(0.05, base_t * 0.5),
                        max(0.01, base_t * 0.25)]

        # Composite direction
        if composite_weight > 0:
            direction = (1 - composite_weight) * g_lp + composite_weight * (-g_grad / (np.linalg.norm(g_grad) + 1e-10))
            direction = np.maximum(direction, 0)
            sd = float(np.sum(direction))
            if sd > 1e-10:
                direction = direction / sd
            else:
                direction = g_lp
        else:
            direction = g_lp

        best_local_obj = best_obj
        best_local_a = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * direction
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a
            stall_count = 0
            last_improve_it = it
            # Reduce composite weight when making progress
            composite_weight = max(0.0, composite_weight * 0.95)
            if verbose and it % 15 == 0:
                print(f"    FW it{it:3d}: obj={best_obj:.10f}, M={M_lp:.6f}, cw={composite_weight:.2f}")
        else:
            stall_count += 1
            # Increase composite weight when stalling
            composite_weight = min(0.5, composite_weight + 0.05)

            # Perturb to escape
            perturb_scale = 0.01 + 0.005 * (stall_count // 5)
            pert = rng.normal(0, perturb_scale, n)
            a = a + pert
            a = np.maximum(a, 0)
            a = a / a.sum()

            if stall_count > stall_threshold:
                # Larger restart
                pert = rng.normal(0, 0.05, n)
                a = best_a + pert
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                stall_count = 0
                composite_weight = 0.0

                if stall_count == 0 and it - last_improve_it > 40:
                    if verbose:
                        print(f"    FW: large restart at it{it}, obj={best_obj:.10f}")

    return best_a


def gd_nesterov(a, time_budget, verbose=False, tau_levels=None, lr_scale=2.0):
    """Nesterov accelerated projected gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 20, 50, 150, 400, 1000, 2500, 5000, 10000]

    beta = 0.9
    v = np.zeros(n)
    it = 0

    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        it_level = 0
        max_per_level = 4000
        local_best = best_obj

        while time.monotonic() - start < time_budget:
            if it_level >= max_per_level:
                break
            it_level += 1
            it += 1

            # Nesterov lookahead
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            # Compute gradient
            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            # Update
            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                local_best = obj_new

            a = a_new
            lr = max(1e-8, lr * 0.99999)

        if verbose:
            print(f"    GD tau={tau:5d}: obj={best_obj:.10f}, iters={it_level}")

    return best_a


def gd_adam(a, time_budget, verbose=False, tau_levels=None, lr=0.005):
    """Adam optimizer on the smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [10, 50, 200, 500, 1000, 3000, 8000]

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    max_per_level = 10000

    for tau in tau_levels:
        it_level = 0
        while time.monotonic() - start < time_budget:
            if it_level >= max_per_level:
                break
            it_level += 1
            t += 1

            g, _ = smoothed_grad(a, tau)

            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * g ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new

    return best_a


def support_prune(a, time_budget=60, verbose=False):
    """Try to improve by zeroing out small elements and re-optimizing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a_norm = a / s
    best_a = a_norm.copy()
    best_obj = compute_obj(a_norm)
    start = time.monotonic()

    sorted_idx = np.argsort(a_norm)

    fracs = [0.01, 0.02, 0.05, 0.10, 0.15, 0.20, 0.30]
    for frac in fracs:
        if time.monotonic() - start > time_budget - 10:
            break
        k = int(n * frac)
        if k == 0 or k >= n:
            continue

        a_try = a_norm.copy()
        a_try[sorted_idx[:k]] = 0
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Quick GD refinement
        alloc = min(20, time_budget - (time.monotonic() - start))
        if alloc < 5:
            break
        a_opt = gd_nesterov(a_try, time_budget=alloc, tau_levels=[50, 200, 800, 2000])
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    Prune {frac*100:.0f}%%: obj={best_obj:.10f} **")

    # Also try zeroing out large elements (counterintuitive but sometimes helps)
    for frac in [0.02, 0.05]:
        if time.monotonic() - start > time_budget - 10:
            break
        k = int(n * frac)
        a_try = a_norm.copy()
        a_try[sorted_idx[-k:]] = 0  # Zero out largest
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        alloc = min(15, time_budget - (time.monotonic() - start))
        if alloc < 5:
            break
        a_opt = gd_nesterov(a_try, time_budget=alloc, tau_levels=[50, 200, 800])
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    Prune top {frac*100:.0f}%%: obj={best_obj:.10f} **")

    return best_a


def resample_a(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_init(n, mode, rng=None):
    """Create diverse initial sequences of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.0001))
        return a / a.sum()
    elif mode == 3:
        c = n // 2
        w = max(1, n // 6)
        a = np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 4:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, np.pi, n)
        a = np.maximum(0.5 + 0.5 * np.cos(x), 0.001)
        return a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [2, 3, 5, 7, 11]:
            a += 0.1 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:
        x = np.abs(np.arange(n) - n / 2)
        a = 1.0 / (x + 1) ** 0.7
        return a / a.sum()
    elif mode == 8:
        x = np.linspace(0, 1, n)
        a = np.where((x > 0.1) & (x < 0.9), 1.0,
                     np.maximum(0.7 - np.abs(x - 0.5) * 2, 0.01))
        return a / a.sum()
    elif mode == 9:
        # Two peaks
        c1, c2 = n // 3, 2 * n // 3
        w = max(1, n // 10)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 10:
        # Broad Gaussian
        x = np.arange(n)
        sigma = n / 3
        a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
        return a / a.sum()
    elif mode == 11:
        # Alternating pattern with envelope
        x = np.linspace(0, 1, n)
        envelope = np.maximum(1 - 4 * (x - 0.5) ** 2, 0.01)
        a = envelope * (1 + 0.3 * np.cos(2 * np.pi * 3 * x))
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 12:
        # Power law
        x = np.abs(np.arange(n) - n / 2)
        a = 1.0 / (x + 1) ** 1.2
        return a / a.sum()
    elif mode == 13:
        # Flat center, decay edges
        center = n // 2
        half = n // 4
        a = np.ones(n)
        a[:max(0, center - half)] *= 0.3
        a[min(n, center + half):] *= 0.3
        return a / a.sum()
    else:
        return np.ones(n) / n


def symmetric_optimize(a, time_budget, verbose=False):
    """Optimize under palindromic constraint a[i] = a[n-1-i]."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    half = (n + 1) // 2

    # Work with half-vector, enforce symmetry via full vector
    a_half = np.zeros(half)
    for i in range(half):
        a_half[i] = (a[i] + a[n - 1 - i]) / 2.0
    a_half = np.maximum(a_half, 0)

    # Reconstruct full
    a_sym = np.zeros(n)
    for i in range(half):
        a_sym[i] = a_half[i]
        a_sym[n - 1 - i] = a_half[i]
    ss = float(np.sum(a_sym))
    if ss > 1e-10:
        a_sym = a_sym / ss

    tau_levels = [20, 100, 400, 1500, 4000]
    beta = 0.9
    v_full = np.zeros(n)
    lr = 2.0 / np.sqrt(n)

    for tau in tau_levels:
        it_level = 0
        v_full = np.zeros(n)
        while time.monotonic() - start < time_budget:
            if it_level >= 3000:
                break
            it_level += 1

            y = a_sym + beta * v_full
            y = np.maximum(y, 0)
            y = (y + y[::-1]) / 2.0
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = a_sym.copy()
                v_full = np.zeros(n)
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g = (g + g[::-1]) / 2.0

            v_full = beta * v_full - lr * g
            a_new = y + v_full
            a_new = np.maximum(a_new, 0)
            a_new = (a_new + a_new[::-1]) / 2.0
            sn = float(np.sum(a_new))
            if sn < 1e-10:
                a_new = best_a.copy()
                v_full = np.zeros(n)
            else:
                a_new = a_new / sn

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()

            a_sym = a_new
            lr = max(1e-8, lr * 0.99999)

    return best_a

    if p2_budget > 30 and n_current <= 2500:
        for restart in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = max(25, p2_budget / max(1, 5 - restart))
            alloc = min(alloc, remaining * 0.65)

            if restart == 0:
                a_fw = best_seq.copy()
            elif restart == 1:
                pert = rng.normal(0, 0.01, n_current)
                a_fw = best_seq + pert
                a_fw = np.maximum(a_fw, 0)
                a_fw = a_fw / a_fw.sum()
            elif restart == 2:
                # Fourier perturbation
                freqs = rng.integers(1, 8, size=3)
                phases = rng.uniform(0, 2 * np.pi, size=3)
                x = np.arange(n_current)
                pert = np.zeros(n_current)
                for f, p in zip(freqs, phases):
                    pert += 0.03 * np.sin(2 * np.pi * f * x / n_current + p)
                a_fw = best_seq + pert
                a_fw = np.maximum(a_fw, 0)
                a_fw = a_fw / a_fw.sum()
            else:
                # Larger perturbation
                pert = rng.normal(0, 0.03, n_current)
                a_fw = best_seq + pert
                a_fw = np.maximum(a_fw, 0)
                a_fw = a_fw / a_fw.sum()

            a_result = lp_frank_wolfe_v2(a_fw, time_budget=alloc, verbose=True,
                                          max_iters=250, seed=100 + restart * 13,
                                          )
            obj_result = compute_obj(a_result)
            if obj_result < best_score:
                best_score = obj_result
                best_seq = a_result.copy()
                print(f"  LP FW r{restart}: obj={best_score:.10f} **")
            else:
                print(f"  LP FW r{restart}: obj={obj_result:.10f}")

    # === Phase 3: GD Polish ===
    print(f"\n=== Phase 3: GD Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 250.0)

    if p3_budget > 20:
        a3 = gd_nesterov(best_seq, time_budget=p3_budget, verbose=True,
                         tau_levels=[10, 50, 150, 400, 1000, 2500, 5000, 10000])
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  GD: obj={best_score:.10f} **")
        else:
            print(f"  GD: obj={obj3:.10f}")

    # === Phase 4: LP + Symmetric ===
    print(f"\n=== Phase 4: LP + Symmetric ===")
    remaining = deadline - time.monotonic()

    # Another LP round
    if remaining > 40 and n_current <= 2500:
        alloc = min(remaining * 0.2, 200.0)
        a4 = lp_frank_wolfe_v2(best_seq, time_budget=alloc, verbose=True,
                                max_iters=150, seed=777)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  LP r2: obj={best_score:.10f} **")
        else:
            print(f"  LP r2: obj={obj4:.10f}")

    # Symmetric optimization
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc_sym = min(remaining * 0.15, 80.0)
        a_sym = symmetric_optimize(best_seq, time_budget=alloc_sym)
        obj_sym = compute_obj(a_sym)
        if obj_sym < best_score:
            best_score = obj_sym
            best_seq = a_sym.copy()
            print(f"  Sym: obj={best_score:.10f} **")
        else:
            print(f"  Sym: obj={obj_sym:.10f}")

    # === Phase 5: Support Pruning ===
    print(f"\n=== Phase 5: Support Pruning ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc_prune = min(remaining * 0.12, 80.0)
        a_pruned = support_prune(best_seq, time_budget=alloc_prune, verbose=True)
        obj_pruned = compute_obj(a_pruned)
        if obj_pruned < best_score:
            best_score = obj_pruned
            best_seq = a_pruned.copy()
            print(f"  Pruned: obj={best_score:.10f} **")
        else:
            print(f"  Pruned: obj={obj_pruned:.10f}")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 200),
        max(400, n_current - 100),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        600, 800, 1000, 1200, 1500, 2000, 2500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 3000 and n != len(best_seq)]
    n_options.sort(key=lambda x: abs(x - n_current))
    n_options = n_options[:6]

    print(f"  n candidates: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 70:
            break

        a_try = resample_a(best_seq, n_try)

        # GD
        alloc1 = min(remaining * 0.12, 35.0)
        if alloc1 > 10:
            a_try = gd_nesterov(a_try, time_budget=alloc1,
                                tau_levels=[20, 100, 400, 1500])

        # LP if small enough
        remaining = deadline - time.monotonic()
        if n_try <= 2000 and remaining > 40:
            alloc2 = min(remaining * 0.1, 35.0)
            if alloc2 > 15:
                a_try = lp_frank_wolfe_v2(a_try, time_budget=alloc2, verbose=False,
                                           max_iters=80, seed=n_try % 500 + 1)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    # LP
    if remaining > 30 and n_final <= 2500:
        alloc = min(remaining * 0.25, 100.0)
        a_f1 = lp_frank_wolfe_v2(best_seq, time_budget=alloc, verbose=True,
                                  max_iters=100, seed=31415)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    # High-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 80.0)
        a_f2 = gd_nesterov(best_seq, time_budget=alloc, verbose=False,
                           tau_levels=[200, 600, 1500, 4000, 10000, 20000])
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final GD: obj={best_score:.10f} **")

    # Adam GD
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 60.0)
        a_f3 = gd_adam(best_seq, time_budget=alloc, verbose=False,
                      tau_levels=[100, 400, 1000, 3000], lr=0.003)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Adam GD: obj={best_score:.10f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 20 and n_final <= 2500:
        alloc = min(remaining * 0.3, 50.0)
        a_f4 = lp_frank_wolfe_v2(best_seq, time_budget=alloc, verbose=False,
                                  max_iters=50, seed=27182)
        obj_f4 = compute_obj(a_f4)
        if obj_f4 < best_score:
            best_score = obj_f4
            best_seq = a_f4.copy()
            print(f"  Last LP: obj={best_score:.10f} **")

    # One more GD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 40.0)
        a_f5 = gd_nesterov(best_seq, time_budget=alloc, verbose=False,
                           tau_levels=[500, 1500, 5000, 10000, 20000])
        obj_f5 = compute_obj(a_f5)
        if obj_f5 < best_score:
            best_score = obj_f5
            best_seq = a_f5.copy()
            print(f"  Last GD: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
