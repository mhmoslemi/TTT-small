"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent (exponentiated gradient) with adaptive tau smoothing,
combined with LP-based refinement and Polyak averaging.
Key innovations: FFT-based subgradient computation, multiplicative updates
that naturally preserve non-negativity, and dual-guided LP polishing.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_smoothed_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using softmax weights.
    Returns (gradient, conv, p_weights)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    # Softmax weights
    shifted = tau * (conv - max_conv)
    # Numerical stability
    shifted -= np.max(shifted)
    p = np.exp(shifted)
    p_sum = float(np.sum(p))
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = p / p_sum
    
    # Cross-correlation: g[j] = sum_k p[k] * a[k-j]
    N = 1
    while N < len(p) + n:
        N <<= 1
    
    P = rfft(p, N)
    A = rfft(a, N)
    corr = irfft(P * np.conj(A), N)
    g = 2.0 * corr[:n]
    
    return g, conv, p


def mirror_descent(a, time_budget=100.0, tau_init=50.0, tau_end=500.0, verbose=False):

    best_a = a.copy()
    best_obj = compute_obj(a)

    running_sum = a.copy()
    count = 0

    momentum = 0.0
    v = np.zeros(n)
    
    it = 0

        v = np.maximum(v, 0)  # Ensure non-negative "gradient" for multiplicative update

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    best_g = None
    best_M = np.inf
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    if M_val < best_M:
                        best_g = g_val
                        best_M = M_val
                        break
        except Exception:
            continue
    
    if best_g is None:
        return None, None
    return best_g, best_M


def lp_iteration(a, time_budget=100.0, verbose=False):
    """LP-based iteration with adaptive step size and Polyak averaging."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Polyak average
    running_sum = a.copy()
    count = 0
    
    it = 0
    stale = 0
    prev_g = None
    momentum = 0.5
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        
        it += 1
        
        # Compute step size using line search
        # Try several t values
        t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        
        # If we have a previous LP solution, try extrapolation
        if prev_g is not None:
            direction = g_lp - prev_g
            norm_dir = np.linalg.norm(direction)
            if norm_dir > 1e-10:
                # Try extrapolated step
                t_ext = min(1.5, 0.5 / (momentum + 0.1))
                a_ext = a + t_ext * (g_lp - a)
                a_ext = np.maximum(a_ext, 0)
                s_ext = float(np.sum(a_ext))
                if s_ext > 1e-10:
                    a_ext = a_ext / s_ext
                    obj_ext = compute_obj(a_ext)
                    if obj_ext < best_obj:
                        t_candidates.insert(0, t_ext)
        
        best_t_obj = best_obj
        best_t_val = 0.0
        best_a_local = best_a.copy()
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stale = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, M={M_lp:.6f}")
        else:
            stale += 1
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: no improvement (stale={stale})")
        
        # Update Polyak average
        running_sum += a
        count += 1
        
        prev_g = g_lp.copy()
        
        # Reset if stale
        if stale > 10:
            # Perturb and restart
            rng = np.random.default_rng(it * 12345)
            strength = min(0.1, 0.02 * stale)
            pert = 1.0 + rng.normal(0, strength, n)
            a = best_a * np.maximum(pert, 0)
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
            prev_g = None
            stale = 0
    
    # Check Polyak average
    polyak_a = running_sum / count
    polyak_a = np.maximum(polyak_a, 0)
    s_p = float(np.sum(polyak_a))
    if s_p > 1e-10:
        polyak_a = polyak_a / s_p
        obj_polyak = compute_obj(polyak_a)
        if obj_polyak < best_obj:
            best_a = polyak_a.copy()
            best_obj = obj_polyak
    
    return best_a


def structured_init(n, seed=42):
    """Create a structured initial sequence based on difference set ideas."""
    rng = np.random.default_rng(seed)
    # Start with a cosine-shaped sequence with some randomness
    x = np.arange(n)
    a = 0.5 + 0.3 * np.cos(2 * np.pi * x / n) + 0.2 * np.cos(4 * np.pi * x / (n * 1.7))
    a = a + rng.uniform(0, 0.1, n)
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def fourier_refine(a, time_budget=50.0):
    """Refine in Fourier domain: project to reduce peak of |FFT(a)^2|."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Compute FFT
        N = 1
        while N < 2 * n - 1:
            N <<= 1
        A = rfft(a, N)
        
        # We want |A|^2 to be as flat as possible (in the time domain, this means conv is flat)
        # Actually, flatness in time domain (conv) means we want A to have specific properties
        # Instead, let's do a multiplicative update in Fourier domain
        # Reduce the magnitude of A at frequencies that contribute to the peak
        
        conv = compute_conv(a)
        peak_idx = np.argmax(conv)
        max_conv = conv[peak_idx]
        
        # The contribution of frequency k to the peak is related to |A[k]|^2
        # Reduce |A| uniformly (which corresponds to making a smoother)
        # Or: reduce |A| more at frequencies that correlate with the peak pattern
        
        # Simple approach: apply a low-pass filter (reduce high frequencies)
        # This makes a smoother, which tends to reduce conv peaks
        freqs = np.arange(N // 2 + 1)
        # Gaussian low-pass in frequency
        sigma = max(5, n // 20)
        mask = np.exp(-freqs**2 / (2 * sigma**2))
        
        A_filtered = A * mask
        a_new = irfft(A_filtered, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        # Decrease sigma over time (less aggressive filtering)
        sigma *= 0.99
        
        it += 1
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
