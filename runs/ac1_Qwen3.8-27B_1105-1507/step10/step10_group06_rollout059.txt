"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:
- LP-based best response (Frank-Wolfe) with adaptive step sizes
- Nesterov-accelerated subgradient descent with log-sum-exp smoothing
- Coordinate descent with exact 1D minimization
- Multiple diverse restarts for escaping local minima

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient(a, tau):
    """Gradient of log-sum-exp smoothed max of conv."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, p


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0)
    s = float(np.sum(v))
    if s < 1e-10:
        return np.ones(len(v)) / len(v)
    return v / s


def nesterov_smoothed(a, time_budget, tau_init=50.0, tau_max=5000.0, lr_init=0.5):
    """Nesterov accelerated gradient descent on smoothed objective with simplex constraint."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = lr_init / np.sqrt(n)
    v = np.zeros(n)
    tau = tau_init
    it = 0
    
    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = a + beta * v
        y = project_simplex(y)
        
        grad, p = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(grad))
        
        if g_norm > 1e-15:
            adaptive_lr = 0.5 / (g_norm + 1e-10)
            lr = 0.7 * lr + 0.3 * min(adaptive_lr, 3.0 * lr_init / np.sqrt(n))
        
        v = beta * v - lr * grad
        a_new = project_simplex(y + v)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        lr *= 0.99995
        tau = min(tau * 1.03, tau_max)
        it += 1
        
        if it % 2000 == 0 and compute_obj(a) > best_obj + 1e-4:
            a = best_a.copy()
            v = np.zeros(n)
    
    return best_a

def coordinate_descent(a, time_budget, verbose=True):
    """
    Coordinate descent on the simplex.
    For each coordinate i, we find the optimal value by considering
    the trade-off between increasing a_i and decreasing others proportionally.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    # Precompute convolution
    conv = compute_conv(a)
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Random coordinate order
        order = rng.permutation(n)
        
        for i in order:
            # Try several values for a_i
            # When we change a_i from a[i] to a[i]+delta, we reduce all other a_j proportionally
            # to maintain sum = 1
            
            old_val = a[i]
            if old_val < 1e-10:
                continue
            
            # Try: keep a_i, reduce others. Or: increase a_i, reduce others.
            # The "others" sum is 1 - a[i]
            # If we set a[i] = x, others sum to 1-x, each scaled by (1-x)/(1-old_val)
            
            # This is complex. Instead, let's try a simpler approach:
            # Swap: move some mass from coordinate j to coordinate i
            # Try a few random j's
            
            for _ in range(3):
                j = rng.integers(0, n)
                if j == i or a[j] < 1e-10:
                    continue
                
                # Move fraction f of a[j] to a[i]
                for f in [0.1, 0.25, 0.5, 0.75]:
                    delta = f * a[j]
                    a_new = a.copy()
                    a_new[i] += delta
                    a_new[j] -= delta
                    if a_new[j] < 0:
                        continue
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new.copy()
                        conv = compute_conv(a)
        
        it += 1
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD pass {it}: obj={best_obj:.10f}, time={elapsed:.1f}s")
    
    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_to_length(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

Let me also think about whether there's a structural insight I'm missing.

The objective is 2n * max(a*a) / (sum a)^2. For a normalized sequence (sum=1), we want to minimize 2n * max(a*a).

The Fourier transform connection: a*a (autocorrelation) in the time domain corresponds to |F(a)|^2 in the frequency domain. So max(a*a) is related to the maximum of the inverse Fourier transform of |F(a)|^2.

By Parseval: sum_k (a*a)[k] = sum_i (|F(a)|_i)^2 = (sum a)^2 = 1 (for normalized a).

So the average value of (a*a)[k] over all 2n-1 lags is 1/(2n-1). We want to make the distribution as flat as possible.

This connects to the problem of designing a "flat-top" spectrum. If |F(a)|^2 were constant, then a*a would be constant (= 1/(2n-1)), and the objective would be 2n/(2n-1) ≈ 1.

But a is non-negative, which constrains F(a) significantly. A non-negative sequence has a spectrum that can't be arbitrary.

One approach: try to make the spectrum as flat as possible subject to the non-negativity constraint. This is like a spectral design problem.

The spectral approach:
- Minimize max_k (IFFT(|F(a)|^2))[k] subject to IFFT(F(a)) >= 0, sum(IFFT(F(a))) = 1
- This is still non-convex, but we can alternate between:
  a) Fix a, compute |F(a)|^2, find the "gaps" in the spectrum
  b) Adjust the spectrum to fill gaps, then inverse FFT and project onto non-negative

This is essentially what the LP approach does, but in the time domain.

I think the most practical improvement is to:
1. Use the LP more efficiently (better solver settings, warm starts)
2. Use a better step size schedule
3. Add a spectral domain polish step
4. Try multiple diverse initializations

Let me also think about a "dual" LP approach. The dual of the best-response LP gives us weights w_k >= 0 on the convolution lags, with sum(w) = 1 (since sum(g) = 1 is in the primal). These weights tell us which lags are "important" (active at the optimum).

Using the dual information, we can construct a better subgradient:

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based Frank-Wolfe with adaptive step sizes, Nesterov-accelerated 
smoothed gradient descent, spectral equalization, and multiple restarts.
Key improvements: FW-gap-based step sizing, top-lag LP reduction, 
and aggressive multi-phase polish with increasing smoothing.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def build_conv_matrix_reduced(a, top_k_indices):
    """Build reduced convolution matrix using only top-k lag indices."""
    n = len(a)
    A = A_full = np.zeros((len(top_k_indices), n))
    for idx, k in enumerate(top_k_indices):
        A[idx, :] = np.convolve(a, np.eye(n)[k % n if k < n else 0])  # placeholder
    # Actually, let's just build it properly
    A = np.zeros((len(top_k_indices), n))
    for idx, k in enumerate(top_k_indices):
        for j in range(n):
            if 0 <= k - j < n:
                A[idx, j] = a[k - j]
            # Also the symmetric contribution
            if 0 <= k - j < n:
                pass  # already handled since conv[a,g][k] = sum_j a[k-j]*g[j]
    # Wait, conv(a,g)[k] = sum_j a[j]*g[k-j] = sum_j a[k-j]*g[j] (by substitution)
    # So A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((len(top_k_indices), n))
    for idx, k in enumerate(top_k_indices):
        for j in range(n):
            jj = k - j
            if 0 <= jj < n:
                A[idx, j] = a[jj]
    return A


def solve_conv_lp(a, top_k=None):
    """Solve LP: min M s.t. (a*g)[k] <= M for selected k, sum(g)=1, g>=0.
    If top_k is None, use all lags."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    if top_k is not None:
        A = build_conv_matrix_reduced(a_norm, top_k)
    else:
        A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient(a, tau):
    """Gradient of (1/tau)*log(sum exp(tau*conv_k)) w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, p


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0)
    s = float(np.sum(v))
    if s < 1e-10:
        return np.ones(len(v)) / len(v)
    return v / s


def nesterov_smoothed(a, time_budget, tau_init=50.0, tau_max=10000.0, lr_scale=1.0):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = lr_scale / np.sqrt(n)
    v = np.zeros(n)
    tau = tau_init
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = project_simplex(y)
        
        grad, p = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(grad))
        
        if g_norm > 1e-15:
            adaptive_lr = 0.3 / (g_norm + 1e-10)
            lr = 0.8 * lr + 0.2 * min(adaptive_lr, 2.0 * lr_scale / np.sqrt(n))
        
        v = beta * v - lr * grad
        a_new = project_simplex(y + v)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        lr *= 0.99995
        tau = min(tau * 1.02, tau_max)
        it += 1
        
        # Reset if stuck
        if it - best_it > 3000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = lr_scale / np.sqrt(n)
            beta = 0.5
            it = best_it + 1800  # allow some more before resetting again
    
    return best_a


def lp_alternate_v2(a, time_budget, seed=42, verbose=True, use_top_k=200):
    """LP-based alternating minimization with adaptive step size and top-k reduction."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    iter_count = 0
    stall_count = 0
    t_blend = 0.3
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        # Determine which lags to use in LP
        conv = compute_conv(a / np.sum(a))
        if use_top_k and use_top_k < 2 * n - 1:
            # Use top k lags plus some random ones for diversity
            top_indices = np.argsort(conv)[::-1][:use_top_k]
            if iter_count % 10 == 0:
                # Occasionally add random lags
                rand_indices = rng.choice(2 * n - 1, size=20, replace=False)
                top_indices = np.unique(np.concatenate([top_indices, rand_indices]))
        else:
            top_indices = None
        
        g_lp, M_lp = solve_conv_lp(a, top_k=top_indices)
        if g_lp is None:
            # Fallback: use full LP
            g_lp, M_lp = solve_conv_lp(a, top_k=None)
            if g_lp is None:
                break
        
        iter_count += 1
        
        # Adaptive step size candidates
        # FW theoretical: 2/(t+2)
        fw_t = 2.0 / (iter_count + 2)
        candidates = [fw_t, t_blend, t_blend * 0.5, t_blend * 2.0]
        candidates = [min(max(t, 0.005), 1.0) for t in candidates]
        if iter_count > 100:
            candidates.extend([0.02, 0.05, 0.1])
        if iter_count > 300:
            candidates.extend([0.01, 0.03])
        
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend
        
        for t in candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t
        
        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            t_blend = max(0.01, best_local_t * 0.95)
            stall_count = 0
        else:
            stall_count += 1
            if stall_count >= 5:
                # Perturb to escape
                strength = 0.005 + 0.003 * min(stall_count, 5)
                pert = np.zeros(n)
                for k in range(1, 6):
                    freq = k * 2 * np.pi / n
                    amp = strength * rng.uniform(0.5, 1.5) / np.sqrt(k)
                    phase = rng.uniform(0, 2 * np.pi)
                    pert += amp * np.cos(freq * np.arange(n) + phase)
                a = project_simplex(best_a * np.exp(pert))
                t_blend = max(0.1, min(t_blend * 1.3, 0.6))
                stall_count = 0
            elif stall_count >= 2:
                a = project_simplex(best_a + 0.008 * rng.standard_normal(n) * np.maximum(best_a, 0.005))
        
        if verbose and iter_count % 30 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP iter {iter_count}: obj={best_obj:.10f}, t={t_blend:.4f}, stall={stall_count}, time={elapsed:.1f}s")
    
    return best_a


def spectral_equalize(a, n_iters=50, time_budget=30):
    """Try to equalize the power spectrum to flatten autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n:
        N <<= 1
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute current spectrum
        A = fft(a, N)
        power = np.abs(A) ** 2
        power = np.concatenate([power, power[N - n:N - 1][::-1]])  # real spectrum
        
        # Target: flat spectrum
        target_power = np.full(N, np.sum(power) / N)
        
        # Blend toward flat
        alpha = 0.1
        new_power = (1 - alpha) * power + alpha * target_power
        
        # Inverse FFT (keeping the magnitude, adjusting phase)
        # This is tricky - we want to change magnitudes while keeping a >= 0
        # Simple approach: scale a to reduce peak power
        a_new = a.copy()
        peak_power = np.max(power)
        avg_power = np.mean(power)
        if peak_power > 0:
            # Reduce elements that contribute to peak
            scale = 1.0 - 0.05 * (peak_power / avg_power - 1.0)
            scale = max(0.9, min(1.0, scale))
            a_new = a * scale
            a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        else:
            # Try the opposite: boost low-power regions
            a_new2 = a * (1.0 + 0.02 * (avg_power / (power[:n] + 1e-10)))
            a_new2 = project_simplex(a_new2)
            obj_new2 = compute_obj(a_new2)
            if obj_new2 < best_obj:
                best_obj = obj_new2
                best_a = a_new2.copy()
    
    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.01):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
