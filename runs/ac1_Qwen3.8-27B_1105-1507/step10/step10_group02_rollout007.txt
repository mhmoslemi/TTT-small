from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def smoothed_gradient(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*conv(a,a))) w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Numerical stability: subtract max
    max_shifted = float(np.max(shifted))
    exp_vals = np.exp(shifted - max_shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    
    # Gradient: dF/d(a[i]) = 2 * sum_j p[i+j] * a[j]
    # = 2 * conv(a_rev, p)[i + n - 1] for i = 0,...,n-1
    a_rev = a[::-1]
    size = 1
    while size < n + len(p) - 1:
        size <<= 1
    fa_rev = rfft(a_rev, size)
    fp = rfft(p, size)
    conv_full = irfft(fa_rev * fp, size)
    
    gradient = 2.0 * conv_full[n-1 : 2*n-1]
    return gradient, conv, p

def project_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Hybrid optimizer: FISTA smoothed gradient + exact coordinate descent + LP polishing.
Uses FFT-based gradient computation for fast coarse optimization, then switches
to exact coordinate descent for fine-tuning. LP polishing with adaptive line search
provides additional improvement. Multi-n exploration and diverse restarts help
escape local optima.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a / s)))

def smoothed_max_gradient(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*conv(a,a)))."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    # For numerical stability
    max_shifted = float(np.max(shifted))
    exp_vals = np.exp(shifted - max_shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp  # probability distribution on lag
    
    # grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    # This is 2 * (a_rev * p)[n-1 : 2n-1] where * denotes convolution
    a_rev = a[::-1].copy()
    size = 1
    while size < n + (2*n-1) - 1:
        size <<= 1
    fa = rfft(a_rev, size)
    fp = rfft(p, size)
    conv_result = irfft(fa * fp, size)
    grad = 2.0 * conv_result[n-1 : 2*n-1]
    return grad

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fista_optimize(a, time_budget=200.0, tau_schedule=None, verbose=True):
    """FISTA on the smoothed objective, projected onto simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    if tau_schedule is None:
        tau_schedule = [5, 10, 20, 40, 80, 160, 320, 640]
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    
    for tau in tau_schedule:
        if time.monotonic() - start >= time_budget:
            break
        
        x = a.copy()
        y = a.copy()
        t_k = 1.0
        lr = 1.0 / (tau * 2.0)  # Conservative initial LR
        
        for it in range(500):
            if time.monotonic() - start >= time_budget:
                break
            
            grad = smoothed_max_gradient(y, tau)
            
            # FISTA update
            x_new = np.maximum(y - lr * grad, 0.0)
            x_new = project_simplex(x_new)
            
            # Check objective
            obj = compute_obj_normalized(x_new)
            if obj < best_obj:
                best_obj = obj
                best_a = x_new.copy()
            
            # Momentum update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            momentum = (t_k - 1.0) / t_new
            y = x_new + momentum * (x_new - x)
            y = np.maximum(y, 0.0)
            y = project_simplex(y)
            x = x_new
            t_k = t_new
            
            # Adaptive LR: if no improvement for a while, reduce
            # Simple heuristic
            if it > 0 and it % 50 == 0:
                # Check if we're making progress
                pass
        
        a = best_a
        if verbose:
            elapsed = time.monotonic() - start
            print(f"    FISTA tau={tau}: obj={best_obj:.10f}, t={elapsed:.1f}s")
    
    return best_a
