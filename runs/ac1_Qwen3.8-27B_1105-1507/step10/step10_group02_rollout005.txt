def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_bound=None, verbose=False):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g>=0.
    If M_bound is None, use M = max(conv(a,a))."""
    n = len(a)
    a_sum = float(np.sum(a))
    if a_sum < 1e-10:
        return None, None
    a_norm = a / a_sum
    
    if M_bound is None:
        conv_aa = compute_conv(a_norm)
        M_bound = float(np.max(conv_aa))
    
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M_bound, cp.sum(g) >= 1]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:

    return None, None

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Half-on pattern
        a = np.zeros(n)
        a[:n//2] = 1.0
        a = a / a.sum()
    elif mode == 3:
        # Triangular
        a = np.sin(np.pi * np.arange(n) / n)
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 4:
        # Two bumps
        centers = [n // 3, 2 * n // 3]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Anti-symmetric-ish
        a = np.ones(n)
        a[:n//2] = 2.0
        a = a / a.sum()
    elif mode == 6:
        # Cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
