        if total_run_time + alloc > time_budget_total:
            continue
            
        start_opt = time.monotonic()
        polished_seq, obj_opt = optimize_step_function(n_try)
        time_spent_opt = time.monotonic() - start_opt
        total_run_time += time_spent_opt
        if obj_opt < best_objective:
            best_sequence = polished_seq
            best_objective = obj_opt
        elif obj_opt - best_objective > 1e-5:
            # Save multiple candidates to allow switching
            pass
            
        # Check for over optimization or time constraints
        if total_run_time > time_budget_total:
            break
        print(f"Trying length {n_try}, candidate objective: {obj_opt}")
    
    # Final validation and return the best candidate
    a_return = a_initial if best_sequence is None else best_sequence
    # Ensure nonnegative and normalized (but the previous system maintained normalization)
    a_return = normalize(a_return)
    # Clip values for optimization if necessary? But the evaluate_sequence function handles that.
    final_obj = compute_objective(a_return)
    if final_obj < best_objective:
        best_sequence, best_objective = a_return, final_obj
        
    # We must return the sequence in the format as required by the evaluator?
    return a_return.tolist()
