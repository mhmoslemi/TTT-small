"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based minimax iteration with dual-guided smoothed gradient descent,
multi-length exploration, and randomized restarts. Key ideas: fast FFT-based
convolution, warm-started LP solves, adaptive tau softmax gradient, and
systematic n-value sweeping with coarse-to-fine resampling.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, max_time=10.0):
    """Solve LP: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > max_time:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.asarray(g.value, dtype=np.float64).flatten()
                g_val = np.maximum(g_val, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:
            continue
    return None, None


def lp_phase(a, time_budget, verbose=True):
    """Iterative LP polishing with adaptive line search and stall detection."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    last_t = 1.0
    stall = 0
    iter_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        g, M = solve_lp(a, max_time=min(5.0, remaining * 0.3))
        if g is None:
            break

        # Line search over mixing parameter t
        t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0,
                        max(0.02, last_t * 0.5), max(0.02, last_t * 1.5),
                        min(1.0, last_t)]
        t_candidates = sorted(set(np.clip(t_candidates, 0.01, 1.0)))

        best_t_obj = compute_obj(a)
        best_t = 0.0
        best_a_local = a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try

                best_t = t

            last_t = best_t
            stall = 0
            obj = compute_obj(a)
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                iter_count += 1
                if verbose and iter_count % 5 == 0:
                    print(f"    LP iter {iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 3:
                break

    return best_a


def smoothed_grad(a, tau):
    """Compute smoothed gradient and probability vector.
    Returns (grad, p) where grad has same length as a."""
    n = len(a)
    a = a / float(np.sum(a))
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    # Gradient: grad_j = 2 * sum_k p_k * a[k-j] (with a out-of-range = 0)
    # This is a cross-correlation. Using FFT:
    size = 1
    while size < 3 * n:
        size <<= 1
    P = rfft(p, size)
    A = rfft(a, size)
    corr = irfft(P * np.conj(A), size)
    grad = 2.0 * corr[:n]
    return grad, p


def gd_phase(a, time_budget, tau_start=50, tau_end=8000, verbose=True):
    """Nesterov-accelerated smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    beta = 0.85
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.6

        a = a / float(np.sum(a))
        grad, p = smoothed_grad(a, tau)

        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-16:
            break

        # Adaptive learning rate
        lr = min(1.0 / (g_norm + 1e-12), 4.0 / np.sqrt(n))
        lr = max(lr, 1e-12)

        v = beta * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            a = a_new
        else:
            a_new = a_new / s_new
            a = a_new

        obj_new = compute_obj(a)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a.copy()
            best_it = it

        it += 1
        beta = min(0.98, 0.85 + 0.0002 * it)

        # Periodic check for divergence
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it
            if it > 50000:
                break

        if verbose and it % 3000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, elapsed={time.monotonic()-start:.0f}s")

    return best_a


def dual_guided_update(a, time_budget, verbose=True):
    """Use LP dual information to guide a more informed update of a."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Get LP solution
        g, M = solve_lp(a, max_time=min(5.0, remaining * 0.25))
        if g is None:
            break

        # Compute where the convolution of a*a is largest
        conv_a = compute_conv(a)
        max_conv_a = float(np.max(conv_a))

        # Find "hot spots" in the convolution of a
        threshold = max_conv_a * 0.8
        hot = np.where(conv_a >= threshold)[0]

        # Compute convolution of a and g at hot spots
        ag_conv = np.zeros(2 * n - 1)
        for k in hot:
            # (a*g)[k] = sum_i a[i] * g[k-i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                ag_conv[k] = float(np.dot(a[lo:hi+1], g[k-hi:k-lo+1][::-1]))

        # Blend: push a towards g more where ag_conv is high
        weight = np.maximum(0, (ag_conv[:n] - M * 0.9) / max(M, 1e-10))
        weight = np.clip(weight, 0, 1)
        weight = 0.5 + 0.5 * weight  # at least 50% towards g

        alpha = 0.7
        a_new = (1 - alpha) * a + alpha * weight * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                if verbose:
                    print(f"    Dual: obj={best_obj:.8f}, elapsed={time.monotonic()-start:.0f}s")
            else:
                # Fall back to simple mixing
                a_new = (1 - 0.3) * a + 0.3 * g
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                else:
                    break

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initial sequences."""
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x)
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * (x - 0.5) ** 2)
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.5))
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = x ** 0.5 * (1 - x) ** 0.3
    elif mode == 6:
        a = np.cos(np.pi * np.arange(n) / n) ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

def randomized_restart(a, best_score, time_budget, rng, verbose=True):
    """Try random perturbations and re-optimize."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_s = best_score
    n_restarts = max(1, int(time_budget / 40))

    for r in range(n_restarts):
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 30:
            break
        alloc = min(35, time_budget - elapsed - 10)
        if alloc < 15:
            break

        # Create perturbed version
        mode = r % 4
        if mode == 0:
            # Add small noise
            noise = rng.normal(0, 0.05, n)
            a_r = a + noise
        elif mode == 1:
            # Blend with random
            a_rand = rng.uniform(0, 1, n)
            a_rand = a_rand / a_rand.sum()
            a_r = 0.6 * a + 0.4 * a_rand
        elif mode == 2:
            # Flip part of the sequence
            k = n // 2
            a_r = np.concatenate([a[:k], a[k:][::-1]])
        else:
            # Circular shift
            shift = rng.integers(1, n // 2)
            a_r = np.roll(a, shift)

        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r < 1e-10:
            continue
        a_r = a_r / s_r

        # Quick optimization
        half = alloc * 0.5
        a_r = lp_phase(a_r, time_budget=half, verbose=False)
        remaining = alloc - (time.monotonic() - start - elapsed)
        if remaining > 8:
            a_r = gd_phase(a_r, time_budget=min(remaining * 0.7, 15),

        obj_r = compute_obj(a_r)

            best_a = a_f.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Round 3: One more LP
    remaining = DEADLINE - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 60.0)
        a_f = lp_phase(best_a, time_budget=alloc, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score - 1e-13:
            best_score = obj_f
            best_a = a_f.copy()
            print(f"  Ultra LP: obj={best_score:.8f} **")

    # Round 4: Final GD
    remaining = DEADLINE - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.5, 40.0)
        a_f = gd_phase(best_a, time_budget=alloc,
                       tau_start=1000, tau_end=20000, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score - 1e-13:
            best_score = obj_f
            best_a = a_f.copy()
            print(f"  Ultra GD: obj={best_score:.8f} **")

    # ===== Final Validation =====
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        final_score = val
    else:
        # Fallback: return a valid sequence
        result = [1.0] * 100
        final_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={final_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    return result
