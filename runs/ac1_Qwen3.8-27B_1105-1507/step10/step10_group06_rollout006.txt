```python
from scipy.optimize import linprog

def solve_lp(a):
    """Solve: min M s.t. A@g <= M, sum(g)=1, g>=0"""
    n = len(a)
    a = a / a.sum()
    
    # A is (2n-1) x n, A[k,j] = a[k-j] if 0<=k-j<n else 0
    # We need: A@g - M <= 0 for all k
    #         sum(g) = 1
    #         g >= 0, M free
    
    # Variables: [g_0, ..., g_{n-1}, M]
    # Objective: minimize M => c = [0, ..., 0, 1]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # Inequality constraints: A@g - M <= 0 => A@g + (-1)*M <= 0
    # A_aug is (2n-1) x (n+1)
    A_aug = np.zeros((2*n-1, n+1))
    # Fill in the banded structure of A
    for j in range(n):
        A_aug[j:j+n, j] = a
    A_aug[:, -1] = -1.0  # coefficient of M
    
    b_ub = np.zeros(2*n-1)
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    # Bounds: g_j >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_aug, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        g = res.x[:n]
        M = res.x[-1]
        g = np.maximum(g, 0)
        g = g / g.sum()
        return g, M
    return None, None
```

                        a = a_try
                        conv = compute_conv(a)
                        peak = float(np.max(conv))
                        peak_pos = int(np.argmax(conv))
                        improved = True
                        break
        
        if not improved:
            # Try random coordinate swaps
            j1 = np.random.randint(n)
            j2 = np.random.randint(n)
            if j1 != j2:
                a_try = a.copy()
                a_try[j1], a_try[j2] = a_try[j2], a_try[j1]
                a_try = a_try / a_try.sum()
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
        
        if time.monotonic() - start > time_budget:
            break
    
    if verbose:
        print(f"    CD done: obj={best_obj:.10f}")
    
    return best_a
