def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""

    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

"""
Optimizes step function coefficients using Alternating Linear Minimax (ALM)
combined with block coordinate descent and multi-n exploration.
Key innovation: bidirectional LP-based optimization that alternates between
fixing one sequence and solving for the other, with block CD for local escape.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    fa = rfft(a, fft_size)
    result = irfft(fa * fa, fft_size)
    return result[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2n*max(conv)/sum^2 assuming a is pre-normalized (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            A[k, i] = a[k - i]
    return A


def solve_lp_minmax(a, solver_list=None):
    """Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None)."""
    n = len(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, max(float(M.value), 0.0)
        except Exception:
            continue
    return None, None


def coordinate_descent(a, time_budget=60.0, verbose=False, seed=42):
    """Coordinate descent with golden section for 1D subproblems."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            
            v = a[i]
            S = float(np.sum(a))
            S_other = S - v
            if S_other < 1e-12 and v < 1e-12:
                continue
            
            # 1D subproblem: minimize f(x) where x = a[i]
            # conv changes:
            #   For k in [i, i+n-1], k != 2i: conv[k] += 2*(x-v)*a[k-i]
            #   For k = 2i: conv[2i] += x^2 - v^2
            
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            k_quad = 2 * i
            
            # Precompute the linear part
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()
            
            mask = (ks != k_quad)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]
            
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0
            
            def f_1d(x):
                if x < 0:
                    return 1e30
                S_new = S
                if len(ks_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = S_new * S_new
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom
            
            # Golden section search
            scale = max(v, 1.0 / n, 1e-6)
            x_lo = max(0.0, v - 3.0 * scale)
            x_hi = v + 3.0 * scale
            
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)
            
            for _ in range(40):
                if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)
            
            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            
            if abs(x_opt - v) > 1e-12:
                delta = x_opt - v
                a[i] = x_opt
                if 0 <= k_quad <= 2 * n - 2:
                    conv[k_quad] += x_opt * x_opt - v * v
                if len(ks_lin) > 0:
                    conv[ks_lin] += 2.0 * delta * a_ki_lin
                
                # Check improvement
                obj_now = 2.0 * n * float(np.max(conv)) / (float(np.sum(a)) ** 2)
                if obj_now < best_obj:
                    best_obj = obj_now
                    best_a = a.copy()
    
    return best_a


def alm_optimize(a, time_budget=100.0, verbose=True, seed=42):
    """
    Alternating Linear Minimax: alternately solve LPs
    min_g max_k (a*g)_k and min_a max_k (g*a)_k.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    
    it = 0
    stall = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        # Step 1: Fix a, solve for g
        g_sol, M1 = solve_lp_minmax(a)
        if g_sol is None:
            break
        
        # Step 2: Fix g, solve for a_new
        a_sol, M2 = solve_lp_minmax(g_sol)
        if a_sol is None:
            break
        
        # Try blending
        improved = False
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * a_sol
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved:
            # Try g_sol directly
            obj_g = 2.0 * n * float(np.max(compute_conv(g_sol)))
            if obj_g < best_obj - 1e-12:
                best_obj = obj_g
                best_a = g_sol.copy()
                a = g_sol
                improved = True
        
        if not improved:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0
        
        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    ALM it{it}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a


def block_cd_2d(a, time_budget=60.0, verbose=False, seed=42):
    """
    Block coordinate descent optimizing pairs of coordinates at once.
    Uses scipy.optimize.minimize for the 2D subproblem.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    def obj_2d(params, i, j):
        x, y = params
        if x < 0 or y < 0:
            return 1e30
        vi, vj = a[i], a[j]
        # Update conv incrementally
        conv_new = conv.copy()
        S = float(np.sum(a))
        
        # Update for coordinate i
        k_min_i = max(0, i)
        k_max_i = min(2*n-2, i+n-1)
        for k in range(k_min_i, k_max_i+1):
            if k == 2*i:
                conv_new[k] += x*x - vi*vi
            else:
                conv_new[k] += 2.0*(x-vi)*a[k-i]
        
        # Update for coordinate j (using original a for cross terms)
        k_min_j = max(0, j)
        k_max_j = min(2*n-2, j+n-1)
        for k in range(k_min_j, k_max_j+1):
            if k == 2*j:
                conv_new[k] += y*y - vj*vj
            else:
                conv_new[k] += 2.0*(y-vj)*a[k-j]
        
        # Cross term: conv[i+j] gets extra 2*(x-vi)*(y-vj) if i != j
        if i != j:
            k_cross = i + j
            if 0 <= k_cross <= 2*n-2:
                conv_new[k_cross] += 2.0*(x-vi)*(y-vj)
        
        S_new = S - vi - vj + x + y
        if S_new < 1e-10:
            return 1e30
        
        max_conv = float(np.max(conv_new))
        return 2.0 * n * max_conv / (S_new * S_new)
    
    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        # Random pairs
        for _ in range(min(20, n // 2)):
            if time.monotonic() - start >= time_budget:
                break
            
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            
            vi, vj = a[i], a[j]
            x0 = [vi, vj]
            
            result = scipy_minimize(
                fun=lambda p: obj_2d(p, i, j),
                x0=x0,
                method='Nelder-Mead',
                options={'maxiter': 200, 'xatol': 1e-10, 'fatol': 1e-12,
                         'adaptive': True}
            )
            
            if result.success and result.fun < 1e20:
                x_new, y_new = max(0, result.x[0]), max(0, result.x[1])
                if abs(x_new - vi) > 1e-12 or abs(y_new - vj) > 1e-12:
                    a[i] = x_new
                    a[j] = y_new
                    conv = compute_conv(a)  # Recompute for consistency
                    obj_now = 2.0 * n * float(np.max(conv)) / (float(np.sum(a))**2)
                    if obj_now < best_obj:
                        best_obj = obj_now
                        best_a = a.copy()
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0], dtype=np.float64)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0).astype(np.float64)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        c1, c2 = n // 3, 2 * n // 3
        w = n // 10
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-((x - c1)**2) / (2*w*w)) + np.exp(-((x - c2)**2) / (2*w*w))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        # Low-frequency wave
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * rng.integers(2, 6) * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a
