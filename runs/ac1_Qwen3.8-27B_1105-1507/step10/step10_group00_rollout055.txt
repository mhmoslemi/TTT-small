    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

    n = len(a)

    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lbfqg_phase(a_init, time_budget, start_T=1.0, end_T=80.0, n_anneal_steps=8):
    """
    L-BFGS optimization on smoothed objective with temperature annealing.
    Uses softmax parameterization: a = softmax(x).
    """
    n = len(a_init)
    a = a_init / np.sum(a_init)
    # Initialize x from a: x = log(a) (with floor to avoid -inf)
    x = np.log(np.maximum(a, 1e-15))
    x = x - np.mean(x)  # center for numerical stability

    best_a = a.copy()
    best_obj = obj_value(a)
    start = time.monotonic()
    total_anneal = time_budget * 0.85

    for step in range(n_anneal_steps):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 30:
            break

        # Anneal temperature
        frac = step / max(1, n_anneal_steps - 1)
        T = start_T + (end_T - start_T) * frac ** 0.7
        step_budget = min(total_anneal / n_anneal_steps, remaining - 10)

        if step_budget < 15:
            break

        # Define objective and gradient for scipy
        def fun_and_grad(x_eval):
            return smooth_obj_and_grad_x(x_eval, T)

        try:
            res = minimize(
                fun_and_grad, x, method='L-BFGS-B',
                jac=True, options={
                    'maxiter': 500,
                    'maxfun': 2000,
                    'ftol': 1e-14,
                    'gtol': 1e-10,
                    'maxls': 50,
                }
            )
            if res.x is not None:
                a_new = softmax_fn(res.x)
                a_new = a_new / a_new.sum()
                obj_new = obj_value(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    x = res.x.copy()
                    print(f"    L-BFGS anneal step {step} (T={T:.1f}): obj={best_obj:.8f} **")
                else:
                    print(f"    L-BFGS anneal step {step} (T={T:.1f}): obj={obj_new:.8f} (no improve)")
        except Exception as e:
            print(f"    L-BFGS anneal step {step} (T={T:.1f}): error {e}")

    # Final evaluation at high T
    if time.monotonic() - start < time_budget - 20:
        T_final = min(end_T * 2, 200.0)
        x_final = np.log(np.maximum(best_a, 1e-15))
        x_final = x_final - np.mean(x_final)
        try:
            def fun_and_grad_final(x_eval):
                return smooth_obj_and_grad_x(x_eval, T_final)
            res = minimize(
                fun_and_grad_final, x_final, method='L-BFGS-B',
                jac=True, options={
                    'maxiter': 300,
                    'maxfun': 1000,
                    'ftol': 1e-15,
                    'gtol': 1e-12,
                }
            )
            if res.x is not None:
                a_final = softmax_fn(res.x)
                a_final = a_final / a_final.sum()
                obj_final = obj_value(a_final)
                if obj_final < best_obj:
                    best_obj = obj_final
                    best_a = a_final.copy()
                    print(f"    L-BFGS final (T={T_final:.1f}): obj={best_obj:.8f} **")
        except Exception:
            pass

    return best_a


def subgrad_projected_descent(a_init, time_budget, lr_init=0.02):
    """
    Projected subgradient descent on the true (non-smooth) objective.
    Uses adaptive step size and momentum.
    """
    n = len(a_init)
    a = a_init / np.sum(a_init)
    best_a = a.copy()
    best_obj = obj_value(a)
    start = time.monotonic()
    lr = lr_init
    momentum = np.zeros(n)
    beta = 0.3
    it = 0
    best_it = 0
    decay = 0.9997

    while time.monotonic() - start < time_budget:
        grad = subgradient_of_max_conv(a)
        # Remove mean to stay on simplex
        grad_centered = grad - np.sum(grad) / n

        momentum = beta * momentum - lr * grad_centered
        a_new = a + momentum
        a_new = project_simplex(a_new)

        obj_new = obj_value(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            a = a_new
        else:
            a = a_new  # still move, but keep best

        lr *= decay
        it += 1

        if it - best_it > 5000:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = max(lr, 0.005)
            beta = min(beta + 0.05, 0.6)
            best_it = it

        if it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SubGrad it{it}: obj={best_obj:.8f}, lr={lr:.6f}, t={elapsed:.1f}s")

    return best_a


def frank_wolfe_true(a_init, time_budget, max_iters=50000):
    """
    Frank-Wolfe method on the true non-smooth objective.
    Uses subgradient for direction, with away-step variant.
    """
    n = len(a_init)
    a = a_init / np.sum(a_init)
    best_a = a.copy()
    best_obj = obj_value(a)
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        if it % 10000 == 0 and it > 0:
            print(f"    FW it{it}: obj={best_obj:.8f}, stall={stall}, t={time.monotonic()-start:.1f}s")

        grad = subgradient_of_max_conv(a)

        # Standard FW direction
        m_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[m_min] = 1.0

        # Away-step: also try moving away from max coordinate
        m_max = int(np.argmax(a))
        w = np.zeros(n)
        w[m_max] = 1.0

        # Try standard step
        def eval_step(direction, max_t=1.0):
            best_t = 0.0
            best_f = best_obj
            # Try several t values
            for t_try in [1.0/(it+2), 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
                if t_try > max_t:
                    continue
                a_try = (1 - t_try) * a + t_try * direction
                a_try = np.maximum(a_try, 0)
                s = np.sum(a_try)
                if s < 1e-10:
                    continue
                a_try = a_try / s
                f_try = obj_value(a_try)
                if f_try < best_f:
                    best_f = f_try
                    best_t = t_try
            return best_t, best_f

        t_std, f_std = eval_step(v)

        # Try away-step (move toward v and away from w)
        t_away, f_away = 0.0, np.inf
        if a[m_max] > 0.05:
            # away direction: v - a[m_max]*w, projected
            for t_try in [0.05, 0.1, 0.15, 0.2]:
                a_try = (1 - t_try) * a + t_try * (v - a[m_max] * w)
                a_try = np.maximum(a_try, 0)
                s = np.sum(a_try)
                if s < 1e-10:
                    continue
                a_try = a_try / s
                f_try = obj_value(a_try)
                if f_try < f_away:
                    f_away = f_try
                    t_away = t_try

        # Pick best
        if f_away < f_std:
            t_best = t_away
            f_best = f_away
            a_new = (1 - t_best) * a + t_best * (v - a[m_max] * w)
        else:
            t_best = t_std
            f_best = f_std
            if t_best > 0:
                a_new = (1 - t_best) * a + t_best * v
            else:
                a_new = a

        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 0:
            a_new = a_new / s_new

        obj_new = obj_value(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new
            stall += 1
            if stall > 200:
                # Restart with perturbation
                rng = np.random.default_rng(42 + it)
                pert = 0.01 * rng.normal(0, 1, n)
                a = best_a + pert
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

        it += 1

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a_init, time_budget, max_iters=40):
    """LP-based polishing with line search."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = obj_value(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start >= time_budget - 5:
            break

        g_lp, _ = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Line search
        best_t = 0
        best_f = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            if a_try.sum() < 0.01:
                continue
            a_try = a_try / a_try.sum()
            f_try = obj_value(a_try)
            if f_try < best_f:
                best_f = f_try
                best_t = t

        if best_t > 0 and best_f < best_obj - 1e-14:
            a_new = (1 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_obj = best_f
            best_a = a_new.copy()
            a = a_new
        else:
            break

    return best_a


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    return a / a.sum()


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 4:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    return a / a.sum()


def try_length(a_best, n_try, time_budget, rng):
    """Resample to different length and optimize."""
    n_old = len(a_best)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        return None, np.inf
    a_try = a_try / s

    # Quick L-BFGS
    remaining = time_budget
    if remaining > 30:
        a_try = lbfqg_phase(a_try, time_budget=min(30.0, remaining * 0.4),
                            start_T=5.0, end_T=50.0, n_anneal_steps=3)

    # Quick subgradient
    remaining = time_budget - 0  # approximate
    if remaining > 20:
        a_try = subgrad_projected_descent(a_try, time_budget=min(15.0, remaining * 0.3))

    obj_try = obj_value(a_try)
    return a_try, obj_try


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines L-BFGS smoothed optimization,
    projected subgradient descent, Frank-Wolfe, and LP polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent if available
    n_default = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = obj_value(parent)
                best_score = obj
                best_seq = parent.copy()
                n_default = parent.size
                print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = u_shaped_init(n_default, sharpness=3.0)
        best_score = obj_value(best_seq)
        print(f"Default U-shaped init (n={n_default}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: L-BFGS smoothed optimization ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 1: L-BFGS Smoothed (n={n_current}, budget={remaining:.0f}s) ===")
        alloc = min(remaining * 0.3, 400.0)
        a_lg = lbfqg_phase(best_seq, time_budget=alloc,
                           start_T=1.0, end_T=80.0, n_anneal_steps=8)
        obj_lg = obj_value(a_lg)
        if obj_lg < best_score:
            best_score = obj_lg
            best_seq = a_lg.copy()
            print(f"  Phase 1 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: no improvement (obj={obj_lg:.8f})")

    # === Phase 2: Projected subgradient descent ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 2: Subgradient + Projection (budget={remaining:.0f}s) ===")
        alloc = min(remaining * 0.2, 200.0)
        a_sg = subgrad_projected_descent(best_seq, time_budget=alloc, lr_init=0.015)
        obj_sg = obj_value(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase 2 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: no improvement (obj={obj_sg:.8f})")

    # === Phase 3: Frank-Wolfe on true objective ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 3: Frank-Wolfe (budget={remaining:.0f}s) ===")
        alloc = min(remaining * 0.15, 150.0)
        a_fw = frank_wolfe_true(best_seq, time_budget=alloc, max_iters=30000)
        obj_fw = obj_value(a_fw)
        if obj_fw < best_score:
            best_score = obj_fw
            best_seq = a_fw.copy()
            print(f"  Phase 3 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: no improvement (obj={obj_fw:.8f})")

    # === Phase 4: LP polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: LP Polishing (budget={remaining:.0f}s) ===")
        alloc = min(remaining * 0.2, 150.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, max_iters=30)
        obj_lp = obj_value(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 4 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4: no improvement (obj={obj_lp:.8f})")

    # === Phase 5: Diverse restarts + re-optimize ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 5: Diverse Restarts (budget={remaining:.0f}s) ===")
        n_cur = len(best_seq)

        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            mode = r
            a_r = diverse_init(n_cur, rng, mode=mode)
            # Blend with current best
            blend = 0.4
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            # Quick L-BFGS
            alloc_r = min(remaining * 0.25, 60.0)
            if alloc_r > 15:
                a_r = lbfqg_phase(a_r, time_budget=alloc_r,
                                  start_T=5.0, end_T=50.0, n_anneal_steps=3)

            # LP polish
            remaining_r = deadline - time.monotonic()
            if remaining_r > 20:
                a_r = lp_polish(a_r, time_budget=min(20.0, remaining_r * 0.4), max_iters=10)

            obj_r = obj_value(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 6: Length Exploration (budget={remaining:.0f}s) ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 100, n_best - 50,
            n_best + 50, n_best + 100, n_best + 200,
            600, 700, 900, 1000, 1100, 1200, 1500
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) >= 30]
        n_options = n_options[:4]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            a_try, obj_try = try_length(best_seq, n_try, remaining * 0.6, rng)
            if a_try is not None and obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 7: Final Polish (budget={remaining:.0f}s) ===")
        # Quick subgradient
        alloc_sg = min(remaining * 0.3, 30.0)
        if alloc_sg > 10:
            a_final = subgrad_projected_descent(best_seq, time_budget=alloc_sg, lr_init=0.005)
            obj_final = obj_value(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()

        # LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_final2 = lp_polish(best_seq, time_budget=min(remaining - 5, 40.0), max_iters=15)
            obj_final2 = obj_value(a_final2)
            if obj_final2 < best_score:
                best_score = obj_final2
                best_seq = a_final2.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

    # Validate and return
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = obj_value(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
