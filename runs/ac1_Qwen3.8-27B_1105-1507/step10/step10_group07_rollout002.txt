import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_subgradient(a, top_k=5):
    """Compute subgradient of max_k conv(a,a)[k]."""
    n = len(a)
    conv = compute_conv(a)
    # Find top k peaks
    top_idx = np.argsort(conv)[::-1][:top_k]
    peak_vals = conv[top_idx]
    max_pv = peak_vals[0]
    # Weights: emphasize the maximum, decay for others
    weights = np.exp((peak_vals - max_pv) * 20.0)
    weights /= weights.sum()
    
    sg = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        for j in range(n):
            if 0 <= k - j < n:
                if 2 * j == k:
                    sg[j] += w * a[j]
                else:
                    sg[j] += w * 2 * a[k - j]
    return sg, conv

def mirror_descent_aaccelerated(a, time_budget=200.0, top_k=10, verbose=False):
    """Nesterov-accelerated mirror descent with entropy proximal."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = a.copy()  # Current point
    y = a.copy()  # Yurmov-Malyshev point (accelerated)
    it = 0
    eta = 1.0  # Step size parameter
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Nesterov: y = (1 + (it+1)/it) * x - ... 
        # Actually standard Nesterov: y_k = (beta_k - 1)/(beta_k) * x_k + (1/beta_k) * y_{k-1}
        # where beta_k = 2/(k+2)
        # Simplified: t_k = (1 + sqrt(1+4*t_{k-1}^2))/2, y = x + (t_{k-1}/t_k)*(y - x_prev)
        
        # Simpler version: heavy-ball momentum
        if it == 0:
            y = x.copy()
            x_prev = x.copy()
        else:
            momentum = 2.0 / (it + 2.0)
            y = (1.0 - momentum) * y + momentum * x
            # Actually let's use a simpler acceleration
            # y = x + alpha * (x - x_prev)
            # alpha = (sqrt(1+4*it^2) - 1) / (2*it) approx 1 for large it
            alpha = min(0.95, 1.0 - 2.0/(it + 2.0))
            y = x + alpha * (x - x_prev)
            y = np.maximum(y, 0)
            s_y = y.sum()
            if s_y < 1e-10:
                y = x.copy()
            else:
                y = y / s_y
        
        x_prev = x.copy()
        
        # Compute subgradient at y
        sg, conv = compute_subgradient(y, top_k=top_k)
        
        # Mirror step: x_new = y * exp(-eta * sg) / Z
        # Adaptive eta
        if it < 50:
            eta = 5.0 / np.sqrt(n)
        else:
            eta = 5.0 / np.sqrt(n) * (1.0 + 0.5 * np.sin(progress * np.pi))
        
        weights_sg = y * np.exp(-eta * sg)
        Z = weights_sg.sum()
        if Z < 1e-30:
            x_new = y.copy()
        else:
            x_new = weights_sg / Z
        
        # Project to simplex
        x_new = np.maximum(x_new, 0)
        s_xn = x_new.sum()
        if s_xn > 1e-10:
            x_new = x_new / s_xn
        else:
            x_new = np.ones(n) / n
        
        x = x_new
        obj = compute_obj(x)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = x.copy()
        
        it += 1
        if verbose and it % 50 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.4f}, t={time.monotonic()-start:.0f}s")
    
    return best_a

def compute_conv(a):

def compute_obj(a):

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def golden_section_1d(f, a, b, tol=1e-6, max_iter=100):
    """Golden section search for minimum of f on [a, b]."""
    gr = (np.sqrt(5) - 1) / 2
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
