Let me also think about the LP approach. The key insight of the LP alternating is:

Given current a, find g that minimizes max_k conv(a, g)[k] subject to g >= 0, sum(g) = 1.

This is a linear program because conv(a, g)[k] is linear in g (since a is fixed).

The LP is:
min M
s.t. sum_i a[i] * g[k-i] <= M for all k = 0, ..., 2n-2
     sum(g) = 1
     g >= 0

This has 2n-1 + 1 constraints and n+1 variables. For n=800, that's about 1600 constraints and 801 variables. This is a medium-sized LP.

The previous code uses cvxpy with multiple solvers. Let me check if this is the bottleneck.

For each LP iteration, we need to build the constraint matrix (2n-1 x n) and solve the LP. The matrix building is O(n^2) which for n=800 is about 640K operations - fast. The LP solve is the expensive part.

With modern solvers, an LP with 1600 constraints and 800 variables should take maybe 0.1-1 second. So we can do about 100-1000 LP iterations in 1500 seconds.

**Key idea: Instead of just blending toward the LP solution, use the LP dual to get additional information.**

The dual of the LP is:

The dual variables lambda_k tell us which convolution peaks are "active" (binding). At the optimum, the active constraints tell us where the peaks are.

We can use this information to directly modify a to reduce the peak at those locations.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternating minimization, smoothed mirror descent, and multi-n exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_smoothed_obj_and_grad(a, tau):
    """Compute smoothed objective value and gradient."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    # Avoid overflow
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
        smooth_val = max_c
    else:
        p = exp_vals / sum_exp
        smooth_val = max_c + np.log(sum_exp) / tau
    
    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j] for valid indices
    # = 2 * cross_correlation(p, a)[j]
    N = 1
    while N < len(p) + n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cc = ifft(P * np.conj(A)).real
    grad = 2.0 * cc[:n]
    
    return smooth_val, grad


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def mirror_descent_smoothed(a, time_budget, tau_init=5.0, tau_final=500.0, verbose=False):
    """Smoothed mirror descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    tau = tau_init
    eta = 1.0 / np.sqrt(n)  # initial step size
    patience = 200
    counter = 0
    
    while time.monotonic() - start < time_budget:
        elapsed_frac = (time.monotonic() - start) / time_budget
        # Animate tau from tau_init to tau_final
        tau = tau_init * (tau_final / tau_init) ** elapsed_frac
        
        smooth_val, grad = compute_smoothed_obj_and_grad(a, tau)
        
        # Mirror descent step: a_new = a * exp(-eta * grad) / Z
        log_update = -eta * grad
        log_update = np.clip(log_update, -50, 50)
        weights = a * np.exp(log_update)
        Z = float(np.sum(weights))
        if Z < 1e-300:
            a_new = a.copy()
        else:
            a_new = weights / Z
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            counter = 0
            a = a_new
        else:
            counter += 1
            if counter >= patience:
                # Reduce step size
                eta *= 0.7
                counter = 0
                if eta < 1e-10:
                    break
                a = best_a.copy()
    
    return best_a

def fourier_escape(a, rng, strength=0.05):
    """Add Fourier perturbation for escape."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2 * np.pi / n
        amp = strength / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_to_length(a, new_n):

    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent(a, time_budget, verbose=False):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    # Precompute: for each coordinate i, we need to know how changing a[i] affects the convolution
    # conv[k] changes by 2*a[k-i]*delta for k in [i, i+n-1]
    # So we can do a "greedy" coordinate update.
    
    for i in range(n):
        if time.monotonic() - start >= time_budget:
            break
        
        # Binary search for optimal a[i]
        # The objective as a function of a[i] (with others fixed) is:
        # 2n * max_k (base_conv[k] + 2*a[i]*a[k-i]) / (sum_base + a[i])^2
        # This is unimodal in a[i] (for a[i] >= 0), so we can binary search.
        
        # For efficiency, just try a few values
        old_val = a[i]
        candidates = [old_val * 0.5, old_val * 0.8, old_val * 1.2, old_val * 2.0, 0.0, old_val + 0.01]
        candidates = [max(0, c) for c in candidates]
        
        best_i_obj = best_obj
        best_i_val = old_val
        
        for cv in candidates:
            a[i] = cv
            s_j = float(np.sum(a))
            if s_j < 1e-10:
                a[i] = old_val
                continue
            obj_j = compute_obj(a)
            if obj_j < best_i_obj:
                best_i_obj = obj_j
                best_i_val = cv
        
        a[i] = best_i_val
        
        if best_i_obj < best_obj - 1e-12:
            best_obj = best_i_obj
            best_a = a.copy()
    
    return best_a
