"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP fixed-point with Anderson acceleration, adaptive perturbation,
support refinement, multi-n exploration, and frequency-domain insights.
"""

import numpy as np
import time
import cvxpy as cp
import scipy.sparse as sp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build sparse convolution matrix A where A@b = a*b (linear conv)."""
    n = len(a)
    idx = np.arange(n)
    rows = (idx[None, :] + idx[:, None]).ravel()
    cols = np.repeat(idx, n)
    vals = np.tile(a, n)
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve(a, preferred_solvers=None):
    """Solve: min M s.t. (a*b)[k] <= M for all k, b >= 0, sum(b) = 1.
    Returns (b_optimal_normalized, M_optimal) or (None, None)."""
    n = len(a)
    if n > 4000:
        return None, None

    A = build_conv_matrix(a)

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if preferred_solvers is None:
        preferred_solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]

    for solver in preferred_solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if b_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(b_var.value, 0)
                sb = float(np.sum(b_val))
                if sb > 1e-10:
                    return b_val / sb, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_fixed_point(a, time_budget, verbose=False, seed=42, max_iters=500,
                   restart_prob=0.05, pert_scale=0.1):
    """
    LP fixed-point iteration with adaptive step size and random restarts.
    Finds local minimum of max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Anderson acceleration memory
    aa_mem_size = 5
    aa_a = [a.copy()]
    aa_f = [a.copy()]  # F(a) = LP solution

    it = 0
    stall_count = 0
    last_best_it = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Solve LP
        b_lp, M_lp = lp_solve(a)
        if b_lp is None:
            break

        # KKT residual: how far is a from being the LP solution
        residual = float(np.max(np.abs(a - b_lp)))

        # Anderson acceleration: extrapolate
        trial = b_lp.copy()
        if len(aa_a) >= 2:
            # Simple extrapolation: if improvement was consistent, extrapolate
            prev_diff = aa_a[-1] - aa_a[-2]
            cur_diff = b_lp - aa_a[-1]
            # Estimate extrapolation factor
            dot = float(np.dot(prev_diff, cur_diff))
            norm_prev = float(np.dot(prev_diff, prev_diff))
            if norm_prev > 1e-20:
                alpha = 0.5 * dot / norm_prev  # half-step extrapolation
                alpha = np.clip(alpha, 0.0, 1.5)
                trial = b_lp + alpha * cur_diff
                trial = np.maximum(trial, 0)
                st = float(np.sum(trial))
                if st > 1e-10:
                    trial = trial / st
                else:
                    trial = b_lp.copy()
                # Verify trial is better
                obj_trial = compute_obj(trial)
                obj_bp = compute_obj(b_lp)
                if obj_trial > obj_bp:
                    trial = b_lp.copy()

        # Update memory
        aa_a.append(a.copy())
        aa_f.append(b_lp.copy())
        if len(aa_a) > aa_mem_size:
            aa_a.pop(0)
            aa_f.pop(0)

        # Evaluate and track
        obj_new = compute_obj(trial)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = trial.copy()
            stall_count = 0
            last_best_it = it
        else:
            stall_count += 1

        a = trial
        it += 1

        # Random perturbation to escape local minima
        if stall_count > 30 and rng.random() < restart_prob:
            # Structured perturbation based on where conv is close to max
            conv = compute_conv(a)
            max_conv = float(np.max(conv))
            # Perturb to reduce the peaks
            peak_lags = np.where(conv > 0.9 * max_conv)[0]
            if len(peak_lags) > 0:
                # Move mass away from positions that contribute to peaks
                pert = rng.normal(0, pert_scale, n)
                # Weight perturbation by how much each position contributes to peaks
                contrib = np.zeros(n)
                for k in peak_lags:
                    k = int(k)
                    lo = max(0, k - n + 1)
                    hi = min(k, n - 1)
                    contrib[lo:hi + 1] += a[lo:hi + 1]
                pert *= (0.5 + contrib / max(1e-10, contrib.max()))
                a = a + pert
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s < 1e-10:
                    a = best_a.copy()
                else:
                    a = a / s
            stall_count = 0
            if verbose:
                print(f"    LP-FP: perturbed at it={it}, obj={compute_obj(a):.10f}")

        # Support refinement: try zeroing small elements
        if it % 50 == 0:
            a_test = a.copy()
            threshold = 0.005 * float(np.mean(a))
            mask = a_test < threshold
            if mask.sum() > 0 and mask.sum() < n // 2:
                a_test[mask] = 0
                s_test = float(np.sum(a_test))
                if s_test > 1e-10:
                    a_test = a_test / s_test
                    obj_test = compute_obj(a_test)
                    if obj_test < best_obj:
                        best_obj = obj_test
                        best_a = a_test.copy()
                        a = a_test.copy()
                        if verbose:
                            print(f"    LP-FP: support pruned at it={it}, obj={best_obj:.10f}")

        if verbose and it % 20 == 0:
            print(f"    LP-FP it={it}: obj={best_obj:.10f}, res={residual:.6f}, stall={stall_count}")

    return best_a


def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def gd_refine(a, time_budget, tau_levels=None, verbose=False):
    """Nesterov accelerated gradient descent with staircase tau."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 20, 80, 300, 1000, 3000]

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0

    for tau in tau_levels:
        it_level = 0
        max_level_iters = 3000
        while time.monotonic() - start < time_budget:
            if it_level >= max_level_iters:
                break
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
            it += 1
            it_level += 1
            lr = max(1e-8, lr * 0.99999)

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation with normalization."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_diverse_init(n, rng):
    """Generate a diverse set of initial sequences."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangle
    x = np.linspace(0, 1, n)
    a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
    inits.append(a / a.sum())

    # Semicircle
    x = np.linspace(-1, 1, n)
    a = np.sqrt(np.maximum(1 - x ** 2, 0.0001))
    inits.append(a / a.sum())

    # Cosine
    x = np.linspace(0, np.pi, n)
    a = np.maximum(0.5 + 0.5 * np.cos(x), 0.001)
    inits.append(a / a.sum())

    # Two Gaussian bumps
    c1, c2 = n // 4, 3 * n // 4
    w = max(1, n // 6)
    a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
    a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
    inits.append(a / a.sum())

    # Flat top with tapered edges
    x = np.linspace(0, 1, n)
    a = np.where((x > 0.15) & (x < 0.85), 1.0,
                 np.maximum(0.6 - 2 * np.abs(x - 0.5) + 0.1, 0.01))
    inits.append(a / a.sum())

    # Random smooth
    a = rng.uniform(0, 1, n)
    # Smooth with a window
    w = max(3, n // 20)
    kernel = np.ones(w) / w
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.001)
    inits.append(a / a.sum())

    # Low-rank sinusoidal
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    for f in rng.integers(1, 6, size=3):
        a += 0.3 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())

    # Edge-concentrated
    x = np.arange(n)
    a = np.exp(-x / (n / 3)) + np.exp(-(n - 1 - x) / (n / 3))
    inits.append(a / a.sum())

    # W-shaped (for even n)
    if n >= 4:
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.abs(4 * t - 2)
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())

    return inits


def frequency_perturb(a, rng, scale=0.05):
    """Perturb in frequency domain to explore different shapes."""
    n = len(a)
    N = 1
    while N < 4 * n:
        N <<= 1
    A = fft(a, N)
    # Add random phase perturbation to non-DC components
    mask = np.ones(N, dtype=bool)
    mask[0] = False
    phases = rng.uniform(-scale, scale, N) * mask
    A *= (1 + 1j * phases)
    a_new = ifft(A).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining LP fixed-point with
    Anderson acceleration, adaptive perturbation, frequency-domain exploration,
    support refinement, and multi-n search.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # === Load parent ===
    parent_n = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                parent_n = parent.size
                print(f"Parent loaded (n={parent_n}): obj={best_score:.10f}")
    except Exception:
        parent = None

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        parent_n = n0
        print(f"Default init (n={n0}): obj={best_score:.10f}")

    n_cur = len(best_seq)

    # === Phase 1: Diverse initialization and quick screening ===
    print(f"\n=== Phase 1: Diverse Init Screening (n={n_cur}) ===")
    remaining = deadline - time.monotonic()

    candidates = [best_seq.copy()]
    inits = make_diverse_init(n_cur, rng)
    for i, init_a in enumerate(inits):
        candidates.append(init_a.copy())

    # Quick screen with short GD
    screen_time_per = min(remaining * 0.03, 8.0)
    for i, cand in enumerate(candidates):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if i == 0:  # Skip parent (already optimized)
            continue
        alloc = min(screen_time_per, remaining * 0.1)
        if alloc < 3:
            continue
        cand_screened = gd_refine(cand, time_budget=alloc,
                                   tau_levels=[5, 30, 100], verbose=False)
        obj_s = compute_obj(cand_screened)
        if obj_s < best_score:
            best_score = obj_s
            best_seq = cand_screened.copy()
            print(f"  Init [{i}]: obj={best_score:.10f} **")
        else:
            print(f"  Init [{i}]: obj={obj_s:.10f}")

    print(f"  After screening: best={best_score:.10f}")

    # === Phase 2: Primary LP fixed-point on best candidate ===
    print(f"\n=== Phase 2: LP Fixed-Point (primary) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.30, 400.0)
    if p2_budget > 30 and n_cur <= 2500:
        a2 = lp_fixed_point(best_seq, time_budget=p2_budget, verbose=True,
                            seed=100, max_iters=1000)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 LP: obj={best_score:.10f} **")
        else:
            print(f"  Phase2 LP: obj={obj2:.10f}")

        # GD after LP to escape
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.05, 30.0)
            a2g = gd_refine(a2, time_budget=alloc, tau_levels=[50, 200, 800])
            obj2g = compute_obj(a2g)
            if obj2g < best_score:
                best_score = obj2g
                best_seq = a2g.copy()
                print(f"  Phase2 LP+GD: obj={best_score:.10f} **")

    # === Phase 3: LP fixed-point on diverse candidates ===
    print(f"\n=== Phase 3: LP on Multiple Candidates ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.25, 300.0)

    # Select top candidates for further optimization
    top_candidates = [best_seq.copy()]
    for init_a in inits[:5]:
        top_candidates.append(init_a.copy())

    per_cand_budget = p3_budget / len(top_candidates)
    for i, cand in enumerate(top_candidates):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_cand_budget, remaining * 0.4)
        if alloc < 15:
            continue
        if i == 0:
            continue  # Already done in Phase 2

        cand = gd_refine(cand, time_budget=min(alloc * 0.2, 10),
                         tau_levels=[10, 50], verbose=False)
        if n_cur <= 2500:
            cand = lp_fixed_point(cand, time_budget=alloc * 0.7, verbose=False,
                                  seed=200 + i, max_iters=200)
        obj_c = compute_obj(cand)
        if obj_c < best_score:
            best_score = obj_c
            best_seq = cand.copy()
            print(f"  Cand [{i}]: obj={best_score:.10f} **")
        else:
            print(f"  Cand [{i}]: obj={obj_c:.10f}")

    # === Phase 4: Frequency-domain perturbation + re-optimize ===
    print(f"\n=== Phase 4: Frequency Perturbation ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for attempt in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            a_pert = frequency_perturb(best_seq, rng, scale=0.03 + 0.02 * attempt)
            # Quick GD
            alloc1 = min(remaining * 0.08, 15.0)
            a_pert = gd_refine(a_pert, time_budget=alloc1, tau_levels=[20, 100])
            # LP if time
            remaining = deadline - time.monotonic()
            if remaining > 30 and n_cur <= 2500:
                alloc2 = min(remaining * 0.15, 40.0)
                a_pert = lp_fixed_point(a_pert, time_budget=alloc2, verbose=False,
                                        seed=500 + attempt, max_iters=100)
            obj_p = compute_obj(a_pert)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_pert.copy()
                print(f"  Freq pert {attempt}: obj={best_score:.10f} **")
            else:
                print(f"  Freq pert {attempt}: obj={obj_p:.10f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()

    n_cur = len(best_seq)
    n_options = sorted(set([
        n_cur,
        n_cur - 100, n_cur + 100,
        n_cur - 200, n_cur + 200,
        n_cur + 400, n_cur + 600,
        600, 700, 900, 1000, 1200, 1500, 2000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 3000 and n != n_cur]
    # Prioritize: close to current, then larger
    n_options.sort(key=lambda x: abs(x - n_cur))
    n_options = n_options[:5]
    print(f"  Trying n: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample(best_seq, n_try)
        # Quick GD
        alloc1 = min(remaining * 0.15, 25.0)
        if alloc1 > 10:
            a_try = gd_refine(a_try, time_budget=alloc1, tau_levels=[10, 50, 200])

        # LP if feasible
        remaining = deadline - time.monotonic()
        if n_try <= 2000 and remaining > 40:
            alloc2 = min(remaining * 0.2, 40.0)
            if alloc2 > 15:
                a_try = lp_fixed_point(a_try, time_budget=alloc2, verbose=False,
                                       seed=n_try, max_iters=100)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 6: Final intensive optimization ===
    print(f"\n=== Phase 6: Final Intensive Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    # LP fixed-point with high iteration count
    if remaining > 40 and n_final <= 2500:
        alloc = min(remaining * 0.3, 100.0)
        a_f1 = lp_fixed_point(best_seq, time_budget=alloc, verbose=True,
                              seed=999, max_iters=2000,
                              restart_prob=0.08, pert_scale=0.05)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final LP: obj={best_score:.10f} **")
        else:
            print(f"  Final LP: obj={obj_f1:.10f}")

    # High-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 50.0)
        a_f2 = gd_refine(best_seq, time_budget=alloc,
                         tau_levels=[100, 500, 2000, 5000], verbose=False)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final GD: obj={best_score:.10f} **")

    # Another LP round
    remaining = deadline - time.monotonic()
    if remaining > 30 and n_final <= 2500:
        alloc = min(remaining * 0.25, 60.0)
        a_f3 = lp_fixed_point(best_seq, time_budget=alloc, verbose=True,
                              seed=12345, max_iters=1000,
                              restart_prob=0.03, pert_scale=0.03)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final LP2: obj={best_score:.10f} **")

    # Ultra high tau GD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 40.0)
        a_f4 = gd_refine(best_seq, time_budget=alloc,
                         tau_levels=[500, 2000, 5000, 10000], verbose=False)
        obj_f4 = compute_obj(a_f4)
        if obj_f4 < best_score:
            best_score = obj_f4
            best_seq = a_f4.copy()
            print(f"  Ultra GD: obj={best_score:.10f} **")

    # === Phase 7: Support refinement ===
    print(f"\n=== Phase 7: Support Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 15 and n_final <= 2500:
        # Try progressive thresholding
        for thresh_mult in [0.001, 0.003, 0.005, 0.01]:
            a_ref = best_seq.copy()
            threshold = thresh_mult * float(np.mean(a_ref))
            mask = a_ref < threshold
            if 0 < mask.sum() < n_final // 2:
                a_ref[mask] = 0
                s_ref = float(np.sum(a_ref))
                if s_ref > 1e-10:
                    a_ref = a_ref / s_ref
                    # Quick LP to re-optimize
                    if remaining > 10:
                        a_ref = lp_fixed_point(a_ref, time_budget=5,
                                               verbose=False, seed=777, max_iters=30)
                    obj_ref = compute_obj(a_ref)
                    if obj_ref < best_score:
                        best_score = obj_ref
                        best_seq = a_ref.copy()
                        print(f"  Support thresh={thresh_mult}: obj={best_score:.10f} **")
                        remaining = deadline - time.monotonic()
                    else:
                        print(f"  Support thresh={thresh_mult}: obj={obj_ref:.10f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
        print(f"\n  Validated: score={best_score:.10f}")
    else:
        result = [1.0] * 100
        best_score = 2.0
        print("\n  WARNING: validation failed, using fallback")

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
