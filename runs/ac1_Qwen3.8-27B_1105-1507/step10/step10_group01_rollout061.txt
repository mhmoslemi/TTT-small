def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])

    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    """Build (2n-1) x n convolution matrix: A[k,j] = a[j] if 0<=k-j<n else 0."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A

        if remaining < 1.0:
            break

        g, M_lp, dual_p = solve_lp_minpeak(a, warm_g, warm_M)
        if g is None:
            break

        warm_g = g.copy()
        warm_M = M_lp
        iter_count += 1

        # Line search
        best_t, obj_at_t = line_search_aggressive(a, g)

        if obj_at_t < best_obj - 1e-13:
            a_new = (1 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                best_obj = obj_at_t
                best_a = a_new.copy()
                a = a_new
                stall_count = 0
                consecutive_no_improve = 0
                if verbose and (iter_count % 15 == 0 or iter_count <= 3):
                    elapsed = time.monotonic() - start
                    print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, M_lp={M_lp:.6f}, time={elapsed:.0f}s")
        else:
            consecutive_no_improve += 1
            stall_count += 1

            if consecutive_no_improve <= 3:
                for t_small in [0.05, 0.02, 0.01, 0.005]:
                    a_small = (1 - t_small) * a + t_small * g
                    a_small = np.maximum(a_small, 0)
                    s_small = float(np.sum(a_small))
                    if s_small > 1e-10:
                        a_small = a_small / s_small
                        obj_small = compute_obj(a_small)
                        if obj_small < best_obj - 1e-13:
                            best_obj = obj_small
                            best_a = a_small.copy()
                            a = a_small
                            stall_count = 0
                            consecutive_no_improve = 0
                            break

            if stall_count >= 8:
                # Perturb using subgradient
                conv_p = compute_conv(a)
                max_p = float(np.max(conv_p))
                tau = 300.0
                exp_p = np.exp(tau * (conv_p - max_p))
                p = exp_p / exp_p.sum()
                N_fft = 1
                while N_fft < 3 * n:
                    N_fft <<= 1
                P = fft(p, N_fft)
                Af = fft(a, N_fft)
                grad = 2.0 * ifft(P * np.conj(Af)).real[:n]

                lr_pert = 0.03 / (float(np.linalg.norm(grad)) + 1e-10)
                a = a - lr_pert * grad
                a = np.maximum(a, 0)
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                else:
                    a = best_a.copy()

                # Add random perturbation
                a = a + 0.008 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()

                stall_count = 0
                consecutive_no_improve = 0
                warm_g = None
                warm_M = None

                if verbose:
                    print(f"    LP it{iter_count}: perturbed, obj={compute_obj(a):.8f}")

                if iter_count > 200:
                    if verbose:
                        print(f"    LP: stopping after {iter_count} iterations, obj={best_obj:.8f}")
                    break

    return best_a


def smoothed_gradient(a, tau):
    """Smoothed gradient of log-sum-exp of convolution."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def gd_polish(a, time_budget, tau_init=100.0, tau_end=2000.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.3 * base_lr), 3.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adapt

        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 8.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9997)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it

    if verbose:
        print(f"    GD: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_raised_cosine(n):
    """Raised cosine initialization."""
    x = np.arange(n)
    a = np.cos(np.pi * x / n) ** 2
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian initialization."""
    x = np.arange(n) - n / 2.0
    a = np.exp(-0.5 * (x / (sigma_frac * n)) ** 2)
    a = a / a.sum()
    return a


def make_triangular(n):
    """Triangular (tent) initialization."""
    x = np.arange(n)
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(x - mid) / mid
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def make_exponential(n, rate=1.0):
    """Exponential ramp initialization."""
    x = np.arange(n)
    a = np.exp(-rate * x / n)
    a = a / a.sum()
    return a


def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    return (a + a[::-1]) / 2.0


def flat_top_polish(a, time_budget, verbose=False):
    """Try to equalize the top autocorrelation values (flat-top optimization)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Use high-tau gradient focusing on the top few peaks
    tau = 500.0
    lr = 0.5 / np.sqrt(n)
    momentum = 0.9
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Focus on top 5 peaks
        sorted_idx = np.argsort(conv)[::-1]
        top_k = sorted_idx[:min(10, len(conv))]
        top_vals = conv[top_k]
        top_max = float(np.max(top_vals))

        # Gradient: weighted by proximity to max
        weights = np.exp(3.0 * (top_vals - top_max))
        weights = weights / weights.sum()

        # Build gradient from top peaks
        grad = np.zeros(n)
        for k_idx, w in zip(top_k, weights):
            k = int(k_idx)
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    grad[j] += w * a[kj]
                kj2 = j - k
                if 0 <= kj2 < n:
                    grad[j] += w * a[kj2]
        grad *= 2.0

        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = best_a.copy()
            v = np.zeros(n)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            v = v * 0.5
            lr = lr * 0.7
            a = best_a.copy()

        it += 1
        tau *= 1.001
        lr = max(1e-8, lr * 0.999)

    if verbose:
        print(f"    Flat-top: obj={best_obj:.8f}, iters={it}")
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP-driven optimization, gradient polishing,
    multi-n exploration, and diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    # Load parent sequence
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = compute_obj(best_seq)
        print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    else:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: LP-Driven Optimization from Parent ===
    print(f"\n=== Phase 1: LP-Driven from parent (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 500.0)
    if p1_budget > 30:
        t0 = time.monotonic()
        a_result = lp_driven_optimize(best_seq, time_budget=p1_budget, verbose=True, rng=rng)
        obj_result = compute_obj(a_result)
        if obj_result < best_score:
            best_score = obj_result
            best_seq = a_result.copy()
            print(f"  Phase1: obj={best_score:.8f} ** (improved!)")
        else:
            print(f"  Phase1: obj={obj_result:.8f} (no improvement)")
        print(f"  Time used: {time.monotonic()-t0:.1f}s")

    # === Phase 2: High-tau GD Polish ===
    print(f"\n=== Phase 2: GD Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.10, 120.0)
    if p2_budget > 15:
        t0 = time.monotonic()
        a_gd = gd_polish(best_seq, time_budget=p2_budget, tau_init=100.0, tau_end=3000.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase2: obj={best_score:.8f} ** (improved!)")
        else:
            print(f"  Phase2: obj={obj_gd:.8f} (no improvement)")
        print(f"  Time used: {time.monotonic()-t0:.1f}s")

    # === Phase 3: LP Round 2 ===
    print(f"\n=== Phase 3: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 300.0)
    if p3_budget > 20:
        t0 = time.monotonic()
        a_lp2 = lp_driven_optimize(best_seq, time_budget=p3_budget, verbose=True, rng=np.random.default_rng(seed + 1))
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase3: obj={best_score:.8f} ** (improved!)")
        else:
            print(f"  Phase3: obj={obj_lp2:.8f} (no improvement)")
        print(f"  Time used: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    # Generate candidate n values
    n_candidates = set()
    for delta in [-200, -100, -50, 50, 100, 200, 300, 400]:
        n_candidates.add(n_cur + delta)
    n_candidates.update([500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600])
    n_candidates = sorted([n for n in n_candidates if 400 <= n <= 1800 and abs(n - n_cur) >= 40])
    # Limit to 5 candidates
    n_candidates = n_candidates[:5]
    print(f"  Trying n values: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            print("  Time running out, skipping remaining n values")
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # LP optimization (shorter budget per n)
        alloc_lp = min(remaining * 0.25, 50.0)
        if alloc_lp > 12:
            a_try = lp_driven_optimize(a_try, time_budget=alloc_lp, verbose=False, rng=rng)

        # GD polish
        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.08, 25.0)
        if alloc_gd > 8:
            a_try = gd_polish(a_try, time_budget=alloc_gd, tau_init=100.0, tau_end=1000.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} ** (improved!)")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Different Initializations ===
    print(f"\n=== Phase 5: Alternative Initializations ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    inits = [
        ("raised_cos", make_raised_cosine(n_best)),
        ("gaussian", make_gaussian(n_best, 0.35)),
        ("triangular", make_triangular(n_best)),
        ("exp_ramp", make_exponential(n_best, 1.5)),
        ("sym_parent", symmetrize(best_seq) / np.sum(symmetrize(best_seq))),
    ]

    for name, a_init in inits:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            print("  Time running out, skipping remaining inits")
            break

        a_init = np.maximum(a_init, 0)
        s_init = float(np.sum(a_init))
        if s_init < 1e-10:
            continue
        a_init = a_init / s_init
        obj_init = compute_obj(a_init)

        if obj_init >= best_score:
            print(f"  {name}: init obj={obj_init:.8f} (worse than best, skipping)")
            continue

        print(f"  {name}: init obj={obj_init:.8f}, optimizing...")
        alloc_lp = min(remaining * 0.20, 40.0)
        if alloc_lp > 10:
            a_init = lp_driven_optimize(a_init, time_budget=alloc_lp, verbose=False, rng=rng)

        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.06, 20.0)
        if alloc_gd > 8:
            a_init = gd_polish(a_init, time_budget=alloc_gd, tau_init=100.0, tau_end=800.0)

        obj_final = compute_obj(a_init)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_init.copy()
            print(f"  ** {name}: obj={best_score:.8f} ** (improved!)")
        else:
            print(f"  {name}: final obj={obj_final:.8f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final LP round
    if remaining > 30:
        alloc = min(remaining * 0.40, 250.0)
        t0 = time.monotonic()
        a_final = lp_driven_optimize(best_seq, time_budget=alloc, verbose=True, rng=np.random.default_rng(seed + 100))
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} ** (improved!)")
        print(f"  Time used: {time.monotonic()-t0:.1f}s")

    # Flat-top polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.20, 60.0)
        t0 = time.monotonic()
        a_ft = flat_top_polish(best_seq, time_budget=alloc, verbose=True)
        obj_ft = compute_obj(a_ft)
        if obj_ft < best_score:
            best_score = obj_ft
            best_seq = a_ft.copy()
            print(f"  Flat-top: obj={best_score:.8f} ** (improved!)")
        print(f"  Time used: {time.monotonic()-t0:.1f}s")

    # Final high-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.50, 80.0)
        t0 = time.monotonic()
        a_gd_final = gd_polish(best_seq, time_budget=alloc, tau_init=300.0, tau_end=5000.0, verbose=True)
        obj_gd_final = compute_obj(a_gd_final)
        if obj_gd_final < best_score:
            best_score = obj_gd_final
            best_seq = a_gd_final.copy()
            print(f"  Final GD: obj={best_score:.8f} ** (improved!)")
        print(f"  Time used: {time.monotonic()-t0:.1f}s")

    # Final validation
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    # Ensure non-negative and within bounds
    best_seq = np.maximum(best_seq, 0)
    result = [float(min(1000.0, x)) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
