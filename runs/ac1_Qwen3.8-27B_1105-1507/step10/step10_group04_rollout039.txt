"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Anderson-accelerated LP fixed-point iteration, direct subgradient
polishing with softmax smoothing, and multi-length exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    g_val = None
    M_val = None

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            kwargs = {}
            if solver == cp.MOSEK and warm_start is not None:
                # MOSEK warm start via setting initial values
                g.value = warm_start
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                break
        except Exception:
            continue

    return g_val, M_val


def anderson_accelerate(f, x0, m=5, time_budget=100, verbose=True):

    f should return the fixed-point image.
    """
    start = time.monotonic()
    n = len(x0)
    
    # Memory
    X = [x0.copy()]  # iterates

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Anderson-accelerated LP fixed-point iteration, high-tau smoothed gradient

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft

import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT. Returns array of length len(a)+len(b)-1."""
    na = len(a)
    nb = len(b)
    if na == 0 or nb == 0:
        return np.array([])
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:na + nb - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value) if M.value is not None else float(np.max(compute_conv_cross(a_norm, g_val)))
                    return g_val, M_val
        except Exception:
            continue
    return None, None


def lp_fixed_point(a):
    """Apply one LP fixed-point map: returns LP solution for input a."""
    g, _ = solve_lp(a)
    if g is None:
        return a.copy()
    return g


def anderson_lp_iteration(a0, m=5, max_iter=300, time_budget=600, verbose=True):
    """
    Anderson acceleration on the LP fixed-point iteration.
    T(a) = LP(a) = argmin_g max_k (a*g)_k s.t. sum(g)=1, g>=0
    """
    start = time.monotonic()
    n = len(a0)

    x = a0.copy()
    F = lp_fixed_point(x)

    # History for Anderson
    X_hist = [x.copy()]
    F_hist = [F.copy()]

    best_x = x.copy()
    best_obj = compute_obj(x)
    best_res = np.linalg.norm(F - x)

    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break

        res = F_hist[-1] - X_hist[-1]
        res_norm = float(np.linalg.norm(res))

        if res_norm < best_res:
            best_res = res_norm
            obj_check = compute_obj(X_hist[-1])
            if obj_check < best_obj:
                best_obj = obj_check
                best_x = X_hist[-1].copy()

        if res_norm < 1e-11:
            if verbose:
                print(f"    Anderson converged at it{it}, res={res_norm:.2e}")
            break

        if len(X_hist) < m + 1:
            # Picard step
            x_new = F_hist[-1].copy()
        else:
            # Anderson step: solve min ||sum alpha_i * r_i||^2 s.t. sum alpha = 1
            R = np.array([F_hist[-m + i] - X_hist[-m + i] for i in range(m)])
            G = R @ R.T + 1e-10 * np.eye(m)

            KKT = np.zeros((m + 1, m + 1))
            KKT[:m, :m] = G
            KKT[:m, m] = 1.0
            KKT[m, :m] = 1.0
            rhs = np.zeros(m + 1)
            rhs[m] = 1.0

            try:
                sol = np.linalg.solve(KKT, rhs)
                alphas = sol[:m]
            except np.linalg.LinAlgError:
                alphas = np.ones(m) / m

            # Stabilize: clip and renormalize
            alphas = np.clip(alphas, -1.0, 3.0)
            s_a = alphas.sum()
            if abs(s_a) < 1e-10:
                alphas = np.ones(m) / m
            else:
                alphas = alphas / s_a

            x_new = np.zeros(n)
            for i in range(m):
                x_new += alphas[i] * F_hist[-m + i]

            # Damp with Picard for stability
            x_new = 0.6 * x_new + 0.4 * F_hist[-1]
            x_new = np.maximum(x_new, 0)
            s_new = float(np.sum(x_new))
            if s_new < 1e-10:
                x_new = F_hist[-1].copy()
            else:
                x_new = x_new / s_new

        F_new = lp_fixed_point(x_new)

        X_hist.append(x_new)
        F_hist.append(F_new)
        if len(X_hist) > m + 1:
            X_hist.pop(0)
            F_hist.pop(0)

        if verbose and (it % 20 == 0 or it < 5):
            obj_now = compute_obj(x_new)
            print(f"    Anderson it{it}: res={res_norm:.2e}, obj={obj_now:.8f}, t={time.monotonic()-start:.0f}s")

        # Track best
        obj_new = compute_obj(x_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_x = x_new.copy()

    return best_x, best_obj


def smoothed_gradient(a, tau):
    """Compute smoothed objective value and gradient for f(a) = max_k (a*a)_k (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    obj_val = max_conv + np.log(sum_exp) / tau

    # Gradient: grad_j = 2 * sum_k p_k * a[k-j]
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A_f = fft(a, N)
    g_full = ifft(P * np.conj(A_f)).real
    grad = 2.0 * g_full[:n]

    return obj_val, grad, p

def kkt_correction(a, time_budget, top_k=10, n_iter=200, verbose=True):
    """
    KKT-based correction: minimize the KKT residual by adjusting a.
    At optimum: sum_{k in S} lambda_k * 2*a[k-j] = const for all j.
    We use this as a direction to further reduce the max convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Get top-k coefficients
        top_idx = np.argsort(conv)[::-1][:top_k]
        top_vals = conv[top_idx]

        # Softmax weights over top-k
        tau_w = 50.0
        exp_w = np.exp(tau_w * (top_vals - top_vals.max()))
        lambdas = exp_w / exp_w.sum()

        # Compute KKT direction: d_j = sum_i lambda_i * 2*a[k_i - j]
        d = np.zeros(n)
        for idx, lam in zip(top_idx, lambdas):
            k = int(idx)
            for j in range(n):
                ki = k - j
                if 0 <= ki < n:
                    d[j] += lam * 2.0 * a[ki]

        # The KKT condition says d should be constant. The "deficit" is d - mean(d).
        # We want to move a in a direction that makes d more constant AND reduces max conv.
        # Gradient of max conv is proportional to d (when concentrated on one k).
        # So we subtract a small multiple of (d - mean(d)) from a's log.

        # Use multiplicative update: a_new_j = a_j * exp(-eta * (d_j - mean(d)))
        eta = 0.1 / (np.std(d) + 1e-10)
        d_centered = d - d.mean()
        a_new = a * np.exp(-eta * d_centered)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step
            eta *= 0.5
            a_new = a * np.exp(-eta * d_centered)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                if eta < 1e-6:
                    break

        if verbose and it % 50 == 0:
            print(f"    KKT it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations (asymmetric to break symmetry)."""
    if mode == 0:
        # Asymmetric ramp (increasing)
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 1:
        # Asymmetric ramp (decreasing)
        a = np.arange(n, 0, -1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 2:
        # Quadratic asymmetric
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.3
        a = a / a.sum()
    elif mode == 3:
        # Exponential asymmetric
        a = np.exp(-2.0 * np.arange(n) / n)
        a = a / a.sum()
    elif mode == 4:
        # Two-bump asymmetric
        c1 = 0.3 * n
        c2 = 0.7 * n
        a = np.exp(-0.5 * ((np.arange(n) - c1) / (0.1 * n)) ** 2) + \
            0.5 * np.exp(-0.5 * ((np.arange(n) - c2) / (0.08 * n)) ** 2)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        a = (np.arange(1, n + 1)) ** 0.5
        a = a / a.sum()
    elif mode == 6:
        # Random
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 7:
        # Sinusoidal modulation on ramp
        x = np.linspace(0, 1, n)
        a = x * (1 + 0.3 * np.sin(4 * np.pi * x))
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

    a_norm = a / a.sum()

    # Initialize g uniformly
    theta = np.zeros(n)
    best_g = np.ones(n) / n
    best_val = np.inf

    for it in range(max_iter):
        # Compute g from theta
        theta_shifted = theta - theta.max()
        g = np.exp(theta_shifted)
        g /= g.sum()

        # Compute conv(a, g)
        conv = compute_conv_cross(a_norm, g)
        curr_val = float(np.max(conv))

        if curr_val < best_val:
            best_val = curr_val
            best_g = g.copy()

        if curr_val - best_val < 1e-12:
            break

        # Softmax weights
        max_c = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_c))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient: grad_j = sum_k p_k * a_norm[k-j]
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(a_norm, N)
        grad = ifft(P * np.conj(A_f)).real[:n]

        # Mirror descent step with diminishing step size
        lr = 2.0 / (np.sqrt(it + 1) * (np.linalg.norm(grad) + 1e-10))
        lr = min(lr, 1.0)
        theta -= lr * grad

    return best_g


def lp_iteration_fast(a, use_fast=False):
    """One LP iteration, using fast solver or cvxpy."""
    if use_fast:
        g = fast_inner_lp(a, tau=1000, max_iter=300)
        return g
    else:
        g, _ = solve_lp(a)
        if g is None:
            return a.copy()
        return g


def parent_blend(parent, n, rng, blend_frac=0.5):
    """Resample parent to length n and blend with random init."""
    if parent is not None and len(parent) > 0:
        p_resampled = resample_sequence(parent, n)
        a_rand = diverse_init(n, rng, mode=int(rng.integers(0, 8)))
        a = (1 - blend_frac) * p_resampled + blend_frac * a_rand
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s
            return a
    return diverse_init(n, rng, mode=0)

    deadline = start_time + budget_s - 30.0

    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent_arr = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")

        pass
