"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses aggressive LP fixed-point iteration with line search blending,
smoothed gradient descent with adaptive tau, multiplicative weights,
and multi-length exploration.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Returns (gradient of length n, exact_max of conv).
    """
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def lp_polish_iteration(a):
    """
    LP: minimize M s.t. conv(a,g) <= M for all k, g >= 0, sum(g) = 1.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    if n > 3000:
        return None, None

    # Build convolution matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue

    return None, None


def line_search_blend(a, g, max_iters=200):
    """
    Line search: find best alpha in [0,1] minimizing obj((1-alpha)*a + alpha*g).
    Uses golden section search on the 1D function.
    """
    def f(alpha):
        x = (1.0 - alpha) * a + alpha * g
        x = np.maximum(x, 0)
        s = float(np.sum(x))
        if s < 1e-10:
            return np.inf
        x = x / s
        return compute_obj(x)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # ~0.618
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    
    for _ in range(max_iters):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
        if hi - lo < 1e-6:
            break
    
    best_alpha = (lo + hi) / 2
    x = (1.0 - best_alpha) * a + best_alpha * g
    x = np.maximum(x, 0)
    s = float(np.sum(x))
    if s < 1e-10:
        return a.copy(), 0.0
    x = x / s
    return x, best_alpha

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def coordinate_descent(a, time_budget, verbose=False, max_rounds=5000):
    """

            break

            idxs = range(n)

            old_val = a[i]
            if old_val < 1e-15:

                candidates = [old_val * 0.5, old_val * 0.8, old_val, old_val * 1.2, old_val * 1.5, 
                              old_val + 0.005, old_val + 0.01]
            candidates = [max(0, c) for c in candidates]
            
            best_c = old_val
            best_obj_i = best_obj  # use current obj as baseline

                a[i] = c

                if s_curr < 1e-10:
                    a[i] = old_val

                a_norm = a / s_curr
                obj_c = compute_obj(a_norm)

                    best_c = c
            
            a[i] = best_c
            s_curr = float(np.sum(a))
            if s_curr > 1e-10:
                a = a / s_curr
                if compute_obj(a) < best_obj:
                    best_obj = compute_obj(a)
                    best_a = a.copy()

        if verbose and round_i % 100 == 0:
            print(f"    CD round {round_i}: obj={best_obj:.10f}, t={time.monotonic()-start:.0f}s")

def make_init(n, mode, rng=None):

    if rng is None:

        w = max(1, n // 8)

    elif mode == 6:  # Power law

        a = x ** 0.5
        return a / a.sum()
    elif mode == 7:  # Exponential center
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 4))
        return a / a.sum()
    elif mode == 8:  # Flat top with tapered edges
        x = np.linspace(0, 1, n)
        a = np.where((x > 0.2) & (x < 0.8), 1.0,
                     np.maximum(0.5 - np.abs(x - 0.5), 0.01))

    elif mode == 9:  # B-spline like (cubic)
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - (2*x - 1)**4, 0.001)
        return a / a.sum()
    else:
        return np.ones(n) / n

"""

Key improvements over previous: LP fixed-point with adaptive over-relaxation and

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Returns (gradient of length n, exact_max of conv).
    """
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    # Use safer exp to avoid overflow
    shifted = tau * (conv - exact_max)
    shifted = np.maximum(shifted, -700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def lp_solve(a):
    """
    LP: minimize M s.t. conv(a,g) <= M for all k, g >= 0, sum(g) = 1.
    Returns (g_optimal, M_optimal) or (None, None).

    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

def parabolic_line_search(a, g, coarse_steps=20):
    """
    Line search on [(1-t)*a + t*g] for t in [0, 1.2] with parabolic refinement.
    Returns (best_x, best_t).
    """

    ts = np.linspace(0, 1.2, coarse_steps)
    best_obj = np.inf
    best_t = 0.0

    for t in ts:
        x = (1.0 - t) * a + t * g
        x = np.maximum(x, 0)
        s = float(np.sum(x))
        if s < 1e-10:
            objs.append(np.inf)
            continue
        x = x / s
        o = compute_obj(x)
        objs.append(o)

            best_t = t

    # Parabolic refinement around best

            a = best_x
            if momentum < 0.5:
                momentum = min(0.5, momentum + 0.05)
            else:
                momentum = momentum * 0.95
            best_obj = best_obj_local
            best_a = a.copy()
            stall = 0
            if verbose:
                print(f"    LP {it:2d}: obj={best_obj:.10f}, t={best_t:.3f}, mom={momentum:.2f}")
        else:
            stall += 1
            # Decay momentum on stall
            momentum *= 0.8
            if verbose:
                print(f"    LP {it:2d}: stall={stall}")
            if stall % 5 == 0:
                # Perturb to escape local structure
                freqs = rng.integers(1, 7, size=4)
                phases = rng.uniform(0, 2 * np.pi, size=4)
                x = np.arange(n)
                pert = np.zeros(n)
                for f, p in zip(freqs, phases):
                    pert += 0.008 * np.sin(2 * np.pi * f * x / n + p)
                a = a + pert
                a = np.maximum(a, 0)
                a = a / a.sum()
                a_prev = a.copy()
                momentum = 0.0
            if stall > 25:
                break

    return best_a


def fast_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9):
    """
    Nesterov accelerated gradient descent with staircase tau schedule.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 20, 80, 300, 1000, 3000, 8000]

    stall_patience = 5000

    for tau in tau_levels:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        lr = 2.5 / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0
        max_iters = 5000
        it = 0
        best_it = 0

        while time.monotonic() - start < time_budget:
            if tau_iters >= max_iters:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it
            a = a_new
            it += 1
            tau_iters += 1
            lr = max(1e-9, lr * 0.99999)

            if it - best_it > stall_patience:
                a = best_a.copy()
                v = np.zeros(n)
                lr = 2.5 / np.sqrt(n)
                best_it = it

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    GD tau={tau:5d}: obj={best_obj:.10f}, iters={tau_iters}, t={elapsed:.0f}s")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def resample_fine(a, new_n):
    """Cubic-like resampling using scipy for smoother results."""
    from scipy.interpolate import interp1d
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    a_new = f(x_new)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: intensive LP fixed-point with adaptive over-relaxation and

    """
    start_time = time.monotonic()

    best_score = np.inf

    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        parent = None

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.10f}")

    n_current = len(best_seq)
    print(f"Starting: n={n_current}, obj={best_score:.10f}, time_limit={budget_s}s")

    # === Phase 1: Quick multi-n screening ===
    print(f"\n=== Phase 1: Multi-n Quick Screening ===")
    remaining = deadline - time.monotonic()

    n_screen = sorted(set([600, 700, 800, 900, 1000, 1100, 1200]))
    n_screen = [n for n in n_screen if abs(n - n_current) <= 500 or n in (700, 800, 900)]
    n_screen = n_screen[:6]

    for n_try in n_screen:
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break
        if abs(n_try - n_current) < 50:
            continue
        a_try = resample_fine(best_seq, n_try)
        alloc = min(remaining * 0.05, 20.0)
        if alloc < 8:
            break
        a_try = fast_gd(a_try, time_budget=alloc, tau_levels=[20, 200], verbose=False)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    print(f"  Best after screening: n={len(best_seq)}, obj={best_score:.10f}")

    # === Phase 2: Initial GD refinement ===
    print(f"\n=== Phase 2: Initial GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 150.0)
    if p2_budget > 15:
        a2 = fast_gd(best_seq, time_budget=p2_budget,
                     tau_levels=[5, 20, 80, 300, 1000], verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()

        else:
            print(f"  Phase2: obj={obj2:.10f}")

    # === Phase 3: Main LP Fixed-Point (primary workhorse) ===
    print(f"\n=== Phase 3: LP Fixed-Point (Main) ===")
    remaining = deadline - time.monotonic()
    if remaining > 100 and len(best_seq) <= 2500:
        # Run 3 rounds of LP with increasing patience

            remaining = deadline - time.monotonic()
            if remaining < 60:
