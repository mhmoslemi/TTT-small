"""
Soft-max gradient descent with LP polishing for step function optimization.
Uses FFT-based soft-max gradients for fast smooth optimization,
periodic LP solves for direction polishing, adaptive temperature annealing,
and systematic n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    p = a / s
    conv = compute_conv(p)
    return 2.0 * n * float(np.max(conv))


def soft_max_gradient_weighted(a, T):
    """
    Compute the gradient of the soft-max of conv(a,a) values weighted by temperature T.
    Uses FFT for efficient computation.
    Returns (grad, soft_max_value, weights).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    exp_vals = np.exp(T * (conv - max_c))
    sum_exp = np.sum(exp_vals)
    if sum_exp < 1e-300:
        weights = np.ones(len(conv)) / len(conv)
    else:
        weights = exp_vals / sum_exp
    soft_max_val = max_c + np.log(sum_exp) / T
    
    # term2[i] = sum_j weights[j] * a[i-j] = conv(weights, a)[i]
    size = 2
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(weights, size)
    term2 = irfft(fw * fa, size)[:n]
    
    # term1[i] = sum_j weights[i+j] * a[j] = conv(weights, flip(a))[i + n - 1]
    fa_flip = rfft(a[::-1], size)
    term1_full = irfft(fw * fa_flip, size)
    term1 = term1_full[n - 1 : n - 1 + n]
    
    grad = term1 + term2
    return grad, soft_max_val, weights


def hard_subgradient(a):
    """Compute the subgradient at the argmax of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    k_max = int(np.argmax(conv))
    grad = np.zeros(n)
    for i in range(n):
        j1 = k_max - i
        if 0 <= j1 < n:
            grad[i] += a[j1]
        j2 = i - k_max
        if 0 <= j2 < n:
            grad[i] += a[j2]
    return grad, k_max


def project_simplex(v, s=1.0):
    """Project v onto {x : sum(x) = s, x >= 0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    d = np.arange(1, n + 1)
    cond = u * d > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_direction(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns g."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def line_search_backtracking(a, direction, alpha0=0.1, c=1e-4):
    """Backtracking line search. Returns best t and objective."""
    n = len(a)
    a0 = a.copy()
    f0 = compute_obj(a0)
    # Gradient at a0 (for the descent direction check)
    grad, _ = hard_subgradient(a0)
    # Directional derivative (subgradient): grad @ direction
    dgrad = np.dot(grad, direction)
    # We want direction to be a descent direction: grad @ direction < 0
    # If direction = -grad, then dgrad = -||grad||^2 < 0 (good)
    
    t = alpha0
    best_t = 0.0
    best_f = f0
    
    for _ in range(30):
        a_try = a0 + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            t *= 0.5
            continue
        a_try = a_try / s
        f_try = compute_obj(a_try)
        if f_try < best_f:
            best_f = f_try
            best_t = t
        if f_try <= f0 + c * t * dgrad:
            return best_t, best_f
        t *= 0.5
    
    return best_t, best_f

def proj_simplex(v, s=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    d = np.arange(1, n + 1, dtype=np.float64)
    cond = u * d > (cs - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = int(np.max(np.where(cond)[0]))
    theta = (cs[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_mat(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    n = len(a)
    an = a / float(np.sum(a))
    A = build_conv_mat(an)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    con = [A @ g <= M, cp.sum(g) == 1]
    p = cp.Problem(cp.Minimize(M), con)
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            p.solve(solver=solver)
            if g.value is not None and p.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                sv = float(np.sum(gv))
                if sv > 1e-10:
                    return gv / sv
        except Exception:
            continue
    return None


def golden_line_search(a, g_lp, t_max=1.2):
    """Golden section search for optimal blend t in [0, t_max]."""
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    lo, hi = 0.0, t_max
    
    def f_at(t):
        a_t = (1 - t) * a + t * g_lp
        a_t = np.maximum(a_t, 0)
        s = float(np.sum(a_t))
        if s < 0.01:
            return np.inf
        return compute_obj(a_t / s)
    
    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc, fd = f_at(c), f_at(d)
    
    for _ in range(20):
        if fc < fd:
            hi, d, fd = d, c, fc
            c = lo + gr * (hi - lo)
            fc = f_at(c)
        else:
            lo, c, fc = c, d, fd
            d = lo + gr * (hi - lo)
            fd = f_at(d)
    
    t_best = c if fc < fd else d
    f_best = min(fc, fd)
    
    # Also check some standard values
    for t_try in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.1]:
        ft = f_at(t_try)
        if ft < f_best:
            f_best = ft
            t_best = t_try
    
    return t_best, f_best


def u_init(n, sharp=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharp * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n


def bimodal_init(n):
    x = np.linspace(0, 1, n)
    a = np.exp(-20 * (x - 0.1) ** 2) + np.exp(-20 * (x - 0.9) ** 2) + 0.01
    return a / a.sum()


def diverse_init(n, rng, mode):
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = (2 * x - 1) ** 6 + 0.02
    elif mode == 1:
        a = np.exp(-((np.arange(n) - n * 0.15) ** 2) / (2 * (n * 0.08) ** 2)) + \
            np.exp(-((np.arange(n) - n * 0.85) ** 2) / (2 * (n * 0.08) ** 2)) + 0.01
    elif mode == 2:
        a = 1.0 / (0.02 + 8.0 * (2 * x - 1) ** 2)
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        k = np.ones(7) / 7
        a = np.convolve(a, k, 'same')
    elif mode == 4:
        a = np.exp(3.0 * (2 * x - 1) ** 2)
    else:
        a = 1.0 / np.sqrt(1 - 0.9 * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n


def optimize_sequence(a, time_budget, use_lp=True, verbose=False):
    """Main optimization: soft-max gradient descent with periodic LP polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Parameters
    T = 15.0  # initial temperature for soft-max
    T_min = 3.0
    T_decay = 0.9995
    lr = 0.05
    momentum = np.zeros(n)
    beta = 0.7
    lp_interval = 20  # solve LP every 20 iterations
    it = 0
    stall = 0
    rng = np.random.default_rng(42 + n)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        
        # Compute soft-max gradient
        if T > T_min:
            grad, smv = soft_max_grad(a, T)
        else:
            grad, k_max = subgrad_hard(a)
        
        # Momentum update
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = proj_simplex(a_new)
        
        # Clip and renormalize
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        lr *= 0.9998
        T *= T_decay
        it += 1
        beta = min(0.9, 0.7 + 0.001 * it)
        
        # Periodic LP polishing
        if use_lp and it % lp_interval == 0 and remaining > 10:
            g_lp = solve_lp(a)
            if g_lp is not None:
                t_opt, f_opt = golden_line_search(a, g_lp)
                if f_opt < best_obj:
                    a_blend = (1 - t_opt) * a + t_opt * g_lp
                    a_blend = np.maximum(a_blend, 0)
                    s_blend = float(np.sum(a_blend))
                    if s_blend > 0.01:
                        a_blend = a_blend / s_blend
                        best_obj = f_opt
                        best_a = a_blend.copy()
                        a = a_blend
                        stall = 0
                        momentum = np.zeros(n)
                        lr = 0.03
                        beta = 0.7
        
        # Stall handling
        if stall > 100:
            # Perturb and restart
            noise = rng.normal(0, 0.02, n)
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            momentum = np.zeros(n)
            lr = 0.03
            beta = 0.7
            T = 10.0
            stall = 0
        
        if verbose and it % 200 == 0:
            print(f"    opt it{it}: obj={best_obj:.8f}, T={T:.2f}, lr={lr:.4f}, "
                  f"stall={stall}, elapsed={elapsed:.1f}s")
    
    return best_a
