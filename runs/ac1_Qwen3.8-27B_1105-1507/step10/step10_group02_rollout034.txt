"""
Optimizes step function coefficients using LP-based iterative refinement with
adaptive active set management, multiplicative updates, and exact coordinate descent polish.
Key innovations: (1) aggressive LP iteration with adaptive blending, (2) active-set
reduction for faster LPs, (3) multiplicative power iterations for escape, (4) symmetric
restriction to halve search space, (5) multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_rows(a, indices):
    """Build rows of the convolution matrix for specific indices k.
    Returns sparse-like structure: for row k, A[k,i] = a[k-i] if valid."""
    n = len(a)
    rows = []
    for k in indices:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            row[j] = a[k - j]
        rows.append(row)
    return np.array(rows)


def solve_lp_minpeak(a, active_indices=None, verbose=False):
    """Solve LP: min M s.t. (a*g)[k] <= M for k in active_indices (or all), sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    if active_indices is None:
        active_indices = list(range(2 * n - 1))
    
    A = build_conv_matrix_rows(a_norm, active_indices)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None


def solve_lp_topk(a, k=20, verbose=False):
    """Solve LP using only top-k convolution indices as constraints."""
    n = len(a)
    a_norm = a / np.sum(a)
    conv = compute_conv(a_norm)
    # Get top k indices
    top_idx = np.argsort(conv)[-k:]
    top_idx = sorted(top_idx.tolist())
    
    A = build_conv_matrix_rows(a_norm, top_idx)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    
    return None


def multiplicative_update(a, conv, top_k=10, eta=0.5):
    """Multiplicative update based on active constraints.
    a_new[i] = a[i] / (1 + eta * grad[i]) where grad comes from active peaks."""
    n = len(a)
    a = a.copy()
    
    # Find top convolution indices
    top_idx = np.argsort(conv)[-top_k:]
    
    # Compute gradient of max at active indices
    # d(conv[k])/da[i] = 2*a[k-i] (if valid) + a[i] (if k=2i, extra term)
    # Actually for conv = a*a, d(conv[k])/da[i] = a[k-i] + a[i] (if k-i = i, i.e., k=2i)
    # More precisely: conv[k] = sum_j a[j]*a[k-j]
    # d(conv[k])/da[i] = a[k-i] + a[i] (when k-i != i)
    #                   = 2*a[i] (when k-i = i)
    
    grad = np.zeros(n)
    for k in top_idx:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                if j == i:
                    grad[i] += 2.0 * a[i]
                else:
                    grad[i] += a[j]
    
    # Normalize gradient
    if np.max(np.abs(grad)) > 1e-15:
        grad = grad / np.max(np.abs(grad))
    
    # Multiplicative update
    a_new = a / (1.0 + eta * np.maximum(grad, 0))
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    
    return a_new


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """Exact coordinate descent with golden section search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 5 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v

            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)

            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks]

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            def f_1d(x, v=v, S_other=S_other, n=n, c_k_lin=c_k_lin,
                     a_ki_lin=a_ki_lin, c_2i=c_2i):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            scale = max(v, S / n, 1e-6)
            candidates = [0.0, v * 0.5, v, v * 1.5, v * 2.0,
                          S_other * 0.5, S_other]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(30):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def lp_iteration(a, time_budget=60.0, blend_factors=None, verbose=False, seed=42, top_k=None):

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    if blend_factors is None:
        blend_factors = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        it += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Solve LP
        if top_k is not None and top_k < 2 * n - 1:
            g_lp = solve_lp_topk(a, k=top_k)
        else:
            g_lp, _ = solve_lp_minpeak(a)

        if g_lp is None:
            break

        # Try blending
        improved = False
        for t in blend_factors:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_obj - 1e-12:

        else:

                stall = 0

        if verbose and it % 5 == 0:

def symmetric_cd(a, time_budget=100.0, verbose=False, seed=42):
    """Coordinate descent on the symmetric half of a palindromic sequence."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Symmetrize
    a_sym = 0.5 * (a + a[::-1])
    a_sym = a_sym / a_sym.sum()

    conv = compute_conv(a_sym)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a_sym.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    half_n = (n + 1) // 2  # we only optimize the first half

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        if sweep % 5 == 1:
            conv = compute_conv(a_sym)
            obj = 2.0 * n * float(np.max(conv))
            if obj < best_obj:
                best_obj = obj
                best_a = a_sym.copy()

        order = rng.permutation(half_n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            # When we change a[i], we also change a[n-1-i] (they're equal)
            # Let's handle this by treating the pair as a single variable
            v = a_sym[i]
            j = n - 1 - i

            if i == j:
                # Middle element, single coordinate
                # Standard 1D update
                S = float(np.sum(a_sym))
                S_other = S - v
                if S_other < 1e-12:
                    continue

                k_min = max(0, i)
                k_max = min(2 * n - 2, i + n - 1)
                ks = np.arange(k_min, k_max + 1)
                a_ki = a_sym[ks - i]
                c_k = conv[ks]
                mask = (ks != 2 * i)
                ks_lin = ks[mask]
                a_ki_lin = a_ki[mask]
                c_k_lin = c_k[mask]
                k_quad = 2 * i
                c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

                # For symmetric, changing a[i] also changes conv at positions
                # but since i == n-1-i, it's just one coordinate
                def f_mid(x):
                    if len(a_ki_lin) > 0:
                        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                        max_lin = float(np.max(lin_vals))
                    else:
                        max_lin = 0.0
                    quad_val = c_2i + x * x - v * v
                    max_val = max(max_lin, quad_val)
                    denom = (S_other + x) * (S_other + x)
                    if denom < 1e-30:
                        return 1e30
                    return 2.0 * n * max_val / denom

            else:
                # Pair (i, j) where i < j
                # Changing a[i] = a[j] = x changes:
                # - a[i] and a[j] both become x
                # - conv changes at all positions involving i or j
                # This is more complex; for simplicity, do sequential updates
                # Update i first, then j (they should end up equal)
                # Actually, let's just do standard CD on the full vector
                # but enforce symmetry after each step

                # For now, just update coordinate i (the symmetric partner will be
                # handled when we get to it in the permutation)
                S = float(np.sum(a_sym))
                S_other = S - v
                if S_other < 1e-12:
                    continue

                k_min = max(0, i)
                k_max = min(2 * n - 2, i + n - 1)
                ks = np.arange(k_min, k_max + 1)
                a_ki = a_sym[ks - i]
                c_k = conv[ks]
                mask = (ks != 2 * i)
                ks_lin = ks[mask]
                a_ki_lin = a_ki[mask]
                c_k_lin = c_k[mask]
                k_quad = 2 * i
                c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

                def f_pair(x):
                    if len(a_ki_lin) > 0:
                        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                        max_lin = float(np.max(lin_vals))
                    else:
                        max_lin = 0.0
                    quad_val = c_2i + x * x - v * v
                    max_val = max(max_lin, quad_val)
                    denom = (S_other + x) * (S_other + x)
                    if denom < 1e-30:
                        return 1e30
                    return 2.0 * n * max_val / denom

            # Unified 1D search (same for both cases since we're doing coordinate descent)
            S = float(np.sum(a_sym))
            S_other = S - v
            scale = max(v, S / half_n, 1e-6)
            candidates = [0.0, v * 0.5, v, v * 1.5, v * 2.0]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_mid if i == j else f_pair
            best_f_val = best_f(v)
            for x in candidates:
                fx = best_f(x)
                if fx < best_f_val:
                    best_f_val = fx
                    best_x = x

            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = best_f(c_pt)
            fd = best_f(d_pt)

            for _ in range(25):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = best_f(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = best_f(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            # Update: set both a[i] and a[n-1-i] to x_opt
            j_sym = n - 1 - i
            a_sym[i] = x_opt
            a_sym[j_sym] = x_opt

            # Update conv (approximate - will be corrected on next recompute)
            # For efficiency, just flag for recompute

        # Enforce symmetry and normalize
        a_sym = 0.5 * (a_sym + a_sym[::-1])
        a_sym = np.maximum(a_sym, 0)
        s_check = float(np.sum(a_sym))
        if s_check > 1e-10:
            a_sym = a_sym / s_check

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    SymCD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_rows(a_norm, indices):
    """Build rows of convolution matrix for specific indices k.
    Row k has A[k,j] = a_norm[k-j] if 0<=k-j<n, else 0."""
    n = len(a_norm)
    A = np.zeros((len(indices), n))
    for row_idx, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[row_idx, j] = a_norm[k - j]
    return A


def solve_lp_full(a):
    """Solve LP: min M s.t. (a*g)[k] <= M all k, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a_norm = a / s
    all_idx = list(range(2 * n - 1))
    A = build_conv_matrix_rows(a_norm, all_idx)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv
        except Exception:
            continue
    return None


def solve_lp_topk(a, k=30):
    """Solve LP using only top-k convolution indices as constraints. Returns g or None."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a_norm = a / s
    conv = compute_conv(a_norm)
    top_idx = np.argsort(conv)[-k:]
    top_idx = sorted(top_idx.tolist())
    A = build_conv_matrix_rows(a_norm, top_idx)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv
        except Exception:
            continue
    return None


def eval_1d_subproblem(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i):
    """Evaluate 1D subproblem for coordinate descent at coordinate i.
    x: new value, v: old value, S_other: sum of other coords,
    c_k_lin: conv values at linear k positions, a_ki_lin: a[k-i] at those positions,
    c_2i: conv value at quadratic position k=2i."""
    if len(a_ki_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom


def golden_section_1d(v, S, S_other, n, c_k_lin, a_ki_lin, c_2i):
    """Solve 1D subproblem via golden section search. Returns optimal x >= 0."""
    scale = max(v, S / n, 1e-6)
    candidates = [0.0, v * 0.3, v * 0.5, v, v * 1.5, v * 2.0, v * 3.0]
    candidates = sorted(set(max(0.0, float(c)) for c in candidates))
    best_x = v
    best_f = eval_1d_subproblem(v, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
    for x in candidates:
        fx = eval_1d_subproblem(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
        if fx < best_f:
            best_f = fx
            best_x = x
    x_lo = max(0.0, best_x - scale)
    x_hi = best_x + scale * 2.0
    if x_hi - x_lo < 1e-12:
        x_hi = x_lo + 1e-8
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = x_hi - gr * (x_hi - x_lo)
    d_pt = x_lo + gr * (x_hi - x_lo)
    fc = eval_1d_subproblem(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
    fd = eval_1d_subproblem(d_pt, v, S_other, n, c_k_pin, c_2i) if False else eval_1d_subproblem(d_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
    for _ in range(35):
        if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
            break
        if fc < fd:
            x_hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = x_hi - gr * (x_hi - x_lo)
            fc = eval_1d_subproblem(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
        else:
            x_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = x_lo + gr * (x_hi - x_lo)
            fd = eval_1d_subproblem(d_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
    return max(0.0, (x_lo + x_hi) / 2.0)


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """Exact coordinate descent with golden section search on 1D subproblems."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        if sweep % 3 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
        order = rng.permutation(n)
        S = float(np.sum(a))
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks]
            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]
            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0
            x_opt = golden_section_1d(v, S, S_other, n, c_k_lin, a_ki_lin, c_2i)
            if abs(x_opt - v) < 1e-15:
                continue
            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    return best_a


def lp_iteration(a, time_budget=60.0, blend_factors=None, verbose=False, seed=42,
                 use_full=False, top_k=30):
    """Iterative LP-based refinement with adaptive blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    if blend_factors is None:
        blend_factors = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    stall = 0
    while time.monotonic() - start < time_budget:
        it += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        if use_full:
            g_lp = solve_lp_full(a)
        else:
            g_lp = solve_lp_topk(a, k=top_k)
        if g_lp is None:
            break
        improved = False
        for t in blend_factors:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 2:
                pert = rng.normal(0, 0.015, n)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP iter {it}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    return best_a


def multiplicative_update(a, conv, top_k=15, eta=0.5):
    """Multiplicative update based on active convolution peaks."""
    n = len(a)
    a = a.copy()
    top_idx = np.argsort(conv)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if j == i:
                grad[i] += 2.0 * a[i]
            else:
                grad[i] += a[j]
    max_grad = float(np.max(np.abs(grad)))
    if max_grad > 1e-15:
        grad = grad / max_grad
    a_new = a / (1.0 + eta * np.maximum(grad, 0))
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def mu_escape(a, time_budget=60.0, verbose=False):
    """Multiplicative update escape: try various eta, iterate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    etas = [0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]
    for eta in etas:
        if time.monotonic() - start >= time_budget:
            break
        a_curr = best_a.copy()
        for step in range(80):
            if time.monotonic() - start >= time_budget:
                break
            conv_curr = compute_conv(a_curr)
            a_curr = multiplicative_update(a_curr, conv_curr, top_k=15, eta=eta)
            obj_curr = 2.0 * n * float(np.max(compute_conv(a_curr)))
            if obj_curr < best_obj:
                best_obj = obj_curr
                best_a = a_curr.copy()
        if verbose:
            print(f"    MU eta={eta}: obj={best_obj:.8f}")
    return best_a


def symmetric_cd(a, time_budget=100.0, verbose=False, seed=42):
    """Coordinate descent enforcing palindromic symmetry."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    a = 0.5 * (a + a[::-1])
    a = a / a.sum()
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    half_n = (n + 1) // 2
    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        if sweep % 3 == 1:
            conv = compute_conv(a)
            obj = 2.0 * n * float(np.max(conv))
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        order = rng.permutation(half_n)
        S = float(np.sum(a))
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            v = a[i]
            j = n - 1 - i
            if i == j:
                S_other = S - v
            else:
                S_other = S - 2.0 * v
            if S_other < 1e-12:
                continue
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks]
            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]
            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0
            if i == j:
                x_opt = golden_section_1d(v, S, S_other, n, c_k_lin, a_ki_lin, c_2i)
                if abs(x_opt - v) < 1e-15:
                    continue
                delta = x_opt - v
                a[i] = x_opt
                if 0 <= k_quad <= 2 * n - 2:
                    conv[k_quad] += x_opt * x_opt - v * v
                if len(ks_lin) > 0:
                    conv[ks_lin] += 2.0 * delta * a_ki_lin
            else:
                # Update both a[i] and a[j]: approximate by treating as single coord with 2x weight
                # For simplicity, update i then j sequentially
                x_opt_i = golden_section_1d(v, S, S_other, n, c_k_lin, a_ki_lin, c_2i)
                if abs(x_opt_i - v) > 1e-14:
                    a[i] = x_opt_i
                    a[j] = x_opt_i
                    # Recompute conv for accuracy (approximation)
                    conv = compute_conv(a)
                    S = float(np.sum(a))
        # Enforce symmetry
        a = 0.5 * (a + a[::-1])
        a = np.maximum(a, 0)
        s_chk = float(np.sum(a))
        if s_chk > 1e-10:
            a = a / s_chk
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    SymCD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x ** 2 + 1e-6)
        a = a / a.sum()
    elif mode == 3:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

            print(f"  SymCD: obj={obj_sym:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print("\n=== Phase 5: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    n_options = sorted(set([
        max(200, n_current - 150), n_current - 50, n_current + 50,
        n_current + 150, n_current + 300, 600, 900, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_current) > 30]
    n_options = n_options[:4]
    print(f"  Trying n: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()
        alloc = min(remaining * 0.35, 80.0)
        if alloc > 15:
            # Quick LP + CD
            a_try = lp_iteration(a_try, time_budget=alloc * 0.5,
                                 blend_factors=[0.2, 0.4, 0.6, 1.0],
                                 verbose=False, seed=n_try, top_k=25)
            remaining_inner = deadline - time.monotonic()
            if remaining_inner > 15:
                a_try = coordinate_descent(a_try, time_budget=min(remaining_inner * 0.4, 30.0),
                                           verbose=False, seed=n_try + 1)
        obj_try = 2.0 * a_try.size * float(np.max(compute_conv(a_try / a_try.sum())))
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print("\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_curr = len(best_seq)
    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        mode = r + 1
        a_r = diverse_init(n_curr, rng, mode=mode)
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        alloc = min(remaining * 0.3, 60.0)
        if alloc > 15:
            a_r = coordinate_descent(a_r, time_budget=alloc,
                                     verbose=False, seed=5000 + r * 100)
            obj_r = 2.0 * a_r.size * float(np.max(compute_conv(a_r / a_r.sum())))
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final Refinement Loop ===
    print("\n=== Phase 7: Final Refinement ===")
    for rnd in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        # LP with top-k
        alloc_lp = min(remaining * 0.2, 40.0)
        if alloc_lp > 10:
            a_lp = lp_iteration(best_seq, time_budget=alloc_lp,
                                blend_factors=[0.05, 0.1, 0.15, 0.2, 0.3],
                                verbose=False, seed=10000 + rnd, top_k=35)
            obj_lp = 2.0 * a_lp.size * float(np.max(compute_conv(a_lp / a_lp.sum())))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP {rnd}: obj={best_score:.8f} **")
        # CD
        remaining = deadline - time.monotonic()
        alloc_cd = min(remaining * 0.2, 40.0)
        if alloc_cd > 10:
            a_cd = coordinate_descent(best_seq, time_budget=alloc_cd,
                                      verbose=False, seed=20000 + rnd)
            obj_cd = 2.0 * a_cd.size * float(np.max(compute_conv(a_cd / a_cd.sum())))
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final CD {rnd}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
