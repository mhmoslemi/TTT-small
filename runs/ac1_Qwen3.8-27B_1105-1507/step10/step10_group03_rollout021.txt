import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Inputs: a (assumed normalized to sum=1), tau (smoothing parameter).
    Returns: gradient vector of length n.
    """
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    # Gradient via FFT cross-correlation
    N = 1
    while N < 4 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g

def make_ramp_init(n, beta, rng=None):
    """Exponential ramp initialization."""
    if beta == 0:
        return np.ones(n) / n
    x = np.linspace(0, 1, n)
    a = np.exp(beta * x)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_power_init(n, beta):
    """Power law initialization."""
    j = np.arange(1, n + 1, dtype=np.float64)
    a = j ** beta
    a = np.maximum(a, 1e-10)
    return a / a.sum()
