"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative gradient descent with adaptive log-sum-exp smoothing,
direct peak-targeting subgradient descent, and multi-n exploration.
Key differences from previous: multiplicative updates, adaptive tau schedule,

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """Compute gradient of smoothed max of conv(a,a) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -700, 0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n], conv, p


def multiplicative_gd(a, time_budget, tau_init=5.0, tau_end=5000.0, lr_init=0.5, verbose=True):
    """
    Multiplicative (mirror descent) gradient descent with adaptive tau.
    Update: a_i <- a_i * exp(-lr * grad_i), then normalize.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        
        # Adaptive tau: exponential schedule
        tau = tau_init * (tau_end / tau_init) ** progress
        
        grad, conv, p = smoothed_grad(a, tau)
        
        # Adaptive learning rate based on gradient magnitude
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm > 1e-15:
            lr = lr_init / (grad_norm + 1e-10)
            lr = min(lr, 2.0)  # Cap to prevent explosion
        else:
            lr = lr_init
        
        # Multiplicative update
        log_update = -lr * grad
        log_update = np.clip(log_update, -10, 10)  # Safety clip
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        
        if verbose and it % 500 == 0:
            print(f"      MG it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.4f}, time={elapsed:.0f}s")
        
        # Restart if stuck
        if it - best_it > 5000:
            a = best_a.copy()
            best_it = it
            # Reduce lr
            lr_init *= 0.7
    
    if verbose:
        print(f"      MG done: obj={best_obj:.8f}, iters={it}")
    return best_a

def nesterov_gd(a, time_budget, tau_init=20.0, tau_end=3000.0, beta=0.9, verbose=True):
    """
    Nesterov accelerated gradient descent on smoothed objective.
    Uses additive updates with proper simplex projection.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init * (tau_end / tau_init) ** progress
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        grad, conv, p = smoothed_grad(y, tau)
        
        # Center gradient for simplex constraint
        grad_centered = grad - np.mean(grad)
        
        # Adaptive LR
        grad_norm = float(np.linalg.norm(grad_centered))
        if grad_norm > 1e-15:
            adapt_lr = 1.0 / (grad_norm + 1e-10)
            lr = 0.7 * lr + 0.3 * adapt_lr
            lr = min(max(lr, 0.1 * base_lr), 3.0 * base_lr)
        
        v = beta * v - lr * grad_centered
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)
        
        if verbose and it % 1000 == 0:
            print(f"      NG it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.5f}, time={elapsed:.0f}s")

            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it
    
    if verbose:
        print(f"      NG done: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.05):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, rng, strength=0.02):
    """Add small random perturbation."""
    n = len(a)
    noise = rng.standard_normal(n) * strength
    a_new = a + noise
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of Nesterov accelerated GD with adaptive log-sum-exp smoothing,
multiplicative (mirror) descent, direct subgradient peak targeting, and aggressive
multi-n exploration with diverse restarts.
Key improvements: adaptive tau responds to plateau detection, multiplicative updates
for better non-negativity handling, direct peak targeting, and broader n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """Compute gradient of smoothed max of conv(a,a) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -700, 0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n], conv, p


def nesterov_gd(a, time_budget, tau_init=10.0, tau_end=3000.0, beta=0.9, verbose=True):
    """
    Nesterov accelerated gradient descent on smoothed objective with adaptive tau.
    Uses plateau detection to adjust tau and learning rate.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr

    # Adaptive tau state
    current_tau = tau_init
    plateau_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Adaptive tau: exponential base schedule modulated by plateau
        base_tau = tau_init * (tau_end / tau_init) ** (progress ** 0.8)
        if plateau_count > 0:
            base_tau = base_tau * (1.0 + 0.5 * min(plateau_count, 5))
        current_tau = min(current_tau + 0.3 * (base_tau - current_tau) + 0.7 * base_tau, 10000.0)
        current_tau = max(current_tau, 1.0)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
        else:
            y = y / s_y

        grad, conv, p = smoothed_grad(y, current_tau)

        grad_centered = grad - np.mean(grad)

        grad_norm = float(np.linalg.norm(grad_centered))
        if grad_norm > 1e-15:
            adapt_lr = 1.5 / (grad_norm + 1e-10)

        v = beta * v - lr * grad_centered
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))

        it += 1

                v = np.zeros(n)

def multiplicative_gd(a, time_budget, tau_init=5.0, tau_end=5000.0, lr_init=0.4, verbose=True):
    """
    Multiplicative (mirror/sinkhorn) gradient descent with adaptive tau.
    Update: a_i <- a_i * exp(-lr * grad_i), then normalize.
    Never produces negative values.
    """
    n = len(a)
    a = np.maximum(a, 1e-12).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    current_tau = tau_init
    plateau_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Adaptive tau
        base_tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        if plateau_count > 3:
            base_tau *= (1.0 + 0.3 * min(plateau_count, 8))
        current_tau = min(max(current_tau + 0.5 * (base_tau - current_tau), 1.0), 10000.0)

        grad, conv, p = smoothed_grad(a, current_tau)

        # Adaptive learning rate
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm > 1e-15:
            lr = lr_init / (grad_norm * np.sqrt(n) / 2.0 + 1e-10)
            lr = min(lr, 3.0)
        else:
            lr = lr_init

        # Multiplicative update
        log_update = -lr * grad
        log_update = np.clip(log_update, -15, 15)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            plateau_count = 0
        else:
            plateau_count += 1

        a = a_new
        it += 1

        if verbose and it % 1000 == 0:
            print(f"      MG it{it}: obj={best_obj:.8f}, tau={current_tau:.0f}, lr={lr:.4f}, "
                  f"plateau={plateau_count}, time={elapsed:.0f}s")

        # Reset if stuck
        if it - best_it > 5000:
            a = best_a.copy()
            best_it = it
            lr_init *= 0.8
            plateau_count = 0

    if verbose:
        print(f"      MG done: obj={best_obj:.8f}, iters={it}")
    return best_a


def subgradient_peak_descent(a, time_budget, verbose=True):
    """
    Direct subgradient descent targeting the peak (and near-peaks) of conv(a,a).
    Uses an average of subgradients over near-max lags for stability.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    lr = 0.3
    peak_tolerance = 0.98  # Consider lags within 98% of max

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Find near-peaks with soft weighting
        threshold = max_conv * peak_tolerance
        weights = np.maximum(conv - threshold, 0)
        total_w = float(np.sum(weights))
        if total_w < 1e-15:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = weights / total_w

        # Compute gradient via FFT
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        grad = 2.0 * ifft(P * np.conj(A)).real[:n]

        # Project onto simplex tangent (center)
        grad_centered = grad - np.mean(grad)

        # Adaptive lr
        grad_norm = float(np.linalg.norm(grad_centered))
        if grad_norm > 1e-15:
            target_lr = 1.0 / (grad_norm * np.sqrt(n) / 3.0 + 1e-10)
            lr = 0.8 * lr + 0.2 * target_lr
            lr = min(lr, 2.0)
            lr = max(lr, 0.01)

        # Gradient step
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = a.copy()
            lr *= 0.5
        else:
            a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            a = a_new
            lr = min(lr * 1.05, 3.0)
        else:
            lr *= 0.85

        a = np.maximum(a, 0)
        a = a / a.sum()
        it += 1

        if verbose and it % 200 == 0:
            n_peaks = int(np.sum(conv >= threshold))
            print(f"      SG it{it}: obj={best_obj:.8f}, n_peaks={n_peaks}, lr={lr:.4f}, time={elapsed:.0f}s")

        if it - best_it > 3000:
            a = best_a.copy()
            best_it = it
            lr = 0.1

    if verbose:
        print(f"      SG done: obj={best_obj:.8f}, iters={it}")
    return best_a


def coordinate_descent(a, time_budget, verbose=True):
    """
    Coordinate-wise optimization: for each coordinate i, find the best value
    of a_i (holding others fixed) that minimizes the objective.
    Cycles through coordinates multiple times.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    passes = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 2.0:
            break

        # One pass through all coordinates
        for i in range(n):
            # Save current value
            save = a[i]
            others_sum = float(np.sum(a) - save)

            # The conv values that depend on a_i:
            # conv_k = sum_j a_j * a_{k-j}
            # a_i appears in conv_k when j=i (gives a_{k-i}) or j=k-i (gives a_i * a_{k-i})
            # Specifically: conv_k depends on a_i^2 when k=2i, and on a_i*a_{k-i} otherwise

            # To find optimal a_i, we need to minimize max over k of conv_k
            # where conv_k = (terms not involving a_i) + a_i * (terms linear in a_i)
            # Let's compute: for each k, conv_k = c_k + d_k * a_i + e_k * a_i^2
            # where e_k = 1 if k = 2i, else 0
            # d_k = a_{k-i} if 0 <= k-i < n and k-i != i, else (2*a_i if k=2i)

            # Actually, let's just do a 1D search: try a few values of a_i
            # The objective as a function of a_i (with others fixed, then renormalize)
            # is complex, so let's do a simpleshrinking search

            # Try: a_i = 0, a_i = current, a_i = 2*current, a_i = current/2
            candidates = [0.0, save * 0.25, save * 0.5, save * 0.75, save, save * 1.25, save * 1.5, save * 2.0]
            best_val = save
            best_cand_obj = compute_obj(a)

            for cv in candidates:
                a[i] = cv
                s = float(np.sum(a))
                if s < 1e-10:
                    a[i] = save
                    continue
                a_norm = a / s
                obj_c = compute_obj(a_norm)
                if obj_c < best_cand_obj - 1e-15:
                    best_cand_obj = obj_c
                    best_val = cv

            a[i] = best_val

        passes += 1
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        a = a / s
        obj_now = compute_obj(a)
        if obj_now < best_obj - 1e-14:
            best_obj = obj_now
            best_a = a.copy()

        if verbose and passes % 50 == 0:
            print(f"      CD pass {passes}: obj={best_obj:.8f}, time={elapsed:.0f}s")

    if verbose:
        print(f"      CD done: obj={best_obj:.8f}, passes={passes}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.05):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, rng, strength=0.02):
    """Add small random perturbation."""
    n = len(a)
    noise = rng.standard_normal(n) * strength
    a_new = a + noise
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def windowed_init(n, window_type='hann'):
    """Create a windowed initialization."""
    if window_type == 'hann':
        a = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    elif window_type == 'gaussian':
        x = np.linspace(-3, 3, n)
        a = np.exp(-x**2 / 2)
    elif window_type == 'raised_cos':
        a = np.cos(np.pi * np.arange(n) / (2 * (n - 1)))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    return a / a.sum()


def optimize_block(a, time_budget, verbose=False):
    """Run a quick optimization cycle: NG + SG."""
    remaining = time_budget
    alloc_ng = remaining * 0.6
    alloc_sg = remaining * 0.4

    if alloc_ng > 5:
        a = nesterov_gd(a, time_budget=alloc_ng, tau_init=20.0, tau_end=1500.0,
                        beta=0.9, verbose=False)
    remaining = time_budget - (time.monotonic() - time.monotonic())  # Not accurate, use remaining time
    # Just do SG with remaining budget estimate
    a = subgradient_peak_descent(a, time_budget=alloc_sg, verbose=False)
    return a
