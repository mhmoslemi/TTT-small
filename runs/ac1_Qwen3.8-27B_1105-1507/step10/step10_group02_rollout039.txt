"""
Hybrid optimizer combining Nesterov-accelerated smoothed gradient descent,
spectral equalization, and exact coordinate descent to minimize
2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
Uses FFT-based gradient computation with log-sum-exp smoothing of the peak,
plus periodic coordinate descent sweeps for local refinement.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def _fft_size(n):
    """Next power of 2 >= n."""
    s = 1
    while s < n:
        s <<= 1
    return s


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = _fft_size(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj_raw(a):
    """Compute the raw objective 2n*max_b/S^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def smooth_max_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum(exp(tau*conv[k]))) w.r.t. a.
    Uses FFT for efficiency.
    Returns (gradient array of length n, conv array, smooth_max value).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    # Numerically stable softmax
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_vals / sum_exp

    # Cross-correlation: (w *round* a)[j] = sum_k w[k] * a[k-j]
    # Use FFT with sufficient padding
    N = _fft_size(3 * n)
    W = fft(w, N)
    A = fft(a, N)
    cross_corr = ifft(np.conj(W) * A).real[:n]

    # Box sum: for each j, sum_{k=j}^{min(j+n-1, 2n-2)} w[k]
    # Use prefix sums
    w_ext = np.zeros(3 * n)
    w_ext[:len(w)] = w
    prefix = np.cumsum(w_ext)
    box_sums = np.zeros(n)
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        if k_end >= 0:
            box_sums[j] = prefix[k_end + 1] - prefix[k_start]

    grad = cross_corr + a * box_sums
    smooth_max_val = (np.log(sum_exp) + tau * max_conv) / tau if sum_exp > 0 else max_conv
    return grad, conv, smooth_max_val


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=50.0, tau_end=500.0,
                               verbose=False, seed=42):
    """Nesterov accelerated projected gradient descent with adaptive tau."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    beta = 0.9
    v = np.zeros(n)
    lr_base = 2.0 / np.sqrt(n)
    lr = lr_base
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 0.5

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient
        grad, conv, _ = smooth_max_gradient(y, tau)

        # Learning rate adaptation
        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            target_lr = lr_base / (1.0 + g_norm * 0.5)
            lr = 0.9 * lr + 0.1 * target_lr
            lr = min(lr, lr_base * 2)
            lr = max(lr, lr_base * 0.01)

        # Update
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        # Evaluate
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 2000 == 0:
            print(f"      NSD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}")

    return best_a


def spectral_equalize(a, time_budget=30.0, alpha=0.3, verbose=False):
    """
    Iteratively flatten the power spectrum to reduce autocorrelation peak.
    a[j] = |IFFT(target_mag * exp(i*phase))|, then normalize.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    N = _fft_size(2 * n - 1)

    while time.monotonic() - start < time_budget:
        A = fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: compress the dynamic range of the spectrum
        # Raise the floor and lower the ceiling
        mag_floor = np.min(mag[mag > 0]) if np.any(mag > 0) else 0
        mag_max = np.max(mag)
        if mag_max < 1e-15:
            break

        # Soft flattening
        mag_target = mag ** (1.0 - alpha)
        mag_target = mag_target / np.max(mag_target) * mag_max

        # Reconstruct
        A_new = mag_target * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller alpha
            alpha = max(0.05, alpha * 0.7)
            if alpha < 0.06:
                break

    return best_a


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """
    Exact coordinate descent with golden section search on 1D subproblems.
    Optimizes each coordinate exactly given all others fixed.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 5 == 1:
            conv = compute_conv(a)
            s_check = float(np.sum(a))
            if s_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (s_check * s_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        order = rng.permutation(n)
        S = float(np.sum(a))
        improved = 0

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)

            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k_lin_arr = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # 1D objective: f(x) = 2n * max(max_lin(x), quad(x)) / (S_other + x)^2
            def f_1d(x, v=v, S_other=S_other, n=n, c_k_lin=c_k_lin,
                     a_ki_lin=a_ki_lin, c_2i=c_2i, ks_lin=ks_lin):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Quick scan
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, max(0, v - scale), v * 0.25, v * 0.5, v,
                          v * 1.5, v * 2.0, v * 3.0, min(1.0, v * 5)]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refine
            x_lo = max(0.0, best_x - scale * 1.5)
            x_hi = best_x + scale * 2.5
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(40):
                if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt
            improved += 1

            # Update conv
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

            s_new = float(np.sum(a))
            if s_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (s_new * s_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, improved={improved}, t={elapsed:.1f}s")

        if improved == 0:
            # Try perturbation to escape
            if sweep % 10 == 0:
                pert_idx = rng.integers(0, n, size=min(10, n))
                a[pert_idx] *= np.exp(rng.normal(0, 0.1, size=min(10, n)))
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = compute_conv(a)

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps
        c1, c2 = n // 4, 3 * n // 4
        w = n // 6
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x ** 2 + 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Fourier-based
        M = 10
        coeffs = rng.normal(0, 0.3, M)
        phases = rng.uniform(0, 2 * np.pi, M)
        a = np.ones(n)
        for m in range(1, M + 1):
            a += coeffs[m - 1] * np.cos(2 * np.pi * m * np.arange(n) / n + phases[m - 1])
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Edge concentration
        a = np.abs(np.sin(np.pi * np.arange(n) / n)) ** rng.uniform(0.5, 3.0)
        a = a / a.sum()
    elif mode == 6:
        # Random Fourier
        M = 20
        coeffs = rng.normal(0, 0.5, M)
        phases = rng.uniform(0, 2 * np.pi, M)
        a = np.ones(n)
        for m in range(1, M + 1):
            a += coeffs[m - 1] * np.cos(2 * np.pi * m * np.arange(n) / n + phases[m - 1])
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if n == 2:
        x_new = np.linspace(0, 1, new_n)
        a_new = a[0] * (1 - x_new) + a[1] * x_new
        return np.maximum(a_new, 0)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def block_coordinate_descent(a, time_budget=60.0, block_size=3, seed=42):

        return a

    while time.monotonic() - start < time_budget:
        # Pick a random block
        i0 = rng.integers(0, n - block_size + 1)
        indices = list(range(i0, i0 + block_size))

        # For each coordinate in the block, do a 1D optimization
        for i in indices:
            if time.monotonic() - start >= time_budget:
                break

            S_other = float(np.sum(a)) - v

            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]
            k_quad = 2 * i

                     a_ki_lin=a_ki_lin, c_2i=c_2i):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))

                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Scan + golden section
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, max(0, v * 0.3), v * 0.7, v, v * 1.4, v * 2.0, v * 3.0]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))
            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)
            for _ in range(30):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            a[i] = x_opt

        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s
            obj = 2.0 * n * float(np.max(compute_conv(a)))
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a

        if s < 1e-10:
            return
        a = a / s

        for method_name, budget_frac in methods:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * budget_frac, 200.0)
            if alloc < 10:
                break

            t0 = time.monotonic()
            if method_name == "cd":
                a = coordinate_descent(a, time_budget=alloc, verbose=False,
                                       seed=seed + seed_offset)
            elif method_name == "nsd":
                a = nesterov_smoothed_descent(a, time_budget=alloc,
                                              tau_init=30.0, tau_end=300.0,
                                              verbose=False, seed=seed + seed_offset)
            elif method_name == "spectral":
                a = spectral_equalize(a, time_budget=alloc, alpha=0.4, verbose=False)
            elif method_name == "block_cd":
                a = block_coordinate_descent(a, time_budget=alloc,
                                             block_size=3, seed=seed + seed_offset)
            elif method_name == "cd_fine":
                a = coordinate_descent(a, time_budget=alloc, verbose=False,
                                       seed=seed + seed_offset + 7777)
            else:
                continue

            elapsed = time.monotonic() - t0
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
                obj = compute_obj_normalized(a)
                if obj < best_score:
                    best_score = obj
                    best_seq = a.copy()
                    print(f"  {label}/{method_name}: obj={best_score:.8f} ** ({elapsed:.1f}s)")
                else:
                    print(f"  {label}/{method_name}: obj={obj:.8f} ({elapsed:.1f}s)")

        return a

    # === Phase 1: Coordinate Descent on parent (main workhorse) ===
    print("\n=== Phase 1: CD on parent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 400.0)
    if p1_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget,
                                  verbose=True, seed=seed)
        obj_cd = compute_obj_normalized(a_cd / a_cd.sum())
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD parent: obj={best_score:.8f} **")
        else:
            print(f"  CD parent: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: Nesterov Smoothed GD ===
    print("\n=== Phase 2: Nesterov Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 200.0)
    if p2_budget > 30:
        a_nsd = nesterov_smoothed_descent(best_seq, time_budget=p2_budget,
                                          tau_init=40.0, tau_end=400.0,
                                          verbose=True, seed=seed + 1)
        obj_nsd = compute_obj_normalized(a_nsd / a_nsd.sum())
        if obj_nsd < best_score:
            best_score = obj_nsd
            best_seq = a_nsd.copy()
            print(f"  NSD: obj={best_score:.8f} **")
        else:
            print(f"  NSD: obj={obj_nsd:.8f}")

    # === Phase 3: Spectral Equalization ===
    print("\n=== Phase 3: Spectral Equalization ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 80.0)
    if p3_budget > 15:
        a_spec = spectral_equalize(best_seq, time_budget=p3_budget,
                                   alpha=0.3, verbose=False)
        obj_spec = compute_obj_normalized(a_spec / a_spec.sum())
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Spectral: obj={best_score:.8f} **")
        else:
            print(f"  Spectral: obj={obj_spec:.8f}")

    # === Phase 4: CD refinement ===
    print("\n=== Phase 4: CD Refinement ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.2, 200.0)
    if p4_budget > 20:
        a_cd2 = coordinate_descent(best_seq, time_budget=p4_budget,
                                   verbose=True, seed=seed + 100)
        obj_cd2 = compute_obj_normalized(a_cd2 / a_cd2.sum())
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  CD2: obj={best_score:.8f} **")
        else:
            print(f"  CD2: obj={obj_cd2:.8f}")

    # === Phase 5: Multi-n exploration ===
    print("\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = []
    for n_try in [n_current - 200, n_current + 200, n_current + 400,
                  600, 1000, 1200, 1500, 2000]:
        if 400 <= n_try <= 2500 and n_try != n_current:
            n_options.append(n_try)
    n_options = sorted(set(n_options))[:5]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        methods = [("cd", 0.5), ("nsd", 0.3), ("cd_fine", 0.3)]
        try_optimize(a_try, n_try, methods, f"n={n_try}", seed_offset=n_try)
        # After resampling and optimizing, also try CD on the result
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_try2 = coordinate_descent(a_try, time_budget=min(remaining * 0.3, 60.0),
                                        verbose=False, seed=n_try + 500)
            obj_try2 = compute_obj_normalized(a_try2 / a_try2.sum())
            if obj_try2 < best_score:
                best_score = obj_try2
                best_seq = a_try2.copy()
                print(f"  n={n_try} refined: obj={best_score:.8f} **")

    # === Phase 6: Diverse restarts ===
    print("\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = r + 1
        a_r = diverse_init(n_cur, rng, mode=mode)
        # Blend with current best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        methods = [("cd", 0.5), ("nsd", 0.3), ("cd_fine", 0.3)]
        try_optimize(a_r, n_cur, methods, f"restart{r}", seed_offset=5000 + r * 100)

    # === Phase 7: Block CD on best ===
    print("\n=== Phase 7: Block CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_bc = block_coordinate_descent(best_seq,
                                        time_budget=min(remaining * 0.3, 80.0),
                                        block_size=3, seed=seed + 9000)
        obj_bc = compute_obj_normalized(a_bc / a_bc.sum())
        if obj_bc < best_score:
            best_score = obj_bc
            best_seq = a_bc.copy()
            print(f"  Block CD: obj={best_score:.8f} **")
        else:
            print(f"  Block CD: obj={obj_bc:.8f}")

    # === Phase 8: Final CD polish ===
    print("\n=== Phase 8: Final CD Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.7, 150.0)
        a_final = coordinate_descent(best_seq, time_budget=alloc,
                                     verbose=False, seed=seed + 9999)
        obj_final = compute_obj_normalized(a_final / a_final.sum())
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
