```python
import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = a[:end - j]
    return A

def solve_lp_fast(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0 using scipy HiGHS."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    # Variables: g[0],...,g[n-1], M
    # Minimize: [0,...,0,1] . [g,M]
    # Constraints: A @ g - M <= 0 (i.e., A @ g + (-1)*M <= 0)
    #              -sum(g) <= 0 and sum(g) <= 0 (for equality)
    # Bounds: g >= 0, M unbounded below (or M >= 0)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # g >= 0, M >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            g = g / s
        M = float(res.x[n])
        return g, M
    return None, None
```

```python
            # So: conv[k](x) = conv[k] - (terms with old_val) + (terms with x)
            # For k != 2i (and k-i in [0,n)):
            #   conv[k](x) = conv[k] - 2*old_val*a[k-i] + 2*x*a[k-i]
            #   = (conv[k] - 2*old_val*a[k-i]) + 2*a[k-i]*x

            #   conv[2i](x) = conv[2i] - old_val^2 + x^2

            #   conv[k](x) = conv[k] (unchanged)

            # k_start = i, k_end = min(2n-2, i+n-1)
            # For k in [k_start, k_end]:

            #   Else: linear term with slope 2*a[k-i]

            # max( max_{k in [k_start,k_end], k!=2i} (base[k] + 2*a[k-i]*x), base[2i] + x^2 )

            # 2. x where the "dominant" linear function crosses zero (if applicable)
            # 3. x = old_val (current)
            # 4. x that equalizes the top few affected positions with unaff_max

            # the value that makes the top affected position equal to unaff_max.
            
            candidates = [0.0]
            if old_val > 0:
                candidates.extend([old_val * 0.1, old_val * 0.5, old_val * 1.5, old_val * 3.0])
            
            # Add a candidate that makes max_affected(x) = unaff_max (if possible)
            # max_affected(x) >= max of the linear terms
            # The max linear term at x is: max_k (base[k] + slope[k]*x)
            # At x=0: max_k base[k]
            # The "break-even" x where max_affected = unaff_max...
            # This is complex, let's just use the simple candidates.
            
            best_x = old_val
            best_f = best_obj  # current objective
            
            for x in candidates:
                new_s = S + x
                if new_s < 1e-10:
                    continue
                # Compute max over affected positions
                # For speed, evaluate the max of (base[k] + slope[k]*x) for affected k
                # And compare with unaff_max
                
                # Affected: k in [k_start, k_end]
                # slope[k] = 2*a[k-i] for k != 2i
                # base[k] = conv[k] - 2*old_val*a[k-i] for k != 2i
                # For k=2i: base2i = conv[2i] - old_val^2, value = base2i + x^2
                
                ks = np.arange(k_start, k_end + 1)
                ki = ks - i  # = a[k-i] index
                
                # Mask for k != 2i
                mask = ks != 2 * i
                aff_max = unaff_max  # start with unaffected max
                
                if np.any(mask):
                    base_lin = conv[ks[mask]] - 2 * old_val * a[ki[mask]]
                    slope = 2 * a[ki[mask]]
                    vals = base_lin + slope * x
                    aff_max = max(aff_max, float(np.max(vals)))
                
                if 2 * i <= k_end and 2 * i >= k_start:
                    base2i = conv[2 * i] - old_val ** 2
                    aff_max = max(aff_max, base2i + x * x)
                
                obj = 2.0 * n * aff_max / (new_s ** 2)
                if obj < best_f:
                    best_f = obj
                    best_x = x
            
            if best_x != old_val and best_f < best_obj - 1e-12:
                # Update
                a[i] = best_x
                s = S + best_x
                # Recompute conv (expensive but necessary for correctness)
                a_norm = a / s
                conv = compute_conv(a_norm)
                # Update prefix/suffix max
                prefix_max[0] = conv[0]
                for k in range(1, 2 * n - 1):
                    prefix_max[k] = max(prefix_max[k-1], conv[k])
                suffix_max[-1] = conv[-1]
                for k in range(2 * n - 3, -1, -1):
                    suffix_max[k] = max(suffix_max[k+1], conv[k])
                new_obj = compute_obj(a)
                best_obj = new_obj
                improved = True
        
        if not improved:
            break
        it += 1
        if verbose and it % 10 == 0:
            print(f"    CD it{it}: obj={best_obj:.8f}")
    
    return a / np.sum(a)
```

But each LP solve gives us the "best response" `g` to the current `a`. The iteration `a <- (1-t)*a + t*g` should converge. The question is how many iterations we need.

Let me think about convergence. The best-response dynamics (alternating between `a` and its best response `g(a)`) should converge to a fixed point where `a = g(a)`, which is a Nash equilibrium of the 2-player zero-sum game where both players choose the same strategy. At this fixed point, `max_k (a*a)[k] <= M` where `M` is the LP optimum, and `a` is the best response to itself.

Actually, I realize that the fixed point of the LP iteration is exactly a local (or global?) minimum of our objective. Let me verify:

At a fixed point, `a = g(a)` where `g(a)` minimizes `max_k (a*g)[k]` over the simplex. Since `a = g(a)`, we have `max_k (a*a)[k] = max_k (a*g(a))[k] = M(a)`.

Is this a minimum? The LP gives us that for any `h` on the simplex, `max_k (a*h)[k] >= M(a)`. In particular, for `h = a`, `max_k (a*a)[k] >= M(a)`. But we also have `max_k (a*a)[k] = M(a)` (since `a = g(a)` achieves the LP optimum). So the LP bound is tight.

But is it a minimum of the original objective? The original objective is `2n * max_k (a*a)[k]` (for normalized a). At the fixed point, this equals `2n * M(a)`.

To check if it's a minimum, consider a perturbation `a + ε*h`. The new objective is `2n * max_k ((a+εh)*(a+εh))[k] ≈ 2n * max_k (a*a + 2ε a*h)[k]`. The change is `2ε * max_k (a*h)[k]` (approximately, if the argmax doesn't change). Since `M(a) = min_h max_k (a*h)[k]`, we have `max_k (a*h)[k] >= M(a)` for all `h`. So the perturbation increases the objective (for small positive ε). Thus the fixed point is a local minimum!

Great, so the LP iteration converges to a local minimum. The question is: how good is this local minimum, and can we do better?

**Key insight for improvement**: The LP iteration always converges to the same local minimum regardless of the starting point (within a basin of attraction). To find a better minimum, we need to:
1. Start from different initial points
2. Use a different optimization method that can escape local minima
3. Use a different problem formulation

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Fast LP best-response iteration (scipy HiGHS), coordinate descent polish,
smoothed gradient descent, multi-start diversity, and length exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = a[:end - j]
    return A

def solve_lp_fast(a):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            g = g / s
        return g, float(res.x[n])
    return None, None

def lp_iteration(a, time_budget=100.0, verbose=False, seed=42, line_search=True):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    it = 0
    best_it = 0
    prev_g = None
    momentum = 0.0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break
        
        g, M = solve_lp_fast(a)
        if g is None:
            break
        
        # Momentum
        if prev_g is not None:
            momentum = 0.8 * momentum + 0.2 * np.linalg.norm(g - prev_g)
        prev_g = g.copy()
        
        it += 1
        
        # Line search
        if line_search:
            t_cands = [0.5, 0.7, 0.9, 1.0, 1.2, 1.5]
        else:
            t_cands = [1.0]
        
        best_t_obj = best_obj
        best_t_a = best_a.copy()
        
        for t in t_cands:
            if t <= 1.0:
                a_try = (1.0 - t) * a + t * g
            else:
                a_try = t * g - (t - 1.0) * a
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()
        
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_a
            best_it = it
            if verbose and it % 50 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}")
        else:
            if it - best_it > 20:
                # Perturb and restart
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                prev_g = None
                momentum = 0
                best_it = it
        
        # Check convergence
        if M < best_obj / (2 * n) * 1.001:
            # Very close to fixed point
            if verbose:
                print(f"    LP it{it}: converged, M={M:.8f}")
            break
    
    return best_a

def coordinate_descent(a, time_budget=30.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    conv = compute_conv(a)
    best_obj = float(np.max(conv))  # Since sum=1, obj = 2n*max(conv)
    start = time.monotonic()
    
    # Prefix and suffix max
    m = 2 * n - 1
    prefix_max = np.zeros(m)
    prefix_max[0] = conv[0]
    for k in range(1, m):
        prefix_max[k] = max(prefix_max[k - 1], conv[k])
    suffix_max = np.zeros(m)
    suffix_max[-1] = conv[-1]
    for k in range(m - 2, -1, -1):
        suffix_max[k] = max(suffix_max[k + 1], conv[k])
    
    it = 0
    while time.monotonic() - start < time_budget:
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget - 1:
                break
            
            old_val = a[i]
            S = 1.0 - old_val
            if S < 1e-10:
                continue
            
            k_start = i
            k_end = min(m - 1, i + n - 1)
            
            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, prefix_max[k_start - 1])
            if k_end < m - 1:
                unaff_max = max(unaff_max, suffix_max[k_end + 1])
            
            # Get affected indices
            ks = np.arange(k_start, k_end + 1)
            ki = ks - i  # indices into a for a[k-i]
            
            # base and slope for linear positions (k != 2i)
            mask_lin = (ks != 2 * i)
            base_lin = np.zeros(len(ks))
            slope = np.zeros(len(ks))
            
            if np.any(mask_lin):
                base_lin[mask_lin] = conv[ks[mask_lin]] - 2 * old_val * a[ki[mask_lin]]
                slope[mask_lin] = 2 * a[ki[mask_lin]]
            
            # Quadratic at k=2i
            has_quad = (2 * i >= k_start and 2 * i <= k_end)
            if has_quad:
                base_quad = conv[2 * i] - old_val ** 2
                quad_idx = 2 * i - k_start  # index in ks array
            
            # Candidate x values
            candidates = [0.0]
            if old_val > 1e-10:
                candidates.extend([old_val * 0.3, old_val * 0.7, old_val * 1.3, old_val * 2.0])
            
            best_x = old_val
            best_f = 2.0 * n * best_obj  # current objective (with sum=1)
            
            for x in candidates:
                new_s = S + x
                if new_s < 1e-10:
                    continue
                
                # Compute max over affected positions
                aff_max = unaff_max
                
                if np.any(mask_lin):
                    vals = base_lin[mask_lin] + slope[mask_lin] * x
                    aff_max = max(aff_max, float(np.max(vals)))
                
                if has_quad:
                    aff_max = max(aff_max, base_quad + x * x)
                
                obj = 2.0 * n * aff_max / (new_s ** 2)
                if obj < best_f:
                    best_f = obj
                    best_x = x
            
            if best_x != old_val and best_f < 2.0 * n * best_obj - 1e-12:
                a[i] = best_x
                a = a / a.sum()
                conv = compute_conv(a)
                best_obj = float(np.max(conv))
                
                # Update prefix/suffix
                prefix_max[0] = conv[0]
                for k in range(1, m):
                    prefix_max[k] = max(prefix_max[k - 1], conv[k])
                suffix_max[-1] = conv[-1]
                for k in range(m - 2, -1, -1):
                    suffix_max[k] = max(suffix_max[k + 1], conv[k])
                
                improved = True
        
        if not improved:
            break
        it += 1
        if verbose and it % 5 == 0:
            print(f"    CD it{it}: obj={2.0 * n * best_obj:.8f}")
    
    return a

def smoothed_gradient_descent(a, time_budget=60.0, tau_init=20.0, tau_end=500.0, verbose=False):
    from scipy.fft import fft, ifft
    
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp
        
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]
        
        # Update
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                bb_lr = min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb_lr
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
