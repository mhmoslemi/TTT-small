"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous: asymmetric initialization, accelerated LP fixed-point
with double-iteration, active-set tracking for subgradient methods, adaptive n-search,
and Fourier-domain perturbation for escaping local minima.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def convolve_with_fixed(a_fixed, g):
    """Compute conv(a_fixed, g) efficiently via FFT."""
    n = len(a_fixed)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a_fixed, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:2 * n - 1]


def lp_solve(a_fixed, solver_list=None):
    """
    Solve LP: min M s.t. conv(a_fixed, g) <= M, g >= 0, sum(g) = 1.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a_fixed)
    if n > 3000:
        return None, None

    # Build convolution matrix A: A[k,j] = a_fixed[k-j] if 0 <= k-j < n else 0
    # Use a banded structure for efficiency
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_fixed

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.ECOS, cp.SCS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue

    return None, None

        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    stall = 0
    iteration = 0
    solver_list = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.ECOS, cp.SCS]

    while time.monotonic() - start < time_budget and iteration < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Solve LP with current a as fixed

        if g_lp is None:
            break

        # Try accelerated blend: first use t=1 (full replacement), then smaller t

        best_local_a = best_a.copy()

                best_local_a = a_try.copy()
                best_t = t

            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a
            last_t = best_t
            stall = 0
            iteration += 1

                    if s_try > 1e-10:

def smoothed_subgrad(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Returns (gradient of length n, exact_max of conv).
    """
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = conv - exact_max
    max_shifted = 0.0
    exp_vals = np.exp(tau * shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    # Gradient: d/d a[i] of smoothed_max = sum_k p[k] * 2*a[k-i] (with bounds)
    # Use FFT for efficiency
    N = 1
    while N < 2 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def nesterov_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9, lr_scale=3.0):
    """Nesterov accelerated gradient with staircase tau schedule."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 15, 50, 150, 400, 1000, 2500, 5000]

    it = 0
    best_it = 0
    stall_patience = 3000

    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0
        max_iters_per_level = 5000

        while time.monotonic() - start < time_budget:
            if tau_iters >= max_iters_per_level:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_subgrad(y, tau)

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it
            a = a_new
            it += 1
            tau_iters += 1
            lr = max(1e-8, lr * 0.99999)

            if it - best_it > stall_patience:
                a = best_a.copy()
                v = np.zeros(n)
                lr = lr_scale / np.sqrt(n)
                best_it = it

        if verbose:
            print(f"    GD tau={tau:5d}: obj={best_obj:.10f}, iters={tau_iters}")

    return best_a


def make_asymmetric_init(n, mode, rng=None):
    """Create diverse asymmetric initial sequences."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.arange(n, dtype=np.float64)

    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        # Off-center Gaussian
        c = n * 0.3
        sigma = n * 0.15
        a = np.exp(-(x - c) ** 2 / (2 * sigma ** 2))
        return a / a.sum()
    elif mode == 2:
        # Two asymmetric bumps
        c1, c2 = n * 0.25, n * 0.6
        w1, w2 = n * 0.1, n * 0.15
        a = np.exp(-(x - c1) ** 2 / (2 * w1 ** 2)) + 0.7 * np.exp(-(x - c2) ** 2 / (2 * w2 ** 2))
        return a / a.sum()
    elif mode == 3:
        # Exponential ramp with taper
        a = np.exp(x / (n * 0.3))
        a *= np.exp(-(x - n) ** 2 / (2 * (n * 0.2) ** 2))
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 4:
        # Cosine-based asymmetric
        t = x / n
        a = 0.5 + 0.3 * np.cos(2 * np.pi * t) + 0.2 * np.cos(4 * np.pi * t + 0.5)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 5:
        # Power law
        a = (1 + x / n) ** 3
        return a / a.sum()
    elif mode == 6:
        # Random smooth (low-pass filtered noise)
        a = rng.uniform(0, 1, n)
        # Apply low-pass filter
        freq = 1 - np.fft.rfftfreq(n)
        mask = 1.0 / (1.0 + (freq * n / 5) ** 2)
        A = np.fft.rfft(a)
        A *= mask
        a = np.fft.irfft(A, n)
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 7:
        # "Anti-triangle": high at edges, low in middle
        t = x / n
        a = 0.3 + 0.7 * (t ** 2 + (1 - t) ** 2)
        return a / a.sum()
    elif mode == 8:
        # Sine with phase shift (asymmetric)
        a = 0.5 + 0.4 * np.sin(2 * np.pi * x / n + 0.3) + 0.1 * np.sin(4 * np.pi * x / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 9:
        # Concentrated at one end with long tail
        a = np.exp(-x / (n * 0.1)) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def subgradient_step(a, n_active=1):
    """
    Single subgradient step: move opposite to the subgradient at the active
    convolution positions. Returns updated a.
    """
    n = len(a)
    a = a / a.sum()
    conv = compute_conv(a)
    
    # Find active set (positions where conv is near max)
    max_val = float(np.max(conv))
    threshold = max_val * 0.99
    active = np.where(conv >= threshold)[0]
    
    # Subgradient: d/d a[i] of conv[k] = 2*a[k-i] (with a[j]=0 outside [0,n-1])
    # We want to DECREASE the active conv values, so move in -gradient direction
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                if i == j:  # k = 2i
                    grad[i] += 2.0 * a[i]
                else:
                    grad[i] += 2.0 * a[j]
    
    # Normalize gradient
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a
    
    # Step size: choose to reduce max conv by a fixed fraction
    # Approximate: step = alpha / gnorm where alpha controls reduction
    alpha = 0.1 * max_val / gnorm
    a_new = a - alpha * grad
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a
    return a_new / s_new


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Asymmetric multi-start with quick GD (150s)
    Phase 2: Accelerated LP fixed-point polishing (600s)
    Phase 3: Multi-n exploration (300s)
    Phase 4: Intensive final LP (300s)
    Phase 5: Final GD + LP polish (100s)
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        parent = None

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.10f}")

    n_current = len(best_seq)
    print(f"Starting n={n_current}, score={best_score:.10f}")

    # === Phase 1: Asymmetric multi-start ===
    print(f"\n=== Phase 1: Asymmetric Multi-Start ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.12, 180.0)
    
    if p1_budget > 20:
        modes = [1, 2, 3, 4, 5, 6, 7, 8, 9]
        per_mode = p1_budget / len(modes)
        for mode in modes:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            a_init = make_asymmetric_init(n_current, mode, rng)
            alloc = min(per_mode * 1.2, remaining * 0.4, 15.0)
            if alloc < 3:
                break
            a_screen = nesterov_gd(a_init, time_budget=alloc,
                                    tau_levels=[10, 50, 200], verbose=False)
            obj_screen = compute_obj(a_screen)
            if obj_screen < best_score:
                best_score = obj_screen
                best_seq = a_screen.copy()
                n_current = len(best_seq)
                print(f"  Mode {mode}: obj={best_score:.10f} **")
            else:
                print(f"  Mode {mode}: obj={obj_screen:.10f}")

    # Also run GD on parent
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 40.0)
        a_gd = nesterov_gd(best_seq, time_budget=alloc,
                            tau_levels=[5, 20, 80, 300], verbose=False)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            n_current = len(best_seq)
            print(f"  GD parent: obj={best_score:.10f} **")

    # === Phase 2: Main LP polishing ===
    print(f"\n=== Phase 2: Accelerated LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.4, 600.0)
    
    if p2_budget > 30 and n_current <= 2500:
        # Run multiple rounds with different seeds for perturbation
        for rnd in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(p2_budget / max(1, 8 - rnd) * 1.5, remaining * 0.7, 100.0)
            if alloc < 15:
                break
            a_lp = lp_polish_accelerated(best_seq, time_budget=alloc,
                                          verbose=True, seed=100 + rnd * 17)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.10f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.10f}")
    else:
        print(f"  Skipped (n={n_current}, budget={p2_budget:.0f}s)")

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 300.0)
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current - 100, n_current - 50, n_current + 50, n_current + 100,
        n_current + 200, n_current + 300,
        500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600, 2000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500]
    # Prioritize: close to current, then larger
    n_options.sort(key=lambda x: (abs(x - n_current), -x))
    n_options = n_options[:8]
    print(f"  n candidates: {n_options}")

    per_n = p3_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample(best_seq, n_try)
        alloc_gd = min(per_n * 0.4, remaining * 0.3, 20.0)
        if alloc_gd > 5:
            a_try = nesterov_gd(a_try, time_budget=alloc_gd,
                                 tau_levels=[10, 50, 200], verbose=False)

        remaining = deadline - time.monotonic()
        if n_try <= 1500:
            alloc_lp = min(per_n * 0.5, remaining * 0.4, 30.0)
            if alloc_lp > 10:
                a_try = lp_polish_accelerated(a_try, time_budget=alloc_lp,
                                               verbose=False, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 4: Intensive final LP ===
    print(f"\n=== Phase 4: Intensive Final LP ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.25, 300.0)
    n_current = len(best_seq)

    if p4_budget > 30 and n_current <= 2500:
        # Long LP run
        a_f = lp_polish_accelerated(best_seq, time_budget=p4_budget,
                                     verbose=True, seed=9999, max_iters=500)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Intensive LP: obj={best_score:.10f} **")
        else:
            print(f"  Intensive LP: obj={obj_f:.10f}")
    else:
        print(f"  Skipped")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau GD
    if remaining > 30:
        alloc = min(remaining * 0.25, 50.0)
        a_f1 = nesterov_gd(best_seq, time_budget=alloc,
                            tau_levels=[100, 400, 1200, 3000, 6000], verbose=False)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final GD: obj={best_score:.10f} **")

    # One more LP pass
    remaining = deadline - time.monotonic()
    if remaining > 20 and len(best_seq) <= 2500:
        alloc = min(remaining * 0.3, 40.0)
        a_f2 = lp_polish_accelerated(best_seq, time_budget=alloc,
                                      verbose=True, seed=4242, max_iters=200)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    # Targeted subgradient steps on the final solution
    remaining = deadline - time.monotonic()
    if remaining > 5:
        a_sub = best_seq.copy()
        for _ in range(50):
            a_sub = subgradient_step(a_sub)
            obj_sub = compute_obj(a_sub)
            if obj_sub < best_score:
                best_score = obj_sub
                best_seq = a_sub.copy()
                print(f"  Subgrad: obj={best_score:.10f} **")
            if remaining - (time.monotonic() - start_time) < 3:
                break

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
