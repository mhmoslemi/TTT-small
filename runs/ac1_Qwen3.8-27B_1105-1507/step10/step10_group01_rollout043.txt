def optimize_smoothed_max(a, time_budget, tau_start=50, tau_end=5000, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov momentum
    beta = 0.9
    v = np.zeros(n)
    
    N = 1
    while N < 3*n:
        N <<= 1
    
    it = 0
    best_it = 0
    base_lr = 5.0 / n  # adaptive
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_start + (tau_end - tau_start) * progress ** 2
        
        # Nesterov step
        y = a + beta * v
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-10:
            y = best_a.copy()
            sy = y.sum()
        y = y / sy
        
        # Compute gradient
        conv = compute_conv(y)
        max_c = conv.max()
        # Clip for numerical stability
        exp_vals = np.exp(tau * (conv - max_c))
        p = exp_vals / exp_vals.sum()
        
        A_fft = fft(y, N)
        P_fft = fft(p, N)
        grad = 2.0 * ifft(P_fft * np.conj(A_fft)).real[:n]
        
        grad_norm = np.linalg.norm(grad)
        
        # Step size: for Lipschitz smooth function, lr <= 1/L
        # L is related to the spectral norm of the Hessian
        # For now, use adaptive step size
        lr = base_lr / (1 + grad_norm * 0.1)
        lr = min(lr, 10.0 / n)
        
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        
        # Reset if stuck
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it
            base_lr *= 0.5
        
        if verbose and it % 1000 == 0:
            print(f"  SmoothMax: it={it}, obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.4f}")
    
    return best_a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a multi-start strategy with fast smoothed-max gradient descent as the core.
Each start uses a different initialization shape (uniform, U-shaped, W-shaped, etc.)
and runs Nesterov-accelerated gradient descent on the log-sum-exp smoothed maximum
of the self-convolution, with progressively increasing temperature.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_max_gradient(a, tau, N):
    """Compute gradient of (1/tau)*log(sum exp(tau*conv(a,a))) w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    # Numerical stability
    shifted = tau * (conv - max_c)
    shifted = np.clip(shifted, -700, 0)
    exp_vals = np.exp(shifted)
    p = exp_vals / float(np.sum(exp_vals))
    
    A_fft = fft(a, N)
    P_fft = fft(p, N)
    grad = 2.0 * ifft(P_fft * np.conj(A_fft)).real[:n]
    return grad


def nesterov_smoothed_max(a, time_budget, tau_start=30.0, tau_end=3000.0, verbose=False):
    """Nesterov-accelerated gradient descent on smoothed max of self-convolution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    beta = 0.9
    v = np.zeros(n)
    lr = 3.0 / n
    it = 0
    best_it = 0
    eval_count = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 1.5
        
        # Nesterov extrapolation
        y = a + beta * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_a.copy()
            sy = float(np.sum(y))
        y = y / sy
        
        # Gradient
        grad = smoothed_max_gradient(y, tau, N)
        g_norm = float(np.linalg.norm(grad))
        
        if g_norm < 1e-15:
            break
        
        # Adaptive learning rate
        # The Lipschitz constant of the gradient is roughly O(n) for the smoothed max
        # Use a conservative step: lr ~ 1/sqrt(n) scaled by gradient magnitude
        lr_eff = lr / (1.0 + 0.5 * g_norm / np.sqrt(n))
        lr_eff = min(lr_eff, 5.0 / np.sqrt(n))
        lr_eff = max(lr_eff, 1e-10)
        
        # Update
        v = beta * v - lr_eff * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        eval_count += 1
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        
        # Decay learning rate slowly
        lr *= 0.9995
        
        # Reset if stuck for too long
        if it - best_it > 3000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = max(lr * 0.5, 0.5 / n)
            best_it = it
            if lr < 1e-8:
                break
        
        if verbose and it % 500 == 0:
            print(f"    Nesterov: it={it}, obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.5f}")
    
    return best_a


def variance_gradient_descent(a, time_budget, verbose=False):
    """Minimize variance of self-convolution using GD."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    mean_val = 1.0 / (2 * n - 1)
    beta = 0.9
    v = np.zeros(n)
    lr = 2.0 / n
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_a.copy()
        y = y / sy
        
        conv = compute_conv(y)
        residual = conv - mean_val
        
        A_fft = fft(y, N)
        R_fft = fft(residual, N)
        grad = 4.0 * ifft(R_fft * np.conj(A_fft)).real[:n]
        
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break
        
        lr_eff = lr / (1.0 + g_norm / np.sqrt(n))
        lr_eff = min(lr_eff, 3.0 / np.sqrt(n))
        
        v = beta * v - lr_eff * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        lr *= 0.999
        
        if it - best_it > 2000:
            a = best_a.copy()
            v = np.zeros(n)
            lr *= 0.5
            best_it = it
            if lr < 1e-10:
                break
    
    return best_a


def generate_initialization(n, method, rng):
    """Generate an initial sequence of length n using the specified method."""
    x = np.linspace(0, 1, n)
    
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'u_shaped':
        c = rng.uniform(0.5, 2.0)
        a = 1.0 + c * (2.0 * x - 1.0) ** 2
    elif method == 'w_shaped':
        c = rng.uniform(0.3, 1.5)
        a = 1.0 + c * np.cos(4.0 * np.pi * x)
    elif method == 'raised_cosine':
        c = rng.uniform(0.3, 0.9)
        a = 1.0 + c * np.cos(2.0 * np.pi * x)
    elif method == 'parabolic':
        c = rng.uniform(0.5, 2.0)
        a = 1.0 + c * (1.0 - (2.0 * x - 1.0) ** 2)
    elif method == 'random_smooth':
        # Random with low-pass filtering
        a = rng.uniform(0.5, 1.5, n)
        # Simple smoothing
        for _ in range(3):
            a = 0.5 * a + 0.25 * np.roll(a, 1) + 0.25 * np.roll(a, -1)
    elif method == 'ripple':
        freq = rng.integers(2, 8)
        amp = rng.uniform(0.2, 0.8)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase)
    elif method == 'two_hump':
        # Concentrated in two bumps
        center1 = rng.uniform(0.1, 0.4)
        center2 = 1.0 - center1
        width = rng.uniform(0.05, 0.2)
        a = np.exp(-((x - center1) / width) ** 2) + np.exp(-((x - center2) / width) ** 2)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def polexit_step(a, time_budget=5.0):
    """
    Polexit-style step: identify the peak lag, compute subgradient,
    and take a projected step to reduce it.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.0 / n
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_idx = int(np.argmax(conv))
        
        # Subgradient at position i: 2*a[max_idx - i] for valid max_idx - i
        grad = np.zeros(n)
        for i in range(n):
            j = max_idx - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
            j2 = max_idx - i
            # Actually the gradient is: d(conv[max_idx])/d(a[i])
            # = a[max_idx - i] (from a[i] as first factor) 
            #   + a[max_idx - i] (from a[i] as second factor, if max_idx - i != i)
            # = 2 * a[max_idx - i] if 0 <= max_idx - i < n
            # (The case max_idx - i == i gives 2*a[i] which is correct for d(a[i]^2)/d(a[i]) = 2*a[i])
        
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break
        
        a_new = a - lr * grad / g_norm
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.005, 3.0 / n)
        else:
            lr *= 0.9
        
        it += 1
        if it > 5000:
            break
    
    return best_a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses multi-start Nesterov accelerated gradient descent on the log-sum-exp smoothed
maximum of the self-convolution, with variance minimization pre-processing,
symmetry-breaking initializations, and progressive tau annealing.
Key insight: optimal sequences are asymmetric (obj < 2), so we explicitly
explore asymmetric shapes (ramps, offset cosines, etc.) alongside symmetric ones.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_max_gradient(a, tau, N):
    """Compute gradient of (1/tau)*log(sum exp(tau*conv(a,a))) w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    s_exp = float(np.sum(exp_vals))
    if s_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / s_exp

    A_fft = fft(a, N)
    P_fft = fft(p, N)
    grad = 2.0 * ifft(P_fft * np.conj(A_fft)).real[:n]
    return grad


def nesterov_smoothed_max(a, time_budget, tau_start=30.0, tau_end=3000.0, verbose=False):
    """Nesterov-accelerated GD on smoothed max of self-convolution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 3 * n:
        N <<= 1

    beta = 0.9
    v = np.zeros(n)
    lr = 4.0 / n
    it = 0
    best_it = 0
    patience = 4000

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 1.5

        y = a + beta * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_a.copy()
            sy = float(np.sum(y))
        y = y / sy

        grad = smoothed_max_gradient(y, tau, N)
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break

        lr_eff = lr / (1.0 + 0.3 * g_norm / np.sqrt(n))
        lr_eff = min(lr_eff, 8.0 / np.sqrt(n))
        lr_eff = max(lr_eff, 1e-12)

        v = beta * v - lr_eff * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr *= 0.9997

        if it - best_it > patience:
            a = best_a.copy()
            v = np.zeros(n)
            lr = max(lr * 0.4, 0.3 / n)
            best_it = it
            patience = int(patience * 0.7)
            if patience < 200:
                break

        if verbose and it % 1000 == 0:
            print(f"      Nesterov: it={it}, obj={best_obj:.8f}, tau={tau:.0f}")

    return best_a

def subgradient_peel(a, time_budget, verbose=False):
    """Subgradient method on the actual max of the self-convolution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.5 / np.sqrt(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))

        # Use top-5 lags for a richer subgradient
        top_k = min(5, len(conv))
        top_indices = np.argsort(conv)[-top_k:]
        weights = np.zeros(len(conv))
        for idx in top_indices:
            weights[idx] = 1.0 / top_k

        # Compute weighted subgradient
        N = 1
        while N < 3 * n:
            N <<= 1
        A_fft = fft(a, N)
        W_fft = fft(weights, N)
        grad = 2.0 * ifft(W_fft * np.conj(A_fft)).real[:n]

        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break

        a_new = a - lr * grad / g_norm
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            it += 1
            if it - best_it > 3000:
                break
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            a = a_new
            lr = min(lr * 1.003, 2.0 / np.sqrt(n))
        else:
            lr *= 0.97

        it += 1
        if it - best_it > 3000:
            break

    return best_a

        for _ in range(5):
            a = 0.5 * a + 0.25 * np.roll(a, 1) + 0.25 * np.roll(a, -1)
    elif method == 'ripple':
        freq = rng.integers(2, 12)
        amp = rng.uniform(0.2, 0.8)
        phase = rng.uniform(0, 2.0 * np.pi)
        a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.01)
    s = float(np.sum(a))
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-start with asymmetric initializations, Nesterov smoothed-max
    GD as primary optimizer, variance pre-processing, subgradient peeling, and
    progressive refinement with increasing tau.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                parent_obj = compute_obj(parent)
                print(f"Parent: n={parent.size}, obj={parent_obj:.8f}")
    except Exception:
        pass

    n_base = 800
    if parent is not None:
        n_base = parent.size

    # Build list of (name, sequence) starts
    start_seqs = []

    if parent is not None:
        p_res = resample_sequence(parent, n_base)
        start_seqs.append(('parent', p_res))

        # Asymmetric modifications of parent
        x = np.linspace(0, 1, n_base)
        for shift_c in [0.5, 1.0, -0.5, -1.0]:
            a_mod = p_res * (1.0 + shift_c * (x - 0.5))
            a_mod = np.maximum(a_mod, 0.01)
            a_mod = a_mod / a_mod.sum()
            start_seqs.append((f'parent_shift_{shift_c}', a_mod))

        # Fourier perturbation
        for _ in range(2):
            k = rng.integers(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            pert = 0.1 * np.cos(2 * np.pi * k * x + phase)
            a_p = p_res * np.exp(pert)
            a_p = np.maximum(a_p, 0.01)
            a_p = a_p / a_p.sum()
            start_seqs.append(('parent_fourier', a_p))

    # Diverse random starts (emphasize asymmetric shapes)
    asymmetric_methods = [
        'linear_ramp_up', 'linear_ramp_down', 'exp_decay', 'exp_growth',
        'asym_cosine', 'asym_cosine_q', 'para_left', 'para_right',
        'two_bump_asym', 'random_smooth', 'ripple',
        'u_shaped', 'w_shaped', 'raised_cosine', 'uniform'
    ]
    for m in asymmetric_methods:
        a_init = generate_init(n_base, m, rng)
        start_seqs.append((m, a_init))

    # Shuffle
    idx = list(range(len(start_seqs)))
    rng.shuffle(idx)
    start_seqs = [start_seqs[i] for i in idx]

    # === Phase 1: Multi-start ===
    print(f"\n=== Phase 1: Multi-start (n={n_base}, {len(start_seqs)} starts) ===")
    p1_budget = min(deadline - time.monotonic(), 650.0)
    time_per_start = p1_budget / max(len(start_seqs), 1)

    for name, a_init in start_seqs:
        remaining = deadline - time.monotonic()
        if remaining < 12:
            break
        alloc = min(time_per_start, remaining * 0.92)

        # Variance pre-opt (short)
        t_var = min(alloc * 0.25, 12.0)
        if t_var > 3:
            a_opt = variance_gd(a_init, time_budget=t_var)
        else:
            a_opt = a_init.copy()

        # Nesterov main
        remaining = deadline - time.monotonic()
        t_main = min(alloc - t_var - 2.0, remaining * 0.6)
        if t_main > 5:
            a_opt = nesterov_smoothed_max(a_opt, time_budget=t_main,
                                          tau_start=20.0, tau_end=2500.0)

        # Peel
        remaining = deadline - time.monotonic()
        t_peel = min(4.0, remaining * 0.08)
        if t_peel > 1:
            a_opt = subgradient_peel(a_opt, time_budget=t_peel)

        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  [{name}] obj={best_score:.8f} **")
        else:
            print(f"  [{name}] obj={obj:.8f}")

    # === Phase 2: Intensive refinement of best ===
    print(f"\n=== Phase 2: Intensive refinement ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        # Round A: Variance + Nesterov (medium tau)
        t_a = min(remaining * 0.25, 120.0)
        if t_a > 10:
            a_ref = variance_gd(best_seq, time_budget=min(t_a * 0.3, 30.0))
            remaining = deadline - time.monotonic()
            t_a2 = min(t_a - 5.0, remaining * 0.5)
            if t_a2 > 5:
                a_ref = nesterov_smoothed_max(a_ref, time_budget=t_a2,
                                              tau_start=50.0, tau_end=3000.0)
            obj_ref = compute_obj(a_ref)
            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_ref.copy()
                print(f"  Round A: obj={best_score:.8f} **")
            else:
                print(f"  Round A: obj={obj_ref:.8f}")

        # Round B: Nesterov (high tau)
        remaining = deadline - time.monotonic()
        t_b = min(remaining * 0.3, 150.0)
        if t_b > 10:
            a_ref = nesterov_smoothed_max(best_seq, time_budget=t_b,
                                          tau_start=200.0, tau_end=8000.0)
            obj_ref = compute_obj(a_ref)
            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_ref.copy()
                print(f"  Round B (high tau): obj={best_score:.8f} **")
            else:
                print(f"  Round B (high tau): obj={obj_ref:.8f}")

        # Round C: Subgradient peel
        remaining = deadline - time.monotonic()
        t_c = min(30.0, remaining * 0.1)
        if t_c > 3:
            a_ref = subgradient_peel(best_seq, time_budget=t_c)
            obj_ref = compute_obj(a_ref)
            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_ref.copy()
                print(f"  Round C (peel): obj={best_score:.8f} **")
            else:
                print(f"  Round C (peel): obj={obj_ref:.8f}")

    # === Phase 3: Symmetry breaking from best ===
    print(f"\n=== Phase 3: Symmetry breaking ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    x = np.linspace(0, 1, n_cur)

    sb_strategies = [
        ('tilt_up', best_seq * (1.0 + 0.8 * x)),
        ('tilt_down', best_seq * (1.0 + 0.8 * (1.0 - x))),
        ('tilt_weak', best_seq * (1.0 + 0.3 * (2.0 * x - 1.0))),
        ('exp_tilt', best_seq * np.exp(0.5 * x)),
        ('exp_tilt_r', best_seq * np.exp(0.5 * (1.0 - x))),
        ('cos_tilt', best_seq * (1.0 + 0.5 * np.sin(2.0 * np.pi * x))),
    ]

    sb_time = min(remaining * 0.15, 90.0)
    if sb_time > 10:
        t_each = sb_time / len(sb_strategies)
        for sname, a_sb in sb_strategies:
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            a_sb = np.maximum(a_sb, 0.01)
            a_sb = a_sb / a_sb.sum()
            alloc = min(t_each, remaining * 0.9)

            a_opt = nesterov_smoothed_max(a_sb, time_budget=alloc * 0.8,
                                          tau_start=100.0, tau_end=5000.0)
            remaining = deadline - time.monotonic()
            if remaining > 3:
                a_opt = subgradient_peel(a_opt, time_budget=min(3.0, remaining * 0.3))

            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  [{sname}] obj={best_score:.8f} **")
            else:
                print(f"  [{sname}] obj={obj:.8f}")

    # === Phase 4: n-sweep ===
    print(f"\n=== Phase 4: n-sweep ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 100), max(400, n_best - 50),
        n_best + 50, n_best + 100, n_best + 200,
        600, 900, 1100, 1300
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1600 and abs(n - n_best) > 20]
    n_options = n_options[:4]

    if n_options and remaining > 40:
        t_each_n = min(remaining * 0.25 / max(len(n_options), 1), 80.0)
        print(f"  Trying: {n_options}")
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_try = resample_sequence(best_seq, n_try)
            alloc = min(t_each_n, remaining * 0.9)

            a_try = nesterov_smoothed_max(a_try, time_budget=alloc * 0.7,
                                          tau_start=50.0, tau_end=4000.0)
            remaining = deadline - time.monotonic()
            if remaining > 3:
                a_try = subgradient_peel(a_try, time_budget=min(3.0, remaining * 0.3))

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Final prolonged polish ===
    print(f"\n=== Phase 5: Final polish ===")
    remaining = deadline - time.monotonic()

    # Multiple Nesterov rounds with progressively higher tau
    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 8:
            break
        tau_s = 100.0 * (r + 1)
        tau_e = 3000.0 * (r + 1)
        t_round = min(remaining * 0.3, 100.0)

        a_pol = nesterov_smoothed_max(best_seq, time_budget=t_round,
                                      tau_start=tau_s, tau_end=tau_e)
        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Final r{r} (tau={tau_s:.0f}-{tau_e:.0f}): obj={best_score:.8f} **")
        else:
            print(f"  Final r{r} (tau={tau_s:.0f}-{tau_e:.0f}): obj={obj_pol:.8f}")

        # Quick peel between rounds
        remaining = deadline - time.monotonic()
        if remaining > 3:
            a_pol = subgradient_peel(best_seq, time_budget=min(4.0, remaining * 0.2))
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()

    # Ultimate: very high tau Nesterov
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_ult = nesterov_smoothed_max(best_seq, time_budget=min(remaining * 0.6, 80.0),
                                      tau_start=1000.0, tau_end=20000.0)
        obj_ult = compute_obj(a_ult)
        if obj_ult < best_score:
            best_score = obj_ult
            best_seq = a_ult.copy()
            print(f"  Ultimate (tau=1000-20000): obj={best_score:.8f} **")
        else:
            print(f"  Ultimate: obj={obj_ult:.8f}")

    # Final peel
    remaining = deadline - time.monotonic()
    if remaining > 3:
        a_final = subgradient_peel(best_seq, time_budget=min(remaining * 0.5, 20.0))
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()

    # Validate
    if best_seq is None:
        best_seq = np.ones(800)
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
