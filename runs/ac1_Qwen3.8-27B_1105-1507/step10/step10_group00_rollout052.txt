The key equation at optimality (for the LP dual): If g is the LP solution, then the "dual variables" correspond to the weights on the active constraints. At the optimum, we have:
- a[i] > 0 => sum_k mu[k] * a[k-i] = lambda (for some Lagrange multiplier lambda)
- where mu[k] >= 0 and mu[k] > 0 only for active constraints (where C(k) = M)

This gives us a fixed-point equation: a[i] * sum_k mu[k] * a[k-i] / sum_j mu[j] * a[j-i] = 1 (for all i with a[i] > 0)

Or equivalently: a[i] / sum_k mu[k] * a[k-i] = constant (for active i)

This is reminiscent of power iteration. If we define the operator T(a)[i] = 1 / (sum_k mu[k] * a[k-i]) (for i with a[i] > 0), then at the fixed point, a[i] = c / T(a)[i] for some constant c.

But we don't know mu. At the first-order optimality condition, mu is proportional to the "weight" on each constraint, which is related to how "active" each constraint is.

In the soft max approach, mu[k] = p(k) = exp(beta * C(k)) / sum_j exp(beta * C(j)). So the fixed-point equation becomes:

a[i] = c / sum_k p(k) * a[k-i]  (for all i in the support)

Or: a[i] * sum_k p(k) * a[k-i] = c

This is exactly what mirror descent with the soft max gradient computes! The update a_new[i] = a[i] * exp(-eta * grad[i]) / sum_j a[j] * exp(-eta * grad[j]) where grad[i] = sum_k p(k) * a[k-i].

So the fixed point of mirror descent with the soft max is the same as the fixed point of the power iteration with mu = p.

As beta -> inf, p concentrates on the argmax, and we get the LP solution.

"""
Mirror descent / power iteration with adaptive soft-max temperature for minimizing
the peak autocorrelation of non-negative step functions.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv_fftpadded(a):
    """Linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = compute_conv_fftpadded(a)
    return 2.0 * n * float(np.max(conv))


def compute_soft_max_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*C(k))) and its value.
    Returns (grad, soft_max_value)
    grad[i] = sum_k p(k) * a[k-i] where p = softmax(beta * C)
    """
    n = len(a)
    conv = compute_conv_fftpadded(a)
    
    max_c = np.max(conv)
    shifted = beta * (conv - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    if sum_exp < 1e-300:
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                grad[i] += a[j]
        return grad, max_c
    
    p = exp_shifted / sum_exp
    soft_max_val = max_c + np.log(sum_exp) / beta
    
    # grad[i] = sum_j a[j] * p[i+j] for j=0..n-1
    # This is cross-correlation of a and p
    size = 1
    while size < 2 * n - 1 + n:  # need size >= (2n-1) + n - 1 = 3n-2
        size <<= 1
    
    a_pad = np.zeros(size)
    a_pad[:n] = a
    p_pad = np.zeros(size)
    p_pad[:2*n-1] = p
    
    fa = rfft(a_pad)
    fp = rfft(p_pad)
    cross_corr = irfft(fp * np.conj(fa), size)
    
    grad = cross_corr[:n]
    # Ensure numerical non-negativity
    grad = np.maximum(grad, 1e-15)
    
    return grad, soft_max_val


def mirror_descent_step(a, grad):
    """
    Mirror descent step on the simplex.
    a_new[i] = a[i] / grad[i], then normalize.
    This is the power iteration step.
    """
    n = len(a)
    # Multiplicative update: a_new[i] = a[i] / grad[i]
    a_new = a / grad
    s = np.sum(a_new)
    if s < 1e-15:
        return a.copy()
    return a_new / s


def blend_step(a, a_new, t):
    """Blend two sequences."""
    result = (1 - t) * a + t * a_new
    result = np.maximum(result, 0)
    s = np.sum(result)
    if s < 1e-15:
        return a.copy()
    return result / s


def soften_and_anneal(beta, schedule, it):
    """Anneal beta according to schedule."""
    if schedule == 'linear':
        return beta * (1 - 0.999 * it / 10000)
    elif schedule == 'exponential':
        return beta * (0.9995 ** it)
    else:
        return beta


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:

    elif mode == 2:

    elif mode == 3:

        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:

    elif mode == 5:

    elif mode == 6:

    elif mode == 7:
        a = np.exp(-5 * (2 * x - 1) ** 2) + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:

    Main optimization loop using mirror descent with soft-max gradient.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    return best_a


def subgradient_descent(a, time_budget=30.0, verbose=False):
    """Standard subgradient descent with momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    lr = 0.01
    momentum = np.zeros(n)
    beta_mom = 0.7
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv_fftpadded(a)
        k_max = int(np.argmax(conv))
        
        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                grad[i] = a[j]
        
        # Normalized gradient (make it zero-mean on the support)
        support = a > 1e-10
        if np.any(support):
            grad[support] -= np.mean(grad[support])
        
        momentum = beta_mom * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        lr *= 0.9998
        it += 1
        
        if it - best_it > 3000:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = 0.005
            best_it = it
    
    return best_a


def soften_line_search(a, a_target, time_budget=2.0):
    """Find best t for blend (1-t)*a + t*a_target."""
    lo, hi = 0.0, 1.5
    best_t = 0.0
    best_obj = compute_obj_normalized(a)
    
    # Try a few values
    candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    for t in candidates:
        a_try = (1 - t) * a + t * a_target
        a_try = np.maximum(a_try, 0)
        s = np.sum(a_try)
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
    
    return best_t, best_obj


def try_length(a_best, n_try, time_budget):
    """Try a different sequence length by interpolating."""
    n_old = len(a_best)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        return None, np.inf
    a_try = a_try / s
    return a_try, compute_obj_normalized(a_try)
