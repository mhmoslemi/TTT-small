"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses improved LP alternating minimization with adaptive line search,
entropic mirror descent polishing, and multi-start exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return ir fft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a, solver=cp.CLARABEL, tol=1e-8):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for s in (solver, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS):
        try:
            prob.solve(solver=s, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 3 * n:
        N <<= 1
    from scipy.fft import fft, ifft
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def mirror_descent_polish(a, time_budget, eta_init=0.1, eta_final=0.001, tau=50.0):
    """Entropic mirror descent on the simplex for polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init
    it = 0
    
    while time.monotonic() - start < time_budget:
        grad = smoothed_gradient(a, tau)
        # Mirror descent update: g_k = a_k * exp(-eta * grad_k) / Z
        log_update = -eta * grad
        log_update -= np.max(log_update)  # numerical stability
        weights = np.exp(log_update)
        g = a * weights
        s_g = float(np.sum(g))
        if s_g < 1e-10:
            break
        g = g / s_g
        
        obj_g = compute_obj(g)
        if obj_g < best_obj:
            best_obj = obj_g
            best_a = g.copy()
        
        a = g
        eta = max(eta * 0.995, eta_final)
        tau = min(tau * 1.01, 500.0)
        it += 1
    
    return best_a

def nesterov_polish(a, time_budget, tau_init=30.0, tau_max=2000.0):
    """Nesterov accelerated gradient descent for polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.85
    base_lr = 1.5 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0
    
    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))
        
        if g_norm > 1e-15:
            adaptive_lr = 0.5 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.1 * base_lr), 3.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adaptive_lr
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        lr *= 0.9997
        tau = min(tau * 1.015, tau_max)
        it += 1
    
    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.

entropic mirror descent polishing, Nesterov gradient polishing,

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a, solvers=None):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solvers is None:
        solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.MOSEK]

    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of (a*a)_k using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]

def fourier_perturb(a, rng, strength=0.01, n_modes=8):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            a_try = resample_to_length(best_a, n_try)
            alloc = min(remaining * 0.35, 80.0)
            if alloc < 20:
                break
            a_try = lp_alternate_improved(a_try, time_budget=alloc, seed=n_try + 100, verbose=False)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                print(f"  ** n={n_try}: obj={best_obj:.10f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 7: Ultra-final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 7: Ultra-final ({remaining:.0f}s) ===")
        alloc = min(remaining * 0.7, 50.0)
        a_final = lp_alternate_improved(best_a, time_budget=alloc, seed=seed + 9999, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final: obj={best_obj:.10f} **")
        else:
            print(f"  Final: obj={obj_final:.10f}")

    # Final validation
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
