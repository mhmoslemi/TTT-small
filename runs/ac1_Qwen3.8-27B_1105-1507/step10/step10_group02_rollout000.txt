import numpy as np
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, max_time=30.0):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    t0 = time.monotonic()
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        if time.monotonic() - t0 > max_time:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 2:
                # Perturb to escape
                pert = rng.normal(0, 0.02 + 0.01 * stall_count, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                stall_count = 0

        if verbose and iteration % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    AltLP iter {iteration}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 10, size=7)
        phases = rng.uniform(0, 2 * np.pi, size=7)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        a = a / a.sum()
    elif mode == 7:
        # Raised cosine
        a = 0.5 * (1 + np.cos(np.pi * np.arange(n) / (n - 1)))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def block_cd(a, time_budget=100.0, block_size=2, verbose=False, seed=42):
    """
    Block coordinate descent: optimize small blocks of coordinates simultaneously
    using Nelder-Mead on the block.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    from scipy.optimize import minimize

    def block_obj(x_block, indices, a_full, S_other):
        a_try = a_full.copy()
        a_try[indices] = x_block
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return 1e10
        a_try = a_try / s
        return 2.0 * n * float(np.max(compute_conv(a_try)))

    sweep = 0
    while time.monotonic() - start < time_budget:
        if time.monotonic() - start >= time_budget:
            break
        sweep += 1
        order = rng.permutation(n)

        for block_start in range(0, n, block_size):
            if time.monotonic() - start >= time_budget:
                break

            block_end = min(block_start + block_size, n)
            indices = list(order[block_start:block_end])
            if len(indices) < 2:
                continue

            x0 = a[indices]
            S_other = S - float(np.sum(x0))

            # Nelder-Mead on the block
            try:
                res = minimize(
                    lambda x: block_obj(x, indices, a, S_other),
                    x0,
                    method='Nelder-Mead',
                    options={'maxiter': 200, 'fatol': 1e-12, 'xatol': 1e-10}
                )
                x_opt = np.maximum(res.x, 0)
                if float(np.sum(x_opt)) > 1e-10:
                    a[indices] = x_opt
                    S = float(np.sum(a))
                    conv = compute_conv(a)
                    obj = 2.0 * n * float(np.max(conv)) / (S * S)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
            except Exception:
                continue

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    BlockCD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining CD, alternating LP, block CD, and n-exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = 2.0 * n_default * float(np.max(compute_conv(best_seq)))
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    # === Phase 1: Coordinate Descent (primary workhorse) ===
    print(f"\n=== Phase 1: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.40, 500.0)
    if p1_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget,
                                  verbose=True, seed=seed)
        obj_cd = 2.0 * a_cd.size * float(np.max(compute_conv(a_cd / a_cd.sum())))
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: Alternating LP ===
    print(f"\n=== Phase 2: Alternating LP ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 30:
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(p2_budget / (3 - rnd), remaining * 0.6)
            if alloc < 15:
                break
            a_lp = alternating_lp(best_seq, time_budget=alloc, verbose=False,
                                  seed=2000 + rnd * 100)
            obj_lp = 2.0 * a_lp.size * float(np.max(compute_conv(a_lp / a_lp.sum())))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  AltLP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  AltLP round {rnd}: obj={obj_lp:.8f}")
                # Follow up with CD
                if remaining > 30:
                    a_cd2 = coordinate_descent(a_lp, time_budget=min(30, remaining * 0.1),
                                               verbose=False, seed=3000 + rnd)
                    obj_cd2 = 2.0 * a_cd2.size * float(np.max(compute_conv(a_cd2 / a_cd2.sum())))
                    if obj_cd2 < best_score:
                        best_score = obj_cd2
                        best_seq = a_cd2.copy()
                        print(f"  CD after AltLP {rnd}: obj={best_score:.8f} **")

    # === Phase 3: CD Refinement ===
    print(f"\n=== Phase 3: CD Refinement ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 20:
        a_cd2 = coordinate_descent(best_seq, time_budget=p3_budget,
                                   verbose=True, seed=seed + 500)
        obj_cd2 = 2.0 * a_cd2.size * float(np.max(compute_conv(a_cd2 / a_cd2.sum())))
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  CD2: obj={best_score:.8f} **")
        else:
            print(f"  CD2: obj={obj_cd2:.8f}")

    # === Phase 4: Block CD for escape ===
    print(f"\n=== Phase 4: Block CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        # Perturb slightly then block CD
        pert = rng.normal(0, 0.01, len(best_seq))
        a_pert = best_seq * np.exp(pert)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.2, 80.0)
        a_block = block_cd(a_pert, time_budget=alloc, block_size=2,
                           verbose=False, seed=seed + 1000)
        obj_block = 2.0 * a_block.size * float(np.max(compute_conv(a_block / a_block.sum())))
        if obj_block < best_score:
            best_score = obj_block
            best_seq = a_block.copy()
            print(f"  BlockCD: obj={best_score:.8f} **")
        else:
            print(f"  BlockCD: obj={obj_block:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(200, n_best - 200), max(200, n_best - 100),
        n_best + 100, n_best + 200, n_best + 400,
        600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]
    # Sort by distance from current best, try closest different ones first
    n_options = sorted([n for n in n_options if n != n_best], key=lambda x: abs(x - n_best))
    n_options = n_options[:4]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick CD on new length
        alloc = min(remaining * 0.2, 60.0)
        if alloc > 15:
            a_cd_try = coordinate_descent(a_try, time_budget=alloc, verbose=False,
                                          seed=n_try)
            obj_try = 2.0 * a_cd_try.size * float(np.max(compute_conv(a_cd_try / a_cd_try.sum())))
        else:
            a_cd_try = a_try
            obj_try = 2.0 * a_try.size * float(np.max(compute_conv(a_try / a_try.sum())))

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_cd_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = 1 + r * 2  # modes 1, 3, 5
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.25, 60.0)
        if alloc > 20:
            a_r = coordinate_descent(a_r, time_budget=alloc, verbose=False,
                                     seed=6000 + r * 100)
            obj_r = 2.0 * a_r.size * float(np.max(compute_conv(a_r / a_r.sum())))
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 100.0)
        a_final = coordinate_descent(best_seq, time_budget=alloc,
                                     verbose=False, seed=seed + 9999)
        obj_final = 2.0 * a_final.size * float(np.max(compute_conv(a_final / a_final.sum())))
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # Final LP pass
    remaining = deadline - time.monotonic()
    if remaining > 25:
        a_lp_final = alternating_lp(best_seq, time_budget=min(remaining * 0.6, 50.0),
                                    verbose=False, seed=7777)
        obj_lp_final = 2.0 * a_lp_final.size * float(np.max(compute_conv(a_lp_final / a_lp_final.sum())))
        if obj_lp_final < best_score:
            best_score = obj_lp_final
            best_seq = a_lp_final.copy()
            print(f"  Final AltLP: obj={best_score:.8f} **")

    # One more CD if time
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = coordinate_descent(best_seq, time_budget=min(remaining * 0.7, 60.0),
                                    verbose=False, seed=8888)
        obj_last = 2.0 * a_last.size * float(np.max(compute_conv(a_last / a_last.sum())))
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
