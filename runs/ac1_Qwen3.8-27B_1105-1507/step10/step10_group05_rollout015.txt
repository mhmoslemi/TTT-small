def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smoothed_gradient(a, tau):
    """Compute gradient of smoothed max of conv(a,a) using FFT. a should have sum=1."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    grad = 2.0 * g_full[:n]
    return grad, conv

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""

Combines fast multiplicative mirror descent, Nesterov-accelerated smoothed
subgradient descent with adaptive tau, multi-length exploration, structured

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_smoothed_gradient(a, tau):
    """
    Compute gradient of tau-smoothed max of conv(a,a) using FFT.
    a should have sum=1 for the gradient to be in the right scale.
    Returns (gradient, convolution).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    grad = 2.0 * g_full[:n]
    return grad, conv


def nesterov_smoothed_descent(a, time_budget=300.0, tau_init=10.0, tau_end=2000.0,
                              verbose=False):
    """
    Nesterov accelerated smoothed gradient descent with adaptive step size.
    Uses cosine annealing for tau and backtracking for step size.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Initial learning rate based on Lipschitz constant estimate
    lr = 1.0 / (2.0 * max(tau_init, 1.0))
    momentum = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0
    stuck_counter = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Cosine annealing for tau (increasing over time for more precise max approximation)
        tau = tau_init + (tau_end - tau_init) * (1.0 - 0.5 * (1.0 + math.cos(math.pi * progress)))

        # Nesterov extrapolation
        y = a + momentum * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = compute_smoothed_gradient(y, tau)

        # Gradient step with projection
        a_new = y - lr * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = y.copy()
            s_new = 1.0
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)

        # Backtracking: if objective worsened significantly, reduce lr
        if obj_new > best_obj * 1.005:
            lr *= 0.7
            continue

        # Update momentum
        v = (a_new - a) / momentum if momentum > 0 else (a_new - a)
        a = a_new
        it += 1

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stuck_counter = 0
        else:
            stuck_counter += 1

            a = best_a.copy()
            v = np.zeros(n)
            lr = 1.0 / (2.0 * max(tau, 2.0))
            momentum = 0.8
            best_it = it
            stuck_counter = 0

        if verbose and it % 20000 == 0:
            print(f"    NSGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.8f}, t={elapsed:.0f}s")

def multiplicative_descent(a, time_budget=100.0, tau=50.0, eta=None, verbose=False):
    """
    Fast multiplicative mirror descent with entropy regularization.
    Much faster per iteration than projected gradient methods.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if eta is None:
        eta = 1.0 / max(n * 0.05, 10.0)

    it = 0
    best_it = 0
    stuck_counter = 0

    while time.monotonic() - start < time_budget:
        g, conv = compute_smoothed_gradient(a, tau)

        # Mirror descent: a_new ∝ a * exp(-eta * g)
        log_update = -eta * g
        log_update -= log_update.max()
        update = np.exp(log_update)
        a_new = a * update
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        a = a_new
        it += 1

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stuck_counter = 0
        else:
            stuck_counter += 1

        # Increase eta and tau when stuck
        if stuck_counter > 20000:
            eta *= 0.7
            tau *= 1.5
            a = best_a.copy()
            best_it = it
            stuck_counter = 0

        if verbose and it % 20000 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.6f}, tau={tau:.1f}, t={time.monotonic()-start:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, mode=0, rng=None):
    """Generate diverse structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x)
        return np.maximum(a, 0.01) / np.sum(a)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4 * np.pi * x)
        return np.maximum(a, 0.01) / np.sum(a)
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (x - 0.5) ** 2)
        return a / np.sum(a)
    elif mode == 4:
        if rng is None:
            rng = np.random.default_rng(0)
        a = rng.random(n) + 0.1
        return a / np.sum(a)
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 2 * x) * np.cos(2 * np.pi * x)
        return np.maximum(a, 0.01) / np.sum(a)
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.sin(2 * np.pi * 5 * x)
        return np.maximum(a, 0.01) / np.sum(a)
    else:
        return np.ones(n) / n


def lp_polish_step(a):
    """
    Single LP-based polishing step using cvxpy.
    Solves: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Returns the optimal g or None if solver fails.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(a)
    a_norm = a / np.sum(a)

    # Build convolution matrix efficiently
    # A @ g = conv(a_norm, g), shape (2n-1, n)
    size = 2 * n - 1
    A = np.zeros((size, n))
    for j in range(n):
        A[j:j + n, j] = a_norm[::-1][:n]
        # Actually, conv(a, g)[k] = sum_j a[j] * g[k-j]
        # So A[k, g_idx] = a[k - g_idx] if valid
    # Let me rebuild properly
    A = np.zeros((size, n))
    for k in range(size):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a_norm[idx]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val
        except Exception:
            continue

    return None


def lp_polish(a, time_budget=60.0, verbose=False):
    """Multiple LP polishing steps with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 5:
            break

        g_lp = lp_polish_step(a)
        if g_lp is None:
            break

        it += 1
        best_t = 0.0
        best_t_obj = best_obj

        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            a = best_a.copy()
            if verbose:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            if verbose:
                print(f"    LP it{it}: no improvement, stopping")
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining fast gradient methods, multi-length exploration,
    diversification, and LP polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: Fast multiplicative descent ===
    remaining = deadline - time.monotonic()
    p1 = min(remaining * 0.12, 120.0)
    if p1 > 15:
        print(f"\n=== Phase 1: Multiplicative Descent ({p1:.0f}s) ===")
        a1 = multiplicative_descent(best_seq, time_budget=p1, tau=30.0, verbose=False)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  P1: obj={best_score:.8f} **")
        else:
            print(f"  P1: obj={obj1:.8f}")

    # === Phase 2: Nesterov smoothed descent (main workhorse) ===
    remaining = deadline - time.monotonic()
    p2 = min(remaining * 0.30, 350.0)
    if p2 > 30:
        print(f"\n=== Phase 2: Nesterov Smoothed Descent ({p2:.0f}s) ===")
        a2 = nesterov_smoothed_descent(best_seq, time_budget=p2,
                                       tau_init=5.0, tau_end=1500.0, verbose=False)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  P2: obj={best_score:.8f} **")
        else:
            print(f"  P2: obj={obj2:.8f}")

    # === Phase 3: Second pass NSGD with different params ===
    remaining = deadline - time.monotonic()
    p3 = min(remaining * 0.12, 120.0)
    if p3 > 20:
        print(f"\n=== Phase 3: NSGD pass 2 ({p3:.0f}s) ===")
        a3 = nesterov_smoothed_descent(best_seq, time_budget=p3,
                                       tau_init=50.0, tau_end=3000.0, verbose=False)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  P3: obj={best_score:.8f} **")
        else:
            print(f"  P3: obj={obj3:.8f}")

    # === Phase 4: MD with high tau for fine structure ===
    remaining = deadline - time.monotonic()
    p4 = min(remaining * 0.08, 80.0)
    if p4 > 15:
        print(f"\n=== Phase 4: MD high-tau ({p4:.0f}s) ===")
        a4 = multiplicative_descent(best_seq, time_budget=p4, tau=300.0, verbose=False)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  P4: obj={best_score:.8f} **")
        else:
            print(f"  P4: obj={obj4:.8f}")

    # === Phase 5: Multi-n exploration ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_current = len(best_seq)

    n_candidates = sorted(set([
        n_current - 200, n_current - 100, n_current - 50, n_current - 20,
        n_current + 20, n_current + 50, n_current + 100, n_current + 200,
        600, 700, 900, 1000, 1100, 1200
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 2000]
    # Prioritize close to current
    n_candidates.sort(key=lambda x: abs(x - n_current))
    n_candidates = n_candidates[:6]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        s = a_try.sum()
        if s > 0:
            a_try = a_try / s
        else:
            a_try = np.ones(n_try) / n_try

        alloc = min(remaining * 0.1, 40.0)
        if alloc > 10:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc,
                                              tau_init=10.0, tau_end=500.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diversification restarts ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 6: Diversification ===")
    n_cur = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if r == 0:
            a_r = 0.85 * best_seq + 0.15 * np.ones(n_cur) / n_cur
            a_r = a_r / a_r.sum()
        elif r == 1:
            a_r = 0.9 * best_seq + 0.1 * structured_init(n_cur, 1, rng)
            a_r = a_r / a_r.sum()
        elif r == 2:
            a_r = 0.9 * best_seq + 0.1 * structured_init(n_cur, 4, rng)
            a_r = a_r / a_r.sum()
        elif r == 3:
            # Fourier perturbation
            pert = np.zeros(n_cur)
            for k in range(1, 5):
                freq = k * 2 * np.pi / n_cur
                amp = 0.04 * rng.uniform(0.5, 1.0) / k
                phase = rng.uniform(0, 2 * np.pi)
                pert += amp * np.cos(freq * np.arange(n_cur) + phase)
            a_r = best_seq * np.exp(pert)
            a_r = np.maximum(a_r, 0)
            s = a_r.sum()
            if s > 1e-10:
                a_r = a_r / s
            else:
                a_r = best_seq.copy()
        else:
            a_r = 0.7 * best_seq + 0.3 * structured_init(n_cur, 5, rng)
            a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.1, 55.0)
        if alloc > 15:
            a_r = nesterov_smoothed_descent(a_r, time_budget=alloc,
                                            tau_init=10.0, tau_end=800.0)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: LP Polish (if available and time permits) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 7: LP Polish ({remaining:.0f}s remaining) ===")
    if remaining > 30:
        lp_budget = min(remaining * 0.25, 80.0)
        try:
            a_lp = lp_polish(best_seq, time_budget=lp_budget, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP polish: obj={best_score:.8f} **")
            else:
                print(f"  LP polish: obj={obj_lp:.8f}")
        except Exception as e:
            print(f"  LP polish failed: {e}")

    # === Phase 8: Final fine-tuning ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 8: Final Fine-tuning ({remaining:.0f}s) ===")

    if remaining > 30:
        # High-tau NSGD for very fine adjustments
        alloc = min(remaining * 0.4, 100.0)
        a_f = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                        tau_init=200.0, tau_end=5000.0, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final NSGD: obj={best_score:.8f} **")
        else:
            print(f"  Final NSGD: obj={obj_f:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Very high tau MD
        alloc = min(remaining * 0.3, 60.0)
        a_f = multiplicative_descent(best_seq, time_budget=alloc, tau=1000.0, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final MD: obj={best_score:.8f} **")
        else:
            print(f"  Final MD: obj={obj_f:.8f}")

    # Final validation and return
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
