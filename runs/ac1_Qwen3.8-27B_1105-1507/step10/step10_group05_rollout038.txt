import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def subgradient_max_conv(a):
    """
    Compute a subgradient of max_k (a*a)[k] w.r.t. a.
    Returns (grad, active_indices).
    grad has the same length as a.
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    
    # Find active set (indices near the max)
    # Use a small tolerance to include near-optimal indices
    threshold = max_val * (1 - 1e-6)
    active = np.where(conv >= threshold)[0]
    
    # For each active index k, the gradient of (a*a)[k] w.r.t. a is:
    # grad[j] = 2*a[k-j] if 0 <= k-j < n, else 0
    # This is a shifted and reversed copy of a.
    
    total_grad = np.zeros(n)
    for k in active:
        # grad_k[j] = 2*a[k-j] for valid j
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            # idx = k - j, where j goes from j_start to j_end
            # So idx goes from k-j_end to k-j_start
            idx_start = k - j_end
            idx_end = k - j_start
            total_grad[j_start:j_end+1] += 2.0 * a[idx_start:idx_end+1]
    
    total_grad /= len(active)
    return total_grad, active, max_val

def smoothed_gradient_conv(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Uses FFT for efficiency.
    Returns (grad, conv, max_conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    # Softmax weights
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # Gradient: sum_k p_k * grad_k
    # grad_k[j] = 2*a[k-j] if 0<=k-j<n
    # Using FFT: grad = 2 * ifft(fft(p) * conj(fft(a)))[:n] (approximately)
    # Actually, let's think about this more carefully.
    # The gradient of sum_k p_k * (a*a)[k] w.r.t. a[j] is:
    # sum_k p_k * 2*a[k-j] (if 0<=k-j<n)
    # = 2 * sum_k p_k * a[k-j] (if valid)
    # Let's define b[k] = p_k for k=0..2n-2, padded to length 2n-1.
    # Then the gradient is 2 * (b * a)[j] where * is cross-correlation... 
    # Actually, sum_k p_k * a[k-j] is the cross-correlation of p and a evaluated at lag j.
    # In FFT terms: ifft(fft(p, N) * conj(fft(a, N)))[:n] gives the cross-correlation.
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    P = np.fft.fft(p, N)
    A = np.fft.fft(a, N)
    corr = np.fft.ifft(P * np.conj(A)).real
    grad = 2.0 * corr[:n]
    
    return grad, conv, max_conv

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def nesterov_smoothed_descent(a, time_budget=50.0, tau_init=50.0, tau_end=2000.0, verbose=False):
    """
    Nesterov accelerated smoothed gradient descent for minimizing max(conv(a,a))
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov parameters
    beta = 0.9  # momentum
    lr = 5.0 / np.sqrt(n)  # initial learning rate
    v = np.zeros(n)  # velocity
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Cosine annealing for tau (increase over time for sharper max approximation)
        cos_val = 0.5 * (1 + np.cos(np.pi * progress))
        tau = tau_init + (tau_end - tau_init) * (1 - cos_val)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Compute smoothed gradient
        g, conv, max_conv = smoothed_gradient_conv(y, tau)
        
        # Update velocity and position
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        
        # Decay learning rate
        lr *= 0.9999
        lr = max(lr, 1e-10)
        
        # Stagnation detection
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n) * 0.5
            best_it = it
        
        if verbose and it % 1000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}, t={elapsed:.1f}s")
    
    return best_a

def subgradient_descent(a, time_budget=30.0, verbose=False):
    """
    Subgradient descent with averaging for minimizing max(conv(a,a)).
    Uses direct subgradient computation (O(n) per step).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 10.0 / n  # initial learning rate
    avg_a = a.copy()
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        g, active, max_val = subgradient_max_conv(a)
        
        # Subgradient step
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        # Running average (for convergence guarantee)
        avg_a = (avg_a * it + a_new) / (it + 1)
        
        obj_new = compute_obj(a_new)
        obj_avg = compute_obj(avg_a)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        if obj_avg < best_obj - 1e-14:
            best_obj = obj_avg
            best_a = avg_a.copy()
            best_it = it
        
        a = a_new
        it += 1
        lr = 10.0 / (n * np.sqrt(it + 1))  # Nesterov-like decay
        
        if it - best_it > 10000:
            a = best_a.copy()
            avg_a = best_a.copy()
            best_it = it
            lr = 10.0 / n
        
        if verbose and it % 5000 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, lr={lr:.6f}, t={time.monotonic()-start:.1f}s")
    
    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns optimal g or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

def lp_iteration(a, time_budget=30.0, verbose=False):
    """LP-based iteration: solve LP for best response, blend with current."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Try different blend parameters
        best_t = 0.0
        best_t_obj = best_obj
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1-t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-12:
            optimal_t = best_t
            a_new = (1-optimal_t) * best_a + optimal_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_obj = compute_obj(a_new)
            best_a = a_new
            a = a_new
            it += 1
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={optimal_t:.2f}")
        else:
            # No improvement, try perturbation
            a = a + 0.01 * np.random.randn(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            it += 1
            if it > 30:
                break
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast FFT-based Nesterov accelerated smoothed gradient descent as the primary
method (O(n log n) per iteration), combined with periodic LP-based refinement,
subgradient averaging, symmetric optimization, and multi-length exploration.
Key innovation: the smoothed gradient of max-convolution is computed via a single
FFT-based convolution, enabling 100x+ more iterations than LP-based approaches.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def smoothed_gradient_conv(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*(a*a)[k])) using FFT.
    Returns (grad, conv, max_conv).
    g[j] = 2 * sum_m p[j+m] * a[m], computed via FFT convolution.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    a_rev = a[::-1]
    # g[j] = 2 * (p * a_rev)[j + n - 1]
    N = 1
    while N < 3 * n - 1:
        N <<= 1

    P = np.fft.fft(p, N)
    AR = np.fft.fft(a_rev, N)
    conv_pa = np.fft.ifft(P * AR).real[:3 * n - 2]

    grad = 2.0 * conv_pa[n - 1: 2 * n - 1]
    return grad, conv, max_conv


def subgradient_max_conv(a):
    """
    Compute a subgradient of max_k (a*a)[k] w.r.t. a.
    O(n * |active|) but typically |active| is small.
    Returns (grad, max_val).
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))

    # Find active set
    threshold = max_val * (1.0 - 1e-7) + 1e-15
    active = np.where(conv >= threshold)[0]

    total_grad = np.zeros(n)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            idx_start = k - j_end
            idx_end = k - j_start
            total_grad[j_start:j_end + 1] += 2.0 * a[idx_start:idx_end + 1]

    total_grad /= len(active)
    return total_grad, max_val


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_smoothed_descent(a, time_budget=50.0, tau_init=50.0, tau_end=5000.0, verbose=False, lr_scale=5.0):
    """
    Nesterov accelerated smoothed gradient descent for minimizing max(conv(a,a))
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = lr_scale / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Smoothstep for tau scheduling
        tau = tau_init + (tau_end - tau_init) * (progress * progress * (3 - 2 * progress))

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        g, conv, max_conv = smoothed_gradient_conv(y, tau)

        # Adaptive learning rate: scale by tau (higher tau -> smaller step)
        effective_lr = lr / max(1.0, tau / tau_init)

        # Update
        v = beta * v - effective_lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1

        a = a_new
        it += 1

        # Decay learning rate
        lr *= 0.99995
        lr = max(lr, 1e-10)

        # Stagnation: restart from best
        if stall_count > 3000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = lr_scale / np.sqrt(n) * 0.7
            stall_count = 0

        if verbose and it % 2000 == 0:
            print(f"    NSGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}, t={elapsed:.1f}s")

    return best_a


def subgradient_averaged_descent(a, time_budget=30.0, verbose=False):
    """
    Subgradient descent with Polyak-style step size and running average.
    Uses exact subgradient (O(n*|active|) per step).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Initial step size based on Lipschitz estimate
    L = 4.0 * n  # rough upper bound on Lipschitz constant of max-conv
    gamma = 1.0 / np.sqrt(L)
    avg_a = a.copy()
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        g, max_val = subgradient_max_conv(a)

        # Polyak step
        a_new = a - gamma * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        # Running average
        avg_a = (avg_a * it + a_new) / (it + 1)

        obj_new = compute_obj(a_new)
        obj_avg = compute_obj(avg_a)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        if obj_avg < best_obj - 1e-14:
            best_obj = obj_avg
            best_a = avg_a.copy()
            best_it = it

        a = a_new
        it += 1
        gamma = 1.0 / np.sqrt(L * (it + 1))  # O(1/sqrt(t)) decay

        if it - best_it > 20000:
            a = best_a.copy()
            avg_a = best_a.copy()
            best_it = it
            gamma = 1.0 / np.sqrt(L)

        if verbose and it % 10000 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, gamma={gamma:.6f}, t={time.monotonic()-start:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_refine(a, time_budget=30.0, verbose=False):
    """LP-based refinement: solve best-response LP, blend, repeat."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        t0 = time.monotonic()
        g_lp = solve_lp_minpeak(a)
        lp_time = time.monotonic() - t0
        if g_lp is None:
            break

        # Try blending
        found = False
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                found = True
                break

        it += 1
        if found:
            no_improve = 0
        else:
            no_improve += 1
            # Perturb to escape
            rng = np.random.default_rng(it * 12345)
            a = a + rng.normal(0, 0.02, n)
            a = np.maximum(a, 0)
            a = a / a.sum()

        if no_improve > 20:
            break

        if verbose and it % 5 == 0:
            print(f"    LP it{it} ({lp_time:.1f}s): obj={best_obj:.8f}")

    return best_a


def symmetric_optimize(a, time_budget=50.0, verbose=False):
    """Optimize within the symmetric subspace a[i] = a[n-1-i]."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Make symmetric
    a_sym = (a + a[::-1]) / 2.0
    a_sym = a_sym / a_sym.sum()

    # Optimize the first half freely, then mirror
    half = (n + 1) // 2
    a_first = a_sym[:half].copy()

    best_a = a_sym.copy()
    best_obj = compute_obj(a_sym)
    start = time.monotonic()

    beta = 0.85
    lr = 3.0 / np.sqrt(half)
    v_half = np.zeros(half)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        # Reconstruct full symmetric vector
        if n % 2 == 0:
            full = np.concatenate([a_first, a_first[::-1]])
        else:
            full = np.concatenate([a_first, a_first[-2::-1]])

        full = np.maximum(full, 0)
        fs = float(np.sum(full))
        if fs < 1e-10:
            break
        full = full / fs

        # Compute gradient
        tau = 100.0 + 100.0 * (it / max(1, time_budget * 10))
        grad_full, conv, max_conv = smoothed_gradient_conv(full, tau)

        # Project gradient onto symmetric subspace
        grad_half = (grad_full[:half] + grad_full[::-1][:half]) / 2.0
        if n % 2 == 0:
            grad_half = grad_full[:half].copy()
        else:
            grad_half = (grad_full[:half] + grad_full[n - 1:n - 1 - half:-1]) / 2.0

        # Nesterov update on half
        y_half = a_first + beta * v_half
        y_half = np.maximum(y_half, 0)
        ys = float(np.sum(y_half)) + (0.0 if n % 2 == 0 else y_half[-1])
        if ys < 1e-10:
            y_half = a_first.copy()
        else:
            # Scale to maintain proper sum
            pass

        v_half = beta * v_half - lr * grad_half
        a_first_new = y_half + v_half
        a_first_new = np.maximum(a_first_new, 0)

        # Reconstruct and normalize
        if n % 2 == 0:
            full_new = np.concatenate([a_first_new, a_first_new[::-1]])
        else:
            full_new = np.concatenate([a_first_new, a_first_new[-2::-1]])

        full_new = np.maximum(full_new, 0)
        fn = float(np.sum(full_new))
        if fn < 1e-10:
            a_first = a_first.copy()
            v_half = np.zeros(half)
        else:
            full_new = full_new / fn
            obj_new = compute_obj(full_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = full_new.copy()
                best_it = it
            a_first = a_first_new.copy()

        it += 1
        lr *= 0.9999
        lr = max(lr, 1e-10)

        if it - best_it > 5000:
            a_first = best_a[:half].copy()
            v_half = np.zeros(half)
            lr = 3.0 / np.sqrt(half) * 0.5
            best_it = it

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_initialization(n, method, rng):
    """Create various initializations."""
    if method == 'uniform':
        return np.ones(n) / n
    elif method == 'cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif method == 'gaussian':
        x = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (x - 0.5) ** 2)
        return a / a.sum()
    elif method == 'bimodal':
        x = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (x - 0.25) ** 2) + np.exp(-20.0 * (x - 0.75) ** 2)
        return a / a.sum()
    elif method == 'random':
        a = rng.random(n) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines fast FFT-based Nesterov smoothed gradient descent,
    subgradient averaging, LP refinement, symmetric optimization, and multi-length
    exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Initialize from parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_current) / n_current
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # === Phase 1: Primary Nesterov Smoothed GD ===
    print(f"\n=== Phase 1: Nesterov Smoothed GD (primary) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 350.0)
    if p1_budget > 30:
        a1 = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                       tau_init=30.0, tau_end=3000.0,
                                       verbose=True, lr_scale=5.0)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj1:.8f}")

    # === Phase 2: Subgradient with averaging ===
    print(f"\n=== Phase 2: Subgradient Averaged Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.08, 80.0)
    if p2_budget > 15:
        a2 = subgradient_averaged_descent(best_seq, time_budget=p2_budget, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj2:.8f}")

    # === Phase 3: LP Refinement ===
    print(f"\n=== Phase 3: LP Refinement ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 120.0)
    if p3_budget > 15:
        a3 = lp_refine(best_seq, time_budget=p3_budget, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj3:.8f}")

    # === Phase 4: Symmetric Optimization ===
    print(f"\n=== Phase 4: Symmetric Optimization ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 80.0)
    if p4_budget > 20:
        a4 = symmetric_optimize(best_seq, time_budget=p4_budget, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase 4: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4: obj={obj4:.8f}")

    # === Phase 5: Alternating GD + LP cycles ===
    print(f"\n=== Phase 5: Alternating Cycles ===")
    remaining = deadline - time.monotonic()
    cycles = 0
    while remaining > 80 and cycles < 6:
        cycles += 1
        # GD burst
        alloc_gd = min(remaining * 0.25, 60.0)
        if alloc_gd > 15:
            a5 = nesterov_smoothed_descent(best_seq, time_budget=alloc_gd,
                                           tau_init=200.0, tau_end=20000.0,
                                           lr_scale=3.0)
            obj5 = compute_obj(a5)
            if obj5 < best_score:
                best_score = obj5
                best_seq = a5.copy()
                print(f"  Cycle {cycles} GD: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        # LP step
        if remaining > 30:
            alloc_lp = min(remaining * 0.15, 30.0)
            if alloc_lp > 10:
                a5 = lp_refine(best_seq, time_budget=alloc_lp)
                obj5 = compute_obj(a5)
                if obj5 < best_score:
                    best_score = obj5
                    best_seq = a5.copy()
                    print(f"  Cycle {cycles} LP: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_candidates = set()
    for dn in [-100, -50, 50, 100, 200, 300, 500]:
        n_try = n_best + dn
        if 300 <= n_try <= 3000:
            n_candidates.add(n_try)
    for n_abs in [400, 600, 700, 900, 1000, 1200, 1500, 2000]:
        if abs(n_abs - n_best) > 50:
            n_candidates.add(n_abs)

    # Sort by distance to current best (try closest first)
    n_candidates = sorted(n_candidates, key=lambda x: abs(x - n_best))
    n_candidates = n_candidates[:6]
    print(f"  Candidates: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # Quick GD
        alloc = min(remaining * 0.12, 40.0)
        if alloc > 10:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc,
                                              tau_init=50.0, tau_end=5000.0,
                                              lr_scale=5.0)

        remaining = deadline - time.monotonic()
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Different initializations (diversification) ===
    print(f"\n=== Phase 7: Diversification ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    inits = ['cosine', 'gaussian', 'bimodal', 'random']
    for init_name in inits:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_init = make_initialization(n_cur, init_name, rng)
        # Blend with current best to keep some structure
        a_init = 0.5 * best_seq + 0.5 * a_init
        a_init = np.maximum(a_init, 0)
        a_init = a_init / a_init.sum()

        alloc = min(remaining * 0.12, 60.0)
        if alloc > 15:
            a7 = nesterov_smoothed_descent(a_init, time_budget=alloc,
                                           tau_init=50.0, tau_end=5000.0,
                                           lr_scale=5.0)
            obj7 = compute_obj(a7)
            if obj7 < best_score:
                best_score = obj7
                best_seq = a7.copy()
                print(f"  Init '{init_name}': obj={best_score:.8f} **")
            else:
                print(f"  Init '{init_name}': obj={obj7:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Long GD with high tau
    if remaining > 40:
        alloc = min(remaining * 0.35, 200.0)
        a_final = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                            tau_init=500.0, tau_end=50000.0,
                                            verbose=True, lr_scale=3.0)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    # Final LP
    if remaining > 25:
        alloc = min(remaining * 0.3, 80.0)
        a_final = lp_refine(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    # One more GD
    if remaining > 20:
        alloc = min(remaining * 0.4, 60.0)
        a_final = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                            tau_init=1000.0, tau_end=100000.0,
                                            lr_scale=2.0)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra-final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
