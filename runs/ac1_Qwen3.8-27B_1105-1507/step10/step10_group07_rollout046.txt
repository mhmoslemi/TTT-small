from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a, n):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_direction(a):
    """Solve LP to find best direction."""
    import cvxpy as cp
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a, n)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)

    return None, None

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a, n):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_value) or (None, None).
    """
    import cvxpy as cp
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a, n)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def compute_smooth_gradient(a, tau, top_k=None):
    """
    Compute gradient of sum_k p[k]*C[k] where p[k] ~ exp(tau*(C[k]-max(C))).
    Uses FFT for efficiency. Returns gradient vector of length n.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    if max_conv < 1e-15:
        return np.zeros(n), conv

    if top_k is not None and top_k < len(conv):
        # Zero out all but top_k peaks
        idx = np.argsort(conv)[::-1][:top_k]
        conv_soft = np.zeros_like(conv)
        conv_soft[idx] = conv[idx]
    else:
        conv_soft = conv

    exp_vals = np.exp(tau * (conv_soft - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    # Gradient: g[i] = 2 * sum_k p[k] * a[k-i] (circular with padding)
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]

    return g, conv

        else:
            stall += 1
            if stall >= 4:
                # Perturb to escape
                scale = 0.01 + 0.008 * stall
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_p = a.sum()
                if s_p > 1e-10:
                    a = a / s_p
                stall = 0
            if stall > 20:
                break

        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Spectral gradient + LP jump hybrid with multi-length exploration.
    Phase 1: Fast spectral gradient optimization on parent
    Phase 2: LP jump refinement
    Phase 3: Wide multi-length exploration
    Phase 4: Multi-restart with peak-aware perturbation
    Phase 5: Final polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Spectral Gradient Optimization ===
    print(f"\n=== Phase 1: Spectral Gradient Optimization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 350.0)
    if p1_budget > 30:
        a_spec = spectral_gradient_optimize(best_seq, time_budget=p1_budget,
                                            verbose=True, seed=42,
                                            tau_init=8.0, tau_end=400.0,
                                            lr_scale=3.0, momentum_init=0.75)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Phase1 Spectral: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 Spectral: obj={obj_spec:.8f}")

    # === Phase 2: LP Jump Refinement ===
    print(f"\n=== Phase 2: LP Jump Refinement ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 150.0)
    if p2_budget > 15:
        a_lp = lp_jump_optimize(best_seq, time_budget=p2_budget,
                                verbose=True, seed=77, max_iter=30)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f}")

    # === Phase 3: Multi-length Exploration ===
    print(f"\n=== Phase 3: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Generate diverse candidate lengths
    n_candidates = sorted(set([
        max(300, n_current - 300), max(400, n_current - 150),
        max(500, n_current - 50),
        n_current + 50, n_current + 150, n_current + 300, n_current + 500,
        500, 700, 900, 1000, 1100, 1200, 1400, 1600, 1800, 2000, 2400
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 3000 and abs(n - n_current) > 30]
    # Sort by proximity to current best (try nearby lengths first)
    n_candidates.sort(key=lambda x: abs(x - n_current))

    for n_try in n_candidates[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)

        # Quick spectral gradient optimization
        alloc1 = min(remaining * 0.15, 40.0)
        if alloc1 > 10:
            a_try = spectral_gradient_optimize(a_try, time_budget=alloc1,
                                               seed=100 + n_try,
                                               tau_init=5.0, tau_end=200.0,
                                               lr_scale=2.5)

        # Short LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.08, 20.0)
        if alloc2 > 5:
            a_try = lp_jump_optimize(a_try, time_budget=alloc2,
                                     seed=200 + n_try, max_iter=5)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Multi-restart with diverse strategies ===
    print(f"\n=== Phase 4: Multi-restart ===")
    remaining = deadline - time.monotonic()

    restart_configs = [
        {"scale": 0.01, "tau_init": 5.0, "tau_end": 300.0, "lr_scale": 2.0},
        {"scale": 0.02, "tau_init": 8.0, "tau_end": 400.0, "lr_scale": 3.0},
        {"scale": 0.035, "tau_init": 12.0, "tau_end": 500.0, "lr_scale": 4.0},
        {"scale": 0.06, "tau_init": 15.0, "tau_end": 600.0, "lr_scale": 5.0},
        {"scale": 0.10, "tau_init": 20.0, "tau_end": 800.0, "lr_scale": 5.0},
    ]

    for r, cfg in enumerate(restart_configs):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Peak-aware perturbation
        scale = cfg["scale"]
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_p = a_pert.sum()
        if s_p < 1e-10:
            continue
        a_pert = a_pert / s_p

        alloc = min(remaining * 0.22, 60.0)
        if alloc > 10:
            a_pert = spectral_gradient_optimize(
                a_pert, time_budget=alloc,
                seed=500 + r,
                tau_init=cfg["tau_init"], tau_end=cfg["tau_end"],
                lr_scale=cfg["lr_scale"], momentum_init=0.7)

            # Quick LP polish if time permits
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_pert = lp_jump_optimize(a_pert, time_budget=min(remaining * 0.15, 15.0),
                                          seed=600 + r, max_iter=3)

        obj_pert = compute_obj(a_pert)
        if obj_pert < best_score:
            best_score = obj_pert
            best_seq = a_pert.copy()
            print(f"  Restart {r} (scale={scale:.4f}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 5: Final Intensive Polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        # Final spectral gradient with tight parameters
        alloc = min(remaining * 0.5, 80.0)
        a_final = spectral_gradient_optimize(best_seq, time_budget=alloc,
                                             verbose=False, seed=999,
                                             tau_init=30.0, tau_end=600.0,
                                             lr_scale=2.0, momentum_init=0.8)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Spectral: obj={best_score:.8f} **")
        else:
            print(f"  Final Spectral: obj={obj_final:.8f}")

    # Final LP polish
    remaining = deadline - time.monotonic()
    if remaining > 12:
        alloc = min(remaining * 0.4, 30.0)
        a_final_lp = lp_jump_optimize(best_seq, time_budget=alloc,
                                      seed=2000, max_iter=5)
        obj_flp = compute_obj(a_final_lp)
        if obj_flp < best_score:
            best_score = obj_flp
            best_seq = a_final_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
