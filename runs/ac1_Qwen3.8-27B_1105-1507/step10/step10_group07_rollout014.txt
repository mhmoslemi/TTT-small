def compute_conv(a):

    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def pair_exchange_optimize(a, time_budget=50.0, verbose=False, seed=42):

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def gaussian_init(n, sigma_frac=0.3):
    """Create a Gaussian-shaped initial sequence."""
    x = np.linspace(0, 1, n)
    center = 0.5
    sigma = sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    s = a.sum()
    return a / s


def triangle_init(n):
    """Create a triangular initial sequence."""
    x = np.linspace(0, 1, n)
    a = np.maximum(0, 1 - 2 * np.abs(x - 0.5))
    s = a.sum()
    return a / s


def two_hump_init(n, frac=0.3):
    """Create a two-hump initial sequence."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - frac) / 0.1) ** 2) + np.exp(-0.5 * ((x - (1-frac)) / 0.1) ** 2)
    s = a.sum()
    return a / s

"""
Frequency-domain multiplicative update optimizer for minimum peak autocorrelation.
Uses FFT-based gradient computation with adaptive peak weighting (homotopy from L2 to L-infty),
multiplicative mirror descent preserving non-negativity, multi-start with diverse initializations,
pairwise exchange local search, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_grad_weighted(a, weights):
    """
    Compute gradient of sum_k weights[k] * (a*a)[k] w.r.t. a.
    grad[i] = 2 * sum_j a[j] * weights[i+j]
    Computed via FFT: 2 * (a * weights_flipped)[2n-2-i]
    """
    n = len(a)
    w_flipped = weights[::-1]
    N = 1
    while N < n + len(weights):
        N <<= 1
    A = fft(a, N)
    WF = fft(w_flipped, N)
    conv_full = ifft(A * WF).real
    indices = 2 * n - 2 - np.arange(n)
    return 2.0 * conv_full[indices]


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    Early iterations use low tau (L2-like), later use high tau (L-inf-like).
    """

    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

        progress = min(1.0, elapsed / time_budget)

        # Homotopy: tau increases over time
        tau = tau_start + (tau_end - tau_start) * (progress ** 1.2)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break

        # Peak weights
        weights = np.exp(tau * (conv / max_conv - 1.0))
        weights /= weights.sum()

        # Gradient
        grad = compute_grad_weighted(a, weights)
        grad_max = float(np.max(np.abs(grad)))
        if grad_max < 1e-20:
            break

        base_eta = 0.8 / (grad_max * np.sqrt(n))

        best_eta = 0.0
        best_new_obj = best_obj
        for mult in [0.3, 0.5, 0.7, 1.0, 1.4, 2.0, 3.0, 5.0]:

            if s_try < 1e-30:

            obj_try = compute_obj(a_try)
            if obj_try < best_new_obj - 1e-15:
                best_new_obj = obj_try
                best_eta = eta

        if best_eta > 0.0 and best_new_obj < best_obj - 1e-15:
            log_factor = -best_eta * grad
            log_factor = np.clip(log_factor, -50.0, 50.0)

            s_a = a.sum()

            best_obj = best_new_obj

        it += 1
        if verbose and it % 25 == 0:
            print(f"      MU it{it}: obj={best_obj:.8f}, tau={tau:.1f}, eta={best_eta:.4f}, t={elapsed:.0f}s")

    return best_a


def additive_mu_optimize(a, time_budget=80.0, verbose=False, seed=42):
    """
    Additive mirror descent with simplex projection.
    Alternative to multiplicative: a_new = project_simplex(a - eta * grad)
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0
    lr = 2.0 / np.sqrt(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = 5.0 + 80.0 * (progress ** 1.5)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break

        weights = np.exp(tau * (conv / max_conv - 1.0))
        weights /= weights.sum()

        grad = compute_grad_weighted(a, weights)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-20:
            break

        # Try a few step sizes
        best_eta = 0.0
        best_new_obj = best_obj
        base_eta = lr / (grad_norm + 1e-10)
        for mult in [0.5, 1.0, 1.5, 2.0, 3.0]:
            eta = base_eta * mult
            a_try = a - eta * grad
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_new_obj - 1e-15:
                best_new_obj = obj_try
                best_eta = eta

        if best_eta > 0.0 and best_new_obj < best_obj - 1e-15:
            a = a - best_eta * grad
            a = project_simplex(a)
            best_obj = best_new_obj
            best_a = a.copy()
            no_improve = 0
            lr *= 1.01
        else:
            no_improve += 1
            lr *= 0.9
            if no_improve >= 30:
                scale = 0.01 + 0.005 * min(no_improve, 8)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                no_improve = 0
                if it > 200:
                    break

        it += 1
        if verbose and it % 30 == 0:
            print(f"      AMU it{it}: obj={best_obj:.8f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def coordinated_descent(a, time_budget=60.0, seed=42):
    """
    Coordinated descent: iterate over positions, find optimal value for each
    given all others fixed.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - start < time_budget:
        improved = False
        # Random order each sweep
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            # Try a few values for a[i]
            # Current contribution to autocorrelation from position i:
            # Changing a[i] to v changes autocorr[k] by (v - a[i]) * (sum of relevant a[k-i] terms)
            # For simplicity, try geometric spacing around current value
            current = a[i]
            if current < 1e-15:
                # Try giving it some mass
                candidates = [0.001 / n, 0.005 / n, 0.01 / n]
            else:
                candidates = [current * f for f in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0]]

            best_cand = current
            best_cand_obj = compute_obj(a)  # current

            for c in candidates:
                if c < 0 or c > 1.0:
                    continue
                a[i] = c
                s_new = a.sum()
                if s_new < 1e-10:
                    a[i] = current
                    continue
                a_norm = a / s_new
                obj_c = compute_obj(a_norm)
                if obj_c < best_cand_obj - 1e-14:
                    best_cand = c
                    best_cand_obj = obj_c

            a[i] = best_cand
            if best_cand != current:
                improved = True

        s_a = a.sum()
        if s_a > 1e-10:
            a = a / s_a
        obj_now = compute_obj(a)
        if obj_now < best_obj - 1e-14:
            best_obj = obj_now
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                break

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def make_init(n, kind='uniform', **kw):
    """Create various initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'gaussian':
        sigma = kw.get('sigma', 0.3)
        x = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        return a / a.sum()
    elif kind == 'triangle':
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - 2 * np.abs(x - 0.5))
        return a / a.sum()
    elif kind == 'two_hump':
        frac = kw.get('frac', 0.3)
        x = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((x - frac) / 0.08) ** 2) + np.exp(-0.5 * ((x - (1 - frac)) / 0.08) ** 2)
        return a / a.sum()
    elif kind == 'v_shape':
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * x - 1)
        return a / a.sum()
    elif kind == 'ramp':
        a = np.linspace(0.05, 1.0, n)
        return a / a.sum()
    elif kind == 'w_shape':
        x = np.linspace(0, 1, n)
        a = 1.0 - 0.5 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: FFT-based multiplicative/additive mirror descent with homotopy peak weighting,
    coordinated descent, multi-start, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={n_parent}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_parent}): obj={best_score:.8f}")

    # === Phase 1: MU on parent ===
    print(f"\n=== Phase 1: MU on parent (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.18, 180.0)
    if p1_budget > 20:
        a_mu = mu_optimize(best_seq, time_budget=p1_budget, verbose=True, seed=42)
        obj_mu = compute_obj(a_mu)
        if obj_mu < best_score:
            best_score = obj_mu
            best_seq = a_mu.copy()
            print(f"  MU: obj={best_score:.8f} **")
        else:
            print(f"  MU: obj={obj_mu:.8f}")

    # === Phase 2: Additive MU ===
    print(f"\n=== Phase 2: Additive Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 100.0)
    if p2_budget > 15:
        a_amu = additive_mu_optimize(best_seq, time_budget=p2_budget, seed=77)
        obj_amu = compute_obj(a_amu)
        if obj_amu < best_score:
            best_score = obj_amu
            best_seq = a_amu.copy()
            print(f"  AMU: obj={best_score:.8f} **")
        else:
            print(f"  AMU: obj={obj_amu:.8f}")

    # === Phase 3: Coordinated descent ===
    print(f"\n=== Phase 3: Coordinated Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.08, 60.0)
    if p3_budget > 10:
        a_cd = coordinated_descent(best_seq, time_budget=p3_budget, seed=123)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 4: Multi-start with diverse inits ===
    print(f"\n=== Phase 4: Multi-Start ===")
    remaining = deadline - time.monotonic()

    init_configs = [
        ('gauss_0.15', 'gaussian', {'sigma': 0.15}),
        ('gauss_0.25', 'gaussian', {'sigma': 0.25}),
        ('gauss_0.35', 'gaussian', {'sigma': 0.35}),
        ('gauss_0.5', 'gaussian', {'sigma': 0.5}),
        ('triangle', 'triangle', {}),
        ('two_hump_0.25', 'two_hump', {'frac': 0.25}),
        ('two_hump_0.35', 'two_hump', {'frac': 0.35}),
        ('v_shape', 'v_shape', {}),
        ('ramp', 'ramp', {}),
        ('w_shape', 'w_shape', {}),
    ]

    # Sort by initial objective
    scored_inits = []
    for name, kind, kw in init_configs:
        a_init = make_init(n_parent, kind, **kw)
        a_init = np.maximum(a_init, 0)
        s_init = a_init.sum()
        if s_init > 0.01:
            a_init = a_init / s_init
            scored_inits.append((compute_obj(a_init), name, a_init))
    scored_inits.sort(key=lambda x: x[0])

    for idx in range(min(5, len(scored_inits))):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        obj_init, name, a_init = scored_inits[idx]
        alloc = min(remaining * 0.12, 70.0)
        if alloc < 15:
            break

        print(f"  Init '{name}': obj={obj_init:.8f}, alloc={alloc:.0f}s")
        a_opt = mu_optimize(a_init, time_budget=alloc * 0.75, seed=200 + idx * 13)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"    ** {name}: obj={best_score:.8f}")
        else:
            print(f"    {name}: obj={obj_opt:.8f}")

    # === Phase 5: Multi-length ===
    print(f"\n=== Phase 5: Multi-Length ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(300, n_current - 200), max(300, n_current - 100),
        n_current + 100, n_current + 200, n_current + 400, n_current + 600,
        500, 700, 900, 1100, 1300, 1500, 1800, 2000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500 and abs(n - n_current) > 50]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 45 or n_tried >= 5:
            break

        alloc = min(remaining * 0.14, 70.0)
        if alloc < 20:
            break

        print(f"  n={n_try} (alloc={alloc:.0f}s)")
        a_try = resample(best_seq, n_try)
        a_opt = mu_optimize(a_try, time_budget=alloc * 0.8, seed=500 + n_try % 100)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")
        n_tried += 1

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 120.0)
        a_final = mu_optimize(best_seq, time_budget=alloc * 0.6, verbose=True, seed=999,
                              tau_start=30.0, tau_end=200.0)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final MU (high tau): obj={best_score:.8f} **")
        else:
            print(f"  Final MU (high tau): obj={obj_final:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.35, 80.0)
        a_final2 = additive_mu_optimize(best_seq, time_budget=alloc, seed=1999)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final AMU: obj={best_score:.8f} **")
        else:
            print(f"  Final AMU: obj={obj_final2:.8f}")

    # === Phase 7: Perturbation + reoptimize ===
    print(f"\n=== Phase 7: Perturbation Restarts ===")
    remaining = deadline - time.monotonic()
    for r_idx in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        scale = 0.004 + 0.004 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            break
        a_pert = a_pert / s_pert

        alloc = min(remaining * 0.2, 50.0)
        if alloc < 12:
            break

        a_opt = mu_optimize(a_pert, time_budget=alloc, seed=3000 + r_idx * 37)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Pert{r_idx} (s={scale:.4f}): obj={best_score:.8f} **")
        else:
            print(f"  Pert{r_idx} (s={scale:.4f}): obj={obj_opt:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
