"""
Optimized step function coefficient search using LP-guided direction with optimal line search,
mirror-pair coordinate descent for asymmetry creation, and multi-length exploration.
Key improvements over previous: (1) proper 1D line search for LP blending, (2) mirror-pair 
2D updates that directly control asymmetry, (3) more aggressive multi-n search.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a):
    """Build sparse convolution matrix: A[k, j] such that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            idx = k - j  # index into a
            if idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[idx])
    A = cp.SparseMatrix(rows, cols, vals, shape=(2 * n - 1, n))
    return A


def solve_lp_direction(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Return (g, M)."""
    n = len(a)
    A = build_conv_matrix_sparse(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g, max_iter=30):
    """Find optimal t in [0,1] minimizing obj((1-t)*a + t*g)."""
    best_t = 0.0
    best_obj = compute_obj_normalized(a)
    
    # Golden section search on [0, 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    
    def obj_at_t(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0)
        s = float(np.sum(a_t))
        if s < 1e-10:
            return 1e10
        a_t = a_t / s
        return compute_obj_normalized(a_t)
    
    fc = obj_at_t(c)
    fd = obj_at_t(d)
    
    for _ in range(max_iter):
        if hi - lo < 1e-8:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at_t(d)
    
    best_t = (lo + hi) / 2.0
    best_obj = obj_at_t(best_t)
    return best_t, best_obj

    """LP-guided descent with optimal line search."""

            break

            a_new = (1.0 - t_opt) * a + t_opt * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_obj = obj_at_t
            best_a = a_new.copy()
            a = a_new
            stall = 0

                    stall = 0

            print(f"    Mirror CD sweep {sweep}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
        
        order = rng.permutation(n_pairs)
        
        for p in order:
            i = p
            j = n - 1 - p

                lo = 0.0

                c = hi - gr * (hi - lo)
                d = lo + gr * (hi - lo)
                
                def obj_1d(x):
                    a[i] = x
                    return compute_obj_normalized(a)
                
                # Restore

                fc = obj_1d(c)
                a[i] = v
                fd = obj_1d(d)
                a[i] = v

                        break

                        d = c
                        fd = fc

                        fc = obj_1d(c)
                        a[i] = v
                    else:
                        lo = c
                        c = d
                        fc = fd
                        d = lo + gr * (hi - lo)
                        a[i] = d
                        fd = obj_1d(d)

def resample_sequence(a, new_n):
