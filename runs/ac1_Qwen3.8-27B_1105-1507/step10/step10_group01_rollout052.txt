    n = len(a)

    diff = k_idx - j_idx

    A[valid] = a_norm[diff[valid]]
    return A

    a_norm = a / np.sum(a)
    
    # Build constraint matrix
    # Variables: [g_0, ..., g_{n-1}, M]
    # Constraints: A@g - M <= 0 (2n-1 inequalities)
    #              sum(g) = 1 (1 equality)
    #              g >= 0, M free
    
    A_ineq = np.zeros((2*n-1, n+1))

    A_eq[0, :n] = 1.0
    A_eq[0, n] = 0.0
    b_eq = [1.0]
    
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize M

        g = np.maximum(g, 0)
        s = float(np.sum(g))
        if s > 1e-10:

    return None, None

    new_a = np.exp(new_log)
    new_a = new_a / new_a.sum()

def mirror_descent_optimize(a, time_budget, tau_init=30.0, tau_end=500.0, 
                            lr_init=0.1, verbose=False):
    """Mirror descent (exponential weighting) on the simplex with adapted tau."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    no_improve = 0
    lr = lr_init
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init * (tau_end / tau_init) ** progress
        
        g, conv, max_conv = smoothed_gradient(a, tau)
        g_norm = float(np.linalg.norm(g))
        
        # Adaptive learning rate
        if g_norm > 1e-15:
            lr = lr * 0.9995 + 0.0005 * (2.0 / (g_norm * np.sqrt(n)))
            lr = min(max(lr, 1e-8), 1.0)
        
        # Mirror descent step: a_new = a * exp(-lr * g) / normalization
        log_a = np.log(np.maximum(a, 1e-300))
        new_log = log_a - lr * g
        new_log -= new_log.max()
        a_new = np.exp(new_log)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 2000:
            # Perturb
            a = best_a + 0.01 * np.random.randn(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            lr = lr_init
            no_improve = 0
        
        a = a_new
        it += 1
        
        if verbose and it % 500 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.4f}")
    
    if verbose:
        print(f"    MD: obj={best_obj:.8f}, iters={it}")
    return best_a
