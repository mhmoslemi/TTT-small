"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses asymmetric spectral shaping, aggressive smoothed GD, LP fixed-point iteration,
coordinate descent on active peaks, and multi-n exploration.
Key insight: breaking palindromic symmetry reduces the central convolution peak.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad_full(a, tau):
    """
    Compute smoothed gradient of max(conv(a,a)).
    Returns gradient of length n.
    """
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = conv - exact_max
    max_exp = 0.0
    for v in shifted:
        if v > max_exp:
            max_exp = v
    exp_vals = np.exp(tau * (shifted - max_exp))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def single_lag_grad(a, lag):
    """
    Compute gradient of (a*a)[lag] with respect to a.
    (a*a)[lag] = sum_j a[j] * a[lag-j]
    d/d a[i] = a[lag-i] + a[i-lag] (where valid indices)
    """
    n = len(a)
    g = np.zeros(n)
    for i in range(n):
        val = 0.0
        j1 = lag - i
        if 0 <= j1 < n:
            val += a[j1]
        j2 = i - lag
        if 0 <= j2 < n:
            val += a[j2]
        g[i] = val
    return g


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def asymmetric_init(a, alpha=0.3):
    """
    Create an asymmetric variant of a by blending with a ramp.
    This breaks palindromic symmetry to reduce central convolution peak.
    """
    n = len(a)
    a = a / a.sum()
    # Create a ramp that's high on the left, low on the right
    ramp = np.linspace(1.0 + alpha, 1.0 - alpha, n)
    mixed = (1.0 - 0.5) * a + 0.5 * ramp / ramp.sum()
    mixed = np.maximum(mixed, 0)
    s = mixed.sum()
    if s > 1e-10:
        mixed = mixed / s
    return mixed


def staircase_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9, lr_scale=2.0):
    """
    Nesterov accelerated gradient descent with staircase tau schedule.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 15, 40, 100, 250, 600, 1200, 2500, 5000]

    it = 0
    best_it = 0
    stall_patience = 8000

    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0
        max_iters_per_level = 6000

        while time.monotonic() - start < time_budget:
            if tau_iters >= max_iters_per_level:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad_full(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it
            a = a_new
            it += 1
            tau_iters += 1
            lr = max(1e-8, lr * 0.99999)

            if it - best_it > stall_patience:
                a = best_a.copy()
                v = np.zeros(n)
                lr = lr_scale / np.sqrt(n)
                best_it = it

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    tau={tau:5d}: obj={best_obj:.10f}, iters={tau_iters}, t={elapsed:.0f}s")

    return best_a


def spectral_flatten(a, n_iters=100, time_budget=30.0, verbose=False):
    """
    Iteratively flatten the power spectrum of a to reduce autocorrelation peaks.
    Keeps the sequence non-negative and sum-normalized.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, size)
        power = np.abs(A) ** 2
        total_power = float(np.sum(power))
        target = total_power / size

        # For each frequency, scale toward flat (but keep DC and its conjugate)
        for k in range(1, size // 2 + 1):
            if k == 0:
                continue
            p = power[k]
            if p > 1e-30:
                ratio = np.sqrt(target / p)
                # Gentle update: blend toward flat
                blend = 0.3
                A[k] *= (1.0 - blend + blend * ratio)
                A[size - k] = np.conj(A[k])

        # Keep DC fixed
        A[0] = float(np.sum(a))

        a_new = ifft(A).real[:n]
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            a_new = best_a.copy()
            continue
        a_new = a_new / sn

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

        if verbose and it % 20 == 0:
            print(f"    spectral it{it}: obj={best_obj:.10f}")

    return best_a


def peak_targeting_gd(a, time_budget, n_active=5, verbose=False):
    """
    GD that targets only the top few peaks in the convolution.
    This gives harder gradients focused on the worst peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.0 / np.sqrt(n)
    beta = 0.85
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Find top n_active peaks
        top_lags = np.argsort(conv)[::-1][:n_active]
        peak_val = conv[top_lags[0]]

        # Gradient = sum of gradients for active peaks
        g = np.zeros(n)
        for lag in top_lags:
            g += single_lag_grad(a, int(lag))

        # Normalize gradient
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break
        g = g / gn * peak_val

        # Nesterov step
        y = a + beta * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            break
        y = y / sy

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / sn

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            it = 0
        else:
            it += 1
            if it > 3000:
                break

        a = a_new
        lr = max(1e-7, lr * 0.9999)

    return best_a


def lp_polish_iteration(a):
    """
    Single LP polishing step: solve min M s.t. conv(a,g)<=M, g>=0, sum(g)=1.
    """
    n = len(a)
    if n > 3000:
        return None, None

    # Build convolution matrix: (a*g)[k] = sum_j a[j]*g[k-j]
    # A[k,j] = a[j] if 0 <= k-j < n, else 0
    # Actually A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    # Let's use: (a*g)[k] = sum_{j=0}^{min(n-1,k)} a[k-j] * g[j] for k=0..2n-2
    # So A[k,j] = a[k-j] if 0 <= k-j <= n-1, else 0

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # For each j (column), A[k,j] is nonzero when 0 <= k-j < n, i.e., j <= k <= j+n-1
        A[j:j + n, j] = a[:n]

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK, cp.CBC]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue

    return None, None


def lp_polish(a, time_budget, verbose=False, seed=42, max_iters=30):
    """
    Iterative LP polishing with adaptive blending and perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    last_t = 0.4
    stall = 0

    for _ in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = lp_polish_iteration(a)
        if g_lp is None:
            break

        # Try multiple blending factors
        t_candidates = sorted(set([
            0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7,
            0.8, 0.9, 1.0,
            max(0.02, last_t * 0.5), min(1.0, last_t * 1.5),
            max(0.02, last_t * 0.7), min(1.0, last_t * 1.3),
        ]))

        best_t_obj = best_obj
        best_t_val = last_t
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose:
                print(f"    LP it: obj={best_obj:.10f}, t={best_t_val:.3f}")
        else:
            stall += 1
            # Fourier perturbation to escape
            freqs = rng.integers(1, 8, size=4)
            phases = rng.uniform(0, 2 * np.pi, size=4)
            x = np.arange(n)
            pert = np.zeros(n)
            for f, p in zip(freqs, phases):
                pert += 0.015 * np.sin(2.0 * np.pi * f * x / n + p)
            a = a + pert
            a = np.maximum(a, 0)
            a = a / a.sum()
            if stall > 8:
                break

    return best_a


def make_init(n, mode, rng=None):
    """Create diverse initial sequences of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.0001))
        return a / a.sum()
    elif mode == 3:
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 4:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 5:
        # Anti-symmetric ramp (breaks symmetry)
        a = np.linspace(0.3, 1.0, n)
        return a / a.sum()
    elif mode == 6:
        # Bi-modal asymmetric
        c1, c2 = n // 3, 2 * n // 3
        w = max(1, n // 10)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += 0.7 * np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 7:
        # Asymmetric: heavy on left
        a = np.exp(-np.arange(n) / (n / 3.0))
        return a / a.sum()
    elif mode == 8:
        # Multi-scale: combine different frequencies
        x = np.arange(n) / n
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x + 0.5) + 0.2 * np.sin(2 * np.pi * 7 * x + 1.2)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 9:
        # Quantized random
        a = rng.integers(0, 5, n).astype(float)
        a = np.maximum(a, 0.1)
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def coordinate_descent_peaks(a, time_budget, verbose=False):
    """
    Alternating: pick the worst lag, compute optimal single-coordinate update
    to reduce that lag's contribution.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_lag = int(np.argmax(conv))
        max_val = float(conv[max_lag])

        # Find the coordinate that contributes most to conv[max_lag]
        # conv[max_lag] = sum_j a[j] * a[max_lag - j]
        contributions = np.zeros(n)
        for j in range(n):
            j2 = max_lag - j
            if 0 <= j2 < n:
                contributions[j] = a[j] * a[j2]
            else:
                contributions[j] = 0.0

        # Pick the coordinate with the largest contribution
        worst_j = int(np.argmax(contributions))

        # Try reducing a[worst_j] and distributing to a random "good" coordinate
        # Find a coordinate that doesn't contribute to any of the top peaks
        top_lags = np.argsort(conv)[::-1][:10]
        good_candidates = []
        for j in range(n):
            contrib_total = 0.0
            for lag in top_lags:
                j2 = int(lag) - j
                if 0 <= j2 < n:
                    contrib_total += a[j] * a[j2]
                j2b = j - int(lag)
                if 0 <= j2b < n:
                    contrib_total += a[j] * a[j2b]
            if contrib_total < 0.1 * max_val / n:
                good_candidates.append(j)

        if len(good_candidates) < 5:
            good_candidates = list(range(n))

        delta = a[worst_j] * 0.1
        good_j = good_candidates[rng.integers(0, len(good_candidates))]

        a_try = a.copy()
        a_try[worst_j] -= delta
        a_try[good_j] += delta
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st

        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
        else:
            # Try smaller step
            for frac in [0.05, 0.02, 0.01]:
                delta2 = a[worst_j] * frac
                a_try2 = a.copy()
                a_try2[worst_j] -= delta2
                a_try2[good_j] += delta2
                a_try2 = np.maximum(a_try2, 0)
                st2 = float(np.sum(a_try2))
                if st2 < 1e-10:
                    continue
                a_try2 = a_try2 / st2
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj - 1e-12:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    a = a_try2
                    break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining asymmetric spectral shaping,
    aggressive smoothed GD, LP fixed-point polishing, peak targeting,
    coordinate descent, multi-n exploration, and diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        parent = None

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.10f}")

    n_current = len(best_seq)
    print(f"Starting optimization with n={n_current}, obj={best_score:.10f}")

    # === Phase 1: Diverse initialization and screening (10% time) ===
    print(f"\n=== Phase 1: Diverse Init Screening ===")
    p1_budget = (deadline - start_time) * 0.10
    p1_deadline = start_time + p1_budget

    init_modes = [0, 1, 2, 5, 6, 7, 8, 9]
    for mode in init_modes:

        else:
            print(f"  Init mode {mode}: obj={obj_screen:.10f}")

                                  tau_levels=[10, 50, 200], verbose=False)

            if obj_asym < best_score:

    # === Phase 2: Aggressive smoothed GD (20% time) ===
    print(f"\n=== Phase 2: Aggressive Smoothed GD ===")
    p2_budget = min((deadline - start_time) * 0.20, 280.0)
