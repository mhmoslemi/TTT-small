    n = len(a)
    # p has length 2n-1
    # grad_i = 2 * sum_j a_j * p[i+j] - a_i * p[2i]

    # q_i = <a, p[i:i+n]>

    a_rev = a[::-1]
    full_conv = np.convolve(a_rev, p, mode='full')
    # full_conv has length n + 2n-1 - 1 = 3n-2
    # q_i corresponds to full_conv[i + n - 1] for i = 0, 1, ..., 2n-2

    valid = diag_idx < 2*n-1
    grad[valid] -= a[valid] * p[diag_idx[valid]]
    
    return grad

"""
Soft-max gradient descent with Nesterov momentum for step function optimization,
combined with LP polishing for final refinement.
Key innovation: smooth approximation of the max-convolution objective via log-sum-exp,
enabling fast O(n log n) gradient computations via FFT cross-correlation.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes sum(a)=1 for speed."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_soft_gradient(a, p):
    """
    Compute gradient of soft-max objective.
    grad_i = 2 * sum_j a_j * p_{i+j}
    Uses FFT cross-correlation for O(n log n) efficiency.
    """
    n = len(a)
    p_len = len(p)  # should be 2n-1
    a_rev = a[::-1]
    size = 1
    while size < n + p_len - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]
    return 2.0 * q


def soft_max_gradient_descent(a, time_budget=200.0, beta_start=10.0, beta_end=50.0,
                              lr_start=0.008, momentum=0.9, verbose=True, seed=42):
    """
    Soft-max gradient descent with momentum on the simplex.
    Minimizes (1/beta) * log(sum_k exp(beta * conv(a,a)_k)) where sum(a)=1.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)  # momentum velocity
    lr = lr_start
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        # Anneal beta
        beta = beta_start + (beta_end - beta_start) * progress
        # Decay learning rate gently
        lr = lr_start * (1.0 - 0.3 * progress)
        
        # Compute convolution and softmax
        conv = compute_conv(a)
        # For numerical stability, subtract max
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Compute gradient
        grad = compute_soft_gradient(a, p)
        
        # Momentum update
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        # Evaluate
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Reseed momentum
                v = np.zeros(n)
                stall = 0
                # Small perturbation
                pert = rng.normal(0, 0.002, n)
                a = project_simplex(best_a + pert)
        
        a = a_new
        it += 1
        
        if verbose and it % 500 == 0:
            print(f"    SmaxGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, "
                  f"lr={lr:.5f}, t={elapsed:.1f}s")
        
        # Early stopping if converging
        if it - best_it > 5000 and it > 1000:
            a = best_a.copy()
            v = np.zeros(n)
            it = best_it
            lr *= 0.7
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g_lp, time_budget=3.0):
    """Golden section search for optimal blend factor."""
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    lo, hi = 0.0, 1.5
    
    def obj_at(t):
        if t < 0:
            t = 0
        a_try = (1 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)
    
    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)
    
    start = time.monotonic()
    for _ in range(20):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)
    
    t_best = c if fc < fd else d
    obj_best = min(fc, fd)
    
    for t_try in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.2]:
        obj_try = obj_at(t_try)
        if obj_try < obj_best:
            obj_best = obj_try
            t_best = t_try
    
    return t_best, obj_best


def lp_polish(a, time_budget=120.0, max_iters=60, verbose=False):
    """LP-based polishing: iteratively find best direction and blend."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=min(2.0, remaining * 0.05))
        
        if t_opt > 0 and obj_opt < best_obj - 1e-12:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = project_simplex(np.maximum(a_new, 0))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"elapsed={time.monotonic()-start:.1f}s")
        
        if stall >= 4:
            # Perturb and restart

                noise = rng.normal(0, 0.005 * (stall - 3), n)
                a = project_simplex(best_a + noise)
            else:

                a_div = 1.0 / (0.05 + 4.0 * (2 * x - 1) ** 2)
                a = project_simplex(0.5 * a_div + 0.5 * best_a)
            stall = 0
    
    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.01, 0.99, n)
    a = 1.0 / np.sqrt(1 - (2 * x - 1) ** 2)

            print(f"  Phase2: obj={obj2:.8f} (no improvement)")
    
    # === Phase 3: LP Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 3: LP Polish, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.20, 200.0)
        a3, obj3 = lp_polish(best_seq, time_budget=alloc, max_iters=50, verbose=True)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj3:.8f} (no improvement)")
    
    # === Phase 4: Multi-restart with diverse initializations ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 4: Diverse Restarts, time={remaining:.0f}s ===")
        n_cur = len(best_seq)
        
        for r in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            
            mode = r % 8
            a_r = diverse_init(n_cur, rng, mode=mode)
            # Blend with current best
            blend = 0.6
            a_r = project_simplex((1 - blend) * a_r + blend * best_seq)
            
            # Quick soft-max GD
            alloc_sgd = min(remaining * 0.25, 80.0)
            if alloc_sgd > 10:
                a_r, obj_r = soft_max_gradient_descent(
                    a_r, time_budget=alloc_sgd,
                    beta_start=10.0, beta_end=40.0,
                    lr_start=0.008, momentum=0.9,
                    verbose=False, seed=seed + 100 + r
                )
            
            # Quick LP
            remaining = deadline - time.monotonic()
            if remaining > 30:
                alloc_lp = min(remaining * 0.2, 50.0)
                a_r, obj_r = lp_polish(a_r, time_budget=alloc_lp, max_iters=20, verbose=False)
            
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")
    
    # === Phase 5: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 5: Length Exploration, time={remaining:.0f}s ===")
        n_best = len(best_seq)
        
        n_options = sorted(set([
            max(400, n_best - 150), max(400, n_best - 75),
            n_best + 75, n_best + 150, n_best + 300,
            600, 750, 900, 1000, 1100, 1200, 1400
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) >= 50]
        n_options = n_options[:4]
        print(f"  Trying n values: {n_options}")
        
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            
            # Interpolate to new length
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = project_simplex(np.maximum(a_try, 0))
            
            # Optimize at new length
            alloc_sgd = min(remaining * 0.4, 60.0)
            if alloc_sgd > 15:
                a_try, obj_try = soft_max_gradient_descent(
                    a_try, time_budget=alloc_sgd,
                    beta_start=10.0, beta_end=40.0,
                    lr_start=0.008, momentum=0.9,
                    verbose=False, seed=seed + 200
                )
            
            remaining = deadline - time.monotonic()
            if remaining > 30:
                alloc_lp = min(remaining * 0.3, 40.0)
                a_try, obj_try = lp_polish(a_try, time_budget=alloc_lp, max_iters=20, verbose=False)
            
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: Final Polish, time={remaining:.0f}s ===")
        # One more soft-max GD round with high beta
        alloc1 = min(remaining * 0.4, 100.0)
        if alloc1 > 20:
            a_f, obj_f = soft_max_gradient_descent(
                best_seq, time_budget=alloc1,
                beta_start=30.0, beta_end=80.0,
                lr_start=0.003, momentum=0.93,
                verbose=True, seed=seed + 300
            )
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final SG: obj={best_score:.8f} **")
        
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc2 = min(remaining * 0.5, 80.0)
            a_f2, obj_f2 = lp_polish(best_seq, time_budget=alloc2, max_iters=30, verbose=True)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final LP: obj={best_score:.8f} **")
    
    # === Phase 7: Last-ditch efforts ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        # Try a few random perturbations + quick LP
        for attempt in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            noise = rng.normal(0, 0.003, len(best_seq))
            a_p = project_simplex(best_seq + noise)
            alloc = min(remaining * 0.4, 20.0)
            if alloc > 5:
                a_p, obj_p = lp_polish(a_p, time_budget=alloc, max_iters=10, verbose=False)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Last-ditch {attempt}: obj={best_score:.8f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
