"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based smoothed gradient descent with Nesterov momentum, adaptive learning rate,
tau annealing, asymmetric perturbations, and multi-length exploration.
Key innovations: efficient O(n log n) gradient via FFT cross-correlation,
momentum-based acceleration, and systematic asymmetry exploration.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_smoothed(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*c[k])) where c = conv(a,a).
    This approximates the gradient of max_k c[k] for large tau.
    Returns (gradient, convolution_values).
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Numerically stable softmax
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # Cross-correlation: grad[j] = 2 * sum_i p[j+i] * a[i]
    N = 1
    while N < 2 * len(p) + n:
        N <<= 1
    
    # We need: r[j] = sum_{i=0}^{n-1} p[j+i] * a[i] for j=0,...,n-1
    # This is cross-correlation (p ⋆ a)[j]
    # In FFT: r = ifft(conj(fft(p, N)) * fft(a, N))
    p_padded = np.zeros(N)
    p_padded[:len(p)] = p
    a_padded = np.zeros(N)
    a_padded[:n] = a
    
    P = fft(p_padded)
    A = fft(a_padded)
    r = ifft(np.conj(P) * A).real[:n]
    
    grad = 2.0 * r
    return grad, conv

def fourier_asym_perturb(a, rng, mode=1, strength=0.02):

    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))

    return a_new / s


def ramp_perturb(a, rng, strength=0.03):

    return a_new / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based smoothed gradient descent with Nesterov momentum, adaptive learning rate,
tau annealing, asymmetric perturbations, and multi-length exploration.
Key: efficient O(n log n) gradient via FFT cross-correlation, systematic asymmetry
exploration (since optimal solutions must be non-symmetric), and aggressive
high-temperature polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_smoothed(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*c[k])) where c = conv(a,a).
    Returns (gradient, convolution_values, max_conv_value).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    # Cross-correlation: grad[j] = 2 * sum_i p[j+i] * a[i]
    N = 1
    while N < 2 * len(p) + n:
        N <<= 1

    p_padded = np.zeros(N)
    p_padded[:len(p)] = p
    a_padded = np.zeros(N)
    a_padded[:n] = a

    P = fft(p_padded)
    A = fft(a_padded)
    r = ifft(np.conj(P) * A).real[:n]

    grad = 2.0 * r
    return grad, conv, max_conv


def smoothed_gd(a, time_budget, tau_start=5.0, tau_end=2000.0,
                lr_init=0.01, beta=0.9, verbose=False, seed=0):
    """
    Nesterov accelerated smoothed gradient descent on the simplex.
    Uses cosine annealing for tau and exponential decay for lr.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    lr = lr_init
    it = 0
    best_it = 0
    stale_count = 0
    prev_grad_norm = 0.0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        cos_val = 0.5 * (1.0 + np.cos(np.pi * progress))
        tau = tau_start + (tau_end - tau_start) * (1.0 - cos_val)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = a.copy()
        else:
            y = y / s_y

        grad_y, _, _ = compute_grad_smoothed(y, tau)
        grad_t = grad_y - np.mean(grad_y)
        grad_norm = float(np.linalg.norm(grad_t))

        # Adaptive LR: Barzilai-Borwein style
        if prev_grad_norm > 1e-10 and it % 3 == 0:
            ratio = grad_norm / prev_grad_norm
            if 0.5 < ratio < 2.0:
                lr = lr * (1.0 / ratio) ** 0.3

        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stale_count = 0
        else:
            stale_count += 1

        a = a_new
        it += 1
        lr = max(1e-6, lr * 0.9997)

        # Staleness detection: restart from best

            a = best_a.copy()
            v = np.zeros(n)
            lr = max(lr_init * 0.3, lr)
            stale_count = 0
            best_it = it

def fourier_asym_perturb(a, rng, mode=1, strength=0.03):
    """Add an asymmetric Fourier perturbation (sine breaks mirror symmetry)."""
    n = len(a)
    x = np.arange(n, dtype=np.float64)
    phase = rng.uniform(0, 2 * np.pi)
    pert = strength * np.sin(2 * np.pi * mode * x / n + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def ramp_perturb(a, rng, strength=0.04):
    """Add a linear ramp perturbation to induce asymmetry."""
    n = len(a)
    x = np.linspace(0, 1, n)
    sign = rng.choice([-1, 1])
    pert = strength * sign * (2 * x - 1) * rng.uniform(0.5, 1.5)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def quadratic_asym_perturb(a, rng, strength=0.03):
    """Add a quadratic asymmetry (parabolic offset)."""
    n = len(a)
    x = np.linspace(0, 1, n)
    center = rng.uniform(0.2, 0.8)
    pert = -strength * (x - center) ** 2
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, rng, strength=0.05):
    """Random multiplicative perturbation."""
    n = len(a)
    pert = np.ones(n) + rng.normal(0, strength, n)
    a_new = a * np.maximum(pert, 0)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 12.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        n_current = 800
        x = np.linspace(0, 1, n_current)
        best_seq = 0.5 + x
        best_seq = best_seq / best_seq.sum()
        best_score = compute_obj(best_seq)
        print(f"Default asymmetric init (n={n_current}): obj={best_score:.8f}")

    # === Phase 1: Initial GD on parent ===
    print(f"\n=== Phase 1: Initial GD ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.15, 180.0)
    if alloc > 30:
        a1 = smoothed_gd(best_seq, time_budget=alloc,
                         tau_start=5, tau_end=500, lr_init=0.02,
                         beta=0.9, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Initial GD: obj={best_score:.8f} **")
        else:
            print(f"  Initial GD: obj={obj1:.8f}")

    # === Phase 2: Systematic perturbation exploration ===
    print(f"\n=== Phase 2: Perturbation exploration ===")
    perturb_types = ["asym1", "asym2", "ramp", "asym3", "ramp_strong",
                     "quad", "blend_ramp", "asym5", "random"]
    for ptype in perturb_types:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(remaining * 0.08, 80.0)
        if alloc < 15:
            break

        a_r = try_perturbation_and_optimize(best_seq, ptype, rng, alloc,
                                            seed=seed + hash(ptype) % 10000)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  {ptype}: obj={best_score:.8f} **")
        else:
            print(f"  {ptype}: obj={obj_r:.8f}")

    # === Phase 3: High-precision polishing with high tau ===
    print(f"\n=== Phase 3: High-precision polish ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.25, 300.0)
    if alloc > 30:
        a2 = smoothed_gd(best_seq, time_budget=alloc,
                         tau_start=50, tau_end=5000, lr_init=0.005,
                         beta=0.9, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  High-tau: obj={best_score:.8f} **")
        else:
            print(f"  High-tau: obj={obj2:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_candidates = []
    for dn in [-100, -50, -200, 50, 100, 200, 300, 400, -300]:
        n_try = n_best + dn
        if 300 <= n_try <= 3000:
            n_candidates.append(n_try)
    for n_abs in [500, 700, 900, 1100, 1400, 1800]:
        if n_abs not in n_candidates and abs(n_abs - n_best) > 80:
            n_candidates.append(n_abs)

    n_candidates = sorted(set(n_candidates))[:7]
    print(f"  Trying n: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Add slight asymmetry
        a_try = fourier_asym_perturb(a_try, rng, mode=1, strength=0.02)

        alloc1 = min(remaining * 0.08, 60.0)
        if alloc1 > 15:
            a_try = smoothed_gd(a_try, time_budget=alloc1,
                                tau_start=10, tau_end=2000, lr_init=0.02)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diversification restarts ===
    print(f"\n=== Phase 5: Diversification ===")
    remaining = deadline - time.monotonic()
    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if r == 0:
            a_r = fourier_asym_perturb(best_seq, rng, mode=7, strength=0.04)
        elif r == 1:
            a_r = random_perturb(best_seq, rng, strength=0.08)
        else:
            a_r = ramp_perturb(best_seq, rng, strength=0.06)

        alloc = min(remaining * 0.1, 100.0)
        if alloc > 15:
            a_r = smoothed_gd(a_r, time_budget=alloc,
                              tau_start=10, tau_end=3000, lr_init=0.02,
                              beta=0.85)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        alloc = min(remaining * 0.4, 200.0)
        a_final = smoothed_gd(best_seq, time_budget=alloc,
                              tau_start=200, tau_end=10000, lr_init=0.003,
                              beta=0.9, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final polish 1: obj={best_score:.8f} **")
        else:
            print(f"  Final polish 1: obj={obj_final:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.35, 150.0)
        a_final = smoothed_gd(best_seq, time_budget=alloc,
                              tau_start=500, tau_end=20000, lr_init=0.002,
                              beta=0.7)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final polish 2: obj={best_score:.8f} **")

    # One more medium-tau pass to potentially escape ultra-high-tau local min
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 100.0)
        a_final = smoothed_gd(best_seq, time_budget=alloc,
                              tau_start=50, tau_end=1000, lr_init=0.005,
                              beta=0.95)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final polish 3: obj={best_score:.8f} **")

    # Validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
