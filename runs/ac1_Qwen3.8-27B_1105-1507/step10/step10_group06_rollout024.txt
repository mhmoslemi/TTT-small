def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_lp_fast(a):
    """Solve the LP: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, inf) on failure."""
    n = len(a)
    a_norm = a / np.sum(a)

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

                s = float(np.sum(g_val))
                if s > 1e-10:

            continue
    return None, np.inf

        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Accept with some probability (simulated annealing style)
            ttemp = max(0.01, 0.5 / (1 + t * 0.001))
            if np.random.random() < np.exp(-(obj_new - best_obj) / (ttemp * best_obj)):
                a = a_new
            else:
                a = best_a  # revert
        
        # Adaptive tau
        tau = min(tau * 1.05, tau_max)
        # Decay learning rate
        lr = 0.05 / (1 + 0.0001 * t)
    
    return best_a

def fourier_perturb(a, rng, strength=0.03):

def block_perturb(a, rng, block_size=20):

"""
Hybrid LP-alternation + adaptive gradient optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses diversified multi-start, adaptive blending, symmetry exploitation,
and multi-phase optimization with Nesterov/Adam gradient polishing.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp_fast(a):
    """Solve LP: min M s.t. conv(a_norm, g)_k <= M, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, inf) on failure."""
    n = len(a)
    a_sum = np.sum(a)
    if a_sum < 1e-15:
        return None, np.inf
    a_norm = a / a_sum

    # Build sparse convolution matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, np.inf


def smoothed_grad_1d(a, tau):
    """Gradient of log-sum-exp smoothed max of conv(a,a) w.r.t. a (on simplex)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    # grad_i = 2 * sum_k p_k * a_{k-i} for valid indices
    # This is 2 * (cross-correlation of p and a) at lags 0..n-1
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(np.conj(P) * A).real
    grad = 2.0 * cross[:n]
    return grad


def nesterov_polish(a, time_budget, tau_init=50.0, tau_max=8000.0):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    tau = tau_init
    stall = 0

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_grad_1d(y, tau)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new
            stall += 1
            if stall > 300:
                a = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
                tau = min(tau * 1.5, tau_max)
                stall = 0

        tau = min(tau * 1.04, tau_max)
        lr *= 0.9998

    return best_a


def adam_polish(a, time_budget, tau_init=100.0, tau_max=10000.0):
    """Adam optimizer on the smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    base_lr = 0.08
    tau = tau_init
    t = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        t += 1
        g = smoothed_grad_1d(a, tau)

        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = base_lr / (1 + 0.00005 * t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            ttemp = max(0.005, 0.3 / (1 + t * 0.0005))
            if rng.random() < np.exp(-(obj_new - best_obj) / (ttemp * best_obj)):
                a = a_new
            else:
                a = best_a.copy()

        tau = min(tau * 1.06, tau_max)

    return best_a


def fourier_perturb(a, rng, strength=0.03):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def block_perturb(a, rng, block_size=25):
    """Replace a random block with scaled random values."""
    n = len(a)
    a_new = a.copy()
    bs = min(block_size, n // 2)
    start = rng.integers(0, max(1, n - bs))
    a_new[start:start + bs] = rng.random(bs) * np.mean(a) * rng.uniform(0.3, 2.5)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def gradient_perturb(a, rng, strength=0.02):
    """Perturb in a random gradient-like direction."""
    n = len(a)
    g = smoothed_grad_1d(a, 50.0)
    noise = rng.standard_normal(n) * 0.5
    direction = -g + strength * noise * np.abs(a).mean()
    a_new = a + strength * direction
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_alternate(a, time_budget, seed=42, verbose=False):
    """LP-based alternating minimization with adaptive blending and diversification."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    t_blend = 0.4
    stall_count = 0
    last_improvement_iter = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = solve_lp_fast(a)
        if g_lp is None:
            break

        iter_count += 1

        # Adaptive blending factor candidates
        if iter_count < 20:
            candidates_t = [t_blend, 0.5, 0.7, 1.0]
        elif iter_count < 100:
            candidates_t = [t_blend, t_blend * 0.5, t_blend * 1.5, 0.1, 0.3, 0.5]
        else:
            candidates_t = [t_blend, t_blend * 0.5, 0.05, 0.02, 0.01, 0.03]
            if iter_count > 300:
                candidates_t.extend([0.005, 0.002])

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend

        for t in candidates_t:
            t = min(max(t, 0.001), 1.0)
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-15:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            t_blend = max(0.003, best_local_t * 0.92)
            stall_count = 0
            last_improvement_iter = iter_count
            if verbose and iter_count % 30 == 0:
                print(f"    LP alt {iter_count}: obj={best_obj:.10f}, t={t_blend:.5f}, M={M_lp:.6f}")
        else:
            stall_count += 1
            # Diversification strategies
            if stall_count >= 4:
                strategy = rng.integers(0, 4)
                if strategy == 0:
                    strength = 0.01 + 0.003 * min(stall_count, 15)
                    a = fourier_perturb(best_a, rng, strength=strength)
                elif strategy == 1:
                    a = block_perturb(best_a, rng, block_size=20)
                elif strategy == 2:
                    a = gradient_perturb(best_a, rng, strength=0.015)
                else:
                    # Reversal symmetry
                    a_rev = best_a[::-1].copy()
                    a = a_rev + 0.01 * rng.standard_normal(n) * best_a
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                t_blend = max(0.1, min(t_blend * 1.5, 0.8))
                stall_count = 0

    if verbose:
        print(f"    LP alternating: {iter_count} iter, final obj={best_obj:.10f}")
    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
