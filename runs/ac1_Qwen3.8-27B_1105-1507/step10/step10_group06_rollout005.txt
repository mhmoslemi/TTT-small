def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):

    n = len(a)

    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):

**Projected gradient for best response:**
To minimize max_k (A g)_k subject to g >= 0, sum g = 1:
- Use the log-sum-exp smoothing: f(g) = (1/tau) * log(sum_k exp(tau * (Ag)_k))
- Gradient: grad_g f = A^T @ p / sum_k p_k where p_k = exp(tau * (Ag)_k) / sum exp(tau * (Aj)_j)
- Project onto simplex {g >= 0, sum g = 1}

The projection onto the simplex can be done in O(n log n) time.

This gives a fast "soft" best response that approximates the LP solution.

def projected_simplex(v):
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def soft_best_response(a, tau=50.0, iters=100):

    Approximate best response via projected gradient on log-sum-exp.
    """

    g = np.ones(n) / n

        conv = compute_conv_fast(a_norm, g)

        max_conv = np.max(conv)
        p = np.exp(tau * (conv - max_conv))
        p = p / np.sum(p)

        # Actually, the gradient of log-sum-exp of (Ag) with respect to g is A^T @ p

        # d/dg log(sum exp(tau * (Ag)_k)) = tau * A^T @ (exp(tau*(Ag))/sum exp(tau*(Ag)))
        # = tau * A^T @ p
        # Let's use a simpler approach: just use A^T @ p as the gradient direction
        
        # A^T @ p: the j-th component is sum_k A[k,j] * p[k]
        # A[k,j] = a_norm[k-j] if 0 <= k-j < n, else 0
        # So (A^T p)_j = sum_{k=j}^{j+n-1} a_norm[k-j] * p[k]

        # Use FFT for fast computation
        # (A^T p)_j = sum_i a_norm[i] * p[i+j] = cross_corr(a_norm, p)[n-1-j]
        # Or use direct FFT
        
        grad = compute_atranspose_p(a_norm, p)

        lr = 1.0 / (tau + 1.0)  # Lipschitz constant is roughly tau * ||A||^2
        g_new = g - lr * grad
        g_new = projected_simplex(g_new)
        
        # Check convergence
        if np.max(np.abs(g_new - g)) < 1e-10:

        g = g_new
    
    return g
