**Alternative: Direct minimization via SOCP with better formulation**

- Set new target M' < M

6. Repeat with new M_target

**Another idea: Frank-Wolfe (Conditional Gradient) method**

The Frank-Wolfe method for minimizing a convex function over the simplex:
1. Compute subgradient g at current point

3. Take step: x_new = (1-t)*x + t*v
4. Choose t via line search or t = 2/(k+2)

"""
FISTA with adaptive log-sum-exp smoothing + LP polish + multi-length exploration.
Uses Nesterov-accelerated gradient descent on a smoothed peak objective,
combined with periodic LP direction finding for global improvements.
Key innovations: efficient FFT-based gradient computation, adaptive temperature
schedule, backtracking line search, and systematic length exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(a, tau):
    """
    Compute gradient of (1/tau)*logsumexp(tau*conv(a,a)).
    Returns (gradient, convolution, max_conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    w = exp_vals / float(np.sum(exp_vals))
    # Gradient: g[m] = sum_k w[k]*(a[k-m] + a[m-k])
    # = np.correlate(w, a, 'valid')[m] + np.convolve(a, w)[m]
    T1 = np.correlate(w, a, 'valid')
    T2 = np.convolve(a, w)[:n]
    grad = T1 + T2
    return grad, conv, max_c


def fista_smoothed(a, time_budget=100.0, tau_start=5.0, tau_end=100.0,

    FISTA (Nesterov accelerated gradient) on smoothed peak objective.
    Adaptive temperature schedule with backtracking line search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    t_k = 1.0
    alpha = max(0.1, n / (tau_end * 4.0))
    no_improve = 0
    stagnant_count = 0
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed >= time_budget:
            break
        progress = min(1.0, elapsed / max(time_budget, 1e-6))
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.6)

        # Nesterov extrapolation
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = x + ((t_k - 1.0) / t_next) * (x - x_prev)
        z = np.maximum(z, 0)
        s_z = z.sum()
        if s_z > 1e-10:
            z = z / s_z
        else:
            z = x.copy()

        # Compute gradient at z
        grad, conv_z, max_c = compute_smoothed_grad(z, tau)

        # Backtracking line search
        t_alpha = alpha
        z_obj = compute_obj(z)
        best_step = 0
        best_step_obj = z_obj
        found = False

            y = project_simplex(y)
            y_obj = compute_obj(y)
            if y_obj < z_obj - 1e-14 or y_obj < best_obj - 1e-14:
                best_step = t_alpha
                best_step_obj = y_obj
                found = True
                break
            t_alpha *= 0.5
        if not found:
            # Try a few fixed steps as fallback
            for t_try in [0.01, 0.005, 0.001]:
                y = z - t_try * grad
                y = project_simplex(y)
                y_obj = compute_obj(y)
                if y_obj < z_obj:
                    best_step = t_try
                    best_step_obj = y_obj
                    found = True
                    break

        if found and best_step > 0:
            x_prev = x.copy()
            x = z - best_step * grad
            x = project_simplex(x)
            x_obj = compute_obj(x)
            if x_obj < best_obj - 1e-13:
                best_obj = x_obj
                best_a = x.copy()
                no_improve = 0
                alpha = min(alpha * 1.08, n / tau_start)
            else:
                no_improve += 1
        else:
            x_prev = x.copy()
            no_improve += 1

        t_k = t_next
        it += 1

        if no_improve > 100:
            stagnant_count += 1
            if stagnant_count > 8:
                break
            # Perturb and reset
            x = best_a * np.exp(rng.normal(0, 0.01, n))
            x = np.maximum(x, 0)
            s_x = x.sum()
            if s_x > 1e-10:
                x = x / s_x
            x_prev = x.copy()
            t_k = 1.0
            alpha = max(0.05, n / (tau * 8.0))
            no_improve = 0
        elif no_improve > 40:
            t_k = 1.0
            x_prev = x.copy()
            alpha *= 0.7
            no_improve = 20

        # Periodically reset alpha to track the changing tau

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, M_target=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_value) or (None, None).
    """
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value).flatten(), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget=60.0, verbose=False, seed=42):
    """LP-based polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        best_t = 0
        best_t_obj = best_obj
        for t_val in [0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]:
            a_try = (1 - t_val) * best_a + t_val * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-15:
                best_t_obj = obj_try
                best_t = t_val

        if best_t > 0 and best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = (1 - best_t) * best_a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            s_ba = best_a.sum()
            if s_ba > 1e-10:
                best_a = best_a / s_ba
            a = best_a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                scale = 0.005 + 0.004 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                no_improve = 0
                if it > 40:
                    break

        it += 1
        if verbose and it % 5 == 0:
            print(f"      LP it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def subgradient_refine(a, time_budget=40.0, K=10):
    """Subgradient method with top-K peak weighting and adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 2.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break

        weights = np.exp((peak_vals - max_pv) * 10.0 / max_pv)
        weights /= float(weights.sum())

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-13:
                    break
            else:
                lr *= 0.998

        it += 1

    return best_a


def tr_sscp_polish(a, time_budget=100.0, r_init=0.25, verbose=False, seed=42):

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)

        return a
    a = a / s

    r = r_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        # Solve trust region SOCP
        a_curr = a / a.sum()
        C_old = compute_conv(a_curr)
        A_cross = build_conv_matrix(a_curr)

        M = cp.Variable()

            cp.sum(a_new_var) == 1,
            a_new_var >= 0,
            C_old + 2 * (A_cross @ delta) <= M,
            cp.norm(delta, 2) <= r,
        ]

        prob = cp.Problem(cp.Minimize(M), constraints)
        delta_val = None
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    delta_val = np.asarray(delta.value).flatten()
                    break
            except Exception:
                continue

        if delta_val is None:
            break

        norm_d = np.linalg.norm(delta_val)
        if norm_d > r:
            delta_val = delta_val * (r / norm_d)

        # Line search
        best_alpha = 0
        best_alpha_obj = best_obj
        alphas = [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0, 1.5, 2.0]
        for t_alpha in alphas:
            a_try = a + t_alpha * delta_val
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj - 1e-15:
                best_alpha_obj = obj_try
                best_alpha = t_alpha

        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-14:
            a = a + best_alpha * delta_val
            a = np.maximum(a, 0)
            s_n = a.sum()
            if s_n > 1e-10:
                a = a / s_n
            best_obj = best_alpha_obj
            best_a = a.copy()
            no_improve = 0
            r = min(r * 1.15, 1.5)
        else:
            no_improve += 1
            r *= 0.6
            if no_improve >= 6:
                if r < 0.02:
                    a = best_a * np.exp(rng.normal(0, 0.01, n))
                    a = np.maximum(a, 0)
                    s_p = a.sum()
                    if s_p > 1e-10:
                        a = a / s_p
                    r = r_init
                    no_improve = 0
                    if it > 60:
                        break
                else:
                    r = max(r * 0.5, 0.02)

        it += 1
        if verbose and it % 5 == 0:
            print(f"      TR it{it}: obj={best_obj:.8f}, r={r:.4f}, t={time.monotonic()-start:.0f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: FISTA (Nesterov) + LP + TR-SOCP hybrid with adaptive
    temperature, multi-length exploration, and systematic perturbations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Quick FISTA warmup ===
    print(f"\n=== Phase 1: FISTA Warmup (20s) ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        p1_budget = min(remaining * 0.05, 20.0)
        a_f = fista_smoothed(best_seq, time_budget=p1_budget,
                             tau_start=3.0, tau_end=30.0, seed=1)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  FISTA warmup: obj={best_score:.8f} **")
        else:
            print(f"  FISTA warmup: obj={obj_f:.8f}")

    # === Phase 2: Main optimization loop ===
    print(f"\n=== Phase 2: Main Loop (FISTA + LP + TR) ===")
    remaining = deadline - time.monotonic()
    main_budget = min(remaining * 0.55, 550.0)

    # Main loop: alternate FISTA and LP with periodic TR
    if main_budget > 30:
        sub_round = 0
        rng_m = np.random.default_rng(100 + seed)
        while time.monotonic() < deadline - main_budget + 20:
            round_budget = min(remaining * 0.1, 80.0)
            if round_budget < 20:
                break

            # FISTA phase
            f_budget = round_budget * 0.5
            a_f = fista_smoothed(best_seq, time_budget=f_budget,
                                 tau_start=5.0, tau_end=80.0,
                                 verbose=(sub_round % 3 == 0), seed=50 + sub_round)
            obj_f = compute_obj(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Round {sub_round} FISTA: obj={best_score:.8f} **")
            else:
                print(f"  Round {sub_round} FISTA: obj={obj_f:.8f}")

            # LP phase
            remaining2 = deadline - time.monotonic()
            if remaining2 > 40:
                lp_budget = min(round_budget * 0.3, 40.0)
                a_lp = lp_polish(best_seq, time_budget=lp_budget,
                                 verbose=(sub_round % 3 == 0), seed=200 + sub_round)
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    print(f"  Round {sub_round} LP: obj={best_score:.8f} **")
                else:
                    print(f"  Round {sub_round} LP: obj={obj_lp:.8f}")

            # TR phase (less frequent)
            remaining2 = deadline - time.monotonic()
            if remaining2 > 40 and sub_round % 2 == 1:
                tr_budget = min(round_budget * 0.2, 30.0)
                a_tr = tr_sscp_polish(best_seq, time_budget=tr_budget,
                                      r_init=0.2, seed=300 + sub_round)
                obj_tr = compute_obj(a_tr)
                if obj_tr < best_score:
                    best_score = obj_tr
                    best_seq = a_tr.copy()
                    print(f"  Round {sub_round} TR: obj={best_score:.8f} **")

            sub_round += 1
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break

    # === Phase 3: Length exploration ===
    print(f"\n=== Phase 3: Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 100),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_current) > 50]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)

        # Quick FISTA
        alloc1 = min(remaining * 0.15, 35.0)
        if alloc1 > 10:
            a_try = fista_smoothed(a_try, time_budget=alloc1,
                                   tau_start=3.0, tau_end=60.0, seed=500)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.1, 25.0)
        if alloc2 > 10:
            a_try = lp_polish(a_try, time_budget=alloc2, seed=600)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Perturbation + FISTA ===
    print(f"\n=== Phase 4: Perturbation + FISTA ===")
    remaining = deadline - time.monotonic()

    for p_idx in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        scale = 0.006 + 0.006 * p_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            break
        a_pert = a_pert / s_pert

        alloc = min(remaining * 0.2, 70.0)
        if alloc > 15:
            a_pert = fista_smoothed(a_pert, time_budget=alloc,
                                    tau_start=10.0, tau_end=100.0,
                                    seed=800 + p_idx)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Pert{p_idx} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Pert{p_idx} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        # Long FISTA with high temperature (sharp peaks)
        final_f_budget = min(remaining * 0.4, 120.0)
        a_final = fista_smoothed(best_seq, time_budget=final_f_budget,
                                 tau_start=30.0, tau_end=200.0,
                                 verbose=True, seed=1000)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final FISTA: obj={best_score:.8f} **")
        else:
            print(f"  Final FISTA: obj={obj_final:.8f}")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        lp_final_budget = min(remaining * 0.3, 50.0)
        a_final_lp = lp_polish(best_seq, time_budget=lp_final_budget, seed=2000)
        obj_flp = compute_obj(a_final_lp)
        if obj_flp < best_score:
            best_score = obj_flp
            best_seq = a_final_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final subgradient
    remaining = deadline - time.monotonic()
    if remaining > 15:
        sg_budget = min(remaining * 0.3, 30.0)
        a_sg = subgradient_refine(best_seq, time_budget=sg_budget, K=12)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
