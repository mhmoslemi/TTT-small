def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a) = 1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    return compute_obj_normalized(a_norm)


def smoothed_gradient_fft(a, tau, N=None):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*conv[k])) w.r.t. a.
    Uses FFT for efficiency.

    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    # Compute gradient: grad[i] = sum_k p[k] * (a[k-i] + a[i-k])

    # = (a * p_rev)[i] where p_rev is p reversed
    # Or equivalently: ifft(fft(a) * conj(fft(p)))
    
    A = fft(a, N)
    P = fft(p, N)
    # Cross-correlation: ifft(conj(A) * P) gives sum_j a[j] * p[j+i] ... 
    # Let me just use: grad[i] = sum_k p[k]*a[k-i] + sum_k p[k]*a[i-k]
    # First part: let u = k-i, then sum_u p[u+i]*a[u] = (a * p_shifted)[i]
    # This is getting complicated. Let me use the property:
    # The gradient of conv(a,a)[k] w.r.t. a[i] is a[k-i] + a[i-k]
    # So grad_smooth[i] = sum_k p[k] * (a[k-i] + a[i-k])
    # = sum_k p[k]*a[k-i] + sum_k p[k]*a[i-k]
    # First term: for fixed i, sum_k p[k]*a[k-i] where k-i must be in [0,n-1]

    # Define p as a sequence of length 2n-1.
    # grad[i] = sum_{k=0}^{2n-2} p[k] * a[k-i] + sum_{k=0}^{2n-2} p[k] * a[i-k]
    # Note: a[x] = 0 if x not in [0, n-1]
    # First sum: let m = k-i, so k = m+i. Sum over m where m in [0,n-1] and m+i in [0, 2n-2]
    #   = sum_{m=max(0,-i)}^{min(n-1, 2n-2-i)} p[m+i] * a[m]
    # Second sum: let m = i-k, so k = i-m. Sum over m where m in [0,n-1] and i-m in [0, 2n-2]
    #   = sum_{m=max(0,i-(2n-2))}^{min(n-1, i)} p[i-m] * a[m]
    
    # Both of these can be computed as convolutions!
    # First sum = (a * p)[i] where * is linear convolution (a has length n, p has length 2n-1)
    #   Wait, (a*p)[i] = sum_j a[j]*p[i-j]. That's not what we want.
    #   We want sum_m a[m]*p[m+i] = (a * p_rev_pad)[i] ... 
    
    # OK let me just use a simpler approach: 
    # grad[i] = sum_k p[k]*(a[k-i] + a[i-k])
    # Note that a[k-i] + a[i-k] is the same as (a * delta at k) contributing to position i
    # The full gradient is: sum_k p[k] * (shifted_a_k + reversed_shifted_a_k)
    # where shifted_a_k[i] = a[k-i] (a shifted by k)
    
    # Using FFT: 
    # The sequence "a[k-i]" for fixed k is just a reversed and shifted by k.
    # sum_k p[k] * a[k-i] = sum_k a[k-i] * p[k]
    # Let j = k-i: sum_j a[j] * p[j+i]  (for valid j)

    # Cross-correlation can be computed as: ifft(conj(FFT(a)) * FFT(p))
    # But we need to be careful about lengths.
    
    # a has length n, p has length 2n-1.
    # Cross-correlation: R[i] = sum_j a[j] * p[j+i]
    # This is the "offset" convolution. We can compute it as:
    # R = ifft(conj(fft(a, N)) * fft(p, N))[:2n-1+n] ... 
    # Actually, the cross-correlation of a (length n) and p (length 2n-1) has length n + (2n-1) - 1 = 3n-2.
    # R[i] for i in [0, 3n-3]
    
    # But we only need R[i] for i in [0, n-1] (first term) and the second term is R'[i] = sum_j a[j] * p[i-j]... 
    
    # I think the cleanest way is:
    # grad[i] = (cross_corr_1)[i] + (cross_corr_2)[i]

    # and cross_corr_2[i] = sum_k p[k]*a[i-k] = cross-correlation of p and a at lag -i (p shifted left by i)
    
    # By symmetry, cross_corr_2[i] = cross_corr_1[i] if p is symmetric (which it is if a is palindromic).
    # In general, cross_corr_2[i] = cross_corr_1 of p and a reversed at lag (n-1-i)... 
    
    # Let me just compute both directly using FFT:
    # cross_corr_1[i] = sum_{k} p[k] * a[k-i] 
    #   = sum_{j} a[j] * p[j+i]  (sub j = k-i)
    #   This is (convolution of a with p, evaluated at position ... ) 
    #   Actually: sum_j a[j] * p[j+i] = (a * p_flipped)[i + (2n-2)] ... no.

    # That's fine for a few thousand iterations.

    # sum_k p[k] * a[k-i]: Think of this as the (i)-th element of the "cross-correlation" of sequences p and a.
    # In signal processing: (p ⋆ a)[i] = sum_k p[k] * conj(a[k-i]) = sum_k p[k] * a[k-i] (for real a)
    # This is computed as: ifft(FFT(p) * conj(FFT(a)))
    # The result has length N (the FFT size), and we take the first (2n-1 + n - 1) = 3n-2 elements... 
    # Actually the linear cross-correlation of p (length 2n-1) and a (length n) has length (2n-1) + n - 1 = 3n-2.
    # We need elements [0, n-1] of this cross-correlation.
    
    # And sum_k p[k] * a[i-k] = (a ⋆ p)[i] = ifft(FFT(a) * conj(FFT(p)))
    # We need elements [0, n-1] of this.
    
    # So: grad = ifft(FFT(p,N) * conj(FFT(a,N)))[:n] + ifft(FFT(a,N) * conj(FFT(p,N)))[:n]
    #       = 2 * Re(ifft(FFT(p,N) * conj(FFT(a,N)))[:n])
    
    # Wait, for real sequences: ifft(FFT(p)*conj(FFT(a))) is real, and it equals ifft(FFT(a)*conj(FFT(p))).
    # So grad = 2 * ifft(FFT(p,N) * conj(FFT(a,N)))[:n]
    
    # Let me verify: ifft(FFT(p)*conj(FFT(a)))[i] = sum_k p[k] * a[k-i] (modulo N)
    # With N large enough (N >= 3n-2), there's no wraparound, so this is correct.
    
    P_fft = fft(p, N)
    A_fft = fft(a, N)
    grad_full = ifft(P_fft * np.conj(A_fft)).real
    grad = 2.0 * grad_full[:n]
    
    return grad, float(np.max(conv))


def nesterov_smoothed(a, time_budget, tau_init=5.0, tau_end=500.0, verbose=True):
    """
    Nesterov accelerated gradient descent on smoothed max-conv objective.
    Adaptive tau from tau_init to tau_end.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    beta = 0.9
    base_lr = 1.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        # Adaptive tau: exponential schedule
        progress = (time.monotonic() - start) / time_budget
        tau = tau_init * (tau_end / tau_init) ** progress
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g, max_conv = smoothed_gradient_fft(y, tau, N)
        
        # Gradient step with adaptive lr
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-15:
            # Step size: we want ||lr * g|| to be reasonable relative to ||a||
            lr = base_lr * 0.5 / max(1.0, g_norm / base_lr)
            lr = min(max(lr, 0.01 * base_lr), 5.0 * base_lr)
        else:
            lr = base_lr
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        beta = min(0.9 + 0.05 * progress, 0.98)
        it += 1
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    NSD iter {it}: obj={best_obj:.10f}, tau={tau:.1f}, "
                  f"no_improve={no_improve}, time={elapsed:.1f}s")
        
        # Reset if stuck
        if no_improve > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            no_improve = 0
    
    return best_a


def build_conv_matrix_fast(a_norm):
    """Build the convolution matrix using a circulant-free approach with FFT."""

def solve_lp_best_response(a, verbose=False):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix A (2n-1) x n where A[k, i] = a_norm[k-i] if valid
    # A @ g = conv(a_norm, g)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i+n, i] = a_norm
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate_fast(a, time_budget, max_iters=50, verbose=True):
    """
    LP alternating with early stopping. Solves LP, blends, checks improvement.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        g_lp, M_lp = solve_lp_best_response(a)
        if g_lp is None:
            break
        
        # Try several blending factors
        best_local = best_obj
        best_local_a = best_a.copy()
        
        for t in [0.3, 0.5, 0.7, 1.0, 0.1, 0.2]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_local:
                best_local = obj_try
                best_local_a = a_try.copy()
        
        if best_local < best_obj - 1e-12:
            best_obj = best_local
            best_a = best_local_a
            a = best_a.copy()
            if verbose:
                print(f"    LP iter {it}: obj={best_obj:.10f}, M_lp={M_lp:.8f}")
        else:
            if verbose:
                print(f"    LP iter {it}: no improvement (M_lp={M_lp:.8f})")
            break
    
    return best_a


def raised_cosine_init(n):
    """Raised cosine (Hann) window."""
    k = np.arange(n)
    return 0.5 * (1 + np.cos(2 * np.pi * k / (n - 1)))


def sum_cosines_init(n):
    """Sum of cosines (Nuttall-like) window."""
    k = np.arange(n)
    x = 2 * np.pi * k / (n - 1)
    # 4-term Nuttall
    a0, a1, a2, a3 = 0.36358, 0.48917, 0.13659, 0.01064
    w = a0 - a1 * np.cos(x) + a2 * np.cos(2*x) - a3 * np.cos(3*x)
    return np.maximum(w, 0)


def smooth_random_init(n, rng, smoothness=5):
    """Random initialization with Gaussian smoothing."""
    a = rng.standard_normal(n)
    # Apply Gaussian smoothing
    kernel_size = 2 * smoothness + 1
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - smoothness)**2 / (smoothness/2)**2)
    kernel /= kernel.sum()
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated adaptive smoothed gradient descent (Nesterov) as the primary
fast method, combined with periodic LP best-response polish and multi-start
diverse initialization. Key improvements: much faster per-iteration cost (no LP
in the main loop), adaptive log-sum-exp temperature scheduling, and diverse
initializations (parent, Hann, Nuttall, smooth random) to escape local minima.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj_full(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    return compute_obj_normalized(a_norm)


def lse_gradient(a, tau, N):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*conv[k])) w.r.t. a.
    Uses FFT. Returns (gradient, max_conv_value).
    Formula: grad = 2 * ifft(conj(FFT(a,N)) * FFT(p,N)).real[:n]
    where p[k] = exp(tau*(conv[k]-max_conv)) / sum(exp(...))
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Numerical stability: clip to avoid overflow
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        idx = np.argmax(conv)
        p[idx] = 1.0
    else:
        p = exp_vals / sum_exp

    A_fft = fft(a, N)
    P_fft = fft(p, N)
    grad_full = ifft(np.conj(A_fft) * P_fft).real
    grad = 2.0 * grad_full[:n]
    return grad, max_conv


def lse_gradient_direct(a, tau):
    """Direct (non-FFT) computation of LSE gradient for verification/small n."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    # grad[i] = 2 * sum_j a[j] * p[j+i]
    grad = np.zeros(n)
    for i in range(n):
        # sum_j a[j]*p[j+i] where j in [0,n-1] and j+i in [0, 2n-2]
        j_max = min(n - 1, 2 * n - 2 - i)
        if j_max >= 0:
            grad[i] = 2.0 * np.dot(a[:j_max + 1], p[i:i + j_max + 1])
    return grad, max_conv


def nesterov_smoothed(a, time_budget, tau_init=5.0, tau_end=500.0, verbose=True, verbose_every=2000):
    """
    Nesterov accelerated gradient descent on smoothed max-conv objective.
    Adaptive tau from tau_init to tau_end over the time budget.
    Projects onto simplex {x >= 0, sum(x) = 1} each step.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N = 1
    while N < 3 * n:
        N <<= 1

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 1.5 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        progress = (time.monotonic() - start) / time_budget
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, _ = lse_gradient(y, tau, N)

        # Adaptive learning rate
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-15:
            lr = base_lr * min(2.0, max(0.1, base_lr / g_norm))
        else:
            lr = base_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        beta = min(0.9 + 0.08 * progress, 0.98)
        it += 1

        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - start
            print(f"    NSD iter {it}: obj={best_obj:.10f}, tau={tau:.1f}, "
                  f"no_imp={no_improve}, t={elapsed:.1f}s")

        if no_improve > 8000:
            a = best_a.copy()
            v = np.zeros(n)
            no_improve = 0

    return best_a


def peak_subgradient_descent(a, time_budget, verbose=True):
    """
    Fast subgradient descent tracking only the argmax of conv.
    Each step: find peak, compute subgradient, update.
    Very fast per iteration (one FFT for conv + one FFT for gradient).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N = 1
    while N < 3 * n:
        N <<= 1

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    lr = 2.0 / n
    v = np.zeros(n)
    beta = 0.9
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: sg[i] = 2*a[k_star - i] if 0 <= k_star - i < n, else 0
        # Vectorized: sg = 2 * (shifted and reversed a)
        sg = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if hi >= lo:
            # sg[i] = 2*a[k_star-i] for i in [lo, hi]
            # Let j = k_star - i, so i = k_star - j, j in [k_star-hi, k_star-lo]
            j_lo = k_star - hi
            j_hi = k_star - lo
            sg[lo:hi+1] = 2.0 * a[j_lo:j_hi+1][::-1]

        # Nesterov step
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Recompute gradient at y for Nesterov
        conv_y = compute_conv(y)
        k_star_y = int(np.argmax(conv_y))
        sg_y = np.zeros(n)
        lo2 = max(0, k_star_y - n + 1)
        hi2 = min(n - 1, k_star_y)
        if hi2 >= lo2:
            j_lo2 = k_star_y - hi2
            j_hi2 = k_star_y - lo2
            sg_y[lo2:hi2+1] = 2.0 * a[j_lo2:j_hi2+1][::-1]

        v = beta * v - lr * sg_y
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr *= 0.99995
        it += 1

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    PSD iter {it}: obj={best_obj:.10f}, t={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a_norm
    return A


def solve_lp_best_response(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget, max_iters=30, verbose=True):
    """
    LP alternating minimization for final polish.
    Each iteration: solve LP for best response, try blending, accept if improves.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    blend = 0.5

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_lp_best_response(a)
        lp_time = time.monotonic() - t0

        if g_lp is None:
            break

        best_local = best_obj
        best_local_a = best_a.copy()

        # Try multiple blending factors
        ts = [blend, blend * 0.5, blend * 1.5, 0.1, 0.3, 0.5, 0.7, 1.0]
        if it > 10:
            ts.extend([0.05, 0.02, 0.01])
        ts = [min(max(t, 0.005), 1.0) for t in ts]

        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_local:
                best_local = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local < best_obj - 1e-13:
            best_obj = best_local
            best_a = best_local_a
            a = best_a.copy()
            blend = max(0.02, getattr(locals(), 'best_local_t', blend) * 0.95)
            if verbose:
                print(f"    LP {it}: obj={best_obj:.10f}, M={M_lp:.8f}, "
                      f"t={blend:.3f}, lp_time={lp_time:.2f}s")
        else:
            if verbose:
                print(f"    LP {it}: no improvement (M={M_lp:.8f})")
            # Try a small perturbation
            a = a + 0.01 * np.random.default_rng(it).standard_normal(n) * a
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            if it > 5:
                break

    return best_a


def hann_init(n):
    """Hann (raised cosine) window."""
    k = np.arange(n)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * k / (n - 1)))


def nuttall_init(n):
    """4-term Nuttall sum-of-cosines window."""
    k = np.arange(n)
    x = 2.0 * np.pi * k / (n - 1)
    a0, a1, a2, a3 = 0.36358, 0.48917, 0.13659, 0.01064
    w = a0 - a1 * np.cos(x) + a2 * np.cos(2 * x) - a3 * np.cos(3 * x)
    return np.maximum(w, 0)


def smooth_random_init(n, rng, bandwidth=0.05):
    """Random init with spectral smoothing."""
    N = 1
    while N < 2 * n:
        N <<= 1
    spectrum = rng.standard_normal(N // 2 + 1) + 1j * rng.standard_normal(N // 2 + 1)
    # Low-pass filter in frequency domain
    freqs = np.fft.rfftfreq(N)
    weight = np.exp(-(freqs / bandwidth) ** 2)
    spectrum *= weight
    a = np.fft.irfft(spectrum, N)[:n]
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def fourier_escape(a, rng, strength=0.02):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 3
    for k in range(1, n_modes + 1):
        freq = 2.0 * np.pi * k / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-start with Nesterov smoothed gradient + LP polish.
    Strategy:
      Phase 1: Quick screening of diverse initializations (40s each)
      Phase 2: Deep Nesterov optimization on top candidates (majority of time)
      Phase 3: LP polish on best candidates
      Phase 4: Final gradient refinement
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0  # 20s safety buffer
    rng = np.random.default_rng(seed)

    best_a = None
    best_obj = np.inf

    # --- Generate initializations ---
    inits = []

    # Parent
    try:
        a_parent = np.asarray(height_sequence_1, dtype=np.float64)
        a_parent = np.maximum(a_parent, 0)
        if a_parent.size > 0 and a_parent.sum() > 0.01:
            a_parent = a_parent / a_parent.sum()
            inits.append((a_parent.copy(), "parent"))
    except Exception:
        pass

    # Various windows at different lengths
    for n_try in [800, 1000]:
        w = hann_init(n_try)
        w = w / w.sum()
        inits.append((w, f"hann_{n_try}"))

    for n_try in [800, 1000]:
        w = nuttall_init(n_try)
        w = np.maximum(w, 0)
        s = float(w.sum())
        if s > 1e-10:
            w = w / s
            inits.append((w, f"nuttall_{n_try}"))

    # Smooth random
    for n_try in [800, 1000, 1200]:
        w = smooth_random_init(n_try, rng, bandwidth=0.04)
        inits.append((w, f"rand_{n_try}"))

    # Very smooth (nearly uniform with slight variation)
    n_smooth = 800
    k = np.arange(n_smooth)
    w = 1.0 + 0.1 * np.cos(2 * np.pi * k / n_smooth) + 0.05 * np.cos(4 * np.pi * k / n_smooth)
    w = np.maximum(w, 0)
    w = w / w.sum()
    inits.append((w, "near_uniform"))

    print(f"Total initializations: {len(inits)}")

    # --- Phase 1: Quick screening ---
    print("\n=== Phase 1: Screening ===")
    screen_budget_each = 35.0
    screened = []

    for a_init, name in inits:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(screen_budget_each, remaining * 0.6)
        if alloc < 15:
            break

        a_scr = nesterov_smoothed(a_init, time_budget=alloc,
                                   tau_init=5.0, tau_end=80.0, verbose=False)
        obj_scr = compute_obj_normalized(a_scr)
        screened.append((obj_scr, a_scr, name))
        print(f"  {name:15s}: obj={obj_scr:.10f}")

    screened.sort(key=lambda x: x[0])
    print(f"  Best after screening: {screened[0][2]} with obj={screened[0][0]:.10f}")

    # --- Phase 2: Deep optimization on top candidates ---
    print("\n=== Phase 2: Deep Optimization ===")
    top_k = min(3, len(screened))

    for idx in range(top_k):
        obj_s, a_s, name = screened[idx]
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break
        if obj_s >= best_obj and best_a is not None:
            continue

        alloc = min(remaining * 0.35, 280.0)
        if alloc < 60:
            break

        print(f"\n  Deep: {name} (budget={alloc:.0f}s, start obj={obj_s:.8f})")

        # Sub-phase A: Nesterov with moderate tau
        a_deep = nesterov_smoothed(a_s, time_budget=alloc * 0.5,
                                    tau_init=10.0, tau_end=300.0, verbose=True,
                                    verbose_every=3000)
        obj_deep = compute_obj_normalized(a_deep)

        # Sub-phase B: Peak subgradient for fine tuning
        remaining_b = deadline - time.monotonic()
        if remaining_b > 30 and obj_deep < best_obj or best_a is None:
            psd_budget = min(alloc * 0.2, remaining_b * 0.3)
            if psd_budget > 15:
                a_psd = peak_subgradient_descent(a_deep, time_budget=psd_budget, verbose=False)
                obj_psd = compute_obj_normalized(a_psd)
                if obj_psd < obj_deep:
                    a_deep = a_psd
                    obj_deep = obj_psd
                    print(f"    PSD improved: {obj_deep:.10f}")

        # Sub-phase C: Nesterov with high tau (near-exact max tracking)
        remaining_c = deadline - time.monotonic()
        if remaining_c > 20:
            final_budget = min(alloc * 0.2, remaining_c * 0.3)
            if final_budget > 10:
                a_fin = nesterov_smoothed(a_deep, time_budget=final_budget,
                                           tau_init=200.0, tau_end=2000.0, verbose=False)
                obj_fin = compute_obj_normalized(a_fin)
                if obj_fin < obj_deep:
                    a_deep = a_fin
                    obj_deep = obj_fin

        if obj_deep < best_obj or best_a is None:
            best_obj = obj_deep
            best_a = a_deep.copy()
            print(f"  ** New best from {name}: obj={best_obj:.10f}")
        else:
            print(f"  {name}: obj={obj_deep:.10f} (no improvement over {best_obj:.10f})")

    # --- Phase 3: LP polish on the best ---
    print("\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40 and best_a is not None:
        lp_budget = min(remaining * 0.35, 150.0)
        print(f"  LP polish budget: {lp_budget:.0f}s")
        a_lp = lp_polish(best_a, time_budget=lp_budget, max_iters=25, verbose=True)
        obj_lp = compute_obj_normalized(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            print(f"  ** LP improved: obj={best_obj:.10f}")
        else:
            print(f"  LP: obj={obj_lp:.10f} (no improvement)")

    # --- Phase 4: Try escape + re-optimize if stuck ---
    remaining = deadline - time.monotonic()
    if remaining > 60 and best_a is not None:
        print(f"\n=== Phase 4: Escape & Re-optimize ===")
        for attempt in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            strength = 0.01 + 0.01 * attempt
            a_esc = fourier_escape(best_a, rng, strength=strength)
            alloc = min(remaining * 0.3, 80.0)
            if alloc < 20:
                break
            a_new = nesterov_smoothed(a_esc, time_budget=alloc,
                                       tau_init=10.0, tau_end=200.0, verbose=False)
            obj_new = compute_obj_normalized(a_new)
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                print(f"  ** Escape {attempt}: obj={best_obj:.10f}")
            else:
                print(f"  Escape {attempt}: obj={obj_new:.10f} (no improvement)")

    # --- Phase 5: Final LP + gradient on best ---
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_a is not None:
        print(f"\n=== Phase 5: Final Polish ===")
        lp_final_budget = min(remaining * 0.4, 60.0)
        if lp_final_budget > 15:
            a_final = lp_polish(best_a, time_budget=lp_final_budget, max_iters=10, verbose=True)
            obj_final = compute_obj_normalized(a_final)
            if obj_final < best_obj:
                best_obj = obj_final
                best_a = a_final.copy()

        remaining = deadline - time.monotonic()
        if remaining > 15:
            grad_budget = min(remaining * 0.5, 30.0)
            if grad_budget > 10:
                a_fin = nesterov_smoothed(best_a, time_budget=grad_budget,
                                           tau_init=500.0, tau_end=5000.0, verbose=False)
                obj_fin = compute_obj_normalized(a_fin)
                if obj_fin < best_obj:
                    best_obj = obj_fin
                    best_a = a_fin.copy()

    # --- Fallback ---
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(800) / 800
        best_obj = compute_obj_normalized(best_a)

    # Convert to output
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        print("  WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
