"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. FFT-accelerated gradient via cross-correlation (O(n log n) per eval)
2. L-BFGS quasi-Newton optimization with log-sum-exp smoothing
3. Adaptive tau annealing schedule
4. Hybrid LP + gradient refinement
5. Multi-restart with diverse initializations (Gaussian, cosine, prolate, Fourier)
"""

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_obj_and_grad(a, tau, C=100.0):
    """
    Compute smoothed objective and gradient.
    Objective: 2n * LSE(a*a, tau) + C*(sum(a)-1)^2
    Gradient computed via FFT cross-correlation.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    lse = max_conv + math.log(sum_exp) / tau
    obj = 2.0 * n * lse + C * (float(np.sum(a)) - 1.0) ** 2

    # Gradient via cross-correlation: grad_i = 2 * sum_j a[j] * p[j+i]
    # This is 2 * correlate(a, p, 'full') at lags 0..n-1
    # Using FFT for efficiency
    N = 1
    while N < 3 * n:
        N <<= 1
    FA = fft(a, N)
    FP = fft(p, N)
    corr_full = ifft(FA * np.conj(FP), N).real[:3 * n - 2]
    # correlate(a, p, 'full') has length n + (2n-1) - 1 = 3n-2
    # The lag m=0 element is at index (2n-1-1) = 2n-2
    corr = corr_full[2 * n - 2: 2 * n - 2 + n]

    sum_a = float(np.sum(a))
    grad = 4.0 * n * corr + 2.0 * C * (sum_a - 1.0)
    return obj, grad


def precise_obj_and_subgrad(a):
    """Compute exact objective and subgradient (for active set)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf, np.zeros(n)
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_val = float(np.max(conv))
    obj = 2.0 * n * max_val
    # Find active set (within tolerance)
    tol = 1e-10
    active = np.where(conv >= max_val - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    alpha = np.ones(len(active)) / len(active)
    # Subgradient: sum of alpha_k * 2 * d(a*a)[k]/da (in normalized space)
    sg = np.zeros(n)
    for k in active:
        for i in range(n):
            d = abs(k - i)
            if d < n:
                # Check validity conditions
                if k < n:
                    if 0 <= i <= k:
                        sg[i] += alpha[list(active).index(k)] * 2 * a_norm[d]
                else:
                    if k - n + 1 <= i <= n - 1:
                        sg[i] += alpha[list(active).index(k)] * 2 * a_norm[d]
    return obj, sg


def lbfgs_optimize(a, tau_schedule, time_budget, verbose=False):
    """
    Run L-BFGS optimization with a schedule of tau values.
    a: initial point (will be normalized to sum=1)
    tau_schedule: list of tau values, low to high
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_schedule:
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        remaining = time_budget - elapsed
        tau_budget = remaining * 0.4

        def obj_and_grad(x):
            x = np.maximum(x, 0)
            return smoothed_obj_and_grad(x, tau)

        res = minimize(
            lambda x: obj_and_grad(x)[0],
            a.copy(),
            jac=lambda x: obj_and_grad(x)[1],
            method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options={'maxiter': 500, 'maxfun': 2000, 'ftol': 1e-16, 'gtol': 1e-12}
        )

        if res.x is not None:
            a_new = np.maximum(res.x, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    if verbose:
                        elapsed = time.monotonic() - start
                        print(f"      tau={tau:.0f}: obj={best_obj:.10f}, t={elapsed:.1f}s")
                a = a_new
            else:
                a = best_a.copy()
        else:
            a = best_a.copy()

    return best_a


def nesterov_smoothed_descent(a, tau, lr, time_budget, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    v = np.zeros(n)
    start = time.monotonic()
    it = 0
    best_it = 0
    t = 1.0

    while time.monotonic() - start < time_budget:
        t += 1
        # Nesterov lookahead
        y = a + (1 - 1.0 / (t + 1.5)) * (a - a - v)  # standard Nesterov
        # Actually: y = a + (t-1)/(t+1.5) * (a - a_prev) ... let me use standard form
        # y = a + beta * v where beta = (t_prev - 1) / (t + 1.5)
        beta = (max(t - 1, 1) - 1) / (t + 1.5)
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        obj_y, grad_y = smoothed_obj_and_grad(y, tau)
        v = (1.0 / (t + 1.5)) * (y - a) * beta + (1.0 / (t + 1.5)) * v - lr * grad_y / np.sqrt(n)

        a_new = y - lr * grad_y / np.sqrt(n)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()

            best_a = a_new.copy()

        it += 1
        lr = lr * 0.9999

        if it - best_it > 50000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it
            lr *= 0.5

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[:n]
        # Actually conv(a,g)[k] = sum_j a[j]*g[k-j], so (A@g)[k] = sum_j A[k,j]*g[j]
        # A[k,j] = a[k-j] if 0<=k-j<n else 0
        # So A[k,j] = a[k-j] for max(0,k-n+1)<=j<=min(k,n-1)
    # Redo properly
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, M) or (None, None).
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None, None

    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:
            continue
    return None, None


def lp_iteration(a, time_budget, seed=42, verbose=False):
    """LP-based block coordinate descent."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        it += 1

        # Line search
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            best_it = it
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: obj={best_obj:.10f}")
        else:
            # Perturbation
            if it - best_it > 10:
                strength = min(0.1, 0.01 * (it - best_it))
                pert = 1.0 + rng.normal(0, strength, n)
                a = best_a * np.maximum(pert, 0)
                a = a / a.sum()

    return best_a


def init_gaussian(n, sigma_frac=0.3):
    """Gaussian initialization."""
    x = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-(x - center) ** 2 / (2 * sigma ** 2))
    return a / a.sum()


def init_cosine(n):
    """Half-cosine window."""
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(np.pi * x / (n - 1)))
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def init_uniform(n):
    """Uniform initialization."""
    return np.ones(n) / n


def init_fourier(n, rng):
    """Random Fourier-based initialization."""
    a = np.ones(n) / n
    n_modes = max(3, n // 50)
    for k in range(1, n_modes + 1):
        amp = rng.uniform(0.1, 0.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * np.arange(n) / n + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def init_prolate(n, bandwidth=0.3):
    """Approximate prolate spheroidal wave function (Dolph-Chebyshev-like)."""
    x = np.arange(n) - n / 2.0
    # Use a raised cosine with adjustable bandwidth
    a = np.sinc(bandwidth * x / (n / 2.0))
    a = np.maximum(a, 0.001)
    return a / a.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
