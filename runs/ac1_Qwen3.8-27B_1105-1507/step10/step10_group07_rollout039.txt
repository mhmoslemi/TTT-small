def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Exact Coordinate Descent + Smoothed Gradient + Multi-length Search for peak autocorrelation minimization.
Primary method: coordinate descent exploiting the bilinear structure of convolution updates.
Escape mechanism: smoothed gradient with softmax peak weighting.
Also explores different sequence lengths to avoid local minima of fixed-n optimization.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_step(a, tau=50.0, lr=None, momentum=0.7):
    """One step of smoothed gradient descent with softmax peak weighting."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]

    if lr is None:
        lr = 3.0 / np.sqrt(n)

    a_new = a - lr * g
    a_new = project_simplex(a_new)
    return a_new, g


def coord_descent(a, time_budget=10.0, verbose=False, seed=42):
    """
    Exact coordinate descent for minimizing 2n*max(conv)/(sum)^2.
    For each coordinate i, finds the best value x given all other coordinates fixed.
    Exploits the structure: changing a_i only affects conv[k] for k in [i, i+n-1].
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    best_obj = obj
    conv_len = 2 * n - 1

    start = time.monotonic()
    sweep = 0
    rng = np.random.default_rng(seed)

    # Precompute prefix and suffix max of conv (updated each sweep)
    def compute_prefix_suffix_max(conv_arr):
        prefix = np.zeros(len(conv_arr))
        prefix[0] = conv_arr[0]
        for j in range(1, len(conv_arr)):
            prefix[j] = max(prefix[j - 1], conv_arr[j])
        suffix = np.zeros(len(conv_arr))
        suffix[-1] = conv_arr[-1]
        for j in range(len(conv_arr) - 2, -1, -1):
            suffix[j] = max(suffix[j + 1], conv_arr[j])
        return prefix, suffix

    while time.monotonic() - start < time_budget:
        prefix, suffix = compute_prefix_suffix_max(conv)

        improved = False
        # Use random order to avoid bias
        order = rng.permutation(n)

        for i in order:

            k_end = min(i + n - 1, conv_len - 1)

            # Max outside affected range
            m_left = prefix[i - 1] if i > 0 else 0.0
            if k_end + 1 <= conv_len - 1:
                m_right = suffix[k_end + 1]
            else:
                m_right = 0.0
            m_out = max(m_left, m_right)

            ai = a[i]

            if ai < 1e-9:
                candidates = np.array([0.0, 1e-4, 5e-4, 1e-3, 3e-3, 0.01, 0.03])
            else:
                # Include equilibration candidates based on top peaks
                top_k = np.argsort(conv)[::-1][:3]
                equil = []
                for k_top in top_k:
                    kt = int(k_top)
                    ki_val = kt - i
                    if 0 <= ki_val < n and kt != 2 * i:
                        a_ki = a[ki_val]
                        if a_ki > 1e-12:
                            # x that makes conv[kt] equal to current value minus some target

                                equil.append(x_eq)
                                equil.append(x_eq * 0.8)
                                equil.append(x_eq * 1.2)
                base_cands = [0.0, ai * 0.4, ai * 0.7, ai * 0.9, ai,

                candidates = np.array(base_cands + equil)

            best_x = ai
            best_local = obj

                if x < 0 or abs(x - ai) < 1e-13:
                    continue
                S_new = S - ai + x
                if S_new < 1e-10:
                    continue

                delta = x - ai
                # New conv in affected range: conv[k] + 2*delta*a[k-i] (+ delta^2 at k=2i)
                length_affected = k_end - k_start + 1
                ki_idx = np.arange(length_affected)
                linear_part = conv[k_start:k_end + 1] + 2.0 * delta * a[ki_idx]

                    idx_2i = 2 * i - k_start
                    linear_part[idx_2i] += delta * delta

                max_in = float(np.max(linear_part))
                new_max = max(max_in, m_out)
                new_obj = 2.0 * n * new_max / (S_new * S_new)

                if new_obj < best_local - 1e-13:
                    best_local = new_obj
                    best_x = x

            if best_x != ai:
                delta = best_x - ai
                length_affected = k_end - k_start + 1

                conv[k_start:k_end + 1] += 2.0 * delta * a[ki_idx]

                    conv[2 * i] += delta * delta
                a[i] = best_x

                improved = True

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

                sp = a_pert.sum()
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        a = a_pert
                        conv = compute_conv(a)
                        S = 1.0
                        obj = obj_pert
                        sweep = 0
                        continue
            break

        sweep += 1
        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start

def smoothed_gd_optimize(a, time_budget=60.0, tau_init=10.0, tau_end=300.0, verbose=False):
    """Smoothed gradient descent with increasing temperature."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 4.0 / np.sqrt(n)
    v = np.zeros(n)
    momentum = 0.75
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        lr = max(1e-10, lr * 0.9997)
        it += 1

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def proposal_path_search(a, time_budget=30.0, verbose=False):
    """
    Path-following: at each step, find the direction that most reduces the top peak,
    then take the largest step in that direction while maintaining the constraint.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:5]
        max_val = float(conv[top_idx[0]])

        # For each top peak k, the gradient direction to reduce it is -a[k-i] for each i
        # Combined: reduce coordinates that contribute to top peaks
        pressure = np.zeros(n)
        for k in top_idx:
            ki = int(k)
            # conv[k] = sum_j a_j * a[k-j], so d(conv[k])/d(a_i) = 2*a[k-i] (or a[i] if k=2i)
            i_start = max(0, ki - (n - 1))
            i_end = min(n - 1, ki)
            if i_start <= i_end:
                j_start = ki - i_end
                j_end = ki - i_start
                w = conv[ki] / max_val
                pressure[i_start:i_end + 1] += w * 2.0 * a[j_start:j_end + 1][::-1]

        # Also add pressure from energy (sum of squares) to encourage spreading
        pressure += 0.5 * a

        # Move: reduce high-pressure coordinates, increase low-pressure
        target = best_a * np.exp(-0.05 * pressure / (np.max(pressure) + 1e-10))
        target = np.maximum(target, 0)
        ts = target.sum()
        if ts > 1e-10:
            target = target / ts
            obj_t = compute_obj(target)
            if obj_t < best_obj:
                best_obj = obj_t
                best_a = target.copy()
                a = target.copy()
            else:
                # Try smaller step
                for frac in [0.5, 0.3, 0.1]:
                    mix = (1 - frac) * a + frac * target
                    mix = np.maximum(mix, 0)
                    ms = mix.sum()
                    if ms > 1e-10:
                        mix = mix / ms
                        obj_m = compute_obj(mix)
                        if obj_m < best_obj:
                            best_obj = obj_m
                            best_a = mix.copy()
                            a = mix.copy()
                            break

    return best_a

            n_options.append(n_abs)

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)

        # Quick SGD
        alloc1 = min(remaining * 0.1, 25.0)
        if alloc1 > 8:
            a_try = smoothed_gd_optimize(a_try, time_budget=alloc1,
                                         tau_init=8.0, tau_end=150.0)

        # Coordinate descent
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 60.0)
        if alloc2 > 10:
            a_try = coord_descent(a_try, time_budget=alloc2)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 6: Perturbation + CD ===
    print(f"\n=== Phase 6: Perturbation + CD ===")
    remaining = deadline - time.monotonic()

    for r_idx in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        scale = 0.004 + 0.004 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            break
        a_pert = a_pert / s_pert

        alloc = min(remaining * 0.2, 60.0)
        if alloc > 10:
            a_pert = coord_descent(a_pert, time_budget=alloc, seed=500 + r_idx)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={best_score:.10f} **")
            else:
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={obj_pert:.10f}")

    # === Phase 7: Final intensive CD ===
    print(f"\n=== Phase 7: Final Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.6, 100.0)
        a_final = coord_descent(best_seq, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.10f} **")
        else:
            print(f"  Final CD: obj={obj_final:.10f}")

    # Final quick SGD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 30.0)
        a_sg = smoothed_gd_optimize(best_seq, time_budget=alloc,
                                    tau_init=50.0, tau_end=500.0)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SGD: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
