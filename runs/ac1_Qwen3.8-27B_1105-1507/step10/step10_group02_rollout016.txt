"""
Optimizes step function coefficients to minimize 2n*max(conv(a,a))/sum(a)^2.
Strategy: Fast coordinate descent with subgradient-ordered sweeps, LP polishing
with line-search blending, pair-perturbation for diversification, and systematic
multi-n exploration. Key improvements over parent: subgradient prioritization,
faster 1D solves with reduced golden-section iterations, and more aggressive
LP-CD alternation.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Assumes sum(a) ~= 1, returns 2n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def f_1d_coord(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i):
    """1D objective for coordinate descent subproblem."""
    if len(a_ki_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom


def golden_section_1d(func, x_lo, x_hi, max_iter=25, tol=1e-12):
    """Golden section search for minimum of a unimodal function."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = x_hi - gr * (x_hi - x_lo)
    d_pt = x_lo + gr * (x_hi - x_lo)
    fc = func(c_pt)
    fd = func(d_pt)
    for _ in range(max_iter):
        if x_hi - x_lo < tol * max(1.0, abs(x_hi)):
            break
        if fc < fd:
            x_hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = x_hi - gr * (x_hi - x_lo)
            fc = func(c_pt)
        else:
            x_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = x_lo + gr * (x_hi - x_lo)
            fd = func(d_pt)
    return (x_lo + x_hi) / 2.0


def compute_subgradient_priority(a, conv, threshold_frac=0.90):
    """Compute subgradient-based priority ordering for coordinate descent.
    Returns permutation of indices ordered by decreasing subgradient magnitude."""
    n = len(a)
    max_conv = float(np.max(conv))
    active_thresh = max_conv * threshold_frac
    active = (conv >= active_thresh).astype(np.float64)

    # Subgradient: grad_i = 2 * sum_j a[j] * active[i+j] for valid indices
    # This is a sliding dot product: grad_i = 2 * dot(a, active[i:i+n])
    # Use FFT: cross-correlation
    N = 1
    while N < 2 * n:
        N <<= 1
    A_fft = fft(a, N)
    Ac = fft(active, N)
    corr = ifft(np.conj(A_fft) * Ac).real[:n]
    grad = 2.0 * np.abs(corr)

    # Also add a small random component to break ties
    order = np.argsort(-grad, kind='stable')
    return order

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-4)
        a = a / a.sum()
    elif mode == 3:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x) + 0.3 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    # Gradient: grad_i = 2 * sum_j a[j] * p[i+j]
    N = 1
    while N < 2 * n:
        N <<= 1
    A_fft = fft(a, N)
    Pc = ifft(np.conj(A_fft) * fft(p, N)).real[:n]
    grad = 2.0 * Pc
    return grad, max_conv


def smoothed_descent(a, time_budget=100.0, tau_init=50.0, tau_end=500.0, verbose=False):
    """Nesterov-accelerated smoothed gradient descent on the normalized objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * (progress ** 0.7)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = 1.0
        else:
            y = y / s_y

        grad, _ = smoothed_gradient_obj(y, tau)
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break

        # Adaptive step size
        lr = max(0.5 * 2.0 / np.sqrt(n), min(2.0 / np.sqrt(n), 1.0 / (g_norm * n ** 0.5 + 1e-8)))

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 2000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining fast CD, LP polishing,
    smoothed gradient descent, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = 2.0 * n_default * float(np.max(compute_conv(best_seq)))
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Fast CD (primary workhorse) ===
    print(f"\n=== Phase 1: Fast Coordinate Descent (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 400.0)
    if p1_budget > 30:
        a_cd = fast_cd(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_cd = 2.0 * a_cd.size * float(np.max(compute_conv(a_cd / a_cd.sum())))
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            n_best = len(best_seq)
            print(f"  Phase 1 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: LP Polishing ===
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 250.0)
    if p2_budget > 30:
        # Multiple LP rounds with different perturbations
        for rnd in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            # Slight perturbation each round to explore
            if rnd > 0:
                pert = rng.normal(0, 0.01 * rnd, len(best_seq))
                a_try = best_seq * np.exp(pert)
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
            else:
                a_try = best_seq.copy()

            alloc = min(p2_budget / (5 - rnd), remaining * 0.5)
            if alloc < 15:
                break

            a_lp = lp_polish_line_search(a_try, time_budget=alloc, verbose=True,
                                         seed=500 + rnd * 7)
            obj_lp = 2.0 * a_lp.size * float(np.max(compute_conv(a_lp / a_lp.sum())))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                n_best = len(best_seq)
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")

    # === Phase 3: CD refinement after LP ===
    print(f"\n=== Phase 3: CD Refinement ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 200.0)
    if p3_budget > 20:
        a_cd2 = fast_cd(best_seq, time_budget=p3_budget, verbose=True,
                        seed=seed + 12345)
        obj_cd2 = 2.0 * a_cd2.size * float(np.max(compute_conv(a_cd2 / a_cd2.sum())))
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            n_best = len(best_seq)
            print(f"  Phase 3 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 CD: obj={obj_cd2:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()

    # Generate candidate n values around current best
    n_candidates = sorted(set([
        max(400, n_best - 200),
        max(400, n_best - 100),
        n_best,
        n_best + 100,
        n_best + 200,
        n_best + 400,
        n_best + 600,
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 3000]
    # Remove duplicates and keep the most diverse
    n_candidates = list(dict.fromkeys(n_candidates))[:6]
    print(f"  Trying n values: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try == n_best:
            continue

        a_try = resample_sequence(best_seq, n_try)

        # Quick CD
        alloc_cd = min(remaining * 0.15, 60.0)
        if alloc_cd > 15:
            a_try = fast_cd(a_try, time_budget=alloc_cd, verbose=False, seed=n_try)

        obj_try = 2.0 * a_try.size * float(np.max(compute_conv(a_try / a_try.sum())))

        # If promising, do LP polish
        if obj_try < best_score + 0.005 and remaining > 80:
            alloc_lp = min(remaining * 0.1, 40.0)
            if alloc_lp > 10:
                a_try = lp_polish_line_search(a_try, time_budget=alloc_lp,
                                              verbose=False, seed=n_try + 100)
                obj_try = 2.0 * a_try.size * float(
                    np.max(compute_conv(a_try / a_try.sum())))

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts with CD ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = 1 + r
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.3 + 0.2 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.3, 80.0)
        if alloc > 20:
            a_r = fast_cd(a_r, time_budget=alloc, verbose=False, seed=9000 + r * 111)
            obj_r = 2.0 * a_r.size * float(np.max(compute_conv(a_r / a_r.sum())))
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                n_best = len(best_seq)
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 6: Smoothed GD + CD ===
    print(f"\n=== Phase 6: Smoothed Gradient Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        # Perturb and run GD
        pert = rng.normal(0, 0.02, len(best_seq))
        a_pert = best_seq * np.exp(pert)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc_gd = min(remaining * 0.25, 80.0)
        if alloc_gd > 20:
            a_gd = smoothed_descent(a_pert, time_budget=alloc_gd,
                                    tau_init=50.0, tau_end=400.0, verbose=False)
            obj_gd = 2.0 * a_gd.size * float(np.max(compute_conv(a_gd / a_gd.sum())))
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
                n_best = len(best_seq)
                print(f"  GD: obj={best_score:.8f} **")
            else:
                print(f"  GD: obj={obj_gd:.8f}")

            # CD after GD
            remaining2 = deadline - time.monotonic()
            if remaining2 > 30:
                a_cd3 = fast_cd(best_seq, time_budget=min(remaining2 * 0.3, 50.0),
                                verbose=False, seed=77777)
                obj_cd3 = 2.0 * a_cd3.size * float(np.max(compute_conv(a_cd3 / a_cd3.sum())))
                if obj_cd3 < best_score:
                    best_score = obj_cd3
                    best_seq = a_cd3.copy()
                    print(f"  CD after GD: obj={best_score:.8f} **")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Final CD
        alloc = min(remaining * 0.4, 100.0)
        if alloc > 10:
            a_final = fast_cd(best_seq, time_budget=alloc, verbose=False, seed=seed + 99999)
            obj_final = 2.0 * a_final.size * float(np.max(compute_conv(a_final / a_final.sum())))
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

        # Final LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_lp_final = lp_polish_line_search(best_seq, time_budget=min(remaining * 0.5, 40.0),
                                               verbose=False, seed=31415)
            obj_lp_final = 2.0 * a_lp_final.size * float(
                np.max(compute_conv(a_lp_final / a_lp_final.sum())))
            if obj_lp_final < best_score:
                best_score = obj_lp_final
                best_seq = a_lp_final.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

            # One more CD
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_final2 = fast_cd(best_seq, time_budget=min(remaining * 0.7, 30.0),
                                   verbose=False, seed=27182)
                obj_final2 = 2.0 * a_final2.size * float(
                    np.max(compute_conv(a_final2 / a_final2.sum())))
                if obj_final2 < best_score:
                    best_score = obj_final2
                    best_seq = a_final2.copy()
                    print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
