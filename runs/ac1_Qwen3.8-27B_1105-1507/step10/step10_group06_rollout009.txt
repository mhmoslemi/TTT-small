    a_norm = a / s

    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_conv_lp(a):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

                s = float(np.sum(g_val))
                if s > 1e-10:

"""
Multi-strategy optimizer for step function coefficients minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses LP alternating minimization combined
with adaptive active-set gradient optimization, Barzilai-Borwein step sizes,
interleaved local search, and multi-n exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def top_k_gradient(a, K):
    """Compute gradient of sum of top K convolution values w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[-K:]
    grad = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            m_range = np.arange(lo, hi + 1)
            grad[m_range] += 2.0 * a[k - m_range]
    return grad, top_idx, conv


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def gradient_optimize(a, time_budget, K_init=20, K_max=200, seed=42):
    """
    Gradient-based optimization with adaptive active set size K.
    Uses Barzilai-Borwein step size and Nesterov momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    K = K_init
    lr_bb = 1.0 / max(1.0, float(np.linalg.norm(a) + 1e-10))
    v = np.zeros(n)  # momentum
    beta = 0.9
    prev_grad = None
    prev_a = None
    no_improve = 0
    max_no_improve = 300

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        # Compute gradient of top-K sum
        grad, top_idx, conv = top_k_gradient(a, K)

        # Barzilai-Borwein step size
        if prev_grad is not None and prev_a is not None:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            s_norm = float(np.dot(s_vec, s_vec))
            y_norm = float(np.dot(y_vec, y_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-15 and y_norm > 1e-15 and abs(sy) > 1e-15:
                lr1 = s_norm / abs(sy)
                lr2 = abs(sy) / y_norm
                # Alternate or mix
                lr_bb = 0.5 * (lr1 + lr2)
                lr_bb = min(max(lr_bb, 1e-6), 0.1)

        # Nesterov lookahead
        y = a + beta * v
        y = project_simplex(y)

        # Gradient step
        grad_y, _, _ = top_k_gradient(y, K)
        a_new = y - lr_bb * grad_y
        a_new = project_simplex(a_new)

        # Update momentum
        v = beta * v - lr_bb * grad_y

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        prev_a = (a + lr_bb * grad_y).copy()  # previous iterate approximation
        prev_grad = grad_y.copy()

        # Adapt K: increase when stuck, decrease when progressing well
        if no_improve > max_no_improve:
            K = min(K * 1.3, K_max)
            v = np.zeros(n)
            lr_bb = 0.5 / (float(np.linalg.norm(grad_y)) + 1e-10)
            lr_bb = min(lr_bb, 0.05)
            a = best_a.copy()
            no_improve = 0
        elif no_improve > max_no_improve // 2:
            K = min(int(K * 1.1), K_max)
        elif no_improve == 0 and K > K_init:
            K = max(int(K * 0.95), K_init)

        # Periodic small perturbation to escape
        if no_improve > max_no_improve // 3 and no_improve % 50 == 0:
            a = best_a + 0.01 * rng.standard_normal(n) * (best_a + 1e-6)
            a = project_simplex(a)
            v = np.zeros(n)

    return best_a


def local_search_pair_swap(a, time_budget, seed=42):
    """Local search: try pair swaps (transfer mass between two positions)."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    base_eps = 0.005
    while time.monotonic() - start < time_budget:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try transferring from i to j and j to i
        for src, dst in [(i, j), (j, i)]:
            if a[src] < 1e-10:
                continue
            eps = base_eps * a[src]
            a_try = a.copy()
            a_try[src] -= eps
            a_try[dst] += eps
            if a_try[src] < 0:
                continue
            a_try = a_try / a_try.sum()
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break

    return best_a

def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 8
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                best_obj = obj_p4b
                best_a = a_p4b.copy()
                print(f"  Phase 4b (Grad): obj={best_obj:.10f} **")

        # Local search
        remaining = deadline - time.monotonic()
        if remaining > 20:
            ls_budget = min(remaining * 0.4, 60)
            a_p4c = local_search_pair_swap(best_a, time_budget=ls_budget, seed=seed + 500)
            obj_p4c = compute_obj(a_p4c)
            if obj_p4c < best_obj:
                best_obj = obj_p4c
                best_a = a_p4c.copy()
                print(f"  Phase 4c (LS): obj={best_obj:.10f} **")

    # === Phase 5: Alternative initializations and n exploration (10% of budget) ===
    print(f"\n=== Phase 5: Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    # Try different n values
    n_options = sorted(set([n_best - 50, n_best + 50, n_best - 100, n_best + 100,
                            n_best + 200]))
    n_options = [n for n in n_options if 400 <= n <= 2000]

    for n_try in n_options[:3]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        a_try = resample_to_length(best_a, n_try)
        alloc = min(remaining * 0.3, 100)
        if alloc < 20:
            break

        # Quick gradient optimization at new n
        a_try = gradient_optimize(a_try, time_budget=alloc,
                                   K_init=15, K_max=100, seed=n_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: obj={best_obj:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # Try cosine initialization at current n
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for freq, amp in [(1, 0.3), (2, 0.2), (3, 0.15)]:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            a_cos = cosine_init(n_best, freq, amp, seed=seed + freq * 1000)
            # Blend with best
            a_blend = 0.3 * a_cos + 0.7 * best_a
            a_blend = project_simplex(a_blend)
            a_blend = gradient_optimize(a_blend, time_budget=min(remaining * 0.3, 60),
                                         K_init=15, K_max=80, seed=seed + freq * 2000)
            obj_blend = compute_obj(a_blend)
            if obj_blend < best_obj:
                best_obj = obj_blend
                best_a = a_blend.copy()
                print(f"  ** cosine(f={freq},a={amp}): obj={best_obj:.10f} **")
            else:
                print(f"  cosine(f={freq},a={amp}): obj={obj_blend:.10f}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Short gradient pass with small K (focus on peak)
        a_final = gradient_optimize(best_a, time_budget=min(remaining * 0.5, 80),
                                     K_init=5, K_max=30, seed=seed + 9999)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final: obj={best_obj:.10f} **")
        else:
            print(f"  Final: obj={obj_final:.10f}")

        # One more local search
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_ls = local_search_pair_swap(best_a, time_budget=min(remaining * 0.6, 40),
                                           seed=seed + 8888)
            obj_ls = compute_obj(a_ls)
            if obj_ls < best_obj:
                best_obj = obj_ls
                best_a = a_ls.copy()
                print(f"  Final LS: obj={best_obj:.10f} **")

    # Final validation
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
