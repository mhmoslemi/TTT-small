For the LP, the number of constraints is 2n-1 (for the conv <= M part) + 1 (for sum(g)=1). The number of variables is n+1 (g_0,...,g_{n-1}, M). So the LP size is O(n) variables and O(n) constraints, which should be very fast for modern solvers.

```python
import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def solve_lp_cvx(a, prev_g=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except:
            continue
    return None, None

def lp_nesterov_iterate(a, time_budget, verbose=True):
    """LP-based iteration with Nesterov acceleration."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov state
    y = a.copy()  # extrapolated point
    beta = 0.9
    v = np.zeros(n)
    last_g = None
    stall = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Solve LP at the extrapolated point
        g_lp, M_lp = solve_lp_cvx(y)
        if g_lp is None:
            break
        
        # Line search: try different mixing parameters
        best_t = 0.5
        best_t_obj = best_obj
        best_a_new = a.copy()
        
        t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for t in t_candidates:
            a_try = (1-t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_new = a_try.copy()
        
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a
            stall = 0
            # Update Nesterov state
            v = beta * v - best_t * (y - g_lp)
            y = a + v
            y = np.maximum(y, 0)
            s_y = float(np.sum(y))
            if s_y > 1e-10:
                y = y / s_y
            else:
                y = a.copy()
            beta = min(0.99, beta + 0.001)
        else:
            stall += 1
            a = y.copy()  # Reset to extrapolated point
            if stall > 5:
                break
    
    return best_a
```

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: fast LP iteration with warm-started MOSEK, Nesterov acceleration,
adaptive support identification, direct KKT Newton polish, and systematic multi-n sweep
with better time allocation.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Cross convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_fast(a, prev_g=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            kwargs = {}
            if solver == cp.MOSEK:
                kwargs['mosek_kwargs'] = {'MSK_DPAR_INTPNT_CO_TOL_REL_GAP': 1e-10}
            prob.solve(solver=solver, **kwargs)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(max(M.value, 0))
        except Exception:
            continue
    return None, None


def lp_nesterov(a, time_budget, verbose=True):
    """LP-based iteration with Nesterov acceleration and adaptive line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov state
    y = a.copy()
    beta = 0.85
    v = np.zeros(n)
    stall = 0
    it = 0
    last_t = 0.5

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Solve LP at extrapolated point
        g_lp, M_lp = solve_lp_fast(y)
        if g_lp is None:
            break

        # Line search with adaptive candidates
        t_cands = np.unique(np.clip(
            np.concatenate([
                np.linspace(0.05, 1.0, 10),
                [max(0.02, last_t * 0.5), max(0.02, last_t * 0.8),
                 last_t, min(1.0, last_t * 1.2), min(1.0, last_t * 1.5)]
            ]), 0.01, 1.0))

        best_t_obj = best_obj
        best_t = last_t
        best_a_new = a.copy()

        for t in t_cands:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_new = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a
            last_t = best_t
            stall = 0
            it += 1

            # Nesterov update
            v = beta * v - best_t * (y - g_lp)
            y = a + v
            y = np.maximum(y, 0)
            s_y = float(np.sum(y))
            if s_y > 1e-10:
                y = y / s_y
            else:
                y = a.copy()
            beta = min(0.99, beta + 0.002)

            if verbose and it % 50 == 0:
                print(f"    LP-Ne {it}: obj={best_obj:.8f}, t={best_t:.3f}, {time.monotonic()-start:.0f}s")
        else:
            stall += 1
            a = y.copy()
            v = np.zeros(n)
            if stall > 8:
                break

    return best_a


def smoothed_grad(a, tau):
    """Compute smoothed objective gradient using FFT."""
    n = len(a)
    a = a / np.sum(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A_f = fft(a, N)
    g_full = ifft(P * np.conj(A_f)).real
    return 2.0 * g_full[:n]


def high_tau_gd(a, time_budget, tau_start=200, tau_end=20000, verbose=False):

        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        y = a + beta * v
        y = np.maximum(y, 0)

        if g_norm < 1e-15:
            break

        # Backtracking line search
        lr = 2.0 / (g_norm + 1e-10)

        obj_new = obj_y

def kkt_newton_polish(a, max_iter=50):
    """Newton polish on KKT conditions at the identified support and active set."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)

        return a

    if len(support) < 2:
        return a

        max_conv = float(np.max(conv))
        # Active set: lags where conv is within 1% of max
        active = np.where(conv > max_conv * 0.99)[0]
        n_a = len(active)
        n_s = len(support)

        if n_a < 1 or n_s < 1:
            break

        # Build the KKT linear system for (delta_a on support, y on active, delta_lambda)
        # At optimum: sum_k y_k * a[k-m] = lambda for m in support
        # And: y must be a probability vector on active set
        # Linearize: update a by small delta, re-solve for y

        # Current dual: solve A_active^T @ y = lambda * 1_support, sum(y)=1
        A_sub = np.zeros((n_s, n_a))
        for i, m in enumerate(support):
            for j, k in enumerate(active):
                if 0 <= k - m < n:
                    A_sub[i, j] = a[k - m]

        # System: A_sub @ y - lambda = 0 (n_s eqs), sum(y) = 1 (1 eq)
        B = np.zeros((n_s + 1, n_a + 1))
        rhs = np.zeros(n_s + 1)
        B[:n_s, :n_a] = A_sub
        B[:n_s, n_a] = -1.0
        B[n_s, :n_a] = 1.0
        rhs[n_s] = 1.0

        try:
            sol = np.linalg.solve(B, rhs)
            y_val = np.maximum(sol[:n_a], 0)
            y_sum = float(np.sum(y_val))
            if y_sum < 1e-10:
                break
            y_val = y_val / y_sum
            lambda_val = sol[n_a]
        except np.linalg.LinAlgError:

        y_full = np.zeros(2 * n - 1)
        y_full[active] = y_val

                    D[i, j] = y_full[k]

            step = 0.1 / (np.max(np.abs(grad_supp)) + 1e-10)
            a_supp = a_new[support] - step * (grad_supp - grad_mean)
            a_supp = np.maximum(a_supp, 0)
            s_supp = float(np.sum(a_supp))
            if s_supp < 1e-10:

            a_supp = a_supp / s_supp

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a


def diverse_init(n, rng, mode=0):

        x = np.arange(n, dtype=np.float64)

def resample_sequence(a, new_n):

        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def optimize_for_n(a_init, time_budget, verbose=False):
    """Optimize a sequence of given length using all techniques."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Phase 1: LP Nesterov (60% of budget)
    lp_budget = time_budget * 0.5
    if lp_budget > 10:
        a_lp = lp_nesterov(a, time_budget=lp_budget, verbose=verbose)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp
            if verbose:
                print(f"    LP: obj={best_obj:.8f}")

    # Phase 2: High-tau GD (20% of budget)
    gd_budget = time_budget * 0.25
    if gd_budget > 5:
        a_gd = high_tau_gd(best_a, time_budget=gd_budget,
                           tau_start=500, tau_end=30000, verbose=False)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_obj:
            best_obj = obj_gd
            best_a = a_gd
            if verbose:
                print(f"    GD: obj={best_obj:.8f}")

    # Phase 3: Support shrink + re-optimize (10%)
    a_shr = support_shrink(best_a)
    obj_shr = compute_obj(a_shr)
    if obj_shr < best_obj:
        best_obj = obj_shr
        best_a = a_shr

    # Phase 4: KKT Newton polish
    a_kkt = kkt_newton_polish(best_a, max_iter=30)
    obj_kkt = compute_obj(a_kkt)
    if obj_kkt < best_obj:
        best_obj = obj_kkt
        best_a = a_kkt
        if verbose:
            print(f"    KKT: obj={best_obj:.8f}")

    # Phase 5: Second LP pass on polished result
    remaining = time_budget - 0.85 * time_budget
    if remaining > 5:
        a_lp2 = lp_nesterov(best_a, time_budget=remaining, verbose=False)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP-Nesterov iteration, high-tau GD, KKT Newton polish,
    support reduction, and systematic multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0  # safety margin

    rng = np.random.default_rng(seed)
    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = diverse_init(800, rng, mode=3)
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Intensive optimization of parent (300s) ===
    print(f"\n=== Phase 1: Parent optimization (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 300.0)
    if p1_budget > 30:
        a_opt, obj_opt = optimize_for_n(best_seq, time_budget=p1_budget, verbose=True)
