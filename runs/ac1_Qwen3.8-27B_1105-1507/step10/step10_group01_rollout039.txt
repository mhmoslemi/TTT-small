"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast coordinate descent (O(n^2) per sweep with incremental convolution updates),
spectral initialization for good starting points, smoothed gradient descent for
final polishing, and multi-n exploration with diverse restarts.
Key innovation: incremental coordinate descent that avoids full FFT recomputation,
enabling thousands of sweeps within the time budget.

import numpy as np
from scipy.fft import fft, ifft
import time


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = fft(a, size)
    return ifft(A * A).real[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(conv))


def spectral_init(n, rng, n_iters=300):
    """Initialize with approximately flat power spectrum (Gerchberg-Saxton style)."""
    a = rng.random(n)
    a = a / a.sum()
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for it in range(n_iters):
        A = fft(a, size)
        mag = np.abs(A)
        target = np.sqrt(np.mean(mag ** 2))
        # Compress dynamic range
        alpha = 0.7 + 0.3 * (it / n_iters)
        mag_new = target * (mag / target) ** alpha
        A_new = mag_new * np.exp(1j * np.angle(A))
        a_new = ifft(A_new).real[:n]
        a = np.maximum(a_new, 0)
        s = a.sum()
        if s < 1e-10:
            a = rng.random(n)
            s = a.sum()
        a = a / s

    return a

def smoothed_grad(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    conv = compute_conv_fft(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad_full = ifft(P * np.conj(A)).real
    return 2.0 * grad_full[:n]


def gd_polish(a_init, time_budget=60.0, tau_init=50.0, tau_end=2000.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = float(a.sum())
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    mu = 0.9  # momentum
    base_lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    stuck_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5

        # Nesterov extrapolation
        y = a + mu * v
        y = np.maximum(y, 0)
        s_y = float(y.sum())
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Gradient
        g = smoothed_grad(y, tau)
        g_norm = float(np.linalg.norm(g))

        # Adaptive step size
        lr = base_lr
        if g_norm > 1e-15:
            # Barzilai-Borwein style
            if prev_g is not None and it % 3 == 0:
                s_vec = y - prev_a
                y_vec = g - prev_g
                ss = float(np.dot(s_vec, s_vec))
                sy = float(np.dot(s_vec, y_vec))
                if ss > 1e-15 and abs(sy) > 1e-12:
                    bb_lr = ss / abs(sy)
                    bb_lr = min(max(bb_lr, 0.1 * base_lr), 3.0 * base_lr)
                    lr = 0.6 * lr + 0.4 * bb_lr
            adapt = 2.0 / (g_norm + 1e-8)
            lr = min(lr, adapt)

        v = mu * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            stuck_count += 1
            if stuck_count > 10:
                break
        else:
            a_new = a_new / s_new
            stuck_count = 0

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy() if g is not None else None
        a = a_new
        it += 1

        # Reset if stuck
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it

        # Decay lr slightly
        base_lr *= 0.99999

    if verbose:
        print(f"      GD: obj={best_obj:.10f}, iters={it}, tau_end={tau:.0f}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.05):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / (k ** 0.5)
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(a_new.sum())
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: spectral init + fast coordinate descent + GD polish.
    Explores multiple sequence lengths and diverse starting points.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0  # safety margin

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to use parent as starting point
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = spectral_init(n_init, rng, n_iters=200)
        best_score = compute_obj(best_seq)
        print(f"Spectral init (n={n_init}): obj={best_score:.10f}")

    # === Phase 1: Coordinate Descent on Parent ===
    print(f"\n=== Phase 1: CD on parent (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 400.0)
    if p1_budget > 30:
        a_cd = coord_desc(best_seq, max_sweeps=100000, time_budget=p1_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD parent: obj={best_score:.10f} **")
        else:
            print(f"  CD parent: obj={obj_cd:.10f} (no improvement)")

    # === Phase 2: GD Polish ===
    print(f"\n=== Phase 2: GD Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.1, 100.0)
    if p2_budget > 20:
        a_gd = gd_polish(best_seq, time_budget=p2_budget, tau_init=100.0, tau_end=1500.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  GD polish: obj={best_score:.10f} **")
        else:
            print(f"  GD polish: obj={obj_gd:.10f}")

    # === Phase 3: CD Again (refined) ===
    print(f"\n=== Phase 3: Second CD Round ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 250.0)
    if p3_budget > 30:
        a_cd2 = coord_desc(best_seq, max_sweeps=100000, time_budget=p3_budget, verbose=True)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  CD round 2: obj={best_score:.10f} **")
        else:
            print(f"  CD round 2: obj={obj_cd2:.10f}")

    # === Phase 4: Multi-n Exploration with Spectral Init ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = []
    for n_try in [500, 600, 700, n_best - 100, n_best + 100, n_best + 200,
                  1000, 1200, 1400, 1600, 2000]:
        if 300 <= n_try <= 3000 and n_try not in n_options:
            n_options.append(n_try)
    # Also try resampling the current best
    n_options.sort()
    n_options = [n for n in n_options if abs(n - n_best) > 50][:8]
    if not n_options:
        n_options = [n_best + 100, n_best - 100]

    print(f"  n options: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Try both spectral init and resampled
        a_sp = spectral_init(n_try, rng, n_iters=200)
        a_res = resample_sequence(best_seq, n_try)
        a_res = a_res / a_res.sum()

        # Pick the better initial
        obj_sp = compute_obj(a_sp)
        obj_res = compute_obj(a_res)
        a_try = a_sp if obj_sp < obj_res else a_res
        init_obj = min(obj_sp, obj_res)
        print(f"  n={n_try}: spectral={obj_sp:.6f}, resampled={obj_res:.6f}")

        # CD optimize
        alloc_cd = min(remaining * 0.3, 80.0)
        if alloc_cd > 10:
            a_try = coord_desc(a_try, max_sweeps=50000, time_budget=alloc_cd)

        # GD polish
        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.1, 40.0)
        if alloc_gd > 10:
            a_try = gd_polish(a_try, time_budget=alloc_gd, tau_init=100.0, tau_end=1000.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 5: Diverse Restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    restart_names = ["fourier", "mirror", "noise", "spectral_reinit"]
    for rname in restart_names:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if rname == "fourier":
            a_r = fourier_perturb(best_seq, rng, strength=0.06)
        elif rname == "mirror":
            a_r = (best_seq + best_seq[::-1]) / 2.0
            a_r = a_r / a_r.sum()
        elif rname == "noise":
            a_r = best_seq + 0.02 * rng.standard_normal(n_cur)
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        else:  # spectral_reinit
            a_r = spectral_init(n_cur, rng, n_iters=300)

        # Optimize
        alloc_cd = min(remaining * 0.25, 60.0)
        if alloc_cd > 10:
            a_r = coord_desc(a_r, max_sweeps=50000, time_budget=alloc_cd)

        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.08, 30.0)
        if alloc_gd > 8:
            a_r = gd_polish(a_r, time_budget=alloc_gd, tau_init=200.0, tau_end=1500.0)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({rname}): obj={best_score:.10f} **")
        else:
            print(f"  Restart ({rname}): obj={obj_r:.10f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final CD
    if remaining > 40:
        alloc = min(remaining * 0.4, 200.0)
        a_final = coord_desc(best_seq, max_sweeps=200000, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.10f} **")

    # Final high-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 100.0)
        a_final2 = gd_polish(best_seq, time_budget=alloc, tau_init=500.0, tau_end=5000.0, verbose=True)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final GD: obj={best_score:.10f} **")

    # Final CD after GD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.7, 80.0)
        a_final3 = coord_desc(best_seq, max_sweeps=200000, time_budget=alloc)
        obj_final3 = compute_obj(a_final3)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()
            print(f"  Final CD2: obj={best_score:.10f} **")

    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
