"""
Hybrid optimizer combining Log-Sum-Exp smoothed gradient descent with Nesterov
momentum, L-BFGS on smoothed objective, active-set SOCP trust regions, and
multi-length exploration. Key innovations: FFT-based exact gradient of the
log-sum-exp smoothing, Barzilai-Borwein adaptive step sizes, and iterative
temperature scheduling for progressive peak concentration.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, T, top_k=None):
    """
    Compute gradient of log-sum-exp of convolution peaks.
    Uses FFT for O(n log n) computation.
    Returns (gradient, LSE_value).
    """
    n = len(a)
    conv = compute_conv(a)
    
    if top_k is not None and top_k < len(conv):
        idx = np.argsort(conv)[::-1][:top_k]
        c_sel = conv[idx]
        p_sel = np.exp(T * (c_sel - c_sel[0]))
        p_sel /= p_sel.sum()
        # Build full p vector (sparse)
        p = np.zeros(len(conv))
        p[idx] = p_sel
        lse_val = float(c_sel[0] + np.log(p_sel.sum()) / T)
    else:
        c_max = conv[0]
        p = np.exp(T * (conv - c_max))
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros_like(conv)
            p[conv.argmax()] = 1.0
            p_sum = 1.0
        p /= p_sum
        lse_val = float(c_max + np.log(p_sum) / T)
    
    # Compute gradient using FFT
    # g[l] = sum_k p[k] * (a[k-l] + a[l-k]) for valid indices
    # = (p * a)[l] + (p_rev * a)[l] where p_rev[k] = p[2n-2-k]
    L = 1
    while L < 4 * n:
        L <<= 1
    
    a_pad = np.zeros(L)
    a_pad[:n] = a
    A_fft = fft(a_pad)
    
    p_pad = np.zeros(L)
    p_pad[:2*n-1] = p
    P_fft = fft(p_pad)
    
    # Term 1: sum_k p[k] * a[l-k] = (p * a)[l]
    conv1 = ifft(P_fft * A_fft).real[:n]
    
    # Term 2: sum_k p[k] * a[k-l] = (p_rev * a)[l] where p_rev[k]=p[2n-2-k]
    p_rev = p[::-1].copy()
    p_rev_pad = np.zeros(L)
    p_rev_pad[:2*n-1] = p_rev
    P_rev_fft = fft(p_rev_pad)
    conv2 = ifft(P_rev_fft * A_fft).real[:n]
    
    grad = 2.0 * (conv1 + conv2) / T  # The 1/T from LSE derivative... wait
    
    # Actually: d/d(a[l]) LSE = (1/T) * sum_k p_k * d(conv_k)/d(a[l])
    # where d(conv_k)/d(a[l]) = a[k-l] + a[l-k]
    # So grad = (1/T) * (conv1 + conv2) ... but conv1 and conv2 already include p weighting
    # Let me recheck: conv1[l] = sum_k p[k] * a[l-k], conv2[l] = sum_k p[k] * a[k-l]
    # So the gradient of LSE is (1/T) * (conv1 + conv2)? No wait.
    
    # LSE = (1/T) * log(sum_k exp(T * c_k)) where c_k = conv(a,a)[k]
    # d(LSE)/d(a[l]) = (1/T) * sum_k [exp(T*c_k)/sum(exp(T*c_k))] * d(c_k)/d(a[l])
    # = (1/T) * sum_k p_k * (a[k-l] + a[l-k])
    # = (1/T) * (conv1[l] + conv2[l])
    
    grad = (conv1 + conv2) / T
    
    return grad, lse_val


def lse_gradient_direct(a, T, top_k=None):
    """Direct O(n^2) gradient computation (for verification or small n)."""
    n = len(a)
    conv = compute_conv(a)
    
    if top_k is not None and top_k < len(conv):
        idx = np.argsort(conv)[::-1][:top_k]
        c_sel = conv[idx]
        pm = c_sel[0]
        w = np.exp(T * (c_sel - pm))
        w /= w.sum()
        p = np.zeros(len(conv))
        p[idx] = w
        lse_val = float(pm + np.log(w.sum()) / T)
    else:
        c_max = conv.max()
        w = np.exp(T * (conv - c_max))
        wsum = w.sum()
        if wsum < 1e-300:
            p = np.zeros_like(conv)
            p[conv.argmax()] = 1.0
            lse_val = float(c_max)
        else:
            p = w / wsum
            lse_val = float(c_max + np.log(wsum) / T)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-18:
            continue
        pk = p[k]
        # a[k-l] nonzero for l in [max(0,k-n+1), min(n-1, k)]
        ls = max(0, k - n + 1)
        le = min(n - 1, k)
        if ls <= le:
            # For l in [ls, le]: a[k-l] = a[k-le : k-ls+1] reversed
            grad[ls:le+1] += pk * a[k-le:k-ls+1][::-1]
        # a[l-k] nonzero for l in [max(0,k), min(n-1, k+n-1)]
        ls2 = k
        le2 = min(n - 1, k + n - 1)
        if ls2 <= le2:
            grad[ls2:le2+1] += pk * a[0:le2-ls2+1]
    
    grad /= T
    return grad, lse_val


def lse_nesterov(a, time_budget=120.0, T_start=30, T_end=600, 
                  momentum_start=0.85, momentum_end=0.95,
                  top_k=200, verbose=True, seed=42):
    """
    Nesterov-accelerated gradient descent on the log-sum-exp smoothed objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    
    # Initial step size
    conv0 = compute_conv(a)
    g0, _ = lse_gradient(a, T_start, top_k)
    g0_norm = np.linalg.norm(g0)
    if g0_norm < 1e-15:
        return best_a
    lr = 1.0 / (g0_norm * np.sqrt(n) * 0.5)
    
    # Nesterov state
    y = a.copy()
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = g0.copy()
    
    # BB step size state
    s_prev = None
    y_prev = None
    
    rng = np.random.default_rng(seed)
    no_improve = 0
    T = T_start
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Update temperature (logarithmic increase)
        T = T_start * (T_end / T_start) ** (progress ** 0.7)
        
        # Update momentum (increase slowly)
        beta = momentum_start + (momentum_end - momentum_start) * progress
        
        # Nesterov look-ahead
        v = beta * v + (1 - beta) * (a - prev_a)
        y = a + v
        
        # Gradient at y
        g, lse_val = lse_gradient(y, T, top_k)
        
        # Barzilai-Borwein step size
        if s_prev is not None and y_prev is not None:
            s_bb = g - y_prev
            denom = np.dot(s_bb, s_bb)
            num = np.dot(a - prev_a, s_bb)
            if denom > 1e-20:
                lr_bb = num / denom
                lr_bb = np.clip(lr_bb, 1e-10, 10.0 / g0_norm)
                # Mix with current lr
                lr = 0.5 * lr + 0.5 * lr_bb
        s_prev = g - y_prev
        y_prev = g.copy()
        prev_g = g.copy()
        
        # Gradient step
        a_new = y - lr * g
        a_new = project_simplex(a_new)
        
        # Evaluate
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                # Perturb
                scale = 0.01 + 0.005 * min(no_improve // 100, 5)
                a_new = best_a * np.exp(rng.normal(0, scale, n))
                a_new = project_simplex(a_new)
                no_improve = 0
                s_prev = None
                v = np.zeros(n)
        
        a = a_new
        prev_a = a_new
        
        # Decay learning rate slightly
        lr *= 0.999
        it += 1
        
        if verbose and it % 200 == 0:
            print(f"      LSE-Nest it{it}: T={T:.1f}, obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
    
    return best_a


def lse_lbfgs(a, time_budget=300.0, T_start=50, T_end=800, top_k=150,
              max_iter_per_call=80, verbose=True, seed=42):
    """
    L-BFGS optimization of the log-sum-exp smoothed objective.
    Multiple calls with increasing temperature.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    T = T_start
    call_count = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = T_start * (T_end / T_start) ** (progress ** 0.6)
        
        # Define objective and gradient for scipy
        def obj_and_grad(x):
            x = np.maximum(x, 0)
            sx = x.sum()
            if sx < 1e-10:
                return 1e10, np.zeros(n)
            x = x / sx
            # LSE value + penalty for sum constraint
            g_lse, lse_val = lse_gradient(x, T, top_k)
            # Penalty for deviation from simplex
            penalty = 1e4 * (sx - 1.0) ** 2
            penalty_grad = 2e4 * (sx - 1.0) / n * np.ones(n)
            # Additional smooth penalty for things that are 0
            small_penalty = 1e6 * np.sum(np.minimum(x, 0.0)) ** 2
            val = lse_val + penalty + small_penalty
            grad = g_lse + penalty_grad
            return val, grad
        
        result = scipy_minimize(
            obj_and_grad, a, method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options={'maxiter': max_iter_per_call, 'maxfun': max_iter_per_call * 10,
                     'ftol': 1e-12, 'gtol': 1e-8}
        )
        
        a_try = np.maximum(result.x, 0)
        s_try = a_try.sum()
        if s_try > 1e-10:
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                call_count += 1
                if verbose and call_count % 3 == 0:
                    print(f"      LSE-LBFGS call{call_count}: T={T:.1f}, obj={best_obj:.8f}, "
                          f"t={time.monotonic()-start:.0f}s")
            a = a_try
        else:
            a = best_a
        
        # If stuck, perturb
        if result.fun > 1e6 or not result.success:
            scale = 0.005 + 0.003 * min(call_count, 10)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = project_simplex(a)
        
        call_count += 1
    
    return best_a


def solve_tr_sscp(a, r, verbose=False):
    """
    Trust region SOCP: linearized peak reduction.
    min M s.t. C_old[k] + 2*(A_cross@delta)[k] <= M, ||delta||_2 <= r,
              sum(a+delta)=1, a+delta>=0
    """
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)
    
    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta
    
    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
        cp.norm(delta, 2) <= r,
    ]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None


def solve_active_set_lp(a, top_k=10, M_margin=1e-4):
    """
    Solve LP to find direction that reduces all top peaks simultaneously.
    min t s.t. 2*(A_k a)@d <= -t for k in active set,
              sum(d) = 0, a+d >= 0, ||d||_inf <= 0.1
    """
    n = len(a)
    a = a / a.sum()
    conv = compute_conv(a)
    max_conv = conv.max()
    
    # Active set: peaks within margin of max
    active_idx = np.where(conv >= max_conv - M_margin * max_conv)[0]
    if len(active_idx) > top_k:
        active_idx = np.argsort(conv)[::-1][:top_k]
    active_idx = sorted(set(active_idx.tolist()))
    
    if len(active_idx) == 0:
        return None
    
    A_cross = build_conv_matrix(a)
    d = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [cp.sum(d) == 0, d >= -a + 1e-6, d <= 0.05]
    for k in active_idx:
        constraints.append(2 * (A_cross[k] @ d) <= -t)
    
    prob = cp.Problem(cp.Maximize(t), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if d.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                d_val = np.asarray(d.value).flatten()
                if t.value is not None and t.value > 1e-6:
                    return d_val
        except Exception:
            continue
    return None


def active_set_polish(a, time_budget=300.0, verbose=True, seed=42):
    """Active-set LP direction finding with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        d = solve_active_set_lp(a, top_k=15, M_margin=5e-4)
        
        if d is None:
            no_improve += 1
            if no_improve >= 5:
                # Perturb
                scale = 0.005 + 0.004 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = project_simplex(a)
                no_improve = 0
                if it > 50:
                    break
            it += 1
            continue
        
        # Line search
        best_alpha = 0
        best_alpha_obj = best_obj
        for alpha in [0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]:
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj - 1e-15:
                best_alpha_obj = obj_try
                best_alpha = alpha
        
        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-15:
            a = a + best_alpha * d
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_obj = best_alpha_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 10 == 0:
                print(f"      ActiveSet it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
        else:
            no_improve += 1
        
        it += 1
    
    return best_a


def tr_sscp_polish(a, time_budget=200.0, r_init=0.3, verbose=False, seed=42):
    """Trust region SOCP polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    r = r_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        delta_dir, M_val = solve_tr_sscp(a, r)
        
        if delta_dir is None:
            break
        
        norm_d = np.linalg.norm(delta_dir)
        if norm_d > r:
            delta_dir = delta_dir * (r / norm_d)
        
        best_alpha = 0
        best_alpha_obj = best_obj
        for t_alpha in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0, 1.5]:
            a_try = a + t_alpha * delta_dir
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj - 1e-15:
                best_alpha_obj = obj_try
                best_alpha = t_alpha
        
        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-15:
            a = a + best_alpha * delta_dir
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_obj = best_alpha_obj
            best_a = a.copy()
            no_improve = 0
            r = min(r * 1.1, 1.5)
        else:
            no_improve += 1
            r *= 0.6
            if no_improve >= 6:
                if r < 0.05:
                    a = best_a * np.exp(rng.normal(0, 0.01, n))
                    a = project_simplex(a)
                    r = r_init
                    no_improve = 0
                    if it > 60:
                        break
                else:
                    r = max(r * 0.5, 0.05)
        
        it += 1
        if verbose and it % 10 == 0:
            print(f"      TR it{it}: obj={best_obj:.8f}, r={r:.4f}, t={time.monotonic()-start:.0f}s")
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Multi-phase hybrid optimizer.
    Phase 1: LSE-Nesterov (fast initial reduction)
    Phase 2: LSE-LBFGS (main optimization)
    Phase 3: Active-set LP polish
    Phase 4: TR-SOCP polish
    Phase 5: Multi-length exploration
    Phase 6: Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")
    
    n_current = len(best_seq)
    
    # === Phase 1: LSE-Nesterov warmup ===
    print(f"\n=== Phase 1: LSE-Nesterov (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 100.0)
    if p1_budget > 20:
        a_p1 = lse_nesterov(best_seq, time_budget=p1_budget,
                            T_start=25, T_end=400, top_k=150,
                            verbose=True, seed=42)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase1 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 result: obj={obj_p1:.8f}")
    
    # === Phase 2: LSE-LBFGS (main optimization) ===
    print(f"\n=== Phase 2: LSE-LBFGS (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.30, 450.0)
    if p2_budget > 40:
        a_p2 = lse_lbfgs(best_seq, time_budget=p2_budget,
                         T_start=40, T_end=600, top_k=120,
                         max_iter_per_call=60, verbose=True, seed=77)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_score:
            best_score = obj_p2
            best_seq = a_p2.copy()
            print(f"  Phase2 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 result: obj={obj_p2:.8f}")
    
    # === Phase 3: Active-set LP polish ===
    print(f"\n=== Phase 3: Active-Set LP (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)
    if p3_budget > 30:
        a_p3 = active_set_polish(best_seq, time_budget=p3_budget,
                                 verbose=True, seed=123)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_score:
            best_score = obj_p3
            best_seq = a_p3.copy()
            print(f"  Phase3 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 result: obj={obj_p3:.8f}")
    
    # === Phase 4: TR-SOCP polish ===
    print(f"\n=== Phase 4: TR-SOCP (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.12, 150.0)
    if p4_budget > 25:
        a_p4 = tr_sscp_polish(best_seq, time_budget=p4_budget,
                              r_init=0.25, verbose=True, seed=200)
        obj_p4 = compute_obj(a_p4)
        if obj_p4 < best_score:
            best_score = obj_p4
            best_seq = a_p4.copy()
            print(f"  Phase4 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 result: obj={obj_p4:.8f}")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    
    n_options = sorted(set([
        max(400, n_current - 200),
        max(400, n_current - 100),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        1000, 1200, 1400, 1600
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_current) > 50]
    
    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)
        
        # Quick LSE-Nesterov
        alloc1 = min(remaining * 0.10, 40.0)
        if alloc1 > 10:
            a_try = lse_nesterov(a_try, time_budget=alloc1,
                                 T_start=20, T_end=300, top_k=100,
                                 verbose=False, seed=300 + n_try)
        
        # LSE-LBFGS
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.12, 80.0)
        if alloc2 > 15:
            a_try = lse_lbfgs(a_try, time_budget=alloc2,
                              T_start=30, T_end=400, top_k=100,
                              max_iter_per_call=50, verbose=False, seed=500 + n_try)
        
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    
    # Run all phases briefly again
    if remaining > 80:
        # LSE-LBFGS final
        alloc = min(remaining * 0.35, 150.0)
        if alloc > 20:
            a_f = lse_lbfgs(best_seq, time_budget=alloc,
                            T_start=100, T_end=1200, top_k=80,
                            max_iter_per_call=50, verbose=True, seed=999)
            obj_f = compute_obj(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final LBFGS: obj={best_score:.8f} **")
        
        # Active set
        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc = min(remaining * 0.30, 100.0)
            if alloc > 20:
                a_f = active_set_polish(best_seq, time_budget=alloc,
                                        verbose=True, seed=1111)
                obj_f = compute_obj(a_f)
                if obj_f < best_score:
                    best_score = obj_f
                    best_seq = a_f.copy()
                    print(f"  Final ActiveSet: obj={best_score:.8f} **")
        
        # TR-SOCP
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.25, 80.0)
            if alloc > 15:
                a_f = tr_sscp_polish(best_seq, time_budget=alloc,
                                     r_init=0.15, verbose=True, seed=2222)
                obj_f = compute_obj(a_f)
                if obj_f < best_score:
                    best_score = obj_f
                    best_seq = a_f.copy()
                    print(f"  Final TR: obj={best_score:.8f} **")
    
    # === Phase 7: Small perturbations + quick local search ===
    print(f"\n=== Phase 7: Perturbation Search ===")
    remaining = deadline - time.monotonic()
    
    for r_idx in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        
        scale = 0.003 + 0.002 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = project_simplex(a_pert)
        
        # Quick LSE-Nesterov
        alloc = min(remaining * 0.15, 40.0)
        if alloc > 8:
            a_pert = lse_nesterov(a_pert, time_budget=alloc,
                                  T_start=100, T_end=800, top_k=80,
                                  verbose=False, seed=3000 + r_idx)
        
        obj_pert = compute_obj(a_pert)
        if obj_pert < best_score:
            best_score = obj_pert
            best_seq = a_pert.copy()
            print(f"  Pert{r_idx} (s={scale:.4f}): obj={best_score:.8f} **")
        else:
            print(f"  Pert{r_idx} (s={scale:.4f}): obj={obj_pert:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
