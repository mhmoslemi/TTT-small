"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid approach: LP alternating minimization with peak-directed
perturbation, exponential weights gradient descent, and multi-length
exploration. Key differences from previous approach: (1) structured
initializations based on frequency-domain flatness, (2) peak-directed
perturbations that target specific convolution peaks, (3) exponential
weights (mirror descent) for non-negativity-preserving updates, (4)
adaptive restart strategy.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of (a*a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def exponential_weights_gd(a, time_budget, tau_init=50.0, tau_max=3000.0):
    """
    Exponential weights (mirror descent) on the smoothed objective.
    Update: a_new[j] = a[j] * exp(-eta * grad[j]), then normalize.
    This naturally preserves non-negativity.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 1.0 / np.sqrt(n)
    tau = tau_init
    it = 0
    patience = 0
    best_patience = patience

    while time.monotonic() - start < time_budget:
        g = smoothed_gradient(a, tau)
        
        # Exponential weights update
        log_updates = -eta * g
        # For numerical stability
        log_updates -= np.max(log_updates)
        updates = np.exp(log_updates)
        a_new = a * updates
        s_new = float(np.sum(a_new))
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new

        # Adaptive tuning
        eta *= 0.999
        tau = min(tau * 1.05, tau_max)
        it += 1

        # Restart if stuck
        if patience > 200:
            # Perturb and reset
            rng = np.random.default_rng(it)
            a = best_a * (1 + 0.01 * rng.standard_normal(n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            eta = 1.0 / np.sqrt(n)
            tau = tau_init * 2
            patience = 0
            best_patience = patience

        if it > 5000 and patience > best_patience + 500:
            break

    return best_a


def peak_directed_perturb(a, rng, strength=0.02):
    """
    Perturb a in a direction that targets the current peaks of the convolution.
    Positions that contribute most to peaks are decreased, others increased.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    # Find top peak positions
    top_k = max(3, n // 50)
    peak_positions = np.argsort(conv)[-top_k:]
    
    # Compute sensitivity: for each position j, how much does it contribute to peaks?
    sensitivity = np.zeros(n)
    for k in peak_positions:
        # conv[k] = sum_j a[j] * a[k-j] (with 0<=j<n, 0<=k-j<n)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            sensitivity[j] += a[j] * a[k - j]
    
    # Normalize sensitivity
    s_max = float(np.max(sensitivity))
    if s_max > 1e-15:
        sensitivity = sensitivity / s_max
    
    # Create perturbation: decrease high-sensitivity positions, increase low-sensitivity
    pert = -strength * sensitivity + strength * 0.5 * rng.standard_normal(n) * 0.1
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a.copy()
    return a_new / s_new


def frequency_flat_init(n, rng):
    """
    Initialize a sequence with a flat power spectrum (as much as possible
    while maintaining non-negativity and sum=1).
    """
    # Start with random phases in frequency domain
    N = 1
    while N < 2 * n:
        N <<= 1
    A = np.zeros(N, dtype=complex)
    A[0] = np.sqrt(n)  # DC component for sum
    for k in range(1, N // 2):
        A[k] = np.sqrt(2.0 / n) * complex(rng.standard_normal(), rng.standard_normal())
        A[N - k] = np.conj(A[k])
    
    a = irfft(A, N)[:n]
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def fourier_perturb(a, rng, strength=0.03):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(8, n // 4)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0], dtype=np.float64)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent_peaks(a, time_budget, seed=42):
    """
    Coordinate descent that specifically targets reducing convolution peaks.
    At each step, identify the peak position k*, then find the pair (i, j)
    with i+j=k* (or i-j=k* depending on convention) that contributes most,
    and reduce one of them while increasing another that doesn't contribute.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        peak_pos = int(np.argmax(conv))

        # Find the pair (i, j) with i+j = peak_pos that maximizes a[i]*a[j]
        # conv[k] = sum_{j} a[j] * a[k-j] where 0<=j<n and 0<=k-j<n
        best_pair_product = -1
        best_i, best_j = -1, -1
        j_min = max(0, peak_pos - n + 1)
        j_max = min(n - 1, peak_pos)
        for j in range(j_min, j_max + 1):
            prod = a[j] * a[peak_pos - j]
            if prod > best_pair_product:
                best_pair_product = prod
                best_i = j
                best_j = peak_pos - j

        if best_i < 0:
            break

        # Reduce the larger of the two, increase a random position that
        # doesn't contribute to the peak
        if a[best_i] >= a[best_j]:
            reduce_idx = best_i
        else:
            reduce_idx = best_j

        # Find a position to increase (one that doesn't contribute to this peak)
        contributing = set()
        for k in range(max(0, peak_pos - n + 1), min(2*n-1, peak_pos + n)):
            pass  # This is complex, let's just pick randomly

        # Pick a random position to increase
        increase_idx = rng.integers(0, n)
        while increase_idx == reduce_idx:
            increase_idx = rng.integers(0, n)

        # Transfer mass
        delta = 0.01 * a[reduce_idx]
        if delta >= a[reduce_idx]:
            delta = 0.5 * a[reduce_idx]

        a_new = a.copy()
        a_new[reduce_idx] -= delta
        a_new[increase_idx] += delta
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller delta
            delta = 0.001 * a[reduce_idx]
            a_new = a.copy()
            a_new[reduce_idx] -= delta
            a_new[increase_idx] += delta
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new

    return best_a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid strategy: (1) multi-start initialization with frequency-flat and
random sequences, (2) enhanced LP alternating minimization with peak-directed
perturbation and adaptive blending, (3) exponential weights (mirror descent)
gradient method for non-negativity-preserving polishing, (4) systematic
exploration of nearby sequence lengths. Peak-directed perturbations specifically
target positions contributing most to convolution peaks, enabling escape from
local fixed points of the best-response dynamics.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    cols = np.arange(n)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_sum = float(np.sum(a))
    if a_sum < 1e-15:
        return None, None
    a_norm = a / a_sum
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of (a*a) via FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g

    Exponential weights (mirror descent) on the smoothed objective.
    Update: a_new[j] = a[j] * exp(-eta * grad[j]), then normalize.
    Naturally preserves non-negativity.

    a = a / s

    best_obj = compute_obj(a)

    it = 0

    max_patience = 300
    restart_count = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        g = smoothed_gradient(a, tau)

        log_updates = -eta * g
        log_updates -= float(np.max(log_updates))
        updates = np.exp(log_updates)
        a_new = a * updates
        s_new = float(np.sum(a_new))
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:

        eta *= 0.9995
        tau = min(tau * 1.04, tau_max)
        it += 1

            restart_count += 1
            strength = 0.01 + 0.005 * min(restart_count, 5)
            a = fourier_perturb(best_a, rng, strength=strength)
            eta = 1.5 / np.sqrt(n)
            tau = max(tau_init, tau * 0.5)
            patience = 0

        if it % 5000 == 0 and compute_obj(a) > best_obj + 1e-5:
            a = best_a.copy()
            eta = 1.0 / np.sqrt(n)

    return best_a


def peak_directed_perturb(a, rng, strength=0.02):
    """
    Perturb a targeting positions that contribute most to convolution peaks.
    Decreases high-sensitivity positions, increases low-sensitivity ones.
    """
    n = len(a)
    conv = compute_conv(a)
    top_k = max(3, n // 40)
    peak_positions = np.argsort(conv)[-top_k:]

    sensitivity = np.zeros(n)
    for k in peak_positions:
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            js = np.arange(j_min, j_max + 1)
            sensitivity[js] += a[js] * a[k - js]

    s_max = float(np.max(sensitivity))
    if s_max > 1e-15:
        sensitivity = sensitivity / s_max

    noise = strength * 0.05 * rng.standard_normal(n)
    pert = -strength * sensitivity + noise
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a.copy()
    return a_new / s_new


def fourier_perturb(a, rng, strength=0.03):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(10, max(3, n // 10))
    for k in range(1, n_modes + 1):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.2) / np.sqrt(k)
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_alternate_enhanced(a, time_budget, seed=42, verbose=False):
    """
    Enhanced LP alternating minimization with:
    - Peak-directed perturbation when stuck
    - Adaptive blending with hysteresis and full-jump exploration
    - Multiple solver fallback
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    t_blend = 0.4
    stall_count = 0
    long_stall = 0
    lp_fail_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break

        g_lp, M_lp = solve_conv_lp(a)
        if g_lp is None:
            lp_fail_count += 1
            if lp_fail_count > 5:
                break
            a = fourier_perturb(best_a, rng, strength=0.02)
            continue

        lp_fail_count = 0
        iter_count += 1

        # Build candidate blending factors
        candidates_t = [t_blend, max(0.05, t_blend * 0.5), min(1.0, t_blend * 1.8)]
        # Periodically try full jump
        if iter_count % 10 == 0:
            candidates_t.append(1.0)
        if stall_count > 5:
            candidates_t.extend([0.05, 0.1, 0.2])

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend

        for t in candidates_t:
            t = min(max(t, 0.005), 1.0)
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            t_blend = max(0.08, best_local_t * 0.92)
            stall_count = 0
            long_stall = 0
        else:
            stall_count += 1
            long_stall += 1

            if stall_count >= 2:
                strength = 0.01 + 0.004 * min(stall_count, 8)
                a = peak_directed_perturb(best_a, rng, strength=strength)
                t_blend = max(0.15, min(t_blend * 1.2, 0.7))
                stall_count = 0
            elif stall_count == 1:
                nudge = 0.002 * rng.standard_normal(n)
                a = best_a * (1.0 + nudge)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s

            if long_stall >= 25:
                a = fourier_perturb(best_a, rng, strength=0.04)
                long_stall = 0
                stall_count = 0
                t_blend = 0.5

        if verbose and iter_count % 30 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP {iter_count}: obj={best_obj:.10f}, t={t_blend:.3f}, "
                  f"elapsed={elapsed:.0f}s/{time_budget:.0f}s")

    if verbose:
        print(f"    LP enhanced: {iter_count} iters, obj={best_obj:.10f}")

    return best_a


def frequency_flat_init(n, rng):
    """
    Initialize with approximately flat power spectrum, projected to non-negative.
    This gives a good starting point for peak minimization.
    """
    N = 1
    while N < 2 * n:
        N <<= 1
    A = np.zeros(N, dtype=complex)
    A[0] = np.sqrt(float(n))
    for k in range(1, N // 2):
        A[k] = np.sqrt(2.0 / float(n)) * complex(rng.standard_normal(), rng.standard_normal())
        A[N - k] = np.conj(A[k])

    a = irfft(A, N)[:n].copy()
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]), dtype=np.float64)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def multi_start_lp(a_list, time_budget_per_start, seeds, verbose=False):
    """
    Run short LP alternating on multiple initializations, return the best.
    """
    best_a = None
    best_obj = np.inf

    for i, a_init in enumerate(a_list):
        remaining = time_budget_per_start * len(a_list) - (time.monotonic() - multi_start_lp._start)
        if remaining < 30:
            break
        alloc = min(time_budget_per_start, remaining - 10)
        if alloc < 10:
            break

        seed = seeds[i % len(seeds)]
        a_result = lp_alternate_enhanced(a_init, time_budget=alloc, seed=seed, verbose=False)
        obj = compute_obj(a_result)

        if obj < best_obj:
            best_obj = obj
            best_a = a_result.copy()
            if verbose:
                print(f"    Multi-start {i}: obj={obj:.10f} **")
        else:
            if verbose:
                print(f"    Multi-start {i}: obj={obj:.10f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-start LP + exponential weights GD + n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 12.0
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        a_parent = np.asarray(height_sequence_1, dtype=np.float64)
        a_parent = np.maximum(a_parent, 0)
        if a_parent.size == 0 or a_parent.sum() < 0.01:
            raise ValueError("Invalid parent")
        a_parent = a_parent / a_parent.sum()
        n_init = a_parent.size
        print(f"Parent: n={n_init}, obj={compute_obj(a_parent):.10f}")
    except Exception:
        n_init = 800
        a_parent = np.ones(n_init, dtype=np.float64) / n_init
        print(f"Default init: n={n_init}")

    # === Phase 1: Multi-start initialization ===
    print(f"\n=== Phase 1: Multi-Start LP (budget={int(budget_s*0.35)}s) ===")
    multi_start_lp._start = time.monotonic()

    n_values = [n_init]
    # Add a few other n values for diversity
    for dn in [-50, 50, -100, 100]:
        nv = n_init + dn
        if 400 <= nv <= 1500 and nv not in n_values:
            n_values.append(nv)

    initializations = []
    seeds = []
    for n_val in n_values[:5]:
        # Resampled parent
        if n_val != n_init:
            a_res = resample_to_length(a_parent, n_val)
        else:
            a_res = a_parent.copy()
        initializations.append(a_res)
        seeds.append(seed + 1)

        # Frequency-flat init
        a_ff = frequency_flat_init(n_val, rng)
        initializations.append(a_ff)
        seeds.append(seed + 100)

        # Random init
        a_rand = rng.dirichlet(np.ones(n_val))
        initializations.append(a_rand)
        seeds.append(seed + 200)

    p1_budget_per_start = max(15, (budget_s * 0.35 - (time.monotonic() - start_time)) / len(initializations))
    best_a, best_obj = multi_start_lp(initializations, p1_budget_per_start, seeds, verbose=True)
    print(f"  Phase 1 best: obj={best_obj:.10f}, n={len(best_a)}")

    # === Phase 2: Deep LP on the best ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: Deep LP (budget={int(min(remaining*0.35, 400))}s) ===")
        p2_budget = min(remaining * 0.35, 400.0)
        a_p2 = lp_alternate_enhanced(best_a, time_budget=p2_budget, seed=seed+500, verbose=True)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_obj:
            best_obj = obj_p2
            best_a = a_p2.copy()
            print(f"  Phase 2: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 2: obj={obj_p2:.10f}")

    # === Phase 3: Exponential weights GD ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 3: Exp-Weights GD ===")
        p3_budget = min(remaining * 0.15, 150.0)
        a_p3 = exponential_weights_gd(best_a, time_budget=p3_budget, tau_init=30.0, tau_max=2000.0)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3.copy()
            print(f"  Phase 3: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 3: obj={obj_p3:.10f}")

    # === Phase 4: Second LP with perturbation ===
    remaining = deadline - time.monotonic()

        a_pert = peak_directed_perturb(best_a, rng, strength=0.02)

        obj_p4 = compute_obj(a_p4)
        if obj_p4 < best_obj:
            best_obj = obj_p4
            best_a = a_p4.copy()
            print(f"  Phase 4: obj={best_obj:.10f} **")
        else:

    # === Phase 5: n exploration ===
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    print(f"\n=== Phase 5: n Exploration (best n={n_best}) ===")

    n_options = []
    for dn in [-30, -20, -10, -5, 5, 10, 20, 30, 50]:
        nv = n_best + dn
        if 350 <= nv <= 1800:
            n_options.append(nv)
    n_options = sorted(set(n_options))

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()

            break
        a_try = resample_to_length(best_a, n_try)
        alloc = min(remaining * 0.15, 80.0)

            break

        a_try = lp_alternate_enhanced(a_try, time_budget=alloc * 0.75, seed=n_try, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 15:

                                            tau_init=50.0, tau_max=1500.0)

        else:

    # === Phase 6: Final deep LP ===

        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_obj:

            best_a = a_final2.copy()

    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
