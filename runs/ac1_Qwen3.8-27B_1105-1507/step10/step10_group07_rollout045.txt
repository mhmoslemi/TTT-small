from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(a, M_target=None):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_value) or (None, None).
    """
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_block_lp(a, block_start, block_end, M_target=None):
    """
    Solve LP where only variables in [block_start, block_end) are free.
    Other variables are fixed at their current values.
    """
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g_free = cp.Variable(block_end - block_start, nonneg=True)
    M = cp.Variable()

    # Build full g vector conceptually: g[j] = a[j] for j not in block, g_free for j in block
    # conv(a, g)[k] = sum_j A[k,j] * g[j]
    # = sum_{j in block} A[k,j] * g_free[j-block_start] + sum_{j not in block} A[k,j] * a[j]

    A_block = A[:, block_start:block_end]
    A_fixed = A.copy()
    A_fixed[:, block_start:block_end] = 0
    fixed_part = A_fixed @ a

    constraints = [A_block @ g_free + fixed_part <= M]

    # sum(g) = 1 => sum(g_free) + sum(a[j] for j not in block) = 1
    fixed_sum = float(np.sum(a) - np.sum(a[block_start:block_end]))
    constraints.append(cp.sum(g_free) == 1 - fixed_sum)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_free.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = a.copy()
                g_val[block_start:block_end] = np.maximum(g_free.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def parabolic_line_search(a, direction, obj_current, max_iter=8):
    """
    Parabolic interpolation line search for a + t * direction.
    Returns (best_t, best_obj).
    """
    best_t = 0.0
    best_obj = obj_current

    # First, find bracket
    t1 = 0.1
    obj1 = obj_current
    t2 = 0.3
    a_try = np.maximum(a + t2 * direction, 0)
    s_try = a_try.sum()
    if s_try > 1e-10:
        a_try = a_try / s_try
        obj2 = compute_obj(a_try)
    else:
        obj2 = obj_current

    if obj2 >= obj1:
        # Function is increasing, try smaller step
        t2 = 0.1
        t1 = 0.03
        a_try = np.maximum(a + t2 * direction, 0)
        s_try = a_try.sum()
        if s_try > 1e-10:
            a_try = a_try / s_try
            obj2 = compute_obj(a_try)
        else:
            obj2 = obj_current
        if obj2 >= obj1:
            return best_t, best_obj

    # Now we have obj(t1) > obj(t2), expand t2
    t3 = t2 * 2
    for _ in range(5):
        a_try = np.maximum(a + t3 * direction, 0)
        s_try = a_try.sum()
        if s_try > 1e-10:
            a_try = a_try / s_try
            obj3 = compute_obj(a_try)
        else:
            obj3 = obj_current
        if obj3 > obj2:
            break
        t1, obj1 = t2, obj2
        t2, obj2 = t3, obj3
        t3 = t2 * 1.8
        if t3 > 5.0:
            break

    # Parabolic interpolation within [t1, t2] or [t2, t3]
    ts = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.8, 1.0, 1.5, 2.0]
    for t in ts:
        a_try = np.maximum(a + t * direction, 0)
        s_try = a_try.sum()
        if s_try <= 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t

    return best_t, best_obj


def lp_polish(a, time_budget=300.0, seed=42, verbose=False):
    """Iterated LP polishing with parabolic line search and adaptive perturbation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        direction = g_lp - a
        best_t, best_t_obj = parabolic_line_search(a, direction, best_obj)

        if best_t > 0 and best_t_obj < best_obj - 1e-14:
            a = a + best_t * direction
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-10:
                a = a / s_a
            best_obj = best_t_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                # Adaptive perturbation: scale increases with stall
                scale = 0.005 + 0.004 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                no_improve = 0
                if it > 50:
                    break

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def freq_perturb(a, rng, smoothness=1.0):
    """
    Perturb in frequency domain for different types of perturbations.
    smoothness < 1: rough perturbations (high freq dominant)
    smoothness > 1: smooth perturbations (low freq dominant)
    """
    n = len(a)
    N = 1
    while N < n:
        N <<= 1
    A = fft(a, N)
    # Perturb magnitude and phase
    freqs = np.arange(N)
    # Taper to control smoothness
    taper = np.exp(-freqs / (n * smoothness))
    noise_mag = rng.normal(0, 0.05, N) * taper
    noise_phase = rng.normal(0, 0.05, N) * taper
    # Keep DC component fixed
    noise_mag[0] = 0
    noise_phase[0] = 0
    # Alphabetical: preserve conjugate symmetry for real result
    noise_mag[N // 2:] = noise_mag[N // 2:N // 2][::-1].copy() if N % 2 == 0 else noise_mag[N // 2:].copy()
    noise_phase[N // 2:] = noise_phase[N // 2:N // 2][::-1].copy() if N % 2 == 0 else noise_phase[N // 2:].copy()
    # Actually, let's just add real noise in frequency domain properly
    A_new = A.copy()
    for k in range(1, N // 2):
        A_new[k] = A[k] * (1 + noise_mag[k]) * np.exp(1j * noise_phase[k])
        A_new[N - k] = np.conj(A_new[k])
    a_new = ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def make_palindromic(a):
    """Make sequence palindromic: a[i] = a[n-1-i]."""
    n = len(a)
    a_pal = np.zeros(n)
    half = n // 2
    for i in range(half):
        a_pal[i] = 0.5 * (a[i] + a[n - 1 - i])
        a_pal[n - 1 - i] = a_pal[i]
    if n % 2 == 1:
        a_pal[half] = a[half]
    s = a_pal.sum()
    if s > 1e-10:
        a_pal = a_pal / s
    return a_pal


def make_two_valued(a, rng):
    """Create a two-valued sequence from the structure of a."""
    n = len(a)
    threshold = np.median(a)
    a_two = np.where(a > threshold, 1.0, 0.3).astype(np.float64)
    # Add small random variation
    a_two = a_two * np.exp(rng.normal(0, 0.02, n))
    a_two = np.maximum(a_two, 0)
    s = a_two.sum()
    if s > 1e-10:
        a_two = a_two / s
    return a_two


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def block_lp_polish(a, time_budget=120.0, n_blocks=4, seed=42):
    """Block coordinate LP polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    block_size = n // n_blocks
    block_starts = [i * block_size for i in range(n_blocks)]
    if block_starts[-1] + block_size < n:
        block_starts.append(n - block_size)

    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        improved = False
        for bs in block_starts:
            be = min(bs + block_size, n)
            if be - bs < 2:
                continue
            g_block, M_val = solve_block_lp(a, bs, be)
            if g_block is None:
                continue
            direction = g_block - a
            best_t, best_t_obj = parabolic_line_search(a, direction, best_obj, max_iter=5)
            if best_t > 0 and best_t_obj < best_obj - 1e-12:
                a = a + best_t * direction
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                best_obj = best_t_obj
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            scale = 0.008 + 0.003 * no_improve
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-10:
                a = a / s_a
            if no_improve >= 8:
                break
        else:
            no_improve = 0
        it += 1

    return best_a


def annealing_search(a, time_budget=400.0, seed=42, verbose=False):
    """
    Adaptive annealing: start with large perturbations, decrease over time.
    Uses frequency-domain perturbations for diversity.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    initial_scale = 0.15
    final_scale = 0.003
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        # Exponential decay of perturbation scale
        scale = initial_scale * (final_scale / initial_scale) ** (progress ** 0.6)

        # Choose perturbation type
        r = rng.random()
        if r < 0.4:
            # Gaussian perturbation
            a_try = best_a * np.exp(rng.normal(0, scale, n))
        elif r < 0.7:
            # Frequency-domain perturbation
            smoothness = 0.5 + rng.random() * 2.0
            a_try = freq_perturb(best_a, rng, smoothness)
            # Blend with original
            blend = 0.5 + 0.5 * scale / initial_scale
            a_try = (1 - blend) * best_a + blend * a_try
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
        elif r < 0.85:
            # Palindromic perturbation
            a_temp = best_a * np.exp(rng.normal(0, scale, n))
            a_temp = np.maximum(a_temp, 0)
            a_try = make_palindromic(a_temp)
        else:
            # Block swap perturbation
            a_try = best_a.copy()
            blk = n // 8
            i = rng.integers(0, n - 2 * blk)
            a_try[i:i+blk], a_try[i+blk:i+2*blk] = a_try[i+blk:i+2*blk].copy(), a_try[i:i+blk].copy()
            a_try = a_try * np.exp(rng.normal(0, scale * 0.5, n))
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try

        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Quick polish (short budget)
        remaining = time_budget - (time.monotonic() - start)
        polish_budget = min(remaining * 0.15, 30.0)
        if polish_budget > 5:
            a_polk = lp_polish(a_try, time_budget=polish_budget, seed=seed + it)
            obj_polk = compute_obj(a_polk)
            if obj_polk < best_obj - 1e-12:
                best_obj = obj_polk
                best_a = a_polk.copy()
                if verbose:
                    print(f"    Anneal it{it}: obj={best_obj:.8f}, scale={scale:.4f} **")
        else:
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()

        it += 1

    return best_a

        while time.monotonic() - start_fine < alloc:
            scale = 0.005 + 0.003 * (1.0 - (time.monotonic() - start_fine) / alloc)
            a_try = a_fine * np.exp(rng_fine.normal(0, scale, n))
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try

            # Very quick polish (1-2 LP steps)
            g_lp, _ = solve_lp_direction(a_try)
            if g_lp is not None:
                direction = g_lp - a_try
                for t in [0.1, 0.3, 0.5, 0.8, 1.0]:
                    a_test = np.maximum(a_try + t * direction, 0)
                    s_test = a_test.sum()
                    if s_test > 1e-10:
                        a_test = a_test / s_test
                        obj_test = compute_obj(a_test)
                        if obj_test < best_fine_obj - 1e-13:
                            best_fine_obj = obj_test
                            a_fine = a_test.copy()
                            break

        obj_final_anneal = compute_obj(a_fine)
        if obj_final_anneal < best_score:
            best_score = obj_final_anneal
            best_seq = a_fine.copy()
            print(f"  Final Anneal: obj={best_score:.8f} **")
        else:
            print(f"  Final Anneal: obj={obj_final_anneal:.8f}")

    # === Phase 8: Final LP sweep ===
    print(f"\n=== Phase 8: Final LP Sweep ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 60.0)
        a_final = lp_polish(best_seq, time_budget=alloc, seed=9999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
