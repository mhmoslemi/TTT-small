"""
Optimizes step function coefficients using analytical-broken-point coordinate descent,
LP polishing with bisection blending, adaptive n exploration, and geometric annealing.
Key improvements: (1) Exact 1D subproblem via breakpoint enumeration + golden section,
(2) Bisection on LP blend parameter for tighter convergence,

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def obj_from_conv(a, conv):
    """Compute objective from precomputed convolution."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm,g)[k] <= M, sum(g)=1, g>=0. Returns g, M."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_1d_for_coord(a, conv, S, i, n):
    """Solve exact 1D subproblem for coordinate i using analytical breakpoints + golden section."""
    v = a[i]
    S_other = S - v
    if S_other < 1e-12:
        return v

    k_min = max(0, i)
    k_max = min(2 * n - 2, i + n - 1)

    ks = np.arange(k_min, k_max + 1)
    a_ki = a[ks - i]
    c_k = conv[ks]

    mask = (ks != 2 * i)
    ks_lin = ks[mask]
    a_ki_lin = a_ki[mask]
    c_k_lin = c_k[mask]

    k_quad = 2 * i
    c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

    # Find unchanged max
    all_ks = np.arange(0, 2 * n - 1)
    mask_unch = np.ones(2 * n - 1, dtype=bool)
    mask_unch[k_min:k_max + 1] = False
    if np.any(mask_unch):
        M_other = float(np.max(conv[mask_unch]))
    else:
        M_other = 0.0

    # Define 1D objective
    def f_1d(x):
        if x < 0:
            x = 0.0
        if len(ks_lin) > 0:
            lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
            max_lin = float(np.max(lin_vals))
        else:
            max_lin = 0.0
        quad_val = c_2i + x * x - v * v
        max_val = max(M_other, max_lin, quad_val)
        denom = (S_other + x) * (S_other + x)
        if denom < 1e-30:
            return 1e30
        return 2.0 * n * max_val / denom

    # Generate analytical candidate points
    candidates = [0.0, v, max(0.0, v * 0.3), v * 2.0]

    # Breakpoint: linear_j = M_other => x = (M_other - c_j + 2v*a_j) / (2*a_j)
    if len(ks_lin) > 0:
        pos_mask = a_ki_lin > 1e-15
        if np.any(pos_mask):
            bp = (M_other - c_k_lin[pos_mask] + 2.0 * v * a_ki_lin[pos_mask]) / (2.0 * a_ki_lin[pos_mask])
            bp_valid = bp[(bp > 1e-10) & (bp < 20.0 * max(S, 1e-6))]
            candidates.extend(bp_valid[:15].tolist())

        # Breakpoint: quadratic = M_other => x = sqrt(M_other - c_2i + v^2)
        qbp_sq = M_other - c_2i + v * v
        if qbp_sq > 0:
            candidates.append(float(np.sqrt(qbp_sq)))

        # Breakpoint: quadratic = linear_j => x^2 - 2*a_j*x + (c_2i - v^2 - c_j + 2v*a_j) = 0
        disc = a_ki_lin ** 2 - (c_2i - v * v - c_k_lin + 2.0 * v * a_ki_lin)
        valid_disc = disc > 0
        if np.any(valid_disc):
            r1 = a_ki_lin[valid_disc] - np.sqrt(disc[valid_disc])
            r2 = a_ki_lin[valid_disc] + np.sqrt(disc[valid_disc])
            roots = np.concatenate([r1, r2])
            roots_valid = roots[(roots > 1e-10) & (roots < 20.0 * max(S, 1e-6))]
            if len(roots_valid) > 0:
                # Take a few spread out ones
                candidates.extend(np.linspace(roots_valid[0], roots_valid[-1], min(5, len(roots_valid))).tolist())

    # Evaluate all candidates
    best_x = v
    best_f = f_1d(v)
    for x in candidates:
        x = max(0.0, float(x))
        fx = f_1d(x)
        if fx < best_f:
            best_f = fx
            best_x = x

    # Golden section refinement
    scale = max(best_x * 0.5, S / n * 0.5, 1e-8)
    x_lo = max(0.0, best_x - scale)
    x_hi = best_x + scale
    if x_hi <= x_lo:
        x_hi = x_lo + 1e-10

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = x_hi - gr * (x_hi - x_lo)
    d_pt = x_lo + gr * (x_hi - x_lo)
    fc = f_1d(c_pt)
    fd = f_1d(d_pt)

    for _ in range(45):
        if x_hi - x_lo < 1e-15 * max(1.0, x_hi):
            break
        if fc < fd:
            x_hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = x_hi - gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
        else:
            x_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = x_lo + gr * (x_hi - x_lo)
            fd = f_1d(d_pt)

    x_opt = max(0.0, (x_lo + x_hi) / 2.0)
    f_opt = f_1d(x_opt)
    if f_opt < best_f:
        best_x = x_opt

    return best_x


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """Fast coordinate descent with analytical 1D subproblem solving."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0
    conv = compute_conv(a)

    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        S = float(np.sum(a))
        max_change = 0.0

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            x_new = solve_1d_for_coord(a, conv, S, i, n)
            if x_new < 0:
                x_new = 0.0

            v = a[i]
            if abs(x_new - v) < 1e-15:
                continue

            delta = x_new - v
            a[i] = x_new

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            k_quad = 2 * i

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_new * x_new - v * v

            ks_lin = np.arange(k_min, k_max + 1)
            ks_lin = ks_lin[ks_lin != k_quad]
            if len(ks_lin) > 0:
                a_ki_lin = a[ks_lin - i]
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta
            max_change = max(max_change, abs(delta))

            if sweep % 2 == 0 or sweep == 1:
                S_check = float(np.sum(a))
                if S_check > 1e-10:
                    obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, max_chg={max_change:.2e}, t={elapsed:.1f}s")

        if max_change < 1e-13:
            if verbose:
                print(f"    CD converged at sweep {sweep}")
            break

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use np.interp (linear) which is stable
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 12, size=7)
        phases = rng.uniform(0, 2 * np.pi, size=7)
        x = np.linspace(0, 1, n)
        a = np.ones(n) * 0.5
        for f, p in zip(freqs, phases):
            a += 0.15 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Raised cosine
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        # Hann window
        x = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
        a = a / a.sum()
    elif mode == 8:
        # Sinc-like
        x = np.linspace(-3, 3, n)
        a = np.sinc(x / 3.0)
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

            print(f"  Anneal CD: obj={best_score:.8f} **")

        # Try perturbed versions with decreasing noise
        remaining = deadline - time.monotonic()
        if remaining > 40:
            noise_levels = [0.05, 0.03, 0.01]
            for nl in noise_levels:
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break
                a_pert = best_seq * np.exp(rng.normal(0, nl, len(best_seq)))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                alloc = min(remaining * 0.3, 50.0)
                if alloc > 15:
                    a_pert = coordinate_descent(a_pert, time_budget=alloc, verbose=False, seed=9500)
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        print(f"  Perturb (noise={nl}): obj={best_score:.8f} **")
                    else:
                        print(f"  Perturb (noise={nl}): obj={obj_pert:.8f}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 150.0)
        a_final = coordinate_descent(best_seq, time_budget=alloc, verbose=False, seed=seed + 9999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # Final LP if time
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_lp_final = lp_polish_bisection(best_seq, time_budget=min(remaining * 0.5, 60.0),
                                         verbose=False, seed=7777)
        obj_lp_final = compute_obj(a_lp_final)
        if obj_lp_final < best_score:
            best_score = obj_lp_final
            best_seq = a_lp_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more final CD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = coordinate_descent(best_seq, time_budget=min(remaining * 0.6, 60.0),
                                    verbose=False, seed=12345)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
