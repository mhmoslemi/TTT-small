   import numpy as np
   from scipy.fft import fft, ifft, rfft, irfft
   import time
   
   def fft_convolve(a, b):
       """Full linear convolution via FFT."""
       n = len(a)
       m = len(b)
       size = 1
       while size < n + m - 1:
           size <<= 1
       A = fft(a, size)
       B = fft(b, size)
       return ifft(A * B).real[:n+m-1]
   
   def fft_corr(a, b):
       """Cross-correlation via FFT: (a ⊛ b)[n] = sum_k a[k] * b[k-n]"""
       n = len(a)
       m = len(b)
       size = 1
       while size < n + m - 1:
           size <<= 1
       A = fft(a, size)
       B = fft(b, size)
       return ifft(A * np.conj(B)).real
       # Result has length size, valid indices are 0 to n+m-2
       # (a ⊛ b)[i] = sum_k a[k] * b[k-i] for i = 0, ..., m-1 (when a is longer)
   
   def project_simplex(v):
       """Project v onto the probability simplex {x >= 0, sum x = 1}."""
       n = len(v)
       if n == 0:
           return v
       u = np.sort(v)[::-1]
       cssv = np.cumsum(u)
       rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
       theta = (cssv[rho] - 1) / (rho + 1.0)
       return np.maximum(v - theta, 0)
   
   def compute_obj_normalized(a):
       """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
       n = len(a)
       c = fft_convolve(a, a)
       return 2.0 * n * float(np.max(c))
   
   def compute_conv(a):
       """Full convolution a*a via FFT."""
       return fft_convolve(a, a)
   
   def smoothed_grad_max(a, tau):
       """
       Gradient of smoothed max: F(a) = (1/tau) * log(sum_k exp(tau * c[k]))
       where c = conv(a,a).
       Returns (F, grad) where grad has length n.
       """
       n = len(a)
       c = compute_conv(a)
       c_max = float(np.max(c))
       exp_c = np.exp(tau * (c - c_max))
       sum_exp = float(np.sum(exp_c))
       if sum_exp < 1e-300:
           p = np.zeros(len(c))
           p[int(np.argmax(c))] = 1.0
       else:
           p = exp_c / sum_exp
       F = c_max + np.log(sum_exp) / tau
       
       # grad[i] = 2 * sum_k p[k] * a[k-i] = 2 * (p ⊛ a)[i]
       size = 1
       while size < 3 * n:
           size <<= 1
       P = fft(p, size)
       A = fft(a, size)
       corr = ifft(P * np.conj(A)).real
       grad = 2.0 * corr[:n]
       
       return F, grad
   
   def variance_obj_grad(a):
       """
       Objective: sum_k (c[k] - mu)^2 where c = conv(a,a), mu = 1/(2n-1).
       Returns (obj, grad).
       """
       n = len(a)
       c = compute_conv(a)
       mu = 1.0 / (2*n - 1)
       d = c - mu
       obj = float(np.sum(d * d))
       # grad[i] = 4 * sum_k d[k] * a[k-i] = 4 * (d ⊛ a)[i]
       size = 1
       while size < 3 * n:
           size <<= 1
       D = fft(d, size)
       A = fft(a, size)
       corr = ifft(D * np.conj(A)).real
       grad = 4.0 * corr[:n]
       return obj, grad
