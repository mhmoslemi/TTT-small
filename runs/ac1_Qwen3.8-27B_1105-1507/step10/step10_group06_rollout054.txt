```python
def newton_step(a, active_weights, mu=1e-4):
    """
    Compute Newton step for the convex problem.
    a: current iterate (sum=1, a>=0)
    active_weights: dict or array of weights for active constraints
    mu: regularization parameter
    """
    n = len(a)
    # Compute gradient
    grad = np.zeros(n)
    for k, w in active_weights.items():
        # grad[j] += 2 * w * a[k-j] (if 0<=k-j<n)
        # This is: for j in valid range, grad[j] += 2*w*a[k-j]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += 2 * w * a[k - j]
    
    # Compute Hessian (sparse)
    # H[j,l] = 2 * sum_{k} w_k * delta_{l, k-j}
    # Build as dense for simplicity (n=800 is fine)
    H = np.zeros((n, n))
    for k, w in active_weights.items():
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            l = k - j
            H[j, l] += 2 * w
            H[l, j] += 2 * w
    
    # Add regularization
    H += mu * np.eye(n)
    
    # Solve KKT system
    # [H, 1; 1^T, 0] [d; lambda] = [-grad; 0]
    # (for the equality constraint sum(d) = 0)
    KKT = np.zeros((n + 1, n + 1))
    KKT[:n, :n] = H
    KKT[:n, n] = 1
    KKT[n, :n] = 1
    rhs = np.zeros(n + 1)
    rhs[:n] = -grad
    
    try:
        sol = np.linalg.solve(KKT, rhs)
        d = sol[:n]
    except np.linalg.LinAlgError:
        # Fallback: use pseudo-inverse or regularization
        H += 1e-3 * np.eye(n)
        KKT[:n, :n] = H
        try:
            sol = np.linalg.solve(KKT, rhs)
            d = sol[:n]
        except:
            d = -grad / np.linalg.norm(grad)
    
    return d
```

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:
1. LP-based alternating minimization for global progress
2. Newton-KKT method with active set for fast local convergence
3. Multiplicative weight updates (MWU) as a diversification tool
4. Multi-start strategy with diverse initializations and lengths
5. Careful time management across phases
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_conv_obj_normalized(a):
    """Compute 2n*max_conv assuming a is already normalized (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm
    return A


def solve_conv_lp(a, time_limit=5.0):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        if time.monotonic() - start > time_limit:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def gradient_and_hessian(a, active_indices, weights):
    """
    Compute gradient and Hessian of sum_k w_k * (a*a)[k].
    grad[j] = 2 * sum_k w_k * a[k-j] (if 0<=k-j<n)
    H[j,l] = 2 * sum_k w_k * delta(l, k-j)
    """
    n = len(a)
    grad = np.zeros(n)
    H = np.zeros((n, n))

    for k, w in zip(active_indices, weights):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        vals = a[k - js]
        grad[js] += 2.0 * w * vals
        H[js, k - js] += 2.0 * w
        H[k - js, js] += 2.0 * w

    return grad, H


def newton_kkt_step(a, active_tol=0.01, mu=1e-4):
    """
    Compute one Newton-KKT step for minimizing max_k (a*a)[k]
    subject to sum(a)=1, a>=0.
    Returns (direction, active_indices, max_conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    # Active set
    threshold = max_conv * (1.0 - active_tol)
    active = np.where(conv >= threshold)[0]

    if len(active) == 0:
        active = np.array([np.argmax(conv)])

    # Weights (uniform on active set)
    w = 1.0 / len(active)
    active_indices = active
    weights = np.full(len(active), w)

    grad, H = gradient_and_hessian(a, active_indices, weights)

    # Regularize Hessian
    H += mu * np.eye(n)

    # Solve KKT system: [H, 1; 1^T, 0] [d; lam] = [-grad; 0]
    KKT = np.zeros((n + 1, n + 1))
    KKT[:n, :n] = H
    KKT[:n, n] = 1.0
    KKT[n, :n] = 1.0
    rhs = np.zeros(n + 1)
    rhs[:n] = -grad

    try:
        sol = np.linalg.solve(KKT, rhs)
        d = sol[:n]
    except np.linalg.LinAlgError:
        # Fall back to steepest descent (projected)
        d = grad - np.mean(grad)
        d = -d  # descent direction

    # Ensure descent direction
    if np.dot(grad, d) > 0:
        # Not a descent direction, use projected gradient
        d = -(grad - np.mean(grad))

    return d, active_indices, max_conv

            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_conv_lp(a, time_limit=min(5.0, remaining * 0.3))
        lp_time = time.monotonic() - t0

        if g_lp is None:
            break

        iter_count += 1

        # Try multiple blending factors
        candidates_t = [t_blend, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if iter_count > 100:
            candidates_t.extend([0.05, 0.02, 0.01])

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend

        for t in candidates_t:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_conv_obj_normalized(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            t_blend = max(0.02, best_local_t * 0.98)
            stall_count = 0
            if verbose and iter_count % 50 == 0:
                print(f"    LP alt it={iter_count}: obj={best_obj:.10f}, t={t_blend:.4f}, lp_t={lp_time:.2f}s")
        else:
            stall_count += 1
            if stall_count >= 5:
                # Fourier perturbation
                pert = np.zeros(n)
                for mode in range(1, 4):
                    freq = mode * 2 * np.pi / n
                    amp = 0.02 * rng.uniform(0.5, 1.5) / mode
                    phase = rng.uniform(0, 2 * np.pi)
                    pert += amp * np.cos(freq * np.arange(n) + phase)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                t_blend = max(0.1, min(t_blend * 1.3, 0.8))
                stall_count = 0

    if verbose:
        print(f"    LP alt done: {iter_count} iters, best_obj={best_obj:.10f}")

    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 6):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, seed=42):
    """
    Create diverse structured initializations.
    """
    rng = np.random.default_rng(seed)
    strategies = [
        'uniform',
        'periodic',
        'bimodal',
        'gaussian',
        'random',
        'sawtooth'
    ]
    strategy = strategies[seed % len(strategies)]

    j = np.arange(n)
    if strategy == 'uniform':
        a = np.ones(n)
    elif strategy == 'periodic':
        freq = rng.choice([2, 3, 4, 5, 6, 7, 8])
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * j / n)
    elif strategy == 'bimodal':
        center1 = n * rng.uniform(0.2, 0.4)
        center2 = n * rng.uniform(0.6, 0.8)
        a = np.exp(-((j - center1) / (n * 0.1))**2) + np.exp(-((j - center2) / (n * 0.1))**2)
    elif strategy == 'gaussian':
        center = n * 0.5
        sigma = n * rng.uniform(0.15, 0.3)
        a = np.exp(-((j - center) / sigma)**2)
    elif strategy == 'random':
        a = rng.uniform(0.5, 1.5, n)
    elif strategy == 'sawtooth':
        period = max(2, n // rng.choice([4, 6, 8, 10]))
        a = 0.5 + 0.5 * np.abs(np.sin(np.pi * j / period))
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def full_optimize(a, time_budget, seed=42, verbose=False):
    """
    Full optimization pipeline: LP alternate -> Newton -> MWU -> Newton.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_conv_obj_normalized(a)
    start = time.monotonic()

    # Stage 1: LP Alternating (50% of budget)
    t1 = time_budget * 0.50
    a1 = lp_alternate(a, time_budget=t1, seed=seed, verbose=verbose)
    obj1 = compute_conv_obj_normalized(a1)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()
    if verbose:
        print(f"    After LP: obj={best_obj:.10f}")

    # Stage 2: Newton KKT (25% of budget)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t2 = remaining * 0.35
        a2 = newton_optimize(best_a, time_budget=t2, verbose=verbose)
        obj2 = compute_conv_obj_normalized(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
        if verbose:
            print(f"    After Newton: obj={best_obj:.10f}")

    # Stage 3: Short LP pass with perturbation (15%)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        t3 = remaining * 0.30
        rng = np.random.default_rng(seed + 123)
        a_pert = fourier_perturb(best_a, rng, strength=0.005)
        a3 = lp_alternate(a_pert, time_budget=t3, seed=seed + 123, verbose=False)
        obj3 = compute_conv_obj_normalized(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
        if verbose:
            print(f"    After LP2: obj={best_obj:.10f}")

    # Stage 4: Final Newton polish (10%)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t4 = remaining * 0.40
        a4 = newton_optimize(best_a, time_budget=t4, verbose=False)
        obj4 = compute_conv_obj_normalized(a4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
        if verbose:
            print(f"    After Newton2: obj={best_obj:.10f}")

    # Stage 5: MWU for final touch
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        t5 = min(remaining * 0.3, 30.0)
        a5 = mwu_optimize(best_a, time_budget=t5, eta=0.01, verbose=False)
        obj5 = compute_conv_obj_normalized(a5)
        if obj5 < best_obj:
            best_obj = obj5
            best_a = a5.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating minimization,
    Newton-KKT optimization, MWU, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0)
        if parent.size == 0 or parent.sum() < 0.01:
            raise ValueError("Invalid parent")
        parent = parent / parent.sum()
        n_parent = parent.size
        print(f"Parent loaded: n={n_parent}, obj={compute_obj(parent):.10f}")
    except Exception:
        n_parent = 800
        parent = np.ones(n_parent) / n_parent
        print(f"Using default init: n={n_parent}, obj={compute_obj(parent):.10f}")

    best_a = parent.copy()
    best_obj = compute_conv_obj_normalized(parent)
    n_best = n_parent

    # === Phase 1: Optimize parent length (55% time) ===
    print(f"\n=== Phase 1: Optimize parent (n={n_best}) ===")
    p1_budget = (deadline - time.monotonic()) * 0.55
    a_p1 = full_optimize(best_a, time_budget=p1_budget, seed=seed, verbose=True)
    obj_p1 = compute_conv_obj_normalized(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1.copy()
    print(f"  Phase 1 best: n={n_best}, obj={best_obj:.10f}")

    # === Phase 2: Try different lengths (25% time) ===
    print(f"\n=== Phase 2: Multi-length exploration ===")
    remaining = deadline - time.monotonic()
    p2_total = remaining * 0.25

    # Candidate lengths to try
    n_candidates = sorted(set([
        n_best - 100, n_best - 50, n_best + 50, n_best + 100,
        max(400, n_best - 150), min(1200, n_best + 150)
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 1400 and n != n_best]

    # Also try some structured inits at best length
    n_candidates = list(set(n_candidates + [n_best]))

    per_n_budget = p2_total / max(1, len(n_candidates))
    
    for idx, n_try in enumerate(n_candidates):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_n_budget, remaining * 0.3)
        if alloc < 20:
            break

        # Different initialization for each
        if n_try == n_best:
            a_init = fourier_perturb(best_a, rng, strength=0.01)
        else:
            a_init = resample_to_length(best_a, n_try)
            # Mix with structured init
            a_struct = structured_init(n_try, seed=seed + idx * 7)
            a_init = 0.7 * a_init + 0.3 * a_struct
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()

        a_result = full_optimize(a_init, time_budget=alloc, seed=seed + idx * 31, verbose=False)
        obj_result = compute_conv_obj_normalized(a_result)

        if obj_result < best_obj:
            best_obj = obj_result
            best_a = a_result.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_obj:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_result:.10f}")

    # === Phase 3: Diversified restarts (15% time) ===
    print(f"\n=== Phase 3: Diversified restarts ===")
    remaining = deadline - time.monotonic()
    p3_total = remaining * 0.15

    # Try structured initializations at the best length
    for attempt in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p3_total / 4, remaining * 0.25)
        if alloc < 20:
            break

        a_struct = structured_init(n_best, seed=seed + 100 + attempt * 17)
        # Mix with current best
        a_mix = 0.5 * best_a + 0.5 * a_struct
        a_mix = np.maximum(a_mix, 0)
        a_mix = a_mix / a_mix.sum()

        a_result = full_optimize(a_mix, time_budget=alloc, seed=seed + 200 + attempt * 43, verbose=False)
        obj_result = compute_conv_obj_normalized(a_result)

        if obj_result < best_obj:
            best_obj = obj_result
            best_a = a_result.copy()
            print(f"  ** Restart {attempt}: obj={best_obj:.10f} **")
        else:
            print(f"  Restart {attempt}: obj={obj_result:.10f}")

    # === Phase 4: Final intensive Newton on best (remaining time) ===
    print(f"\n=== Phase 4: Final Newton polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.8, 120.0)
        a_final = newton_optimize(best_a, time_budget=alloc, verbose=True)
        obj_final = compute_conv_obj_normalized(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
        print(f"  Final Newton: obj={best_obj:.10f}")

    # Also try one more LP+Newton cycle on the absolute best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.6, 60.0)
        a_final2 = full_optimize(best_a, time_budget=alloc, seed=seed + 999, verbose=False)
        obj_final2 = compute_conv_obj_normalized(a_final2)
        if obj_final2 < best_obj:
            best_obj = obj_final2
            best_a = a_final2.copy()
        print(f"  Final full: obj={best_obj:.10f}")

    # === Final validation ===
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
