import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * np.max(conv)

def solve_lp(a, solver=None):
    """Solve min M s.t. (a*g) <= M, g>=0, sum(g)=1. Returns (g*, M*)."""
    n = len(a)
    # Build constraint matrix A: A[k,j] = a[k-j] if 0<=k-j<n
    # Instead of building the full matrix, use structure
    # Actually for n up to ~2000, building the matrix is fine
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver is None:
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
        except:
            prob.solve(solver=cp.PDLP, verbose=False)
    
    if g.value is not None:
        g_val = np.maximum(g.value, 0)
        sg = g_val.sum()
        if sg > 1e-10:
            return g_val/sg, float(M.value)
    return None, None

def lp_fixed_point(a, time_budget, verbose=False, max_rounds=200):
    """LP fixed-point iteration with Anderson acceleration."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Anderson acceleration memory
    m = 5  # memory size
    history_a = [a.copy()]
    history_f = [a.copy()]  # f(a) = g*(a)
    
    for round in range(max_rounds):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        
        g_star, M_star = solve_lp(a)
        if g_star is None:
            break
        
        history_f[-1] = g_star.copy()
        
        # Anderson acceleration
        if len(history_a) >= 2:
            # Compute residuals
            n_hist = min(m, len(history_a))
            recent_a = history_a[-n_hist:]
            recent_f = history_f[-n_hist:]
            residuals = [f - a for a, f in zip(recent_a, recent_f)]
            
            # Solve for mixing coefficients
            if n_hist >= 2:
                # Gamma = [r_i - r_{i+1}]
                gammas = [residuals[i] - residuals[i+1] for i in range(n_hist-1)]
                # B = [dot(gamma_i, gamma_j)]
                B = np.array([[np.dot(gammas[i], gammas[j]) for j in range(n_hist-1)] for i in range(n_hist-1)])
                B += 1e-10 * np.eye(n_hist-1)  # regularization
                g_vec = np.array([np.dot(gammas[i], residuals[n_hist-1]) for i in range(n_hist-1)])
                try:
                    alpha = np.linalg.solve(B, g_vec)
                except:
                    alpha = np.zeros(n_hist-1)
                alpha = np.clip(alpha, -1, 1)
                
                # Anderson combination
                a_new = recent_f[-1] - np.sum(alpha * [gammas[i] for i in range(n_hist-1)])
                # Actually the standard formula is:
                # x_new = f(x_k) - sum_i alpha_i * (r_i - r_{i+1})
                # where r_i = f(x_i) - x_i
                a_new = g_star.copy()
                for i in range(n_hist-1):
                    a_new -= alpha[i] * (residuals[i] - residuals[i+1])
                
                a_new = np.maximum(a_new, 0)
                sa = a_new.sum()
                if sa > 1e-10:
                    a_new = a_new / sa
                else:
                    a_new = g_star.copy()
        else:
            a_new = g_star.copy()
        
        # Line search: try several blending factors
        best_blend_obj = compute_obj(a_new)
        best_blend = a_new.copy()
        
        for t in [0.5, 0.7, 1.0]:
            a_try = (1-t)*a + t*a_new
            a_try = np.maximum(a_try, 0)
            st = a_try.sum()
            if st > 1e-10:
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_blend_obj:
                    best_blend_obj = obj_try
                    best_blend = a_try.copy()
        
        a = best_blend
        
        if a_new != a:  # update history only if we actually moved
            history_a.append(a.copy())
            history_f.append(a_new.copy())
            if len(history_a) > m + 1:
                history_a = history_a[-(m+1):]
                history_f = history_f[-(m+1):]
        
        obj_current = compute_obj(a)
        if obj_current < best_obj - 1e-12:
            best_obj = obj_current
            best_a = a.copy()
            if verbose:
                print(f"  LP-FP round {round}: obj={best_obj:.10f}, M_lp={M_star:.10f}")
        
        # Convergence check
        residual = np.max(np.abs(a - g_star))
        if residual < 1e-8:
            if verbose:
                print(f"  Converged at round {round}, residual={residual:.2e}")
            break
    
    return best_a

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, solver=None):
    """Solve min M s.t. (a*g) <= M, g>=0, sum(g)=1. Returns (g*, M*, dual_info)."""
    n = len(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    objective = cp.Minimize(M)
    prob = cp.Problem(objective, constraints)

    if solver is None:
        try:
            prob.solve(solver=cp.MOSEK, verbose=False,
                       msk_api_task=cp.MOSEK_TK_FEASIBILITY)

        except Exception:
            return None, None, None
