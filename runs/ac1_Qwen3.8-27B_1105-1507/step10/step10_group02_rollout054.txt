def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:

    a_norm = a / s

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

    # Actually let me think more carefully

def _f_1d(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i):
    if len(a_ki_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom

def diverse_init(n, rng, mode=0):
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 0.5 * np.sin(2 * np.pi * 3 * x))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Chirp-like
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * (x**2 / 2))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Concentrated in middle
        x = np.linspace(-1, 1, n)
        a = np.exp(-2.0 * x**2)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Hybrid optimizer combining mirror descent (multiplicative weight updates) with
log-sum-exp smoothing, iterative LP polishing, and exact coordinate descent.
Key innovation: multiplicative updates that naturally respect non-negativity and

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def compute_pressure(a, tau, conv=None):
    """Compute pressure vector: gradient of smooth max of conv(a,a)."""
    n = len(a)
    if conv is None:
        conv = compute_conv(a)
    max_conv = float(np.max(conv))
    p = np.exp(tau * (conv - max_conv))
    s = float(np.sum(p))
    if s < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = p / s
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A_rev = fft(a[::-1], N)
    cross = ifft(P * A_rev).real
    pressure = 2.0 * tau * cross[n - 1:n - 1 + n]
    return pressure, conv, max_conv

def _f_1d(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i):
    """1D subproblem objective for coordinate descent."""
    if len(a_ki_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)

    else:

    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom

    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * 3 * x + 0.5)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(-1, 1, n)
        a = np.exp(-1.5 * x ** 2) + 0.3 * np.exp(-6.0 * (x - 0.5) ** 2)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        freqs = [2, 3, 5, 7]
        a = np.ones(n)
        for f in freqs:
            a += 0.2 / f * np.cos(2 * np.pi * f * x)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
