"""
Nesterov Accelerated Smoothed Gradient + LP Polish Optimizer.
Uses fast FFT-based gradient computation on a log-sum-exp smoothed objective,
with adaptive temperature, Nesterov momentum, and multi-length exploration.
Replaces slow SOCP trust regions with many fast gradient iterations.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad(a, tau):
    """
    Compute gradient of F_tau(a) = 2*n*(1/tau)*log(sum_k exp(tau*c_k(a)))
    Returns (gradient, p_weights, max_conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    exp_vals = np.exp(shifted)
    p = exp_vals / float(np.sum(exp_vals))

    # grad F_tau[m] = 4*n * sum_k p[k]*a[k-m] (with bounds)
    # = 4*n * cross_corr(p, a)[m]
    # cross_corr(p,a)[m] = ifft(FFT(p) * conj(FFT(a))).real
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad = 4.0 * n * np.real(ifft(P * np.conj(A)))[:n]
    return grad, p, max_c


def nesterov_smoothed_gd(a, time_budget=300.0, tau_init=5.0, tau_max=500.0,
                         lr_scale=1.0, verbose=False, seed=42):
    """
    Nesterov accelerated gradient on log-sum-exp smoothed objective.
    Adaptive tau that increases over time.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    tau = tau_init
    lr = lr_scale / np.sqrt(n)
    momentum = 0.9
    v = np.zeros(n)
    it = 0
    stall = 0

    y = a.copy()  # Nesterov lookahead point

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1.0))
        # Adaptive tau: increase smoothly
        tau_eff = tau_init * (tau_max / tau_init) ** (progress ** 0.6)

        # Compute gradient at y
        grad_y, p_y, max_c_y = smoothed_grad(y, tau_eff)

        # Nesterov step
        a_new = y - lr * grad_y
        a_new = project_simplex(a_new)

        # Update objective
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive learning rate
        if stall > 100:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                # Perturb and reset
                a_new = best_a * np.exp(rng.normal(0, 0.02, n))
                a_new = np.maximum(a_new, 0)
                s_p = a_new.sum()
                if s_p > 1e-10:
                    a_new = a_new / s_p
                lr = lr_scale / np.sqrt(n)
                v = np.zeros(n)
                y = a_new.copy()
                best_a = a_new.copy()
                best_obj = compute_obj(a_new)
                stall = 0

        # Update Nesterov momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        y = a_new + ((momentum - 1.0) / beta) * (a_new - y + lr * grad_y)
        # Simpler Nesterov: y = a_new + momentum * (a_new - a_old)
        # Let's use the standard form:
        y = (1.0 + momentum) * a_new - momentum * y

        # Project y to stay in reasonable range
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y > 1e-10:
            y = y / s_y

        a = a_new
        it += 1

        if verbose and it % 500 == 0:
            print(f"    NSGD it{it}: obj={best_obj:.8f}, tau={tau_eff:.1f}, "
                  f"lr={lr:.4e}, t={elapsed:.0f}s")

    return best_a


def fast_subgradient(a, time_budget=100.0, K=5, verbose=False, seed=42):
    """
    Direct subgradient method on the non-smooth objective.
    Uses top-K peak weighting for stability.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 8.0 / n
    v = np.zeros(n)
    momentum = 0.8
    it = 0
    stall = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        if max_c < 1e-15:
            break

        # Top-K peaks
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        weights = np.exp(10.0 * (peak_vals - max_pv) / max_pv)
        weights /= weights.sum()

        # Subgradient: 2*n * sum_k w_k * (D_k a)
        # (D_k a)[m] = a[k-m] if 0<=k-m<n
        # sum_k w_k * a[k-m] = cross_corr(w_top, a_padded)[m]
        w_full = np.zeros(2 * n - 1)
        for w, k in zip(weights, top_idx):
            w_full[int(k)] = w

        N = 1
        while N < 3 * n:
            N <<= 1
        W = fft(w_full, N)
        A = fft(a, N)
        sg = 2.0 * n * np.real(ifft(W * np.conj(A)))[:n]

        v = momentum * v + sg
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                lr *= 0.6
                v *= 0.3
                stall = 0
                if lr < 1e-14:
                    a = best_a * np.exp(rng.normal(0, 0.015, n))
                    a = np.maximum(a, 0)
                    s_a = a.sum()
                    if s_a > 1e-10:
                        a = a / s_a
                    lr = 8.0 / n
                    v = np.zeros(n)
                    stall = 0

        it += 1
        if verbose and it % 200 == 0:
            print(f"    SubGrad it{it}: obj={best_obj:.8f}, lr={lr:.4e}, "
                  f"t={time.monotonic()-start:.0f}s")

    return best_a


def build_conv_matrix_rows(a):
    """Build the convolution matrix A such that (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # (A@g)[k] includes a[j]*g[k-j] for valid k-j
        # So A[k, j] = a[j] if 0 <= k-j < n, i.e., j <= k < j+n
        k_start = j
        k_end = min(m - 1, j + n - 1)
        A[k_start:k_end + 1, j] = a[j]
    return A


def lp_polish(a, time_budget=120.0, verbose=False, seed=42):
    """LP-based polishing: find direction g minimizing max conv(a,g)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        A = build_conv_matrix_rows(a)
        m = 2 * n - 1

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        g_val = None
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0).flatten()
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                    break
            except Exception:
                continue

        if g_val is None:
            break

        # Line search
        best_t = 0
        best_t_obj = best_obj
        for t_val in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]:
            a_try = (1 - t_val) * best_a + t_val * g_val
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-15:
                best_t_obj = obj_try
                best_t = t_val

        if best_t > 0 and best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = (1 - best_t) * best_a + best_t * g_val
            best_a = np.maximum(best_a, 0)
            s_ba = best_a.sum()
            if s_ba > 1e-10:
                best_a = best_a / s_ba
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                a = best_a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                no_improve = 0
                if it > 50:
                    break

        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def spectral_init(n, seed=42):
    """
    Initialize using a spectral approach: try to make the DFT magnitude
    of a as flat as possible, then make it nonnegative.
    """
    rng = np.random.default_rng(seed)

    # Method 1: Random phases with flat magnitude
    N = 1
    while N < 2 * n:
        N <<= 1
    # Target: flat power spectrum
    target_mag = np.ones(N) / np.sqrt(N)
    phases = rng.uniform(0, 2 * np.pi, N)
    # Enforce Hermitian symmetry for real signal
    spectrum = target_mag * np.exp(1j * phases)
    # Make it Hermitian
    for k in range(1, N // 2):
        spectrum[N - k] = np.conj(spectrum[k])
    spectrum[0] = abs(spectrum[0])
    if N % 2 == 0:
        spectrum[N // 2] = abs(spectrum[N // 2])

    a = np.real(ifft(spectrum))[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n

    return a


def structured_init(n, seed=42):
    """Create structured initializations with periodic properties."""
    rng = np.random.default_rng(seed)

    # Try several structures and pick the best
    candidates = []

    a1 = np.maximum(1.0 + 0.5 * np.sin(x + rng.uniform(0, 2 * np.pi)), 0.01)
    candidates.append(a1 / a1.sum())

    # 2. Cosine with random phase
    a2 = np.maximum(1.0 + 0.4 * np.cos(2 * x + rng.uniform(0, 2 * np.pi)), 0.01)
    candidates.append(a2 / a2.sum())

    # 3. Two-frequency
    f1 = rng.integers(3, 10)
    f2 = rng.integers(10, 25)
    a3 = np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * f1 * x / n) +
                    0.2 * np.sin(2 * np.pi * f2 * x / n), 0.01)
    candidates.append(a3 / a3.sum())

    # 4. Random with smoothing
    a4 = np.maximum(rng.exponential(1.0, n), 0.001)
    # Smooth with a short window
    w = max(3, n // 100)
    kernel = np.ones(w) / w
    a4 = np.convolve(a4, kernel, mode='same')
    candidates.append(a4 / a4.sum())

    # 5. Exponential decay pair
    a5 = np.zeros(n)
    decay = rng.uniform(0.5, 2.0)
    for i in range(n):
        a5[i] = np.exp(-decay * min(i, n - 1 - i) / n) + 0.1
    candidates.append(a5 / a5.sum())

    # Pick the best
    best = candidates[0]
    best_obj = compute_obj(best)
    for c in candidates[1:]:
        obj = compute_obj(c)
        if obj < best_obj:
            best_obj = obj
            best = c

    return best


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Fast Nesterov smoothed gradient + LP polish + multi-length.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Load parent sequence ---
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        print("No parent available, using default initialization.")

    if best_seq is None:
        best_seq = structured_init(n_target, seed)
        best_score = compute_obj(best_seq)
        print(f"Structured init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Quick Nesterov smoothed GD ===
    print(f"\n=== Phase 1: Nesterov Smoothed GD (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 250.0)
    if p1_budget > 20:
        a_opt = nesterov_smoothed_gd(best_seq, time_budget=p1_budget,
                                     tau_init=5.0, tau_max=300.0,
                                     lr_scale=1.5, verbose=True, seed=42)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase1 NSGD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 NSGD: obj={obj_opt:.8f}")

    # === Phase 2: Subgradient refinement ===
    print(f"\n=== Phase 2: Subgradient Refinement ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.1, 80.0)
    if p2_budget > 15:
        a_sg = fast_subgradient(best_seq, time_budget=p2_budget, K=5,
                                verbose=True, seed=77)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase2 SubGrad: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 SubGrad: obj={obj_sg:.8f}")

    # === Phase 3: LP Polish ===
    print(f"\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 120.0)
    if p3_budget > 20:
        a_lp = lp_polish(best_seq, time_budget=p3_budget, verbose=True, seed=111)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase3 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 LP: obj={obj_lp:.8f}")

    # === Phase 4: Second NSGD pass with different params ===
    print(f"\n=== Phase 4: NSGD Pass 2 ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 120.0)
    if p4_budget > 15:
        a_opt2 = nesterov_smoothed_gd(best_seq, time_budget=p4_budget,
                                      tau_init=20.0, tau_max=800.0,
                                      lr_scale=0.8, verbose=True, seed=200)
        obj_opt2 = compute_obj(a_opt2)
        if obj_opt2 < best_score:
            best_score = obj_opt2
            best_seq = a_opt2.copy()
            print(f"  Phase4 NSGD2: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 NSGD2: obj={obj_opt2:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 150), max(400, n_current - 80),
        n_current + 80, n_current + 150, n_current + 300,
        600, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_current) > 50]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_seq(best_seq, n_try)

        # Quick NSGD
        alloc1 = min(remaining * 0.1, 40.0)
        if alloc1 > 8:
            a_try = nesterov_smoothed_gd(a_try, time_budget=alloc1,
                                         tau_init=10.0, tau_max=200.0,
                                         lr_scale=1.2, seed=300 + n_try % 100)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.08, 30.0)
        if alloc2 > 10:
            a_try = lp_polish(a_try, time_budget=alloc2, seed=400 + n_try % 100)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Random restarts + optimization ===
    print(f"\n=== Phase 6: Random Restarts ===")
    remaining = deadline - time.monotonic()

    for r_idx in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        scale = 0.005 + 0.008 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        # NSGD on perturbed
        alloc = min(remaining * 0.18, 60.0)
        if alloc > 10:
            a_pert = nesterov_smoothed_gd(a_pert, time_budget=alloc,
                                          tau_init=10.0, tau_max=400.0,
                                          lr_scale=1.0, seed=500 + r_idx)
            # Quick LP
            remaining = deadline - time.monotonic()
            alloc_lp = min(remaining * 0.05, 20.0)
            if alloc_lp > 8:
                a_pert = lp_polish(a_pert, time_budget=alloc_lp, seed=600 + r_idx)

            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Restart{r_idx} (s={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Restart{r_idx} (s={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 7: Final intensive optimization ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 100.0)
        a_final = nesterov_smoothed_gd(best_seq, time_budget=alloc,
                                       tau_init=50.0, tau_max=1000.0,
                                       lr_scale=0.5, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final NSGD: obj={best_score:.8f} **")
        else:
            print(f"  Final NSGD: obj={obj_final:.8f}")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 40.0)
        a_flp = lp_polish(best_seq, time_budget=alloc, seed=2000)
        obj_flp = compute_obj(a_flp)
        if obj_flp < best_score:
            best_score = obj_flp
            best_seq = a_flp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final subgradient
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_fsg = fast_subgradient(best_seq, time_budget=alloc, K=3, seed=3000)
        obj_fsg = compute_obj(a_fsg)
        if obj_fsg < best_score:
            best_score = obj_fsg
            best_seq = a_fsg.copy()
            print(f"  Final SubGrad: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
