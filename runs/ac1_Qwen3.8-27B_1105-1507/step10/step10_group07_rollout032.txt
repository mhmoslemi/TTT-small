"""
Exponential Repulsion + Trust Region SOCP Hybrid Optimizer.
Uses a multiplicative exponential (Boltzmann-style) update as the primary
diversification tool, with O(n log n) per-step cost via FFT-based structured
matrix-vector products. Combined with trust-region SOCP for precise polishing,
multi-length exploration, and perturbation-based restarts.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_repulsion(a, w, fft_size):
    """
    Compute R = S @ a where S[i,j] = w[i+j] (the structured matrix).
    R[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    Computed via FFT: R = (w * a_flipped)[n-1 : 2n-1]
    """
    n = len(a)
    a_flipped = a[::-1]
    W = fft(w, fft_size)
    A = fft(a_flipped, fft_size)
    conv = ifft(W * A).real
    return conv[n - 1:2 * n - 1]


def exponential_repulsion(a, time_budget=150.0, tau_init=3.0, tau_end=80.0,
                          eta_init=0.15, momentum=0.5, verbose=False, seed=42):
    """
    Exponential (Boltzmann) repulsion method.
    Multiplicative update: a_new[i] = a[i] * exp(-eta * R[i])
    where R is the weighted peak contribution field computed via FFT.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # FFT size for repulsion computation
    fft_size = 1
    while fft_size < 3 * n - 2:
        fft_size <<= 1

    eta = eta_init
    it = 0
    v_mom = np.zeros(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break

        # Compute weights with softmax
        diffs = conv - max_conv
        exp_vals = np.exp(tau * diffs)
        w = exp_vals / float(np.sum(exp_vals))

        # Compute repulsion field R = S @ a
        R = compute_repulsion(a, w, fft_size)
        R = np.maximum(R, 0)
        R_max = np.max(R)
        if R_max < 1e-15:
            break

        # Momentum-smoothed field
        v_mom = momentum * v_mom + (1 - momentum) * R
        v_max = np.max(v_mom)
        if v_max < 1e-15:
            break
        v_norm = v_mom / v_max

        # Exponential update
        a_new = a * np.exp(-eta * v_norm)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        # Damped update for stability
        alpha = 0.5
        a_try = (1 - alpha) * a + alpha * a_new
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            # Slowly increase eta
            eta = min(eta * 1.01, 2.0)
        else:
            eta = max(eta * 0.95, 0.01)

        it += 1
        if verbose and it % 200 == 0:
            print(f"    ER it{it}: obj={best_obj:.8f}, tau={tau:.1f}, eta={eta:.4f}, t={elapsed:.0f}s")

    return best_a

def solve_tr_sscp(a, r, M_target=None):
    """Solve trust region SOCP for direction finding."""
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)

    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta

    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
    ]

    if r is not None:
        constraints.append(cp.norm(delta, 2) <= r)

    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_direction(a, M_target=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def resample(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

                                          tau_init=3.0, tau_end=50.0,
                                          eta_init=0.1, momentum=0.3, seed=n_try)

        # TR-SOCP on result
        remaining = deadline - time.monotonic()
        alloc_tr = min(remaining * 0.15, 60.0)
        if alloc_tr > 15:
            a_try = tr_sscp_polish(a_try, time_budget=alloc_tr, r_init=0.2, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Perturbation + re-optimization ===
    print(f"\n=== Phase 6: Perturbation + Re-optimize ===")
    remaining = deadline - time.monotonic()

    for r_idx in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        scale = 0.008 + 0.006 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            break
        a_pert = a_pert / s_pert

        # Use ER for quick improvement
        alloc_er = min(remaining * 0.15, 50.0)
        if alloc_er > 10:
            a_pert = exponential_repulsion(a_pert, time_budget=alloc_er,
                                           tau_init=5.0, tau_end=60.0,
                                           eta_init=0.1, momentum=0.3,
                                           seed=500 + r_idx)

        remaining = deadline - time.monotonic()
        alloc_tr = min(remaining * 0.15, 50.0)
        if alloc_tr > 10:
            a_pert = tr_sscp_polish(a_pert, time_budget=alloc_tr, r_init=0.15, seed=600 + r_idx)

        obj_pert = compute_obj(a_pert)
        if obj_pert < best_score:
            best_score = obj_pert
            best_seq = a_pert.copy()
            print(f"  Pert{r_idx} (scale={scale:.4f}): obj={best_score:.8f} **")
        else:
            print(f"  Pert{r_idx} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 7: LP polish ===
    print(f"\n=== Phase 7: LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc_lp = min(remaining * 0.15, 80.0)
        a_lp = lp_polish(best_seq, time_budget=alloc_lp, seed=1000)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase7 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase7 LP: obj={obj_lp:.8f}")

    # === Phase 8: Final intensive TR-SOCP ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_final = min(remaining * 0.6, 100.0)
        a_final = tr_sscp_polish(best_seq, time_budget=alloc_final, r_init=0.12, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final TR: obj={best_score:.8f} **")
        else:
            print(f"  Final TR: obj={obj_final:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
