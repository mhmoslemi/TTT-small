        lr = 0.5 / np.sqrt(n)
    
    best_a = a.copy()
    best_val = float(a @ apply_B(a, y))
    
    for it in range(n_iter):
        grad = apply_B(a, y)  # This is B(y) @ a, gradient is 2*this
        # Mirror descent step
        scaled = -lr * grad
        # For numerical stability
        scaled -= np.max(scaled)
        weights = np.exp(scaled)
        a_new = a * weights
        s = float(np.sum(a_new))
        if s < 1e-30:
            break
        a_new = a_new / s
        
        val = float(a_new @ apply_B(a_new, y))
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
        
        a = a_new
        lr *= 0.999
    
    return best_a

def dual_update(a, sharpen=1.0):
    """Update y based on current convolution. Higher sharpen = more peaky."""
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    if max_conv <= 0:
        return np.ones(len(conv)) / len(conv)
    # Sharpen the distribution
    if sharpen == 1.0:
        y = conv / np.sum(conv)
    else:
        # Exponential weighting
        scaled = sharpen * (conv / max_conv)
        y = np.exp(scaled)
        y = y / np.sum(y)
    return y

def primal_dual_method(a, time_budget, seed=42):
    """
    Fast primal-dual method:
    - Primal: mirror descent on the QP min a^T B(y) a on simplex
    - Dual: update y to concentrate on peak lags
    """
    start = time.monotonic()
    n = len(a)
    a = project_simplex(a.copy())
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    y = np.ones(2 * n - 1) / (2 * n - 1)  # uniform init
    rng = np.random.default_rng(seed)

    n_qp_iter = 200
    sharpen = 3.0
    lr = 1.0 / n
    
    while time.monotonic() - start < time_budget:
        # Primal update
        a = primal_qp_update(a, y, n_iter=n_qp_iter, lr=lr)

        y = dual_update(a, sharpen=sharpen)
        
        # Check
        obj = compute_obj(a)

    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_conv_lp(a):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def lp_alternate(a, time_budget, seed=42, verbose=False):
    """LP-based alternating minimization."""
    n = len(a)
    a = project_simplex(a.copy())

    start = time.monotonic()

    stall_count = 0
    iter_count = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        
        g_lp, M_lp = solve_conv_lp(a)
        if g_lp is None:
            break
        
        iter_count += 1
        
        # Try multiple blending factors
        candidates_t = [t_blend, t_blend*0.5, t_blend*2 if t_blend*2 <= 1 else 1.0, 0.1, 0.3, 0.5, 0.8, 1.0]
        
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        
        for t in candidates_t:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

            best_obj = best_local_obj

                noise = 0.01 * rng.standard_normal(n) * a
                a = project_simplex(a + noise)
                stall_count = 0
            t_blend = max(0.05, min(t_blend * 0.9, 1.0))
    
    if verbose:
        print(f"    LP alt: {iter_count} iters, best_obj={best_obj:.10f}")

def coordinate_descent(a, time_budget, max_passes=100):
    """
    Coordinate descent: at each step, optimally update one coordinate.
    """
    start = time.monotonic()
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(12345)
    
    order = np.arange(n)
    
    for pass_num in range(max_passes):
        if time.monotonic() - start > time_budget:
            break
        
        if pass_num % 5 == 0:
            # Reorder
            if rng.random() < 0.5:
                order = rng.permutation(n)
            else:
                order = np.arange(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                    break
            
            # Compute current convolution
            conv = compute_conv(a)
            S = float(np.sum(a))
            
            # For coordinate i, the new convolution as a function of t = a[i]:
            # new_conv[k] = conv[k] + 2*(t - a[i]) * a[k-i] for k != 2i
            # new_conv[2i] = conv[2i] + (t-a[i])*(t+a[i])
            # new_sum = S - a[i] + t
            
            ai = a[i]
            # Compute the "sensitivity" at each lag
            # sens[k] = coefficient of (t - a[i]) in new_conv[k]
            # For k != 2i: sens[k] = 2 * a[k-i]
            # For k = 2i: the change is (t-ai)*(t+ai) = (t-ai)^2 + ai*(t-ai)
            #   So linear part: ai, quadratic part: 1
            
            # We want to find t that minimizes max_k new_conv[k] / (S - ai + t)^2
            
            # Simple approach: try a few candidate t values
            # The optimal t is where the constraint is most "balanced"
            
            # For efficiency, let's try a 1D golden section search
            # over t in [0, t_max] where t_max is reasonable
            
            # First, find which lags are "active" (near the peak)
            peak = float(np.max(conv))
            # Lags that could become the peak after update
            # k where a[k-i] is large, and k=2i
            
            t_lo = 0.0
            t_hi = min(2.0 * S / n * 10, S)  # generous upper bound
            
            # Golden section search
            gr = (np.sqrt(5) - 1) / 2
            
            def obj_at_t(t):
                if t < 0:
                    return np.inf
                S_new = S - ai + t
                if S_new < 1e-10:
                    return np.inf
                # Compute new conv at key lags
                # For efficiency, only compute at lags that matter
                # But for correctness, let's compute all
                delta = t - ai
                # new_conv[k] = conv[k] + 2*delta*a[k-i] (approximately, ignoring the k=2i special case)
                # For k=2i: new_conv[2i] = conv[2i] + delta*(t+ai)
                
                k_max = min(2i if 2i < 2*n-1 else 2*n-2, 2*n-2)
                # Actually we need to be careful about boundaries
                # a[k-i] is non-zero only when 0 <= k-i <= n-1, i.e., i <= k <= i+n-1
                valid_k_lo = max(0, i)
                valid_k_hi = min(2*n-2, i + n - 1)
                
                new_conv = conv.copy()
                new_conv[valid_k_lo:valid_k_hi+1] += 2 * delta * a[max(0, valid_k_lo-i):min(n, valid_k_hi+1-i)]
                # Fix the k=2i case
                if 0 <= 2*i <= 2*n-2:
                    new_conv[2*i] = conv[2*i] + delta*(t + ai)
                
                max_new = float(np.max(new_conv))
                return max_new / (S_new * S_new)
            
            # Golden section search
            a_gs = t_lo + gr * (t_hi - t_lo)
            b_gs = t_lo + (1 - gr) * (t_hi - t_lo)
            fa = obj_at_t(a_gs)
            fb = obj_at_t(b_gs)
            
            for _ in range(30):  # 30 iterations of golden section gives ~1e-6 relative precision
                if fa < fb:
                    t_hi = b_gs
                    b_gs = a_gs
                    a_gs = t_lo + gr * (t_hi - t_lo)
                    fa = fb
                    fb = obj_at_t(a_gs)
                else:
                    t_lo = a_gs
                    a_gs = b_gs
                    b_gs = t_lo + (1 - gr) * (t_hi - t_lo)
                    fb = fa
                    fa = obj_at_t(b_gs)
            
            t_opt = (t_lo + t_hi) / 2
            new_obj_raw = obj_at_t(t_opt)
            
            # Check if this is actually better
            current_raw = float(np.max(conv)) / (S * S)
            if new_obj_raw < current_raw - 1e-15:
                a[i] = max(0, t_opt)
                # Renormalize
                a = project_simplex(a)
                # Update best
                obj_new = compute_obj(a)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
    
    return best_a

def nesterov_polish(a, time_budget, tau_init=30.0, tau_max=2000.0):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    tau = tau_init
    
    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = project_simplex(y)
        
        # Smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp
        
        # Gradient: 2 * sum_k p[k] * a[k-i] = 2 * cross_corr(y, p)
        N = 1
        while N < 3 * n:
            N <<= 1
        g_full = ifft(np.conj(fft(y, N)) * fft(p, N)).real[:n]
        g = 2.0 * g_full
        
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-15:
            adaptive_lr = 0.5 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.1 * 2.0/np.sqrt(n)), 3.0 * 2.0/np.sqrt(n))
            lr = 0.8 * lr + 0.2 * adaptive_lr
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        lr *= 0.9999
        tau = min(tau * 1.01, tau_max)
    
    return best_a

def band_limited_init(n, n_harmonics=20, seed=42):
    """Create a band-limited non-negative sequence."""
    rng = np.random.default_rng(seed)
    t = np.arange(n)
    # Start with 1 (DC component)
    sig = np.ones(n)
    for m in range(1, n_harmonics + 1):
        amp = rng.uniform(0.1, 0.8) / m
        phase = rng.uniform(0, 2 * np.pi)
        sig += amp * np.cos(2 * np.pi * m * t / n + phase)
    # Square to ensure non-negativity
    a = sig ** 2
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def random_smooth_init(n, seed=42):
    """Create a random smooth non-negative sequence."""
    rng = np.random.default_rng(seed)
    # Random low-pass filtered noise
    spectrum = rng.standard_normal(n // 2 + 1)
    # Low-pass: keep only low frequencies
    cutoff = max(5, n // 20)
    spectrum[cutoff:] = 0
    sig = irfft(spectrum * np.exp(1j * rng.standard_normal(n // 2 + 1)), n)
    a = np.abs(sig) + 0.5
    a = a / a.sum()
    return a

Let me also reconsider the primal-dual method. The key question is: how well does it converge?

The primal update minimizes `a^T B(y) a` where `B(y)` depends on the current `y`. If `y` is concentrated on the peak lags, then `B(y)` emphasizes reducing those peak lags. The dual update makes `y` more concentrated on the current peaks.

- `a` minimizes the "weighted" objective with weights `y`
- `y` is the distribution of the peaks

`min a^T T_{k*} a s.t. a >= 0, sum(a) = 1`

where `T_{k*}` is the Toeplitz matrix for lag `k*`. This is a QP that can be solved exactly.

- LP: exact best response, slow per iteration (~0.5s)
- Primal-dual: approximate best response, fast per iteration (~0.001s)

So in the same time, the primal-dual can do ~500x more iterations. The question is whether the approximation error accumulates or not.

    # Actually, let's arrange variables as [g_0, ..., g_{n-1}, M]
    
    # Constraints: A @ [g; M] <= b
    # For each k: sum_i A[k,i] * g[i] - M <= 0
    # So the row is [a[k], a[k-1], ..., a[k-n+1] (padded), -1]
    
    # Let's build it row by row
    rows = []
    cols = []
    data = []
    
    for k in range(2 * n - 1):
        for i in range(max(0, k - n + 1), min(k + 1, n)):
            rows.append(k)
            cols.append(i)
            data.append(a[k - i])
        # M coefficient
        rows.append(k)
        cols.append(n)  # M is variable n
        data.append(-1.0)
    
    A_sparse = csr_matrix((data, (rows, cols)), shape=(2*n-1, n+1))
    b = np.zeros(2 * n - 1)
    
    # c vector: minimize M (last variable)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Bounds: g[i] >= 0, M unbounded below (but will be positive)
    bounds = [(0, None)] * n + [(None, None)]
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    result = linprog(c, A_ub=A_sparse, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

        g = np.maximum(g, 0)
        s = float(np.sum(g))
        if s > 1e-10:
