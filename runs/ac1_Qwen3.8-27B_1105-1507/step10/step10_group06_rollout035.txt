        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp_help(a):
    """
    Solve LP: min M s.t. (a_norm * g)_k <= M for all k, sum(g)=1, g>=0.
    Uses scipy linprog with HiGHS.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build A_conv matrix: (2n-1) x n, where A_conv @ g = conv(a_norm, g)
    A_conv = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_conv[j:j+n, j] = a_norm
    
    # Decision variables: [g_0, ..., g_{n-1}, M] (total n+1)
    # Minimize M: c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A_conv @ g - M <= 0 for each of 2n-1 constraints
    # [A_conv, -I] @ [g; M] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_conv
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g_i >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        g_val = np.maximum(result.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(result.x[n])
    return None, None


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of (a*a)_k using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0).copy()
    s = float(np.sum(v))
    if s < 1e-15:
        return np.full(len(v), target_sum / len(v))
    return v * (target_sum / s)


def nesterov_polish(a, time_budget, tau_init=30.0, tau_max=2000.0):
    """Nesterov accelerated gradient descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = project_simplex(y, 1.0)

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adaptive_lr = 0.5 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.1 * base_lr), 3.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adaptive_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = project_simplex(a_new, 1.0)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr *= 0.9998
        tau = min(tau * 1.02, tau_max)
        it += 1

    return best_a


def fourier_perturb(a, rng, strength=0.03):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 8
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 2.0) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, rng, strength=0.05):
    """Add random multiplicative perturbation."""
    n = len(a)
    noise = rng.standard_normal(n) * strength
    a_new = a * np.exp(noise)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_alternate_fast(a, time_budget, seed=42, verbose=False, max_iters=None):
    """
    Fast LP-based alternating minimization using scipy HiGHS.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    t_blend = 0.5
    stall_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1:
            break
        if max_iters is not None and iter_count >= max_iters:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_lp_help(a)
        lp_time = time.monotonic() - t0
        iter_count += 1

        if g_lp is None:
            break

        # Try several blending factors
        candidates_t = [t_blend, t_blend * 0.5, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if iter_count > 100:
            candidates_t.extend([0.05, 0.02, 0.01])

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend

        for t in candidates_t:
            t = min(max(t, 0.001), 1.0)
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            t_blend = max(0.01, best_local_t * 0.99)
            stall_count = 0
            if verbose and iter_count % 50 == 0:
                print(f"    LP {iter_count}: obj={best_obj:.10f}, t={t_blend:.4f}, "
                      f"lp_time={lp_time:.3f}s")
        else:
            stall_count += 1
            if stall_count >= 3:
                strength = 0.01 + 0.005 * min(stall_count, 15)
                a = fourier_perturb(best_a, rng, strength=strength)
                t_blend = max(0.1, min(t_blend * 1.3, 0.8))
                stall_count = 0
            elif stall_count >= 1:
                a = best_a + 0.003 * rng.standard_normal(n) * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()

    if verbose:
        print(f"    LP alt: {iter_count} iters, best_obj={best_obj:.10f}")
    return best_a


def block_coordinate_descent(a, time_budget, block_size=32, tau=50.0):
    """
    Block coordinate descent: optimize blocks of coordinates using
    local gradient descent, keeping other blocks fixed.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    n_blocks = (n + block_size - 1) // block_size

    while time.monotonic() - start < time_budget:
        improved = False
        for b in range(n_blocks):
            idx = slice(b * block_size, min((b + 1) * block_size, n))
            n_block = idx.stop - idx.start

            # Local optimization of this block
            a_block = a[idx].copy()
            others_sum = s - float(np.sum(a_block))

            for local_it in range(50):
                # Gradient w.r.t. this block
                g_full = smoothed_gradient(a, tau)
                g_block = g_full[idx].copy()

                # Also account for the constraint that sum should stay = 1
                # (we'll project after each step)
                lr = 0.1 / (n_block + 1)
                a_block -= lr * g_block
                a_block = np.maximum(a_block, 0)

                # Renormalize: keep total sum = 1
                # Scale this block so total sum = 1
                block_sum = float(np.sum(a_block))
                if others_sum + block_sum > 1e-10:
                    a_block = a_block * (1.0 - others_sum) / block_sum if block_sum > 1e-10 else np.full(n_block, (1.0 - others_sum) / n_block)
                    a_block = np.maximum(a_block, 0)
                    # Re-normalize to ensure sum = 1 - others_sum
                    bs = float(np.sum(a_block))
                    if bs > 1e-10:
                        a_block = a_block * (1.0 - others_sum) / bs

                a[idx] = a_block

                obj = compute_obj(a)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            # Try a smaller tau for finer optimization
            tau = max(tau * 0.8, 5.0)

    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def smooth_gradient_step(a, tau=100.0, lr=0.05, n_steps=1000):
    """Simple gradient descent steps with projection."""
    n = len(a)
    for _ in range(n_steps):
        g = smoothed_gradient(a, tau)
        a = a - lr * g
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            return a
        a = a / s
    return a

**Alternative: Direct non-convex optimization.**

The problem is: min_a max_k (a*a)_k s.t. a >= 0, sum(a) = 1.

- Penalty methods

(a*a)_k ≈ (a^t * a^t)_k + 2 * (a^t * delta)_k (first-order Taylor)

where delta = a - a^t. This gives a linear constraint in delta, so the subproblem is an LP. This is essentially the same as the LP alternating (with g = a^t and optimizing over delta).

def line_search_blend(a, g, n_evals=20):

Now, let me also think about **convergence acceleration**. The LP alternating is like a "best response" method. In game theory, the best response method can be accelerated using:
- **Fictitious play**: Use the average of past best responses
- **Regret matching**: Weight past best responses by regret
- **Virtual regret matching**: A variant that converges faster

For our problem, **fictitious play** would be:
a_{t+1} = (t*a_t + g_t) / (t+1)

This is a convex combination with t/(t+1) weight on the old and 1/(t+1) on the new. As t grows, the update slows down, which helps convergence.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast LP-based alternating minimization (scipy HiGHS) with line-search blending,
fictitious play acceleration, multi-start diverse initialization, Nesterov gradient
polishing, and adaptive n-exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_norm):
    """Build matrix (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm
    return A


def solve_lp_help(a):
    """
    Solve LP: min M s.t. (a_norm * g)_k <= M for all k, sum(g)=1, g>=0.
    Uses scipy linprog with HiGHS.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A_conv = build_conv_matrix(a_norm)

    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality: A_conv @ g - M <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_conv
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.success:
            g_val = np.maximum(result.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(result.x[n])
    except Exception:
        pass

    # Fallback to cvxpy
    try:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A_conv @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except Exception:
        pass

    return None, None


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of (a*a)_k using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum} via scaling."""
    v = np.maximum(v, 0).copy()
    s = float(np.sum(v))
    if s < 1e-15:
        return np.full(len(v), target_sum / len(v))
    return v * (target_sum / s)


def line_search_blend(a, g, n_coarse=15, n_fine=15):
    """Find optimal t in [0,1] to minimize obj((1-t)*a + t*g)."""
    best_t = 0.0
    best_obj = compute_obj(a)

    # Coarse search
    ts = np.linspace(0, 1, n_coarse)
    for t in ts[1:]:
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    # Fine search around best
    if 0 < best_t < 1:
        step = 1.0 / n_coarse
        t_lo = max(0, best_t - step)
        t_hi = min(1, best_t + step)
        ts_fine = np.linspace(t_lo, t_hi, n_fine)
        for t in ts_fine:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t

    return best_t, best_obj


def nesterov_polish(a, time_budget, tau_init=40.0, tau_max=3000.0):
    """Nesterov accelerated gradient descent with adaptive step and projection."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.92
    base_lr = 1.5 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = project_simplex(y, 1.0)

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adaptive_lr = 0.3 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.05 * base_lr), 2.0 * base_lr)
            lr = 0.85 * lr + 0.15 * adaptive_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = project_simplex(a_new, 1.0)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr * 0.5
                stall = 0

        a = a_new
        lr *= 0.9997
        tau = min(tau * 1.03, tau_max)
        it += 1

    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 10
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 2.0) / (k ** 0.7)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))

    return a_new / s

def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_starts(a_parent, rng):
    """Generate diverse starting points given a parent."""
    n = len(a_parent)
    starts = []

    # Parent itself
    starts.append(('parent', a_parent.copy()))

    # Perturbed parent
    starts.append(('perturbed', fourier_perturb(a_parent, rng, 0.015)))
    starts.append(('perturbed2', random_perturb(a_parent, rng, 0.03)))

    # Random
    starts.append(('random', rng.dirichlet(np.ones(n) * 0.5)))

    # Smooth (low-freq Fourier)
    freqs = np.arange(n)
    k1 = rng.integers(1, 4)
    k2 = rng.integers(4, 8)
    ph1 = rng.uniform(0, 2 * np.pi)
    ph2 = rng.uniform(0, 2 * np.pi)
    smooth = 1.0 + 0.5 * np.cos(2 * np.pi * k1 * freqs / n + ph1) + \
             0.3 * np.cos(2 * np.pi * k2 * freqs / n + ph2)
    smooth = np.maximum(smooth, 0.01)
    starts.append(('smooth', smooth / smooth.sum()))

    # Bimodal
    center = rng.uniform(0.2, 0.8) * n
    width = rng.uniform(n * 0.05, n * 0.15)
    bimodal = np.exp(-((freqs - center) / width) ** 2) + \
              np.exp(-((freqs - (n - center)) / width) ** 2)
    bimodal = np.maximum(bimodal, 0.001)
    starts.append(('bimodal', bimodal / bimodal.sum()))

    return starts

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 90:
            break
        a_try = resample_to_length(best_a, n_try)
        alloc = min(remaining * 0.18, 130.0)
        if alloc < 40:
            break

        a_try = lp_alternate_linesearch(a_try, time_budget=alloc * 0.7,
                                         seed=n_try * 7, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_try = nesterov_polish(a_try, time_budget=min(remaining * 0.2, 35.0),
                                     tau_init=100.0, tau_max=2500.0)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  n={n_try}: obj={best_obj:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 6: Final intensive refinement loop ===
    print(f"\n=== Phase 6: Final Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_cur = best_a.copy()
        it = 0
        while time.monotonic() < deadline and it < 15:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            # LP step
            alloc_lp = min(remaining * 0.35, 60.0)
            a_cur = lp_alternate_linesearch(a_cur, time_budget=alloc_lp,
                                             seed=seed + 9999 + it * 77, verbose=False,
                                             use_fp=(it % 2 == 0))
            obj_cur = compute_obj(a_cur)
            if obj_cur < best_obj:
                best_obj = obj_cur
                best_a = a_cur.copy()
                print(f"    Final LP {it}: obj={best_obj:.10f} **")

            # Gradient step
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc_gd = min(remaining * 0.25, 30.0)
            a_cur = nesterov_polish(a_cur, time_budget=alloc_gd,
                                     tau_init=150.0, tau_max=5000.0)
            obj_cur = compute_obj(a_cur)
            if obj_cur < best_obj:
                best_obj = obj_cur
                best_a = a_cur.copy()
                print(f"    Final GD {it}: obj={best_obj:.10f} **")
            it += 1

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
