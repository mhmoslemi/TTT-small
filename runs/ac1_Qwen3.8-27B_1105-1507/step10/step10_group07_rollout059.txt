```python
"""
Hybrid optimizer for minimum peak convolution: combines fast FFT-based
smoothed gradient descent (logsumexp), subgradient methods with top-K
peaks, and short convex (QP/SOCP) polishing. Designed to improve upon
TR-SOCP-only approaches by doing many more cheap iterations.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import fft, ifft, rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.argwhere(cond)
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1, 0]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cross_corr_conv(p, a):
    """
    Compute cross-correlation (p ⋆ a)[i] = sum_k p[k]*a[k-i] for i=0..n-1.
    p has length 2n-1, a has length n.
    """
    n = len(a)
    N = 1
    while N < len(p) + n - 1:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    result = ifft(P * np.conj(A)).real
    return result[:n]


def smoothed_gradient_descent(a, time_budget=200.0, beta_start=5.0, beta_end=500.0, verbose=False):
    """
    Smoothed gradient descent using logsumexp smoothing of the max.
    f(a) ≈ 2n * logsumexp(beta * conv(a,a)) with sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 1.0  # step size
    it = 0
    rng = np.random.default_rng(123)

    # Precompute FFT size for cross-correlation
    N_cc = 1
    while N_cc < 3 * n - 2:
        N_cc <<= 1

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-9))
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.5)

        # Compute convolution
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Softmax weights
        shifted = beta * (conv - max_conv)  # for numerical stability
        p = np.exp(shifted)
        p_sum = float(np.sum(p))
        if p_sum < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = p / p_sum

        # Cross-correlation gradient
        g = cross_corr_conv(p, a)
        grad = 2.0 * n * beta * g

        # Gradient step
        a_new = a - eta * grad
        a_new = project_simplex(a_new)

        # Evaluate
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

        # Adaptive step size
        if it % 50 == 0:
            # Reduce step size slowly
            eta *= 0.999
        # Occasionally add small noise to escape local minima
        if it % 200 == 100:
            a = best_a * np.exp(rng.normal(0, 0.003, n))
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-10:
                a = a / s_a

        it += 1
        if verbose and it % 100 == 0:
            print(f"    SGrad it{it}: obj={best_obj:.8f}, beta={beta:.1f}, eta={eta:.4f}, t={elapsed:.0f}s")

    return best_a


def subgradient_descent(a, time_budget=150.0, K=5, verbose=False):
    """
    Subgradient method using top-K peaks of the convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    lr = 5.0 / np.sqrt(n)
    momentum = 0.75
    v = np.zeros(n)
    stall = 0
    rng = np.random.default_rng(456)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Top K peaks
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]

        # Soft weights over top K
        weights = np.exp((peak_vals - max_pv) * 5.0 / max(max_pv, 1e-10))
        weights /= weights.sum()

        # Compute subgradient: g[i] = 2 * sum_{k in top_K} w_k * a[k-i]
        # Build indicator vector for top-K (weighted)
        p = np.zeros(2 * n - 1)
        for w, k in zip(weights, top_idx):
            p[int(k)] = w

        g = 2.0 * cross_corr_conv(p, a)

        # Momentum step
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            lr *= 0.998
            if stall % 500 == 0:
                # Perturb to escape
                a = best_a * np.exp(rng.normal(0, 0.005, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                v *= 0.5
            if lr < 1e-10:
                break

        it += 1
        if verbose and it % 200 == 0:
            print(f"    SubGrad it{it}: obj={best_obj:.8f}, lr={lr:.6f}, stall={stall}, t={time.monotonic()-start:.0f}s")

    return best_a
```

```python
def hankel_matvec(w, a):
    """Compute W @ a where W[m,j] = w[m+j]. Uses FFT cross-correlation."""
    n = len(a)
    return cross_corr_conv(w, a)


def qp_polish(a, time_budget=100.0, p_power=2.0, verbose=False):
    """
    Iteratively reweighted QP polishing.
    Each step: given current a, compute weights w from conv(a,a),
    then solve min a^T W a s.t. sum(a)=1, a>=0 via projected gradient.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        # Compute weights
        conv = compute_conv(a)
        # Use power of convolution values as weights
        w = np.maximum(conv, 1e-15) ** p_power
        w = w / w.sum()  # normalize

        # Solve QP: min x^T W x s.t. sum(x)=1, x>=0
        # via projected gradient
        x = a.copy()
        # Step size: 1 / (2 * lambda_max(W)) approx 1 / (2 * trace(W)/n * 2)
        # trace(W) = sum_m W[m,m] = sum_m w[2m]
        trace_W = float(np.sum(w[0:2*n-1:2]))
        L = 2.0 * trace_W  # upper bound on 2*lambda_max(W)
        eta_qp = 1.0 / L

        for inner in range(200):
            # Gradient: 2 * W @ x
            grad = 2.0 * hankel_matvec(w, x)
            x_new = x - eta_qp * grad
            x_new = project_simplex(x_new)
            x = x_new

        # Check objective
        obj_new = compute_obj(x)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = x.copy()
            a = x.copy()
        else:
            # Try different power
            p_power *= 1.5
            if p_power > 100:
                break

        it += 1
        if verbose and it % 5 == 0:
            print(f"    QP it{it}: obj={best_obj:.8f}, p_power={p_power:.1f}, t={time.monotonic()-start:.0f}s")

    return best_a
```

```python
"""
Hybrid FFT-based optimizer for minimum peak convolution ratio.
Combines fast smoothed gradient descent (logsumexp), subgradient methods,
iteratively reweighted QP polishing, multi-length exploration, and short
SOCP trust-region refinement. Key advantage over pure SOCP approaches:
O(n log n) per iteration enables 10x+ more optimization steps.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import fft, ifft, rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cross_corr_conv(p, a):
    """
    Compute g[i] = sum_k p[k]*a[k-i] for i=0..n-1.
    p has length 2n-1, a has length n. Uses FFT.
    """
    n = len(a)
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    P = fft(p, N)
    Arev = fft(a[::-1], N)
    conv_full = ifft(P * Arev).real
    return conv_full[n - 1:2 * n - 1]


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def smoothed_gradient_descent(a, time_budget=200.0, beta_start=8.0, beta_end=600.0,
                              verbose=False, seed=42, noise_interval=300, noise_scale=0.004):
    """
    Smoothed gradient descent using logsumexp smoothing of the max.
    f(a) ~ 2n * logsumexp(beta * conv(a,a)) with sum(a)=1, a>=0.
    Very fast: O(n log n) per iteration via FFT.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.8 / np.sqrt(n) * 10
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-9))
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.4)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        shifted = beta * (conv - max_conv)
        p = np.exp(shifted)
        p_sum = float(np.sum(p))
        if p_sum < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = p / p_sum

        g = cross_corr_conv(p, a)
        grad = 2.0 * n * beta * g

        a_new = a - eta * grad
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        eta *= 0.9995

        if it > 0 and it % noise_interval == 0:
            a = best_a * np.exp(rng.normal(0, noise_scale, n))
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-10:
                a = a / s_a

        it += 1
        if verbose and it % 200 == 0:
            print(f"    SGrad it{it}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.0f}s")

    return best_a


def subgradient_descent(a, time_budget=150.0, K=5, verbose=False, seed=77):
    """Subgradient method using top-K peaks with momentum and adaptive restarts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    lr = 8.0 / n
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]

        weights = np.exp((peak_vals - max_pv) * 6.0 / max(max_pv, 1e-10))
        weights /= weights.sum()

        p = np.zeros(2 * n - 1)
        for w_i, k_i in zip(weights, top_idx):
            p[int(k_i)] = w_i

        g = 2.0 * cross_corr_conv(p, a)

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall == 200:
                lr *= 0.7
                v *= 0.4
            elif stall == 500:
                a = best_a * np.exp(rng.normal(0, 0.004, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                v = np.zeros(n)
                lr = 8.0 / n
                stall = 0
            if lr < 1e-12:
                break

        it += 1
        if verbose and it % 300 == 0:
            print(f"    SubGrad it{it}: obj={best_obj:.8f}, lr={lr:.6f}, t={time.monotonic()-start:.0f}s")

    return best_a


def qp_polish(a, time_budget=120.0, p_power=3.0, verbose=False, seed=99):
    """
    Iteratively reweighted QP polishing.
    Each outer step: compute weights from conv, solve QP min x^T W x s.t. sum(x)=1, x>=0.
    W is Hankel: W[m,j] = w[m+j]. Solved via cvxpy.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        w = np.maximum(conv, 1e-12) ** p_power
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Build Hankel matrix W (n x n) with W[m,j] = w[m+j]
        W = np.zeros((n, n))
        for m in range(n):
            W[m, :] = w[m:m + n]

        x = cp.Variable(n, nonneg=True)
        try:
            objective = cp.quad_form(x, W, assume_PSD=True)
        except Exception:
            objective = cp.quad_form(x, (W + W.T) / 2, assume_PSD=True)

        constraints = [cp.sum(x) == 1]
        prob = cp.Problem(cp.Minimize(objective), constraints)

        x_val = None
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    x_val = np.maximum(np.asarray(x.value).flatten(), 0)
                    x_s = x_val.sum()
                    if x_s > 1e-10:
                        x_val = x_val / x_s
                    break
            except Exception:
                continue

        if x_val is None:
            # Fallback: projected gradient on the QP
            x_pg = a.copy()
            eta_qp = 0.5 / max(float(np.trace(W)) / n, 1e-6)
            for _ in range(100):
                grad = 2.0 * (W @ x_pg)
                x_pg = project_simplex(x_pg - eta_qp * grad)
            x_val = x_pg

        obj_new = compute_obj(x_val)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = x_val.copy()
            a = x_val.copy()
            p_power = max(1.0, p_power * 0.8)
        else:
            p_power *= 1.3
            if p_power > 200:
                break

        it += 1
        if verbose and it % 3 == 0:
            print(f"    QP it{it}: obj={best_obj:.8f}, p={p_power:.1f}, t={time.monotonic()-start:.0f}s")

    return best_a


def solve_tr_sscp(a, r):
    """Solve trust region SOCP for one step."""
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)

    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta

    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
        cp.norm(delta, 2) <= r,
    ]

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    result = np.interp(x_new, x_old, a)
    return np.maximum(result, 0)


def coordinate_descent_refine(a, time_budget=60.0, verbose=False):
    """
    Coordinate descent: for each coordinate, find the optimal value
    given all others fixed. Uses the structure of the convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        improved = False
        for i in range(n):
            # Current contribution of a[i] to convolution
            # c[k] = sum_m a[m]*a[k-m]
            # If we change a[i] to a[i]+t, c[k] changes by:
            #   t*a[k-i] (from m=i term) + t*a[i-k] (from k-m=i term)
            #   = t*(a[k-i] + a[i-k]) with a[j]=0 outside [0,n-1]
            # But if k-i = i (k=2i), it's just t*a[i] (a[i]^2 term)
            
            # For simplicity, use a small perturbation approach
            # Try a few values of a[i]
            old_val = a[i]
            if old_val < 1e-12:
                continue
            
            s_others = s - old_val
            candidates = [old_val * 0.5, old_val * 0.7, old_val * 1.3, old_val * 1.5,
                          old_val + 0.001, max(0, old_val - 0.001)]
            best_cand = old_val
            best_cand_obj = best_obj
            
            for cand in candidates:
                cand = max(0.0, cand)
                a[i] = cand
                new_s = a.sum()
                if new_s < 1e-10:
                    a[i] = old_val
                    continue
                a_norm = a / new_s
                obj_cand = compute_obj(a_norm)
                if obj_cand < best_cand_obj - 1e-14:
                    best_cand_obj = obj_cand
                    best_cand = cand
            
            a[i] = best_cand
            if best_cand != old_val:
                improved = True
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                if compute_obj(a) < best_obj - 1e-13:
                    best_obj = compute_obj(a)
                    best_a = a.copy()

        it += 1
        if not improved and it > 3:
            break
        if verbose and it % 2 == 0:
            print(f"    CoordDescent it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase hybrid optimizer combining fast gradient methods,
    QP polishing, TR-SOCP, multi-length exploration, and coordinate descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Smoothed GD (primary fast optimizer) ===
    print(f"\n=== Phase 1: Smoothed Gradient Descent (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.15, 200.0)
    if p1_budget > 20:
        a1 = smoothed_gradient_descent(best_seq, time_budget=p1_budget,
                                       beta_start=10.0, beta_end=800.0,
                                       verbose=True, seed=42)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1 SGrad: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 SGrad: obj={obj1:.8f}")

    # === Phase 2: Subgradient refinement ===
    print(f"\n=== Phase 2: Subgradient Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.10, 130.0)
    if p2_budget > 15:
        a2 = subgradient_descent(best_seq, time_budget=p2_budget, K=5,
                                 verbose=True, seed=77)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 SubGrad: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 SubGrad: obj={obj2:.8f}")

    # === Phase 3: QP Polish ===
    print(f"\n=== Phase 3: QP Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 150.0)
    if p3_budget > 20:
        a3 = qp_polish(best_seq, time_budget=p3_budget, p_power=3.0,
                       verbose=True, seed=99)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 QP: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 QP: obj={obj3:.8f}")

    # === Phase 4: TR-SOCP Polish ===
    print(f"\n=== Phase 4: TR-SOCP Polish ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.18, 250.0)
    if p4_budget > 30:
        a4 = tr_sscp_polish(best_seq, time_budget=p4_budget, r_init=0.2,
                            verbose=True, seed=200)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4 TR-SOCP: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 TR-SOCP: obj={obj4:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = []
    for delta in [-200, -100, 100, 200, 300, 400, 500]:
        n_try = n_current + delta
        if 500 <= n_try <= 1800 and n_try != n_current:
            n_options.append(n_try)
    # Also try some absolute values
    for n_abs in [600, 700, 900, 1000, 1100, 1300, 1500]:
        if 500 <= n_abs <= 1800 and n_abs != n_current and n_abs not in n_options:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))[:7]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)

        # Quick smoothed GD
        alloc1 = min(remaining * 0.08, 30.0)
        if alloc1 > 10:
            a_try = smoothed_gradient_descent(a_try, time_budget=alloc1,
                                              beta_start=8.0, beta_end=400.0,
                                              seed=300 + n_try)

        # Quick subgradient
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.06, 25.0)
        if alloc2 > 8:
            a_try = subgradient_descent(a_try, time_budget=alloc2, K=5,
                                        seed=400 + n_try)

        # Short TR-SOCP
        remaining = deadline - time.monotonic()
        alloc3 = min(remaining * 0.10, 60.0)
        if alloc3 > 15:
            a_try = tr_sscp_polish(a_try, time_budget=alloc3, r_init=0.2,
                                   seed=500 + n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Perturbation + re-optimization ===
    print(f"\n=== Phase 6: Perturbation + Re-optimization ===")
    remaining = deadline - time.monotonic()

    for r_idx in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        scale = 0.006 + 0.006 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            break
        a_pert = a_pert / s_pert

        # Smoothed GD
        alloc1 = min(remaining * 0.15, 50.0)
        if alloc1 > 15:
            a_pert = smoothed_gradient_descent(a_pert, time_budget=alloc1,
                                               beta_start=10.0, beta_end=500.0,
                                               seed=600 + r_idx)

        # TR-SOCP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.12, 50.0)
        if alloc2 > 15:
            a_pert = tr_sscp_polish(a_pert, time_budget=alloc2, r_init=0.15,
                                    seed=700 + r_idx)

        obj_pert = compute_obj(a_pert)
        if obj_pert < best_score:
            best_score = obj_pert
            best_seq = a_pert.copy()
            print(f"  Pert{r_idx} (scale={scale:.4f}): obj={best_score:.8f} **")
        else:
            print(f"  Pert{r_idx} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 7: Coordinate descent refinement ===
    print(f"\n=== Phase 7: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.15, 80.0)
        a_cd = coordinate_descent_refine(best_seq, time_budget=alloc, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CoordDescent: obj={best_score:.8f} **")
        else:
            print(f"  CoordDescent: obj={obj_cd:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 100.0)
        a_final = tr_sscp_polish(best_seq, time_budget=alloc, r_init=0.12,
                                 verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final TR: obj={best_score:.8f} **")
        else:
            print(f"  Final TR: obj={obj_final:.8f}")

    # Final quick smoothed GD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 50.0)
        a_sg = smoothed_gradient_descent(best_seq, time_budget=alloc,
                                         beta_start=50.0, beta_end=1000.0,
                                         seed=1111, noise_scale=0.002)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SGrad: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
