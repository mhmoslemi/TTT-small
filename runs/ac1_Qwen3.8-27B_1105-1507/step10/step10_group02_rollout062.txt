```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, verbose=False):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None
```

Let me also add one more technique: **subgradient descent with exponential weights (mirror descent)**.

The idea:
- Objective: f(a) = 2n * max_k (a*a)[k] / (sum(a))^2
- Subgradient of max_k (a*a)[k] with respect to a[i]:
  For each k in argmax, the contribution to the gradient of a[i] is 2*a[k-i] (if valid)
  So the full subgradient component for a[i] is: 2 * sum_{k in argmax} a[k-i]

- For a smooth approximation, use log-sum-exp:
  F(a) = (1/tau) * log(sum_k exp(tau * (a*a)[k]))
  Gradient: (1/tau) * sum_k exp(tau*(a*a)[k]) / sum_j exp(tau*(a*a)[j]) * 2*a[k-i]

- Mirror descent update:
  a[i] <- a[i] * exp(-lr * grad_smooth[i])
  Then normalize: a <- a / sum(a)

This is the "exponential weights" or "Hedge" algorithm applied to our problem.

```python
"""
Optimizes step function coefficients using an enhanced hybrid approach combining:
1. Exact coordinate descent with golden section search (widened brackets)
2. Aggressive LP-based direction finding with continuous blend search
3. Pairwise coordinate descent with Nelder-Mead 2D subproblem solving
4. Mirror descent (exponential weights) with log-sum-exp smoothing
5. Systematic multi-n exploration with warm-starting
6. Stochastic perturbation for escaping local minima
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_obj_from_conv(a, conv):
    """Compute objective given precomputed convolution (assumes a is normalized)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def golden_section_min(f, lo, hi, max_iter=50):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-12 * max(1.0, abs(hi)):
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return max(0.0, (lo + hi) / 2.0)


def coordinate_descent(a, time_budget, seed, verbose=False):
    """
    Exact coordinate descent with golden section search.
    Prioritizes coordinates that contribute to peak convolution values.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        
        # Periodically recompute conv
        if sweep % 3 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
        
        # Priority ordering: coordinates near peak get priority
        conv_max = float(np.max(conv))
        peak_threshold = conv_max * 0.9
        peak_ks = np.where(conv >= peak_threshold)[0]
        
        # Compute priority for each coordinate
        priority = np.zeros(n)
        for k in peak_ks:
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    priority[i] += conv[k] * a[ki]
        
        # Combine priority with random ordering
        order = np.zeros(n)
        order += priority / (priority.max() + 1e-10)
        order += 0.3 * rng.random(n)
        order = np.argsort(-order)
        
        S = float(np.sum(a))
        
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            
            v = a[i]
            S_other = S - v
            
            if S_other < 1e-12:
                continue
            
            # Determine valid k range for this coordinate
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()
            
            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]
            
            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0
            
            # Define 1D objective for golden section
            def f_1d(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom
            
            # Quick scan for good initial bracket
            scale = max(v, S / n, 1e-6)
            # Wider bracket than before
            candidates = [0.0, v * 0.1, v * 0.3, v * 0.5, v * 0.7, v,
                          v * 1.3, v * 1.5, v * 2.0, v * 3.0, v * 5.0,
                          S / (2 * n), S / n, S / (4 * n)]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))
            
            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x
            
            # Golden section in a wider neighborhood
            x_lo = max(0.0, best_x - 2 * scale)
            x_hi = best_x + 3 * scale
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8
            
            x_opt = golden_section_min(f_1d, x_lo, x_hi, max_iter=40)
            
            if abs(x_opt - v) < 1e-14:
                continue
            
            # Update a and conv
            delta = x_opt - v
            a[i] = x_opt
            
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            
            S += delta
            
            # Track best
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a
```

```python
"""
Optimizes step function coefficients using an enhanced hybrid approach combining:
1. Exact coordinate descent with priority-based ordering and widened brackets
2. Aggressive LP-based direction finding with continuous blend search and power-iteration blends
3. Pairwise coordinate descent with 2D Nelder-Mead subproblem solving
4. Mirror descent (exponential weights) with log-sum-exp smoothed gradients
5. Systematic multi-n exploration with warm-starting

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def cd_1d_objective(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i):
    """1D subproblem objective for coordinate descent."""
    if len(a_ki_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom


def golden_section_cd(lo, hi, max_iter, v, S_other, n, c_k_lin, a_ki_lin, c_2i):
    """Golden section search for cd_1d_objective on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = cd_1d_objective(c, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
    fd = cd_1d_objective(d, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
    for _ in range(max_iter):
        if hi - lo < 1e-13 * max(1.0, abs(hi)):
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = cd_1d_objective(c, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = cd_1d_objective(d, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
    return max(0.0, (lo + hi) / 2.0)


def coordinate_descent(a, time_budget, seed, verbose=False):
    """Exact coordinate descent with priority-based ordering."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 3 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        conv_max = float(np.max(conv))
        peak_threshold = conv_max * 0.85
        peak_ks = np.where(conv >= peak_threshold)[0]

        priority = np.zeros(n)
        for k in peak_ks:
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    priority[i] += a[ki]

        order_score = priority / (priority.max() + 1e-10) + 0.4 * rng.random(n)
        order = np.argsort(-order_score)

        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            scale = max(v, S / n, 1e-6)
            candidates = [0.0, v * 0.1, v * 0.3, v * 0.5, v * 0.7, v,
                          v * 1.3, v * 1.5, v * 2.0, v * 3.0, v * 5.0,
                          S / (4.0 * n), S / (2.0 * n), S / n, S / n * 2.0]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = cd_1d_objective(v, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            for x in candidates:
                fx = cd_1d_objective(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            x_lo = max(0.0, best_x - 2.0 * scale)
            x_hi = best_x + 3.0 * scale
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            x_opt = golden_section_cd(x_lo, x_hi, 40, v, S_other, n,
                                      c_k_lin, a_ki_lin, c_2i)

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print("    CD sweep {}: obj={:.8f}, t={:.1f}s".format(sweep, best_obj, elapsed))

    return best_a


def pairwise_obj(params, a, i, j, n):
    """2D objective for pairwise coordinate descent."""
    x, y = float(params[0]), float(params[1])
    if x < 0 or y < 0:
        return 1e10
    a_try = a.copy()
    a_try[i] = x
    a_try[j] = y
    s = float(np.sum(a_try))
    if s < 1e-10:
        return 1e10
    a_try = a_try / s
    conv = compute_conv(a_try)
    return 2.0 * n * float(np.max(conv))


def pairwise_cd(a, time_budget, seed, verbose=False):
    """Pairwise coordinate descent with Nelder-Mead 2D optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0

    conv = compute_conv(a)
    conv_max = float(np.max(conv))
    peak_ks = np.where(conv >= conv_max * 0.9)[0]

    while time.monotonic() - start < time_budget:
        it += 1

        if it % 50 == 0:
            conv = compute_conv(a)
            conv_max = float(np.max(conv))
            peak_ks = np.where(conv >= conv_max * 0.9)[0]

        # Pick pair: prefer coordinates near peak locations
        if len(peak_ks) > 0:
            k = peak_ks[rng.integers(len(peak_ks))]
            valid_i = [i for i in range(n) if 0 <= k - i < n]
            if len(valid_i) >= 2:
                idx = rng.choice(len(valid_i), size=2, replace=False)
                i, j = valid_i[idx[0]], valid_i[idx[1]]
            else:
                i, j = rng.choice(n, size=2, replace=False)
        else:
            i, j = rng.choice(n, size=2, replace=False)

        result = minimize(pairwise_obj, [a[i], a[j]], method='Nelder-Mead',
                          args=(a, i, j, n),
                          options={'maxiter': 40, 'xatol': 1e-10, 'fatol': 1e-12,
                                   'adaptive': True})

        if result.fun < best_obj - 1e-12:
            a[i] = max(0.0, float(result.x[0]))
            a[j] = max(0.0, float(result.x[1]))
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print("    Pairwise it {}: obj={:.8f}, t={:.1f}s".format(it, best_obj, elapsed))

    return best_a

def mirror_descent(a, time_budget, seed, verbose=False):
    """Mirror descent with exponential weights and log-sum-exp smoothing."""
    n = len(a)
    a = np.maximum(a, 1e-12).astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    beta = 0.7
    v = np.zeros(n)
    lr = 2.0 / np.sqrt(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = 30.0 + 300.0 * progress

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Compute gradient via FFT cross-correlation
        N_md = 1
        while N_md < 3 * n:
            N_md <<= 1
        P = fft(p, N_md)
        A_md = fft(a, N_md)
        corr = ifft(P * np.conj(A_md)).real
        grad = 2.0 * corr[:n]

        # Nesterov momentum
        v = beta * v - lr * grad
        a_euclidean = a + v
        a_euclidean = np.maximum(a_euclidean, 1e-12)

        # Mirror (exponential) step
        a_mirror = a * np.exp(-lr * grad)
        a_mirror = np.maximum(a_mirror, 1e-12)

        # Blend
        a_new = 0.4 * a_euclidean + 0.6 * a_mirror
        a_new = np.maximum(a_new, 1e-12)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 2000 == 0:
            print("    MD it {}: obj={:.8f}, tau={:.0f}, lr={:.4f}".format(
                it, best_obj, tau, lr))

        # Adaptive learning rate
        lr = max(lr * 0.9999, 0.5 / np.sqrt(n))

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x ** 2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 8, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Frequency-domain inspired: flat spectrum target
        a = np.ones(n) / n
        # Add small perturbations at high frequency
        a += 0.1 * rng.normal(0, 1, n)
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple optimization strategies.
    Uses CD, LP polishing, pairwise CD, mirror descent, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print("Parent (n={}): obj={:.8f}".format(parent.size, best_score))
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print("Default init (n={}): obj={:.8f}".format(n_default, best_score))

    # === Phase 1: Quick Coordinate Descent ===
    print("\n=== Phase 1: Quick CD ===")
    remaining = deadline - time.monotonic()
    p1 = min(remaining * 0.1, 120.0)
    if p1 > 20:
        a1 = coordinate_descent(best_seq, time_budget=p1, seed=seed, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print("  Phase 1: obj={:.8f} **".format(best_score))
        else:
            print("  Phase 1: obj={:.8f} (no improvement)".format(obj1))

    # === Phase 2: LP-CD Interleaved Loop ===
    print("\n=== Phase 2: LP-CD Loop ===")
    remaining = deadline - time.monotonic()
    p2 = min(remaining * 0.25, 350.0)
    if p2 > 30:
        a2 = lp_cd_loop(best_seq, time_budget=p2, seed=seed + 1, verbose=False)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print("  Phase 2: obj={:.8f} **".format(best_score))
        else:
            print("  Phase 2: obj={:.8f}".format(obj2))

    # === Phase 3: Pairwise CD ===
    print("\n=== Phase 3: Pairwise CD ===")
    remaining = deadline - time.monotonic()
    p3 = min(remaining * 0.12, 150.0)
    if p3 > 20:
        a3 = pairwise_cd(best_seq, time_budget=p3, seed=seed + 2, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print("  Phase 3: obj={:.8f} **".format(best_score))
        else:
            print("  Phase 3: obj={:.8f}".format(obj3))

    # === Phase 4: CD Refinement after Pairwise ===
    print("\n=== Phase 4: CD Refinement ===")
    remaining = deadline - time.monotonic()
    p4 = min(remaining * 0.12, 150.0)
    if p4 > 15:
        a4 = coordinate_descent(best_seq, time_budget=p4, seed=seed + 3, verbose=False)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print("  Phase 4: obj={:.8f} **".format(best_score))
        else:
            print("  Phase 4: obj={:.8f}".format(obj4))

    # === Phase 5: Multi-n Exploration ===
    print("\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 200), max(400, n_best - 100),
        n_best + 100, n_best + 200, n_best + 400,
        600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) > 30]
    n_options = n_options[:6]
    print("  Trying n values: {}".format(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Short CD on new length
        alloc = min(remaining * 0.15, 50.0)
        if alloc > 15:
            a_try = coordinate_descent(a_try, time_budget=alloc, seed=n_try, verbose=False)

        # Quick LP polish
        remaining2 = deadline - time.monotonic()
        alloc_lp = min(remaining2 * 0.05, 20.0)
        if alloc_lp > 8:
            a_try = lp_cd_loop(a_try, time_budget=alloc_lp, seed=n_try + 1000)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print("  ** n={}: obj={:.8f} **".format(n_try, best_score))
        else:
            print("  n={}: obj={:.8f}".format(n_try, obj_try))

    # === Phase 6: Mirror Descent ===
    print("\n=== Phase 6: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p6 = min(remaining * 0.1, 100.0)
    if p6 > 20:
        # Slight perturbation before MD
        pert = rng.normal(0, 0.01, len(best_seq))
        a_pert = best_seq * np.exp(pert)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        a6 = mirror_descent(a_pert, time_budget=p6, seed=seed + 6, verbose=True)
        obj6 = compute_obj(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print("  Phase 6: obj={:.8f} **".format(best_score))
        else:
            print("  Phase 6: obj={:.8f}".format(obj6))

    # === Phase 7: Diverse Restarts ===
    print("\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = 1 + r * 2
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.4 + 0.15 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 80.0)
        if alloc > 20:
            a_r = coordinate_descent(a_r, time_budget=alloc * 0.5, seed=5000 + r * 100)
            a_r = lp_cd_loop(a_r, time_budget=alloc * 0.5, seed=6000 + r * 100)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print("  Restart {}: obj={:.8f} **".format(r, best_score))
            else:
                print("  Restart {}: obj={:.8f}".format(r, obj_r))

    # === Phase 8: Final Polish ===
    print("\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Final CD
        alloc_cd = min(remaining * 0.4, 150.0)
        if alloc_cd > 15:
            a_final = coordinate_descent(best_seq, time_budget=alloc_cd,
                                         seed=seed + 9999, verbose=False)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print("  Final CD: obj={:.8f} **".format(best_score))

        # Final LP
        remaining2 = deadline - time.monotonic()
        alloc_lp = min(remaining2 * 0.4, 80.0)
        if alloc_lp > 15:
            a_lp = lp_cd_loop(best_seq, time_budget=alloc_lp, seed=7777)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print("  Final LP: obj={:.8f} **".format(best_score))

        # One more short CD
        remaining3 = deadline - time.monotonic()
        alloc_cd2 = min(remaining3 * 0.5, 60.0)
        if alloc_cd2 > 10:
            a_cd = coordinate_descent(best_seq, time_budget=alloc_cd2,
                                      seed=seed + 8888)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print("  Final CD2: obj={:.8f} **".format(best_score))

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print("\n=== FINAL: n={}, score={:.8f}, time={:.1f}s ===".format(
        len(result), best_score, elapsed))

    return result
```
