"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:
- LP-based minimax with dual extraction for exact subgradients
- Polyak-averaged subgradient descent with exact argmax
- Nesterov-accelerated smoothed gradient descent
- Support reduction and multi-n exploration
- Diverse restarts with smart blending

Key improvements: Polyak averaging, exact subgradient at high tau,
adaptive tau based on top-2 gap, better LP warm-starting, wider n sweep.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = fft(a, size)
    fb = fft(b, size)
    return ifft(fa * fb).real[:len(a) + len(b) - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g).
    A[k,j] = a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    if verbose:
                        print(f"    LP solved with {solver}, M={M_val:.10f}")
                    return g_val, M_val
        except Exception:
            continue

    return None, None


def polyak_subgradient_gd(a, time_budget, tau_start=50, tau_end=20000, verbose=True):
    """
    Subgradient descent with Polyak averaging.
    Uses smoothed max (softmax) for gradient, but when tau is very high,
    switches to exact subgradient at the argmax positions.
    Polyak averaging gives better convergence for non-smooth objectives.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0

    # Polyak averaging: maintain running average
    avg_a = a.copy()
    avg_weight = 1.0
    avg_obj = compute_obj(avg_a)

    # Nesterov momentum
    beta = 0.9
    v = np.zeros(n)

    # Track best argmax set for exact subgradient
    argmax_set = np.array([np.argmax(compute_conv(a))])

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Adaptive tau: increase over time
        tau = tau_start + (tau_end - tau_start) * progress ** 0.7

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute convolution and gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))

        # Determine if we should use exact subgradient
        # If the gap between top-1 and top-2 is large relative to 1/tau, use exact
        sorted_vals = np.sort(conv)[::-1]
        if len(sorted_vals) >= 2:
            gap = sorted_vals[0] - sorted_vals[1]
        else:
            gap = np.inf

        if tau * gap > 50:  # softmax is essentially one-hot
            # Exact subgradient: uniform over argmax positions
            threshold = max_conv - 1e-12
            argmax_positions = np.where(conv >= threshold)[0]
            p = np.zeros(len(conv))
            p[argmax_positions] = 1.0 / len(argmax_positions)
        else:
            # Softmax smoothing
            exp_vals = np.exp(tau * (conv - max_conv))
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp

        # Gradient via FFT: grad_j = 2 * sum_k p_k * y_{k-j} (with appropriate bounds)
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        Y = fft(y, N)
        g_full = ifft(P * np.conj(Y)).real
        grad = 2.0 * g_full[:n]

        # Step size: use diminishing step
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break
        # Use the fact that we're on the simplex: step = c / sqrt(it+1)
        step = min(1.0 / (g_norm + 1e-10), 1.0 / np.sqrt(it + 1))
        step = max(step, 1e-8)

        # Update with momentum
        v = beta * v - step * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        # Update Polyak average
        avg_a = (avg_weight * avg_a + a_new) / (avg_weight + 1)
        avg_weight += 1

        # Check both current and average
        obj_new = compute_obj(a_new)
        obj_avg = compute_obj(avg_a)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        if obj_avg < best_obj - 1e-14:
            best_obj = obj_avg
            best_a = avg_a.copy()
            best_it = it

        a = a_new
        it += 1
        beta = min(0.98, 0.9 + 0.08 * progress)

        # Reset if no improvement for too long
        if it - best_it > 50000:
            a = best_a.copy()
            v = np.zeros(n)
            avg_a = best_a.copy()
            avg_weight = 1.0
            best_it = it
            if it > 100000:
                break

        if verbose and it % 10000 == 0:
            print(f"    PSGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.0f}s")

    return best_a


def lp_polish_linesearch(a, time_budget, verbose=True):
    """LP polishing with line-search mixing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.3

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp(a, verbose=False)
        if g_lp is None:
            break

        # Line search over mixing parameter t
        t_candidates = np.concatenate([
            np.linspace(0.05, 1.0, 15),
            [max(0.02, last_t * 0.7), max(0.02, last_t * 0.9), last_t,
             min(1.0, last_t * 1.2), min(1.0, last_t * 1.5)]
        ])
        t_candidates = np.unique(np.clip(t_candidates, 0.01, 1.0))

        best_t_obj = best_obj
        best_t = last_t
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-12:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t
            stall = 0
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.4f}, elapsed={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 2:
                break

    return best_a


def support_shrink_refine(a, time_budget, verbose=True):
    """Try zeroing out small entries and re-optimizing on reduced support."""
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    sorted_vals = np.sort(a)[::-1]
    for frac in [0.2, 0.3, 0.5, 0.7, 0.85, 0.95]:
        if time.monotonic() - start > time_budget:
            break
        k = max(2, int(len(a) * frac))
        if k >= len(a):
            continue
        threshold = sorted_vals[k - 1]
        support = np.where(a >= threshold * 0.5)[0]
        if len(support) < 2:
            continue
        a_shrunk = np.zeros(len(a))
        a_shrunk[support] = a[support]
        a_shrunk = a_shrunk / a_shrunk.sum()

        obj_shrunk = compute_obj(a_shrunk)
        if obj_shrunk < best_obj - 1e-12:
            best_obj = obj_shrunk
            best_a = a_shrunk.copy()
            # Further refine with quick GD
            alloc = min(20, time_budget - 5 - (time.monotonic() - start))
            if alloc > 5:
                a_shrunk = polyak_subgradient_gd(a_shrunk, time_budget=alloc,
                                                  tau_start=200, tau_end=5000, verbose=False)
                obj_shrunk = compute_obj(a_shrunk)
                if obj_shrunk < best_obj:
                    best_obj = obj_shrunk
                    best_a = a_shrunk.copy()
            if verbose:
                print(f"    Shrink frac={frac}: obj={best_obj:.8f}, support={len(support)}")

    return best_a


def log_param_optimize(a, time_budget, tau_min=100, tau_max=10000, verbose=True):
    """Optimize using log-parameterization with L-BFGS on smoothed objective."""
    full_len = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Identify support
    threshold = 1e-8
    support = np.where(a > threshold)[0]
    if len(support) < 2:
        support = np.where(a > a.max() * 1e-6)[0]
    if len(support) < 2:
        return a

    b0 = np.log(np.maximum(a[support], 1e-30))
    b0 = b0 - b0.mean()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    n_stages = 4
    tau_stages = [tau_min * (tau_max / tau_min) ** (i / max(1, n_stages - 1)) for i in range(n_stages)]

    current_b = b0.copy()
    for stage_idx, tau in enumerate(tau_stages):
        if time.monotonic() - start > time_budget:
            break

        def obj_grad(b_vec):
            b_shifted = b_vec - b_vec.max()
            a_supp = np.exp(b_shifted)
            a_supp = a_supp / a_supp.sum()
            a_full = np.zeros(full_len)
            a_full[support] = a_supp

            conv = compute_conv(a_full)
            max_conv = float(np.max(conv))
            exp_vals = np.exp(tau * (conv - max_conv))
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp

            obj_val = max_conv + np.log(sum_exp) / tau

            N = 1
            while N < 3 * full_len:
                N <<= 1
            P = fft(p, N)
            A_full_f = fft(a_full, N)
            g_full = ifft(P * np.conj(A_full_f)).real
            grad_full = 2.0 * g_full[:full_len]

            # Chain rule for softmax parameterization
            grad_b = a_supp * (grad_full[support] - np.sum(a_supp * grad_full[support]))

            return obj_val, grad_b

        try:
            res = scipy_minimize(obj_grad, current_b, method='L-BFGS-B',
                                 jac=True, options={'maxiter': 300, 'gtol': 1e-12,
                                                    'ftol': 1e-16, 'maxfun': 1000})
            if res.x is not None:
                b_shifted = res.x - res.x.max()
                a_supp = np.exp(b_shifted)
                a_supp = a_supp / a_supp.sum()
                a_new = np.zeros(full_len)
                a_new[support] = a_supp
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    current_b = res.x.copy()
                    if verbose:
                        print(f"    LogOpt stage{stage_idx}: tau={tau:.0f}, obj={best_obj:.8f}")
        except Exception:
            pass

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.5
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
        a = a / a.sum()
    elif mode == 5:
        a = np.exp(-np.arange(n) / (n * 0.2)) + 0.3 * np.exp(-np.arange(n) / (n * 0.8))
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x ** 3 * (1 - x)
        a = a / a.sum()
    elif mode == 7:
        # Cosine bump
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = a / a.sum()
    elif mode == 8:
        # Symmetric bump
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses corrected cross-correlation gradient, Polyak-averaged subgradient descent,
LP-based minimax polishing, log-parameter L-BFGS, support reduction,
wide multi-n exploration, and random perturbations to escape local minima.
Key improvements: correct FFT gradient (conj(P)*Y), Polyak averaging,
exact subgradient at high tau, wider n sweep, perturbation-based restarts.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_gradient(p, y):
    """Compute gradient of smoothed max: grad[j] = 2 * sum_k p[k] * y[k-j].
    p has length 2n-1, y has length n. Uses FFT cross-correlation.
    """
    n = len(y)
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    P = fft(p, N)
    Y = fft(y, N)
    # Cross-correlation: ifft(conj(P) * Y) gives (p star y)
    corr = ifft(np.conj(P) * Y).real
    return 2.0 * corr[:n]


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, verbose=False):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    if verbose:
                        print(f"    LP solved with {solver}, M={M_val:.10f}")
                    return g_val, M_val
        except Exception:
            continue

    return None, None


def polyak_subgradient_gd(a, time_budget, tau_start=50, tau_end=30000, verbose=True):
    """
    Subgradient descent with Polyak averaging and Nesterov momentum.
    Uses exact subgradient when tau*gap > threshold, softmax otherwise.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0

    # Polyak averaging
    avg_a = a.copy()
    avg_weight = 1.0

    # Nesterov momentum
    beta = 0.9
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Adaptive tau: stays moderate longer, then increases
        tau = tau_start + (tau_end - tau_start) * progress ** 0.6

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute convolution
        conv = compute_conv(y)
        max_conv = float(np.max(conv))

        # Determine if we should use exact subgradient
        sorted_vals = np.sort(conv)[::-1]
        if len(sorted_vals) >= 2:
            gap = sorted_vals[0] - sorted_vals[1]
        else:
            gap = np.inf

        if tau * gap > 30:
            # Essentially one-hot: use exact subgradient at argmax positions
            threshold = max_conv - 1e-14
            argmax_positions = np.where(conv >= threshold)[0]
            p = np.zeros(len(conv))
            p[argmax_positions] = 1.0 / len(argmax_positions)
        else:
            # Softmax smoothing
            exp_vals = np.exp(tau * (conv - max_conv))
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp

        # Gradient via FFT cross-correlation
        grad = compute_gradient(p, y)

        # Step size
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break
        step = min(1.0 / (g_norm * g_norm + 1e-10), 1.0 / (it + 1))
        step = max(step, 1e-10)

        # Update with momentum
        v = beta * v - step * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        # Update Polyak average
        avg_a = (avg_weight * avg_a + a_new) / (avg_weight + 1.0)
        avg_weight += 1.0

        # Check both current and average
        obj_new = compute_obj(a_new)
        obj_avg = compute_obj(avg_a)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        if obj_avg < best_obj - 1e-14:
            best_obj = obj_avg
            best_a = avg_a.copy()
            best_it = it

        a = a_new
        it += 1
        beta = min(0.99, 0.9 + 0.09 * progress)

        # Reset if stalled
        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            avg_a = best_a.copy()
            avg_weight = 1.0
            best_it = it
            if it > 80000:
                break

        if verbose and it % 10000 == 0:
            print(f"    PSGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.0f}s")

    return best_a


def lp_polish_linesearch(a, time_budget, verbose=True):
    """LP polishing with line-search mixing for optimal step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.3

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp(a, verbose=False)
        if g_lp is None:
            break

        # Line search over mixing parameter t
        t_candidates = np.concatenate([
            np.linspace(0.05, 1.0, 12),
            [max(0.02, last_t * 0.6), max(0.02, last_t * 0.8), last_t,
             min(1.0, last_t * 1.3), min(1.0, last_t * 1.6)]
        ])
        t_candidates = np.unique(np.clip(t_candidates, 0.01, 1.0))

        best_t_obj = best_obj
        best_t = last_t
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-12:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t
            stall = 0
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.4f}, elapsed={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 2:
                break

    return best_a


def support_shrink_refine(a, time_budget, verbose=True):
    """Try zeroing out small entries and re-optimizing on reduced support."""
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    sorted_vals = np.sort(a)[::-1]
    for frac in [0.2, 0.3, 0.5, 0.7, 0.85, 0.95]:
        if time.monotonic() - start > time_budget:
            break
        k = max(2, int(len(a) * frac))
        if k >= len(a):
            continue
        threshold = sorted_vals[k - 1]
        support = np.where(a >= threshold * 0.3)[0]
        if len(support) < 2:
            continue
        a_shrunk = np.zeros(len(a))
        a_shrunk[support] = a[support]
        a_shrunk = a_shrunk / a_shrunk.sum()

        obj_shrunk = compute_obj(a_shrunk)
        if obj_shrunk < best_obj - 1e-12:
            best_obj = obj_shrunk
            best_a = a_shrunk.copy()
            alloc = min(15, time_budget - 5 - (time.monotonic() - start))
            if alloc > 5:
                a_shrunk = polyak_subgradient_gd(a_shrunk, time_budget=alloc,
                                                  tau_start=200, tau_end=5000, verbose=False)
                obj_shrunk = compute_obj(a_shrunk)
                if obj_shrunk < best_obj:
                    best_obj = obj_shrunk
                    best_a = a_shrunk.copy()
            if verbose:
                print(f"    Shrink frac={frac}: obj={best_obj:.8f}, support={len(support)}")

    return best_a


def log_param_optimize(a, time_budget, tau_min=100, tau_max=10000, verbose=True):
    """Optimize using log-parameterization with L-BFGS on smoothed objective."""
    full_len = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    threshold = 1e-8
    support = np.where(a > threshold)[0]
    if len(support) < 2:
        support = np.where(a > a.max() * 1e-6)[0]
    if len(support) < 2:
        return a

    b0 = np.log(np.maximum(a[support], 1e-30))
    b0 = b0 - b0.mean()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    n_stages = 4
    for stage_idx in range(n_stages):
        if time.monotonic() - start > time_budget:
            break
        tau = tau_min * (tau_max / tau_min) ** (stage_idx / max(1, n_stages - 1))

        def obj_grad(b_vec):
            b_shifted = b_vec - b_vec.max()
            a_supp = np.exp(b_shifted)
            a_supp = a_supp / a_supp.sum()
            a_full = np.zeros(full_len)
            a_full[support] = a_supp

            conv = compute_conv(a_full)
            max_conv = float(np.max(conv))
            exp_vals = np.exp(tau * (conv - max_conv))
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp

            obj_val = max_conv + np.log(sum_exp) / tau
            grad = compute_gradient(p, a_full)

            # Chain rule for softmax parameterization
            grad_b = a_supp * (grad[support] - np.sum(a_supp * grad[support]))

            return obj_val, grad_b

        try:
            res = scipy_minimize(obj_grad, b0, method='L-BFGS-B',
                                 jac=True, options={'maxiter': 400, 'gtol': 1e-12,
                                                    'ftol': 1e-16, 'maxfun': 2000})
            if res.x is not None:
                b0 = res.x.copy()
                b_shifted = b0 - b0.max()
                a_supp = np.exp(b_shifted)
                a_supp = a_supp / a_supp.sum()
                a_new = np.zeros(full_len)
                a_new[support] = a_supp
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    if verbose:
                        print(f"    LogOpt stage{stage_idx}: tau={tau:.0f}, obj={best_obj:.8f}")
        except Exception:
            pass

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.5
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
        a = a / a.sum()
    elif mode == 5:
        a = np.exp(-np.arange(n) / (n * 0.2)) + 0.3 * np.exp(-np.arange(n) / (n * 0.8))
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x ** 3 * (1 - x)
        a = a / a.sum()
    elif mode == 7:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = a / a.sum()
    elif mode == 8:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
        a = a / a.sum()
    elif mode == 9:
        # Wide Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 0.5)
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a


def perturbation_restart(a, rng, scale=0.05):
    """Add random perturbation and renormalize."""
    n = len(a)
    noise = rng.normal(0, scale, n)
    a_pert = a + noise
    a_pert = np.maximum(a_pert, 0)
    s = float(np.sum(a_pert))
    if s > 1e-10:
        a_pert = a_pert / s
    else:
        a_pert = a.copy()
    return a_pert
