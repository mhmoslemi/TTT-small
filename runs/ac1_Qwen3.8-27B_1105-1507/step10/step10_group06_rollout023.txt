import numpy as np
import cvxpy as cp

from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a):
    """Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(max(0, j - n + 1), min(2 * n - 1, j + n)):
            # A[k, j] = a[j] if 0 <= k-j < n
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses accelerated LP alternating minimization with adaptive line search,
multi-length exploration, and FFT-based gradient polishing.
Key improvements over previous: faster sparse LP, proper line search,
active-set tracking, and multi-restart with cooling.

import numpy as np
import cvxpy as cp
import time
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a, row_indices, col_indices):
    """Build sparse matrix with precomputed structure."""
    n = len(a)
    # For each (row, col) pair, the value is a[col]
    values = a[col_indices]
    return csr_matrix((values, (row_indices, col_indices)), shape=(2 * n - 1, n))


def precompute_conv_pattern(n):
    """Precompute the sparsity pattern of the convolution matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_conv_lp(a, row_idx, col_idx, solver_preference=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_sparse(a_norm, row_idx, col_idx)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.SCS]
    if solver_preference:
        solvers = [solver_preference] + [s for s in solvers if s != solver_preference]

    for solver in solvers:
        try:
            prob.solve(solver=solver)

                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:

    return None, None


def fourier_perturb(a, rng, strength=0.03, n_modes=8):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, n_modes + 1):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / (k + 1)
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, rng, strength=0.05):
    """Random multiplicative perturbation."""
    n = len(a)
    log_pert = strength * rng.standard_normal(n)
    a_new = a * np.exp(log_pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def line_search_blend(a, g, current_obj, max_steps=10):
    """
    Line search: find best t in (0,1] such that f((1-t)*a + t*g) is minimized.
    Uses golden section search on [0, 1].
    """
    best_t = 0.0
    best_obj = current_obj

    # Try a few standard points first (fast path)
    candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    for t in candidates:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    # If we found improvement at t=1, try to refine with smaller t values
    if best_t > 0:
        # Try smaller t values for more precise search
        refine = [best_t * 0.5, best_t * 0.8, best_t * 1.2 if best_t * 1.2 <= 1 else 1.0,
                  max(0.01, best_t - 0.05), min(1.0, best_t + 0.05)]
        for t in refine:
            if t <= 0 or t > 1:
                continue
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t

    return best_t, best_obj, best_t > 0


def lp_alternate_fast(a, time_budget, row_idx, col_idx, seed=42, verbose=False):
    """
    Fast LP-based alternating minimization with line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    stall_count = 0
    no_improve_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_conv_lp(a, row_idx, col_idx)
        lp_time = time.monotonic() - t0

        if g_lp is None:
            if verbose:
                print(f"    LP failed at iter {iter_count}")
            break

        iter_count += 1

        # Line search for best blend
        t_best, obj_best, improved = line_search_blend(a, g_lp, best_obj)

        if improved and obj_best < best_obj - 1e-14:
            a_try = (1.0 - t_best) * a + t_best * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            best_obj = obj_best
            best_a = a_try.copy()
            a = a_try.copy()
            stall_count = 0
            no_improve_count = 0
            if verbose and iter_count % 50 == 0:
                print(f"    LP alt {iter_count}: obj={best_obj:.10f}, t={t_best:.4f}, "
                      f"M_lp={M_lp:.6f}, lp_time={lp_time:.3f}s")
        else:
            stall_count += 1
            no_improve_count += 1

            # Perturbation strategy based on stall count
            if stall_count == 2:
                a = fourier_perturb(best_a, rng, strength=0.01)
            elif stall_count >= 4:
                strength = 0.02 + 0.01 * min(no_improve_count, 20)
                a = random_perturb(best_a, rng, strength=strength)
                stall_count = 0
                # Reset a to the perturbed best
                a = np.maximum(a, 0)
                a = a / a.sum()

        # If we've been stuck for a very long time, do a bigger restart
        if no_improve_count > 100:
            a = fourier_perturb(best_a, rng, strength=0.05, n_modes=12)
            no_improve_count = 0
            stall_count = 0

    if verbose:
        print(f"    LP alternating: {iter_count} iterations, best_obj={best_obj:.10f}, "
              f"time={time.monotonic()-start:.1f}s")

def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of (a*a)_k using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def gradient_polish(a, time_budget, tau_init=30.0, tau_max=2000.0):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 1.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adaptive_lr = 0.5 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.1 * base_lr), 3.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adaptive_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr *= 0.9999
        tau = min(tau * 1.01, tau_max)
        it += 1

    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def precompute_conv_pattern(n):
    """Precompute sparsity pattern of convolution matrix (2n-1 x n)."""
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    idx = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        count = j_end - j_start + 1
        rows[idx:idx + count] = k
        cols[idx:idx + count] = np.arange(j_start, j_end + 1, dtype=np.int32)
        idx += count
    return rows, cols


def build_conv_matrix(a, row_idx, col_idx):
    """Build sparse convolution matrix with precomputed structure."""
    n = len(a)
    values = a[col_idx]
    return csr_matrix((values, (row_idx, col_idx)), shape=(2 * n - 1, n))


def solve_conv_lp(a, row_idx, col_idx):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm, row_idx, col_idx)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g, best_obj):
    """Try blending at a few t values, return best (t, obj)."""
    best_t = 0.0
    best_o = best_obj
    for t in [0.5, 1.0, 0.3, 0.7]:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_o:
            best_o = obj
            best_t = t
    return best_t, best_o


def fourier_perturb(a, rng, strength=0.03, n_modes=8):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, n_modes + 1):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / (k + 1)
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, rng, strength=0.05):
    """Random multiplicative perturbation."""
    n = len(a)
    log_pert = strength * rng.standard_normal(n)
    a_new = a * np.exp(log_pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_alternate(a, time_budget, row_idx, col_idx, seed=42, verbose=False):
    """
    LP-based alternating minimization with line search.
    Fast iteration with minimal overhead.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_conv_lp(a, row_idx, col_idx)
        lp_time = time.monotonic() - t0

        if g_lp is None:
            if verbose:
                print(f"    LP failed at iter {iter_count}")
            break

        iter_count += 1

        t_best, obj_best = line_search_blend(a, g_lp, best_obj)

        if t_best > 0 and obj_best < best_obj - 1e-14:
            a_try = (1.0 - t_best) * a + t_best * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            best_obj = obj_best
            best_a = a_try.copy()
            a = a_try.copy()
            stall_count = 0
            if verbose and iter_count % 50 == 0:
                elapsed = time.monotonic() - start
                print(f"    iter {iter_count}: obj={best_obj:.10f}, t={t_best:.2f}, "
                      f"M={M_lp:.6f}, lp={lp_time:.3f}s, elapsed={elapsed:.0f}s")
        else:
            stall_count += 1
            if stall_count == 2:
                a = fourier_perturb(best_a, rng, strength=0.01, n_modes=6)
            elif stall_count >= 4:
                strength = 0.02 + 0.005 * min(stall_count, 15)
                a = random_perturb(best_a, rng, strength=strength)
                stall_count = 0
            elif stall_count >= 10:
                a = fourier_perturb(best_a, rng, strength=0.04, n_modes=12)
                stall_count = 0

    if verbose:
        print(f"    Done: {iter_count} iters, obj={best_obj:.10f}, "
              f"time={time.monotonic()-start:.1f}s")
    return best_a


def smoothed_gradient(a, tau):
    """Gradient of log-sum-exp smoothed max of (a*a)_k."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def gradient_polish(a, time_budget, tau_init=30.0, tau_max=2000.0):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 1.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adaptive_lr = 0.5 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.1 * base_lr), 3.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adaptive_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr *= 0.9999
        tau = min(tau * 1.01, tau_max)

    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_cosine_window(n):
    """Create a cosine window sequence."""
    x = np.arange(n)
    w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    return w / w.sum()
