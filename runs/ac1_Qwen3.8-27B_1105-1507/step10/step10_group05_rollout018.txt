Uses fast Nesterov-accelerated smoothed gradient descent with adaptive tau,
active-set focusing, spectral initialization, and multi-length exploration.
Key innovations: pure gradient-based approach (no LP), FFT-based gradient,
backtracking line search, diverse restarts, and top-K peak focusing.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    result = ifft(fa * fa, size).real[:2 * n - 1]
    return result


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_gradient(a, conv, tau, top_k=None):
    """
    Compute gradient of log-sum-exp smoothed max of conv w.r.t. a.
    If top_k is specified, only use top-k peaks (harder focusing).
    Returns (gradient, smoothed_value).
    """
    n = len(a)
    m = len(conv)
    if top_k is not None and top_k < m:
        # Use only top-k values
        threshold_idx = np.argpartition(conv, m - top_k)[m - top_k:]
        p = np.zeros(m)
        vals = conv[threshold_idx]
        max_vals = float(np.max(vals))
        exp_vals = np.exp(tau * (vals - max_vals))
        sum_exp = float(np.sum(exp_vals))
        p[threshold_idx] = exp_vals / sum_exp
        smoothed_val = max_vals + np.log(sum_exp) / tau
    else:
        max_conv = float(np.max(conv))
        shifted = conv - max_conv
        exp_vals = np.exp(tau * shifted)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(m)
            p[np.argmax(conv)] = 1.0
            smoothed_val = max_conv
        else:
            p = exp_vals / sum_exp
            smoothed_val = max_conv + np.log(sum_exp) / tau

    # gradient[i] = 2 * sum_k p[k] * a[k-i] (where 0 <= k-i < n)
    # = 2 * cross_correlation(p, a) [valid part]
    size = 1
    while size < m + n - 1:
        size <<= 1
    P = fft(p, size)
    Ar = fft(a[::-1], size)
    conv_pa = ifft(P * Ar).real
    grad = 2.0 * conv_pa[n - 1:2 * n - 1]

    return grad, smoothed_val


def hard_max_gradient(a, conv, top_k=20):
    """
    Compute gradient using uniform weights on top-k peaks.
    This is the tau -> infinity limit of the smoothed gradient.
    """
    n = len(a)
    m = len(conv)
    k = min(top_k, m)
    top_idx = np.argpartition(conv, m - k)[m - k:]
    p = np.zeros(m)
    p[top_idx] = 1.0 / k

    conv_pa = ifft(P * Ar).real

def nesterov_smoothed_descent(a, time_budget, tau_init=50.0, tau_end=5000.0,
                               beta=0.9, lr_scale=2.0, top_k_announce=50,
                               verbose=False, seed=0):
    """
    Nesterov accelerated smoothed gradient descent on the simplex.
    Uses adaptive tau (annealing) and backtracking line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = lr_scale / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Cosine annealing of tau
        cos_val = 0.5 * (1 + np.cos(np.pi * progress))
        tau = tau_init + (tau_end - tau_init) * (1 - cos_val)

        # Adaptive top_k: start with all, narrow down
        m = 2 * n - 1
        if top_k_announce > 0:
            top_k = max(1, int(m * (1.0 - progress * 0.7)))
        else:
            top_k = None

        # Nesterov lookahead
        y = a + beta * v
        y = project_simplex(y)

        # Compute gradient
        conv_y = compute_conv(y)
        if tau > 200:
            grad_y, _ = hard_max_gradient(y, conv_y, top_k=min(top_k or 100, 50))
        elif top_k is not None and top_k < m:
            grad_y, _ = smooth_max_gradient(y, conv_y, tau, top_k=top_k)
        else:
            grad_y, _ = smooth_max_gradient(y, conv_y, tau)

        # Gradient normalization
        grad_norm = float(np.linalg.norm(grad_y))
        if grad_norm > 1e-15:
            grad_y = grad_y / grad_norm

        # Update momentum
        v = beta * v - lr * grad_y

        # Take step
        a_new = y + v
        a_new = project_simplex(a_new)

        # Evaluate
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            patience = 0
        else:
            patience += 1
            # Backtrack: reduce step
            if patience > 5:
                lr *= 0.8
                a_new = a
                patience = 0

        a = a_new
        it += 1

        # Decay learning rate slowly
        lr = max(1e-6, lr * 0.9999)

        # Periodic momentum reset
        if it % 5000 == 0:
            v = 0.5 * v

        # Restart if stuck for too long
        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = lr_scale / np.sqrt(n)
            best_it = it
            patience = 0

        if verbose and it % 5000 == 0:
            print(f"      GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def spectral_init(n, rng, mode='random'):
    """Generate diverse initializations based on spectral properties."""
    if mode == 'uniform':
        return np.ones(n) / n
    elif mode == 'ramp':
        a = np.linspace(0.1, 2.0, n)
        return a / a.sum()
    elif mode == 'cosine':
        k = rng.integers(1, max(2, n // 20))
        x = np.arange(n)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * k * x / n + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 'multi_cosine':
        a = np.ones(n)
        x = np.arange(n)
        for _ in range(rng.integers(2, 5)):
            k = rng.integers(1, max(2, n // 10))
            amp = rng.uniform(0.1, 0.5)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * k * x / n + phase)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 'exp_decay':
        gamma = rng.uniform(0.5, 3.0)
        a = np.exp(-gamma * np.arange(n) / n)
        return a / a.sum()
    elif mode == 'bimodal':
        center = rng.uniform(0.2, 0.8) * n
        sigma = rng.uniform(n * 0.05, n * 0.2)
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2) + 0.1 * np.exp(-0.5 * ((x - (n - center)) / sigma) ** 2)
        return a / a.sum()
    elif mode == 'dirichlet':
        alpha = rng.uniform(0.5, 3.0)
        a = rng.dirichlet(np.full(n, alpha))
        return a
    else:
        return np.ones(n) / n


def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(8, n // 20)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, rng, strength=0.05):
    """Add random multiplicative perturbation."""
    n = len(a)
    pert = 1.0 + rng.normal(0, strength, n)
    a_new = a * pert
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def active_set_refine(a, time_budget, top_k=10, verbose=False):
    """
    Refine by focusing gradient only on the top-k peaks of the convolution.
    This is the tau -> infinity limit, giving exact gradient of the max.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if time_budget - elapsed < 2:
            break

        # Nesterov
        y = a + beta * v
        y = project_simplex(y)

        conv_y = compute_conv(y)
        grad_y, _ = hard_max_gradient(y, conv_y, top_k=top_k)

        grad_norm = float(np.linalg.norm(grad_y))
        if grad_norm < 1e-15:
            break
        grad_y = grad_y / grad_norm

        v = beta * v - lr * grad_y
        a_new = y + v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        else:
            if it - best_it > 5000:
                a = best_a.copy()
                v = np.zeros(n)
                lr *= 0.7
                best_it = it

        a = a_new
        it += 1
        lr = max(1e-7, lr * 0.99995)

        if verbose and it % 10000 == 0:
            print(f"      AS it{it}: obj={best_obj:.8f}, lr={lr:.7f}, t={elapsed:.0f}s")

    return best_a


def multistart_refine(a, time_budget, n_starts=3, verbose=False, seed=0):
    """Try multiple perturbation directions and keep the best."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for i in range(n_starts):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 30:
            break
        alloc = remaining / (n_starts - i)

        # Different perturbation types
        if i == 0:
            a_try = fourier_perturb(best_a, rng, strength=0.02)
        elif i == 1:
            a_try = random_perturb(best_a, rng, strength=0.04)
        else:
            a_try = 0.9 * best_a + 0.1 * np.ones(n) / n
            a_try = a_try / a_try.sum()

        a_try = nesterov_smoothed_descent(a_try, time_budget=alloc * 0.6,
                                           tau_init=100.0, tau_end=3000.0,
                                           beta=0.9, lr_scale=1.5,
                                           top_k_announce=30)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            if verbose:
                print(f"    MS {i}: obj={best_obj:.8f} **")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines diverse initializations, Nesterov smoothed
    gradient descent with adaptive tau, active-set refinement, and multi-length
    exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        best_seq = np.ones(n_current) / n_current
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # === Phase 1: Diverse initialization exploration ===
    print(f"\n=== Phase 1: Diverse Initialization (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.12, 180.0)

    modes = ['cosine', 'multi_cosine', 'exp_decay', 'bimodal', 'ramp', 'dirichlet']
    per_mode = p1_budget / len(modes)

    for mode in modes:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_mode, remaining * 0.15)
        if alloc < 10:
            continue

        a_init = spectral_init(n_current, rng, mode=mode)
        a_try = nesterov_smoothed_descent(a_init, time_budget=alloc,
                                           tau_init=30.0, tau_end=2000.0,
                                           beta=0.9, lr_scale=2.0,
                                           top_k_announce=40)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  {mode}: obj={best_score:.8f} **")
        else:
            print(f"  {mode}: obj={obj_try:.8f}")

    # Also try from parent with perturbation
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 60.0)
        a_pert = fourier_perturb(best_seq, rng, strength=0.02)
        a_try = nesterov_smoothed_descent(a_pert, time_budget=alloc,
                                           tau_init=50.0, tau_end=3000.0,
                                           beta=0.9, lr_scale=2.0,
                                           top_k_announce=30)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  perturbed parent: obj={best_score:.8f} **")

    # === Phase 2: Intensive optimization from best ===
    print(f"\n=== Phase 2: Intensive Nesterov Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.3, 400.0)
    print(f"  Budget: {p2_budget:.0f}s, current best: {best_score:.8f}")

    a_main = nesterov_smoothed_descent(best_seq, time_budget=p2_budget,
                                        tau_init=20.0, tau_end=8000.0,
                                        beta=0.9, lr_scale=2.5,
                                        top_k_announce=80,
                                        verbose=True)
    obj_main = compute_obj(a_main)
    if obj_main < best_score:
        best_score = obj_main
        best_seq = a_main.copy()
        print(f"  Phase 2: obj={best_score:.8f} **")
    else:
        print(f"  Phase 2: obj={obj_main:.8f} (no improvement)")

    # === Phase 3: Active-set refinement ===
    print(f"\n=== Phase 3: Active Set Refinement ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 250.0)
    print(f"  Budget: {p3_budget:.0f}s")

    a_as = active_set_refine(best_seq, time_budget=p3_budget, top_k=10, verbose=True)
    obj_as = compute_obj(a_as)
    if obj_as < best_score:
        best_score = obj_as
        best_seq = a_as.copy()
        print(f"  Phase 3: obj={best_score:.8f} **")
    else:
        print(f"  Phase 3: obj={obj_as:.8f} (no improvement)")

    # === Phase 4: Multi-start refinement ===
    print(f"\n=== Phase 4: Multi-Start Refinement ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 180.0)
    print(f"  Budget: {p4_budget:.0f}s")

    if p4_budget > 30:
        a_ms = multistart_refine(best_seq, time_budget=p4_budget,
                                  n_starts=3, verbose=True, seed=seed + 100)
        obj_ms = compute_obj(a_ms)
        if obj_ms < best_score:
            best_score = obj_ms
            best_seq = a_ms.copy()
            print(f"  Phase 4: obj={best_score:.8f} **")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    print(f"  Current best n={n_best}, score={best_score:.8f}")

    n_candidates = []
    for dn in [-100, -50, 50, 100, 150, 200, 300]:
        n_try = n_best + dn
        if 500 <= n_try <= 1500 and n_try != n_best:
            n_candidates.append(n_try)
    for n_abs in [600, 750, 900, 1000, 1200]:
        if n_abs not in n_candidates and abs(n_abs - n_best) > 30:
            n_candidates.append(n_abs)

    n_candidates = sorted(set(n_candidates))[:4]
    print(f"  Trying n values: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(remaining * 0.12, 80.0)
        if alloc < 20:
            continue

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()
        a_try = nesterov_smoothed_descent(a_try, time_budget=alloc,
                                           tau_init=30.0, tau_end=3000.0,
                                           beta=0.9, lr_scale=2.0,
                                           top_k_announce=40)
        # Quick active set polish
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_try = active_set_refine(a_try, time_budget=min(remaining * 0.1, 30.0), top_k=8)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s, current best: {best_score:.8f}")

    # Long Nesterov with high tau
    if remaining > 60:
        alloc = min(remaining * 0.4, 200.0)
        a_fin = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                           tau_init=200.0, tau_end=20000.0,
                                           beta=0.9, lr_scale=1.5,
                                           top_k_announce=20,
                                           verbose=True)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Active set with very small top_k
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 100.0)
        a_fin = active_set_refine(best_seq, time_budget=alloc, top_k=5, verbose=True)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final AS: obj={best_score:.8f} **")

    # One more Nesterov if time
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 100.0)
        a_fin = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                           tau_init=500.0, tau_end=50000.0,
                                           beta=0.95, lr_scale=1.0,
                                           top_k_announce=10,
                                           verbose=False)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Ultra-final GD: obj={best_score:.8f} **")

    # Final active set
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_fin = active_set_refine(best_seq, time_budget=min(remaining * 0.5, 50.0), top_k=3)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Ultra-final AS: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
