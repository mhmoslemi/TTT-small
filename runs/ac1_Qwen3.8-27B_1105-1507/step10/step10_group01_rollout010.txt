"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of fast exact coordinate descent, LP-based refinement with CLARABEL,
and high-tau smoothed gradient descent. Explores multiple sequence lengths and
initializations to escape local minima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize_scalar

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def coordinate_descent(a, max_sweeps=500, time_budget=300.0, verbose=True, seed=42):
    """
    Fast exact coordinate descent. For each coordinate i, solves the 1D subproblem:
    min over delta: 2n * max_k (C[k] + 2*delta*a[k-i] + delta^2*delta_{k,2i}) / (S+delta)^2
    subject to a[i]+delta >= 0.
    Uses incremental autocorrelation updates for O(n) per evaluation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    s = 1.0  # normalized

    C = compute_conv(a)  # current autocorrelation, length 2n-1
    max_C = float(np.max(C))
    best_obj = 2.0 * n * max_C  # since s=1
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        # Shuffle order each sweep
        rng.shuffle(order)

        improved = 0
        for i in order:
            ai = a[i]
            if ai < 1e-15:
                # Try small positive values
                delta_candidates = np.array([0.001 / n, 0.01 / n, 0.05 / n, 0.1 / n])
            else:
                # Grid of delta values centered around 0
                scale = max(ai, 1.0 / n)
                delta_candidates = np.linspace(-0.95 * ai, 5.0 * scale, 25)

            # Vectorized evaluation
            # For coordinate i, the correction to autocorrelation is:
            # C_new[k] = C[k] + 2*delta*a[k-i] + delta^2 * (1 if k==2i else 0)
            # a[k-i] is valid when 0 <= k-i <= n-1, i.e., i <= k <= i+n-1

            # Build the "d" vector: d[k] = a[k-i] if valid, else 0
            d = np.zeros(2 * n - 1)
            valid_start = i
            valid_end = min(i + n, 2 * n - 1)
            if valid_start < valid_end:
                d[valid_start:valid_end] = a[0:valid_end - valid_start]

            # e[k] = 1 if k == 2i, else 0
            e = np.zeros(2 * n - 1)
            if 2 * i < 2 * n - 1:
                e[2 * i] = 1.0

            # Evaluate f(delta) for all candidates
            # f(delta) = 2n * max(C + 2*delta*d + delta^2*e) / (S+delta)^2
            # S = 1.0 (normalized)
            deltas = delta_candidates[:, None]  # (m, 1)
            correction = 2.0 * deltas * d[None, :] + (deltas ** 2) * e[None, :]  # (m, 2n-1)
            new_C = C[None, :] + correction  # (m, 2n-1)
            new_max = np.max(new_C, axis=1)  # (m,)
            new_s = 1.0 + delta_candidates  # (m,)
            valid = new_s > 0.01
            if not np.any(valid):
                continue
            objs = np.full(len(delta_candidates), np.inf)
            objs[valid] = 2.0 * n * new_max[valid] / (new_s[valid] ** 2)

            best_idx = int(np.argmin(objs))
            best_delta = delta_candidates[best_idx]
            best_new_obj = objs[best_idx]

            if best_new_obj < best_obj - 1e-10:
                # Apply update
                a[i] += best_delta
                s += best_delta
                # Update C incrementally
                C += 2.0 * best_delta * d + (best_delta ** 2) * e
                C = np.maximum(C, 0)  # numerical safety
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                a = a / s
                # Recompute C exactly every so often to avoid drift
                best_obj = 2.0 * n * float(np.max(C))
                improved += 1

        # Periodic exact recomputation
        if (sweep + 1) % 20 == 0:
            C = compute_conv(a)
            max_C = float(np.max(C))
            obj_exact = 2.0 * n * max_C
            if obj_exact < best_obj:
                best_obj = obj_exact
                best_a = a.copy()
            if obj_exact > best_obj + 1e-6:
                # Drift detected, re-sync
                a = best_a.copy()
                C = compute_conv(a)
                best_obj = 2.0 * n * float(np.max(C))

        if verbose and (sweep + 1) % 50 == 0:
            elapsed = time.monotonic() - start
            C = compute_conv(a)
            obj_now = 2.0 * n * float(np.max(C))
            print(f"    CD sweep {sweep+1}: obj={obj_now:.8f}, improved={improved}, time={elapsed:.1f}s")

    # Final exact evaluation
    a = best_a.copy()
    obj_final = compute_obj(a)
    return a / a.sum()


def build_conv_matrix(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    if not HAS_CVXPY:
        return None, None

    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, direction, n_evals=30):
    """Line search: find best t in [0,1] to minimize obj((1-t)*a + t*direction)."""
    ts = np.linspace(0.01, 0.99, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try / s)

    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]

    # Golden section refinement
    if best_obj < np.inf:
        a_lo, a_hi = ts[max(0, best_idx - 2)], ts[min(n_evals - 1, best_idx + 2)]
        result = minimize_scalar(
            lambda t: compute_obj(np.maximum((1 - t) * a + t * direction, 0) /
                                  np.maximum(np.sum(np.maximum((1 - t) * a + t * direction, 0)), 1e-10)),
            bounds=(max(0.001, a_lo), min(0.999, a_hi)),
            method='bounded', options={'xatol': 1e-5}
        )
        if result.fun < best_obj:
            best_t = result.x
            best_obj = result.fun

    return best_t, best_obj

def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def gd_polish(a, time_budget=100.0, tau_init=100.0, tau_end=2000.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.2 * base_lr), 3.0 * base_lr)
            lr = 0.75 * lr + 0.25 * adapt

        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 5.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it

    if verbose:
        print(f"    GD: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def block_coordinate_descent(a, block_size=10, max_sweeps=200, time_budget=300.0, verbose=False):
    """
    Block coordinate descent: update blocks of consecutive coordinates simultaneously.
    For a block, the 1D->multiD subproblem is solved via a small grid or SLSQP.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(77)

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        n_blocks = n // block_size
        order = rng.permutation(n_blocks)

        for b in order:
            idx = slice(b * block_size, (b + 1) * block_size)
            a_block = a[idx].copy()
            scale = max(float(np.mean(a_block)), 1.0 / n)

            # Try perturbations of the block
            best_block_obj = best_obj
            best_block_val = a_block.copy()

            for trial in range(8):
                delta = rng.normal(0, 0.3 * scale, block_size)
                candidate = a_block + delta
                candidate = np.maximum(candidate, 0)
                if candidate.sum() < 1e-10:
                    continue
                a_try = a.copy()
                a_try[idx] = candidate
                a_try = a_try / a_try.sum()
                obj_try = compute_obj(a_try)
                if obj_try < best_block_obj:
                    best_block_obj = obj_try
                    best_block_val = candidate.copy()
                    a = a_try
                    best_a = a_try.copy()
                    best_obj = obj_try

    return best_a

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 15 and HAS_CVXPY:
        alloc = min(remaining * 0.35, 150.0)
        a_final = lp_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final high-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.5, 60.0)
        a_final = gd_polish(best_seq, time_budget=alloc, tau_init=200.0, tau_end=3000.0, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
