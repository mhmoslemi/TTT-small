    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, top_k=None):
    """Solve LP: min M s.t. 2*(A@g)_k - (a*a)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    aa = compute_conv(a_norm)
    
    if top_k is not None and top_k < 2 * n - 1:
        idx = np.argsort(aa)[::-1][:top_k]
        idx_sorted = np.sort(idx)
        A = build_conv_matrix(a_norm)
        A_sub = A[idx_sorted, :]
        aa_sub = aa[idx_sorted]
    else:
        A = build_conv_matrix(a_norm)
        A_sub = A
        aa_sub = aa
    
    n_cons = A_sub.shape[0]
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [2 * (A_sub @ g) - aa_sub <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    g_val = None
    M_val = None
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                break
        except Exception:
            continue
    
    return g_val, M_val


def lp_polish(a, time_budget, t_init=0.3, top_k=None, verbose=False):
    """LP polishing with improved linearization and adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    t = t_init
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        
        g_lp, M_lp = solve_lp(a, top_k=top_k)
        if g_lp is None:
            break
        
        # Adaptive step size: try a few values
        best_t_obj = best_obj
        best_t = t
        best_a_try = a.copy()
        for t_try in [t * 0.5, t, t * 1.5, 0.5]:
            t_try = float(np.clip(t_try, 0.05, 1.0))
            a_try = (1.0 - t_try) * a + t_try * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t_try
                best_a_try = a_try.copy()
        
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_try
            a = best_a_try
            t = best_t
            stall = 0
        else:
            stall += 1
            t = max(0.05, t * 0.7)
            if stall > 5:
                break
        
        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t:.3f}, top_k={top_k}, elapsed={time.monotonic()-start:.0f}s")
    
    return best_a
