import numpy as np

from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])

    while size < 2 * n - 1:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    # Standard algorithm: find theta such that sum(max(v-theta, 0)) = target_sum
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)

    rho = rho[-1]

    # Nesterov parameters
    beta = 0.8
    lr = 0.01 / math.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    patience = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        
        # Adaptively increase top_k as we converge
        current_top_k = max(3, top_k + int(10 * progress))
        
        # Nesterov lookahead
        y = a + beta * v
        y = project_simplex(y, 1.0)
        
        # Compute convolution and gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        obj_current = 2.0 * n * max_conv
        
        grad = subgradient_at_peaks(y, conv, current_top_k)
        
        # Update momentum
        v = beta * v - lr * grad
        a_new = y + v
        a_new = project_simplex(a_new, 1.0)
        
        # Compute objective
        obj_new = compute_obj_normalized(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            patience = 0
        else:
            patience += 1
        
        a = a_new
        it += 1
        
        # Adaptive learning rate
        if patience > 500:
            lr *= 0.9
            patience = 0
            v *= 0.5
        
        # Reduce lr over time
        lr *= 0.99995
        
        if verbose and it % 5000 == 0:
            print(f"    FSD it{it}: obj={best_obj:.8f}, lr={lr:.8f}, top_k={current_top_k}, t={elapsed:.0f}s")
        
        if patience > 5000:
            # Reset to best
            a = best_a.copy()
            v = np.zeros(n)
            lr = 0.01 / math.sqrt(n)
            patience = 0
            # Small perturbation
            a = a * (1 + 0.01 * rng.standard_normal(n))
            a = project_simplex(a, 1.0)
    
    return best_a


def topk_smoothed_descent(a, time_budget=100.0, verbose=False, seed=42):
    """
    Subgradient descent using sum-of-top-k as smoothed objective.
    More stable than raw max because gradient doesn't jump between peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    beta = 0.85
    lr = 0.005 / math.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    patience = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        
        # Adaptive top-k: start small (focus on max), grow larger (smooth out)
        k = max(3, int(3 + 20 * progress))
        
        y = a + beta * v
        y = project_simplex(y, 1.0)
        
        conv = compute_conv(y)
        # Get top-k values and indices
        top_idx = np.argsort(conv)[::-1][:k]
        top_vals = conv[top_idx]
        max_conv = float(top_vals[0])
        obj_current = 2.0 * n * max_conv
        
        # Gradient: average of subgradients at top-k peaks
        grad = subgradient_at_peaks(y, conv, k)
        
        # Add a small "pressure" toward uniform to avoid degeneracy
        grad -= 0.001 / n
        
        v = beta * v - lr * grad
        a_new = y + v
        a_new = project_simplex(a_new, 1.0)
        
        obj_new = compute_obj_normalized(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            patience = 0
        else:
            patience += 1
        
        a = a_new
        it += 1
        
        if patience > 1000:
            lr *= 0.85
            v *= 0.3
            patience = 0
        
        lr *= 0.9999
        
        if verbose and it % 2000 == 0:
            print(f"    TSD it{it}: obj={best_obj:.8f}, lr={lr:.8f}, k={k}, t={elapsed:.0f}s")
        
        if patience > 8000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 0.005 / math.sqrt(n)
            patience = 0
            a = a * (1 + 0.02 * rng.standard_normal(n))
            a = project_simplex(a, 1.0)
    
    return best_a


def build_conv_matrix_sparse(a, m):
    """
    Build structure for LP: A[g] = conv(a, g) at specific positions m.
    Returns a function that computes conv(a, g) at positions m.
    """
    n = len(a)
    
    def compute_conv_at_m(g):
        # conv(a, g) at position m[j] = sum_i a[i] * g[m[j]-i]
        # Use FFT for efficiency
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fa = rfft(a, size)
        fg = rfft(g, size)
        conv_full = irfft(fa * fg, size)[:2*n-1]
        return conv_full[m]
    
    return compute_conv_at_m


def solve_lp_minpeak(a, timeout=5.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns optimal g or None.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix A (2n-1) x n where A @ g = conv(a_norm, g)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    start = time.monotonic()
    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        if time.monotonic() - start > timeout:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    
    return None


def lp_refinement(a, time_budget=100.0, verbose=False, seed=42):
    """LP-based refinement with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        g_lp = solve_lp_minpeak(a, timeout=min(remaining * 0.5, 3.0))
        if g_lp is None:
            break
        
        # Line search
        best_t_obj = best_obj
        best_t_val = 0.0
        best_a_local = best_a.copy()
        
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = project_simplex(a_try, 1.0)
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, t={best_t_val:.2f}")
        else:
            # No improvement, break
            break
    
    return best_a


def diverse_initialization(n, rng, method='random'):
    """Create diverse initial sequences."""
    if method == 'random':
        a = rng.exponential(1.0, n)
    elif method == 'cosine':
        x = np.linspace(0, 1, n)
        k = rng.integers(1, 5)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * k * x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
    elif method == 'gaussian':
        x = np.linspace(-3, 3, n)
        mu = rng.uniform(-1, 1)
        sigma = rng.uniform(0.5, 1.5)
        a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    elif method == 'piecewise':
        n_pieces = rng.integers(3, 8)
        heights = rng.uniform(0.5, 1.5, n_pieces)
        a = np.repeat(heights, n // n_pieces + 1)[:n]
    elif method == 'fourier':
        a = np.ones(n)
        n_modes = rng.integers(2, 6)
        for _ in range(n_modes):
            freq = rng.integers(1, n // 4)
            amp = rng.uniform(0.1, 0.4)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n) / n
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def combined_optimize(a, time_budget=100.0, verbose=False, seed=42):
    """
    Combined optimization: fast subgradient + periodic LP + top-k smoothing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    phase1_time = min(time_budget * 0.4, 40.0)
    phase2_time = min(time_budget * 0.35, 35.0)
    phase3_time = max(time_budget - phase1_time - phase2_time - 5, 10.0)
    
    # Phase 1: Fast subgradient
    if phase1_time > 5:
        a1 = fast_subgradient_descent(a, time_budget=phase1_time, top_k=5, verbose=verbose, seed=seed)
        obj1 = compute_obj_normalized(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
            if verbose:
                print(f"    Combined P1: obj={best_obj:.8f}")
    
    # Phase 2: Top-k smoothed descent
    remaining = time_budget - (time.monotonic() - start)
    p2_time = min(phase2_time, remaining * 0.5)
    if p2_time > 5:
        a2 = topk_smoothed_descent(best_a, time_budget=p2_time, verbose=verbose, seed=seed + 1)
        obj2 = compute_obj_normalized(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            if verbose:
                print(f"    Combined P2: obj={best_obj:.8f}")
    
    # Phase 3: LP refinement
    remaining = time_budget - (time.monotonic() - start)
    p3_time = min(phase3_time, remaining * 0.7)
    if p3_time > 10:
        a3 = lp_refinement(best_a, time_budget=p3_time, verbose=verbose, seed=seed + 2)
        obj3 = compute_obj_normalized(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            if verbose:
                print(f"    Combined P3: obj={best_obj:.8f}")
    
    return best_a
