At the optimum `a*`, with `sum(a*) = 1` and `a* >= 0`:
- If `a*[i] > 0`: the gradient of the objective at `a*` projected onto the tangent space of the simplex is 0 (or the subdifferential contains 0).
- If `a*[i] = 0`: the gradient component is >= 0 (can't decrease by increasing `a[i]`).

The subdifferential of `max_k c_k` at a point where the max is achieved at a set S is the convex hull of `{2*R_k(a) : k in S}`.

So the KKT condition is: there exists a convex combination `p = sum_{k in S} p_k * 2*R_k(a*)` (with `p_k >= 0, sum p_k = 1`) such that:
- `p[i] = λ` for all `i` with `a*[i] > 0` (for some λ)
- `p[i] >= λ` for all `i` with `a*[i] = 0`

This gives us a way to check how close we are to the optimum, and potentially to compute the exact optimum by solving a system of equations.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a multi-phase approach: LP best-response alternating, softmax-parameterized
gradient descent, targeted peak flattening with line search, multi-length
exploration with diverse initializations, and final combined polishing.
Key improvements over previous: softmax reparameterization, exact subgradient
line search, structured initializations (sine/cosine/bell), and adaptive
restart strategy.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = normalize(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def compute_subgradient(a, c, peak_indices):
    """
    Compute subgradient of max_k c_k at a, for the given peak indices.
    Returns the average subgradient over peak_indices.
    grad[i] = 2 * mean_{k in peak_indices} a[k-i] (with out-of-range = 0)
    """
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        for i in range(n):
            idx = k - i
            if 0 <= idx < n:
                grad[i] += 2.0 * a[idx]
    grad /= len(peak_indices)
    return grad


def compute_smoothing_weights(c, tau):
    """Compute softmax weights for the convolution values."""
    max_c = float(np.max(c))
    shifted = tau * (c - max_c)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    s = float(np.sum(exp_vals))
    if s < 1e-300:
        p = np.zeros(len(c))
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / s
    return p


def smoothed_gradient_fft(a, c, p):
    """
    Compute smoothed gradient using FFT.
    grad[i] = 2 * sum_k p[k] * a[k-i]
    This is 2 * cross_correlation(p, a).
    """
    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    # cross_correlation: sum_k p[k] * a[k-i]
    # This is ifft(conj(P) * A) appropriately indexed
    gc = ifft(np.conj(P) * A).real
    # gc[i] = sum_k p[k] * a[k-i] (circular, need to handle linear)
    # Actually: sum_k p[k] * a[k-i] for k=0..2n-2, i=0..n-1
    # gc[i] in circular correlation with period N
    grad = 2.0 * gc[:n]
    return grad


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    s = float(np.sum(w))
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def softmax_gradient_descent(a, time_budget, tau_init=50.0, tau_max=50000.0):
    """
    Gradient descent in softmax parameterization.
    a[j] = exp(f[j]) / sum(exp(f)). Optimizes over f.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Initialize f from a
    f = np.log(np.maximum(a, 1e-30))

    lr = 1.0
    tau = tau_init
    prev_obj = best_obj
    stall = 0

    while time.monotonic() - start < time_budget:
        c = compute_conv(a)
        p = compute_smoothing_weights(c, tau)
        g = smoothed_gradient_fft(a, c, p)

        # GradDesc in f-space: df = a * (g - sum(a*g))
        df = a * (g - float(np.dot(a, g)))
        gnorm = float(np.linalg.norm(df))

        if gnorm < 1e-15:
            break

        # Adaptive step size
        lr = min(lr * 1.001, 2.0 / (gnorm + 1e-10))

        f_new = f - lr * df
        # Numerical stability
        f_new = f_new - np.max(f_new)
        a_new = np.exp(f_new)
        a_new = a_new / float(np.sum(a_new))

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Reduce learning rate and increase tau
                lr *= 0.5
                tau = min(tau * 1.5, tau_max)
                stall = 0

        a = a_new
        tau = min(tau * 1.01, tau_max)

    return best_a


def targeted_peak_flatten(a, time_budget):
    """
    Targeted peak flattening: identify peaks, compute subgradient,
    take optimized step via line search.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    base_step = 0.5 / n

    while time.monotonic() - start < time_budget:
        c = compute_conv(a)
        M = float(np.max(c))
        # Find peaks within 1% of max
        tol = 0.01 * M
        peak_idx = np.where(c > M - tol)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(c)])

        # Compute subgradient (average over peaks)
        sg = compute_subgradient(a, c, peak_idx)
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-15:
            break

        # Line search: try several step sizes
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_alpha = 0.0

        alphas = [base_step * mult for mult in [0.1, 0.25, 0.5, 1.0, 2.0, 4.0, 8.0]]
        for alpha in alphas:
            a_try = a - alpha * sg
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_alpha = alpha

        if best_local_alpha > 0:
            a = best_local_a
            best_a = best_local_a
            best_obj = best_local_obj
            base_step = max(base_step * 0.99, 0.01 / n)
        else:
            # Try smoothed gradient instead
            tau = 100.0
            p = compute_smoothing_weights(c, tau)
            g = smoothed_gradient_fft(a, c, p)
            g = g / (float(np.linalg.norm(g)) + 1e-15)

            found = False
            for alpha in [base_step * mult for mult in [0.1, 0.25, 0.5, 1.0, 2.0, 4.0]]:
                a_try = a - alpha * g
                a_try = project_simplex(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    a = a_try
                    best_a = a_try.copy()
                    best_obj = obj_try
                    found = True
                    break

            if not found:
                # Increase tau for sharper focus
                tau *= 2.0
                if tau > 100000:
                    break

    return best_a


def fourier_perturb(a, rng, strength=0.03):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 8
    x = np.arange(n)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * x + phase)
    a_new = a * np.exp(pert)
    return normalize(a_new)


def random_sparse_perturb(a, rng, strength=0.05, sparsity=0.1):
    """Perturb a random subset of coordinates."""
    n = len(a)
    mask = rng.random(n) < sparsity
    pert = strength * rng.standard_normal(n) * mask
    a_new = a + pert * a
    return normalize(a_new)


def make_sine_init(n):
    """Sine-shaped initialization."""
    j = np.arange(n)
    a = np.sin(np.pi * (j + 1) / (n + 1))
    return normalize(a)


def make_cosine_init(n):
    """Raised cosine initialization."""
    j = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * j / (n - 1))) if n > 1 else np.ones(1)
    return normalize(a)


def make_bell_init(n, width=0.3):
    """Gaussian bell initialization."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    a = np.exp(-((j - center) / (width * n)) ** 2)
    return normalize(a)


def make_trapezoid_init(n, edge_frac=0.2):
    """Trapezoidal initialization."""
    j = np.arange(n)
    edge = int(edge_frac * n)
    a = np.ones(n)
    a[:edge] = np.linspace(0.3, 1.0, edge)
    a[-edge:] = np.linspace(1.0, 0.3, edge)
    return normalize(a)


def lp_alternate_v2(a, time_budget, seed=42, verbose=True):

            break

        iter_count += 1

        # Try multiple blending factors
        candidates_t = [t_blend, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if iter_count > 50:
            candidates_t.extend([0.05, 0.02, 0.01])

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend

        for t in candidates_t:

            a_try = (1.0 - t) * a + t * g_lp
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)

def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

    if remaining > 20:

        alloc_lp = min(remaining * 0.4, 120.0)
        if alloc_lp > 15:
            a_final = lp_alternate_v2(best_a, time_budget=alloc_lp,
                                       seed=seed + 99999, verbose=False)
            obj_final = compute_obj(a_final)
            if obj_final < best_obj:
                best_obj = obj_final
                best_a = a_final.copy()
                print(f"  LP final: obj={best_obj:.10f} **")

        # Peak flatten
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc_pf = min(remaining * 0.3, 60.0)
            a_final = targeted_peak_flatten(best_a, time_budget=alloc_pf)
            obj_final = compute_obj(a_final)
            if obj_final < best_obj:
                best_obj = obj_final
                best_a = a_final.copy()
                print(f"  Flatten final: obj={best_obj:.10f} **")

        # Softmax GD
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc_sg = min(remaining * 0.3, 60.0)
            a_final = softmax_gradient_descent(best_a, time_budget=alloc_sg,
                                                tau_init=200.0, tau_max=100000.0)
            obj_final = compute_obj(a_final)
            if obj_final < best_obj:
                best_obj = obj_final
                best_a = a_final.copy()
                print(f"  Softmax final: obj={best_obj:.10f} **")

    # === Final validation ===
    if best_a is None:
        best_a = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        final_obj = val
    else:
        result = [1.0] * 100
        final_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={final_obj:.10f}, time={elapsed:.1f}s ===")

    return result
