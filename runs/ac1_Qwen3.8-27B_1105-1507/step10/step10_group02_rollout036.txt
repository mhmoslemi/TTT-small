from scipy.fft import rfft, irfft
import time
# cvxpy for LP if needed

def compute_conv(a):
    """Full convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj_normalized(a):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def exact_1d_solve(i, v, a, conv, S):
    """
    Find optimal x for coordinate i, given all others fixed.
    a: current sequence (sum = S)
    conv: current convolution
    v: current value of a[i]
    Returns optimal x and new objective.
    """
    n = len(a)
    D = S - v  # sum of all other elements
    
    if D < 1e-15:
        return v  # can't change if all mass is in this coordinate
    
    k_min = i
    k_max = i + n - 1
    
    # Precompute: max of conv outside the active range
    if k_min == 0 and k_max == 2*n-2:
        C_out = 0.0
    else:
        # conv has length 2n-1, indices 0 to 2n-2
        # Active range is [k_min, k_max]
        outside = np.concatenate([conv[:k_min], conv[k_max+1:]]) if k_max+1 <= 2*n-2 else (conv[:k_min] if k_min > 0 else np.array([0.0]))
        C_out = float(np.max(outside)) if len(outside) > 0 else 0.0
    
    # Active linear terms: k in [k_min, k_max], k != 2*i
    # For each such k: term = conv[k] + 2*a[k-i]*(x-v)
    # Slope: 2*a[k-i], intercept: conv[k] - 2*a[k-i]*v
    quad_idx = 2*i
    quad_active = (0 <= quad_idx <= 2*n-2)
    
    # Get indices for linear terms
    ks = np.arange(k_min, k_max+1)
    mask = ks != quad_idx
    ks_lin = ks[mask]
    
    if len(ks_lin) > 0:
        a_ki = a[ks_lin - i]  # a[k-i] for each k
        c_k = conv[ks_lin]
    else:
        a_ki = np.array([])
        c_k = np.array([])
    
    # Quadratic term
    c_2i = float(conv[quad_idx]) if quad_active else 0.0
    
    # Build list of candidate x values
    candidates = [0.0, v]
    
    # Stationary point of quadratic/(D+x)^2
    if D > 1e-15:
        x_quad = (c_2i - v*v) / D
        if x_quad > 0:
            candidates.append(x_quad)
    
    # Stationary points of linear terms
    if len(ks_lin) > 0:
        for idx in range(len(ks_lin)):
            if a_ki[idx] > 1e-15:
                x_lin = D + 2*v - c_k[idx] / a_ki[idx]
                if x_lin > 0:

    # Kink points: where quadratic = linear
    if quad_active and len(ks_lin) > 0:
        for idx in range(len(ks_lin)):
            # c_2i + x^2 - v^2 = c_k + 2*a[k-i]*(x-v)
            # x^2 - 2*a[k-i]*x + (c_2i - v^2 - c_k + 2*a[k-i]*v) = 0
            b = -2 * a_ki[idx]
            c = c_2i - v*v - c_k[idx] + 2*a_ki[idx]*v
            disc = b*b - 4*c
            if disc >= 0:
                sqrt_disc = np.sqrt(disc)
                x1 = (-b + sqrt_disc) / 2
                x2 = (-b - sqrt_disc) / 2
                if x1 > 0:
                    candidates.append(x1)
                if x2 > 0:
                    candidates.append(x2)
    
    # Kink points: where C_out = linear (if C_out > 0)
    if C_out > 0 and len(ks_lin) > 0:

            if a_ki[idx] > 1e-15:

                if x_kink > 0:
                    candidates.append(x_kink)
    
    # Kink points: where C_out = quadratic
    if C_out > 0 and quad_active:
        # C_out = c_2i + x^2 - v^2
        # x^2 = C_out - c_2i + v^2
        x2_val = C_out - c_2i + v*v

            best_x = x

    width = max(1e-4 * max(best_x, v, S/n), 1e-8)

    x_hi = best_x + width

    c_pt = x_hi - gr * (x_hi - x_lo)

    fc = f_1d(c_pt)

    for _ in range(20):

            x_hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = x_hi - gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
        else:
            x_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = x_lo + gr * (x_hi - x_lo)
            fd = f_1d(d_pt)
    
    x_gs = (x_lo + x_hi) / 2.0
    f_gs = f_1d(x_gs)
    if f_gs < best_f:
        best_f = f_gs
        best_x = x_gs
    
    return best_x, best_f

"""
Optimizes step function coefficients using exact 1D coordinate descent (solving each
subproblem via candidate enumeration of stationary and kink points), combined with
iterative LP polishing (mirror descent) and multi-n exploration.
Key innovation: exact 1D subproblem solves replace heuristic golden section search,
finding true per-coordinate optima. LP polishing uses iterative blending to
converge to strong local optima.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def obj_normalized(a):
    """Compute 2*n*max(conv) assuming sum(a)=1 (normalized)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def exact_1d_candidates(i, v, a, conv, S, n):
    """
    For coordinate i with current value v, compute all candidate x values
    where the 1D objective could have a minimum.
    Returns sorted list of candidate x values (all >= 0).
    """
    D = S - v
    if D < 1e-15:
        return [0.0]

    k_min = i
    k_max = min(i + n - 1, 2 * n - 2)

    # Max of conv outside active range
    if k_min > 0 or k_max < 2 * n - 2:
        parts = []
        if k_min > 0:
            parts.append(conv[:k_min])
        if k_max < 2 * n - 2:
            parts.append(conv[k_max + 1:])
        C_out = float(np.max(np.concatenate(parts))) if parts else 0.0
    else:
        C_out = 0.0

    quad_idx = 2 * i
    quad_active = (0 <= quad_idx <= 2 * n - 2)
    c_2i = float(conv[quad_idx]) if quad_active else 0.0

    # Linear terms
    ks = np.arange(k_min, k_max + 1)
    mask = ks != quad_idx
    ks_lin = ks[mask]

    a_ki = np.zeros(len(ks_lin))
    c_k = np.zeros(len(ks_lin))
    if len(ks_lin) > 0:
        a_ki = a[ks_lin - i].copy()
        c_k = conv[ks_lin].copy()

    candidates = set()
    candidates.add(0.0)
    candidates.add(v)

    # Stationary point of quadratic term
    x_quad = (c_2i - v * v) / D
    if x_quad > 1e-12:
        candidates.add(x_quad)

    # Stationary points of linear terms
    if len(ks_lin) > 0:
        pos_mask = a_ki > 1e-15
        if np.any(pos_mask):
            x_lins = D + 2.0 * v - c_k[pos_mask] / a_ki[pos_mask]
            for xl in x_lins:
                if xl > 1e-12:
                    candidates.add(float(xl))

    # Kink points: quadratic = linear
    if quad_active and len(ks_lin) > 0:
        pos_mask = a_ki > 1e-15
        if np.any(pos_mask):
            for idx in np.where(pos_mask)[0]:
                # x^2 - 2*a_ki[idx]*x + (c_2i - v^2 - c_k[idx] + 2*a_ki[idx]*v) = 0
                A2 = 1.0
                B2 = -2.0 * a_ki[idx]
                C2 = c_2i - v * v - c_k[idx] + 2.0 * a_ki[idx] * v
                disc = B2 * B2 - 4.0 * A2 * C2
                if disc >= 0:
                    sqrt_d = np.sqrt(disc)
                    x1 = (-B2 + sqrt_d) / (2.0 * A2)
                    x2 = (-B2 - sqrt_d) / (2.0 * A2)
                    if x1 > 1e-12:
                        candidates.add(float(x1))
                    if x2 > 1e-12:
                        candidates.add(float(x2))

    # Kink: C_out = linear
    if C_out > 1e-15 and len(ks_lin) > 0:
        pos_mask = a_ki > 1e-15
        if np.any(pos_mask):
            x_kinks = (C_out - c_k[pos_mask]) / (2.0 * a_ki[pos_mask]) + v
            for xk in x_kinks:
                if xk > 1e-12:
                    candidates.add(float(xk))

    # Kink: C_out = quadratic
    if C_out > 1e-15 and quad_active:
        x2_val = C_out - c_2i + v * v
        if x2_val > 0:
            candidates.add(float(np.sqrt(x2_val)))

    # Heuristic candidates
    mean_val = S / n
    candidates.add(mean_val * 0.5)
    candidates.add(mean_val * 1.5)
    candidates.add(v * 0.5)
    candidates.add(v * 2.0)

    # Filter to reasonable range
    max_x = max(10.0 * v, 3.0 * S, 1.0)
    result = sorted(x for x in candidates if 0.0 <= x <= max_x)
    return result


def eval_1d_batch(x_arr, v, a_ki, c_k, c_2i, quad_active, C_out, D, n):
    """Evaluate the 1D objective at multiple x values (vectorized)."""
    # x_arr: shape (M,)
    # a_ki, c_k: shape (L,) - linear terms
    # Returns: shape (M,) - objective values

    max_vals = np.full_like(x_arr, C_out, dtype=np.float64)

    if quad_active:
        quad_vals = c_2i + x_arr * x_arr - v * v
        max_vals = np.maximum(max_vals, quad_vals)

    if len(a_ki) > 0:
        # lin_vals[k, m] = c_k[k] + 2*a_ki[k]*(x_arr[m] - v)
        # = (c_k[k] - 2*a_ki[k]*v) + 2*a_ki[k]*x_arr[m]
        precomp = c_k - 2.0 * a_ki * v
        # Shape: (L, M)
        lin_matrix = precomp[:, np.newaxis] + 2.0 * a_ki[:, np.newaxis] * x_arr[np.newaxis, :]
        max_lin = np.max(lin_matrix, axis=0)
        max_vals = np.maximum(max_vals, max_lin)

    denom = (D + x_arr) ** 2
    denom = np.maximum(denom, 1e-30)
    return 2.0 * n * max_vals / denom


def exact_coordinate_descent(a, time_budget=300.0, verbose=True, seed=42):
    """
    Exact coordinate descent: for each coordinate, find the true 1D optimum
    by enumerating all candidate points (stationary points and kinks).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-15:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        elapsed = time.monotonic() - start

        if elapsed > time_budget:
            break

        # Recompute conv periodically
        if sweep % 5 == 1 or sweep == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        order = rng.permutation(n)
        S = float(np.sum(a))
        improved = False

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            D = S - v
            if D < 1e-12:
                continue

            k_min = i
            k_max = min(i + n - 1, 2 * n - 2)

            # Max outside active range
            if k_min > 0 or k_max < 2 * n - 2:
                parts = []
                if k_min > 0:
                    parts.append(conv[:k_min])
                if k_max < 2 * n - 2:
                    parts.append(conv[k_max + 1:])
                C_out = float(np.max(np.concatenate(parts))) if parts else 0.0
            else:
                C_out = 0.0

            quad_idx = 2 * i
            quad_active = (0 <= quad_idx <= 2 * n - 2)
            c_2i = float(conv[quad_idx]) if quad_active else 0.0

            ks = np.arange(k_min, k_max + 1)
            mask = ks != quad_idx
            ks_lin = ks[mask]
            a_ki = a[ks_lin - i].copy() if len(ks_lin) > 0 else np.array([])
            c_k = conv[ks_lin].copy() if len(ks_lin) > 0 else np.array([])

            # Get candidates
            candidates = exact_1d_candidates(i, v, a, conv, S, n)
            if len(candidates) == 0:
                candidates = [0.0, v]

            x_arr = np.array(candidates, dtype=np.float64)

            # Evaluate all candidates
            f_vals = eval_1d_batch(x_arr, v, a_ki, c_k, c_2i, quad_active, C_out, D, n)
            best_idx = int(np.argmin(f_vals))
            x_opt = float(x_arr[best_idx])
            f_opt = float(f_vals[best_idx])

            # Refine with golden section around best candidate
            width = max(0.01 * max(abs(x_opt), abs(v), S / n), 1e-6)
            x_lo = max(0.0, x_opt - width)
            x_hi = x_opt + width

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = eval_1d_batch(np.array([c_pt]), v, a_ki, c_k, c_2i, quad_active, C_out, D, n)[0]
            fd = eval_1d_batch(np.array([d_pt]), v, a_ki, c_k, c_2i, quad_active, C_out, D, n)[0]

            for _ in range(25):
                if x_hi - x_lo < 1e-12 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = eval_1d_batch(np.array([c_pt]), v, a_ki, c_k, c_2i, quad_active, C_out, D, n)[0]
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = eval_1d_batch(np.array([d_pt]), v, a_ki, c_k, c_2i, quad_active, C_out, D, n)[0]

            x_gs = (x_lo + x_hi) / 2.0
            f_gs = float(eval_1d_batch(np.array([x_gs]), v, a_ki, c_k, c_2i, quad_active, C_out, D, n)[0])
            if f_gs < f_opt:
                x_opt = x_gs
                f_opt = f_gs

            if abs(x_opt - v) < 1e-14:
                continue

            # Update a and conv
            delta = x_opt - v
            a[i] = x_opt
            S += delta

            if quad_active and 0 <= quad_idx <= 2 * n - 2:
                conv[quad_idx] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki

            improved = True

            # Track best
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, t={elapsed:.1f}s, improved={improved}")

        # Early stopping if no improvement for many sweeps
        if not improved and sweep > 20:
            # Try a perturbation to escape
            if sweep % 40 == 0:
                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                S = 1.0
                conv = compute_conv(a)

    # Final recompute
    S_final = float(np.sum(a))
    if S_final > 1e-10:
        obj_final = 2.0 * n * float(np.max(compute_conv(a / S_final)))
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a.copy()

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=30.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]
    start = time.monotonic()
    for solver in solvers:
        if time.monotonic() - start > timeout:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None


def lp_polish(a, time_budget=60.0, verbose=False, seed=42):
    """Iterative LP polishing with adaptive blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    iteration = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        iteration += 1
        g_lp, M_lp = solve_lp_minpeak(a, timeout=min(remaining - 2, 30.0))
        if g_lp is None:
            break

        # Line search for blending factor
        best_t = 0.0
        best_t_obj = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-12 and best_t > 0:
            best_obj = best_t_obj
            best_a = ((1.0 - best_t) * a + best_t * g_lp)
            best_a = np.maximum(best_a, 0)
            best_a = best_a / best_a.sum()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 4:
                # Perturb
                pert = rng.normal(0, 0.01, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if time.monotonic() - start > time_budget * 0.85:
                    break

        if verbose and iteration % 2 == 0:
            print(f"    LP iter {iteration}: obj={best_obj:.10f}, t={best_t}, M_lp={M_lp:.6f}")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x ** 2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        # Periodic with random frequency
        freq = rng.uniform(2, 8)
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Symmetric with gap in middle
        a = np.ones(n)
        gap_start = n // 3
        gap_end = 2 * n // 3
        a[gap_start:gap_end] = rng.uniform(0.1, 0.5)
        a = a / a.sum()
    elif mode == 7:
        # Alternating high/low
        a = np.ones(n)
        a[::2] = rng.uniform(0.5, 1.5)
        a[1::2] = rng.uniform(0.2, 0.8)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: exact coordinate descent + LP polishing + multi-n exploration +
    diverse restarts. Uses exact 1D subproblem solves for faster, more accurate
    coordinate updates.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = 2.0 * n_default * float(np.max(compute_conv(best_seq)))
        print(f"Default init (n={n_default}): obj={best_score:.10f}")

    # === Phase 1: Exact Coordinate Descent ===
    print(f"\n=== Phase 1: Exact Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.40, 550.0)
    if p1_budget > 30:
        a_cd = exact_coordinate_descent(best_seq, time_budget=p1_budget,
                                        verbose=True, seed=seed)
        obj_cd = 2.0 * a_cd.size * float(np.max(compute_conv(a_cd / a_cd.sum())))
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase 1 CD: obj={best_score:.10f} **")
        else:
            print(f"  Phase 1 CD: obj={obj_cd:.10f} (no improvement)")

    # === Phase 2: LP Polishing ===
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 250.0)
    if p2_budget > 30:
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(p2_budget / (3 - rnd), remaining * 0.5)
            if alloc < 15:
                break
            a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True,
                             seed=1000 + rnd * 111)
            obj_lp = 2.0 * a_lp.size * float(np.max(compute_conv(a_lp / a_lp.sum())))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.10f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.10f}")
            # Quick CD after LP
            remaining = deadline - time.monotonic()
            if remaining > 30 and best_score < np.inf:
                a_cd2 = exact_coordinate_descent(best_seq, time_budget=min(30, remaining * 0.15),
                                                 verbose=False, seed=2000 + rnd * 100)
                obj_cd2 = 2.0 * a_cd2.size * float(np.max(compute_conv(a_cd2 / a_cd2.sum())))
                if obj_cd2 < best_score:
                    best_score = obj_cd2
                    best_seq = a_cd2.copy()
                    print(f"  CD after LP {rnd}: obj={best_score:.10f} **")

    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 100), n_best + 100, n_best + 200, n_best + 500,
        600, 700, 800, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2500 and abs(n - n_best) > 30]
    n_options = n_options[:6]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.2, 60.0)
        if alloc > 20:
            a_try = exact_coordinate_descent(a_try, time_budget=alloc, verbose=False,
                                             seed=n_try + 7)

        obj_try = 2.0 * a_try.size * float(np.max(compute_conv(a_try / a_try.sum())))
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 4: Diverse Restarts ===
    print(f"\n=== Phase 4: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = 1 + r
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 60.0)
        if alloc > 25:
            a_r = exact_coordinate_descent(a_r, time_budget=alloc, verbose=False,
                                           seed=5000 + r * 100)
            obj_r = 2.0 * a_r.size * float(np.max(compute_conv(a_r / a_r.sum())))
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.10f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.10f}")

    # === Phase 5: Final CD + LP ===
    print(f"\n=== Phase 5: Final Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc_cd = min(remaining * 0.5, 200.0)
        a_final = exact_coordinate_descent(best_seq, time_budget=alloc_cd,
                                           verbose=True, seed=seed + 9999)
        obj_final = 2.0 * a_final.size * float(np.max(compute_conv(a_final / a_final.sum())))
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.10f} **")
        else:
            print(f"  Final CD: obj={obj_final:.10f}")

    # Final LP pass
    remaining = deadline - time.monotonic()
    if remaining > 25:
        a_lp_final = lp_polish(best_seq, time_budget=min(remaining * 0.6, 60.0),
                               verbose=False, seed=7777)
        obj_lp_f = 2.0 * a_lp_final.size * float(np.max(compute_conv(a_lp_final / a_lp_final.sum())))
        if obj_lp_f < best_score:
            best_score = obj_lp_f
            best_seq = a_lp_final.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    # One more quick CD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = exact_coordinate_descent(best_seq, time_budget=min(remaining * 0.7, 40.0),
                                          verbose=False, seed=424242)
        obj_last = 2.0 * a_last.size * float(np.max(compute_conv(a_last / a_last.sum())))
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last CD: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
