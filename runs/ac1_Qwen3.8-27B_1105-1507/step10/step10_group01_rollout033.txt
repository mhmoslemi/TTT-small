Derivative: 2*a_i

Actually, let me think about what FISTA would look like here.

FISTA solves: min f(x) + g(x) where f is smooth and g is simple.

In our case, f(a) = (1/tau) * log(sum_k exp(tau * (a*a)_k)) is smooth (for tau > 0), and g(a) = indicator of {a >= 0, sum(a) = 1}.

The proximal operator of g is: project onto {a >= 0, sum(a) = 1}, which is the "softmax-like" projection:
prox_g(a)_i = max(0, a_i - lambda) where lambda is chosen so sum_i max(0, a_i - lambda) = 1.

This is tractable (bisection on lambda).

So FISTA iterations:
1. y_k = a_k + (t_k - 1)/(t_{k+1}) * (a_k - a_{k-1})
2. a_{k+1} = prox_{g, L^{-1}}(y_k - (1/L) * grad f(y_k))
3. t_{k+1} = (1 + sqrt(1 + 4*t_k^2))/2

where L is the Lipschitz constant of grad f.

The gradient of f(a) = (1/tau) * log(sum_k exp(tau * (a*a)_k)) is:
grad f(a) = sum_k p_k * grad((a*a)_k) where p_k = exp(tau*(a*a)_k) / sum_j exp(tau*(a*a)_j)

And grad((a*a)_k) = 2 * b^{(k)} (as derived above).

So grad f(a) = 2 * sum_k p_k * b^{(k)} = 2 * (p convolved with a, truncated to n components).

This can be computed efficiently with FFT!

The Lipschitz constant L of grad f: |Hessian of f| <= tau * max eigenvalue of the "convolution matrix" ... this is tricky. We can estimate it empirically.

Hessian f = sum_k p_k * Hessian q_k + tau * Cov_p(q) (the covariance term is related to the variance of the q_k's)

The first term has norm at most max_k ||Hessian q_k|| = 2 * max_k ||C_k||.

For our convolution matrices, ||C_k|| (spectral norm) is at most 2 (since C_k is a "permutation-like" matrix with entries 0 or 1, and at most 2 non-zero entries per row).

Actually, C_k has at most 2 non-zeros per row (since for each i, there's at most one j such that i+j = k, and at most one j such that j+i = k, but these are the same). So C_k is a (weighted) permutation matrix, and ||C_k||_op <= 1.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-driven direction finding, FISTA on smoothed objectives,
subgradient polishing, and multi-n exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import fft, ifft
import time

# Global variables
best_score_global = np.inf
best_seq_global = None

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa).real[:2*n-1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A such that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    m = 2*n - 1
    A = np.zeros((m, n))
    for k in range(m):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a_norm, g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.clip(g.value, 0, None)
                s = g_val.sum()
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except:
            continue
    return None, None

def smoothed_gradient(a, tau):
    """Gradient of (1/tau)*log(sum exp(tau*(a*a)_k))."""
    n = len(a)
    conv = compute_conv(a)
    max_c = conv.max()
    exp_vals = np.exp(tau * (conv - max_c))
    p = exp_vals / exp_vals.sum()
    
    N = 1
    while N < 3*n:
        N <<= 1
    grad = 2.0 * ifft(fft(p, N) * np.conj(fft(a, N))).real[:n]
    return grad, p

def fista_smoothed(a, tau, L, time_budget, max_iter=50000):
    """FISTA on smoothed objective with projection onto {a>=0, sum(a)=1}."""
    n = len(a)
    a = a / a.sum()
    a_prev = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < time_budget and it < max_iter:
        # Momentum
        t_new = (1 + np.sqrt(1 + 4*t_mom*t_mom)) / 2
        y = a + (t_mom - 1) / t_new * (a - a_prev)
        y = np.maximum(y, 0)
        s = y.sum()
        if s < 1e-10:
            y = best_a.copy()
        else:
            y = y / s
        
        # Gradient
        grad, p = smoothed_gradient(y, tau)
        
        # Gradient step
        z = y - grad / L
        
        # Proximal: project onto {a >= 0, sum(a) = 1}
        z_sorted = np.sort(z)[::-1]
        cumsum = np.cumsum(z_sorted)
        rho = np.arange(1, len(z)+1)
        cond = z_sorted - (cumsum - 1) / rho > 0
        lam_idx = int(np.sum(cond))
        if lam_idx > 0:
            lam = (cumsum[lam_idx-1] - 1) / lam_idx
        else:
            lam = (z.sum() - 1) / n
        a_new = np.maximum(z - lam, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        # Check
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a_prev = a.copy()
        a = a_new
        t_mom = t_new
        it += 1
    
    return best_a, best_obj

def subgradient_polish(a, time_budget, lr_init=0.01):
    """Polish using subgradient of exact max objective."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = conv.max()
        
        # Find active set (top convolution values)
        threshold = max_c - 1e-10
        active = np.where(conv >= threshold)[0]
        
        # Subgradient: average gradient over active set
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                idx = k - i
                if 0 <= idx < n:
                    grad[i] += 2.0 * a[idx]
        grad /= len(active)
        
        # Step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:

        else:
            a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        else:
            lr *= 0.9
        
        a = a_new
    
    return best_a, best_obj

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-driven direction finding combined with FISTA on smoothed objectives,
subgradient polishing, and multi-n exploration. Key improvements over previous:
- FISTA with adaptive Lipschitz estimation for faster convergence
- Higher tau smoothing for final polish (up to 10000)
- Subgradient polishing on exact objective
- More efficient line search (fewer evaluations)
- Better time management and multi-n exploration
"""

import numpy as np
import cvxpy as cp
from scipy.fft import fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa).real[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    m = 2 * n - 1
    k_idx = np.arange(m)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((m, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a_norm, g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.clip(g.value, 0, None)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_grad_and_weights(a, tau):
    """Compute gradient and probability weights of smoothed max."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 3 * n:
        N <<= 1
    grad = 2.0 * ifft(fft(p, N) * np.conj(fft(a, N))).real[:n]
    return grad, p


def project_simplex_nonneg(z):
    """Project z onto {a >= 0, sum(a) = 1}."""
    n = len(z)
    z_sorted = np.sort(z)[::-1]
    cumsum = np.cumsum(z_sorted)
    rho = np.arange(1, n + 1)
    cond = z_sorted - (cumsum - 1.0) / rho > 0
    lam_idx = int(np.sum(cond))
    if lam_idx > 0:
        lam = (cumsum[lam_idx - 1] - 1.0) / lam_idx
    else:
        lam = (z.sum() - 1.0) / n
    a_new = np.maximum(z - lam, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_new / s


def fista_smoothed(a, tau, time_budget, verbose=False):
    """FISTA on smoothed objective with projection onto {a>=0, sum(a)=1}."""
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    a_prev = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    # Estimate Lipschitz constant
    L = 2.0 * tau
    converged_L = False

    while time.monotonic() - start < time_budget:
        # Momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = a + (t_mom - 1.0) / t_new * (a - a_prev)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Gradient
        grad, p = smoothed_grad_and_weights(y, tau)

        # Adaptive L: backtracking
        step_count = 0
        while step_count < 20:
            z = y - grad / L
            a_cand = project_simplex_nonneg(z)
            # Check if gradient step decreases the smoothed objective
            f_y = (1.0 / tau) * (np.log(float(np.sum(np.exp(tau * (compute_conv(y) - compute_conv(y).max()))))) + tau * compute_conv(y).max())
            f_cand_conv = compute_conv(a_cand)
            f_cand = (1.0 / tau) * (np.log(float(np.sum(np.exp(tau * (f_cand_conv - f_cand_conv.max()))))) + tau * f_cand_conv.max())
            f_quad = f_y + float(np.dot(grad, a_cand - y)) + 0.5 * L * float(np.dot(a_cand - y, a_cand - y))
            if f_cand <= f_quad + 1e-10:
                break
            L *= 2.0
            step_count += 1

        if not converged_L:
            converged_L = True
        else:
            L = max(L * 0.95, 2.0 * tau)

        a_new = a_cand

        # Check exact objective
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        a_prev = a.copy()
        a = a_new
        t_mom = t_new
        it += 1

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"      FISTA it{it}: obj={best_obj:.8f}, L={L:.1f}, time={elapsed:.0f}s")

    return best_a, best_obj


def subgradient_polish(a, time_budget, lr_init=0.005, verbose=False):
    """Polish using subgradient of exact max objective with FFT."""
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    it = 0

    N = 1
    while N < 3 * n:
        N <<= 1
    A_fft = fft(a, N)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))

        # Active set: top values within relative tolerance
        threshold = max_c * 0.999
        active = conv >= threshold
        n_active = int(np.sum(active))
        if n_active == 0:
            active[int(np.argmax(conv))] = True
            n_active = 1

        # Subgradient via FFT
        p_active = active.astype(np.float64) / n_active
        A_fft = fft(a, N)
        grad = (2.0 / n_active) * ifft(fft(p_active, N) * np.conj(A_fft)).real[:n]

        # Gradient step with backtracking
        best_step_obj = best_obj
        best_step_a = best_a
        for step_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr_try = lr * step_mult
            a_try = a - lr_try * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_step_obj:
                best_step_obj = obj_try
                best_step_a = a_try.copy()

        if best_step_obj < best_obj:
            best_obj = best_step_obj
            best_a = best_step_a.copy()
            a = best_step_a
            lr = min(lr * 1.1, 0.05)
        else:
            lr *= 0.7

        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"      Subgrad it{it}: obj={best_obj:.8f}, lr={lr:.6f}, time={elapsed:.0f}s")

    return best_a, best_obj


def lp_optimize(a, time_budget, verbose=False):
    """LP-driven optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget:
        if time_budget - (time.monotonic() - start) < 2.0:
            break

        g, M = solve_lp(a)
        if g is None:
            break

        # Line search: try several step sizes
        conv_curr = compute_conv(a)
        max_curr = float(np.max(conv_curr))
        gap = max_curr - M
        if gap < 1e-8:
            stall += 2
        else:
            # Oriented step sizes
            t_max = min(0.95, max(0.1, gap / (max_curr + 1e-10) * 2.0))
            ts = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, t_max, min(0.95, t_max * 1.3)]
            ts = sorted(set([max(0.01, min(0.99, t)) for t in ts]))

            best_t_obj = best_obj
            best_t_a = best_a
            for t in ts:
                a_try = (1.0 - t) * a + t * g
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj - 1e-12:
                    best_t_obj = obj_try
                    best_t_a = a_try.copy()

            if best_t_obj < best_obj - 1e-12:
                best_obj = best_t_obj
                best_a = best_t_a.copy()
                a = best_t_a
                stall = 0
            else:
                stall += 1
                # Try multiplicative update as alternative
                a_mult = a * np.power(g + 1e-10, 0.3)
                a_mult = np.maximum(a_mult, 0)
                s_mult = a_mult.sum()
                if s_mult > 1e-10:
                    a_mult = a_mult / s_mult
                    obj_mult = compute_obj(a_mult)
                    if obj_mult < best_obj - 1e-12:
                        best_obj = obj_mult
                        best_a = a_mult.copy()
                        a = best_t_a
                        stall = 0

        it += 1
        if verbose and (it % 10 == 0 or stall > 10):
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, M={M:.6f}, max={max_curr:.6f}, stall={stall}, time={elapsed:.0f}s")

        if stall >= 8:
            # Perturb to escape
            a_pert = a + 0.01 * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            a = a_pert
            stall = 0
            if it > 50:
                break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
