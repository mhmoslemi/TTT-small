"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid approach: fast Nesterov-accelerated gradient descent with adaptive
soft-max smoothing, periodic LP-guided direction correction, multi-length exploration,
and smart restart strategies. Key innovations: backtracking line search with
armijo condition, dual LP gap monitoring, and progressive tau scheduling.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gradient_softmax(a, tau):
    """Compute gradient of soft-max smoothed objective using FFT.
    Returns (gradient, convolution, max_conv, p_weights).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Clamp for numerical stability
    shifted = np.minimum(shifted, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]

    return g, conv, max_conv, p


def gradient_exact_max(a):
    """Compute subgradient at the exact max position(s) of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    # Find all positions within tolerance of max
    tol = 1e-10 * max(1.0, max_conv)
    active = np.where(conv >= max_conv - tol)[0]

    if len(active) == 1:
        k_star = active[0]
        grad = np.zeros(n)
        for i in range(n):
            idx = k_star - i
            if 0 <= idx < n:
                grad[i] = 2.0 * a[idx]
            elif 0 <= (k_star - i) < 2 * n - 1 and (k_star - i) < n:
                grad[i] = 2.0 * a[k_star - i]
        return grad, conv, max_conv

    # Multiple active positions: average subgradient
    grad = np.zeros(n)
    for k_star in active:
        for i in range(n):
            idx = k_star - i
            if 0 <= idx < n:
                grad[i] += 2.0 * a[idx]
    grad /= len(active)
    return grad, conv, max_conv


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_direction(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a/g_normalized, g) <= M, sum(g)=1, g>=0.
    Returns optimal g or None.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    return g_val
        except Exception:
            continue
    return None


def nesterov_gd_cycle(a, time_budget, tau_init, tau_end, momentum=0.9,

        return a

    best_obj = compute_obj(a)
    start = time.monotonic()

    if base_lr is None:
        base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    it = 0
    best_it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Cosine schedule for tau
        cos_val = 0.5 * (1 + np.cos(np.pi * progress))
        tau = tau_init + (tau_end - tau_init) * (1 - cos_val)

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient

            g, conv, max_c, p = gradient_softmax(y, tau)
        else:
            g, conv, max_c = gradient_exact_max(y)

            remaining = time_budget - (time.monotonic() - start)
            if remaining > 10:
                g_lp = solve_lp_direction(y)
                if g_lp is not None:
                    # Blend: use LP direction as a "pull"
                    g = 0.5 * g + 0.5 * n * (g_lp / np.sum(g_lp))
                    # Actually, let's use the LP solution as the step direction
                    # and blend with the gradient
                    pass

        # Adaptive learning rate using Barzilai-Borwein style
        if it > 0 and it % 5 == 0:
            # Simple adaptive: scale lr by recent progress
            recent_obj = compute_obj(a)

            else:
                lr = lr * 0.85

        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))

            a_new = best_a.copy()
            v = np.zeros(n)

            a_new = a_new / s_new

            a_new = y - lr * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                break
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            backtrack += 1

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        v = a_new - y
        a = a_new
        it += 1

        # Decay lr slowly
        lr = max(1e-10, lr * 0.9999)

        # Restart if stuck
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * (0.3 + 0.7 * rng.random())
            best_it = it

        if verbose and it % 2000 == 0:
            print(f"      GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}")

def lp_refinement_cycle(a, time_budget, verbose=False, seed=42):
    """
    LP-based refinement: repeatedly solve LP and blend with current solution.

    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)

    best_it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp = solve_lp_direction(a)
        if g_lp is not None:
            it += 1
            # Golden section line search on [0, 1] for blend parameter
            # f(t) = obj((1-t)*a + t*g_lp)
            best_t = 0.0
            best_t_obj = best_obj

            # Try a set of candidate t values
            t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
            for t in t_candidates:
                a_try = (1.0 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_t = t

            if best_t_obj < best_obj - 1e-12:
                a_try = (1.0 - best_t) * a + best_t * g_lp
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                best_obj = best_t_obj
                best_a = a_try.copy()
                a = a_try
                best_it = it
                if verbose and it % 5 == 0:
                    print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}")
            else:
                # No improvement from LP direction
                pass

        # If stuck, perturb
        if it - best_it > 10:
            strength = min(0.1, 0.01 * (it - best_it))
            pert = 1.0 + rng.normal(0, strength, n)
            a_pert = a * np.maximum(pert, 0)
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 1e-10:
                a = a_pert / s_p
            best_it = it
            it = 0  # Reset counter

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.03, n_modes=5):
    """Add low-frequency Fourier modes as structured perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(n_modes, n // 10)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.0) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

            a_try = nesterov_gd_cycle(a_try, time_budget=alloc,
                                      tau_init=100, tau_end=2000,
                                      momentum=0.92,
                                      base_lr=3.0/np.sqrt(n_try),
                                      seed=n_try)
        # Quick polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.06, 25.0)
        if alloc2 > 10:
            a_try = fast_gradient_polish(a_try, time_budget=alloc2, seed=n_try + 1)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diversification restarts (13% budget) ===
    print(f"\n=== Phase 5: Diversification ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        # Different perturbation strategies
        if r == 0:
            a_r = fourier_perturb(best_seq, rng, strength=0.02, n_modes=8)
        elif r == 1:
            pert = 1.0 + rng.normal(0, 0.05, n_cur)
            a_r = best_seq * np.maximum(pert, 0)
            a_r = a_r / a_r.sum()
        elif r == 2:
            # Blend with symmetric version
            a_sym = (best_seq + best_seq[::-1]) / 2.0
            a_r = 0.6 * best_seq + 0.4 * a_sym / a_sym.sum()
            a_r = a_r / a_r.sum()
        elif r == 3:
            # Blend with uniform
            a_r = 0.8 * best_seq + 0.2 * np.ones(n_cur) / n_cur
            a_r = a_r / a_r.sum()
        else:
            # Stronger Fourier perturbation
            a_r = fourier_perturb(best_seq, rng, strength=0.05, n_modes=12)

        alloc = min(remaining * 0.18, 70.0)
        if alloc > 15:
            a_r = nesterov_gd_cycle(a_r, time_budget=alloc,
                                    tau_init=100, tau_end=3000,
                                    momentum=0.93,
                                    base_lr=2.5/np.sqrt(n_cur),
                                    seed=500 + r * 7)
            # Polish
            remaining = deadline - time.monotonic()
            alloc_p = min(remaining * 0.06, 25.0)
            if alloc_p > 10:
                a_r = fast_gradient_polish(a_r, time_budget=alloc_p, seed=600 + r)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 6: Final intensive LP + GD (5% budget) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        alloc = min(remaining * 0.5, 120.0)
        a_final = lp_refinement_cycle(best_seq, time_budget=alloc,
                                      verbose=True, seed=9999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.6, 60.0)
        a_final = fast_gradient_polish(best_seq, time_budget=alloc, seed=7777)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.7, 50.0)
        a_final = fast_gradient_polish(best_seq, time_budget=alloc, seed=8888)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra-final: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
