"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast scipy HiGHS LP with active-set (cutting plane) reduction for the inner
best-response problem, combined with adaptive blending, gradient polishing via
FFT-smoothed subgradients, and multi-scale exploration.
Key improvements over previous: 10x faster LP via scipy.linprog+HiGHS,
cutting-plane constraint reduction, line-search step sizes, and diverse restarts.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n1, n2 = len(a), len(b)
    size = 1
    while size < n1 + n2 - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n1 + n2 - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_rows(a_norm, indices):
    """Build rows of the convolution matrix for given constraint indices."""
    n = len(a_norm)
    rows = np.zeros((len(indices), n))
    for idx, k in enumerate(indices):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start < j_end:
            # row[j] = a_norm[k-j] for j in [j_start, j_end)
            # k-j ranges from k-j_start down to k-j_end+1
            vals = a_norm[k - j_end + 1:k - j_start + 1][::-1]
            rows[idx, j_start:j_end] = vals
    return rows


def solve_lp_full(a_norm, prev_g=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy linprog with HiGHS.
    Returns (g_opt, M_opt, lambda_dual) or (None, None, None).
    """
    n = len(a_norm)
    num_constraints = 2 * n - 1

    A_conv = build_conv_rows(a_norm, np.arange(num_constraints))

    A_ub = np.hstack([A_conv, -np.ones((num_constraints, 1))])
    b_ub = np.zeros(num_constraints)
    A_eq = np.hstack([np.ones((1, n)), [[0.0]]])
    b_eq = [1.0]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    bounds = [(0, None)] * n + [(None, None)]

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.success:
            g_val = np.maximum(result.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                g_val /= s
                # Extract dual variables (shadow prices) for inequality constraints
                lam = result.ineqlin.marginals
                if lam is not None:
                    lam = np.maximum(lam, 0)
                    lam_sum = float(np.sum(lam))
                    if lam_sum > 1e-15:
                        lam = lam / lam_sum
                    else:
                        lam = np.ones(num_constraints) / num_constraints
                else:
                    lam = np.ones(num_constraints) / num_constraints
                return g_val, float(np.max(compute_conv_cross(a_norm, g_val))), lam
    except Exception:
        pass
    return None, None, None


def solve_lp_cutting_plane(a_norm, K_init=80, K_max=400, max_rounds=10):
    """
    Solve LP using cutting plane / active set reduction.
    Start with K_init most likely binding constraints, add violated ones iteratively.
    """
    n = len(a_norm)
    num_constraints = 2 * n - 1

    # Initial active set: use uniform prior to guess binding constraints
    # The center constraints are most likely binding for "nice" sequences
    center = n - 1
    active_set = list(range(max(0, center - K_init // 2), min(num_constraints, center + K_init // 2)))

    active_set = np.array(active_set)
    g_opt = None
    M_opt = None

    for round_idx in range(max_rounds):
        A_sub = build_conv_rows(a_norm, active_set)
        K = len(active_set)

        A_ub = np.hstack([A_sub, -np.ones((K, 1)))]
        b_ub = np.zeros(K)
        A_eq = np.hstack([np.ones((1, n)), [[0.0]]])
        b_eq = [1.0]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        bounds = [(0, None)] * n + [(None, None)]

        try:
            result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs')
        except Exception:
            break

        if not result.success:
            break

        g_candidate = np.maximum(result.x[:n], 0)
        s = float(np.sum(g_candidate))
        if s < 1e-10:
            break
        g_candidate /= s
        M_candidate = float(result.x[-1])

        # Check all constraints for violation
        conv_all = compute_conv_cross(a_norm, g_candidate)
        violations = conv_all - M_candidate
        max_violation = float(np.max(violations))

        if max_violation < 1e-8:
            g_opt = g_candidate
            M_opt = M_candidate
            break

        # Add most violated constraints
        active_set_set = set(active_set.tolist())
        violated = np.where(violations > 1e-10)[0]
        # Add top violated not already in active set
        to_add = [v for v in violated if v not in active_set_set]
        to_add = to_add[:min(K_init // 2, len(to_add))]

        if len(to_add) == 0:
            # Can't add more, accept as approximate
            g_opt = g_candidate
            M_opt = float(np.max(conv_all))
            break

        active_set = np.array(list(active_set_set | set(to_add)))
        if len(active_set) > K_max:
            # Keep only top K_max most violated
            conv_on_active = conv_all[active_set]
            # Recompute all to find top
            active_set = np.argsort(-conv_all)[:K_max]

    if g_opt is None:
        # Fallback to full LP
        return solve_lp_full(a_norm)

    # Compute dual approximately: use the objective values as weights
    conv_final = compute_conv_cross(a_norm, g_opt)
    M_final = float(np.max(conv_final))
    # Softmax weights on the convolution values
    shifted = conv_final - M_final
    exp_vals = np.exp(50 * shifted)
    lam = exp_vals / float(np.sum(exp_vals))

    return g_opt, M_final, lam


def smoothed_gradient_fft(a, lam):

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    # Standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def nesterov_polish(a, lam_init, time_budget, tau=50.0):
    """
    Nesterov accelerated gradient descent on smoothed f(a) = 2n * sum_k lam[k] * h_k(a)
    projected onto simplex. Uses the given lambda distribution for smoothing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adaptive tau
    beta = 0.9
    base_lr = 1.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = a + beta * v
        y = project_simplex(y)

        # Compute gradient using current lambda (from max of a*a)
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            lam = np.zeros(len(conv))
            lam[np.argmax(conv)] = 1.0
        else:
            lam = exp_vals / sum_exp

        g = smoothed_gradient_fft(y, lam)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adaptive_lr = 0.5 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.1 * base_lr), 5.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adaptive_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        beta = min(0.9 + 0.01 * (it // 1000), 0.99)
        lr *= 0.9995
        tau = min(tau * 1.05, 500.0)
        it += 1

    return best_a

            break

                a_pert = np.maximum(a_pert, 0)
                a_pert /= a_pert.sum()
                a = a_pert

    if verbose:
        print(f"  Main loop: {iter_count} iters, total_lp_time={total_lp_time:.1f}s, "
              f"best_obj={best_obj:.10f}")

    return best_a


def fourier_perturb(a, rng, strength=0.03):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 6):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_init(n, rng, pattern='mixed'):
    """Generate diverse random initializations."""
    if pattern == 'uniform':
        return np.ones(n) / n
    elif pattern == 'fourier':
        a = np.ones(n)
        for k in range(1, 8):
            freq = k * 2 * np.pi / n
            amp = 0.5 / k
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(freq * np.arange(n) + phase)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif pattern == 'bimodal':
        a = np.zeros(n)
        center = n // 2
        width = n // 4
        for i in range(n):
            d = min(abs(i - center), abs(i - center + n), abs(i - center - n))
            a[i] = np.exp(-d**2 / (2 * width**2))
        a += 0.01 * rng.standard_normal(n)
        a = np.maximum(a, 0)
        return a / a.sum()
    elif pattern == 'random_sparse':
        a = np.zeros(n)
        n_nonzero = int(n * 0.6)
        indices = rng.choice(n, size=n_nonzero, replace=False)
        a[indices] = rng.exponential(1.0, size=n_nonzero)
        a += 0.001
        return a / a.sum()
    else:  # 'mixed'
        a = np.ones(n) / n
        a = fourier_perturb(a, rng, strength=0.1)
        return a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization with fast LP, gradient polishing,
    and multi-scale exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0

    # Initialize
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0)
        if a.size == 0 or a.sum() < 0.01:
            raise ValueError("Invalid parent")
        a = a / a.sum()
        n_init = a.size
        print(f"Parent loaded: n={n_init}, obj={compute_obj(a):.10f}")
    except Exception:
        n_init = 800
        a = np.ones(n_init) / n_init
        print(f"Default init: n={n_init}, obj={compute_obj(a):.10f}")

    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)

    # === Phase 1: Main LP loop on parent length ===
    print(f"\n=== Phase 1: Main LP (n={n_init}) ===")
    p1_deadline = start_time + budget_s * 0.50
    p1_budget = max(30, p1_deadline - time.monotonic())
    a1 = main_loop(best_a, time_budget=p1_budget, seed=seed, verbose=True)
    obj1 = compute_obj(a1)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()
        print(f"  Phase 1: obj={best_obj:.10f} **")
    else:
        print(f"  Phase 1: obj={obj1:.10f}")

    # === Phase 2: Gradient polish ===
    print(f"\n=== Phase 2: Gradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = min(remaining * 0.2, 120.0)
        a2 = nesterov_polish(best_a, None, time_budget=p2_budget, tau=30.0)
        obj2 = compute_obj(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            print(f"  Phase 2: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 2: obj={obj2:.10f}")

    # === Phase 3: Second LP pass with perturbation ===
    print(f"\n=== Phase 3: LP with Perturbation ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_pert = fourier_perturb(best_a, rng, strength=0.02)
        p3_budget = min(remaining * 0.3, 200.0)
        a3 = main_loop(a_pert, time_budget=p3_budget, seed=seed + 123, verbose=False)
        obj3 = compute_obj(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            print(f"  Phase 3: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 3: obj={obj3:.10f}")

    # === Phase 4: Try different n values ===
    print(f"\n=== Phase 4: Multi-scale Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    n_options = []
    for delta in [50, 100, 150, 200]:
        n_options.append(n_best + delta)
        n_options.append(n_best - delta)
    # Add some standard sizes
    n_options.extend([600, 700, 900, 1000, 1200])
    n_options = sorted(set(n for n in n_options if 400 <= n <= 2000))

    # Sort by proximity to current best
    n_options.sort(key=lambda x: abs(x - n_best))

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_to_length(best_a, n_try)
        alloc = min(remaining * 0.2, 100.0)
        if alloc < 20:
            break

        a_try = main_loop(a_try, time_budget=alloc * 0.7, seed=n_try, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_try = nesterov_polish(a_try, None, time_budget=min(remaining * 0.2, 30.0), tau=50.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: obj={best_obj:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 5: Fresh random starts ===
    print(f"\n=== Phase 5: Random Restarts ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        patterns = ['fourier', 'bimodal', 'random_sparse']
        for pattern in patterns:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            a_rand = random_init(n_best, rng, pattern=pattern)
            alloc = min(remaining * 0.25, 80.0)
            if alloc < 20:
                break
            a_rand = main_loop(a_rand, time_budget=alloc, seed=seed + hash(pattern) % 10000, verbose=False)
            obj_rand = compute_obj(a_rand)
            if obj_rand < best_obj:
                best_obj = obj_rand
                best_a = a_rand.copy()
                print(f"  ** {pattern} n={n_best}: obj={best_obj:.10f} **")
            else:
                print(f"  {pattern}: obj={obj_rand:.10f}")

    # === Phase 6: Final refinement ===
    print(f"\n=== Phase 6: Final Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p6_budget = min(remaining * 0.6, 100.0)
        a6 = main_loop(best_a, time_budget=p6_budget, seed=seed + 9999, verbose=False)
        obj6 = compute_obj(a6)
        if obj6 < best_obj:
            best_obj = obj6
            best_a = a6.copy()
            print(f"  Final LP: obj={best_obj:.10f} **")
        else:
            print(f"  Final LP: obj={obj6:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a7 = nesterov_polish(best_a, None, time_budget=min(remaining * 0.5, 40.0), tau=100.0)
            obj7 = compute_obj(a7)
            if obj7 < best_obj:
                best_obj = obj7
                best_a = a7.copy()
                print(f"  Final grind: obj={best_obj:.10f} **")
            else:
                print(f"  Final grind: obj={obj7:.10f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
