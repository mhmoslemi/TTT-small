"""
Improved optimizer combining adaptive-temperature smoothed gradient,
enhanced LP direction finding with golden-section line search,
multi-scale initialization, and coordinate-wise refinement.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a):
    """Solve LP: min M s.t. A @ g <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def adaptive_smoothed_gd(a, time_budget=100.0, tau_init=20.0, tau_max=500.0, seed=42):
    """Smoothed gradient descent with adaptive temperature (simulated annealing style)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 5.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    tau = tau_init
    no_improve = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Decrease temperature when making progress
            tau = max(5.0, tau * 0.97)

            a = a_new  # Still accept (like SA)

            if no_improve > 200:
                # Reset with perturbation

                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                v = np.zeros(n)
                lr = 5.0 / np.sqrt(n)
                tau = tau_init
                no_improve = 0
                if lr < 1e-12:
                    break

        # Adaptive learning rate
        lr = max(1e-12, lr * 0.999)

    return best_a

def coordinate_descent_refine(a, time_budget=60.0, seed=42):
    """Coordinate-wise line search to refine individual elements."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)

        return a

    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0

        improved = False
        for idx in order:
            idx = int(idx)

            base = best_a[idx]
            candidates = [base * f for f in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]]

            best_local_obj = best_obj

                a_try = best_a.copy()

                if s_try < 1e-10:

                s_a = best_a.sum()
                if s_a > 1e-10:
                    best_a = best_a / s_a
                best_obj = best_local_obj
                improved = True
        
        it += 1
        if not improved and it > 3:
            break

"""
Improved hybrid optimizer: LP with golden-section line search (primary),
TR-SOCP trust region, adaptive-temperature smoothed gradient, coordinate
descent refinement, and aggressive multi-length exploration.
Key improvements: golden-section line search, more LP iterations,
broader n exploration, structured perturbations, and coordinate-wise polish.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a):
    """Solve LP: min M s.t. A @ g <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_tr_sscp(a, r):
    """Solve trust region SOCP with given radius r."""
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)

    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta

    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
        cp.norm(delta, 2) <= r,
    ]

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None


def golden_section_line_search(base_a, direction, lo=0.0, hi=2.0, max_iter=18, tol=1e-10):
    """Golden section search for optimal step size along direction."""
    golden_ratio = (np.sqrt(5) - 1) / 2.0

    def obj_at_alpha(alpha):
        a_try = base_a + alpha * direction
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            return np.inf
        a_try = a_try / s_try
        return compute_obj(a_try)

    # Initial bracketing
    obj_lo = obj_at_alpha(lo)
    obj_mid = obj_at_alpha((lo + hi) / 2.0)
    obj_hi = obj_at_alpha(hi)

    if obj_mid > obj_lo and obj_hi > obj_lo:
        return 0.0, obj_lo
    elif obj_hi < obj_mid:
        lo, obj_lo = (lo + hi) / 2.0, obj_mid
        hi, obj_hi = hi, obj_hi
    else:
        hi, obj_hi = (lo + hi) / 2.0, obj_mid

    # Golden section iterations
    c = hi - golden_ratio * (hi - lo)
    d = lo + golden_ratio * (hi - lo)
    f_c = obj_at_alpha(c)
    f_d = obj_at_alpha(d)

    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if f_c < f_d:
            hi = d
            d = c
            f_d = f_c
            c = hi - golden_ratio * (hi - lo)
            f_c = obj_at_alpha(c)
        else:
            lo = c
            c = d
            f_c = f_d
            d = lo + golden_ratio * (hi - lo)
            f_d = obj_at_alpha(d)

    best_alpha = (lo + hi) / 2.0
    best_obj = obj_at_alpha(best_alpha)
    return best_alpha, best_obj


def adaptive_smoothed_gd(a, time_budget=100.0, tau_init=20.0, tau_max=500.0, seed=42):
    """Smoothed gradient descent with adaptive temperature."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 5.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    tau = tau_init
    no_improve = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        else:
            break

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            tau = max(5.0, tau * 0.97)
        else:
            a = a_new
            no_improve += 1
            tau = min(tau_max, tau * 1.04)
            if no_improve > 150:
                a = best_a * np.exp(rng.normal(0, 0.015, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                v = np.zeros(n)
                lr = 5.0 / np.sqrt(n)
                tau = tau_init
                no_improve = 0

        lr = max(1e-12, lr * 0.9992)

    return best_a


def lp_golden_polish(a, time_budget=200.0, verbose=False, seed=42):
    """LP direction with golden-section line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_direction(best_a)
        if g_lp is None:
            break

        direction = g_lp - best_a
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-12:
            no_improve += 1
        else:
            best_alpha, best_alpha_obj = golden_section_line_search(
                best_a, direction, lo=0.0, hi=2.5, max_iter=16)

            if best_alpha_obj < best_obj - 1e-14 and best_alpha > 0.01:
                best_obj = best_alpha_obj
                best_a = best_a + best_alpha * direction
                best_a = np.maximum(best_a, 0)
                s_ba = best_a.sum()
                if s_ba > 1e-10:
                    best_a = best_a / s_ba
                no_improve = 0
            else:
                no_improve += 1

        if no_improve >= 3:
            scale = 0.008 + 0.005 * min(no_improve, 8)
            best_a = best_a * np.exp(rng.normal(0, scale, n))
            best_a = np.maximum(best_a, 0)
            s_ba = best_a.sum()
            if s_ba > 1e-10:
                best_a = best_a / s_ba
            no_improve = 0
            if it > 60:
                break

        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP-GS it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def tr_sscp_polish(a, time_budget=300.0, r_init=0.3, verbose=False, seed=42):
    """Trust region SOCP with golden-section line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    r = r_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        delta_dir, M_val = solve_tr_sscp(best_a, r)

        if delta_dir is None:
            g_lp, M_val = solve_lp_direction(best_a)
            if g_lp is None:
                break
            delta_dir = g_lp - best_a
        else:
            norm_d = np.linalg.norm(delta_dir)
            if norm_d > r:
                delta_dir = delta_dir * (r / norm_d)

        finite_mask = np.isfinite(delta_dir)
        if not finite_mask.all():
            delta_dir = np.where(finite_mask, delta_dir, 0.0)

        dir_norm = np.linalg.norm(delta_dir)
        if dir_norm < 1e-12:
            no_improve += 1
        else:
            best_alpha, best_alpha_obj = golden_section_line_search(
                best_a, delta_dir, lo=0.0, hi=2.0, max_iter=14)

            if best_alpha_obj < best_obj - 1e-14 and best_alpha > 0.02:
                best_obj = best_alpha_obj
                best_a = best_a + best_alpha * delta_dir
                best_a = np.maximum(best_a, 0)
                s_n = best_a.sum()
                if s_n > 1e-10:
                    best_a = best_a / s_n
                no_improve = 0
                r = min(r * 1.08, 2.0)
            else:
                no_improve += 1
                r *= 0.68

        if no_improve >= 4:
            if r < 0.04:
                best_a = best_a * np.exp(rng.normal(0, 0.012, n))
                best_a = np.maximum(best_a, 0)
                s_p = best_a.sum()
                if s_p > 1e-10:
                    best_a = best_a / s_p
                r = r_init
                no_improve = 0
                if it > 50:
                    break
            else:
                r = max(r * 0.5, 0.04)

        it += 1
        if verbose and it % 8 == 0:
            elapsed = time.monotonic() - start
            print(f"    TR it{it}: obj={best_obj:.8f}, r={r:.4f}, t={elapsed:.0f}s")

    return best_a


def coordinate_descent_refine(a, time_budget=60.0, seed=42):
    """Coordinate-wise line search for fine refinement."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    order = rng.permutation(n)
    it = 0

    while time.monotonic() - start < time_budget:
        improved = False
        for idx in order:
            idx = int(idx)
            base = best_a[idx]
            if base < 1e-10:
                candidates = [base * 10 + 0.001, base * 20 + 0.002]
            else:
                candidates = [base * f for f in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.4, 1.8]]
                candidates.append(0.0)

            best_local = base
            best_local_obj = best_obj

            for c in candidates:
                a_try = best_a.copy()
                a_try[idx] = c
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_local_obj - 1e-14:
                    best_local_obj = obj_try
                    best_local = c

            if best_local != base:
                best_a[idx] = best_local
                s_a = best_a.sum()
                if s_a > 1e-10:
                    best_a = best_a / s_a
                best_obj = best_local_obj
                improved = True

        it += 1
        if not improved and it > 2:
            break

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation, preserving non-negativity."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    result = np.maximum(np.interp(x_new, x_old, a), 0)
    s = result.sum()
    if s > 1e-10:
        result = result / s
    return result


def propose_candidate(seed=42, budget_s=1500):
    """
    Main multi-phase search: adaptive GD warmup, TR-SOCP, LP with golden section,
    coordinate descent, multi-length exploration, perturbation escapes, and final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Adaptive smoothed GD warmup ===
    print(f"\n=== Phase 1: Adaptive Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 80.0)
    if p1_budget > 15:
        a_sgd = adaptive_smoothed_gd(best_seq, time_budget=p1_budget,
                                     tau_init=20.0, tau_max=400.0, seed=42)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 GD: obj={obj_sgd:.8f}")

    # === Phase 2: TR-SOCP Polish ===
    print(f"\n=== Phase 2: TR-SOCP Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.28, 320.0)
    if p2_budget > 40:
        a_tr = tr_sscp_polish(best_seq, time_budget=p2_budget,
                              r_init=0.3, verbose=True, seed=42)
        obj_tr = compute_obj(a_tr)
        if obj_tr < best_score:
            best_score = obj_tr
            best_seq = a_tr.copy()
            print(f"  Phase2 TR: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 TR: obj={obj_tr:.8f}")

    # === Phase 3: LP with Golden Section (primary) ===
    print(f"\n=== Phase 3: LP + Golden Section ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.22, 220.0)
    if p3_budget > 30:
        a_lp = lp_golden_polish(best_seq, time_budget=p3_budget,
                                verbose=True, seed=77)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase3 LP-GS: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 LP-GS: obj={obj_lp:.8f}")

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.05, 50.0)
    if p4_budget > 12:
        a_cd = coordinate_descent_refine(best_seq, time_budget=p4_budget, seed=123)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase4 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 CD: obj={obj_cd:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_candidates = set()
    # Relative offsets
    for delta in [-200, -100, -50, 50, 100, 200, 300, 400, 500, 700]:
        nc = n_current + delta
        if 400 <= nc <= 2500:
            n_candidates.add(nc)
    # Absolute values
    for na in [500, 600, 700, 900, 1000, 1100, 1200, 1400, 1600, 1800, 2000]:
        if 400 <= na <= 2500 and na != n_current:
            n_candidates.add(na)

    n_list = sorted(n_candidates)
    # Prioritize: close to current first, then larger
    n_list.sort(key=lambda x: (abs(x - n_current) * 0.7 + max(0, x - n_current) * 0.3))

    explored = 0
    for n_try in n_list[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        explored += 1
        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")

        a_try = resample_seq(best_seq, n_try)

        # Quick GD
        alloc1 = min(remaining * 0.08, 20.0)
        if alloc1 > 8:
            a_try = adaptive_smoothed_gd(a_try, time_budget=alloc1,
                                         tau_init=10.0, tau_max=200.0, seed=100 + explored)

        # TR-SOCP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 60.0)
        if alloc2 > 12:
            a_try = tr_sscp_polish(a_try, time_budget=alloc2, r_init=0.25, seed=200 + explored)

        # Quick LP
        remaining = deadline - time.monotonic()
        alloc3 = min(remaining * 0.08, 30.0)
        if alloc3 > 10:
            a_try = lp_golden_polish(a_try, time_budget=alloc3, seed=300 + explored)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Perturbation + Polish ===
    print(f"\n=== Phase 6: Perturbation + Polish ===")
    remaining = deadline - time.monotonic()

    for r_idx in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        scale = 0.005 + 0.005 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            break
        a_pert = a_pert / s_pert

        # Alternate between TR and LP
        use_lp = (r_idx % 2 == 1)
        alloc = min(remaining * 0.20, 60.0)
        if alloc > 15:
            if use_lp:
                a_pert = lp_golden_polish(a_pert, time_budget=alloc, seed=400 + r_idx)
            else:
                a_pert = tr_sscp_polish(a_pert, time_budget=alloc,
                                        r_init=0.2, seed=400 + r_idx)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Pert{r_idx} (scale={scale:.4f}, {'LP' if use_lp else 'TR'}): obj={best_score:.8f} **")
            else:
                print(f"  Pert{r_idx} (scale={scale:.4f}, {'LP' if use_lp else 'TR'}): obj={obj_pert:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        # TR-SOCP
        alloc_tr = min(remaining * 0.35, 100.0)
        if alloc_tr > 20:
            a_final = tr_sscp_polish(best_seq, time_budget=alloc_tr,
                                     r_init=0.12, verbose=True, seed=999)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final TR: obj={best_score:.8f} **")

        # LP
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.30, 60.0)
        if alloc_lp > 15:
            a_flp = lp_golden_polish(best_seq, time_budget=alloc_lp, seed=2000)
            obj_flp = compute_obj(a_flp)
            if obj_flp < best_score:
                best_score = obj_flp
                best_seq = a_flp.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # Coordinate descent
        remaining = deadline - time.monotonic()
        alloc_cd = min(remaining * 0.25, 40.0)
        if alloc_cd > 12:
            a_cd = coordinate_descent_refine(best_seq, time_budget=alloc_cd, seed=5000)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
