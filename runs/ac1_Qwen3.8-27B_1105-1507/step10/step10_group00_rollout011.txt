def softmax_max_gradient(a, beta=50.0):

"""
Hybrid optimization for step function coefficients minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines vectorized mirror descent (exact subgradient), LP-based Frank-Wolfe polishing,
symmetry exploitation, smoothed multi-peak gradients, and adaptive length exploration.
Key improvements over parent: faster gradient eval, multi-peak smoothing for escape,
symmetry projection, and more aggressive LP time allocation.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def max_conv_gradient(a):
    """Vectorized subgradient of max_k(c_k) where c = conv(a,a).
    Returns (gradient, k_max, c_max)."""
    n = len(a)
    conv = compute_conv(a)
    k_max = int(np.argmax(conv))
    c_max = float(conv[k_max])
    grad = np.zeros(n)
    m_start = max(0, k_max - n + 1)
    m_end = min(n - 1, k_max)
    if m_start <= m_end:
        indices = np.arange(m_start, m_end + 1)
        a_indices = k_max - indices
        grad[indices] = 2.0 * a[a_indices]
    return grad, k_max, c_max


def multi_peak_gradient(a, top_k=3, decay=0.6):
    """Smoothed gradient using top-k peaks with exponential decay weighting."""
    n = len(a)
    conv = compute_conv(a)
    # Find top k distinct peaks (not too close together)
    flat_conv = np.sort(conv)[::-1]
    threshold = flat_conv[min(top_k - 1, len(flat_conv) - 1)] * 0.95
    peak_candidates = np.where(conv >= threshold)[0]
    # Select up to top_k spread-out peaks
    if len(peak_candidates) > top_k:
        # Greedy selection: pick highest, then farthest from existing
        selected = [int(np.argmax(conv))]
        conv_copy = conv.copy()
        for _ in range(top_k - 1):
            # Suppress region around already-selected peaks
            for s in selected:
                radius = max(2, n // 20)
                lo = max(0, s - radius)
                hi = min(len(conv_copy), s + radius + 1)
                conv_copy[lo:hi] = -np.inf
            next_peak = int(np.argmax(conv_copy))
            if conv_copy[next_peak] > 0:
                selected.append(next_peak)
    else:
        selected = list(peak_candidates)

    grad = np.zeros(n)
    for idx, k in enumerate(selected):
        w = decay ** idx
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            indices = np.arange(m_start, m_end + 1)
            a_indices = k - indices
            grad[indices] += w * 2.0 * a[a_indices]
    total_w = sum(decay ** i for i in range(len(selected)))
    grad /= total_w
    return grad


def mirror_descent(a, time_budget=80.0, verbose=False, use_multi_peak=False, top_k=3):
    """Mirror descent (exponential family) with exact subgradient on simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.08
    it = 0
    patience = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        if use_multi_peak:
            grad = multi_peak_gradient(a, top_k=top_k)
        else:
            grad, k_max, c_max = max_conv_gradient(a)

        # Mirror descent: a_m <- a_m * exp(-lr * grad_m)
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        lr *= 0.9992

            lr = 0.08

def symmetry_project(a):
    """Project to palindromic: a_i = a_{n-1-i}."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    s = float(np.sum(a_sym))
    if s > 1e-10:
        a_sym = a_sym / s
    return a_sym


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g_dir, time_budget=3.0):
    """Golden-section line search for optimal blend (1-t)*a + t*g_dir."""
    lo, hi = 0.0, 1.0

    def obj_at(t):
        a_try = (1 - t) * a + t * g_dir
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    start = time.monotonic()
    for _ in range(30):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)
    return t_best, obj_best


def lp_polish(a, time_budget=120.0, max_iters=60, verbose=False):
    """LP-based Frank-Wolfe polishing with line search and adaptive restarts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(99999)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=min(3.0, remaining * 0.04))

        if t_opt > 0 and obj_opt < best_obj - 1e-13:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                best_obj = obj_opt
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 5 == 0:
            print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, M_lp={M_lp:.6f}, elapsed={time.monotonic()-start:.1f}s")

        if stall >= 3:
            # Adaptive perturbation
            if stall <= 5:
                freq = rng.integers(1, max(2, n // 30))
                phase = rng.uniform(0, 2 * np.pi)
                x = np.arange(n, dtype=np.float64) / max(1, n - 1)
                pert = 0.015 * stall * np.cos(2 * np.pi * freq * x + phase)
                a = best_a * np.exp(pert)
            elif stall <= 8:
                noise = rng.normal(0, 0.02 * stall, n)
                a = best_a + noise
            else:
                a = best_a + rng.normal(0, 0.03, n)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 0.01:
                a = a / s
            else:
                a = best_a.copy()
            stall = 0

    return best_a


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.01, 0.99, n)
    a = 1.0 / np.sqrt(1 - (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4.0 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(4.0 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s


def try_length(a_best, n_try, time_budget):
    """Try a different sequence length via interpolation + optimization."""
    n_old = len(a_best)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        return None, np.inf
    a_try = a_try / s

    remaining = time_budget
    # Quick MD
    if remaining > 30:
        a_try = mirror_descent(a_try, time_budget=min(25.0, remaining * 0.25))
    # LP polish
    remaining = time_budget - (time.monotonic() - time.monotonic())  # approximate
    alloc = min(40.0, remaining * 0.5)
    if alloc > 10:
        a_try = lp_polish(a_try, time_budget=alloc, max_iters=25)

    obj_try = compute_obj(a_try)
    return a_try, obj_try


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining mirror descent, LP polishing,
    multi-peak smoothing, symmetry, and length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={n_current}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = u_shaped_init(n_current, 3.0)
        best_score = compute_obj(best_seq)
        print(f"Default U-shaped (n={n_current}): obj={best_score:.8f}")

    # Also try arcsine
    a_arc = arcsine_init(n_current)
    obj_arc = compute_obj(a_arc)
    if obj_arc < best_score:
        best_score = obj_arc
        best_seq = a_arc.copy()
        print(f"Arcsine init better: obj={best_score:.8f}")

    # === Phase 1: Mirror descent warm-up ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 1: Mirror Descent (n={n_current}) ===")
        alloc1 = min(80.0, remaining * 0.12)
        a_md1 = mirror_descent(best_seq, time_budget=alloc1, verbose=False)
        obj_md1 = compute_obj(a_md1)
        if obj_md1 < best_score:
            best_score = obj_md1
            best_seq = a_md1.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 MD: obj={obj_md1:.8f} (no improvement)")

    # === Phase 2: LP polishing (main) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 2: LP Polishing ===")
        alloc2 = min(remaining * 0.30, 350.0)
        a_lp = lp_polish(best_seq, time_budget=alloc2, max_iters=80, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Multi-peak mirror descent ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 3: Multi-Peak MD ===")
        alloc3 = min(60.0, remaining * 0.10)
        a_mp = mirror_descent(best_seq, time_budget=alloc3, use_multi_peak=True, top_k=4, verbose=False)
        obj_mp = compute_obj(a_mp)
        if obj_mp < best_score:
            best_score = obj_mp
            best_seq = a_mp.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj_mp:.8f} (no improvement)")

    # === Phase 4: LP polishing round 2 ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: LP Round 2 ===")
        # Small perturbation before
        noise = rng.normal(0, 0.003, len(best_seq))
        a_pert = best_seq + noise
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        alloc4 = min(remaining * 0.22, 250.0)
        a_lp2 = lp_polish(a_pert, time_budget=alloc4, max_iters=60, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase4 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 LP: obj={obj_lp2:.8f}")

    # === Phase 5: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print(f"\n=== Phase 5: n Exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 100, n_best - 200, n_best + 100, n_best + 200, n_best + 400,
            600, 700, 900, 1000, 1200, 1400
        ]))
        n_options = [n for n in n_options if 500 <= n <= 2000 and abs(n - n_best) >= 80]
        n_options = n_options[:5]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            a_try, obj_try = try_length(best_seq, n_try, remaining * 0.6)
            if a_try is not None and obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final intensive LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: Final Polish ===")
        alloc6 = min(remaining - 15, 180.0)
        if alloc6 > 20:
            a_final = lp_polish(best_seq, time_budget=alloc6, max_iters=50, verbose=True)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final: obj={best_score:.8f} **")
            else:
                print(f"  Final: obj={obj_final:.8f}")

    # === Phase 7: Quick final MD + LP ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        a_q = mirror_descent(best_seq, time_budget=min(15.0, remaining * 0.4))
        obj_q = compute_obj(a_q)
        if obj_q < best_score:
            best_score = obj_q
            best_seq = a_q.copy()
            print(f"  Quick MD: obj={best_score:.8f} **")
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_q2 = lp_polish(best_seq, time_budget=min(remaining - 5, 30.0), max_iters=15)
            obj_q2 = compute_obj(a_q2)
            if obj_q2 < best_score:
                best_score = obj_q2
                best_seq = a_q2.copy()
                print(f"  Quick LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score_final = val
    else:
        result = [1.0] * 100
        best_score_final = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score_final:.8f}, time={elapsed:.1f}s ===")
    return result
