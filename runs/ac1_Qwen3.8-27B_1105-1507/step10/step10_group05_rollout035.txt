KL mirror descent, coordinate descent polish, and multi-length search.
No LP solving - pure gradient methods for maximum throughput.
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time
import math

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_grad_fft(a, tau, N_cache=None):
    """
    Compute gradient of (1/T)*log(sum_k exp(T*conv(a,a)[k])) using FFT.
    Returns (gradient array of length n, probability vector p of length 2n-1, conv).
    """
    n = len(a)
    conv = compute_conv(a)
    
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # Compute gradient via FFT
    # grad[j] = sum_k p[k] * (a[k-j] + a[j-k]) with out-of-range = 0
    # = ifft(fft(p) * conj(fft(a))).real[:n] + ifft(fft(p) * fft(a)).real[:n]
    
    if N_cache is None or N_cache < 4 * n:
        N = 1
        while N < 4 * n:
            N <<= 1
    else:
        N = N_cache
    
    A = fft(a, N)
    P = fft(p, N)
    
    # Cross-correlation term: sum_k p[k] * a[k-j]
    term1 = ifft(P * np.conj(A)).real[:n]
    # Convolution term: sum_k p[k] * a[j-k]  
    term2 = ifft(P * A).real[:n]
    
    grad = term1 + term2
    return grad, p, conv

def nesterov_smoothed_descent(a, time_budget, tau_init=50.0, tau_end=500.0, verbose=False, use_kl=True):
    """
    Nesterov-accelerated smoothed subgradient descent on the simplex.
    Uses KL mirror descent when use_kl=True, Euclidean projection otherwise.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    mu = 0.9  # momentum
    lr = 1.0 / np.sqrt(n)  # initial learning rate
    v = np.zeros(n)  # momentum velocity
    y = a.copy()  # Nesterov lookahead
    
    it = 0
    best_it = 0
    N = 1
    while N < 4 * n:
        N <<= 1
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Cosine annealing for tau
        tau = tau_init + (tau_end - tau_init) * (1 - math.cos(math.pi * progress)) / 2
        
        # Nesterov lookahead
        y = a + mu * v
        if use_kl:
            # For KL mirror descent, the lookahead is in log-space
            # y = a * exp(mu * log(v_component))  -- actually let's keep it simple
            y = np.maximum(y, 1e-15)
            y = y / y.sum()
        else:
            y = project_simplex(y)
        
        # Compute smoothed gradient
        grad, p, conv = smoothed_grad_fft(y, tau, N)
        grad = 2.0 * n * grad  # Scale by 2n for the objective
        
        # Update
        if use_kl:
            # KL mirror descent: a_new = a * exp(-lr * grad) / Z
            log_update = -lr * grad
            # Clip for numerical stability
            log_update = np.clip(log_update, -50, 50)
            a_new = a * np.exp(log_update)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
            else:
                a_new = a_new / s_new
            a_new = np.maximum(a_new, 0)
        else:
            # Euclidean: a_new = a - lr * grad, then project
            a_new = a - lr * grad
            a_new = project_simplex(a_new)
        
        # Check objective
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        # Update momentum
        v = mu * v + (a - a_new)
        a = a_new
        
        # Adaptive learning rate (reduce if oscillating)
        if it > 0 and it % 1000 == 0:
            lr = lr * 0.95
        
        it += 1
        
        if verbose and it % 5000 == 0:
            print(f"      NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.1f}s")
        
        # Early stopping on stagnation
        if it - best_it > 50000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = lr * 0.5
            tau = min(tau * 1.5, tau_end)
            best_it = it
    
    return best_a

def coordinate_descent(a, time_budget, verbose=False):

                c = current + delta * (1.0 / n)
                if c > 0:
                    candidates.append(c)
            
            best_val = current
            best_local_obj = best_obj  # This isn't quite right, we need to recompute
            
            # Actually, let's just try a binary search approach
            # Try 5 values around current
            for factor in [0.7, 0.85, 1.0, 1.15, 1.3]:
                new_val = current * factor
                a[j] = new_val
                a = a / a.sum()  # Renormalize
                obj = compute_obj(a)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = current

            a[j] = current

        it += 1

def subgradient_active_set(a, time_budget, verbose=False):
    """
    Subgradient method using the active set of max-achieving convolution entries.
    More accurate than smoothing but noisier.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 4 * n:
        N <<= 1
    
    step_num = 0
    avg_grad = np.zeros(n)
    window = 50  # averaging window
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        # Active set: entries within 1e-6 of max
        active = np.where(conv >= max_c - 1e-8)[0]
        
        # Uniform distribution over active set
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / len(active)
        
        # Compute subgradient
        A = fft(a, N)
        P = fft(p, N)
        term1 = ifft(P * np.conj(A)).real[:n]
        term2 = ifft(P * A).real[:n]
        grad = 2.0 * n * (term1 + term2)
        
        # Averaging
        avg_grad = avg_grad * (1 - 1/window) + grad / window
        
        # Learning rate: O(1/sqrt(k))
        step_size = 5.0 / np.sqrt(step_num + 1) / np.sqrt(n)
        
        a_new = a - step_size * avg_grad
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        step_num += 1
        
        if verbose and step_num % 10000 == 0:
            print(f"      SAD it{step_num}: obj={best_obj:.8f}, |active|={len(active)}, t={time.monotonic()-start:.1f}s")
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast FFT-based smoothed subgradient with Nesterov momentum, KL mirror descent,
adaptive tau annealing, coordinate descent polish, and multi-length exploration.
Key innovation: correct O(n log n) gradient via cross-correlation identity,
enabling millions of iterations vs. the parent's LP-based approach.
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gradient_obj(a, tau, N_cache=None):
    """
    Compute smoothed gradient and true objective for F(a) = 2n*max(conv(a,a))
    assuming sum(a)=1. Uses cross-correlation identity:
    grad[j] = 4n * sum_k p[k] * a[k-j] = 4n * (p cross-corr a)[j]
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    obj = 2.0 * n * max_c

    # Smoothed probabilities
    exp_vals = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    # FFT-based cross-correlation
    if N_cache is None:
        N = 1
        while N < 4 * n:
            N <<= 1
    else:
        N = N_cache

    A = fft(a, N)
    P = fft(p, N)
    grad = 4.0 * n * ifft(P * np.conj(A)).real[:n]

    return grad, obj, conv


def nesterov_descent(a, time_budget, tau_init=30.0, tau_end=800.0,
                     verbose=False, use_kl=True, lr_scale=1.0, seed=42):
    """
    Nesterov-accelerated smoothed subgradient on the simplex.
    use_kl=True: multiplicative (KL mirror) updates.
    use_kl=False: additive (Euclidean projected) updates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 4 * n:
        N <<= 1

    mu = 0.75  # momentum
    lr = lr_scale * 1.0 / math.sqrt(n)
    v = np.zeros(n)
    y = a.copy()

    it = 0
    best_it = 0
    eval_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Smoothstep for tau (faster initial increase, slower later)
        tau = tau_init + (tau_end - tau_init) * (3 * progress * progress - 2 * progress * progress * progress)

        # Nesterov lookahead
        y = a + mu * v
        if use_kl:
            y = np.maximum(y, 1e-15)
            y = y / y.sum()
        else:
            y = project_simplex(y)

        # Gradient at y
        grad, smooth_obj, conv = gradient_obj(y, tau, N)

        # Update
        if use_kl:
            log_step = -lr * grad
            log_step = np.clip(log_step, -40, 40)
            a_new = a * np.exp(log_step)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
            else:
                a_new = a_new / s_new
        else:
            a_new = y - lr * grad
            a_new = project_simplex(a_new)

        # Evaluate periodically (every 10 steps or at the end)
        if it % 10 == 0 or it < 5:
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it
            eval_count += 1

        # Momentum update
        v = mu * v + (a - a_new)
        a = a_new.copy()

        # Adaptive learning rate
        if it > 0 and it % 2000 == 0:
            lr *= 0.92

        it += 1

        # Stagnation restart
        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = lr_scale * 0.5 / math.sqrt(n)
            best_it = it

        if verbose and it % 10000 == 0:
            cur_obj = compute_obj(a)
            print(f"      ND it{it}: smooth_obj={smooth_obj:.6f}, best={best_obj:.8f}, "
                  f"tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.1f}s")

    # Final evaluation
    final_obj = compute_obj(a)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()

    return best_a


def active_set_subgradient(a, time_budget, verbose=False):
    """
    Subgradient method using uniform distribution over active (max-achieving) entries.
    This is equivalent to tau->infinity smoothing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 4 * n:
        N <<= 1

    step_num = 0
    avg_grad = np.zeros(n)
    window = 30

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        active = np.where(conv >= max_c - 1e-10)[0]

        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / len(active)

        A = fft(a, N)
        P = fft(p, N)
        grad = 4.0 * n * ifft(P * np.conj(A)).real[:n]

        # Smoothing via averaging
        avg_grad = avg_grad * (1.0 - 1.0 / window) + grad / window

        # Polyak step size
        step_size = 8.0 / (math.sqrt(step_num + 1.0) * math.sqrt(n))

        a_new = a - step_size * avg_grad
        a_new = project_simplex(a_new)

        if step_num % 50 == 0:
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()

        a = a_new
        step_num += 1

        if verbose and step_num % 20000 == 0:
            cur_obj = compute_obj(a)
            print(f"      ASD it{step_num}: obj={cur_obj:.8f}, best={best_obj:.8f}, "
                  f"|active|={len(active)}, t={time.monotonic()-start:.1f}s")

    return best_a


def coordinate_refine(a, time_budget, verbose=False):
    """
    Coordinate-wise refinement: for each j, try scaling a[j] by factors
    and keep the best. Cyclic until no improvement or time runs out.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    passes = 0

    factors = [0.7, 0.85, 1.15, 1.3, 0.5, 1.5]

    while time.monotonic() - start < time_budget:
        improved = False
        for j in range(n):
            orig = a[j]
            if orig < 1e-12:
                # Try small positive values
                for new_val in [0.01 / n, 0.02 / n]:
                    a[j] = new_val
                    a = a / a.sum()
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
                a[j] = orig
                a = a / a.sum()
            else:
                for f in factors:
                    a[j] = orig * f
                    a = a / a.sum()
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        # Try to fine-tune around the new value
                        new_orig = a[j]
                        for f2 in [0.9, 1.1]:
                            a[j] = new_orig * f2
                            a = a / a.sum()
                            obj2 = compute_obj(a)
                            if obj2 < best_obj - 1e-12:
                                best_obj = obj2
                                best_a = a.copy()
                                improved = True
                        break
                a[j] = orig
                a = a / a.sum()

        passes += 1
        if verbose and passes % 2 == 0:
            print(f"      CD pass {passes}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, seed, rng):
    """Generate diverse structured initializations."""
    x = np.arange(n)
    choice = seed % 7
    if choice == 0:
        a = np.ones(n) / n
    elif choice == 1:
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x / n)
    elif choice == 2:
        a = 1.0 + 0.5 * np.cos(4 * np.pi * x / n)
    elif choice == 3:
        # Triangle
        center = n // 2
        a = 1.0 - 0.5 * np.abs(x - center) / center
        a = np.maximum(a, 0.1)
    elif choice == 4:
        # Two peaks
        a = 1.0 + 0.8 * np.cos(2 * np.pi * x * 3.0 / n)
        a = np.maximum(a, 0.05)
    elif choice == 5:
        # Exponential taper
        mid = n / 2.0
        a = np.exp(-0.5 * ((x - mid) / (n / 4.0)) ** 2)
        a = a + 0.1 * np.ones(n)
    else:
        a = np.ones(n) + 0.3 * rng.standard_normal(n)
        a = np.maximum(a, 0.05)
    return a / a.sum()

        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj1:.8f} (no improvement)")

    # === Phase 2: Euclidean Nesterov (high tau, high precision) ===
    print(f"\n=== Phase 2: Euclidean Nesterov ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.20, 300.0)
    if p2_budget > 30:
        a2 = nesterov_descent(best_seq, time_budget=p2_budget,
                              tau_init=100.0, tau_end=3000.0,
                              verbose=True, use_kl=False, lr_scale=0.8, seed=seed + 1)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj2:.8f} (no improvement)")

    # === Phase 3: KL with different hyperparams ===
    print(f"\n=== Phase 3: KL (different params) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)
    if p3_budget > 20:
        a3 = nesterov_descent(best_seq, time_budget=p3_budget,
                              tau_init=50.0, tau_end=100.0,
                              verbose=True, use_kl=True, lr_scale=2.0, seed=seed + 2)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj3:.8f} (no improvement)")

    # === Phase 4: Active-set subgradient ===
    print(f"\n=== Phase 4: Active-Set Subgradient ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.10, 150.0)
    if p4_budget > 20:
        a4 = active_set_subgradient(best_seq, time_budget=p4_budget, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase 4: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4: obj={obj4:.8f} (no improvement)")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.08, 80.0)
    if p5_budget > 15:
        a5 = coordinate_refine(best_seq, time_budget=p5_budget, verbose=True)
        obj5 = compute_obj(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Phase 5: obj={best_score:.8f} **")
        else:
            print(f"  Phase 5: obj={obj5:.8f} (no improvement)")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_candidates = []
    for dn in [-150, -100, -50, 50, 100, 150, 200, 300]:
        n_try = n_best + dn
        if 500 <= n_try <= 1500:
            n_candidates.append(n_try)
    # Add some absolute values
    for n_abs in [600, 700, 900, 1000, 1200]:
        if n_abs not in n_candidates and abs(n_abs - n_best) > 40:
            n_candidates.append(n_abs)

    n_candidates = sorted(set(n_candidates))
    # Limit to 5 most promising (closest to n_best first)
    n_candidates.sort(key=lambda x: abs(x - n_best))
    n_candidates = n_candidates[:5]
    print(f"  Trying n values: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        alloc = min(remaining * 0.10, 60.0)
        if alloc > 20:
            a_try = nesterov_descent(a_try, time_budget=alloc,
                                     tau_init=50.0, tau_end=500.0,
                                     use_kl=True, lr_scale=1.0, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diversification restarts ===
    print(f"\n=== Phase 7: Diversification ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    rng2 = np.random.default_rng(seed + 100)

    restarts = [
        ("cos1", structured_init(n_current, 1, rng2)),
        ("cos3", structured_init(n_current, 4, rng2)),
        ("mix", 0.6 * best_seq + 0.4 * structured_init(n_current, 0, rng2)),
        ("pert", best_seq * (1 + 0.05 * rng2.standard_normal(n_current))),
    ]
    restarts = [(name, np.maximum(a, 0)) for name, a in restarts]
    restarts = [(name, a / a.sum()) for name, a in restarts]

    for name, a_r in restarts:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        alloc = min(remaining * 0.08, 50.0)
        if alloc > 15:
            a_r = nesterov_descent(a_r, time_budget=alloc,
                                   tau_init=30.0, tau_end=300.0,
                                   use_kl=True, lr_scale=1.0, seed=seed + 200)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart ({name}): obj={best_score:.8f} **")
            else:
                print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau KL (near-exact max)
    if remaining > 50:
        alloc = min(remaining * 0.35, 180.0)
        a_f = nesterov_descent(best_seq, time_budget=alloc,
                               tau_init=500.0, tau_end=5000.0,
                               verbose=True, use_kl=True, lr_scale=0.5, seed=9999)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final KL high-tau: obj={best_score:.8f} **")
        else:
            print(f"  Final KL high-tau: obj={obj_f:.8f}")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 60.0)
        a_f = coordinate_refine(best_seq, time_budget=alloc)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_f:.8f}")

    # One more KL pass
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 80.0)
        a_f = nesterov_descent(best_seq, time_budget=alloc,
                               tau_init=1000.0, tau_end=8000.0,
                               use_kl=True, lr_scale=0.3, seed=7777)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Ultra-final KL: obj={best_score:.8f} **")
        else:
            print(f"  Ultra-final KL: obj={obj_f:.8f}")

    # Final coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_f = coordinate_refine(best_seq, time_budget=alloc)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Ultra-final CD: obj={best_score:.8f} **")

    # Validation and cleanup
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure all values are in valid range
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
