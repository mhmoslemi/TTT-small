def init_cosine(n, k=1):
    """Initialize with cosine structure."""
    x = np.arange(n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * k * x / n)
    a = np.maximum(a, 0.001)
    return a / a.sum()


def init_exponential(n, rate=0.5):
    """Initialize with exponential decay."""
    x = np.arange(n)
    a = np.exp(-rate * x / n)
    return a / a.sum()


def init_bimodal(n):
    """Initialize with two peaks."""
    x = np.arange(n)
    center1 = n * 0.25
    center2 = n * 0.75
    width = n * 0.1
    a = np.exp(-((x - center1) / width) ** 2) + np.exp(-((x - center2) / width) ** 2)
    a = np.maximum(a, 0.001)
    return a / a.sum()


def init_random_structured(n, rng, alpha=0.3):
    """Initialize with random + smooth structure."""
    a_raw = rng.standard_normal(n)
    # Smooth it
    kernel_size = max(3, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    a_smooth = np.convolve(a_raw, kernel, mode='same')
    a = alpha * a_smooth + (1 - alpha)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def mirror_symmetrize(a):
    """Make sequence mirror-symmetric."""
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-10:
        return a.copy()
    return a_sym / s
