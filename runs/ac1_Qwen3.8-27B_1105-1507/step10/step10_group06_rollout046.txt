def compute_conv(a):

    n = len(a)
    if n == 0:
        return np.array([])
    size = 1

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

def simplex_project(a):
    """Project a onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-300:
        n = len(a)
        return np.ones(n) / n
    return a / s

def resample_to_length(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n + 1)[:-1]
    x_new = np.linspace(0, 1, new_n + 1)[:-1]
    a_new = np.interp(x_new, x_old, a, period=None)
    return np.maximum(a_new, 0)
