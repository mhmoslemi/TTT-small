import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = 0
        k_end = min(2 * n - 1, j + n)
        for k in range(k_start, k_end):
            A[k, j] = a[k - j]
    return A


def solve_lp(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.SCS, cp.PDLP, cp.MOSEK]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)

    return None, None

def coordinate_descent(a, time_budget=200.0, verbose=False, seed=42):

    if S < 1e-10:

    best_a = a.copy()

                if fx < best_f:

                    best_x = x

            # Golden section refinement
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(30):
                if x_hi - x_lo < 1e-12 * max(1.0, abs(x_hi)):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-13:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse asymmetric initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        # Linear ramp (increasing)
        a = np.arange(1, n + 1, dtype=float)
        a = a / a.sum()
    elif mode == 2:
        # Linear ramp (decreasing)
        a = np.arange(n, 0, -1, dtype=float)
        a = a / a.sum()
    elif mode == 3:
        # Quadratic (convex)
        x = np.linspace(0, 1, n)
        a = x ** 2 + 0.01
        a = a / a.sum()
    elif mode == 4:
        # Quadratic (concave)
        x = np.linspace(0, 1, n)
        a = np.sqrt(x + 0.01)
        a = a / a.sum()
    elif mode == 5:
        # U-shape
        x = np.linspace(0, 1, n)
        a = (x - 0.5) ** 2 + 0.01
        a = a / a.sum()
    elif mode == 6:
        # Random
        a = rng.exponential(1, n)
        a = a / a.sum()
    elif mode == 7:
        # Sine-modulated
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * 3 * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 8:
        # Exponential decay
        a = np.exp(-np.linspace(0, 3, n))
        a = a / a.sum()
    elif mode == 9:
        # Two-sided exponential
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.3)) + np.exp(-4 * (x - 0.7)) + 0.01
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining LP iteration, coordinate descent, multi-n exploration,
    and diverse asymmetric initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_default = 800
        best_seq = diverse_init(n_default, rng, mode=1)
        best_score = compute_obj(best_seq)
        print(f"Default ramp init (n={n_default}): obj={best_score:.8f}")

    # === Phase 1: Try diverse initializations with short LP ===
    print(f"\n=== Phase 1: Diverse Inits with Short LP ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 200.0)
    n_best = len(best_seq)

    modes_to_try = [1, 2, 3, 4, 7, 8, 9]
    for mode in modes_to_try:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        a_try = diverse_init(n_best, rng, mode=mode)
        # Short LP to see if this direction is promising
        alloc = min(remaining * 0.1, 25.0)
        if alloc > 10:
            a_try = lp_iteration(a_try, time_budget=alloc, verbose=False, seed=seed + mode)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  Init mode {mode}: obj={best_score:.8f} **")
        else:
            print(f"  Init mode {mode}: obj={obj_try:.8f}")

    # === Phase 2: Extended LP Iteration (main workhorse) ===
    print(f"\n=== Phase 2: Extended LP Iteration ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 500.0)
    if p2_budget > 30:
        a_lp = lp_iteration(best_seq, time_budget=p2_budget, verbose=True, seed=seed)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 3: Coordinate Descent Refinement ===
    print(f"\n=== Phase 3: CD Refinement ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 300.0)
    if p3_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p3_budget,
                                  verbose=True, seed=seed)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 4: More LP after CD ===
    print(f"\n=== Phase 4: LP after CD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 200.0)
    if p4_budget > 20:
        a_lp2 = lp_iteration(best_seq, time_budget=p4_budget, verbose=False, seed=seed + 7)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP2: obj={best_score:.8f} **")
        else:
            print(f"  LP2: obj={obj_lp2:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(200, n_current - 200),
        n_current + 200,
        n_current + 400,
        500, 600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000]
    # Prioritize n values near current best
    n_options.sort(key=lambda x: abs(x - n_current))
    n_options = n_options[:4]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try == n_current:
            continue

        a_try = resample_sequence(best_seq, n_try)
        # LP on new length
        alloc = min(remaining * 0.2, 60.0)
        if alloc > 15:
            a_try = lp_iteration(a_try, time_budget=alloc, verbose=False, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final CD Polish ===
    print(f"\n=== Phase 6: Final CD Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 200.0)
        a_final = coordinate_descent(best_seq, time_budget=alloc,
                                     verbose=False, seed=seed + 99)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # === Phase 7: Final LP ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_lp_final = lp_iteration(best_seq, time_budget=min(remaining * 0.5, 60.0),
                                  verbose=False, seed=seed + 77)
        obj_lp_final = compute_obj(a_lp_final)
        if obj_lp_final < best_score:
            best_score = obj_lp_final
            best_seq = a_lp_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_lp_final:.8f}")

    # Final validation via evaluate_sequence
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
        print(f"  evaluate_sequence confirms: {val:.8f}")
    elif val == np.inf:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
