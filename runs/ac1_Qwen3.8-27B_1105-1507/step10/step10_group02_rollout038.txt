def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def fft_cross_correlation(w, a):
    """Compute R[i] = sum_k w[k] * a[k-i] for i=0..n-1 via FFT.
    w has length 2n-1, a has length n."""
    n = len(a)
    N = 1
    while N < len(w) + n:
        N <<= 1
    W = rfft(w, N)
    A = rfft(a, N)
    # R[i] = sum_k w[k] a[k-i] = ifft(conj(FFT(a)) * FFT(w))[i]
    result = irfft(np.conj(A) * W, N)[:n]
    return result

        else:
            a_new = a_new / s_new
        momentum = a_new - y

        # Evaluate
        conv_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(conv_new))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 200:
                # Perturb to escape
                rng = np.random.default_rng(int(start * 1000) + it)
                pert = rng.normal(0, 0.02, n)
                a_new = best_a * np.exp(pert)
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                momentum = np.zeros(n)
                stall_count = 0
                beta = 0.7  # Reduce momentum temporarily

        a = a_new
        it += 1

        if verbose and it % 5000 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}, gnorm={g_norm:.4f}")

    return best_a


def multiplicative_sgd(a, time_budget=300.0, tau_init=15.0, tau_max=400.0,
                       verbose=True, seed=42):
    """Multiplicative (exponential) smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    tau = tau_init
    it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1.0))
        tau = tau_init + (tau_max - tau_init) * progress ** 0.6

        grad, conv = smoothed_gradient(a, tau)
        g_norm = float(np.linalg.norm(grad))

        if g_norm > 1e-15:
            lr = min(lr * 0.9998 + 0.0002 * 2.0 / (np.sqrt(n)), 5.0 / np.sqrt(n))
            lr = max(lr, 0.005 / np.sqrt(n))

        # Multiplicative update: a[i] <- a[i] * exp(-lr * grad[i])
        log_update = -lr * grad
        log_update = np.clip(log_update, -50, 50)  # Prevent overflow
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 300:
                rng = np.random.default_rng(12345 + it)
                pert = rng.normal(0, 0.03, n)
                a_new = best_a * np.exp(pert)
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                stall_count = 0
                lr = lr * 0.5

        a = a_new
        it += 1

        if verbose and it % 5000 == 0:
            print(f"    MulSGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None


def lp_polish(a, time_budget=60.0, verbose=False, seed=42):
    """LP-based polishing: find g minimizing max(a*g), blend with current."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 3:
                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if time.monotonic() - start > time_budget * 0.85:
                    break

    return best_a


def f1d_subeval(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i):
    """Evaluate 1D subproblem objective."""
    if len(a_ki_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom


def coordinate_descent(a, time_budget=300.0, verbose=True, seed=42):
    """
    Exact coordinate descent with golden section search.
    Smart coordinate ordering based on gradient magnitude.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1

        # Periodically recompute conv
        if sweep % 3 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        # Smart ordering: sort by "importance" (how much each coord affects max)
        if sweep % 5 == 1:
            # Compute approximate gradient magnitude for ordering
            max_conv = float(np.max(conv))
            # Coords that are large and affect the peak lags are important
            importance = np.zeros(n)
            active_ks = np.where(conv > 0.95 * max_conv)[0]
            for k in active_ks[:20]:  # Limit for speed
                for i in range(n):
                    ki = k - i
                    if 0 <= ki < n:
                        importance[i] += a[ki]
            order = np.argsort(-importance + rng.uniform(0, 0.1, n))
        else:
            order = rng.permutation(n)

        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v

            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)

            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks]

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Quick scan to find good bracket
            scale = max(v, S / n, 1e-6)
            candidates = [0.0, v * 0.3, v * 0.5, v * 0.7, v,
                          v * 1.3, v * 1.5, v * 2.0, v * 3.0]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f1d_subeval(v, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            for x in candidates:
                fx = f1d_subeval(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f1d_subeval(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            fd = f1d_subeval(d_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)

            for _ in range(40):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f1d_subeval(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f1d_subeval(d_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def block_coordinate_descent(a, time_budget=120.0, block_size=2, verbose=True, seed=42):
    """Block coordinate descent: optimize pairs of coordinates jointly."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0

    while time.monotonic() - start < time_budget:
        # Pick a random pair (i, j)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        v_i, v_j = a[i], a[j]
        S_other = S - v_i - v_j

        if S_other < 1e-12:
            continue

        # Build the 2D subproblem
        # conv[k] changes as: for each k, the contribution involving a[i] and a[j]
        # This is complex; use a simple 2D grid search + refinement

        # Quick grid search
        best_xi, best_xj = v_i, v_j
        best_f = 2.0 * n * float(np.max(conv)) / (S * S)

        scale_i = max(v_i, S / n, 1e-6)
        scale_j = max(v_j, S / n, 1e-6)

        # 5x5 grid
        for di in [-1, -0.5, 0, 0.5, 1]:
            for dj in [-1, -0.5, 0, 0.5, 1]:
                xi = max(0.0, v_i + di * scale_i)
                xj = max(0.0, v_j + dj * scale_j)
                # Compute new conv (approximate: only update affected entries)
                new_conv = conv.copy()
                # Remove old contributions
                for k in range(2 * n - 1):
                    contrib = 0.0
                    if 0 <= k - i < n:
                        contrib += v_i * a[k - i] if k - i != j else v_i * v_j
                    if 0 <= i - k + 2 * i < n and i != j:  # This is getting complex
                        pass
                    # This is too slow for direct computation; skip and use evaluate
                # Just evaluate directly (expensive but correct)
                a_try = a.copy()
                a_try[i] = xi
                a_try[j] = xj
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                conv_try = compute_conv(a_try / s_try)
                f_try = 2.0 * n * float(np.max(conv_try))
                if f_try < best_f:
                    best_f = f_try
                    best_xi, best_xj = xi, xj

        if best_xi != v_i or best_xj != v_j:
            # Refine with local search
            for _ in range(5):
                improved = False
                for step in [0.3, 0.1, 0.03]:
                    for di in [-1, 0, 1]:
                        for dj in [-1, 0, 1]:
                            if di == 0 and dj == 0:
                                continue
                            xi = max(0.0, best_xi + di * step * scale_i)
                            xj = max(0.0, best_xj + dj * step * scale_j)
                            a_try = a.copy()
                            a_try[i] = xi
                            a_try[j] = xj
                            s_try = float(np.sum(a_try))
                            if s_try < 1e-10:
                                continue
                            conv_try = compute_conv(a_try / s_try)
                            f_try = 2.0 * n * float(np.max(conv_try))
                            if f_try < best_f - 1e-15:
                                best_f = f_try
                                best_xi, best_xj = xi, xj
                                improved = True
                    if improved:
                        break

            # Apply update
            a[i] = best_xi
            a[j] = best_xj
            conv = compute_conv(a)
            S = float(np.sum(a))
            obj = 2.0 * n * float(np.max(conv)) / (S * S)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    BCD it{it}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def generate_msequence_init(n, rng, seed=0):
    """Generate an m-sequence-inspired initialization."""
    # Use a binary sequence with good correlation properties
    # Approximate: use a pseudo-random binary sequence with balanced 0s and 1s
    rng_ms = np.random.default_rng(seed)

    # Try to create a sequence with near-flat autocorrelation
    # Method: iterative balancing
    b = rng_ms.integers(0, 2, n).astype(float)

    # Iteratively flip bits to reduce peak autocorrelation
    for _ in range(50):
        # Compute autocorrelation
        ac = np.correlate(b, b, mode='full')[n-1:]  # lags 0 to n-1
        peak = np.max(ac)
        mean_ac = np.mean(ac)
        # Find the lag with maximum deviation
        worst_lag = np.argmax(np.abs(ac - mean_ac))

        # Try flipping random bits to reduce peak
        best_flip = -1
        best_val = peak
        for trial in range(min(50, n)):
            idx = rng_ms.integers(0, n)
            b_trial = b.copy()
            b_trial[idx] = 1 - b_trial[idx]
            ac_trial = np.correlate(b_trial, b_trial, mode='full')[n-1:]
            val = np.max(ac_trial)
            if val < best_val:
                best_val = val
                best_flip = idx
        if best_flip >= 0:
            b[best_flip] = 1 - b[best_flip]

    # Convert to non-negative sequence
    a = b / b.sum()
    return a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussians
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        # Sinusoidal modulation
        freqs = rng.integers(1, 10, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # M-sequence inspired
        a = generate_msequence_init(n, rng, seed=mode * 100)
    elif mode == 7:
        # Alternating pattern with noise
        a = np.ones(n)
        a[::2] = 1.0
        a[1::2] = 0.3
        a += rng.normal(0, 0.05, n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 8:
        # Chirp-like
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * (5 * x**2 + 0.5))
        a += rng.normal(0, 0.03, n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            print(f"  Nesterov GD: obj={obj_sgd:.8f}")

    # === Phase 2: Multiplicative SGD ===
    print(f"\n=== Phase 2: Multiplicative SGD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 200.0)
    if p2_budget > 30:
        # Start from slightly perturbed version
        pert = rng.normal(0, 0.01, len(best_seq))
        a_start = best_seq * np.exp(pert)
        a_start = np.maximum(a_start, 0)
        a_start = a_start / a_start.sum()

        a_mul = multiplicative_sgd(a_start, time_budget=p2_budget,
                                    tau_init=12.0, tau_max=350.0,
                                    verbose=True, seed=seed + 1)
        obj_mul = 2.0 * a_mul.size * float(np.max(compute_conv(a_mul / a_mul.sum())))
        if obj_mul < best_score:
            best_score = obj_mul
            best_seq = a_mul.copy()
            print(f"  Mul SGD: obj={best_score:.8f} **")
        else:
            print(f"  Mul SGD: obj={obj_mul:.8f}")

    # === Phase 3: Exact Coordinate Descent ===
    print(f"\n=== Phase 3: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.3, 350.0)

            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")
