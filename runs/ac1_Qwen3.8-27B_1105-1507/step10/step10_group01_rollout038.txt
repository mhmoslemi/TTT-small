Minimize `M` s.t. `conv(a,a)[k] <= M` for all k, `||a - a_old||^2 <= delta^2`, `a >= 0`

The constraints `conv(a,a)[k] <= M` are quadratic (non-convex in general due to the product of variables). However, if we linearize around `a_old`:
`conv(a,a)[k] ≈ 2*conv(a_old, a)[k] - conv(a_old, a_old)[k]`

Then the problem becomes:
Minimize `M` s.t. `2*conv(a_old, a)[k] - conv(a_old, a_old)[k] <= M`, `||a - a_old||^2 <= delta^2`, `a >= 0`

The first constraint is linear in `a` (since `a_old` is fixed), and the second is a ball constraint. This can be solved as a QP!

This is essentially an SQP (Sequential Quadratic Programming) approach. Let me implement this.

Actually, for `cvxpy`, a ball constraint `||a - a_old||^2 <= delta^2` can be expressed as `cp.SOC(delta, a - a_old)` (second-order cone constraint). And the linear constraints are straightforward. So this is a SOCP, which CVXPY can handle efficiently.

```python
def sqp_step(a, delta, solver=cp.MOSEK):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    a_norm = a / s  # work with normalized
    
    # Precompute conv(a_norm, a_norm)
    conv_aa = compute_conv(a_norm)
    
    # Build linearized constraint matrix: 2*conv(a_norm, g)[k] - conv(a_norm, a_norm)[k] <= M
    # conv(a_norm, g)[k] = sum_j a_norm[j] * g[k-j]
    # So the constraint is: 2 * sum_j a_norm[j] * g[k-j] - conv_aa[k] <= M
    
    # Build A matrix: A[k, j] = 2 * a_norm[k-j] if 0 <= k-j < n, else 0
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = 2.0 * a_norm[diff[valid]]
    
    b = conv_aa  # the constant term
    
    # Solve: min M s.t. A@g - M <= b, ||g - a_norm||_2 <= delta, g >= 0, sum(g) = 1
    g = cp.Variable(n)
    M = cp.Variable()
    constraints = [
        A @ g - M <= b,
        cp.sum(g) == 1,
        g >= 0,
        cp.norm(g - a_norm, 2) <= delta
    ]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    prob.solve(solver=solver)
    if g.value is not None:
        g_val = np.maximum(g.value, 0)
        s_val = np.sum(g_val)
        if s_val > 1e-10:
            return g_val / s_val, float(M.value)
    return None, None
```

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses SOCP-based trust region (SQP) as the main optimization engine,
combined with high-temperature smoothed gradient descent for polishing,
and multi-length exploration for diversity.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A

def socp_step(a_old, delta):
    """
    Solve trust region SOCP:
    min M s.t. 2*conv(a_old, g)[k] - conv(a_old, a_old)[k] <= M  (all k)
               ||g - a_old||_2 <= delta
               g >= 0, sum(g) = 1
    Returns (g, M) or (None, None).
    """
    n = len(a_old)
    a_old = np.maximum(a_old, 0).astype(np.float64)
    s = float(np.sum(a_old))
    if s < 1e-10:
        return None, None
    a_norm = a_old / s
    
    # Build constraint matrix
    A = build_conv_matrix(a_norm)
    conv_aa = compute_conv(a_norm)
    
    g = cp.Variable(n)
    M = cp.Variable()
    constraints = [
        2.0 * (A @ g) - M <= conv_aa,
        cp.sum(g) == 1,
        g >= 0,
        cp.norm(g - a_norm, 2) <= delta
    ]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val, float(M.value)
        except Exception:
            continue
    return None, None

def sqp_optimize(a, time_budget=600.0, verbose=True):
    """
    SQP (Sequential Quadratic Programming) optimization via SOCP trust region.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    delta = 0.05
    min_delta = 1e-5
    iter_count = 0
    no_improve = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break
        
        g, M_pred = socp_step(a, delta)
        if g is None:
            # Fallback: try smaller delta
            delta *= 0.5
            if delta < min_delta:
                break
            continue
        
        iter_count += 1
        obj_new = compute_obj(g)
        
        if obj_new < best_obj - 1e-12:
            a = g.copy()
            best_obj = obj_new
            best_a = g.copy()
            no_improve = 0
            # Expand trust region
            delta = min(delta * 1.3, 0.15)
            
            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"  SQP it{iter_count}: obj={best_obj:.8f}, delta={delta:.4f}, time={elapsed:.0f}s")
        else:

                delta = 0.03
                no_improve = 0

def smooth_max_obj_and_grad(a, tau):
    """
    Compute smoothed objective and gradient.
    obj = 2n * logsumexp(tau * conv(a,a)) / (sum a)^2
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    
    S = float(np.sum(a))
    L = max_conv + np.log(sum_exp) / tau
    obj = 2.0 * n * L / (S * S)
    
    # Gradient via cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    corr = ifft(P * np.conj(A)).real[:n]
    
    grad_conv = 2.0 * tau * corr
    grad = 2.0 * n * (grad_conv * S - 2.0 * L) / (S ** 3)
    
    return obj, grad

def gd_polish(a, time_budget=100.0, tau_start=100.0, tau_end=3000.0, verbose=True):
    """High-tau smoothed gradient descent with Nesterov acceleration."""
    n = len(a)
```

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses SOCP-based Sequential Quadratic Programming (trust region) as the main engine,
combined with high-temperature smoothed gradient descent for polishing,
adaptive trust region management, and multi-length exploration.
Key innovation: SOCP trust region steps avoid the need for separate line search,
and adaptive delta management balances local accuracy with global progress.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_socp(a_old, delta):
    """
    Solve trust region SOCP:
    min M s.t. 2*conv(a_old, g)[k] - conv(a_old, a_old)[k] <= M  (all k)
               ||g - a_old||_2 <= delta
               g >= 0, sum(g) = 1
    Returns (g, M) or (None, None).
    """
    n = len(a_old)
    a_old = np.maximum(a_old, 0).astype(np.float64)
    s = float(np.sum(a_old))
    if s < 1e-10:
        return None, None
    a_norm = a_old / s

    A = build_conv_matrix(a_norm)
    conv_aa = compute_conv(a_norm)

    g = cp.Variable(n)
    M = cp.Variable()
    constraints = [
        2.0 * (A @ g) - M <= conv_aa,
        cp.sum(g) == 1,
        g >= 0,
        cp.norm(g - a_norm, 2) <= delta
    ]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val, float(M.value)
        except Exception:
            continue
    return None, None

def smooth_max_grad(a, tau):
    """
    Compute gradient of smoothed max: 2n*logsumexp(tau*conv(a,a))/S^2.
    Returns (obj, grad).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    S = float(np.sum(a))
    L = tau * max_conv + np.log(sum_exp)
    obj = 2.0 * n * L / (S * S)

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    corr = ifft(P * np.conj(A)).real[:n]

    grad_L = 2.0 * tau * corr
    grad = 2.0 * n * (grad_L * S - 2.0 * L) / (S ** 3)

    return obj, grad

        else:
            y = y / s_y

        _, grad = smooth_max_grad(y, tau)
        g_norm = float(np.linalg.norm(grad))

        # Adaptive learning rate
        if g_norm > 1e-15:
            # Barzilai-Borwein update
            if prev_g is not None and it % 5 == 0:
                s_vec = y - prev_a
                y_vec = grad - prev_g
                ss = float(np.dot(s_vec, s_vec))
                sy = float(np.dot(s_vec, y_vec))
                if ss > 1e-15 and abs(sy) > 1e-12:
                    bb_lr = ss / abs(sy)
                    bb_lr = min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))
                    lr = 0.7 * lr + 0.3 * bb_lr

            step = lr / (g_norm + 1e-10)
            step = min(step, 1.0)
        else:
            step = 0.0

        v = beta * v - step * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = grad.copy()
        a = a_new
        it += 1
        lr = max(1e-12, lr * 0.9995)

        if it - best_it > 8000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            best_it = it

        if verbose and it % 5000 == 0:
            print(f"  GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.4f}, time={time.monotonic()-start:.0f}s")

    if verbose:
        print(f"  GD done: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 7):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val, float(M.value)
        except Exception:
            continue
    return None, None


def lp_line_search_step(a, time_budget=300.0, verbose=False):
    """
    LP direction + line search as a secondary optimization method.
    Used to complement the SOCP approach.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.5:
            break

        g, M = solve_lp_direction(a)
        if g is None:
            break

        # Fast line search: evaluate at a few points
        ts = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
        best_t = 0.0
        best_obj_t = best_obj
        for t in ts:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj_t:
                    best_obj_t = obj_try
                    best_t = t

        if best_t > 0 and best_obj_t < best_obj - 1e-12:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_obj = best_obj_t
            best_a = a.copy()
            stall = 0
            if verbose:
                print(f"  LP it: obj={best_obj:.8f}, t={best_t:.2f}, time={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 10:
                break

    return best_a
