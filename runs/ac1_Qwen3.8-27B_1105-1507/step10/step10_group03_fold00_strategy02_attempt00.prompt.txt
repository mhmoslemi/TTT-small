def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:

        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    sequence = [min(1000.0, x) for x in sequence]

    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Returns (gradient of length n, exact_max of conv).
    """
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def staircase_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9, lr_scale=2.0):
    """
    Nesterov accelerated gradient descent with staircase tau schedule.
    At each tau level, runs until a max iteration count or time runs out.

    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0
        max_iters_per_level = 4000
        local_best = best_obj
        local_best_it = 0

        while time.monotonic() - start < time_budget:
            if tau_iters >= max_iters_per_level:
                break

            # Nesterov lookahead
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            # Compute gradient at y
            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            # Update
            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5

                a_new = a_new / s_new

            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it
            if obj_new < local_best - 1e-15:
                local_best = obj_new
                local_best_it = tau_iters

            a = a_new
            it += 1
            tau_iters += 1

            # Slow learning rate decay
            lr = max(1e-7, lr * 0.99998)

            # Reset if stuck
            if it - best_it > stall_patience:
                a = best_a.copy()
                v = np.zeros(n)
                lr = lr_scale / np.sqrt(n)
                best_it = it
                it = it  # keep counting

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    tau={tau:5d}: obj={best_obj:.10f}, iters={tau_iters}, t={elapsed:.0f}s")

def lp_polish_iteration(a, solver_list=None):
    """
    Single LP polishing step: solve min M s.t. conv(a,g)<=M, g>=0, sum(g)=1.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    if n > 2500:
        return None, None

    # Build convolution matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue

    return None, None

def make_init(n, mode, rng=None):
    """Create diverse initial sequences of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:  # Triangle
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
        return a / a.sum()
    elif mode == 2:  # Semicircle
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.0001))
        return a / a.sum()
    elif mode == 3:  # Two bumps
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 4:  # Uniform random
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 5:  # Cosine bump
        x = np.linspace(0, np.pi, n)
        a = np.maximum(0.5 + 0.5 * np.cos(x), 0.001)
        return a / a.sum()
    elif mode == 6:  # Multi-sine
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [2, 3, 5]:
            a += 0.2 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:  # Exponential center
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 4))
        return a / a.sum()
    elif mode == 8:  # Flat top with tapered edges
        x = np.linspace(0, 1, n)
        a = np.where((x > 0.2) & (x < 0.8), 1.0,
                     np.maximum(0.5 - np.abs(x - 0.5), 0.01))
        return a / a.sum()
    else:
        return np.ones(n) / n


def population_de(n, pop_size, time_budget, verbose=False, seed=42):
    """
    Differential evolution population search.
    Each trial gets a short GD refinement.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    # Initialize population with diverse modes
    pop = np.zeros((pop_size, n))
    for i in range(pop_size):
        pop[i] = make_init(n, mode=i % 9, rng=rng)

    objs = np.array([compute_obj(pop[i]) for i in range(pop_size)])
    best_idx = np.argmin(objs)
    best_a = pop[best_idx].copy()
    best_obj = objs[best_idx]

    if verbose:
        print(f"    DE init: best={best_obj:.10f}, mean={np.mean(objs):.10f}")

    gen = 0
    per_trial_budget = 3.0  # seconds per trial including GD

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < per_trial_budget * 2:
            break

        # Differential evolution: x_trial = x_r1 + F*(x_r2 - x_r3)
        candidates = list(range(pop_size))
        r1, r2, r3 = rng.choice(candidates, 3, replace=False)
        F = 0.5 + 0.5 * rng.random()

        trial = pop[r1] + F * (pop[r2] - pop[r3])
        trial = np.maximum(trial, 0)
        st = float(np.sum(trial))
        if st < 1e-10:
            gen += 1
            continue
        trial = trial / st

        # Quick GD refinement
        trial = staircase_gd(trial, time_budget=min(per_trial_budget, remaining * 0.3),
                             tau_levels=[20, 100, 500], verbose=False)
        obj_trial = compute_obj(trial)

        # Arena selection: replace worst if better
        worst_idx = np.argmax(objs)
        if obj_trial < objs[worst_idx]:
            pop[worst_idx] = trial
            objs[worst_idx] = obj_trial

        # Track global best
        if obj_trial < best_obj:
            best_obj = obj_trial
            best_a = trial.copy()
            if verbose and gen % 20 == 0:
                print(f"    DE gen{gen}: best={best_obj:.10f}")

        gen += 1

    return best_a, best_obj


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def symmetric_optimize(a, time_budget, verbose=False):
    """
    Optimize under the constraint that a is palindromic (a[i] = a[n-1-i]).
    Effectively halves the search space.
    """
    n = len(a)
    half = (n + 1) // 2
    # Extract half
    a_half = np.zeros(half)
    for i in range(half):
        a_half[i] = (a[i] + a[n - 1 - i]) / 2.0
    a_half = np.maximum(a_half, 0)
    s = float(np.sum(a_half))
    if s < 1e-10:
        return a.copy()

    # For even n: total = 2*sum(half)
    # For odd n: total = 2*sum(half) - a_half[half-1]
    # We'll just work with the full vector and enforce symmetry after each step

    best_a = a.copy()
    best_obj = compute_obj(a)
    # Normalize full vector
    s_full = float(np.sum(a))
    if s_full > 1e-10:
        a = a / s_full
        best_a = a.copy()
        best_obj = compute_obj(a)

    start = time.monotonic()
    tau_levels = [10, 50, 200, 800]
    lr = 2.0 / np.sqrt(n)
    beta = 0.85
    v = np.zeros(n)
    it = 0

        it_level = 0
        while time.monotonic() - start < time_budget:
            if it_level >= 2000:
                break
            y = a + beta * v
            y = np.maximum(y, 0)
            # Enforce symmetry
            y = (y + y[::-1]) / 2.0
            sy = float(np.sum(y))
            if sy < 1e-10:
                break
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            # Symmetrize gradient
            g = (g + g[::-1]) / 2.0

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            a_new = (a_new + a_new[::-1]) / 2.0
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:

                v = np.zeros(n)
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
            it += 1
            it_level += 1
            lr = max(1e-7, lr * 0.99998)

    return best_a
