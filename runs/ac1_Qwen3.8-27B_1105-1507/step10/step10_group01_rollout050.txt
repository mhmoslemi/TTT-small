"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent (multiplicative weights) with adaptive temperature as the primary
method, combined with improved LP-driven direction finding and Nesterov GD polishing.
Key improvements: natural gradient updates handling non-negativity naturally,
improved LP linearization, top-k subgradient mixing, larger n exploration,
and simulated annealing for escaping local minima.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_from_p(a, p):
    """
    Compute 2 * sum_j a[j] * p[i+j] for i=0..n-1.
    p has length 2n-1 (distribution over convolution lags).
    Uses FFT: g[i] = (b * p)[i+n-1] where b = a[::-1].
    """
    n = len(a)
    b = a[::-1]
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def smoothed_gradient(a, conv, tau):
    """Compute gradient of LSE(tau, conv(a,a)) with respect to a."""
    n = len(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Numerical stability
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    return gradient_from_p(a, p)


def topk_subgradient(a, conv, k=3):
    """Subgradient using only the top-k peaks (uniform weights)."""
    n = len(a)
    idx = np.argsort(conv)[-k:]
    p = np.zeros(2 * n - 1)
    p[idx] = 1.0 / k
    return gradient_from_p(a, p)


def blended_gradient(a, conv, tau, k_top=3, alpha=0.3):
    """Blend smoothed gradient with top-k subgradient."""
    g_smooth = smoothed_gradient(a, conv, tau)
    g_topk = topk_subgradient(a, conv, k_top)
    return alpha * g_topk + (1.0 - alpha) * g_smooth


def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update: a_i <- a_i * exp(-eta * grad_i), then normalize."""
    exponent = -eta * grad
    # Clip for numerical stability
    exponent = np.clip(exponent, -50, 50)
    a_new = a * np.exp(exponent)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s


def lp_direction_improved(a):
    """
    Solve improved LP: min M s.t. 2*conv(a,g)_k - conv(a,a)_k <= M for all k,
    sum(g) = 1, g >= 0.
    This is the first-order Taylor approximation of min max conv(g,g)_k.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)

    # Build matrix A such that (A @ g)[k] = conv(a,g)[k]
    # conv(a,g)[k] = sum_j a[k-j] * g[j] for valid j
    A = np.zeros((2 * n - 1, n))
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    # A[k, j] = a[k-j] if 0 <= k-j < n
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A[valid] = a_norm[diff[valid]]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    # 2*A@g - conv_aa <= M  =>  2*A@g - M <= conv_aa
    constraints = [2.0 * (A @ g) - M <= conv_aa, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_direction_standard(a):
    """
    Standard LP: min M s.t. conv(a,g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)

    A = np.zeros((2 * n - 1, n))
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A[valid] = a_norm[diff[valid]]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g, n_evals=40):
    """Line search for best t in [0,1] to minimize obj((1-t)*a + t*g)."""
    n = len(a)
    ts = np.linspace(0.01, 0.99, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = np.maximum((1 - t) * a + t * g, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try / s)

    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]

    # Parabolic refinement
    if 0 < best_idx < n_evals - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.01, 0.99))
            a_par = np.maximum((1 - t_par) * a + t_par * g, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                f_par = compute_obj(a_par / s_par)
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par

    return best_t, best_obj


def mirror_descent_optimize(a, time_budget=300.0, tau_init=30.0, tau_end=1500.0,
                            verbose=True, blend_k=3, blend_alpha=0.2):
    """
    Mirror descent (multiplicative weights) with adaptive temperature.
    Alternates between smoothed gradient and top-k subgradient.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    it = 0
    best_it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Temperature schedule: start low, increase, then slightly decrease
        if progress < 0.7:
            tau = tau_init + (tau_end - tau_init) * progress / 0.7
        else:
            tau = tau_end

        conv = compute_conv(a)

        # Choose gradient type: alternate or random
        if it % 5 == 0:
            # Use top-k subgradient for diversity
            g = topk_subgradient(a, conv, k=blend_k)
        elif it % 5 == 3:
            # Use blended gradient
            g = blended_gradient(a, conv, tau, k_top=blend_k, alpha=blend_alpha)
        else:
            # Use smoothed gradient
            g = smoothed_gradient(a, conv, tau)

        # Adaptive step size for mirror descent
        g_max = float(np.max(np.abs(g)))
        if g_max < 1e-15:
            break
        # Step size: balance between exploration and stability
        eta = 2.0 / (g_max + 1e-10)
        # Scale with progress (larger steps early, smaller later)
        eta *= (1.0 + 0.5 * (1.0 - progress))

        # Mirror descent step
        a_new = mirror_descent_step(a, g, eta)

        # Occasionally add small random perturbation (simulated annealing)
        if it % 30 == 20 and progress < 0.8:
            temp = 0.01 * (1.0 - progress)
            a_new = a_new + temp * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            # Try halving the step
            a_half = mirror_descent_step(a, g, eta * 0.3)
            obj_half = compute_obj(a_half)
            if obj_half < best_obj - 1e-13:
                best_obj = obj_half
                best_a = a_half.copy()
                best_it = it
                a = a_half
                stall = 0
            elif stall >= 20:
                # Perturb to escape local minimum
                strength = 0.02 * (1.0 + progress)
                a_pert = a * np.exp(strength * rng.standard_normal(n))
                a_pert = np.maximum(a_pert, 0)
                s_pert = float(np.sum(a_pert))
                if s_pert > 1e-10:
                    a = a_pert / s_pert
                else:
                    a = best_a.copy()
                stall = 0

        a = a_new
        it += 1

        if verbose and (it % 100 == 0 or it == 1):
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, eta={eta:.4f}, "
                  f"stall={stall}, time={time.monotonic()-start:.0f}s")

    return best_a


def nesterov_gd_optimize(a, time_budget=150.0, tau_init=50.0, tau_end=1000.0,
                         verbose=False):
    """Nesterov accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv_y = compute_conv(y)
        g = smoothed_gradient(y, conv_y, tau)
        g_norm = float(np.linalg.norm(g))

        # Adaptive LR
        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.3 * base_lr), 2.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adapt

        # Barzilai-Borwein every 5 iterations
        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 4.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it

    if verbose:
        print(f"    NesterovGD: obj={best_obj:.8f}, iters={it}")
    return best_a


def lp_driven_optimize(a, time_budget=200.0, verbose=True):
    """LP-driven optimization with improved linearization and line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    stall = 0
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        # Alternate between improved and standard LP
        if it % 2 == 0:
            g, M = lp_direction_improved(a)
        else:
            g, M = lp_direction_standard(a)

        if g is None:
            # Fallback: use gradient direction
            conv = compute_conv(a)
            g = smoothed_gradient(a, conv, 200.0)
            g = np.maximum(g, 0)
            s_g = float(np.sum(g))
            if s_g < 1e-10:
                break
            g = g / s_g
            M = None

        # Line search
        best_t, obj_t = line_search(a, g, n_evals=30)
        it += 1

        if obj_t < best_obj - 1e-12:
            a_try = np.maximum((1 - best_t) * a + best_t * g, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                best_obj = obj_t
                best_a = a_try.copy()
                a = a_try
                stall = 0
                no_improve = 0
                if verbose and (it % 5 == 0):
                    print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, "
                          f"M={M:.6f}({'std' if it%2==1 else 'imp'}), "
                          f"time={time.monotonic()-start:.0f}s")
        else:
            no_improve += 1
            # Try smaller steps
            for t_small in [0.1, 0.05, 0.02]:

                s_sm = float(np.sum(a_sm))

                    if obj_sm < best_obj - 1e-12:

                        a = best_a

                conv_p = compute_conv(a)
                tau_p = 300.0

                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                else:
                    a = best_a.copy()
                no_improve = 0
                if it > 200:
                    if verbose:
                        print(f"    LP: giving up after {it} iterations")
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            print(f"  Phase4 MD2: obj={obj_md2:.8f} (no improvement)")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = []
    for delta in [-200, -100, 100, 200, 400, 600]:
        n_try = n_best + delta
        if 400 <= n_try <= 2500 and n_try not in n_options:
            n_options.append(n_try)
    # Also try some absolute values
    for n_abs in [600, 1000, 1200, 1600, 2000]:
        if n_abs not in n_options:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))[:5]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Mirror descent
        alloc_md = min(remaining * 0.35, 80.0)
        if alloc_md > 15:
            a_try = mirror_descent_optimize(a_try, time_budget=alloc_md,
                                            tau_init=50.0, tau_end=800.0, verbose=False)

        # LP
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.25, 50.0)
        if alloc_lp > 10:
            a_try = lp_driven_optimize(a_try, time_budget=alloc_lp, verbose=False)

        # GD polish
        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.1, 30.0)
        if alloc_gd > 8:
            a_try = nesterov_gd_optimize(a_try, time_budget=alloc_gd,
                                         tau_init=100.0, tau_end=500.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    restart_names = ["mirror_sym", "fourier_perturb", "smooth_blend", "random_mix"]
    for r, name in enumerate(restart_names):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if name == "mirror_sym":
            a_r = (best_seq + best_seq[::-1]) / 2.0
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
            else:
                a_r = best_seq.copy()
        elif name == "fourier_perturb":
            pert = np.zeros(n_cur)
            for k in range(1, 6):
                freq = k * 2.0 * np.pi / n_cur
                amp = 0.04 * rng.uniform(0.5, 1.5) / k
                phase = rng.uniform(0, 2.0 * np.pi)
                pert += amp * np.cos(freq * np.arange(n_cur) + phase)
            a_r = best_seq * np.exp(pert)
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
        elif name == "smooth_blend":
            ks = max(5, n_cur // 12)
            if ks % 2 == 0:
                ks += 1
            a_smooth = uniform_filter1d(best_seq, size=ks)
            a_r = 0.7 * best_seq + 0.3 * a_smooth
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
        else:
            a_r = 0.8 * best_seq + 0.2 * (rng.uniform(0, 1, n_cur) / n_cur)
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r

        # Optimize with MD + LP
        alloc_md = min(remaining * 0.3, 60.0)
        if alloc_md > 12:
            a_r = mirror_descent_optimize(a_r, time_budget=alloc_md,
                                          tau_init=50.0, tau_end=500.0, verbose=False)

        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.2, 40.0)
        if alloc_lp > 10:
            a_r = lp_driven_optimize(a_r, time_budget=alloc_lp, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final high-tau mirror descent
    if remaining > 30:
        alloc = min(remaining * 0.4, 150.0)
        a_final = mirror_descent_optimize(best_seq, time_budget=alloc,
                                          tau_init=500.0, tau_end=8000.0,
                                          verbose=True, blend_k=2, blend_alpha=0.5)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.4, 100.0)
        a_final2 = lp_driven_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final tiny GD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 50.0)
        a_final3 = nesterov_gd_optimize(best_seq, time_budget=alloc,
                                        tau_init=500.0, tau_end=5000.0, verbose=True)
        obj_final3 = compute_obj(a_final3)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
