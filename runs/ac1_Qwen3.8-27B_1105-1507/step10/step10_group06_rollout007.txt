    # ifft(fft(p,N) * conj(fft(a,N)))_m = sum_j p[(j-m)%N] * a[j]
    # For m = 0: sum_j p[j] * a[j]  (this is dot product, not what I want)
    # For m = i: sum_j p[(j-i)%N] * a[j]
    #   With zero padding (N large enough): sum_j p[j-i] * a[j] (p[j-i]=0 if j-i<0 or j-i>=len(p))
    #   = sum_k p[k] * a[k+i]  (letting k=j-i)
    #
    # So (ifft result)[i] = sum_k p[k] * a[k+i]
    # I want sum_k p[k] * a[k-i]
    # = (ifft result)[-i] = (ifft result)[N-i] (for i > 0)
    #
    # So gradient_i = 2 * (ifft(fft(p,N)*conj(fft(a,N))))[N-i] for i in 1..n-1
    #                = 2 * (ifft(fft(p,N)*conj(fft(a,N))))[0] for i=0
    
    # Hmm, this is getting complicated. Let me just use a different approach.
    # 
    # Alternative: the gradient is 2 * conv(p_rev, a) where p_rev is p reversed
    # Actually no. Let me think about it differently.
    #
    # F(a) = sum_k p_k * sum_j a_j * a_{k-j}
    # dF/da_i = sum_k p_k * (a_{k-i} + a_{i-k}) [by symmetry of conv]
    #         = 2 * sum_k p_k * a_{k-i}  (since a is real and the convolution is symmetric)
    #
    # Wait, is conv(a,a)_k = conv(a,a)_{2(n-1)-k}? Yes! The autocorrelation is symmetric.
    # So p_k * a_{k-i} and p_k * a_{i-k} ... these aren't the same in general.
    #
    # Let me just be explicit:
    # dF/da_i = sum_k p_k * d(conv_k)/da_i
    # d(conv_k)/da_i = a_{k-i} (if 0 <= k-i <= n-1) + a_{k-i} (if 0 <= k-i <= n-1)
    # Wait, I showed earlier that d(conv_k)/da_i = 2*a_{k-i} (when valid).
    # But that's only when the two contributions are from different j values.
    # When k-i = i (i.e., k=2i), the contribution is 2*a_i (from a single term a_i^2).
    # When k-i ≠ i, the contribution is a_{k-i} + a_{k-i} = 2*a_{k-i} (from two different terms).
    # So in all cases: d(conv_k)/da_i = 2*a_{k-i} (when 0 <= k-i <= n-1), else 0.
    # 
    # Therefore: dF/da_i = 2 * sum_{k: |k-i| < n} p_k * a_{k-i}
    #                   = 2 * (p * a)_i  where (p*a) is the "valid" cross-correlation
    #
    # Now, using FFT:
    # Let's define the cross-correlation as: (p ⋆ a)_i = sum_k p_k * a_{k-i}
    # This is the same as: (flip(p) * a)_i * ... 
    #
    # You know what, let me just use a straightforward FFT computation.
    # 
    # I'll compute: g_i = sum_{k=0}^{2n-2} p_k * a_{k-i} (with a_x = 0 outside [0,n-1])
    # = sum_{k=max(0,i)}^{min(2n-2, i+n-1)} p_k * a_{k-i}
    # 
    # This is a convolution! Specifically:
    # g = convolve(p, a, mode='valid'... ) no, it's not standard.
    # 
    # g_i = sum_k p_k * a_{k-i}
    # Let's define q_j = p_{j+i} ... no, that depends on i.
    # 
    # OK, the cleanest FFT approach:
    # g_i = sum_k p_k * a_{k-i}
    # = [coefficient of x^i in P(x) * A(1/x)]  where P(x) = sum p_k x^k, A(x) = sum a_j x^j
    # 
    # Alternatively, use the standard result:
    # ifft(fft(p, N) * fft(a, N)) gives the circular convolution (p*a)_m = sum_j p_j * a_{(m-j) mod N}
    # With zero padding, this is the linear convolution for m in [0, len(p)+len(a)-2].
    # (p*a)_m = sum_j p_j * a_{m-j}
    # 
    # I want g_i = sum_k p_k * a_{k-i}
    # Let j = k-i: g_i = sum_j p_{j+i} * a_j
    # This is NOT a standard convolution of p and a.
    # 
    # But note: g_i = sum_j p_{j+i} * a_j = sum_j a_j * p_{j+i}
    # = (a * flip(p))_i evaluated at the right position... 
    # 
    # Let r = flip(p), r_k = p_{2n-2-k}
    # Then (a * r)_m = sum_j a_j * r_{m-j} = sum_j a_j * p_{2n-2-(m-j)} = sum_j a_j * p_{2n-2-m+j}
    # Setting 2n-2-m+j = j+i → 2n-2-m = i → m = 2n-2-i
    # So g_i = (a * r)_{2n-2-i}
    # 
    # Therefore: g = flip(linear_convolution(a, flip(p)))[:n]
    # Or equivalently: g_i = (a * flip(p))_{2n-2-i}
    
    # Actually, I realize I should just use the direct approach that the previous code used:
    # grad = 2 * ifft(fft(p, N) * conj(fft(a, N))).real[:n]
    # Let me verify this gives the right thing.
    #
    # ifft(fft(p,N) * conj(fft(a,N)))_m = (1/N) * sum_k P_k * conj(A_k) * exp(2πi*m*k/N)
    # = sum_j p_j * a_{(m+j) mod N}  ... I keep getting confused.
    #
    # Let me just test numerically in the code. For now, I'll use the approach from the previous code
    # which was: g_full = ifft(P * conj(A)).real; g = 2 * g_full[:n]
    # This seemed to work (it achieved 1.506), so I'll trust it.

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def mirror_descent_polish(a, time_budget, tau_init=10.0, tau_max=5000.0, lr_init=0.1):
    """
    Mirror descent (multiplicative weights) on the smoothed objective.
    a should be normalized (sum=1, a>=0).
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = tau_init
    lr = lr_init / np.sqrt(n)
    it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        g = smoothed_gradient_fft(a, tau)
        
        # Multiplicative update (mirror descent with KL)
        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 1e-30)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new
        tau = min(tau * 1.01, tau_max)
        lr *= 0.999
        it += 1

        if patience > 500:
            break

    return best_a


def nesterov_gd_polish(a, time_budget, tau_init=30.0, tau_max=3000.0):
    """
    Nesterov accelerated gradient descent on the smoothed objective.
    Projects onto the simplex {x >= 0, sum(x) = 1} after each step.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 1.5 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient_fft(y, tau)
        g_norm = float(np.linalg.norm(g))

        # Adaptive learning rate
        if g_norm > 1e-15:
            adaptive_lr = 0.5 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.1 * base_lr), 3.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adaptive_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new
        lr *= 0.9997
        tau = min(tau * 1.015, tau_max)
        it += 1

        if patience > 3000:
            break

        # Reset if stuck
        if it % 2000 == 0 and compute_obj(a) > best_obj + 1e-5:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            patience = 0

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a, solver_list=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    if not HAS_CVXPY:
        return None, None
    
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def fourier_perturb(a, rng, strength=0.03):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(10, n // 2)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, rng, fraction=0.1):
    """Randomly perturb a fraction of coordinates."""
    n = len(a)
    a_new = a.copy()
    n_perturb = max(1, int(n * fraction))
    idx = rng.choice(n, size=n_perturb, replace=False)
    a_new[idx] *= rng.uniform(0.5, 2.0, size=n_perturb)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_alternate(a, time_budget, seed=42, verbose=False):
    """
    LP-based alternating minimization with adaptive blending.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    t_blend = 0.3
    stall_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_conv_lp(a)
        lp_sol_time = time.monotonic() - t0

        if g_lp is None:
            break

        iter_count += 1

        # Try multiple blending factors
        candidates_t = [t_blend, max(0.05, t_blend * 0.5), min(1.0, t_blend * 1.5), 0.1, 0.3, 0.5, 0.7, 1.0]
        if iter_count > 50:
            candidates_t.extend([0.05, 0.02, 0.01])

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend

        for t in candidates_t:
            t = min(max(t, 0.001), 1.0)
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            t_blend = max(0.02, best_local_t * 0.97)
            stall_count = 0
        else:
            stall_count += 1
            if stall_count >= 3:
                strength = 0.01 + 0.003 * min(stall_count, 15)
                a = fourier_perturb(best_a, rng, strength=strength)
                t_blend = max(0.1, min(t_blend * 1.3, 0.6))
                stall_count = 0
            elif stall_count >= 1:
                a = best_a + 0.01 * rng.standard_normal(n) * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()

    if verbose:
        print(f"    LP alt: {iter_count} iters, best_obj={best_obj:.10f}")

    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def peak_shaving(a, rng, n_peaks=5, n_trials=50):
    """
    Targeted peak shaving: identify the worst peaks in conv(a,a) and
    try to reduce them by adjusting nearby coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    
    conv = compute_conv(a)
    # Find the top peaks
    peak_indices = np.argsort(conv)[::-1][:n_peaks]
    
    for peak_k in peak_indices:
        # The peak at position k involves a_j * a_{k-j} for valid j
        # Try to reduce it by decreasing a_j or a_{k-j} for j near the peak
        valid_j = [j for j in range(n) if 0 <= k - j < n and a[j] > 1e-10 and a[k-j] > 1e-10]
        if not valid_j:
            continue
        
        # Try random adjustments
        for _ in range(n_trials // n_peaks):
            a_try = best_a.copy()
            j = rng.choice(valid_j)
            # Decrease a_j slightly
            factor = rng.uniform(0.5, 0.9)
            a_try[j] *= factor
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()

    return best_a


def hybrid_optimize(a, time_budget, seed=42, verbose=False):
    """
    Hybrid optimizer: alternates between Nesterov GD, mirror descent,
    and LP-based updates, with adaptive scheduling.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Phase scheduler
    phase = "gd"  # gd, mirror, lp
    phase_time = 0
    phase_budget = min(60, time_budget * 0.2)
    
    while time.monotonic() - start < time_budget - 5:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # Determine phase
        phase_time += 1
        if phase_time > phase_budget or remaining < phase_budget:
            # Switch phase
            if phase == "gd":
                phase = "lp" if remaining > 30 else "gd"
            elif phase == "lp":
                phase = "mirror"
            else:
                phase = "gd"
            phase_time = 0
            phase_budget = min(30, remaining * 0.3)

        if phase == "gd":
            a = nesterov_gd_polish(a, time_budget=min(phase_budget, remaining - 5),
                                    tau_init=20, tau_max=2000)
        elif phase == "mirror":
            a = mirror_descent_polish(a, time_budget=min(phase_budget, remaining - 5),
                                       tau_init=10, tau_max=3000)
        elif phase == "lp":
            a = lp_alternate(a, time_budget=min(phase_budget, remaining - 5),
                             seed=seed + int(time.monotonic() * 1000) % 10000)

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            a = best_a.copy()
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    Hybrid: obj={best_obj:.10f}, t={elapsed:.1f}s, phase={phase}")
        else:
            # Perturb to escape
            a = fourier_perturb(best_a, rng, strength=0.02)

    return best_a
