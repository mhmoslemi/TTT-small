import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import diags

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix_sparse(a):
    """Build sparse convolution matrix."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_lp_sparse(a, warm_start=None):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_sparse(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

diverse initialization, Nesterov smoothed gradient polishing, and multi-length
exploration. Key improvements: sparse LP matrix, log-space geometric blending,
diverse parametric initializations, coordinate descent polish, and adaptive
time management.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import csr_matrix


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a):
    """Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    idx = 0
    for j in range(n):
        for i in range(n):
            rows[idx] = j + i
            cols[idx] = j
            vals[idx] = a[i]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a_norm*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt, dual_values) or (None, None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_sparse(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    cons = A @ g <= M
    constraints = [cons, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    best_result = None
    best_status = None
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    dual = None
                    try:
                        dual = np.abs(cons.dual.value)
                    except Exception:
                        pass
                    return g_val / s, float(M.value), dual
        except Exception:
            continue
    return None, None, None


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of (a*a)_k using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def nesterov_polish(a, time_budget, tau_init=30.0, tau_max=3000.0, verbose=False):
    """Nesterov accelerated gradient descent on the smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 3.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adaptive_lr = 1.0 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.05 * base_lr), 5.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adaptive_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        lr *= 0.9999
        tau = min(tau * 1.015, tau_max)
        it += 1

        if stall >= 100:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            stall = 0

        if verbose and it % 500 == 0:
            print(f"    GD it={it}, obj={obj_new:.10f}, best={best_obj:.10f}, tau={tau:.1f}")

    return best_a


def coordinate_descent_polish(a, time_budget, verbose=False):
    """Coordinate descent: try adjusting pairs of coordinates to reduce peak conv."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    it = 0
    while time.monotonic() - start < time_budget:
        # Pick two random coordinates and try to shift mass between them
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try shifting mass from i to j (and vice versa)
        for direction in [1, -1]:
            for frac in [0.01, 0.05, 0.1, 0.2]:
                delta = direction * frac * a[i]
                if a[i] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break

        it += 1
        if verbose and it % 1000 == 0:
            print(f"    CD it={it}, obj={best_obj:.10f}")

    return best_a


def fourier_perturb(a, rng, strength=0.03):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 8
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, rng, strength=0.05):
    """Add random multiplicative perturbation."""
    n = len(a)
    noise = 1.0 + strength * rng.standard_normal(n)
    a_new = a * noise
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def geometric_blend(a, g, t):
    """Geometric (log-space) interpolation on the positive simplex."""
    a_safe = np.maximum(a, 1e-15)
    g_safe = np.maximum(g, 1e-15)
    log_a = np.log(a_safe)
    log_g = np.log(g_safe)
    log_new = (1.0 - t) * log_a + t * log_g
    a_new = np.exp(log_new)
    s = float(np.sum(a_new))
    if s < 1e-15:
        return a.copy()
    return a_new / s


def lp_alternate(a, time_budget, seed=42, verbose=True, use_geometric=True):
    """Main LP-based alternating minimization with adaptive blending."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    t_blend = 0.4
    stall_count = 0
    prev_g = None

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        t0 = time.monotonic()
        g_lp, M_lp, dual = solve_lp(a)
        lp_time = time.monotonic() - t0

        if g_lp is None:
            break

        iter_count += 1

        # Adaptive candidate blending factors
        candidates_t = [t_blend, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if iter_count > 100:
            candidates_t.extend([0.05, 0.02, 0.01, 0.005])
        if iter_count < 20:
            candidates_t = [0.3, 0.5, 0.7, 1.0]

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend
        used_geometric = False

        for t in sorted(candidates_t, reverse=True):
            if use_geometric:
                a_try = geometric_blend(a, g_lp, t)
            else:
                a_try = (1.0 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try

            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t
                used_geometric = use_geometric

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            t_blend = max(0.005, best_local_t * 0.95)
            stall_count = 0
            prev_g = g_lp
            if verbose and iter_count % 50 == 0:
                print(f"    LP alt {iter_count}: obj={best_obj:.10f}, t={t_blend:.4f}, "
                      f"M_lp={M_lp:.6f}, lp_time={lp_time:.3f}s")
        else:
            stall_count += 1
            if stall_count >= 5:
                strength = 0.02 + 0.01 * min(stall_count, 15)
                if rng.random() < 0.5:
                    a = fourier_perturb(best_a, rng, strength=strength)
                else:
                    a = random_perturb(best_a, rng, strength=strength)
                t_blend = max(0.1, min(t_blend * 1.5, 0.8))
                stall_count = 0
            elif stall_count >= 2:
                a = best_a + 0.01 * rng.standard_normal(n) * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()

    if verbose:
        print(f"    LP alternating: {iter_count} iterations, best_obj={best_obj:.10f}")

    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def parametric_init(n, kind, rng):
    """Generate diverse parametric initializations."""
    x = np.linspace(0, 1, n)
    if kind == 0:
        a = np.ones(n)
    elif kind == 1:
        # U-shaped
        a = 1.0 + 2.0 * (x - 0.5) ** 2
    elif kind == 2:
        # V-shaped
        a = 1.0 + 3.0 * np.abs(x - 0.5)
    elif kind == 3:
        # Cosine bump
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
    elif kind == 4:
        # Multi-frequency
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
    elif kind == 5:
        # Random smooth
        freqs = rng.choice(range(1, 20), size=5, replace=False)
        a = np.ones(n)
        for f in freqs:
            a += rng.uniform(0.1, 0.5) * np.cos(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
    elif kind == 6:
        # Exponential edges
        a = np.exp(2.0 * (2 * x - 1) ** 2)
    elif kind == 7:
        # Pure random
        a = rng.uniform(0.1, 1.0, n)
    elif kind == 8:
        # Two-peak
        a = 1.0 + 1.5 * np.exp(-50 * (x - 0.2) ** 2) + 1.5 * np.exp(-50 * (x - 0.8) ** 2)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a


def dual_guided_perturb(a, dual, rng, strength=0.02):
    """Perturb guided by LP dual information (which lags are active)."""
    n = len(a)
    if dual is None or len(dual) == 0:
        return random_perturb(a, rng, strength)

    # Active lags (high dual values) indicate where the peak is
    # Try to reduce the sequence at positions that contribute to active lags
    active = np.zeros(n)
    for k, d in enumerate(dual):
        if d > 0.01:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    active[i] += d * a[j]

    active = active / (np.max(active) + 1e-15)
    pert = 1.0 - strength * active + strength * 0.5 * rng.standard_normal(n)
    a_new = a * pert
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

        candidates.append(a_cand)

    best_obj = np.inf
    for i, a_cand in enumerate(candidates):
