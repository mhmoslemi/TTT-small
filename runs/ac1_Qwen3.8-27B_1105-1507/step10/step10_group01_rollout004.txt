import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    p = a / s
    conv = compute_conv_fft(p)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_direction(a_p):

def line_search_accelerated(a, b, t_min, t_max, n_evals=40):
    """
    Line search: minimize obj((1-t)*a + t*b) for t in [t_min, t_max].
    Uses coarse grid + golden section refinement.
    Returns (best_t, best_obj).
    """
    n = len(a)
    ts = np.linspace(t_min, t_max, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        x = (1 - t) * a + t * b
        x = np.maximum(x, 0)
        s = float(np.sum(x))
        if s > 1e-10:
            objs[i] = compute_obj(x)

    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]

    # Golden section refinement around best
    lo = ts[max(0, best_idx - 1)]
    hi = ts[min(n_evals - 1, best_idx + 1)]
    gr = (np.sqrt(5) - 1) / 2
    for _ in range(15):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        x1 = (1 - m1) * a + m1 * b
        x1 = np.maximum(x1, 0)
        x2 = (1 - m2) * a + m2 * b
        x2 = np.maximum(x2, 0)
        s1 = float(np.sum(x1))

        f2 = compute_obj(x2) if s2 > 1e-10 else np.inf

                best_obj = f1
                best_t = m1
        else:
            lo = m1
            if f2 < best_obj:

                best_t = m2

def peak_shaving(p, n_peaks=10, strength=0.1, n_iters=50):
    """
    Reduce peaks in the autocorrelation by modifying p.
    For each peak lag k, reduce p[i] where i and i+k both have large values.
    """
    n = len(p)
    p = p.copy()
    for _ in range(n_iters):
        conv = compute_conv_fft(p)
        # Find the n_peaks largest peaks (excluding lag 0 which is always max)
        c_shifted = conv.copy()
        c_shifted[0] = -np.inf
        peak_indices = np.argsort(c_shifted)[-n_peaks:][::-1]

        improved = False
        for k in peak_indices:
            if k < 0 or k >= n:
                continue
            # c[k] = sum_i p[i]*p[i+k] (for k > 0)
            # Reduce contributions from large products
            if 0 < k < n:
                products = p[:n - k] * p[k:]
                avg_product = float(np.mean(products))
                # Find indices where product is much above average
                threshold = avg_product * 1.5
                mask = products > threshold
                if np.sum(mask) > 0:
                    # Reduce p[i] and p[i+k] at those locations
                    idx_reducing = np.where(mask)[0]
                    for i in idx_reducing[:20]:
                        p[i] *= (1 - strength)
                        p[i + k] *= (1 - strength)
                    improved = True

        p = np.maximum(p, 0)
        s = float(np.sum(p))
        if s < 1e-10:
            return p
        p = p / s

        if not improved:
            break

    return p


def smoothed_gradient(p, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(p,p)."""
    n = len(p)
    conv = compute_conv_fft(p)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        q = np.zeros(len(conv))
        q[int(np.argmax(conv))] = 1.0
    else:
        q = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    Q = fft(q, N)
    P = fft(p, N)
    grad = 2.0 * ifft(Q * np.conj(P)).real[:n]
    return grad

    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 2.0

        y = p + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_p.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        # Adaptive learning rate
        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm * np.sqrt(n) + 1e-10)
            adapt = min(max(adapt, 0.1 * base_lr), 5.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adapt

        # Barzilai-Borwein every 5 iterations
        if prev_g is not None and it % 5 == 4:
            s_vec = y - prev_p
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-10), 8.0 / np.sqrt(n))
                lr = 0.6 * lr + 0.4 * bb

        v = beta * v - lr * g
        p_new = y + v
        p_new = np.maximum(p_new, 0)
        s_new = float(np.sum(p_new))
        if s_new < 1e-10:
            p_new = best_p.copy()
            v = np.zeros(n)
        else:
            p_new = p_new / s_new

        obj_new = compute_obj(p_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_p = p_new.copy()
            best_it = it

        prev_p = p.copy()
        prev_g = g.copy()
        p = p_new
        it += 1
        lr = max(1e-12, lr * 0.9999)

        # Reset if stuck
        if it - best_it > 20000:
            p = best_p.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it
            beta = max(0.5, beta * 0.9)

    if verbose:
        print(f"    GD: obj={best_obj:.8f}, iters={it}, tau_final={tau:.0f}")
    return best_p


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation (spline)."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if new_n < 2:
        return a[:1].copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, kind='cubic')
    a_new = np.maximum(a_new, 0)
    return a_new


def frequency_flatten(p, n_iters=100, verbose=False):
    """
    Try to flatten the power spectrum |FFT(p)|^2 via iterative projection.
    This pushes energy from high-power frequencies to low-power ones.
    """
    n = len(p)
    N = 1
    while N < 2 * n:
        N <<= 1
    p = p.copy()

    for it in range(n_iters):
        P = fft(p, N)
        power = np.abs(P) ** 2
        avg_power = float(np.mean(power))

        # Reduce high-power bins, increase low-power bins
        alpha = 0.01
        P_modified = P * np.exp(-alpha * (power - avg_power) / (avg_power + 1e-10))

        p_new = ifft(P_modified).real[:n]
        p_new = np.maximum(p_new, 0)
        s = float(np.sum(p_new))
        if s < 1e-10:
            break
        p_new = p_new / s
        p = p_new

    return p


def coordinate_descent_poke(p, n_pokes=100, verbose=False):
    """
    Random coordinate-wise perturbations with accept/reject.
    Tries small changes to random elements and keeps improvements.
    """
    n = len(p)
    p = p.copy()
    best_obj = compute_obj(p)
    rng = np.random.default_rng(42)

    for _ in range(n_pokes):
        i = rng.integers(0, n)
        # Try multiplying p[i] by a random factor
        factor = rng.uniform(0.7, 1.3)
        p_try = p.copy()
        p_try[i] *= factor
        p_try = np.maximum(p_try, 0)
        s = float(np.sum(p_try))
        if s < 1e-10:
            continue
        p_try = p_try / s
        obj_try = compute_obj(p_try)
        if obj_try < best_obj - 1e-14:
            p = p_try
            best_obj = obj_try

    return p


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining ACG, peak shaving, GD polish, frequency flattening,
    multi-n exploration, and coordinate descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
            else:
                raise ValueError
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)
    n_best = n_current

    # === Phase 1: Accelerated Conditional Gradient (main workhorse) ===
    print(f"\n=== Phase 1: ACG Optimization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.3, 400.0)
    if p1_budget > 30:
        a_acg = accelerated_conditional_gradient(best_seq, time_budget=p1_budget, verbose=True)
        obj_acg = compute_obj(a_acg)
        if obj_acg < best_score:
            best_score = obj_acg
            best_seq = a_acg.copy()
            n_best = len(best_seq)
            print(f"  Phase1 ACG: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 ACG: obj={obj_acg:.8f} (no improvement)")

    # === Phase 2: Peak Shaving ===
    print(f"\n=== Phase 2: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_psvd = peak_shaving(best_seq, n_peaks=15, strength=0.08, n_iters=80)
        obj_psvd = compute_obj(a_psvd)
        if obj_psvd < best_score:
            best_score = obj_psvd
            best_seq = a_psvd.copy()
            print(f"  Phase2 PSV: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 PSV: obj={obj_psvd:.8f} (no improvement)")

    # === Phase 3: GD Polish (first round) ===
    print(f"\n=== Phase 3: GD Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 100.0)
    if p3_budget > 15:
        a_gd = gd_polish(best_seq, time_budget=p3_budget, tau_init=80.0, tau_end=2000.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase3 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 GD: obj={obj_gd:.8f} (no improvement)")

    # === Phase 4: Second ACG Round ===
    print(f"\n=== Phase 4: Second ACG Round ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.25, 300.0)
    if p4_budget > 30:
        a_acg2 = accelerated_conditional_gradient(best_seq, time_budget=p4_budget, verbose=True)
        obj_acg2 = compute_obj(a_acg2)
        if obj_acg2 < best_score:
            best_score = obj_acg2
            best_seq = a_acg2.copy()
            n_best = len(best_seq)
            print(f"  Phase4 ACG: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 ACG: obj={obj_acg2:.8f} (no improvement)")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    n_options = sorted(set([
        n_cur - 200, n_cur - 100, n_cur - 50, n_cur - 20,
        n_cur + 20, n_cur + 50, n_cur + 100, n_cur + 200, n_cur + 400,
        600, 700, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_cur) >= 20]
    n_options = n_options[:8]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)

        # ACG optimization
        alloc_acg = min(remaining * 0.25, 80.0)
        if alloc_acg > 15:
            a_try = accelerated_conditional_gradient(a_try, time_budget=alloc_acg, verbose=False)

        # Peak shaving
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_try = peak_shaving(a_try, n_peaks=10, strength=0.06, n_iters=40)

        # GD polish
        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.08, 30.0)
        if alloc_gd > 10:
            a_try = gd_polish(a_try, time_budget=alloc_gd, tau_init=100.0, tau_end=1500.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Frequency Flattening + GD ===
    print(f"\n=== Phase 6: Frequency Flattening ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_ff = frequency_flatten(best_seq, n_iters=200)
        obj_ff = compute_obj(a_ff)
        if obj_ff < best_score:
            best_score = obj_ff
            best_seq = a_ff.copy()
            print(f"  Phase6 FF: obj={best_score:.8f} **")
        else:
            print(f"  Phase6 FF: obj={obj_ff:.8f} (no improvement)")
            # Still try GD on the original
            a_gd2 = gd_polish(best_seq, time_budget=min(remaining * 0.15, 60.0),
                              tau_init=200.0, tau_end=5000.0, verbose=False)
            obj_gd2 = compute_obj(a_gd2)
            if obj_gd2 < best_score:
                best_score = obj_gd2
                best_seq = a_gd2.copy()
                print(f"  Phase6 GD: obj={best_score:.8f} **")

    # === Phase 7: Coordinate Descent Poke ===
    print(f"\n=== Phase 7: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_cd = coordinate_descent_poke(best_seq, n_pokes=200)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase7 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase7 CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 8: Final Intensive Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final ACG with small budget
    if remaining > 30:
        alloc = min(remaining * 0.4, 150.0)
        a_final = accelerated_conditional_gradient(best_seq, time_budget=alloc, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final ACG: obj={best_score:.8f} **")

    # Final high-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 80.0)
        a_final2 = gd_polish(best_seq, time_budget=alloc, tau_init=500.0, tau_end=10000.0, verbose=True)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # === Phase 9: Diverse Restarts (if time remains) ===
    print(f"\n=== Phase 9: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    # Fourier perturbation restart
    if remaining > 60:
        n = n_cur
        pert = np.zeros(n)
        for k in range(1, 7):
            freq = k * 2.0 * np.pi / n
            amp = 0.03 * rng.uniform(0.5, 1.5) / k
            phase = rng.uniform(0, 2.0 * np.pi)
            pert += amp * np.cos(freq * np.arange(n) + phase)
        a_r = best_seq * np.exp(pert)
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.3, 80.0)
        a_r = accelerated_conditional_gradient(a_r, time_budget=alloc, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_r = gd_polish(a_r, time_budget=min(remaining * 0.3, 30.0),
                           tau_init=100.0, tau_end=2000.0)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart (fourier): obj={best_score:.8f} **")
        else:
            print(f"  Restart (fourier): obj={obj_r:.8f}")

    # Smoothing restart
    remaining = deadline - time.monotonic()
    if remaining > 50:
        ks = max(5, n_cur // 20)
        if ks % 2 == 0:
            ks += 1
        # Manual uniform filter
        kernel = np.ones(ks) / ks
        a_smooth = np.convolve(best_seq, kernel, mode='same')
        a_r = 0.85 * best_seq + 0.15 * a_smooth
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.3, 60.0)
        a_r = accelerated_conditional_gradient(a_r, time_budget=alloc, verbose=False)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart (smooth): obj={best_score:.8f} **")
        else:
            print(f"  Restart (smooth): obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure all values are valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    # Verify with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
