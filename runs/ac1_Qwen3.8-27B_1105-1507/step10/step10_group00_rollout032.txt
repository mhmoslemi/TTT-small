import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_grad_at_peak(a, k_max):
    """Compute gradient of (a*a)[k_max] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        j1 = k_max - i
        j2 = i - k_max
        if 0 <= j1 < n:
            grad[i] += a[j1]
        if 0 <= j2 < n:
            grad[i] += a[j2]
    return grad

# Vectorized gradient
def compute_grad_at_peak_vec(a, k_max):
    """Vectorized gradient computation."""
    n = len(a)
    grad = np.zeros(n)
    # For i in [0, n-1]:
    #   j1 = k_max - i, valid if 0 <= k_max - i < n, i.e., max(0, k_max-n+1) <= i <= min(n-1, k_max)
    #   j2 = i - k_max, valid if 0 <= i - k_max < n, i.e., max(0, k_max) <= i <= min(n-1, k_max+n-1)
    
    i_min1 = max(0, k_max - n + 1)
    i_max1 = min(n - 1, k_max)
    if i_min1 <= i_max1:
        indices1 = np.arange(i_min1, i_max1 + 1)
        grad[indices1] += a[k_max - indices1]
    
    i_min2 = max(0, k_max)
    i_max2 = min(n - 1, k_max + n - 1)
    if i_min2 <= i_max2:
        indices2 = np.arange(i_min2, i_max2 + 1)
        grad[indices2] += a[indices2 - k_max]

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

    s = float(np.sum(a))

        return np.inf

    return 2.0 * n * float(np.max(conv))


def compute_grad_at_peak(a, k_max):
    """Compute gradient of (a*a)[k_max] with respect to a.
    grad[m] = 2*a[k_max-m] if 0 <= k_max-m < n, else 0."""
    n = len(a)

        grad[idx] = 2.0 * a[k_max - idx]
    return grad

"""
Hybrid optimization for step function coefficients using:
1. Fast Away-Steps Frank-Wolfe with analytic O(n) line search (exploiting
   quadratic structure of autoconvolution under sparse direction changes)
2. Nesterov accelerated subgradient for warm-up
3. Adaptive length exploration
4. Spectral initializations for diversity
Key insight: for pairwise FW moves, conv(a + t*(e_i - e_j)) can be evaluated
in O(n) without FFT, enabling very fast line searches.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def obj_normalized(a):
    """Objective assuming sum(a)=1, a>=0. Returns 2n*max(conv)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_grad_at_peak(a, k_max):
    """Gradient of (a*a)[k_max] w.r.t. a. grad[m] = 2*a[k_max-m] if valid."""
    n = len(a)
    grad = np.zeros(n)
    i_min = max(0, k_max - n + 1)
    i_max = min(n - 1, k_max)
    if i_min <= i_max:
        idx = np.arange(i_min, i_max + 1)
        grad[idx] = 2.0 * a[k_max - idx]
    return grad


def nesterov_subgrad(a, time_budget=20.0):
    """Nesterov accelerated subgradient descent on simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_normalized(a)
    y = a.copy()
    v = np.zeros(n)
    start = time.monotonic()
    it = 0
    lr = 0.05

    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        k_max = int(np.argmax(conv))
        g = compute_grad_at_peak(y, k_max)

        momentum = it / (float(it) + 2.0)
        v = momentum * v - lr * g
        y_new = y + v
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = best_a
            v = np.zeros(n)
        else:
            y_new = y_new / s_new

        y = y_new
        obj = obj_normalized(y)
        if obj < best_obj:
            best_obj = obj
            best_a = y.copy()

        it += 1
        lr = max(0.005, lr * 0.9995)

    return best_a


def fast_fw_pairwise_eval(conv_a, a, i, j, n, t):
    """Evaluate max of conv(a + t*(e_i - e_j)) given conv_a, using O(n) formula."""
    # conv_new[k] = conv_a[k] + 2t*(a[k-i] - a[k-j]) + t^2*(delta_{2i} - 2*delta_{i+j} + delta_{2j})
    # Build delta array: delta[k] = 2*(a[k-i] - a[k-j])
    delta = np.zeros(2 * n - 1)
    # a shifted by i: positions k where 0 <= k-i < n, i.e., i <= k <= i+n-1
    lo_i = max(0, i)
    hi_i = min(2 * n - 2, i + n - 1)
    if lo_i <= hi_i:
        delta[lo_i:hi_i + 1] = a[0:hi_i - lo_i + 1]
    # a shifted by j (subtracted)
    lo_j = max(0, j)
    hi_j = min(2 * n - 2, j + n - 1)
    if lo_j <= hi_j:
        delta[lo_j:hi_j + 1] -= a[0:hi_j - lo_j + 1]
    delta *= 2.0

    # Add t*delta to conv_a
    conv_new = conv_a + t * delta

    # Add t^2 corrections
    t2 = t * t
    if 0 <= 2 * i < 2 * n - 1:
        conv_new[2 * i] += t2
    if 0 <= i + j < 2 * n - 1:
        conv_new[i + j] -= 2.0 * t2
    if 0 <= 2 * j < 2 * n - 1:
        conv_new[2 * j] += t2

    return float(np.max(conv_new))


def fast_fw_standard_eval(conv_a, a, i, n, t):
    """Evaluate max of conv((1-t)*a + t*e_i) given conv_a, using O(n) formula.
    conv_new[k] = (1-t)^2 * conv_a[k] + 2t(1-t) * a[k-i] + t^2 * delta_{k,2i}
    """
    # a shifted by i
    a_shift = np.zeros(2 * n - 1)
    lo_i = max(0, i)
    hi_i = min(2 * n - 2, i + n - 1)
    if lo_i <= hi_i:
        a_shift[lo_i:hi_i + 1] = a[0:hi_i - lo_i + 1]

    one_minus_t = 1.0 - t
    conv_new = (one_minus_t * one_minus_t) * conv_a + 2.0 * t * one_minus_t * a_shift
    if 0 <= 2 * i < 2 * n - 1:
        conv_new[2 * i] += t * t

    return float(np.max(conv_new))


def golden_section(eval_func, lo, hi, iters=30):
    """Golden section search. Returns (best_t, best_val)."""
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = eval_func(c)
    fd = eval_func(d)
    for _ in range(iters):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = eval_func(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = eval_func(d)
    best_t = c if fc < fd else d
    best_val = min(fc, fd)
    return best_t, best_val


def away_steps_fw(a, time_budget=60.0, max_iters=100000, verbose=False,

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(rng_seed)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.01:
            break

        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        g = compute_grad_at_peak(a, k_max)

        # Standard FW: move toward argmin g
        i = int(np.argmin(g))

        # Pairwise: move from j (largest a_j * g_j) to i
        ag = a * g
        j = int(np.argmax(ag))

        best_t = 0.0
        best_obj_try = best_obj
        best_type = 'none'

        # Try standard FW line search

            return fast_fw_standard_eval(conv, a, i, n, t)

        t_fw, obj_fw = golden_section(eval_fw, 0.0, 1.0, iters=25)
        if obj_fw < best_obj_try:
            best_obj_try = obj_fw
            best_t = t_fw
            best_type = 'fw'

        # Try pairwise (if different indices and j has mass)
        if i != j and a[j] > 1e-10:
            def eval_pw(t):
                return fast_fw_pairwise_eval(conv, a, i, j, n, t)

            t_pw, obj_pw = golden_section(eval_pw, 0.0, a[j], iters=25)
            if obj_pw < best_obj_try:
                best_obj_try = obj_pw
                best_t = t_pw
                best_type = 'pw'

        # Also try a few random directions for diversity
        if it > 100 and it % 100 == 0:
            for _ in range(2):
                i_rand = int(rng.integers(0, n))
                j_rand = int(rng.integers(0, n))

                    continue

                def eval_rand(t):
                    return fast_fw_pairwise_eval(conv, a, i_rand, j_rand, n, t)

                t_r, obj_r = golden_section(eval_rand, 0.0, a[j_rand], iters=15)
                if obj_r < best_obj_try:
                    best_obj_try = obj_r
                    best_t = t_r
                    best_type = 'rand'
                    i = i_rand
                    j = j_rand

        # Apply best move
        if best_type == 'fw':
            a_new = (1.0 - best_t) * a
            a_new[i] += best_t
        elif best_type in ('pw', 'rand'):
            a_new = a.copy()
            a_new[i] += best_t
            a_new[j] -= best_t
            a_new = np.maximum(a_new, 0)
        else:
            # No improvement found - perturbation
            pert = rng.normal(0, 0.002, n)
            a_new = a + pert
            a_new = np.maximum(a_new, 0)

        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = obj_normalized(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
            a = a_new
        else:
            a = best_a

        it += 1

        # Reset if stuck too long
        if stall > 50:
            a = best_a + 0.005 * rng.normal(0, 1, n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0

        if verbose and it % 200 == 0:
            print(f"    FW it{it}: obj={best_obj:.10f}, type={best_type}, "
                  f"t={best_t:.5f}, elapsed={time.monotonic()-start:.1f}s")

    return best_a

def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random nonnegative initialization, smoothed."""
    a = rng.uniform(0.1, 1.0, n)
    # Smooth slightly
    kernel_size = 5
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    a = a / s
    return a


def spectral_init(n, rng):
    """Initialize from a random-phase flat spectrum, clipped to nonnegative."""
    # Flat magnitude spectrum with random phase
    mag = np.ones(n)
    phase = rng.uniform(0, 2 * np.pi, n)
    spectrum = mag * np.exp(1j * phase)
    # IFFT to get real sequence
    a = np.fft.ifft(spectrum).real
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return u_shaped_init(n)
    return a / s


def interpolate_sequence(a_src, n_dst):
    """Interpolate a sequence to a different length."""
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 1e-10:
        return np.ones(n_dst) / n_dst
    return a_dst / s

        print(f"  Trying n values: {n_options}")

        # Allocate time evenly
        time_per_n = max(20.0, (remaining - 50) / len(n_options))

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            a_try = interpolate_sequence(best_seq, n_try)
            # Quick FW
            alloc = min(time_per_n, remaining * 0.6)
            a_opt = away_steps_fw(a_try, time_budget=alloc, max_iters=100000,
                                  rng_seed=n_try)
            obj_try = obj_normalized(a_opt)
            print(f"  n={n_try}: obj={obj_try:.10f}")
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: IMPROVED to {best_score:.10f}")

    # === Phase 5: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 5: Diverse restarts ===")
        n_cur = len(best_seq)

        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            if r == 0:
                a_r = u_shaped_init(n_cur, sharpness=rng.uniform(2, 6))
            elif r == 1:
                a_r = spectral_init(n_cur, rng)
            elif r == 2:
                a_r = random_init(n_cur, rng)
            else:
                a_r = best_seq + 0.05 * rng.normal(0, 1, n_cur)
                a_r = np.maximum(a_r, 0)
                a_r = a_r / a_r.sum()

            # Blend with best
            blend = 0.3
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            alloc = min(60.0, remaining * 0.3)
            a_opt = away_steps_fw(a_r, time_budget=alloc, max_iters=50000,
                                  rng_seed=1000 + r)
            obj_r = obj_normalized(a_opt)
            print(f"  Restart {r}: obj={obj_r:.10f}")
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  ** IMPROVED to {best_score:.10f}")

    # === Phase 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Final polish (budget={min(remaining-10, 100):.0f}s) ===")
        alloc = min(remaining - 10, 100.0)
        a_final = away_steps_fw(best_seq, time_budget=alloc, verbose=True,
                                max_iters=200000, rng_seed=999)
        obj_final = obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final: obj={best_score:.10f} ** IMPROVED")
        else:
            print(f"  Final: obj={obj_final:.10f}")

    # Very final: one more quick pass
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_check = away_steps_fw(best_seq, time_budget=min(remaining - 5, 30.0),
                                max_iters=50000, rng_seed=777)
        obj_check = obj_normalized(a_check)
        if obj_check < best_score:
            best_score = obj_check
            best_seq = a_check.copy()
            print(f"  Extra: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
