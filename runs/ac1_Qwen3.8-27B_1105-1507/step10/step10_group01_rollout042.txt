def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    return A


def build_conv_matrix_vectorized(a):
    """Build convolution matrix using vectorized operations."""
    n = len(a)
    k_vals = np.arange(2 * n - 1)[:, None]
    j_vals = np.arange(n)[None, :]
    i_vals = k_vals - j_vals
    valid = (i_vals >= 0) & (i_vals < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[i_vals[valid]]
    return A

def line_search_armijo(a, g, f0, c1=1e-4, rho=0.5):
    """
    Backtracking line search with Armijo condition.
    Returns (t, f(t)) for the best step.

    d = g - a  # direction
    # Gradient approximation (use smoothed gradient)
    # For now, use a simple backtracking

        a_try = np.maximum(a_try, 0)

        if f_try <= f0 + c1 * t * (-f0):  # Armijo (descent)

def line_search_grid(a, g, n_evals=50):
    """Grid-based line search with parabolic refinement."""
    n = len(a)
    ts = np.linspace(0.01, 0.99, n_evals)
    objs = np.full(n_evals, np.inf)

        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try / s)
    
    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]
    
    # Parabolic refinement
    if 0 < best_idx < n_evals - 1:
        t0, t1, t2 = ts[best_idx-1], ts[best_idx], ts[best_idx+1]
        f0, f1, f2 = objs[best_idx-1], objs[best_idx], objs[best_idx+1]
        denom = f0 - 2.0*f1 + f2

                if f_par < best_obj:

def smoothed_gradient(a, tau):
    """Gradient of smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_v = np.exp(tau * (conv - max_c))
    s_exp = float(np.sum(exp_v))
    if s_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_v / s_exp
    
    N = 1
    while N < 3*n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]

    # Or more precisely, solve: max g^T * 1 s.t. A @ g = min(vals) * 1, g >= 0

    return vals, j_star

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_vec(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_vals = np.arange(2 * n - 1)[:, None]
    j_vals = np.arange(n)[None, :]
    i_vals = k_vals - j_vals
    valid = (i_vals >= 0) & (i_vals < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[i_vals[valid]]
    return A


def solve_lp_dir(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    if not HAS_CVXPY:
        return None, None
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_vec(a_norm)
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None

def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)

        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0

        p = exp_v / s_exp

    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    Af = fft(a, N)
    g_full = ifft(P * np.conj(Af)).real
    return 2.0 * g_full[:n]

def gd_polish(a, time_budget=60.0, tau_init=80.0, tau_end=1500.0):
    """Nesterov accelerated gradient descent on smoothed objective."""

    a = np.maximum(a, 0).copy().astype(np.float64)

        return a

    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    prev_a = a.copy()
    prev_g = None

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        g = smoothed_grad(y, tau)
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            lr = 0.8 * lr + 0.2 * min(max(adapt, 0.3 * 2.0 / np.sqrt(n)), 3.0 * 2.0 / np.sqrt(n))
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)

        if s_new < 1e-10:

        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9995)

def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 6):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_to_n(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] * n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        sym_seq = sym_seq / sym_seq.sum()
        a3 = lp_optimize(sym_seq, time_budget=p3_time, use_sym=True)
        # Also try CD on symmetric
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a3 = coordinate_descent(a3, time_budget=min(remaining * 0.3, 50.0))
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 Sym: obj={best_score:.9f} **")
        else:
            print(f"  Phase3 Sym: obj={obj3:.9f}")

    # === Phase 4: GD Polish ===
    print("\n=== Phase 4: GD Polish ===")
    remaining = deadline - time.monotonic()
    p4_time = min(remaining * 0.10, 100.0)
    if p4_time > 15:
        a4 = gd_polish(best_seq, time_budget=p4_time, tau_init=50.0, tau_end=1200.0)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4 GD: obj={best_score:.9f} **")
        else:
            print(f"  Phase4 GD: obj={obj4:.9f}")

    # === Phase 5: Multi-n Exploration ===
    print("\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best - 100, n_best + 100, n_best + 200,
        n_best - 200, n_best + 300,
        400, 500, 600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and n != n_best]
    n_options = n_options[:4]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        a_try = resample_to_n(best_seq, n_try)
        a_try = a_try / a_try.sum()
        # Brief LP
        alloc_lp = min(remaining * 0.35, 70.0)
        if alloc_lp > 15:
            a_try = lp_optimize(a_try, time_budget=alloc_lp)
        # Brief CD
        remaining = deadline - time.monotonic()
        alloc_cd = min(remaining * 0.20, 40.0)
        if alloc_cd > 10:
            a_try = coordinate_descent(a_try, time_budget=alloc_cd)
        # Brief GD
        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.10, 25.0)
        if alloc_gd > 8:
            a_try = gd_polish(a_try, time_budget=alloc_gd, tau_init=100, tau_end=800)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  n={n_try}: obj={best_score:.9f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.9f}")

    # === Phase 6: Diverse Restarts ===
    print("\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    restarts = [
        ("fourier_perturb", lambda: fourier_perturb(best_seq, rng, strength=0.04)),
        ("smooth_blend", None),
        ("anti_sym", None),
    ]

    for name, func in restarts:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        if name == "fourier_perturb":
            a_r = func()
        elif name == "smooth_blend":
            ks = max(5, n_cur // 12)
            if ks % 2 == 0:
                ks += 1
            a_sm = uniform_filter1d(best_seq, size=ks)
            a_r = 0.85 * best_seq + 0.15 * a_sm
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        else:
            # Anti-symmetric perturbation (shift energy to opposite side)
            a_shift = np.roll(best_seq, n_cur // 4)
            a_r = 0.7 * best_seq + 0.3 * a_shift
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

        alloc_lp = min(remaining * 0.3, 50.0)
        if alloc_lp > 12:
            a_r = lp_optimize(a_r, time_budget=alloc_lp)
        remaining = deadline - time.monotonic()
        alloc_cd = min(remaining * 0.15, 30.0)
        if alloc_cd > 10:
            a_r = coordinate_descent(a_r, time_budget=alloc_cd)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.9f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.9f}")

    # === Phase 7: Final Intensive Polish ===
    print("\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final LP
    if remaining > 30:
        alloc = min(remaining * 0.40, 200.0)
        a_f = lp_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.9f} **")

    # Final CD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 100.0)
        a_f2 = coordinate_descent(best_seq, time_budget=alloc, verbose=True)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final CD: obj={best_score:.9f} **")

    # Final high-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 60.0)
        a_f3 = gd_polish(best_seq, time_budget=alloc, tau_init=200.0, tau_end=3000.0)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final GD: obj={best_score:.9f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.9f}, time={elapsed:.1f}s ===")
    return result
