    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_at(a, conv, S):
    """Compute subgradient of max_k (a*a)[k] at a, given argmax set S."""
    n = len(a)
    grad = np.zeros(n)
    for k in S:
        k = int(k)
        # Part 1: a[k-j] for j where 0<=k-j<n
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            idx = k - np.arange(j_start, j_end + 1)
            grad[j_start:j_end+1] += a[idx]
        # Part 2: a[j-k] for j where 0<=j-k<n
        j_start2 = max(k, 0)
        j_end2 = min(k + n - 1, n - 1)
        if j_start2 <= j_end2:
            idx2 = np.arange(j_start2, j_end2 + 1) - k
            grad[j_start2:j_end2+1] += a[idx2]
    return grad / len(S)


def accelerated_subgradient(a_init, time_budget, verbose=True):
    """Nesterov-accelerated proximal subgradient on simplex."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x_prev = a.copy()
    x_curr = a.copy()
    t_val = 1.0
    
    it = 0
    best_it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        # Nesterov interpolation
        t_prev = t_val
        t_val = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0
        alpha = (t_prev - 1.0) / t_val
        
        y = x_curr + alpha * (x_curr - x_prev)
        y = project_simplex(y)
        
        # Compute subgradient at y
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        threshold = max_conv * (1.0 - 1e-8)
        S = np.where(conv >= threshold)[0]
        
        grad = subgradient_at(y, conv, S)
        
        # Step size
        eta = 2.0 / (t_val * (np.linalg.norm(grad) + 1e-15))
        
        # Update
        x_prev = x_curr.copy()
        x_curr = project_simplex(y - eta * grad)
        
        # Track best
        obj_curr = compute_obj(x_curr)
        if obj_curr < best_obj - 1e-14:
            best_obj = obj_curr
            best_a = x_curr.copy()
            best_it = it
            no_improve = 0
        else:
            no_improve += 1
        
        it += 1
        
        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - start
            print(f"    ASG it{it}: obj={best_obj:.8f}, eta={eta:.6f}, elapsed={elapsed:.0f}s")
        
        if no_improve > 50000:
            break
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g approximates conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A
