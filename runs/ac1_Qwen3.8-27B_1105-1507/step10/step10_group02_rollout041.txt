```python
    b = b / b.sum()  # normalize
    
    v = np.zeros(n)  # Nesterov momentum
    beta = 0.9
    
    for i in range(n_iter):
        tau = tau_schedule[i]
        
        # Nesterov lookahead
        y = b + beta * v
        y = np.maximum(y, 0)
        s = y.sum()
        if s < 1e-15:
            y = b.copy()
            s = 1.0
        else:
            y = y / s
        
        # Compute gradient of smoothed max
        conv = compute_conv(y)
        max_conv = conv.max()
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / exp_vals.sum()
        
        # Compute gradient via FFT cross-correlation
        L = 1
        while L < 3*n - 2:
            L *= 2
        P = rfft(p, L)  # p is real
        B = rfft(y, L)
        cross = irfft(conj(P) * B, L)[:n]
        grad = 2.0 * cross
        
        # Gradient step with adaptive learning rate
        grad_norm = np.sqrt((grad**2).sum())
        if grad_norm > 1e-15:
            # For Nesterov on convex problems, optimal step size is related to 
            # the distance to the feasible set and the gradient norm
            lr = 1.0 / (grad_norm + 1e-10)
            # Scale to be reasonable
            lr = min(lr, 10.0 / np.sqrt(n))
        else:
            lr = 0.0
        
        v = beta * v - lr * grad
        b_new = y + v
        b_new = np.maximum(b_new, 0)
        s_new = b_new.sum()
        if s_new < 1e-15:
            b_new = b.copy()
            v = np.zeros(n)
        else:
            b_new = b_new / s_new
        
        # Nesterov restart
        if np.dot(b_new - y, y - b) > 0:
            v = np.zeros(n)
            beta = 0.9
        else:
            beta += 0.01
            beta = min(beta, 0.999)
        
        b = b_new
    
    return b
```

```python
"""
Optimizes step function coefficients using a combination of:
1. Nesterov-accelerated smoothed gradient descent with adaptive temperature
2. Exact coordinate descent for local refinement
3. Active-set identification to focus on critical convolution indices
4. Multi-n exploration and diverse restarts
The key innovation is efficiently computing gradients via FFT-based cross-correlation
and using a multi-scale temperature schedule for fast convergence.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj_normalized(a):
    """Compute objective for normalized a (sum=1): 2n*max(conv)."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def compute_smoothed_obj_and_grad(b, tau):
    """
    Compute smoothed objective and gradient for normalized b (sum=1).
    Returns (smoothed_max, gradient) where gradient is w.r.t. b.
    """
    n = len(b)
    conv = compute_conv(b)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / exp_vals.sum()
    smoothed_max = max_conv + np.log(exp_vals.sum()) / tau
    
    # Gradient: grad_j = 2 * sum_k p_k * b_{k-j}
    # This is the cross-correlation of p (length 2n-1) and b (length n)
    L = 1
    while L < 3 * n - 2:
        L <<= 1
    P = fft(p, L)
    B = fft(b, L)
    cross = ifft(np.conj(P) * B).real[:n]
```

```python
def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def obj_normalized(a):
    n = len(a)
    if n == 0:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a)))

def compute_grad_smooth(b, tau, conv=None):
    """Compute gradient of tau-smoothed max of conv(b,b)."""
    n = len(b)
    if conv is None:
        conv = compute_conv(b)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    s = float(np.sum(exp_vals))
    if s < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / s
    # gradient_j = 2 * sum_k p_k * b_{k-j} = 2 * (p cross-corr b)_j
    L = 1
    while L < 3 * n - 2:
        L <<= 1
    P = fft(p, L)
    B = fft(b, L)
    cross = ifft(P * np.conj(B), L).real[:n]
    return 2.0 * cross, conv

def nesterov_descent(b, time_budget=300.0, tau_min=5.0, tau_max=3000.0,
                     verbose=False, seed=42):
    """Nesterov accelerated smoothed gradient descent on probability simplex."""
    n = len(b)
    b = np.maximum(b, 0).astype(np.float64).copy()
    s = float(np.sum(b))
    if s < 1e-10:
        return b
    b = b / s
    
    best_b = b.copy()
    best_obj = obj_normalized(b)
    start = time.monotonic()
    
    v = np.zeros(n)
    t = 0
    it_count = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(time_budget, 1e-6))
        
        # Temperature schedule: exponential from tau_min to tau_max
        tau = tau_min * (tau_max / tau_min) ** frac
        tau = max(tau, tau_min)
        
        t += 1
        # Nesterov parameter
        beta = (1.0 + 1.0 / np.sqrt(4.0 * t + 1.0)) / 2.0
        beta = min(beta, 0.9999)
        
        # Learning rate: for convex f with L-Lipschitz gradient, optimal lr = 1/(L*t)
        # L ≈ tau * 2n (smoothness of smoothed max)
        # But we also have the constraint, so we use a heuristic
        L_est = tau * 2.0 * n
        lr = 1.0 / (L_est * t) if t > 0 else 1.0 / L_est
        # Also cap by a reasonable value
        lr = min(lr, 2.0 / (np.sqrt(n) * np.sqrt(t + 1)))
        lr = max(lr, 1e-8)

        y = b + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-15:
            y = best_b.copy()
            s_y = 1.0
        else:
            y = y / s_y
        
        # Compute gradient
        conv = compute_conv(y)
        g, _ = compute_grad_smooth(y, tau, conv)
        
        # Gradient step
        v_new = beta * v - lr * g
        b_new = y + v_new
        
        # Project onto simplex (clip + normalize)
        b_new = np.maximum(b_new, 0)
        s_new = float(np.sum(b_new))
        if s_new < 1e-15:
            b_new = best_b.copy()
            v_new = np.zeros(n)
        else:
            b_new = b_new / s_new

def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """Exact coordinate descent with golden section search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        if sweep % 5 == 1:
            conv = compute_conv(a)
            obj = 2.0 * n * float(np.max(conv))
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            
            v = a[i]
            S_other = 1.0 - v  # since sum(a) = 1
            
            if S_other < 1e-12 and v < 1e-12:
                continue
            
            # k range affected by changing a[i]
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[np.clip(ks - i, 0, n-1)]
            c_k = conv[ks]
            
            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]
            
            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0
            
            def f_1d(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (1.0 - v + x) ** 2
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / (denom)
            
            # Golden section search
            # Initial bracket
            scale = max(v, 1.0 / n, 1e-6)
            x_lo = max(0.0, v - 3.0 * scale)
            x_hi = v + 3.0 * scale
            
            # Expand if needed
            f_lo = f_1d(x_lo)
            f_hi = f_1d(x_hi)
            f_mid = f_1d(v)
            
            for _ in range(40):
                if x_hi - x_lo < 1e-12 * max(1.0, x_hi):
                    break
                c_pt = x_hi - phi * (x_hi - x_lo)
                d_pt = x_lo + phi * (x_hi - x_lo)
                fc = f_1d(c_pt)
                fd = f_1d(d_pt)
                if fc < fd:
                    x_hi = d_pt
                else:
                    x_lo = c_pt
            
            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            
            if abs(x_opt - v) < 1e-14:
                continue
            
            delta = x_opt - v
            a[i] = x_opt
            
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
        
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
```

```python
"""
Optimizes step function coefficients using L-BFGS-B quasi-Newton optimization
with O(n) subgradient computation, combined with exact coordinate descent for
local refinement, multi-n exploration, and diverse restarts. Key innovations:
1) Exploiting the O(n) subgradient structure of max-convolution objective
2) Using L-BFGS-B quasi-Newton information for much faster convergence than
   first-order methods
3) Adaptive smoothing that switches between O(n) subgradient and O(n log n)
   FFT gradient based on the gap between top-2 convolution values
4) Multi-start with diverse initializations and n-length exploration
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_normalized(a):
    """Compute objective for normalized a (sum=1): 2n*max(conv)."""
    n = len(a)
    if n == 0:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a)))


def obj_unnormalized(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def objective_and_grad(a_flat):
    """
    Objective and (sub)gradient for L-BFGS-B.
    f(a) = 2n * max(conv(a,a)) / sum(a)^2
    Uses subgradient at the argmax of the convolution.
    """
    a = a_flat
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return 1e10, np.zeros(n)

    conv = compute_conv(a)
    M = float(np.max(conv))
    k_max = int(np.argmax(conv))

    f = 2.0 * n * M / (S * S)

    # Subgradient: df/da_i = 4n/S^3 * (S * a_{k*-i} - M)
    g = np.full(n, -4.0 * n * M / (S * S * S))
    j_min = max(0, k_max - n + 1)
    j_max = min(n - 1, k_max)
    idx = np.arange(j_min, j_max + 1)
    g[idx] += 4.0 * n * S / (S * S * S) * a[k_max - idx]
    # Simplify: g[idx] += 4n/(S*S) * a[k_max - idx]
    # So total: g[idx] = 4n/(S*S) * a[k_max-idx] - 4nM/(S*S*S)

    return f, g


def objective_and_grad_smooth(a_flat, tau=50.0):
    """
    Objective and smooth gradient using log-sum-exp approximation.
    """
    a = a_flat
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return 1e10, np.zeros(n)

    conv = compute_conv(a)
    max_c = float(np.max(conv))

    # Smoothed max
    exp_vals = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        smooth_M = max_c
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        smooth_M = max_c + np.log(sum_exp) / tau
        p = exp_vals / sum_exp

    f = 2.0 * n * smooth_M / (S * S)

    # Smooth gradient: d(smooth_M)/da_i = 2 * sum_k p_k * a_{k-i}
    # Use FFT for this
    L = 1
    while L < 3 * n - 2:
        L <<= 1
    P = fft(p, L)
    A = fft(a, L)
    cross = ifft(P * np.conj(A), L).real[:n]
    dM_da = 2.0 * cross

    # Full gradient: df/da_i = 2n/S^3 * (S * dM/da_i - 2*smooth_M)
    g = (2.0 * n / (S * S * S)) * (S * dM_da - 2.0 * smooth_M)

    return f, g


def l_bfgsb_optimize(a, time_budget=300.0, verbose=False, seed=42):
    """
    Main optimizer using L-BFGS-B with adaptive gradient (smooth or subgradient).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()

    # Determine whether to use smooth or subgradient
    conv = compute_conv(a)
    top2 = np.sort(conv)[-2:]
    use_smooth = (top2[1] - top2[0]) < 0.1 / n

    if use_smooth:
        func = lambda x: objective_and_grad_smooth(x, tau=50.0)
    else:
        func = objective_and_grad

    # Run L-BFGS-B
    try:
        res = minimize(
            func, a.copy(), method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options={'maxiter': 5000, 'maxls': 30, 'ftol': 1e-15,
                     'gtol': 1e-12, 'disp': False}

def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """
    Exact coordinate descent with golden section search on 1D subproblem.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    phi = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1
        if sweep % 5 == 1:
            conv = compute_conv(a)
            obj = 2.0 * n * float(np.max(conv))
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = 1.0 - v

            if S_other < 1e-12 and v < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)

            ks = np.arange(k_min, k_max + 1)
            a_ki = a[np.clip(ks - i, 0, n - 1)]
            c_k = conv[ks]

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            def f_1d(x, v=v, c_k_lin=c_k_lin, a_ki_lin=a_ki_lin, c_2i=c_2i):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (1.0 - v + x) ** 2
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Golden section search with adaptive bracket
            scale = max(v, 1.0 / n, 1e-6)
            x_lo = max(0.0, v - 4.0 * scale)
            x_hi = v + 4.0 * scale

            for _ in range(45):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                c_pt = x_hi - phi * (x_hi - x_lo)
                d_pt = x_lo + phi * (x_hi - x_lo)
                fc = f_1d(c_pt)
                fd = f_1d(d_pt)
                if fc < fd:
                    x_hi = d_pt
                else:
                    x_lo = c_pt

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

    return best_a


def nesterov_descent(b, time_budget=300.0, tau_min=5.0, tau_max=3000.0,
                     verbose=False, seed=42):
    """Nesterov accelerated smoothed gradient descent on probability simplex."""
    n = len(b)
    b = np.maximum(b, 0).astype(np.float64).copy()
    s = float(np.sum(b))
    if s < 1e-10:
        return b
    b = b / s

    best_b = b.copy()
    best_obj = obj_normalized(b)
    start = time.monotonic()

    v = np.zeros(n)
    t = 0
    it_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(time_budget, 1e-6))

        # Exponential temperature schedule
        tau = tau_min * (tau_max / tau_min) ** (frac ** 0.7)
        tau = max(tau, tau_min)

        t += 1
        beta = min(0.9999, (1.0 + 1.0 / np.sqrt(4.0 * t + 1.0)) / 2.0)

        # Adaptive learning rate
        conv = compute_conv(b)
        max_c = float(np.max(conv))
        grad, _ = compute_grad_for_nesterov(b, tau, conv)
        gnorm = float(np.sqrt(np.sum(grad * grad)))
        if gnorm < 1e-15:
            break
        lr = min(2.0 / (np.sqrt(n) * np.sqrt(t + 1)), 1.0 / (gnorm * tau + 1e-10))
        lr = max(lr, 1e-10)

        # Nesterov step
        y = b + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-15:
            y = best_b.copy()
            s_y = 1.0
        else:
            y = y / s_y

        # Recompute gradient at y for better accuracy
        conv_y = compute_conv(y)
        grad_y, _ = compute_grad_for_nesterov(y, tau, conv_y)

        v = beta * v - lr * grad_y
        b_new = y + v
        b_new = np.maximum(b_new, 0)
        s_new = float(np.sum(b_new))
        if s_new < 1e-15:
            b_new = best_b.copy()
            v = np.zeros(n)
        else:
            b_new = b_new / s_new

        if it_count % 200 == 0:
            obj = obj_normalized(b_new)
            if obj < best_obj:
                best_obj = obj
                best_b = b_new.copy()
                if verbose:
                    print(f"      Nesterov it={it_count}, obj={best_obj:.8f}, tau={tau:.1f}")

        v = np.clip(v, -1.0, 1.0)
        b = b_new
        it_count += 1

    return best_b


def compute_grad_for_nesterov(b, tau, conv):
    """Compute gradient with O(n) shortcut when single max dominates."""
    n = len(b)
    max_c = float(np.max(conv))
    sorted_conv = np.sort(conv)
    gap = (sorted_conv[-1] - sorted_conv[-2]) if len(sorted_conv) >= 2 else 0

    if gap * tau > 40.0:
        # Single max dominates - use O(n) formula
        k_max = int(np.argmax(conv))
        g = np.zeros(n)
        j_min = max(0, k_max - n + 1)
        j_max = min(n - 1, k_max)
        idx = np.arange(j_min, j_max + 1)
        g[idx] = 2.0 * b[k_max - idx]
        return g, conv
    else:
        # Use FFT for full smoothed gradient
        exp_vals = np.exp(tau * (conv - max_c))
        s = float(np.sum(exp_vals))
        if s < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = exp_vals / s
        L = 1
        while L < 3 * n - 2:
            L <<= 1
        P = fft(p, L)
        B = fft(b, L)
        cross = ifft(P * np.conj(B), L).real[:n]
        return 2.0 * cross, conv


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] * n / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x ** 2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 10, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Triangle-wave based
        x = np.linspace(0, 1, n)
        a = np.abs(2 * x - 1)
        a = a / a.sum()
    elif mode == 6:
        # Two-peak with uniform base
        a = np.ones(n) * 0.3
        peaks = [n // 3, 2 * n // 3]
        for p in peaks:
            width = n // 6
            for j in range(max(0, p - width), min(n, p + width)):
                a[j] += 1.0
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: L-BFGS-B + Nesterov + coordinate descent + multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 12.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = obj_normalized(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = obj_normalized(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = obj_normalized(best_seq)

    n_best = len(best_seq)

    # === Phase 1: L-BFGS-B (main workhorse) ===
    print("\n=== Phase 1: L-BFGS-B Optimization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 500.0)
    if p1_budget > 60:
        # Run multiple L-BFGS-B with different starting perturbations
        for attempt in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p1_budget / (3 - attempt), remaining * 0.7)
            if alloc < 20:
                break
            if attempt == 0:
                a_start = best_seq.copy()
            else:
                pert = rng.normal(0, 0.02 * attempt, n_best)
                a_start = best_seq * np.exp(pert)
                a_start = np.maximum(a_start, 0)
                a_start = a_start / a_start.sum()

            a_opt = l_bfgsb_optimize(a_start, time_budget=alloc, verbose=True,
                                     seed=seed + attempt * 100)
            obj_opt = obj_normalized(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  L-BFGS-B attempt {attempt}: obj={best_score:.8f} **")
            else:
                print(f"  L-BFGS-B attempt {attempt}: obj={obj_opt:.8f}")

    # === Phase 2: Nesterov Smoothed Descent ===
    print("\n=== Phase 2: Nesterov Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 350.0)
    if p2_budget > 40:
        a_nd = nesterov_descent(best_seq, time_budget=p2_budget,
                                tau_min=10.0, tau_max=5000.0,
                                verbose=True, seed=seed)
        obj_nd = obj_normalized(a_nd)
        if obj_nd < best_score:
            best_score = obj_nd
            best_seq = a_nd.copy()
            n_best = len(best_seq)
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_nd:.8f}")

    # === Phase 3: Coordinate Descent ===
    print("\n=== Phase 3: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.25, 350.0)
    if p3_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p3_budget,
                                  verbose=True, seed=seed)
        obj_cd = obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print("\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        n_best - 50, n_best + 50, n_best + 100, n_best + 200,
        600, 700, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) > 30]
    n_options = n_options[:6]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # L-BFGS-B on new n
        alloc1 = min(remaining * 0.2, 50.0)
        if alloc1 > 15:
            a_try = l_bfgsb_optimize(a_try, time_budget=alloc1, verbose=False, seed=n_try)

        # Coordinate descent on new n
        alloc2 = min(remaining * 0.2, 50.0)
        if alloc2 > 15:
            a_try = coordinate_descent(a_try, time_budget=alloc2, verbose=False, seed=n_try)

        obj_try = obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts ===
    print("\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = 1 + r
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.4 + 0.15 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc1 = min(remaining * 0.2, 40.0)
        alloc2 = min(remaining * 0.15, 30.0)
        if alloc1 > 10:
            a_r = l_bfgsb_optimize(a_r, time_budget=alloc1, verbose=False, seed=6000 + r * 100)
        if alloc2 > 10:
            a_r = coordinate_descent(a_r, time_budget=alloc2, verbose=False, seed=7000 + r * 100)

        obj_r = obj_normalized(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 6: Final Polish ===
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # L-BFGS-B one more time
        alloc1 = min(remaining * 0.4, 150.0)
        if alloc1 > 20:
            a_f1 = l_bfgsb_optimize(best_seq, time_budget=alloc1, verbose=False,
                                    seed=seed + 9999)
            obj_f1 = obj_normalized(a_f1)
            if obj_f1 < best_score:
                best_score = obj_f1
                best_seq = a_f1.copy()
                print(f"  Final LBFGS: obj={best_score:.8f} **")

        # Coordinate descent
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.5, 200.0)
        if alloc2 > 20:
            a_f2 = coordinate_descent(best_seq, time_budget=alloc2, verbose=False,
                                      seed=seed + 8888)
            obj_f2 = obj_normalized(a_f2)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < np.inf:
            best_score = val
        else:
            result = [1.0] * 100
            best_score = 2.0
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
