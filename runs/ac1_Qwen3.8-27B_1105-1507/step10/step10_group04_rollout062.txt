def build_conv_matrix(a):

        A[start:end, j] = a[:end - start]
    return A

But wait, that's for periodic convolution. For aperiodic (linear) convolution, the relationship is different. The linear convolution theorem says: `FFT(linear_conv(a,a)) = FFT(a) * FFT(a)` (pointwise product, not |FFT(a)|^2).

Actually: `FFT(linear_conv(a, b)) = FFT(a) * FFT(b)` (with zero-padding). So for a=b: `FFT(conv(a,a)) = FFT(a)^2`.

If we want `conv(a,a)` to be constant (= c), then `FFT(conv(a,a))` should be c at frequency 0 and 0 elsewhere. from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a, b=None):

def build_conv_matrix(a):
    """Matrix A (2n-1 x n) where A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):

def solve_lp(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0.
    Returns g or None."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            kwargs = {}
            if solver == cp.SCS:
                kwargs = {'eps': 1e-9, 'max_iters': 20000}
            elif solver == cp.ECOS:
                kwargs = {'abstol': 1e-9}
            prob.solve(solver=solver, **kwargs)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value).flatten(), 0.0)
                s = float(np.sum(g_val))
                if s > 1e-12:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_with_line_search(a, time_budget, verbose=True):
    """LP-based polishing with golden-section line search on mixing parameter."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_best_t = 0.5

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 5:
            break

        g_lp = solve_lp(a)
        if g_lp is None:
            break

        # Golden section search for optimal t in [0.01, 1.0]
        phi = (np.sqrt(5) - 1) / 2  # ~0.618
        t_lo, t_hi = 0.01, 1.0
        t1 = t_hi - phi * (t_hi - t_lo)
        t2 = t_lo + phi * (t_hi - t_lo)
        obj_t1 = compute_obj(np.maximum((1 - t1) * a + t1 * g_lp, 0))
        obj_t2 = compute_obj(np.maximum((1 - t2) * a + t2 * g_lp, 0))

        for _ in range(12):
            t_hi = min(1.0, remaining * 0.01 + t_hi)
            if t_hi - t_lo < 0.005:
                break
            if obj_t1 < obj_t2:
                t_hi = t2
                t2 = t1
                obj_t2 = obj_t1
                t1 = t_hi - phi * (t_hi - t_lo)
                obj_t1 = compute_obj(np.maximum((1 - t1) * a + t1 * g_lp, 0))
            else:
                t_lo = t1
                t1 = t2
                obj_t1 = obj_t2
                t2 = t_lo + phi * (t_hi - t_lo)
                obj_t2 = compute_obj(np.maximum((1 - t2) * a + t2 * g_lp, 0))

        t_opt = (t_lo + t_hi) / 2
        obj_opt = compute_obj(np.maximum((1 - t_opt) * a + t_opt * g_lp, 0))

        if obj_opt < best_obj - 1e-13:
            best_obj = obj_opt
            best_a = np.maximum((1 - t_opt) * a + t_opt * g_lp, 0).astype(np.float64)

            a = best_a
            stall_count = 0

                break

        if verbose:
            print(f"    LP: obj={best_obj:.10f}, t={t_opt:.4f}, t={elapsed:.0f}s")

    """FFT-accelerated smoothed subgradient descent in softmax parameterization."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / float(np.sum(a))

    log_w = np.log(a.copy())
    log_w -= log_w.mean()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    no_improve = 0
    v = np.zeros(n)
    beta = 0.85

    N = 1
    while N < 3 * n:
        N <<= 1

    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.6)

        # Current a from softmax
        lw = log_w - log_w.max()
        a = np.exp(lw)
        a = a / float(np.sum(a))

        # Compute gradient of smoothed max(conv(a,a))
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        shifted = tau * (conv - max_c)
        exp_v = np.exp(shifted)
        sum_exp = float(np.sum(exp_v))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = exp_v / sum_exp

        # Gradient w.r.t. a: grad_a[i] = 2 * sum_k p[k] * a[k-i]
        P = fft(p, N)
        A_f = fft(a, N)
        corr = ifft(P * np.conj(A_f), N).real
        grad_a = 2.0 * corr[:n]

        # Chain rule: grad_log_w[j] = a[j] * (grad_a[j] - sum(grad_a * a))
        inner = float(np.sum(grad_a * a))
        grad_bw = a * (grad_a - inner)

        gn = float(np.linalg.norm(grad_bw))
        if gn < 1e-15:
            break

        lr = min(1.0 / (gn + 1e-10), 3.0 / np.sqrt(n))
        v = beta * v - lr * grad_bw
        log_w_new = log_w + v

        lw_new = log_w_new - log_w_new.max()
        a_new = np.exp(lw_new)
        a_new = a_new / float(np.sum(a_new))

        obj_new = compute_obj(a_new)

            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.5
            log_w_new = log_w + v
            lw_new = log_w_new - log_w_new.max()
            a_new = np.exp(lw_new)
            a_new = a_new / float(np.sum(a_new))

def resample_to(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = x * (1 - x)
    elif mode == 4:
        a = np.exp(-((np.arange(n) - n / 2) ** 2) / (n * 0.05))
    elif mode == 5:
        a = (1 + np.cos(2 * np.pi * np.arange(n) / n * 3)) / 2
        a = np.maximum(a, 0.01)
    elif mode == 6:
        a = rng.uniform(0, 1, n)
        a = np.convolve(a, np.ones(5) / 5, mode='same')
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
