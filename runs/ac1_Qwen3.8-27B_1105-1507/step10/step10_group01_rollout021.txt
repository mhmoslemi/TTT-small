"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a novel combination of:
1. LP-driven large moves with adaptive line search
2. Mirror descent (exponential) with FFT-accelerated multi-peak smoothed gradient
3. Adaptive tau scheduling based on peak gap
4. Aggressive multi-n exploration with structured resampling
5. Phase-interleaved LP + mirror descent for continuous improvement
Key difference from previous: mirror descent naturally handles non-negativity,
multi-peak subgradient avoids over-focusing, and adaptive LR gives faster convergence.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_gradient_fft(a, P):
    """
    Compute gradient: grad[p] = 2 * sum_q P[p+q] * a[q]
    where P is length 2n-1 (probability weights on convolution indices).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    # Need N >= (2n-1) + n - 1 = 3n - 2 to avoid wrap-around
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    P_full = np.zeros(N)
    P_full[:len(P)] = P
    a_full = np.zeros(N)
    a_full[:n] = a
    # Cross-correlation: ifft(fft(P_full) * conj(fft(a_full)))
    result = ifft(fft(P_full) * np.conj(fft(a_full))).real
    return 2.0 * result[:n]


def smoothed_gradient(a, tau, top_k=None):
    """
    Compute smoothed gradient of max_k (a*a)[k] using softmax weighting.
    If top_k is specified, only weight the top_k peaks.
    Returns (gradient, conv_max, P).
    """
    n = len(a)
    conv = compute_conv(a)
    conv_max = float(np.max(conv))

    if top_k is not None and top_k < len(conv):
        # Use only top-k peaks
        idx = np.argpartition(conv, -top_k)[-top_k:]
        P = np.zeros(len(conv))
        shifted = conv[idx] - conv_max
        weights = np.exp(tau * shifted)
        P[idx] = weights / weights.sum()
    else:
        # Full softmax
        shifted = conv - conv_max
        weights = np.exp(tau * shifted)
        P = weights / weights.sum()

    grad = compute_gradient_fft(a, P)
    return grad, conv_max, P


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_minpeak(a, epsilon=0.0):
    """
    Solve LP: min M + epsilon*sum(g^2) s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    if epsilon > 0:
        obj = cp.Minimize(M + epsilon * cp.sum_squares(g))
    else:
        obj = cp.Minimize(M)

    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(obj, constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, direction, n_evals=20):
    """Line search: find best t in [0,1] to minimize obj((1-t)*a + t*direction)."""
    n = len(a)
    ts = np.linspace(0.01, 0.99, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try)

    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]

    # Parabolic refinement
    if 0 < best_idx < n_evals - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.01, 0.99))
            a_par = (1 - t_par) * a + t_par * direction
            a_par = np.maximum(a_par, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                f_par = compute_obj(a_par)
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par

    return best_t, best_obj

    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iter_count = 0
    stall = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        g, M_lp = solve_lp_minpeak(a)
        if g is None:
            break

        best_t, obj_new = line_search(a, g, n_evals=18)
        iter_count += 1

        if obj_new < best_obj - 1e-13:
            a_new = (1 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                if verbose and (iter_count % 5 == 1):
                    el = time.monotonic() - start
                    print(f"    LP#{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, M_lp={M_lp:.6f}, ({el:.0f}s)")
        else:
            stall += 1
            # Try small steps
            improved = False
            for t_s in [0.05, 0.02, 0.01]:
                a_s = (1 - t_s) * a + t_s * g
                a_s = np.maximum(a_s, 0)
                s_s = float(np.sum(a_s))
                if s_s > 1e-10:
                    a_s = a_s / s_s
                    obj_s = compute_obj(a_s)
                    if obj_s < best_obj - 1e-13:
                        best_obj = obj_s
                        best_a = a_s.copy()
                        a = a_s
                        stall = 0
                        improved = True
                        break

            if not improved and stall >= 4:
                # Perturb to escape
                a = a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

            if verbose and iter_count % 10 == 0:
                print(f"    LP#{iter_count}: stalled, obj={best_obj:.8f}")

            if iter_count > 80 and stall >= 8:
                break

    return best_a


def mirror_descent(a, time_budget=200.0, tau_init=50.0, tau_end=2000.0,
                   top_k=None, verbose=False):
    """
    Mirror descent (exponential) optimization with adaptive tau.
    Update: a_new = a * exp(-lr * grad) / sum(a * exp(-lr * grad))
    This naturally maintains positivity.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    lr = 5.0 / np.sqrt(n)
    
    # Momentum in log-space
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.2

        # Compute gradient
        grad, conv_max, P = smoothed_gradient(a, tau, top_k=top_k)
        grad_norm = float(np.linalg.norm(grad))

        if grad_norm < 1e-15:
            break

        # Adaptive learning rate
        lr_target = 1.0 / (grad_norm * np.sqrt(n) + 1e-10)
        lr_target = min(max(lr_target, 0.5 / np.sqrt(n)), 10.0 / np.sqrt(n))
        lr = 0.8 * lr + 0.2 * lr_target

        # Mirror descent step with momentum
        # In log-space: log(a_new) = log(a) - lr * grad + momentum
        log_a = np.log(a + 1e-30)
        v = 0.9 * v - lr * grad
        log_a_new = log_a - v
        a_new = np.exp(log_a_new - log_a_new.max())
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        # Periodic check: if stuck, reduce lr
        if it - best_it > 3000:
            lr *= 0.5
            v = np.zeros(n)
            if verbose:
                print(f"    MD: reset at it={it}, obj={best_obj:.8f}, lr={lr:.4f}")

        if verbose and it % 5000 == 0:
            print(f"    MD it={it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.4f}")

    return best_a


def gd_polish(a, time_budget=100.0, tau_init=100.0, tau_end=3000.0, verbose=False):
    """Nesterov accelerated gradient descent (projected) for final polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        grad, _, _ = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(grad))

        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.3 * base_lr), 3.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adapt

        # BB step
        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = grad - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-9), 5.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = grad.copy()
        a = a_new
        it += 1

        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it

    if verbose:
        print(f"    GD: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: interleaved LP + mirror descent with multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # === Initialization ===
    # Try to use parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if parent is None:
        # Try several initial n values
        for n_init in [600, 800, 1000]:
            a_try = np.ones(n_init) / n_init
            best_score = compute_obj(a_try)
            if best_score < best_score:  # always true for first
                best_score = best_score
                best_seq = a_try.copy()
        # Actually just use n=800 as default
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_cur = len(best_seq)
    print(f"Starting: n={n_cur}, obj={best_score:.8f}")

    # === Phase 1: LP-driven optimization (main workhorse) ===
    print(f"\n=== Phase 1: LP Optimization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 450.0)
    if p1_budget > 30:
        a_lp = lp_optimize(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.8f}")

    # === Phase 2: Mirror descent polish ===
    print(f"\n=== Phase 2: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 150.0)
    if p2_budget > 15:
        a_md = mirror_descent(best_seq, time_budget=p2_budget,
                              tau_init=50.0, tau_end=1500.0,
                              top_k=10, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Phase2 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 MD: obj={obj_md:.8f}")

    # === Phase 3: Second LP round ===
    print(f"\n=== Phase 3: Second LP Round ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.18, 250.0)
    if p3_budget > 20:
        a_lp2 = lp_optimize(best_seq, time_budget=p3_budget, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase3 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 LP: obj={obj_lp2:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_candidates = sorted(set([
        max(400, n_best - 200),
        max(400, n_best - 100),
        min(2000, n_best + 100),
        min(2000, n_best + 200),
        min(2000, n_best + 400),
        600, 900, 1100, 1300
    ]))
    # Filter: must be different from current by at least 50
    n_candidates = [n for n in n_candidates if abs(n - n_best) >= 50]
    n_candidates = n_candidates[:5]
    print(f"  Trying n values: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # LP
        alloc = min(remaining * 0.25, 50.0)
        if alloc > 12:
            a_try = lp_optimize(a_try, time_budget=alloc, verbose=False)

        # Mirror descent
        remaining = deadline - time.monotonic()
        alloc_md = min(remaining * 0.08, 25.0)
        if alloc_md > 8:
            a_try = mirror_descent(a_try, time_budget=alloc_md,
                                   tau_init=50.0, tau_end=1000.0, top_k=8)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Final intensive optimization ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau mirror descent
    if remaining > 20:
        alloc = min(remaining * 0.3, 120.0)
        a_final = mirror_descent(best_seq, time_budget=alloc,
                                 tau_init=200.0, tau_end=5000.0,
                                 top_k=5, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 200.0)
        a_final2 = lp_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final GD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 80.0)
        a_final3 = gd_polish(best_seq, time_budget=alloc,
                             tau_init=500.0, tau_end=5000.0, verbose=True)
        obj_final3 = compute_obj(a_final3)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # === Validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
