import numpy as np
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_norm, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm
    return A


def solve_lp_precise(a, warm_g=None, verbose=False):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm, n)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_g is not None:
        g.value = warm_g.copy()

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            kwargs = {}

def peak_shave(a, time_budget, verbose=False):
    """Identify the peak of conv(a,a) and redistribute mass to reduce it."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_val = conv[k_star]

        # Subgradient direction: reduce a[j] where a[j]*a[k_star-j] is large
        # This means reducing values that contribute most to the peak
        contrib = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                contrib[j] += a[j] * a[idx]
            if idx != j:
                contrib[j] += a[idx] * a[j]  # double counting correction handled below

        # Actually, the subgradient of conv[k_star] w.r.t. a[j] is:
        # a[k_star-j] if k_star-j != j, or 2*a[k_star-j] if k_star-j == j

        # = 2*a[k-j] if k-j != j, a[k-j] if k-j == j
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                if idx == j:
                    grad[j] = a[idx]
                else:
                    grad[j] = 2.0 * a[idx]

        # Projected gradient step: reduce along grad direction

        eta_candidates = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]
        best_eta_obj = best_obj

        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.4

        y = a + beta * v
        y = np.maximum(y, 0)

                            val += p[k] * y[idx]

                                pass

                g_direct[m] = val

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:

        # Adaptive step size

        elif mode == 1:
            # Edge emphasis (increase edges, decrease middle)
            weights = np.linspace(1, 0.3, n)
            a_pert = a * weights
        elif mode == 2:
            # Random scaling of random subset
            mask = rng.random(n) < 0.3
            a_pert = a.copy()
            a_pert[mask] *= rng.uniform(0.5, 2.0, mask.sum())
        else:
            # Smooth random perturbation
            freqs = rng.uniform(0.5, 5.0, 3)
            phases = rng.uniform(0, 2 * np.pi, 3)
            t = np.linspace(0, 1, n)
            noise = sum(np.exp(2j * np.pi * f * t + p) for f, p in zip(freqs, phases)).real
            noise = noise / (np.abs(noise).max() + 1e-10) * 0.1
            a_pert = a * (1.0 + noise)

        a_pert = np.maximum(a_pert, 0)
        s_pert = float(np.sum(a_pert))
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        # Quick optimization
        alloc = min(per_pert - 5, remaining - 10)
        if alloc < 5:
            continue

        # LP polish
        lp_alloc = alloc * 0.5
        if lp_alloc > 3:
            a_pert = lp_line_search(a_pert, time_budget=lp_alloc, verbose=False)

        # Peak shave
        ps_alloc = min(alloc * 0.25, remaining - 20)
        if ps_alloc > 3:
            a_pert = peak_shave(a_pert, time_budget=ps_alloc, verbose=False)

        # Gradient descent
        gd_alloc = min(alloc * 0.25, remaining - 15)
        if gd_alloc > 3:
            a_pert = softmax_grad_descent(a_pert, time_budget=gd_alloc,
                                          tau_start=200, tau_end=8000, verbose=False)

        obj_pert = compute_obj(a_pert)
        if obj_pert < best_obj:
            best_obj = obj_pert
            best_a = a_pert.copy()
            if verbose:
                print(f"    Perturb {i} (mode={mode}): obj={best_obj:.8f} **")

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if n == 2:
        x_old = np.array([0.0, 1.0])
    else:
        x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def window_init(n, window_type='hann'):
    t = np.linspace(0, 1, n)
    if window_type == 'hann':
        a = 0.5 * (1 - np.cos(2 * np.pi * t))
    elif window_type == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    elif window_type == 'raised_cos2':
        a = np.cos(np.pi * (t - 0.5)) ** 2
    elif window_type == 'kaiser_approx':
        # Approximate Kaiser with beta=5
        x = (t - 0.5) * 2  # [-1, 1]
        a = np.exp(-5.0 * abs(x) ** 1.5)
    elif window_type == 'flat_top':
        # Flat-top window approximation
        a = 0.2156 - 0.4160 * np.cos(2 * np.pi * t) + 0.2780 * np.cos(4 * np.pi * t) \
            - 0.0835 * np.cos(6 * np.pi * t) + 0.0069 * np.cos(8 * np.pi * t)
    elif window_type == 'sqrt_hann':
        a = np.sqrt(0.5 * (1 - np.cos(2 * np.pi * t)))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def optimize_at_n(a_init, time_budget, verbose=False):
    """Full optimization pipeline for a given initial sequence."""
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(len(a_init)) / len(a_init)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: LP line search (40% budget)
    if time_budget > 20:
        alloc = time_budget * 0.35
        a_new = lp_line_search(a, time_budget=alloc, verbose=verbose)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new
        remaining = time_budget - (time.monotonic() - start)

    # Phase 2: Peak shaving (15% budget)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.2, time_budget * 0.15)
        a_new = peak_shave(best_a, time_budget=alloc, verbose=verbose)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new

    # Phase 3: Softmax GD (20% budget)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.3, time_budget * 0.2)
        a_new = softmax_grad_descent(best_a, time_budget=alloc,
                                     tau_start=200, tau_end=15000, verbose=verbose)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new

    # Phase 4: Another LP pass (10% budget)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.15, time_budget * 0.1)
        a_new = lp_line_search(best_a, time_budget=alloc, verbose=False)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new

    # Phase 5: Random perturbations (remaining)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        n_pert = max(2, int(remaining / 30))
        a_new = random_perturb_refine(best_a, time_budget=remaining * 0.7,
                                      n_perturbations=n_pert, verbose=verbose)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP polishing with peak shaving, softmax GD,
    wide n-sweep, random perturbations, and multi-structure initialization.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = window_init(800, 'hann')
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, obj={best_score:.8f}")

    # === Phase A: Optimize at current n (300s) ===
    print(f"\n=== Phase A: Optimize at n={n_best} ===")
    remaining = deadline - time.monotonic()
    pA_budget = min(remaining * 0.2, 300.0)
    if pA_budget > 30:
        a_opt = optimize_at_n(best_seq, time_budget=pA_budget, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase A: obj={best_score:.8f} **")
        else:
            print(f"  Phase A: obj={obj_opt:.8f} (no improvement)")

    # === Phase B: Wide n-sweep (400s) ===
    print(f"\n=== Phase B: Wide n-sweep ===")
    remaining = deadline - time.monotonic()
    pB_budget = min(remaining * 0.35, 400.0)

    # Generate n candidates: focus around current best, also try extremes
    n_candidates = []
    for delta in range(-400, 801, 100):
        n_try = n_best + delta
        if 200 <= n_try <= 3000:
            n_candidates.append(n_try)
    # Also try specific values
    for n_fixed in [300, 400, 500, 600, 700, 800, 900, 1000, 1200, 1500, 2000, 2500]:
        if n_fixed not in n_candidates and 200 <= n_fixed <= 3000:
            n_candidates.append(n_fixed)

    n_candidates = sorted(set(n_candidates))
    n_candidates = [n for n in n_candidates if n != n_best]

    # Sort by proximity to current best n (likely to be better nearby)
    n_candidates.sort(key=lambda x: abs(x - n_best))
    n_candidates = n_candidates[:12]  # Limit to 12 values
    print(f"  n candidates: {n_candidates}")

    per_n_budget = pB_budget / max(1, len(n_candidates))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_n_budget, remaining - 20)
        if alloc < 20:
            break

        # Initialize with resampled best + window blend
        a_resampled = resample_sequence(best_seq, n_try)
        a_window = window_init(n_try, 'hann')
        a_init = 0.7 * a_resampled + 0.3 * a_window
        a_init = np.maximum(a_init, 0)
        s_init = float(np.sum(a_init))
        if s_init > 1e-10:
            a_init = a_init / s_init

        # Also try pure window
        a_window2 = window_init(n_try, 'blackman')
        obj_w2 = compute_obj(a_window2)

        # Optimize
        a_opt = optimize_at_n(a_init, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)

        # Also try window directly with quick optimization
        a_opt_w = optimize_at_n(a_window2, time_budget=alloc * 0.5, verbose=False)
        obj_opt_w = compute_obj(a_opt_w)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        elif obj_opt_w < best_score:
            best_score = obj_opt_w
            best_seq = a_opt_w.copy()
            n_best = n_try
            print(f"  ** n={n_try} (window): obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: best={min(obj_opt, obj_opt_w):.8f} (no improvement)")

    # === Phase C: Diverse structure exploration (200s) ===
    print(f"\n=== Phase C: Diverse structures ===")
    remaining = deadline - time.monotonic()
    pC_budget = min(remaining * 0.15, 200.0)

    if pC_budget > 30:
        window_types = ['hann', 'blackman', 'raised_cos2', 'kaiser_approx', 'flat_top', 'sqrt_hann']
        per_struct = pC_budget / len(window_types)

        for wt in window_types:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(per_struct, remaining - 10)
            if alloc < 10:
                break

            a_w = window_init(n_best, wt)
            # Blend with current best
            a_blend = 0.5 * a_w + 0.5 * (resample_sequence(best_seq, n_best) if len(best_seq) != n_best else best_seq)
            a_blend = np.maximum(a_blend, 0)
            s_blend = float(np.sum(a_blend))
            if s_blend > 1e-10:
                a_blend = a_blend / s_blend

            a_opt = optimize_at_n(a_blend, time_budget=alloc, verbose=False)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** {wt}: obj={best_score:.8f} **")
            else:
                print(f"  {wt}: obj={obj_opt:.8f}")

    # === Phase D: Intensive final polish (remaining) ===
    print(f"\n=== Phase D: Final intensive polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s, current best: {best_score:.8f}")

    # Multiple rounds of the full pipeline
    for round_idx in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = remaining * 0.3
        a_opt = optimize_at_n(best_seq, time_budget=alloc, verbose=(round_idx == 0))
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Round {round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  Round {round_idx}: obj={obj_opt:.8f} (stable)")

    # Final micro-optimizations
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 60.0)
        a_opt = lp_line_search(best_seq, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.3, 40.0)
        a_opt = peak_shave(best_seq, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final PeakShave: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 40.0)
        a_opt = softmax_grad_descent(best_seq, time_budget=alloc,
                                     tau_start=1000, tau_end=50000, verbose=False)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
