    Returns (g_normalized, M_used) or (None, None).
    """
    n = len(a)
    if n > 3000:
        return None, None
    
    if M is None:
        conv_aa = compute_conv(a)
        M = float(np.max(conv_aa))
    
    # Build convolution matrix A: (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A_matrix = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_matrix[j:j+n, j] = a
    
    g_var = cp.Variable(n, nonneg=True)
    constraints = [A_matrix @ g_var <= M, cp.sum(g_var) >= 1]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate', 'optimal_unknown'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, M
        except Exception:
            continue
    return None, None

def gradient_polish(a, time_budget, tau_levels=None, verbose=False):
    """Nesterov accelerated gradient descent with log-sum-exp smoothing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if tau_levels is None:
        tau_levels = [10, 50, 200, 800, 2000, 5000]
    
    beta = 0.9
    it = 0
    
    for tau in tau_levels:
        if time.monotonic() - start > time_budget - 5:
            break
        lr = 2.0 / np.sqrt(n)
        v = np.zeros(n)
        it_level = 0
        max_per_level = 3000
        local_best = best_obj
        
        while time.monotonic() - start < time_budget - 5:
            if it_level >= max_per_level:
                break
            
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy
            
            g, _ = smoothed_grad(y, tau)
            
            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
            if obj_new < local_best - 1e-15:
                local_best = obj_new
            
            a = a_new
            it += 1
            it_level += 1
            lr = max(1e-8, lr * 0.9999)
        
        if verbose:
            print(f"    GD tau={tau}: obj={best_obj:.10f}")
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_init(n, mode, rng=None):
    """Create diverse initial sequences."""
    if rng is None:
        rng = np.random.default_rng(0)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.0001))
        return a / a.sum()
    elif mode == 3:
        a = rng.uniform(0.1, 1, n)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(0, np.pi, n)
        a = np.maximum(0.5 + 0.5 * np.cos(x), 0.001)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 2 * x + rng.uniform(0, 6.28))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 6:
        a = np.exp(-np.abs(np.arange(n) - n/2) / (n/5))
        return a / a.sum()
    else:
        return np.ones(n) / n

def lp_solve_minM(a, verbose=False):
    """Solve: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0."""
    n = len(a)
    if n > 3000:
        return None, None
    
    A_matrix = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_matrix[j:j+n, j] = a
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A_matrix @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if b_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(b_var.value, 0)
                sb = float(np.sum(b_val))
                if sb > 1e-10:
                    return b_val / sb, float(M_var.value)
        except Exception:
            continue
    return None, None
