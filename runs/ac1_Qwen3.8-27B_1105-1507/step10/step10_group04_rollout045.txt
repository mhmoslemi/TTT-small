def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_g=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    g_val = None
    M_val = None
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            kwargs = {}

                kwargs = {'MSDK_PARM_MS_INFEASIBILITY_TOL': 1e-10,
                          'MSDK_PARM_MS_REL_GAP_TOL': 1e-10}
            prob.solve(solver=solver, **kwargs)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:

def smoothed_grad(a, tau):
    """Compute smoothed gradient of max conv(a,a) w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    if tau * (max_conv - float(np.min(conv))) > 500:
        # Avoid overflow
        shift = tau * conv - np.max(tau * conv)
        exp_vals = np.exp(shift)
    else:
        exp_vals = np.exp(tau * (conv - max_conv))
    
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # Gradient: 2 * sum_k p_k * a[k-j] for each j
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A_f = fft(a, N)
    g_full = ifft(P * np.conj(A_f)).real
    grad = 2.0 * g_full[:n]
    
    return grad, p

def parametric_init(n, rng):
    """Generate parametric initializations with various shapes."""
    inits = []
    x = np.linspace(0, 1, n)
    
    # Various asymmetric shapes
    shapes = [
        np.exp(-5 * x),                                    # left-heavy exponential
        np.exp(-5 * (1 - x)),                              # right-heavy exponential
        x ** 5 * (1 - x),                                  # right-skewed
        x * (1 - x) ** 5,                                  # left-skewed
        1 / (1 + np.exp(10 * (x - 0.3))),                  # sigmoid left
        1 / (1 + np.exp(10 * (x - 0.7))),                  # sigmoid right
        np.exp(-3 * x) * np.exp(-1 * (1 - x)),             # two-sided exp (asym)
        np.exp(-1 * x) * np.exp(-3 * (1 - x)),             # two-sided exp (asym other)
        x ** 2 * (1 - x) ** 3,                             # beta(3,4)
        x ** 3 * (1 - x) ** 2,                             # beta(4,3)
        np.maximum(0, 1 - 8 * np.abs(x - 0.2)),            # triangle left
        np.maximum(0, 1 - 8 * np.abs(x - 0.8)),            # triangle right
    ]
    
    for shape in shapes:
        a = shape.copy()
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s
            inits.append(a)
    
    # Random asymmetric
    for _ in range(5):
        alpha = rng.uniform(1, 10)
        beta = rng.uniform(1, 10)
        a = x ** (alpha - 1) * (1 - x) ** (beta - 1)
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s
            inits.append(a)
    
    return inits


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n == 1 else 1.0/new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
