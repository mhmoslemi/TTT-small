"""
Optimized step function coefficient search using active-set LP, FISTA,
and multi-scale exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a, k_indices=None):
    """Build convolution matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    if k_indices is None:
        k_range = np.arange(2 * n - 1)
    else:
        k_range = np.asarray(k_indices)
    k_idx = k_range[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    m = len(k_range)
    A = np.zeros((m, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp(a, k_indices=None, warm_start=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M, dual_values) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm, k_indices)
    m = A.shape[0]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    obj = cp.Minimize(M)
    prob = cp.Problem(obj, constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    try:
                        dual = np.asarray(constraints[0].dual_value, dtype=np.float64)
                    except Exception:
                        dual = None
                    return g_val, float(M.value), dual
        except Exception:
            continue
    return None, None, None


def line_search(a, g, n_evals=30):
    """Line search: find best t to minimize obj((1-t)*a + t*g)."""
    n = len(a)
    ts = np.linspace(0.005, 0.995, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * g
        objs[i] = compute_obj(a_try)
    
    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]
    
    # Parabolic refinement
    if 0 < best_idx < n_evals - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.005, 0.995))
            a_par = (1 - t_par) * a + t_par * g
            f_par = compute_obj(a_par)
            if f_par < best_obj:
                best_t = t_par
                best_obj = f_par
    
    return best_t, best_obj

        y_new = y - step * grad
        y_new = project_simplex(y_new)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = x_new.copy()
            best_it = it
        
        y = x_new
        t_k = t_new
        it += 1
        
        if it - best_it > 20000:
            x = best_a.copy()
            y = best_a.copy()
            t_k = 1.0
            best_it = it

def init_raised_cosine(n):
    """Initialize with raised cosine."""
    j = np.arange(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * j / n))
    a = np.maximum(a, 0)
    return a / a.sum()


def init_gaussian(n, sigma_frac=0.3):
    """Initialize with Gaussian."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    return a / a.sum()


def init_bimodal(n, frac=0.3):
    """Initialize with bimodal (high at edges)."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    x = (j - center) / (n / 2.0)
    a = np.exp(-frac * (1.0 - x**2))
    a = np.maximum(a, 0)
    return a / a.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimized step function coefficient search using active-set LP with dual-guided
constraint selection, FISTA with entropy-regularized updates, and multi-n
exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: active-set LP reduction for speed, entropy-based FISTA,
broader n exploration with diverse initializations, and adaptive time allocation.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a, k_indices=None):
    """Build convolution matrix A where (A@g)[k_idx] = conv(a,g)[k_idx]."""
    n = len(a)
    if k_indices is None:
        k_range = np.arange(2 * n - 1)
    else:
        k_range = np.asarray(k_indices)
    k_idx = k_range[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    m = len(k_range)
    A = np.zeros((m, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp(a, k_indices=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for k in k_indices, sum(g)=1, g>=0.
    Returns (g, M, dual_values) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm, k_indices)
    m = A.shape[0]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    try:
                        dual = np.asarray(constraints[0].dual_value, dtype=np.float64)
                    except Exception:
                        dual = None
                    return g_val, float(M.value), dual
        except Exception:
            continue
    return None, None, None


def line_search(a, g, n_evals=30):
    """Line search: find best t in [0,1] to minimize obj((1-t)*a + t*g)."""
    n = len(a)
    ts = np.linspace(0.003, 0.997, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = (1.0 - t) * a + t * g
        objs[i] = compute_obj(a_try)

    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]

    # Golden section refinement around best
    if 0 < best_idx < n_evals - 1:
        lo = ts[max(0, best_idx - 2)]
        hi = ts[min(n_evals - 1, best_idx + 2)]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = compute_obj((1.0 - c) * a + c * g)
        fd = compute_obj((1.0 - d) * a + d * g)
        for _ in range(15):
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - gr * (hi - lo)
                fc = compute_obj((1.0 - c) * a + c * g)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = compute_obj((1.0 - d) * a + d * g)
        t_gold = (lo + hi) / 2.0
        f_gold = compute_obj((1.0 - t_gold) * a + t_gold * g)
        if f_gold < best_obj:
            best_t = t_gold
            best_obj = f_gold

    return best_t, best_obj

    rng = np.random.default_rng(42)
    n_constraints = 2 * n - 1
    n_active_target = max(n // 3, int(max_active_frac * n_constraints))

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        # Decide whether to use active set
        if it % revalidate_every == 0 or not use_active:
            # Full LP for revalidation
            g, M_lp, dual = solve_lp(a, None)
            if dual is not None:
                full_dual = np.asarray(dual, dtype=np.float64)
                top_idx = np.argsort(full_dual)[::-1][:n_active_target]
                curr_max_k = int(np.argmax(compute_conv(a)))
                mask = np.zeros(n_constraints, dtype=bool)
                mask[top_idx] = True
                mask[curr_max_k] = True
                active_k = np.where(mask)[0]
                use_active = True
        else:
            # Reduced LP
            g, M_lp, dual = solve_lp(a, active_k)
            if g is None:
                g, M_lp, dual = solve_lp(a, None)
                if g is None:
                    break
                use_active = False

        if g is None:
            break
        it += 1

        # Update active set from reduced LP duals
        if dual is not None and use_active:
            full_dual = np.zeros(n_constraints)
            full_dual[active_k] = dual
            # Add constraints that are now violated
            conv_a = compute_conv(a)
            viol_mask = conv_a > M_lp * 1.01
            if np.any(viol_mask):
                viol_idx = np.where(viol_mask)[0]
                active_k = np.unique(np.concatenate([active_k, viol_idx[:n_active_target // 4]]))
                active_k = active_k[:n_active_target * 2]

        # Line search
        best_t, obj_at_t = line_search(a, g, n_evals=25)

        if obj_at_t < best_obj - 1e-13:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                best_obj = obj_at_t
                best_a = a_new.copy()
                a = a_new
                stall = 0
        else:
            stall += 1
            improved = False
            for t_small in [0.03, 0.01, 0.005]:
                a_sm = (1.0 - t_small) * a + t_small * g
                a_sm = np.maximum(a_sm, 0)
                s_sm = float(np.sum(a_sm))
                if s_sm > 1e-10:
                    a_sm = a_sm / s_sm
                    obj_sm = compute_obj(a_sm)
                    if obj_sm < best_obj - 1e-13:
                        best_obj = obj_sm
                        best_a = a_sm.copy()
                        a = a_sm
                        stall = 0
                        improved = True
                        break

            if not improved and stall >= 4:
                # Structured perturbation using softmax subgradient
                conv_p = compute_conv(a)
                tau_p = 200.0
                exp_p = np.exp(tau_p * (conv_p - conv_p.max()))
                p = exp_p / exp_p.sum()
                L = 1
                while L < 3 * n:
                    L <<= 1
                Af = fft(a, L)
                Pf = fft(p, L)
                grad = 2.0 * ifft(Pf * np.conj(Af)).real[:n]
                gnorm = float(np.linalg.norm(grad))
                if gnorm > 1e-15:
                    lr_p = 0.02 / gnorm
                    a = a - lr_p * grad
                a = np.maximum(a + 0.004 * rng.standard_normal(n), 0)
                a = a / a.sum()
                stall = 0
                use_active = False
                active_k = all_k.copy()

            if stall >= 12 and it > 40:
                break

        if verbose and it % 8 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.4f}, "
                  f"n_act={len(active_k)}/{n_constraints}, time={time.monotonic()-start:.0f}s")

    return best_a


def fista_optimize(a, time_budget=100.0, tau_start=50.0, tau_end=3000.0, verbose=False):
    """
    FISTA on smoothed max of conv(a,a) with simplex projection.
    Uses adaptive step size with backtracking.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64))

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()
    y = a.copy()
    t_k = 1.0
    it = 0
    best_it = 0
    step = 1.0 / n

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start * (tau_end / tau_start) ** progress

        # Compute gradient at y
        conv_y = compute_conv(y)
        max_c = float(np.max(conv_y))
        exp_vals = np.exp(tau * (conv_y - max_c))
        p = exp_vals / exp_vals.sum()

        A = fft(y, L)
        P = fft(p, L)
        grad = (2.0 / tau) * ifft(A * P).real[:n]

        # Backtracking line search for step size
        step_cur = step
        y_new = y - step_cur * grad
        f_new = float(np.sum(np.maximum(y_new, 0) * 0))  # placeholder
        # Simple: just use the step and project
        y_new = project_simplex(y)
        # Actually let's do proper backtracking
        step_cur = step
        obj_y = compute_obj(project_simplex(y))
        for _ in range(10):
            trial = y - step_cur * grad
            trial = project_simplex(trial)
            obj_trial = compute_obj(trial)
            if obj_trial <= obj_y + 0.01 * step_cur * float(np.dot(grad, trial - y)):
                break
            step_cur *= 0.5
        step = max(step_cur, 1e-8)

        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        x_new = trial + ((t_k - 1.0) / t_new) * (trial - y)
        x_new = project_simplex(x_new)

        obj_new = compute_obj(x_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = x_new.copy()
            best_it = it

        y = x_new
        t_k = t_new
        it += 1

        if it - best_it > 30000:
            x = best_a.copy()
            y = best_a.copy()
            t_k = 1.0
            step = 1.0 / n
            best_it = it

    if verbose:
        print(f"    FISTA: obj={best_obj:.8f}, iters={it}, tau_end={tau:.0f}")
    return best_a


def entropy_gd(a, time_budget=100.0, verbose=False):
    """
    Gradient descent with multiplicative weight updates on the simplex.
    Uses smoothed max with adaptive tau.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64))

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = 50.0 * (5000.0 / 50.0) ** progress
        eta = 0.1 / (tau * np.log(n) + 1.0)

        conv_a = compute_conv(a)
        max_c = float(np.max(conv_a))
        exp_vals = np.exp(tau * (conv_a - max_c))
        p = exp_vals / exp_vals.sum()

        A = fft(a, L)
        P = fft(p, L)
        grad = (2.0 / tau) * ifft(A * P).real[:n]

        # Exponentiated gradient (mirror descent on simplex)
        log_w = np.log(a + 1e-30) - eta * grad
        log_w -= np.max(log_w)
        w = np.exp(log_w)
        s_w = float(np.sum(w))
        if s_w < 1e-300:
            w = np.ones(n) / n
        else:
            w = w / s_w

        obj_w = compute_obj(w)
        if obj_w < best_obj - 1e-14:
            best_obj = obj_w
            best_a = w.copy()
        a = w
        it += 1

    if verbose:
        print(f"    EntropyGD: obj={best_obj:.8f}, iters={it}")
    return best_a


def init_raised_cosine(n):
    """Raised cosine initialization."""
    j = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * j / n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def init_gaussian(n, sigma_frac=0.25):
    """Gaussian initialization."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = max(sigma_frac * n, 1.0)
    a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def init_bimodal(n, width=0.3):
    """Bimodal initialization (high at edges)."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (j - center) / (n / 2.0)
    a = np.exp(-width * (1.0 - x * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def init_hamming(n):
    """Hamming window initialization."""
    j = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * j / (n - 1))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def init_flat_with_edges(n):
    """Flat with slightly boosted edges."""
    j = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    edge_boost = 2.0
    fade = n // 5
    for i in range(fade):
        a[i] = 1.0 + (edge_boost - 1.0) * (1.0 - i / fade)
        a[n - 1 - i] = 1.0 + (edge_boost - 1.0) * (1.0 - i / fade)
    return a / a.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
