def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = (a*g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. (a_norm * g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_value).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

    rng = np.random.default_rng(seed)

    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break

        g, M = solve_lp(a)
        if g is None:
            break

        best_t, obj_at_t = line_search_adaptive(a, g)
        it += 1

        if obj_at_t < best_obj - 1e-13:
            a_new = np.maximum((1 - best_t) * a + best_t * g, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                best_obj = obj_at_t
                best_a = a_new.copy()
                a = a_new
                stall = 0
                if verbose and (it % 5 == 0 or it == 1):
                    print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.4f}, M={M:.6f}")
        else:
            stall += 1
            if verbose and it % 10 == 0:
                print(f"    LP it{it}: stalled obj={best_obj:.8f}")

            if stall >= 3:
                # Perturb to escape
                tau = 500.0
                g_pert = smoothed_gradient(a, tau)
                a = a - 0.01 / (np.linalg.norm(g_pert) + 1) * g_pert
                a = np.maximum(a, 0)
                a = a + 0.003 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

        if it > 500:
            break

    return best_a


def coordinate_descent(a, time_budget=60.0, verbose=False):
    """
    Coordinate descent: for each coordinate j, find the optimal a_j
    that minimizes the peak autocorrelation (keeping others fixed,
    adjusting sum to maintain normalization).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99)

    # Precompute: for each j, the "influence" on each lag k
    # (a*a)_k = sum_i a_i * a_{k-i}
    # If we change a_j to a_j + d, the change in (a*a)_k is:
    #   d * a_{k-j} + d * a_{k-j} + d^2 (if k-j == j, i.e., k=2j)
    # = 2*d*a_{k-j} + d^2*[k==2j]
    # We need sum(a) = 1, so we adjust another coordinate or use a "virtual" adjustment.
    # Simpler: parameterize as a_j -> a_j * (1+eps) and rescale the rest.
    # Or: a_j -> a_j + d, and a_j -> a_j + d, a_k -> a_k - d for some k.
    # Simplest: just try increasing a_j and decreasing the mean of others.

    passes = 0
    while time.monotonic() - start < time_budget:
        improved = False
        # Random order for diversity
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            # Try a few values of a_j
            current_val = a[j]
            other_sum = 1.0 - current_val
            if other_sum < 1e-10:
                continue

            best_val = current_val
            best_local_obj = np.inf

            # Try increasing and decreasing a_j
            for frac in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                new_val = current_val * frac
                if new_val < 0 or new_val > 1.0 - 1e-10:
                    continue
                # Rescale others to maintain sum=1
                a_try = a.copy()
                a_try[j] = new_val
                remainder = 1.0 - new_val
                if remainder < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_try[mask] = a[mask] * (remainder / other_sum)
                a_try = np.maximum(a_try, 0)
                s = float(np.sum(a_try))
                if s < 1e-10:
                    continue
                a_try = a_try / s

                # Compute peak
                conv_try = compute_conv(a_try)
                peak = float(np.max(conv_try))
                if peak < best_local_obj - 1e-14:
                    best_local_obj = peak
                    best_val = new_val
                    best_a_try = a_try.copy()

            if best_val != current_val:
                a = best_a_try
                obj_check = compute_obj(a)
                if obj_check < best_obj - 1e-14:
                    best_obj = obj_check
                    best_a = a.copy()
                    improved = True

        passes += 1
        if not improved and passes > 3:
            break
        if verbose and passes % 5 == 0:
            print(f"    CD pass{passes}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n + 1)[:-1]
    x_new = np.linspace(0, 1, new_n + 1)[:-1]
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.05):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: FISTA-primary optimization with LP polishing,
    coordinate descent, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_base = len(best_seq)
    print(f"Starting optimization from n={n_base}, obj={best_score:.8f}")

    # === Phase 1: FISTA on base sequence (main workhorse) ===
    print(f"\n=== Phase 1: FISTA (main) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 450.0)
    if p1_budget > 30:
        a_fista = fista_optimize(best_seq, time_budget=p1_budget,
                                 tau_init=50.0, tau_end=50000.0,
                                 verbose=True, seed=seed)
        obj_fista = compute_obj(a_fista)
        if obj_fista < best_score:
            best_score = obj_fista
            best_seq = a_fista.copy()
            print(f"  Phase1 FISTA: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 FISTA: obj={obj_fista:.8f} (no improvement)")

    # === Phase 2: LP polishing ===
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 20:
        a_lp = lp_optimize(best_seq, time_budget=p2_budget, verbose=True, seed=seed + 1)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Coordinate Descent ===
    print(f"\n=== Phase 3: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.08, 80.0)
    if p3_budget > 15:
        a_cd = coordinate_descent(best_seq, time_budget=p3_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase3 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 4: Second FISTA round with higher tau ===
    print(f"\n=== Phase 4: FISTA Round 2 (high tau) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 150.0)
    if p4_budget > 20:
        a_f2 = fista_optimize(best_seq, time_budget=p4_budget,
                              tau_init=500.0, tau_end=100000.0,
                              verbose=True, seed=seed + 2)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Phase4 FISTA2: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 FISTA2: obj={obj_f2:.8f} (no improvement)")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 100, n_best - 50,
        n_best + 50, n_best + 100, n_best + 200,
        n_best + 300, n_best + 500,
        600, 700, 900, 1000, 1100, 1200, 1400
    ]))
    # Filter: reasonable range, not too close to current
    n_options = [n for n in n_options if 500 <= n <= 2000 and abs(n - n_best) > 40]
    n_options = n_options[:5]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # Quick FISTA
        alloc_f = min(remaining * 0.25, 60.0)
        if alloc_f > 15:
            a_try = fista_optimize(a_try, time_budget=alloc_f,
                                   tau_init=100.0, tau_end=10000.0,
                                   verbose=False, seed=seed + n_try)

        # Quick LP
        remaining = deadline - time.monotonic()
        alloc_l = min(remaining * 0.1, 40.0)
        if alloc_l > 10:
            a_try = lp_optimize(a_try, time_budget=alloc_l, verbose=False, seed=seed + n_try + 100)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    restarts = [
        ("fourier_perturb", lambda: fourier_perturb(best_seq, rng, strength=0.06)),
        ("random_blend", None),
        ("smooth_reblend", None),
    ]

    for r, (name, func) in enumerate(restarts):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if name == "fourier_perturb":
            a_r = func()
        elif name == "random_blend":
            a_r = 0.8 * best_seq + 0.2 * (rng.uniform(0, 1, n_cur) / n_cur)
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        else:  # smooth_reblend
            ks = max(11, n_cur // 20)
            if ks % 2 == 0:
                ks += 1
            a_smooth = uniform_filter1d(best_seq, size=ks)
            a_r = 0.7 * best_seq + 0.3 * a_smooth
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

        # Optimize
        alloc_f = min(remaining * 0.2, 50.0)
        if alloc_f > 12:
            a_r = fista_optimize(a_r, time_budget=alloc_f,
                                 tau_init=200.0, tau_end=20000.0,
                                 verbose=False, seed=seed + 1000 + r)

        remaining = deadline - time.monotonic()
        alloc_l = min(remaining * 0.08, 30.0)
        if alloc_l > 10:
            a_r = lp_optimize(a_r, time_budget=alloc_l, verbose=False, seed=seed + 2000 + r)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP
    if remaining > 30:
        alloc = min(remaining * 0.4, 150.0)
        a_final = lp_optimize(best_seq, time_budget=alloc, verbose=True, seed=seed + 5000)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # High-tau FISTA
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.35, 100.0)
        a_final2 = fista_optimize(best_seq, time_budget=alloc,
                                  tau_init=1000.0, tau_end=200000.0,
                                  verbose=True, seed=seed + 6000)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final FISTA: obj={best_score:.8f} **")
        else:
            print(f"  Final FISTA: obj={obj_final2:.8f}")

    # Final CD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 40.0)
        a_final3 = coordinate_descent(best_seq, time_budget=alloc, verbose=True)
        obj_final3 = compute_obj(a_final3)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final3:.8f}")

    # Final validation with evaluate_sequence
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
