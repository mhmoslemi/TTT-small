    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_v = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_v))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_v / sum_exp
    
    N = 1
    while N < 2 * (2 * n - 1):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def nesterov_polish(a, time_budget, tau_init=30.0, tau_max=2000.0):
    """Nesterov accelerated GD on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 1.0 / np.sqrt(n)
    v = np.zeros(n)
    tau = tau_init
    
    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        grad = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(grad))
        
        if g_norm > 1e-15:
            adaptive = 0.5 / (g_norm + 1e-10)
            lr = 0.8 * lr + 0.2 * min(max(adaptive, 0.01), 3.0 / np.sqrt(n))
        
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        lr *= 0.9995
        tau = min(tau * 1.01, tau_max)
    
    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Structured Fourier perturbation."""
    n = len(a)
    x = np.arange(n)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = 2 * np.pi * k / n
        amp = strength * rng.uniform(0.3, 1.0) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * x + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample(a, new_n):
    """Linear resampling to new length."""
    if len(a) == new_n:
        return a.copy()
    if len(a) == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

This IS a zero-sum game (Player 2 wants to maximize). By von Neumann's minimax theorem, `min_a max_g max_k(a*g)[k] = max_g min_a max_k(a*g)[k] = V` (the value).

"""
Hybrid LP + gradient optimizer for step function coefficient minimization.
Uses CLARABEL LP solver with golden-section line search, Nesterov smoothed
gradient polishing, diverse multi-start initializations, and adaptive
length exploration. Key improvements: exact 1D line search for step sizes,
faster LP via matrix structure, anti-stalling with escalating perturbations,
and parallel diverse initializations.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT, returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    return 2.0 * n * float(np.max(compute_conv(a_n)))


def build_conv_matrix(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. (a_n*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def golden_line_search(a, g, n_evals=8):
    """Golden section search for optimal t in (0,1) for (1-t)*a + t*g."""
    lo = 0.001
    hi = 0.999
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    def obj_at(t):
        x = (1.0 - t) * a + t * g
        x = np.maximum(x, 0)
        s = float(np.sum(x))
        if s < 1e-10:
            return np.inf
        return compute_obj(x / s)

    c = lo + gr * (hi - lo)
    d = lo + (1.0 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    for _ in range(n_evals - 2):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + (1.0 - gr) * (hi - lo)
            fd = obj_at(d)

    t_opt = (lo + hi) / 2.0
    return t_opt

def smoothed_gradient(a, tau):
    """Gradient of tau*log(sum(exp(tau*(a*a)_k))) w.r.t. a."""

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n_old = len(a)
    if n_old == new_n:
        return a.copy()
    if n_old == 1:

    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, new_n)

    """
    Main search function. Combines LP alternating minimization with
    golden-section line search, Nesterov gradient polishing, diverse
    multi-start initializations, and adaptive length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 35.0

    # Load parent sequence
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0)
        if a.size == 0 or float(a.sum()) < 0.01:
            raise ValueError("Invalid parent")
        a = a / a.sum()
    except Exception:
        a = np.ones(800) / 800.0

    best_a = a.copy()
    best_obj = compute_obj(a)
    n_cur = len(a)
    print(f"Init: n={n_cur}, obj={best_obj:.12f}")

    rng = np.random.default_rng(seed)

    # === Phase 1: Main LP optimization on parent ===
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 450.0)
    print(f"\n--- Phase 1: LP alternate (budget={p1_budget:.0f}s) ---")
    a1 = lp_alternate(best_a, p1_budget, seed=seed, verbose=True)
    obj1 = compute_obj(a1)
    if obj1 < best_obj:
        best_obj, best_a = obj1, a1.copy()
        print(f"  Phase 1: {best_obj:.12f} **")
    else:
        print(f"  Phase 1: {obj1:.12f} (no improvement)")

    # === Phase 2: Diverse initializations ===
    remaining = deadline - time.monotonic()
    if remaining > 250:
        print(f"\n--- Phase 2: Diverse starts (remaining={remaining:.0f}s) ---")
        n_use = len(best_a)
        x = np.linspace(0.0, 1.0, n_use)

        inits = []
        inits.append(("concave", np.maximum(1.0 - x**2, 0.001)))
        inits.append(("u_shape", np.maximum(0.5 * (x - 0.5)**2 * 4 + 0.3, 0.001)))
        inits.append(("bimodal", np.maximum(
            np.exp(-((x - 0.3) / 0.12)**2) + np.exp(-((x - 0.7) / 0.12)**2) + 0.1, 0.001)))
        inits.append(("random", rng.dirichlet(np.full(n_use, 0.5))))
        inits.append(("power_law", np.maximum(1.0 / (x + 0.05), 0.001)))

        for name, a_try in inits:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            a_try = np.maximum(a_try, 0).astype(np.float64)
            a_try = a_try / a_try.sum()
            alloc = min(remaining * 0.22, 180.0)

            a_try = lp_alternate(a_try, alloc, seed=seed + 100 * hash(name) % 99999,
                                 verbose=False)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj, best_a = obj_try, a_try.copy()
                print(f"  {name} (n={n_use}): {best_obj:.12f} **")
            else:
                print(f"  {name} (n={n_use}): {obj_try:.12f}")

    # === Phase 3: LP on perturbed best ===
    remaining = deadline - time.monotonic()
    if remaining > 200:
        print(f"\n--- Phase 3: LP on perturbed best ---")
        a3 = fourier_perturb(best_a, rng, strength=0.01)
        p3_budget = min(remaining * 0.25, 250.0)
        a3 = lp_alternate(a3, p3_budget, seed=seed + 5555, verbose=False)
        obj3 = compute_obj(a3)
        if obj3 < best_obj:
            best_obj, best_a = obj3, a3.copy()
            print(f"  Phase 3: {best_obj:.12f} **")
        else:
            print(f"  Phase 3: {obj3:.12f}")

    # === Phase 4: Gradient polish ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n--- Phase 4: Gradient polish ---")
        p4_budget = min(remaining * 0.15, 120.0)
        a4 = nesterov_polish(best_a, p4_budget, tau_init=50.0, tau_max=2000.0)
        obj4 = compute_obj(a4)
        if obj4 < best_obj:
            best_obj, best_a = obj4, a4.copy()
            print(f"  Phase 4: {best_obj:.12f} **")
        else:
            print(f"  Phase 4: {obj4:.12f}")

    # === Phase 5: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 200:
        print(f"\n--- Phase 5: Length exploration ---")
        n_best = len(best_a)
        n_options = sorted(set([
            n_best + 50, n_best + 100, n_best + 200,
            n_best - 50, n_best - 100
        ]))
        n_options = [n for n in n_options if 200 <= n <= 4000]

        for n_try in n_options[:4]:
            remaining = deadline - time.monotonic()
            if remaining < 120:
                break
            a_try = resample_seq(best_a, n_try)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            alloc = min(remaining * 0.25, 180.0)

            a_try = lp_alternate(a_try, alloc * 0.65, seed=n_try + 1, verbose=False)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj, best_a = obj_try, a_try.copy()
                print(f"  n={n_try}: {best_obj:.12f} **")
            else:
                print(f"  n={n_try}: {obj_try:.12f}")

            # Quick polish on the new length
            remaining = deadline - time.monotonic()
            if remaining > 40 and obj_try < best_obj + 0.01:
                a_try = nesterov_polish(a_try, min(remaining * 0.2, 50.0),
                                         tau_init=80.0, tau_max=1000.0)
                obj_try2 = compute_obj(a_try)
                if obj_try2 < best_obj:
                    best_obj, best_a = obj_try2, a_try.copy()
                    print(f"  n={n_try} polished: {best_obj:.12f} **")

    # === Phase 6: Final LP refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n--- Phase 6: Final LP refinement ---")
        a6 = fourier_perturb(best_a, rng, strength=0.003)
        p6_budget = min(remaining * 0.4, 200.0)
        a6 = lp_alternate(a6, p6_budget, seed=seed + 12345, verbose=True)
        obj6 = compute_obj(a6)
        if obj6 < best_obj:
            best_obj, best_a = obj6, a6.copy()
            print(f"  Phase 6: {best_obj:.12f} **")
        else:
            print(f"  Phase 6: {obj6:.12f}")

    # === Phase 7: Ultra-final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        print(f"\n--- Phase 7: Ultra-final polish ---")
        p7_budget = min(remaining * 0.5, 80.0)
        a7 = nesterov_polish(best_a, p7_budget, tau_init=100.0, tau_max=5000.0)
        obj7 = compute_obj(a7)
        if obj7 < best_obj:
            best_obj, best_a = obj7, a7.copy()
            print(f"  Phase 7: {best_obj:.12f} **")
        else:
            print(f"  Phase 7: {obj7:.12f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - t_start
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.12f}, time={elapsed:.1f}s ===")
    return result
