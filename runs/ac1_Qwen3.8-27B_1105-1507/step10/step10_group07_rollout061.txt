```python
    w_new = w - lr * v
    
    return w_new, v, compute_obj(a)
```

Actually, let me think about this differently. The key insight is:

The objective is 2n * max(a*a) / (sum a)^2. 

If I normalize so sum(a) = 1, I want to minimize 2n * max(a*a).

The "soft" version with temperature τ is:
J_τ(a) = 2n * log(sum_k exp(τ*(a*a)[k])) / (τ * (sum a)^2) ... no that's not quite right.

Actually, let me just use: J_τ(a) = 2n * (sum_k exp(τ * conv[k]))^{1/τ} / (sum a)^2

As τ → ∞, this approaches 2n * max(conv) / (sum a)^2.

The gradient of J_τ with respect to a is well-defined and smooth.

3. Try different lengths
4. Final polish

The key difference from the previous approach:

- More effective momentum
- Better restart strategy
- More time spent on the "refinement" phase

Let me also think about whether there's a mathematical shortcut. 

For the problem of minimizing max(a*a) subject to a >= 0, sum(a) = 1:
- This is a convex optimization problem (minimize max of linear functions)
- The optimal solution exists and is unique (up to the non-uniqueness of the max)
- At the optimum, there's a set of "active" peaks where (a*a)[k] = max

The KKT conditions suggest that at the optimum:
- There exists a set S of active peaks
- The gradient of the objective is in the normal cone of the feasible set
- Specifically: sum_{k in S} λ_k * d(a*a)[k]/da + μ * 1 = 0 (for the unconstrained part)
- Where λ_k >= 0 are the Lagrange multipliers

This means: at the optimum, the "weighted sum" of the derivatives of the active peaks is constant (proportional to 1).

This gives us a characterization: a is optimal iff there exists a probability distribution p on the active peaks such that:
sum_k p[k] * d(a*a)[k]/da = constant (vector)

This is exactly what the LP direction-finding is looking for!

So the LP approach is theoretically sound. The question is whether we can do it more efficiently.

Let me try a hybrid approach:
1. Use smoothed GD for most of the time (fast, many iterations)
2. Occasionally use a "smart" direction from a simplified LP
3. Use the structured nature of the problem

For the LP, instead of solving the full LP, I can use an iterative method:
- Start with uniform p on the top K peaks
- Iterate: update a in the direction that reduces the weighted peaks
- Update p to focus on the most violated constraints

This is essentially the dual subgradient method, which should converge.

```python
"""
Adaptive multi-phase optimizer for minimum peak autocorrelation sequences.
Uses smoothed gradient descent with adaptive temperature, subgradient peaking,
multi-length exploration, and structured restarts.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_obj_and_grad(a, tau):
    """
    Compute the soft objective and its gradient w.r.t. a.
    J_tau = 2n * (1/tau) * log(sum_k exp(tau * conv[k]))
    (This approximates 2n * max(conv) as tau -> inf, with sum(a)=1)
    Returns (J_tau, grad_J) where grad_J = dJ/da.
    """
    n = len(a)
    s = float(np.sum(a))
    a_norm = a / s if s > 1e-10 else a
    conv = compute_conv(a_norm)
    
    # Soft max
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    soft_max = max_conv + np.log(float(np.sum(exp_vals))) / tau
    
    J = 2.0 * n * soft_max  # (with sum(a_norm) = 1, so (sum a)^2 = 1)
    
    # Gradient: dJ/da = 2n * sum_k p[k] * d(conv[k])/da / s (chain rule through normalization)
    # Actually, since we're working with a_norm = a/s, and s = sum(a):
    # d(a_norm)/da = (I - a_norm * 1^T) / s ... this gets complicated.
    # 
    # Simpler: just work with a directly assuming sum(a) = 1.
    # If a is already normalized (sum = 1), then:
    # d(conv[k])/da[j] = 2 * a[k-j] * I(0 <= k-j < n)
    # dJ/da[j] = 2n * sum_k p[k] * 2 * a[k-j] * I(0 <= k-j < n)
    # = 4n * sum_{k: 0<=k-j<n} p[k] * a[k-j]
    # = 4n * sum_m p[m+j] * a[m] (substituting m = k-j)
    
    # Compute c[j] = sum_m a[m] * p[m+j] for j = 0..n-1
    # Using FFT:
    a_rev = a[::-1]
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    conv_cross = irfft(rfft(a_rev, N) * rfft(p, N), N)
    c = conv_cross[n - 1: 2 * n - 1]  # c[j] = sum_m a[m] * p[m+j]
    
    grad = 4.0 * n * c  # This is dJ/da assuming sum(a) = 1
    
    return J, grad, p, conv


def smoothed_gd(a, time_budget, tau_start=5.0, tau_end=500.0, lr_start=None, 
                momentum=0.85, seed=42, verbose=False):
    """
    Smoothed gradient descent on the simplex for minimizing 2n*max(a*a).
    Uses adaptive temperature and learning rate.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    if lr_start is None:
        lr_start = 4.0 / np.sqrt(n)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Adaptive temperature: start low, increase over time
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.7)
        
        # Compute soft gradient
        _, grad, p, conv = softmax_obj_and_grad(a, tau)
        
        # Update with momentum (in the "unconstrained" space, then project)
        v = momentum * v + grad
        lr = lr_start * (0.999 ** it)  # Slow decay
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        lr = max(1e-12, lr * 0.9997)
        
        # Periodic restart if stuck
        if no_improve > 200:
            scale = 0.01 + 0.005 * min(no_improve // 100, 5)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-10:
                a = a / s_a
            v *= 0.1
            no_improve = 0
            if it > 500:
                break
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"      SGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")
    
    return best_a


def subgradient_peak_push(a, time_budget, K=10, seed=42, verbose=False):
    """
    Subgradient method focusing on the top K peaks of the convolution.
    More aggressive than smoothed GD for the "late game".
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        if max_conv < 1e-15:
            break
        
        # Get top K peaks
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        
        # Soft weights on peaks
        weights = np.exp(10.0 * (peak_vals - peak_vals[0]) / max_conv)
        weights /= weights.sum()
        
        # Compute subgradient: sum of weighted peak derivatives
        # d(conv[k])/da[j] = 2*a[k-j] * I(0<=k-j<n)
        # So grad[j] = 2 * sum_{k in top} w_k * a[k-j] * I(0<=k-j<n)
        
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            # For this peak k, the derivative w.r.t. a[j] is 2*a[k-j] if 0<=k-j<n
            # So the contribution to grad[j] is 2*w*a[k-j]
            # grad[j] += 2*w*a[k-j] for j such that 0<=k-j<n, i.e., k-n+1 <= j <= k
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                # a[k-j] for j in [j_start, j_end] means a[k-j_start] down to a[k-j_end]
                # = a[k-j_start : k-j_end+1][::-1]
                indices = np.arange(j_start, j_end + 1)
                a_vals = a[k - indices]  # a[k-j] for each j
                grad[j_start:j_end + 1] += 2.0 * w * a_vals
        
        # Update
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.7
                v *= 0.3
                no_improve = 0
                if lr < 1e-10:
                    # Try a restart
                    a = best_a * np.exp(rng.normal(0, 0.01, n))
                    a = np.maximum(a, 0)
                    s_a = a.sum()
                    if s_a > 1e-10:
                        a = a / s_a
                    lr = 2.0 / np.sqrt(n)
                    v = np.zeros(n)
        
        it += 1
        if verbose and it % 50 == 0:
            print(f"      PeakPush it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
```

```python
"""
Adaptive multi-phase optimizer for minimum peak autocorrelation step functions.
Combines fast smoothed gradient descent with adaptive temperature scheduling,
peak-focused subgradient refinement, structured multi-start exploration,
and multi-length search. Avoids slow convex solvers in favor of many fast
gradient iterations with smart restarts and parameter adaptation.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_grad(a, tau):
    """
    Compute soft-max of convolution and its gradient w.r.t. a.
    Returns (soft_max_value, gradient, peak_distribution, full_conv).
    Assumes sum(a) = 1.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    total = float(np.sum(exp_vals))
    p = exp_vals / total
    soft_max_val = max_conv + np.log(total) / tau

    # Gradient of 2n * soft_max w.r.t. a:
    # d(conv[k])/da[j] = 2*a[k-j] * I(0 <= k-j < n)
    # grad[j] = 2n * sum_k p[k] * 2*a[k-j] * I(0<=k-j<n)
    #         = 4n * sum_m a[m] * p[m+j]  (sub m = k-j)
    # Compute c[j] = sum_m a[m] * p[m+j] via FFT
    a_rev = a[::-1]
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    cross = irfft(rfft(a_rev, N) * rfft(p, N), N)
    c = cross[n - 1: 2 * n - 1]
    grad = 4.0 * n * c

    return soft_max_val, grad, p, conv


def smoothed_gd(a, time_budget, tau_start=5.0, tau_end=500.0,
                lr_start=None, momentum=0.85, seed=42, verbose=False,
                restart_scale=0.012, restart_patience=150):
    """
    Smoothed gradient descent on the simplex.
    Adaptive temperature, learning rate decay, momentum, and periodic restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    if lr_start is None:
        lr_start = 4.0 / np.sqrt(n)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    it = 0
    no_improve = 0
    current_lr = lr_start

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Adaptive temperature with smooth schedule
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.6)

        # Compute soft gradient
        _, grad, p, conv = soft_max_grad(a, tau)

        # Momentum update
        v = momentum * v + grad

        # Adaptive learning rate with backtracking
        lr = current_lr
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)

        if not np.isfinite(obj_new) or obj_new > best_obj * 1.5:
            # Step too large, try smaller
            lr = lr * 0.3
            a_new = a - lr * v
            a_new = project_simplex(a_new)
            obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            # Slightly increase momentum when improving
            momentum = min(0.92, momentum * 1.001)
        else:
            no_improve += 1
            momentum = max(0.7, momentum * 0.998)

        a = a_new
        current_lr = max(1e-14, current_lr * 0.9998)
        it += 1

        # Periodic structured restart if stuck
        if no_improve >= restart_patience:
            scale = restart_scale * (1 + 0.5 * (no_improve // restart_patience))
            scale = min(scale, 0.08)
            # Structured perturbation: mix Gaussian with low-frequency noise
            freq = np.fft.rfftfreq(n)
            noise_freq = np.fft.rfft(rng.normal(0, 1, n))
            # Low-pass filter the noise
            noise_freq[1:] *= np.exp(-5.0 * np.abs(freq[1:]))
            low_freq_noise = np.fft.irfft(noise_freq, n)
            perturbation = rng.normal(0, scale * 0.5, n) + low_freq_noise * scale * 0.5
            a = best_a * np.exp(perturbation)
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-10:
                a = a / s_a
            v *= 0.1
            current_lr = lr_start * 0.5
            no_improve = 0
            if it > 500:
                break

        if verbose and it % 200 == 0:
            print(f"        SGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, "
                  f"lr={current_lr:.2e}, t={elapsed:.0f}s")

    return best_a


def peak_subgradient(a, time_budget, K=10, seed=42, verbose=False,
                     lr_start=None, momentum=0.7):
    """
    Subgradient method focusing on top-K peaks.
    More targeted than smoothed GD, good for final refinement.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    if lr_start is None:
        lr_start = 2.5 / np.sqrt(n)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    it = 0
    no_improve = 0
    current_lr = lr_start

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        if max_conv < 1e-15:
            break

        # Get top K peaks with soft weighting
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        # Soft weights: emphasize the largest peak more
        weights = np.exp(12.0 * (peak_vals - peak_vals[0]) / max_conv)
        weights /= weights.sum()

        # Compute subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                indices = np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] += 2.0 * w * a[k - indices]

        # Update
        v = momentum * v + grad
        a_new = a - current_lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            current_lr *= 0.995
            if no_improve > 80:
                current_lr *= 0.6
                v *= 0.3
                no_improve = 0
                if current_lr < 1e-12:
                    a = best_a * np.exp(rng.normal(0, 0.008, n))
                    a = np.maximum(a, 0)
                    s_a = a.sum()
                    if s_a > 1e-10:
                        a = a / s_a
                    v = np.zeros(n)
                    current_lr = lr_start * 0.5
                    if it > 300:
                        break

        it += 1
        if verbose and it % 100 == 0:
            print(f"        Peak it{it}: obj={best_obj:.8f}, K={K}, t={elapsed:.0f}s")

    return best_a


def dual_ascent_direction(a, iterations=200, eta=None):
    """
    Solve the dual of the LP direction-finding problem using
    multiplicative weights / dual ascent.
    Returns a direction g (normalized to sum=1) that minimizes
    max(a*g) approximately.
    
    Primal LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0
    Dual LP:   max sum(y) s.t. sum_k y[k]*a[k-j] <= 1 for all j, y>=0
    """
    n = len(a)
    num_peaks = 2 * n - 1
    a = a / a.sum()

    if eta is None:
        eta = 1.0 / np.sqrt(n)

    y = np.ones(num_peaks) / num_peaks

    # Build the constraint matrix implicitly
    # A^T y: for each j, sum_k y[k] * a[k-j] * I(0<=k-j<n)
    # This is the cross-correlation of y and a

    for it in range(iterations):
        # Compute A^T y (the constraint values)
        # A^T y [j] = sum_k y[k] * a[k-j] * I(0 <= k-j < n)
        # = sum_m y[m+j] * a[m] (sub m = k-j)
        # This is cross-correlation of y and a
        a_rev = a[::-1]
        N = 1
        while N < 3 * n - 1:
            N <<= 1
        cross = irfft(rfft(a_rev, N) * rfft(y, N), N)
        Ay = cross[n - 1: 2 * n - 1]  # Ay[j] = sum_m a[m]*y[m+j]

        # Find violated constraints
        violations = np.maximum(Ay - 1.0, 0)
        max_viol = float(np.max(violations)) if len(violations) > 0 else 0

        if max_viol < 1e-8:
            break

        # Multiplicative update on y
        # Increase y[k] proportionally to how well it helps (lowers max constraint)
        # Dual subgradient: for each k, the "contribution" is
        # 1 - max_j (A^T y)[j] * I(k is in the path to j)
        # Simplified: update y[k] based on the constraint it most affects
        # Use: y[k] *= exp(eta * (1 - Ay_at_k_contribution))
        
        # For efficiency, compute the "impact" of each y[k] on the max violation
        # impact[k] = max over j where a[k-j] > 0 of violations[j] * a[k-j]
        # This is expensive; use a simpler heuristic:
        # y[k] *= exp(eta * (1 - sum_j violations[j] * a[k-j] / sum_j a[j]^2))
        
        # Simpler: use the subgradient of the dual objective
        # The dual is max sum(y) s.t. A^T y <= 1
        # Subgradient of constraint g_j(y) = (A^T y)[j] - 1 is A[:, j]
        # So: y *= exp(-eta * subgradient_of_max_violation)
        
        # Find which constraints are most violated
        violated_idx = np.where(violations > 0.01 * max_viol)[0]
        if len(violated_idx) == 0:
            break
            
        # For each k, compute how much it contributes to violations
        # contribution[k] = sum_{j in violated} violations[j] * a[k-j] * I(0<=k-j<n)
        # This is a weighted sum of shifted a's
        contribution = np.zeros(num_peaks)
        for j in violated_idx:
            j = int(j)
            # y[k] contributes to Ay[j] via a[k-j] when 0<=k-j<n
            # So contribution from constraint j to y[k] is violations[j] * a[k-j]
            k_start = j  # k-j = 0 => k = j
            k_end = j + n - 1  # k-j = n-1 => k = j+n-1
            k_start = max(0, k_start)
            k_end = min(num_peaks - 1, k_end)
            if k_start <= k_end:
                contribution[k_start:k_end+1] += violations[j] * a[:k_end-k_start+1]

        # Update
        max_contrib = float(np.max(contribution))
```

```python
import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_grad(a, tau):
    """Soft-max of convolution and gradient. Assumes sum(a)=1."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    total = float(np.sum(exp_vals))
    p = exp_vals / total
    soft_max_val = max_conv + np.log(total) / tau
    a_rev = a[::-1]
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    cross = irfft(rfft(a_rev, N) * rfft(p, N), N)
    c = cross[n - 1: 2 * n - 1]
    grad = 4.0 * n * c
    return soft_max_val, grad, p, conv


def smoothed_gd(a, time_budget, tau_start=5.0, tau_end=500.0,

def peak_subgradient(a, time_budget, K=8, seed=42, verbose=False):

            k = int(k)

        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

def alternating_optimize(a, time_budget, seed=42, verbose=False):
    """
    Alternate between smoothed GD and peak subgradient with adaptive switching.
    This helps escape plateaus that one method alone might get stuck in.
    """

    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Cycle through: GD -> Peak -> GD -> Peak(...)
    phase = 0
    time_per_phase = 30.0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < time_per_phase:
            time_per_phase = max(remaining, 5.0)

        if phase % 2 == 0:
            # Smoothed GD phase
            tau_s = max(5.0, (elapsed / time_budget) * 200.0)
            tau_e = tau_s * 15.0
            a_opt = smoothed_gd(a, time_budget=time_per_phase,
                                tau_start=tau_s, tau_end=tau_e,
                                seed=seed + phase * 10)
        else:
            # Peak subgradient phase
            K = max(3, 12 - phase)
            a_opt = peak_subgradient(a, time_budget=time_per_phase,
                                     K=K, seed=seed + phase * 10)

        obj = compute_obj(a_opt)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_opt.copy()
            a = a_opt.copy()
        else:
            # Small perturbation before next phase
            a = best_a * np.exp(rng.normal(0, 0.005, n))
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-10:
                a = a / s_a

        phase += 1
        if verbose and phase % 4 == 0:
            print(f"        Alt ph{phase}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a
```
