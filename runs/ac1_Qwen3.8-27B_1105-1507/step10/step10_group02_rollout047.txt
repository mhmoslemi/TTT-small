"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid approach: LP-based direction finding with proper line search,
exact coordinate descent with active set tracking, Nesterov-accelerated
smoothed gradient descent for escape, and multi-n exploration.
Key improvement over previous: proper 1D line search along LP directions
and adaptive block coordinate descent.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT. Returns array of length 2n-1."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv)/sum^2. Assumes a is non-negative."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, 2 * j + n)
        length = min(n, 2 * n - 1 - j)
        A[k_start:k_start + length, j] = a[:length]
    return A


def solve_lp_direction(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def line_search_lp_direction(a, g):
    """
    Proper 1D line search: minimize f(t) = 2n * max_k(conv((1-t)a+tg, (1-t)a+tg)[k])
    over t in [0, 1]. Uses the fact that conv is quadratic in t.
    """
    n = len(a)
    a = a / float(np.sum(a))
    g = g / float(np.sum(g))

    C_aa = compute_conv(a)
    C_ag = compute_conv_cross(a, g)
    C_gg = compute_conv(g)

    def f_of_t(t):
        vals = (1 - t) ** 2 * C_aa + 2 * t * (1 - t) * C_ag + t ** 2 * C_gg
        return 2.0 * n * float(np.max(vals))

    # Golden section search on [0, 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_lo, a_hi = 0.0, 1.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f_of_t(c)
    fd = f_of_t(d)

    for _ in range(60):
        if a_hi - a_lo < 1e-12:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f_of_t(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f_of_t(d)

    t_opt = (a_lo + a_hi) / 2.0
    return t_opt, f_of_t(t_opt)


def lp_optimize(a, time_budget=300.0, verbose=True):
    """Iterative LP direction finding with proper line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(best_a)
    start = time.monotonic()
    iteration = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g = solve_lp_direction(a)
        if g is None:
            break

        t_opt, obj_at_t = line_search_lp_direction(a, g)

        if t_opt > 1e-6 and obj_at_t < best_obj - 1e-10:
            a = (1 - t_opt) * a + t_opt * g
            a = np.maximum(a, 0)
            s_new = float(np.sum(a))
            if s_new > 1e-10:
                a = a / s_new
                obj_new = compute_obj_normalized(a)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
                    if verbose and iteration % 5 == 0:
                        print(f"    LP iter {iteration}: obj={best_obj:.8f}, t={t_opt:.4f}")
        else:
            # Try smaller steps
            found = False
            for t_try in [0.1, 0.05, 0.02, 0.01]:
                a_try = (1 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj_normalized(a_try)
                    if obj_try < best_obj - 1e-12:
                        a = a_try
                        best_obj = obj_try
                        best_a = a_try.copy()
                        found = True
                        break
            if not found:
                # Perturb to escape
                rng = np.random.default_rng(iteration * 7919)
                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()

        iteration += 1

    return best_a


def coordinate_descent(a, time_budget=300.0, verbose=True, seed=42):
    """
    Exact coordinate descent with active set tracking.
    Each coordinate is optimized via golden section on the exact 1D subproblem.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    no_improve_count = 0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 5 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1

        order = rng.permutation(n)
        S = float(np.sum(a))
        sweep_improved = False

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v

            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks]

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            def f_1d(x, v=v, S_other=S_other, n=n, c_k_lin=c_k_lin,
                     a_ki_lin=a_ki_lin, c_2i=c_2i):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.5, v * 0.8, v, v * 1.2, v * 1.5,
                          v * 2.0, 2.0 / n, 1.0 / np.sqrt(n)]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 3.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(40):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta
            sweep_improved = True

        if not sweep_improved and no_improve_count > 10:
            # Restart with perturbation
            rng2 = np.random.default_rng(sweep * 104729)
            pert = rng2.normal(0, 0.05, n)
            a = a * np.exp(pert)
            a = np.maximum(a, 0)
            a = a / a.sum()
            conv = compute_conv(a)
            no_improve_count = 0

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=30.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated smoothed gradient descent for escape from local minima."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    beta = 0.9
    lr_base = 5.0 / np.sqrt(n)
    v = np.zeros(n)

    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_fft = fft(y, N)
        g_full = ifft(P * np.conj(A_fft)).real
        grad = 2.0 * g_full[:n]

        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            lr = lr_base / (g_norm * n + 1.0)
            lr = min(lr, lr_base)
        else:
            lr = lr_base

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 2000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x ** 2)
        a = a / a.sum()
    elif mode == 4:
        # Symmetric window (Hann-like)
        x = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Flat top window
        a = np.ones(n)
        edge = max(1, n // 10)
        for i in range(edge):
            decay = np.cos(np.pi * i / (2 * edge)) ** 2
            a[i] = decay
            a[n - 1 - i] = decay
        a = a / a.sum()
    elif mode == 6:
        # Three bumps
        centers = [n // 6, n // 2, 5 * n // 6]
        widths = [n // 10, n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines LP direction finding with line search,
    coordinate descent, Nesterov smoothed GD, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj_normalized(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj_normalized(best_seq)

    # === Phase 1: LP with proper line search (main innovation) ===
    print(f"\n=== Phase 1: LP + Line Search ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.3, 400.0)
    if p1_budget > 60:
        a_lp = lp_optimize(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj_normalized(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Coordinate Descent ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p2_budget,
                                  verbose=True, seed=seed)
        obj_cd = compute_obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 3: More LP (after CD refinement) ===
    print(f"\n=== Phase 3: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)
    if p3_budget > 30:
        a_lp2 = lp_optimize(best_seq, time_budget=p3_budget, verbose=True)
        obj_lp2 = compute_obj_normalized(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP2: obj={best_score:.8f} **")
        else:
            print(f"  LP2: obj={obj_lp2:.8f}")

    # === Phase 4: CD refinement ===
    print(f"\n=== Phase 4: CD Refinement ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 200.0)
    if p4_budget > 20:
        a_cd2 = coordinate_descent(best_seq, time_budget=p4_budget,
                                   verbose=False, seed=seed + 100)
        obj_cd2 = compute_obj_normalized(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  CD2: obj={best_score:.8f} **")
        else:
            print(f"  CD2: obj={obj_cd2:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 200), max(400, n_best - 100),
        n_best + 100, n_best + 200, n_best + 400, n_best + 800,
        600, 800, 1000, 1200, 1500, 2000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 3000 and abs(n - n_best) > 50]
    # Sort by distance from current best to try closest first
    n_options.sort(key=lambda x: abs(x - n_best))
    n_options = n_options[:6]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick CD on new length
        alloc_cd = min(remaining * 0.15, 60.0)
        if alloc_cd > 15:
            a_try = coordinate_descent(a_try, time_budget=alloc_cd,
                                       verbose=False, seed=n_try)

        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_score:
            # Try LP on the improved sequence
            remaining2 = deadline - time.monotonic()
            alloc_lp = min(remaining2 * 0.1, 50.0)
            if alloc_lp > 10:
                a_try_lp = lp_optimize(a_try, time_budget=alloc_lp, verbose=False)
                obj_lp_try = compute_obj_normalized(a_try_lp)
                if obj_lp_try < obj_try:
                    a_try = a_try_lp
                    obj_try = obj_lp_try
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Nesterov GD escape ===
    print(f"\n=== Phase 6: GD Escape ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        pert = rng.normal(0, 0.04, len(best_seq))
        a_pert = best_seq * np.exp(pert)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.2, 100.0)
        a_gd = nesterov_smoothed_descent(a_pert, time_budget=alloc,
                                         tau_init=20.0, tau_end=600.0, verbose=True)
        obj_gd = compute_obj_normalized(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  GD escape: obj={best_score:.8f} **")
        else:
            print(f"  GD escape: obj={obj_gd:.8f}")

    # === Phase 7: Diverse restarts with LP ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = 1 + r
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 60.0)
        if alloc > 20:
            a_r = coordinate_descent(a_r, time_budget=alloc * 0.6,
                                     verbose=False, seed=5000 + r * 100)
            # Quick LP polish
            remaining_r = deadline - time.monotonic()
            alloc_lp_r = min(remaining_r * 0.1, 20.0)
            if alloc_lp_r > 5:
                a_r = lp_optimize(a_r, time_budget=alloc_lp_r, verbose=False)

            obj_r = compute_obj_normalized(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # LP
        alloc_lp = min(remaining * 0.4, 100.0)
        if alloc_lp > 10:
            a_final = lp_optimize(best_seq, time_budget=alloc_lp, verbose=False)
            obj_final = compute_obj_normalized(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # CD
        remaining = deadline - time.monotonic()
        alloc_cd = min(remaining * 0.4, 80.0)
        if alloc_cd > 10:
            a_final_cd = coordinate_descent(best_seq, time_budget=alloc_cd,
                                            verbose=False, seed=99999)
            obj_final_cd = compute_obj_normalized(a_final_cd)
            if obj_final_cd < best_score:
                best_score = obj_final_cd
                best_seq = a_final_cd.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
        print(f"  evaluate_sequence confirmed: {val:.8f}")
    elif val < np.inf:
        print(f"  evaluate_sequence: {val:.8f} (slightly different from internal {best_score:.8f})")
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
