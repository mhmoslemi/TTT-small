    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    if s < 0.01:

    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):

    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def project_simplex(v):
    # Project onto probability simplex
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
        cp.norm(delta, 2) <= r,

At the optimum (for a fixed set of active peaks), the KKT conditions give:
`2 * sum_k lambda_k * a_shifted_k = mu * 1 + nu` (where nu >= 0 for inactive variables)

For active variables (a_i > 0): `2 * sum_k lambda_k * a[k_i - i] = mu`

This is a system of equations that we can try to solve iteratively.

**Fixed-point iteration:**
1. Initialize `a`
2. Compute `c = conv(a, a)`
3. Find active peaks, compute `lambda_k` (e.g., uniformly over top K)
4. Compute `g_i = 2 * sum_k lambda_k * a[k_i - i]`
5. Find `mu` such that `a_i * (g_i - mu) = 0` for all `i` (complementary slackness)
   - This means: for `a_i > 0`, we need `g_i = mu`
   - So `mu = g_i` for all active `i`
   - This is only possible if all active `g_i` are equal
   - If not, we need to adjust `a`
6. Update: `a_i <- a_i * (mu / g_i)^alpha` for `g_i > 0` (multiplicative update)
7. Renormalize

This is a "power iteration" style approach. The idea is that elements with high gradient (contributing a lot to peaks) get reduced, and elements with low gradient get increased.

"""
KKT Fixed-Point + Trust-Region SOCP Hybrid Optimizer for minimum peak autocorrelation.
Novel peak-equalization via KKT complementary slackness, combined with adaptive
trust-region SOCP polishing, smoothed gradient warmup, and structured multi-restart.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_pressure(a, peak_indices):
    """Compute pressure on each element from given peak lags.
    pressure[i] = sum_k a[k-i] for valid k-i indices."""
    n = len(a)
    pressure = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            pressure[idx] += a[k - idx]
    return pressure


def kkt_fixed_point(a, time_budget=120.0, K=5, alpha_init=0.4, verbose=False, seed=42):
    """
    KKT fixed-point iteration targeting complementary slackness.
    Multiplicative update: a_i <- a_i * (mu/g_i)^alpha for active i.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = alpha_init
    it = 0
    stagnant = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))

        top_idx = np.argsort(conv)[::-1][:K]
        top_vals = conv[top_idx].copy()
        # Weight: peaks closer to max get higher weight
        spread = max(M - top_vals[-1], 1e-15)
        weights = 1.0 + (top_vals - top_vals[-1]) / spread
        weights /= weights.sum()

        # Compute gradient g_i = 2 * sum_k w_k * a[k-i]
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                g[idx] += 2.0 * w * a[k - idx]

        active = a > 1e-10
        if not active.any():
            break
        mu = float(g[active].mean())

        if mu < 1e-15:
            break

        ratio = np.ones(n)
        ratio[active] = (mu / (g[active] + 1e-15)) ** alpha
        a_new = a * ratio
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
            alpha = min(alpha * 1.01, 0.8)
        else:
            stagnant += 1
            alpha *= 0.92
            if stagnant > 40 and alpha < 0.05:

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    KKT it{it}: obj={best_obj:.8f}, alpha={alpha:.4f}, t={elapsed:.0f}s")

def smoothed_gradient_descent(a, time_budget=100.0, tau_init=20.0, tau_end=400.0, verbose=False):
    """Smoothed gradient descent with increasing temperature (softmax over peaks)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 4.0 / np.sqrt(n)
    momentum = 0.82
    v = np.zeros(n)
    it = 0

    N = 1
    while N < 3 * n:
        N <<= 1

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        total = float(np.sum(exp_vals))
        if total < 1e-30:
            break
        p = exp_vals / total

        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr = max(1e-12, lr * 0.9992)
        it += 1

    return best_a


def solve_sscp(a, r, M_target=None):
    """Solve trust region SOCP for direction finding."""
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)

    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta

    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2.0 * (A_cross @ delta) <= M,
        cp.norm(delta, 2) <= r,
    ]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_direction(a, M_target=None):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

        a_sgd = smoothed_gradient_descent(best_seq, time_budget=p2_budget,
                                          tau_init=15.0, tau_end=300.0)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  GD: obj={best_score:.8f} **")
        else:
            print(f"  GD: obj={obj_sgd:.8f}")

    # === Phase 3: SOCP Trust-Region (primary optimizer) ===
    print(f"\n=== Phase 3: SOCP Trust-Region ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.30, 350.0)
    if p3_budget > 30:
        a_sscp = sscp_polish(best_seq, time_budget=p3_budget,
                             r_init=0.35, verbose=True, seed=42)
        obj_sscp = compute_obj(a_sscp)
        if obj_sscp < best_score:
            best_score = obj_sscp
            best_seq = a_sscp.copy()
            print(f"  SOCP: obj={best_score:.8f} **")
        else:
            print(f"  SOCP: obj={obj_sscp:.8f}")

    # === Phase 4: LP Polish (complementary direction) ===
    print(f"\n=== Phase 4: LP Polish ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.12, 100.0)
    if p4_budget > 15:
        a_lp = lp_polish(best_seq, time_budget=p4_budget, verbose=True, seed=77)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 5: KKT again (after SOCP, may find new basin) ===
    print(f"\n=== Phase 5: KKT Refinement ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.08, 70.0)
    if p5_budget > 10:
        a_kkt2 = kkt_fixed_point(best_seq, time_budget=p5_budget, K=7,
                                 alpha_init=0.3, verbose=False, seed=200)
        obj_kkt2 = compute_obj(a_kkt2)
        if obj_kkt2 < best_score:
            best_score = obj_kkt2
            best_seq = a_kkt2.copy()
            print(f"  KKT2: obj={best_score:.8f} **")
        else:
            print(f"  KKT2: obj={obj_kkt2:.8f}")

    # === Phase 6: Multi-restart with structured perturbations ===
    print(f"\n=== Phase 6: Multi-Restart ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r_idx in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        # Structured perturbation: anti-gradient based
        conv = compute_conv(best_seq)
        M = float(np.max(conv))
        top_idx = np.argsort(conv)[::-1][:5]
        pressure = compute_pressure(best_seq, top_idx)
        pressure = pressure / (pressure.max() + 1e-15)

        scale = 0.006 + 0.004 * r_idx
        noise = rng.normal(0, scale, n_current)
        # Bias perturbation away from high-pressure elements
        noise -= 0.5 * scale * pressure
        a_pert = best_seq * np.exp(noise)
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        # Quick KKT warmup
        alloc_kkt = min(remaining * 0.06, 20.0)
        if alloc_kkt > 8:
            a_pert = kkt_fixed_point(a_pert, time_budget=alloc_kkt, K=5,
                                     alpha_init=0.3, seed=300 + r_idx)

        # SOCP polish
        remaining = deadline - time.monotonic()
        alloc_sscp = min(remaining * 0.20, 70.0)
        if alloc_sscp > 15:
            a_pert = sscp_polish(a_pert, time_budget=alloc_sscp,
                                 r_init=0.2, seed=500 + r_idx)

        obj_pert = compute_obj(a_pert)
        if obj_pert < best_score:
            best_score = obj_pert
            best_seq = a_pert.copy()
            print(f"  Restart{r_idx} (scale={scale:.4f}): obj={best_score:.8f} **")
        else:
            print(f"  Restart{r_idx} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(500, n_current - 150),
        max(600, n_current - 50),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        1000, 1200, 1400, 1600
    ]))
    n_options = [n for n in n_options if 500 <= n <= 2000 and abs(n - n_current) > 60]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)

        alloc1 = min(remaining * 0.10, 25.0)
        if alloc1 > 8:
            a_try = kkt_fixed_point(a_try, time_budget=alloc1, K=5,
                                    alpha_init=0.4, seed=700)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.25, 80.0)
        if alloc2 > 15:
            a_try = sscp_polish(a_try, time_budget=alloc2, r_init=0.25)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final intensive SOCP + KKT ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 120.0)
        a_final = sscp_polish(best_seq, time_budget=alloc,
                              r_init=0.15, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final SOCP: obj={best_score:.8f} **")
        else:
            print(f"  Final SOCP: obj={obj_final:.8f}")

    # Quick KKT at the end
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.3, 40.0)
        a_final_kkt = kkt_fixed_point(best_seq, time_budget=alloc, K=8,
                                      alpha_init=0.2, seed=1111)
        obj_fk = compute_obj(a_final_kkt)
        if obj_fk < best_score:
            best_score = obj_fk
            best_seq = a_final_kkt.copy()
            print(f"  Final KKT: obj={best_score:.8f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.25, 30.0)
        a_flp = lp_polish(best_seq, time_budget=alloc, seed=2000)
        obj_flp = compute_obj(a_flp)
        if obj_flp < best_score:
            best_score = obj_flp
            best_seq = a_flp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
