"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses accelerated alternating LP, spectral initialization, multi-scale optimization,
and high-precision smoothed gradient descent with adaptive tau scheduling.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import time
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_minmax_lp(a, time_limit=10):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:
            continue
    return None, np.inf

def alternating_lp(a, time_budget, verbose=False):
    """Accelerated alternating LP with momentum and adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    alpha = 0.5  # mixing parameter
    momentum = 0.0
    v = np.zeros(n)  # momentum velocity
    stall = 0
    
    it = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Solve LP
        g_lp, M_lp = solve_minmax_lp(a)
        if g_lp is None:
            break
        
        # Compute target incorporating momentum
        target = g_lp + momentum * v
        
        # Line search over mixing parameter
        best_t = alpha
        best_obj_t = best_obj
        
        for t in np.concatenate([np.linspace(0.05, 1.0, 15), 
                                   [alpha * 0.5, alpha * 0.7, alpha * 1.3, alpha * 1.5]]):
            t = np.clip(t, 0.01, 1.0)
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t
        
        if best_obj_t < best_obj - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            
            # Update momentum
            v = 0.8 * v + 0.2 * (a_new - a)
            a = a_new
            best_obj = best_obj_t
            best_a = a_new.copy()
            alpha = best_t
            stall = 0
        else:
            stall += 1
            alpha *= 0.8
            if stall > 10:
                break
        
        it += 1
        if verbose and it % 5 == 0:
            print(f"    ALP it{it}: obj={best_obj:.8f}, alpha={alpha:.3f}, t={time.monotonic()-start:.0f}s")
    
    return best_a

def spectral_init(n, rng):
    """Initialize with flat power spectrum, clipped to non-negative."""
    # Create a random phase spectrum
    freqs = np.arange(n)
    phases = rng.uniform(0, 2*np.pi, n)
    amplitudes = np.ones(n)
    spectrum = amplitudes * np.exp(1j * phases)
    
    # Make it Hermitian (for real signal)
    spectrum[0] = 1.0
    if n % 2 == 0:
        spectrum[n//2] = 1.0
    
    a = irfft(spectrum, n).real
    # Take absolute value and normalize
    a = np.abs(a)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a / a.sum()
    elif mode == 3:
        # Spectral init
        a = spectral_init(n, rng)
    elif mode == 4:
        # Gaussian
        x = np.linspace(-3, 3, n)
        a = np.exp(-x**2)
        a = a / a.sum()
    elif mode == 5:
        # Symmetric bump
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
        a = a / a.sum()
    elif mode == 6:
        # Two bumps
        x = np.linspace(0, 1, n)
        a = np.exp(-((x-0.3)/0.1)**2) + np.exp(-((x-0.7)/0.1)**2)
        a = a / a.sum()
    elif mode == 7:
        # Symmetric spectral
        a = spectral_init(n, rng)
        a = (a + a[::-1]) / 2  # enforce symmetry
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a

def smoothed_optimize(a, time_budget, tau_start=100, tau_end=10000, verbose=False):
    """Smoothed gradient descent with Nesterov acceleration and adaptive tau."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0
    
    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.4
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
            beta = 0.9
        else:
            y = y / s_y
        
        # Compute smoothedobjective gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp
        
        # Gradient via FFT: grad_j = 2 * sum_k p_k * y_{k-j}
        P = fft(p, N_fft)
        Y = fft(y, N_fft)
        g_full = ifft(P * np.conj(Y)).real
        g = 2.0 * g_full[:n]
        
        # Project to tangent space of simplex (after normalization)
        g = g - np.sum(y * g)
        
        # Adaptive step size
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        lr = 1.0 / (g_norm + 1e-10)
        lr = min(lr, 2.0 / np.sqrt(n))
        lr = max(lr, 1e-10)
        
        # Update
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        beta = min(0.95, 0.9 + 0.0002 * it)
        
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            beta = 0.9
            best_it = it
        
        if verbose and it % 2000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.0f}s")
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def log_param_optimize(a, time_budget, verbose=False):
    """Optimize using log-parameterization with L-BFGS."""
    full_len = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    # Identify support
    threshold = np.percentile(a[a > 0], 5) if np.any(a > 0) else 0
    support = np.where(a > threshold)[0]
    if len(support) < 2:
        support = np.where(a > a.max() * 1e-8)[0]
    if len(support) < 2:
        return a
    
    b0 = np.log(np.maximum(a[support], 1e-30))
    b0 = b0 - b0.mean()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Multi-stage tau
    tau_values = [100, 500, 2000, 8000]
    current_b = b0.copy()
    
    for tau in tau_values:
        if time.monotonic() - start > time_budget:
            break
        
        stage_budget = time_budget / len(tau_values)
        
        def obj_grad(b_vec):
            b_shifted = b_vec - b_vec.max()
            a_supp = np.exp(b_shifted)
            a_supp = a_supp / a_supp.sum()
            a_full = np.zeros(full_len)
            a_full[support] = a_supp
            
            conv = compute_conv(a_full)
            max_conv = float(np.max(conv))
            exp_vals = np.exp(tau * (conv - max_conv))
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp
            
            obj_val = float(max_conv + np.log(sum_exp) / tau)
            
            # Gradient w.r.t. a_full
            N = 1
            while N < 3 * full_len:
                N <<= 1
            P = fft(p, N)
            A_f = fft(a_full, N)
            g_full = ifft(P * np.conj(A_f)).real
            grad_full = 2.0 * g_full[:full_len]
            
            # Chain rule for log params (softmax)
            grad_supp = grad_full[support]
            grad_b = a_supp * (grad_supp - np.sum(a_supp * grad_supp))
            
            return obj_val, grad_b
        
        try:
            res = scipy_minimize(obj_grad, current_b, method='L-BFGS-B',
                                 jac=True, options={'maxiter': 100, 'gtol': 1e-10, 'ftol': 1e-14})
            if res.x is not None:
                b_shifted = res.x - res.x.max()
                a_supp = np.exp(b_shifted)
                a_supp = a_supp / a_supp.sum()
                a_new = np.zeros(full_len)
                a_new[support] = a_supp
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    current_b = res.x.copy()
                    if verbose:
                        print(f"    LogOpt tau={tau}: obj={best_obj:.8f}")
        except Exception:
            pass
    
    return best_a

def multi_start_optimize(n, rng, time_budget, verbose=False):
    """Try multiple initializations and keep the best."""
    best_a = None
    best_obj = np.inf
    
    n_starts = min(6, max(3, int(time_budget / 30)))
    per_start_budget = time_budget / n_starts
    
    for i in range(n_starts):
        remaining = time_budget - 0  # simplified
        if remaining < 20:
            break
        
        mode = i % 8
        a_init = diverse_init(n, rng, mode=mode)
        
        # Quick LP polish
        a_try = alternating_lp(a_init, time_budget=per_start_budget * 0.4, verbose=False)
        
        # GD refinement
        a_try = smoothed_optimize(a_try, time_budget=per_start_budget * 0.4, 
                                   tau_start=50, tau_end=3000, verbose=False)
        
        # Log param
        a_try = log_param_optimize(a_try, time_budget=per_start_budget * 0.2, verbose=False)
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            if verbose:
                print(f"    MultiStart {i} (mode={mode}): obj={best_obj:.8f}")
    
    return best_a, best_obj
