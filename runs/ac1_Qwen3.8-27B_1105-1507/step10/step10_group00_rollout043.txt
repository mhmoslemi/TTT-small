import numpy as np

from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_smoothed_obj(a, tau):
    """Compute smoothed objective using logsumexp."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf, None
    a_norm = a / s
    conv = compute_conv(a_norm)
    # logsumexp for numerical stability
    c = conv / tau
    c_max = np.max(c)
    log_sum_exp = c_max + np.log(np.sum(np.exp(c - c_max)))
    obj = 2.0 * n * tau * log_sum_exp
    # softmax weights
    weights = np.exp(c - c_max) / np.sum(np.exp(c - c_max))
    return obj, weights

    # where dc_norm[k]/da_norm[j] = 2*a_norm[k-j]
    grad = 2.0 * n * grad
    return grad, obj

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n + 1) - (cssv - 1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    import cvxpy as cp
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def line_search_blend(a, g_lp, time_budget=3.0):

    gr = 2 - phi
    lo, hi = 0.0, 1.2
    
    def obj_at(t):
        if t < 0:
            t = 0
        a_try = (1 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)
    
    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)
    
    start = time.monotonic()
    for _ in range(25):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)
    
    t_best = c if fc < fd else d
    obj_best = min(fc, fd)
    
    for t_try in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
        obj_try = obj_at(t_try)
        if obj_try < obj_best:
            obj_best = obj_try
            t_best = t_try

def try_length(a_best, n_try, time_budget, rng):
    """Try a different sequence length by interpolation and refinement."""
    n_old = len(a_best)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        return None, np.inf
    a_try = a_try / s
    
    # Quick refinement
    if time_budget > 30:
        a_try = fast_subgradient_descent(a_try, time_budget=min(15.0, time_budget * 0.3))
    
    if time_budget > 50:
        a_try = lp_polish(a_try, time_budget=min(30.0, time_budget * 0.5))
    
    obj_try = compute_obj(a_try)
    return a_try, obj_try

"""
Hybrid optimization for step function coefficients using FFT-accelerated
smoothed gradient descent, coordinate descent with golden section search,
and LP polishing. Explores multiple lengths and initializations.
Key improvements: O(n log n) gradient via FFT, coordinate-wise convex
optimization, adaptive temperature annealing, and aggressive length search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) - (cssv - 1) > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def compute_smoothed_grad_fft(a, tau):
    """
    Compute gradient of smoothed objective 2n*tau*logsumexp(conv/tau)
    using FFT for the cross-correlation. Returns (gradient, objective).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.zeros(n), np.inf
    a = a / s
    conv = compute_conv(a)

    c = conv / tau
    c_max = float(np.max(c))
    exp_c = np.exp(c - c_max)
    exp_sum = float(np.sum(exp_c))
    weights = exp_c / exp_sum

    # Cross-correlation R[j] = sum_k weights[k] * a[k-j] via FFT
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    W = rfft(weights, N)
    A = rfft(a, N)
    R = irfft(np.conj(W) * A, N)

    # Gradient = 4n * R[:n]
    grad = 4.0 * n * R[:n]

    # Smoothed objective value
    obj = 2.0 * n * tau * (c_max + np.log(exp_sum))

    return grad, obj


def smoothed_descent(a, time_budget=100.0, verbose=False):
    """Smoothed subgradient descent with adaptive temperature and momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = 0.02
    tau_min = 0.001
    tau_decay = 0.997
    lr = 0.3
    momentum = np.zeros(n)
    beta = 0.85
    it = 0
    no_improve = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        grad, _ = compute_smoothed_grad_fft(a, tau)

        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        tau = max(tau_min, tau * tau_decay)

        if no_improve > 100:
            lr *= 0.85
            no_improve = 0
        elif no_improve > 500:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = 0.15
            tau = 0.01
            no_improve = 0
            best_it = it

        it += 1
        if verbose and it % 1000 == 0:
            print(f"    SD it{it}: obj={best_obj:.8f}, tau={tau:.5f}, lr={lr:.4f}, t={time.monotonic()-start:.1f}s")

            k_lo = j
            k_hi = min(2 * n - 2, j + n - 1)
            is_2j_in_range = (2 * j <= 2 * n - 2) and (2 * j >= k_lo) and (2 * j <= k_hi)

            def eval_trial(val):
                """Evaluate max conv when a[j] = val (keeping others fixed)."""
                # Change in conv:
                # For k in [k_lo, k_hi], k != 2j: delta = 2*(val - old_val)*a[k-j]
                # For k = 2j (if in range): delta = val^2 - old_val^2
                # We need max of (conv + delta) over all k
                # Unaffected max:
                if k_lo > 0 or k_hi < 2 * n - 2:
                    # Compute max of unaffected region
                    if k_lo > 0:
                        m1 = float(np.max(conv[:k_lo])) if k_lo > 0 else 0.0
                    else:
                        m1 = 0.0
                    if k_hi < 2 * n - 2:
                        m2 = float(np.max(conv[k_hi + 1:]))
                    else:
                        m2 = 0.0
                    m_unchanged = max(m1, m2)
                else:
                    m_unchanged = 0.0

                # Affected region
                delta_base = 2.0 * (val - old_val)
                max_aff = -np.inf
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * j and is_2j_in_range:
                        v = conv[k] + val * val - old_val * old_val
                    else:
                        if 0 <= k - j < n:
                            v = conv[k] + delta_base * a[k - j]
                        else:
                            v = conv[k]
                    if v > max_aff:
                        max_aff = v

                return max(m_unchanged, max_aff)

            # Golden section
            phi = (1 + np.sqrt(5)) / 2
            gr = 2 - phi
            c = lo + gr * (hi - lo)
            d = lo + (1 - gr) * (hi - lo)
            fc = eval_trial(c)
            fd = eval_trial(d)

            for _ in range(40):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = lo + gr * (hi - lo)
                    fc = eval_trial(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = eval_trial(d)

            best_c = c if fc < fd else d
            best_val = min(fc, fd)

            # Also try 0 and the old value
            val_0 = eval_trial(0.0)
            val_old = eval_trial(old_val)

            if val_0 <= best_val and val_0 <= val_old:
                best_c = 0.0
                best_val = val_0
            elif val_old <= best_val:
                best_c = old_val
                best_val = val_old

            # Update if improved
            new_obj_candidate = 2.0 * n * best_val
            if best_c != old_val and new_obj_candidate < best_obj - 1e-12:
                # Update conv
                delta_base = 2.0 * (best_c - old_val)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * j and is_2j_in_range:
                        conv[k] += best_c * best_c - old_val * old_val
                    else:
                        if 0 <= k - j < n:
                            conv[k] += delta_base * a[k - j]
                a[j] = best_c
                best_obj = new_obj_candidate
                best_a = a.copy()
                improved = True

        if verbose and sweep % 50 == 0:
            # Recompute exact objective periodically
            exact_obj = compute_obj(a)
            if exact_obj < best_obj:
                best_obj = exact_obj
                best_a = a.copy()
            else:
                best_obj = exact_obj
                a = best_a.copy()
                conv = compute_conv(a)
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(2.0 * (2 * x - 1) - np.log(2))
    elif mode == 7:
        a = np.ones(n)
        a[::2] = 0.3
    elif mode == 8:
        a = (2 * x - 1) ** 6 + 0.02
    elif mode == 9:
        # Gaussian mixture at 1/3 and 2/3
        a = np.exp(-((idx - n / 3.0) ** 2) / (2 * (n / 8) ** 2)) + \
            np.exp(-((idx - 2 * n / 3.0) ** 2) / (2 * (n / 8) ** 2)) + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    import cvxpy as cp
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g_lp, time_budget=3.0):
    """Find optimal blend factor t that minimizes obj((1-t)*a + t*g_lp)."""
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    lo, hi = 0.0, 1.2

    def obj_at(t):
        if t < 0:
            t = 0
        a_try = (1 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    start = time.monotonic()
    for _ in range(25):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)

    for t_try in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
        obj_try = obj_at(t_try)
        if obj_try < obj_best:
            obj_best = obj_try
            t_best = t_try

    return t_best, obj_best


def lp_polish(a, time_budget=60.0, verbose=False, max_iters=30):
    """LP polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(77777)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        T0 = time.monotonic()
        g_lp, M_lp = solve_lp_minpeak(a)
        T1 = time.monotonic()

        if g_lp is None:
            break

        t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=min(3.0, remaining * 0.1))

        if t_opt > 0 and obj_opt < best_obj - 1e-14:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 3 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, elapsed={time.monotonic()-start:.1f}s")

        if stall >= 3:
            noise = rng.normal(0, 0.015 * stall, n)
            a = best_a + noise
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 0.01:
                a = a / s
            else:
                a = best_a.copy()
            stall = 0

    return best_a


def try_length(a_best, n_try, time_budget, rng):
    """Try a different sequence length."""
    n_old = len(a_best)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        return None, np.inf
    a_try = a_try / s

    if time_budget > 30:
        a_try = fast_peak_descent(a_try, time_budget=min(15.0, time_budget * 0.25))

    if time_budget > 60:
        a_try = coordinate_descent(a_try, time_budget=min(30.0, time_budget * 0.4))

    if time_budget > 90:
        a_try = lp_polish(a_try, time_budget=min(25.0, time_budget * 0.25), max_iters=10)

    obj_try = compute_obj(a_try)
    return a_try, obj_try


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: hybrid of FFT-based smoothed descent, coordinate descent,
    and LP polishing. Explores multiple lengths and diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # === Load parent ===
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default U-shaped init (n={n_current}): obj={best_score:.8f}")

    # Try alternative inits at same length
    for mode in [0, 4]:
        a_alt = diverse_init(n_current, rng, mode=mode)
        obj_alt = compute_obj(a_alt)
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"  Alt init mode={mode}: obj={best_score:.8f} **")

    # === Phase 1: Fast peak descent (quick improvement) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 1: Fast Peak Descent ===")
        alloc = min(80.0, remaining * 0.1)
        a_fpd = fast_peak_descent(best_seq, time_budget=alloc, verbose=True)
        obj_fpd = compute_obj(a_fpd)
        if obj_fpd < best_score:
            best_score = obj_fpd
            best_seq = a_fpd.copy()
            print(f"  FPD: obj={best_score:.8f} **")
        else:
            print(f"  FPD: obj={obj_fpd:.8f} (no improvement)")

    # === Phase 2: Smoothed gradient descent (FFT-accelerated) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 2: Smoothed Gradient Descent ===")
        alloc = min(200.0, remaining * 0.18)
        a_sd = smoothed_descent(best_seq, time_budget=alloc, verbose=True)
        obj_sd = compute_obj(a_sd)
        if obj_sd < best_score:
            best_score = obj_sd
            best_seq = a_sd.copy()
            print(f"  SD: obj={best_score:.8f} **")
        else:
            print(f"  SD: obj={obj_sd:.8f} (no improvement)")

    # === Phase 3: Coordinate descent ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print("\n=== Phase 3: Coordinate Descent ===")
        alloc = min(250.0, remaining * 0.22)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, verbose=True, max_sweeps=800)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 4: LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 4: LP Polish ===")
        alloc = min(250.0, remaining * 0.2)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True, max_iters=40)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 5: More coordinate descent (finer) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print("\n=== Phase 5: Fine Coordinate Descent ===")
        alloc = min(200.0, remaining * 0.18)
        a_cd2 = coordinate_descent(best_seq, time_budget=alloc, verbose=True, max_sweeps=600)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Fine CD: obj={best_score:.8f} **")
        else:
            print(f"  Fine CD: obj={obj_cd2:.8f} (no improvement)")

    # === Phase 6: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 6: Length Exploration ===")
        n_best = len(best_seq)

        n_options = []
        for delta in [-100, -50, 50, 100, 200, 300, -200, 400]:
            n_try = n_best + delta
            if 400 <= n_try <= 2500:
                n_options.append(n_try)
        for n_val in [500, 600, 700, 900, 1000, 1100, 1200, 1500, 2000]:
            if n_val not in n_options and abs(n_val - n_best) >= 30:
                n_options.append(n_val)

        n_options = sorted(set(n_options))
        # Prioritize lengths close to current best
        n_options = sorted(n_options, key=lambda x: abs(x - n_best))[:8]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            a_try, obj_try = try_length(best_seq, n_try, remaining * 0.5, rng)
            if a_try is not None and obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 7: Diverse Restarts ===")
        n_cur = len(best_seq)

        for r in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            mode = (r + 3) % 10
            a_r = diverse_init(n_cur, rng, mode=mode)
            blend = 0.3 + 0.1 * r
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            if remaining > 60:
                a_r = fast_peak_descent(a_r, time_budget=min(15.0, remaining * 0.15))
            if remaining > 100:
                a_r = coordinate_descent(a_r, time_budget=min(30.0, remaining * 0.25), max_sweeps=200)
            if remaining > 150:
                a_r = lp_polish(a_r, time_budget=min(30.0, remaining * 0.2), max_iters=10)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final intensive refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 8: Final Refinement ===")
        for round_num in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            if round_num % 3 == 0:
                alloc = min(remaining * 0.3, 60.0)
                a_fin = smoothed_descent(best_seq, time_budget=alloc)
            elif round_num % 3 == 1:
                alloc = min(remaining * 0.3, 60.0)
                a_fin = coordinate_descent(best_seq, time_budget=alloc, max_sweeps=200)
            else:
                alloc = min(remaining * 0.3, 60.0)
                a_fin = lp_polish(best_seq, time_budget=alloc, max_iters=15)

            obj_fin = compute_obj(a_fin)
            if obj_fin < best_score:
                best_score = obj_fin
                best_seq = a_fin.copy()
                print(f"  Round {round_num}: obj={best_score:.8f} **")
            else:
                print(f"  Round {round_num}: obj={obj_fin:.8f}")

    # === Phase 9: Ultra-fine CD if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        a_ultra = coordinate_descent(best_seq, time_budget=min(remaining - 5, 80.0), max_sweeps=400)
        obj_ultra = compute_obj(a_ultra)
        if obj_ultra < best_score:
            best_score = obj_ultra
            best_seq = a_ultra.copy()
            print(f"  Ultra CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
