Now, the general problem: for large n, what's the minimum of 2*n*max(conv(a,a))/(sum(a))^2?

This is related to the "minimum peak-to-average power ratio" (PAPR) problem, or more precisely, the aperiodic autocorrelation minimization problem.

For the aperiodic autocorrelation (which is what conv(a,a) computes for real sequences), the minimum possible maximum is related to the "Borwein-Steigerwald" sequences and related constructions.

Actually, I recall now that this is closely related to the problem of finding sequences with minimum aperiodic autocorrelation. For continuous (real-valued) sequences, the optimal is related to "chirp" sequences or specific polynomial roots.

But there's a simpler approach: the problem can be formulated as:
Minimize M subject to:
- sum(a) = 1
- a_i >= 0
- For all k = 0, 1, ..., 2n-2: sum_j a[j]*a[k-j] <= M

This is a non-convex problem (bilinear constraints). However, the alternating approach (fix a, optimize g; fix g, optimize a) can work well.

The LP approach from the previous code is:
- Fix a, find g >= 0, sum(g) = 1 that minimizes max_k (a*g)[k]
- Update a = (1-t)*a + t*g

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build the convolution matrix A where A[k,j] = a[k-j] if valid."""

    A = np.zeros((2 * n - 1, n))

    return A

            if st > 1e-10:
                a_try = a_try / st
                candidates.append(a_try)

        # Evaluate all candidates
        best_candidate = None
        best_candidate_obj = best_obj
        for cand in candidates:
            obj_cand = compute_obj(cand)
            if obj_cand < best_candidate_obj:
                best_candidate_obj = obj_cand
                best_candidate = cand

        if best_candidate is not None:
            best_obj = best_candidate_obj
            best_a = best_candidate.copy()
            a = best_candidate.copy()
            stall_count = 0
            if verbose and it % 10 == 0:
                print(f"    LP-FP it{it}: obj={best_obj:.10f}, t_opt={t_opt:.10f}, "
                      f"t_opt2={t_opt2:.10f}, |active|={len(active) if active is not None else 0}")
        else:
            stall_count += 1
            if stall_count >= stall_threshold:
                # Perturb to escape
                pert = rng.normal(0, 0.01, n)
                a = a + pert
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0
                if stall_count > 5 * stall_threshold:
                    break

    return best_a


def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def gd_polish(a, time_budget, verbose=False, tau_levels=None, beta=0.9):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [10, 50, 200, 800, 2000, 5000]

    for tau in tau_levels:
        lr = 3.0 / np.sqrt(n)
        v = np.zeros(n)
        it_level = 0
        max_it = 3000

        while time.monotonic() - start < time_budget and it_level < max_it:
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-14:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
            it_level += 1
            lr = max(1e-8, lr * 0.99995)

        if verbose:
            print(f"    GD tau={tau:5d}: obj={best_obj:.10f}, iters={it_level}")

    return best_a


def make_exponential_init(n, rate, rng=None):
    """Monotonically increasing exponential."""
    x = np.arange(n)
    a = np.exp(rate * x / max(1, n - 1))
    return a / a.sum()


def make_polynomial_init(n, power, rng=None):
    """Monotonically increasing polynomial."""
    x = np.linspace(0, 1, n)
    a = x ** power + 0.01
    return a / a.sum()


def make_chirp_init(n, rng=None):
    """Real chirp-like initialization (quadratic phase made real)."""
    x = np.linspace(0, 1, n)
    # Use a smooth increasing function with some oscillation
    a = np.exp(0.5 * x) * (1.0 + 0.1 * np.sin(4 * np.pi * x))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_bimodal_init(n, rng=None):
    """Bimodal: high at both ends but asymmetric."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 1, n)
    c1 = 0.15
    c2 = 0.85
    w = 0.2
    a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + 1.5 * np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    a = np.maximum(a, 0.001)
    return a / a.sum()


def make_geometric_init(n, ratio, rng=None):
    """Geometric sequence."""
    a = ratio ** np.arange(n)
    return a / a.sum()


def make_staircase_init(n, steps=5, rng=None):
    """Staircase: piecewise constant, increasing."""
    a = np.zeros(n)
    for s in range(steps):
        start = int(s * n / steps)
        end = int((s + 1) * n / steps)
        a[start:end] = (s + 1) ** 1.5
    return a / a.sum()


def active_set_optimize(a, time_budget, verbose=False, seed=42):
    """
    Targeted optimization using active set information.
    Identifies which lags are at the maximum and tries to equalize them.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(100):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Find active set: lags where conv is within 1% of max
        threshold = max_val * 0.99
        active = np.where(conv >= threshold)[0]

        # LP: minimize t subject to (a*g)[k] <= t for all k,
        # with equality on active set (to force equalization)
        A = build_conv_matrix(a)
        g_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()

        constraints = [A @ g_var <= t_var, cp.sum(g_var) == 1]
        # Add equality constraints for active set (soft)
        for k in active[:20]:  # Limit to avoid overly constrained LP
            constraints.append(cp.Abs((A @ g_var)[k] - t_var) <= 0.05 * t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    sg = float(np.sum(g_val))
                    if sg > 1e-10:
                        g_val = g_val / sg
                        solved = True
                        break
            except Exception:
                continue

        if not solved:
            break

        # Try blending
        best_t_obj = best_obj
        best_t_a = best_a.copy()
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - alpha) * a + alpha * g_val
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()

        if best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_t_a.copy()
            if verbose and it % 5 == 0:
                print(f"    AS iter{it}: obj={best_obj:.10f}, |active|={len(active)}")
        else:
            # Slight perturbation
            a = a + rng.normal(0, 0.005, n)
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def try_multiple_solvers_lp(a):
    """Try LP with multiple solver configurations for robustness."""
    n = len(a)
    A = build_conv_matrix(a)
    g_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [A @ g_var <= t_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solver_configs = [
        (cp.MOSEK, {}),
        (cp.MOSEK, {'MSK_IPARAM_MAX_NUM_ITERS': 50000}),
        (cp.CLARABEL, {}),
        (cp.PDLP, {}),
        (cp.SCS, {'max_iters': 100000, 'eps': 1e-9}),
    ]
    for solver, kwargs in solver_configs:
        try:
            prob.solve(solver=solver, verbose=False, **kwargs)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(t_var.value)
        except Exception:
            continue
    return None, None


def fine_tune(a, time_budget, verbose=False):
    """
    Fine-tuning: very small perturbations around the current best.
    Uses coordinate descent and small random steps.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    # Coordinate descent: adjust pairs of coordinates
    step_size = 0.01
    improving = True
    while improving and time.monotonic() - start < time_budget * 0.5:
        improving = False
        for _ in range(3):  # Multiple passes
            if time.monotonic() - start > time_budget * 0.5:
                break
            idx = rng.integers(0, n)
            other_idx = rng.integers(0, n)
            if idx == other_idx:
                continue

            original_i = a[idx].copy()
            original_j = a[other_idx].copy()

            # Try moving mass from one to the other
            for delta in [step_size, -step_size, step_size * 0.5, -step_size * 0.5]:
                a[idx] = original_i + delta
                a[other_idx] = original_j - delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s < 1e-10:
                    a[idx] = original_i
                    a[other_idx] = original_j
                    continue
                a = a / s
                obj = compute_obj(a)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    improving = True
                    step_size *= 0.7
                else:
                    a[idx] = original_i
                    a[other_idx] = original_j
                    a = a / a.sum()

    # Small random perturbations
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 2:
        for _ in range(int(remaining * 50)):
            if time.monotonic() - start > time_budget - 1:
                break
            a_try = best_a + rng.normal(0, 0.002, n)
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP-based fixed-point iteration with active set targeting,
    exponential/polynomial initializations, multi-n exploration, and fine polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        parent = None

    if best_seq is None:
        n0 = 800
        best_seq = make_exponential_init(n0, 1.0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.10f}")

    # === Phase 1: Diverse initializations at current n ===
    print("\n=== Phase 1: Diverse	Init Screening ===")
    n_current = len(best_seq)

    init_candidates = []
    init_candidates.append(("parent", best_seq.copy()))
    init_candidates.append(("exp1", make_exponential_init(n_current, 1.0)))
    init_candidates.append(("exp2", make_exponential_init(n_current, 2.0)))
    init_candidates.append(("exp3", make_exponential_init(n_current, 3.0)))
    init_candidates.append(("poly1", make_polynomial_init(n_current, 1.0)))
    init_candidates.append(("poly2", make_polynomial_init(n_current, 2.0)))
    init_candidates.append(("poly3", make_polynomial_init(n_current, 3.0)))
    init_candidates.append(("chirp", make_chirp_init(n_current)))
    init_candidates.append(("bimodal", make_bimodal_init(n_current, rng)))
    init_candidates.append(("geo1.1", make_geometric_init(n_current, 1.1)))
    init_candidates.append(("geo1.05", make_geometric_init(n_current, 1.05)))
    init_candidates.append(("stair", make_staircase_init(n_current, 8)))
    init_candidates.append(("stair5", make_staircase_init(n_current, 5)))
    init_candidates.append(("stair12", make_staircase_init(n_current, 12)))

    # Quick LP pass on each (very short)
    for name, a_init in init_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        obj_init = compute_obj(a_init)
        # Quick 5-iteration LP
        a_quick = a_init.copy()
        for _ in range(5):
            g_opt, t_opt, _ = lp_solve_given_a(a_quick)
            if g_opt is None:
                break
            a_quick = 0.7 * a_quick + 0.3 * g_opt
            a_quick = np.maximum(a_quick, 0)
            a_quick = a_quick / a_quick.sum()
        obj_quick = compute_obj(a_quick)
        if obj_quick < best_score:
            best_score = obj_quick
            best_seq = a_quick.copy()
            print(f"  {name}: obj={best_score:.10f} **")
        else:
            print(f"  {name}: init={obj_init:.6f}, after_quick_LP={obj_quick:.6f}")

    print(f"  After Phase 1: best={best_score:.10f}")

    # === Phase 2: Aggressive LP fixed-point (main workhorse) ===
    print("\n=== Phase 2: LP Fixed-Point Iteration ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.3, 400.0)
    if p2_budget > 30:
        a2 = lp_fixed_point(best_seq, time_budget=p2_budget, verbose=True,
                           seed=seed, max_iters=300, alpha_start=0.5, alpha_end=0.05)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 LP-FP: obj={best_score:.10f} **")
        else:
            print(f"  Phase2 LP-FP: obj={obj2:.10f}")

    # === Phase 3: Active set optimization ===
    print("\n=== Phase 3: Active Set Optimization ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 200.0)
    if p3_budget > 20:
        a3 = active_set_optimize(best_seq, time_budget=p3_budget, verbose=True, seed=seed + 1)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 AS: obj={best_score:.10f} **")

            print(f"  Phase3 AS: obj={obj3:.10f}")

    # === Phase 4: Multi-n exploration ===
    print("\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(200, n_current - 200), max(200, n_current - 100),
        n_current,
        n_current + 100, n_current + 200, n_current + 400,
        400, 600, 800, 1000, 1200, 1500, 2000, 2500
    ]))
    n_options = [n for n in n_options if 200 <= n <= 3000]

    # Sort by distance from current (closest first)
    n_options.sort(key=lambda x: abs(x - n_current))
    n_options = n_options[:8]  # Limit to 8 options

    print(f"  n candidates: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample(best_seq, n_try)

        # LP optimization
        alloc = min(remaining * 0.2, 40.0)
        if alloc > 10:
            a_try = lp_fixed_point(a_try, time_budget=alloc, verbose=False,
                                   seed=n_try, max_iters=100)

        # Quick GD
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.1, 20.0)
            a_try = gd_polish(a_try, time_budget=alloc, tau_levels=[50, 200, 800])

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    print(f"  After Phase 4: best={best_score:.10f}, n={len(best_seq)}")

    # === Phase 5: Second round of LP + AS on best ===
    print("\n=== Phase 5: Second Round Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.25, 150.0)
        a5 = lp_fixed_point(best_seq, time_budget=alloc, verbose=True,
                           seed=seed + 100, max_iters=200, alpha_start=0.3, alpha_end=0.05)
        obj5 = compute_obj(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Phase5 LP: obj={best_score:.10f} **")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.15, 80.0)
            a5b = active_set_optimize(best_seq, time_budget=alloc, verbose=True, seed=seed + 200)
            obj5b = compute_obj(a5b)
            if obj5b < best_score:
                best_score = obj5b
                best_seq = a5b.copy()
                print(f"  Phase5 AS: obj={best_score:.10f} **")

    # === Phase 6: GD polishing with high tau ===
    print("\n=== Phase 6: GD Polishing ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 80.0)
        a6 = gd_polish(best_seq, time_budget=alloc, verbose=True,
                      tau_levels=[100, 500, 2000, 5000, 10000])
        obj6 = compute_obj(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Phase6 GD: obj={best_score:.10f} **")
        else:
            print(f"  Phase6 GD: obj={obj6:.10f}")

    # === Phase 7: Fine-tuning ===
    print("\n=== Phase 7: Fine Tuning ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 120.0)
        a7 = fine_tune(best_seq, time_budget=alloc, verbose=True)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Phase7 fine: obj={best_score:.10f} **")
        else:
            print(f"  Phase7 fine: obj={obj7:.10f}")

    # === Phase 8: Final LP sweep ===
    print("\n=== Phase 8: Final LP Sweep ===")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 60.0)
        a8 = lp_fixed_point(best_seq, time_budget=alloc, verbose=False,
                           seed=seed + 999, max_iters=100, alpha_start=0.2, alpha_end=0.05)
        obj8 = compute_obj(a8)
        if obj8 < best_score:
            best_score = obj8
            best_seq = a8.copy()
            print(f"  Phase8 final LP: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
