def compute_conv(a):

    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = float(np.sum(w))
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

            lr = lr_max
        
        # Update momentum and position
        v = beta * v - lr * obj_grad / (2.0 * n)
        a_new = y + v
        a_new = project_simplex(a_new)
        
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        it += 1
        
        # Stall detection: if no improvement for many iterations, reset momentum
        if stall_count > 50:
            v = np.zeros(n)
            lr = lr * 0.5
            stall_count = 0
        
        # Periodic exact evaluation and CD micro-step
        if it % 200 == 0:
            a_exact = np.maximum(a, 0)
            s_e = float(np.sum(a_exact))
            if s_e > 1e-10:
                a_exact = a_exact / s_e
                obj_exact = 2.0 * n * float(np.max(compute_conv(a_exact)))
                if obj_exact < best_obj:
                    best_obj = obj_exact
                    best_a = a_exact.copy()
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    NGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.1f}s")
    
    return best_a


def compute_conv_incremental(a, i_new, i_old, x_new, x_old):
    """
    Update convolution when changing a[i] from x_old to x_new.
    Returns the delta for each affected position.
    """
    n = len(a)
    delta = x_new - x_old
    affected = []
    for j in range(n):
        k = i_new + j
        if 0 <= k <= 2 * n - 2:
            affected.append((k, delta * a[j]))
        k2 = i_old + j
        if 0 <= k2 <= 2 * n - 2 and k2 != k:
            # Only if j != i_new (to avoid double counting the self-term)
            pass
    return affected


def coordinate_descent_v2(a, time_budget=300.0, verbose=False, seed=42):
    """
    Improved coordinate descent with block-2 analytical solutions.
    For pairs (i,j) where the cross-term position i+j is in the active set,
    uses the analytical solution x=y=c'/(S').
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            
            v_old = a[i]
            S_other = S - v_old
            if S_other < 1e-12:
                continue
            
            # Determine affected positions
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()
            
            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k_lin_lin = c_k[mask]
            c_2i = float(conv[2 * i])
            
            def f_1d(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v_old)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v_old * v_old
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) ** 2
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom
            
            # Quick scan
            candidates = [0.0, v_old * 0.5, v_old, v_old * 1.5, v_old * 2.0,
                          S / n, S_other * 0.5, S_other]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))
            
            best_x = v_old
            best_f = f_1d(v_old)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x
            
            # Golden section refinement
            scale = max(v_old, S / n, 1e-8)
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8
            
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)
            
            for _ in range(40):
                if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)
            
            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            
            if abs(x_opt - v_old) < 1e-14:
                continue
            
            # Update a and conv incrementally
            delta = x_opt - v_old
            a[i] = x_opt
            
            # Update conv at affected positions
            for j in range(n):
                k = i + j
                if 0 <= k <= 2 * n - 2:
                    conv[k] += 2.0 * delta * a[j] if j != i else 0.0
            # Fix the self-term
            conv[2 * i] += x_opt * x_opt - v_old * v_old
            
            S += delta
            
            # Track best
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    if n > 3000:
        # Too large for LP, use a simpler approach
        return None, None
    
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None


def lp_polish(a, time_budget=60.0, verbose=False, seed=42):
    """LP-based polishing with line search."""
    n = len(a)
    if n > 3000:
        # Fallback: just do coordinate descent
        return coordinate_descent_v2(a, time_budget=time_budget, verbose=verbose, seed=seed)
    
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search over blend factors
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in np.concatenate([np.linspace(0.05, 0.5, 10), np.linspace(0.5, 1.0, 10)]):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 2:
                break
    
    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 3:
        # Prolate-like: Bessel function approximation
        x = np.linspace(-1, 1, n)
        a = np.sinc(2 * x)  # sin(2*pi*x)/(2*pi*x)
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 4:
        # Gaussian
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2)
        a = a / a.sum()
    elif mode == 5:
        # Flat top with tapered edges
        a = np.ones(n)
        taper = max(1, n // 10)
        t = np.linspace(0, 1, taper)
        a[:taper] *= 0.5 * (1 - np.cos(np.pi * t))
        a[-taper:] *= 0.5 * (1 - np.cos(np.pi * t[::-1]))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
