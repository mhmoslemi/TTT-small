Note that sum_b = sum_k (a*a)[k] = (sum_j a[j])^2 = (sum_a)^2.

So max_b >= sum_b/(2n-1) = (sum_a)^2/(2n-1).

Therefore objective >= 2n/(2n-1).

from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def subgrad_max_conv(a):
    """Compute subgradient of max_k (a*a)[k] at a."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    # Find all maximizing indices (with tolerance)
    tol = 1e-10 * max_val
    I = np.where(conv >= max_val - tol)[0]
    if len(I) == 0:
        I = np.array([np.argmax(conv)])
    # Subgradient = (1/|I|) * sum_{k in I} grad f_k
    # grad f_k at a: [grad f_k]_i = sum_j delta
    # (a*a)[k] = sum_j a[j]*a[k-j]
    # d/d a[i] = a[k-i] + a[k-i] (if valid) = 2*a[k-i] (approximately)
    # More precisely using FFT:
    p = np.zeros(2*n-1)
    for k in I:
        p[k] = 1.0 / len(I)
    N = 1
    while N < 2 * n + 2*n-1:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, max_val


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_solve(a):
    """Solve min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None

def subgradient_polish(a, time_budget, verbose=False):
    """Project subgradient method for max(conv(a,a)) with simplex constraint."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)

    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lb = 1.0 / (2 * n - 1)

    alpha_decay = 0.9999
    alpha = 0.5 / np.sqrt(n)
    
    while time.monotonic() - start < time_budget:
        g, max_val = subgrad_max_conv(a)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        
        # Polyak-like step size
        gap = max_val - lb
        if gap > 1e-15:
            alpha_polyak = gap / (g_norm * g_norm)

        alpha_use = min(alpha_polyak, alpha)

def staircase_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9, lr_scale=2.0):

    for tau in tau_levels:

            y = np.maximum(y, 0)
            sy = float(np.sum(y))

                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)

            v = beta * v - lr * g

            a = a_new

def make_init(n, mode, rng=None):
    if rng is None:

    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:  # Triangle (Fejer-like)

            a = np.zeros(n)
            a[:half+1] = np.arange(half+1, 0, -1)
            a[half+1:] = np.arange(half-1, -1, -1)  # Wait, this doesn't work for even
            # Let me redo: triangular of length n

            # Actually: a[k] = min(k+1, n-k) for k=0,...,n-1

        else:
            a = np.minimum(np.arange(n) + 1, n - np.arange(n))
        return a / a.sum()
    elif mode == 2:  # Semicircle
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 1e-8))
        return a / a.sum()
    elif mode == 3:  # Gaussian
        x = np.arange(n)
        a = np.exp(-((x - n/2)**2) / (2*(n/4)**2))
        return a / a.sum()
    elif mode == 4:  # Random
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 5:  # Cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        return a / a.sum()
    elif mode == 6:  # Fejer kernel (raised cosine)
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif mode == 7:  # Optimal flat-top (Dolphin-Chebyshev like)
        # Approximate with a sum of cosines
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for k in range(1, 6):
            a += 0.3 / k * np.cos(2 * np.pi * k * (x - 0.5))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 8:  # Two-lobe
        x = np.arange(n)
        a = np.exp(-((x - n*0.3)**2) / (2*(n*0.1)**2)) + np.exp(-((x - n*0.7)**2) / (2*(n*0.1)**2))

    elif mode == 9:  # Power law
        x = np.arange(n, dtype=float)
        a = (1.0 + x / n) ** (-2.0)
        a += (1.0 + (n-1-x) / n) ** (-2.0)

    else:
        return np.ones(n) / n


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s
