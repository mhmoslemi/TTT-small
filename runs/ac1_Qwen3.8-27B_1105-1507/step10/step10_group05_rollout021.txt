import math

    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""

        return np.inf

        return np.inf
    a_norm = a / s

def project_simplex(v):

    """
    Compute gradient of smoothed max of conv(a,a) and the convolution itself.
    Uses the log-sum-exp smoothing: f(a) = (1/tau) * log(sum_k exp(tau * b[k]))

    conv = compute_conv(a)

    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    if base_lr is None:
        base_lr = 5.0 / np.sqrt(n)
    
    beta = 0.85  # momentum
    lr = base_lr
    v = np.zeros(n)
    it = 0
    best_it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Cosine annealing for tau
        cos_val = 0.5 * (1 + np.cos(np.pi * progress))
        tau = tau_start + (tau_end - tau_start) * (1 - cos_val)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g, conv, p = compute_gradient_and_conv(y, tau)
        
        # Gradient step with adaptive learning rate
        # Use the norm of the gradient to scale
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-15:
            g = g / g_norm  # normalize gradient direction
        
        a_new = y - lr * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        # Check if we improved
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            no_improve = 0
        else:
            no_improve += 1
            # If no improvement, try smaller step
            a_new = y - lr * 0.5 * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    best_it = it
                    no_improve = 0
                    lr *= 0.7
                else:
                    lr *= 0.8
        
        v = beta * v - lr * g
        a = a_new
        it += 1
        
        # Decay learning rate
        lr = max(1e-10, lr * 0.9995)
        
        # Restart if stuck
        if no_improve > 3000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            no_improve = 0
            # Add small perturbation
            a = a * (1 + 0.01 * rng.normal(0, 1, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_it = it
        
        if verbose and it % 5000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")
    
    return best_a


def multiplicative_update(a, n_iter=1000, verbose=False):
    """
    Multiplicative update method for flattening the convolution.
    At each step, reduce a[j] where it contributes most to the peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(n_iter):
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Find active set (where conv is close to max)
        threshold = max_conv - 0.01 * max_conv
        active = conv >= threshold
        
        # Compute "pressure" on each a[j]
        # pressure[j] = sum_k active[k] * (a[k-j] + a[j-k]) / (2n-1)
        # Use FFT for efficiency
        N = 1
        while N < 3 * n:
            N <<= 1
        
        P = fft(active.astype(float), N)
        A = fft(a, N)
        
        term1 = ifft(P * np.conj(A)).real[:n]
        term2 = ifft(P * A).real[:n]
        pressure = term1 + term2
        
        # Multiplicative update: reduce a where pressure is high
        # a_new[j] = a[j] / (1 + epsilon * pressure[j] / max_pressure)
        max_pressure = float(np.max(pressure)) if np.max(pressure) > 0 else 1.0
        epsilon = 0.1
        a_new = a / (1.0 + epsilon * pressure / max_pressure)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        
        if verbose and it % 100 == 0:
            print(f"    MU it{it}: obj={best_obj:.8f}")
    
    return best_a


def fast_lp_iteration(a, n_iter=500, verbose=False):
    """
    Fast LP-like iteration using the dual and multiplicative weights.
    Solves the dual: max sum(y) s.t. A^T y <= 1, y >= 0
    using multiplicative weight updates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Dual variables y (length 2n-1)
    y = np.ones(2 * n - 1) / (2 * n - 1)
    
    # Precompute FFT sizes
    N = 1
    while N < 3 * n:
        N <<= 1
    
    for it in range(n_iter):
        # Compute A^T y: for each j, c[j] = sum_k y[k] * (a[k-j] + a[j-k])
        # This is the same as the gradient computation but with y instead of softmax
        P = fft(y, N)
        A = fft(a, N)
        
        term1 = ifft(P * np.conj(A)).real[:n]
        term2 = ifft(P * A).real[:n]
        c = term1 + term2  # c[j] = sum_k y[k] * (a[k-j] + a[j-k])
        
        # Multiplicative weight update for dual
        # Increase y[k] where the corresponding "constraint" is tight
        # The constraint is c[j] <= 1 for all j
        # If c[j] > 1, we need to reduce y at positions that contribute to c[j]
        
        # Alternative: use the primal-dual relationship
        # At optimum, a[j] > 0 implies c[j] = 1 (complementary slackness)
        # So: if c[j] < 1, increase a[j]; if c[j] > 1, decrease a[j]
        
        # Primal update: a_new[j] = a[j] * (1 + eta * (1 - c[j]))
        eta = 0.05
        a_new = a * (1.0 + eta * np.maximum(0, 1.0 - c))
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        
        # Dual update: increase y where conv(a,y) is large
        conv_ay = compute_conv_a_b(a, y)  # need this function
        # Actually, let's skip the dual update and just do primal
        # The primal update above should be sufficient
        
        if verbose and it % 100 == 0:
            print(f"    FLIP it{it}: obj={best_obj:.8f}")
    
    return best_a


def compute_conv_a_b(a, b):
    """Convolution of a and b (not necessarily the same) via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def lp_gcq_iteration(a, time_budget=100.0, verbose=False, seed=42):
    """
    LP iteration using a faster approach: instead of solving the full LP,
    use a greedily-constrained quadratic (GCQ) or active-set heuristic.
    
    At each step:
    1. Find the peak of conv(a,a)
    2. Solve a restricted problem: minimize max_conv subject to 
       the active constraints (where conv is near max)
    3. Use projected gradient on the restricted problem
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    it = 0
    best_it = 0
    no_improve = 0
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    beta = 0.8
    tau = 100.0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 2:
            break
        
        # Adaptive tau
        tau = 100.0 + 500.0 * (elapsed / max(time_budget, 1.0))
        
        g, conv, p = compute_gradient_and_conv(a, tau)
        
        # Nesterov
        y = a + beta * v
        y = np.maximum(y, 0)
        y = y / max(np.sum(y), 1e-10)
        
        g_y, _, _ = compute_gradient_and_conv(y, tau)
        
        v = beta * v - lr * g_y
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9997)
        
        if no_improve > 2000:
            # Perturbation restart
            strength = min(0.1, 0.01 * (no_improve - 2000))
            a = best_a * (1 + strength * rng.normal(0, 1, n))
            a = np.maximum(a, 0)
            a = a / max(a.sum(), 1e-10)
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n)
            no_improve = 0
            best_it = it
        
        if verbose and it % 5000 == 0:
            print(f"    LP-GCQ it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.0f}s")
    
    return best_a


def init_central_binomial(n, c=0.5):
    """Initialize with a[j] proportional to 1/sqrt(j + c)."""
    j = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(j + c)
    a = a / a.sum()
    return a


def init_exponential(n, rate=1.0):
    """Initialize with a[j] proportional to exp(-rate * j / n)."""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * j / n)
    a = a / a.sum()
    return a


def init_cosine(n, order=1):
    """Initialize with a[j] proportional to cos(pi * order * j / (2n))^2."""
    j = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * order * j / (2 * n)) ** 2
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a


def init_polynomial(n, power=0.5):
    """Initialize with a[j] proportional to (1 - j/n)^power."""
    j = np.arange(n, dtype=np.float64)
    a = (1.0 - j / n) ** power
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a
