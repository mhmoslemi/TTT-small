**Key mathematical insight**: The problem `min max(conv(a,a))` subject to `sum(a) = 1, a >= 0` is equivalent to finding a probability distribution on `{0, ..., n-1}` that minimizes the maximum probability of the sum of two independent draws. This is related to the "cocktail party problem" or "sumset" problems.

For continuous distributions, the optimal is the arcsine distribution (on a circle) which gives `max = 1/(2*pi)` approximately. For discrete, we're looking for a distribution that makes the convolution as flat as possible.

**Approach: Active Set Method**

At the optimum, there's a set of "active" convolution entries where the maximum is achieved. If we can identify this set, we can solve a quadratic program:
- Minimize `M`
- Subject to: `conv(a,a)[k] = M` for all active `k`
- `conv(a,a)[k] <= M` for all other `k`
- `sum(a) = 1`
- `a >= 0`

This is a non-convex QP in general, but we can use it as a guiding principle.

**Alternative: Block Coordinate Descent**

Instead of optimizing one coordinate at a time, optimize blocks of coordinates together. For a block of size `b`, we can solve a small QP or use a local search.

**Another idea: Frequency domain approach**

The convolution `a*a` in the time domain corresponds to `|A(f)|^2` in the frequency domain (where `A` is the DFT). So minimizing the max of `a*a` is related to making `|A(f)|^2` have low peaks in the time domain, which is related to the time-domain characteristics of `|A(f)|^2`.

Actually, let me think more carefully. The linear convolution `conv(a,a)` has length `2n-1`. Its DFT is `|A(f)|^2` (circular convolution interpretation). But linear convolution is different from circular.

For the linear convolution, if we zero-pad `a` to length `2n-1`, then the circular convolution of the zero-padded sequences equals the linear convolution. So `FFT(zeropad(a)) * FFT(zeropad(a))` gives us the DFT of the linear convolution.

So `max(conv(a,a))` is the maximum of the inverse DFT of `|FFT(zeropad(a))|^2`.

This means we want `|FFT(zeropad(a))|^2` to have a "flat" inverse DFT. In other words, we want the power spectrum to be as "smooth" as possible in a certain sense.

"""
Optimizes step function coefficients using enhanced active-set coordinate descent,

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a) = 1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_rows(a, rows):
    """Build rows of conv matrix for specific output indices."""
    n = len(a)
    A = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        # conv(a,g)[k] = sum_j a[j] * g[k-j] where 0<=j<n and 0<=k-j<n
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[idx, j_min:j_max+1] = a[j_max - j_min::-1][:j_max - j_min + 1]
    return A


def solve_lp_minpeak(a, active_frac=0.3, time_budget=30.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Only enforce constraints on 'active' entries (near max) for speed.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_ag = compute_conv(a_norm)  # This is a*a, not a*g, but we need the matrix
    # Actually we need the matrix A such that A @ g = conv(a_norm, g)
    
    # For the LP, we need conv(a_norm, g)[k] <= M for all k
    # conv(a_norm, g)[k] = sum_j a_norm[j] * g[k-j]
    # This is a linear function of g.
    
    # Build the full matrix (expensive for large n)
    # A[k, j] = a_norm[j] if 0 <= k-j < n, else 0
    # i.e., A[k, j] = a_norm[j] if max(0, k-n+1) <= j <= min(n-1, k)
    
    # For large n, building the full (2n-1) x n matrix is O(n^2) memory
    # Instead, let's just use all constraints for n <= 1000
    # Or use a subset
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # Use all constraints (for n up to ~1200 this should be fine)
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    start = time.monotonic()
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None


def coordinate_descent_enhanced(a, time_budget=300.0, verbose=False, seed=42):
    """
    Enhanced coordinate descent with:
    - Exact 1D subproblem solving via golden section
    - Active set awareness for better bracketing
    - Periodic 2D block updates
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1

        # Recompute conv every few sweeps to avoid drift
        if sweep % 5 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v

            if S_other < 1e-12:
                continue

            # Determine valid k range for conv entries affected by a[i]
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)

            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Define 1D objective
            def f_1d(x, v=v, S_other=S_other, n=n, c_k_lin=c_k_lin,
                     a_ki_lin=a_ki_lin, c_2i=c_2i):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Quick scan
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.5, v * 0.75, v * 1.25, v * 1.5, v * 2.0,
                          S / n, S / (2 * n)]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(40):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            # Update
            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

            # Track best
            S_new = S
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def multiplicative_descent(a, time_budget=100.0, tau_init=30.0, tau_end=300.0, verbose=False):
    """
    Multiplicative gradient descent: a[i] <- a[i] * exp(-lr * grad[i]).
    Naturally maintains non-negativity.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    lr = 5.0 / np.sqrt(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress
        # Decrease learning rate over time
        lr_eff = lr * (1.0 - 0.5 * progress)

        # Compute smoothed gradient
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        # Exponential surrogate for max
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient of sum_k p[k] * conv(a,a)[k] with respect to a
        # conv(a,a)[k] = sum_j a[j] * a[k-j]
        # d/d a[i] conv(a,a)[k] = a[k-i] + a[i-k] (if in range)
        # grad[i] = sum_k p[k] * (a[k-i] + a[i-k])
        #         = 2 * sum_k p[k] * a[k-i] (by symmetry of conv)
        #         = 2 * (p * a)[i] in circular sense, but we need linear conv
        # Actually: grad[i] = sum_k p[k] * (a[k-i] + a[i-k] if valid)
        # Let's compute via FFT for efficiency
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        # grad[i] = 2 * sum_k p[k] * a[(k-i) mod N] but we need linear
        # For linear conv: grad[i] = 2 * sum_k p[k] * a[k-i] where 0<=k-i<n
        # This is (p * a)[i] in linear convolution sense, times 2
        # Actually let's just do it directly for correctness
        # grad[i] = sum_{k} p[k] * (a[k-i] if 0<=k-i<n else 0 + a[i-k] if 0<=i-k<n else 0)
        # = sum_{j=0}^{n-1} a[j] * (p[i+j] + p[i-j])  ... hmm this is getting complicated
        # Let's use the FFT approach which should be correct for the circular version
        # and approximately correct for linear when n is large
        grad = 2.0 * g_full[:n]

        # Apply multiplicative update
        a_new = a * np.exp(-lr_eff * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 2000 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}")

    return best_a

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 8, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def block_coordinate_descent(a, time_budget=100.0, block_size=4, verbose=False, seed=42):
    """
    Block coordinate descent: optimize groups of coordinates together.
    For each block, solve a small QP via scipy.optimize.minimize.
    """
    from scipy.optimize import minimize as sp_minimize
    
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0

    while time.monotonic() - start < time_budget:
        # Pick a random block
        i0 = rng.integers(0, n - block_size)
        block = np.arange(i0, i0 + block_size)
        
        S_other = float(np.sum(a) - np.sum(a[block]))
        
        def obj_block(x_block):
            a_tmp = a.copy()
            a_tmp[block] = x_block
            x_b = np.maximum(x_block, 0)
            a_tmp[block] = x_b
            s_tmp = S_other + float(np.sum(x_b))
            if s_tmp < 1e-10:
                return 1e10
            conv_tmp = compute_conv(a_tmp / s_tmp)
            return 2.0 * n * float(np.max(conv_tmp))
        
        x0 = a[block].copy()
        
        try:
            res = sp_minimize(obj_block, x0, method='Nelder-Mead',
                            options={'maxiter': 200, 'xatol': 1e-10, 'fatol': 1e-12})
            if res.fun < obj_block(x0) - 1e-12:
                a[block] = np.maximum(res.x, 0)
                conv = compute_conv(a / a.sum())
                obj_now = 2.0 * n * float(np.max(conv))
                if obj_now < best_obj:
                    best_obj = obj_now
                    best_a = a.copy()
        except Exception:
            pass
        
        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    BCD it{it}: best_obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a

"""
Optimizes step function coefficients using a hybrid strategy combining:
1. Exact coordinate descent with gradient-informed bracketing
2. LP-based direction finding with active-set refinement
3. Multiplicative weight updates for natural non-negativity
4. Simulated annealing for escaping local minima
5. Multi-n exploration with cross-length transfer
Key innovations: gradient-aware CD bracketing, LP with proximity penalty,
and annealed restarts that preserve structure.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_budget=25.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_with_proximity(a, lambda_reg=0.1, time_budget=25.0):
    """
    Solve LP with proximity penalty: min M + lambda*||g-a||_1
    s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    objective = cp.Minimize(M + lambda_reg * cp.norm1(g - a_norm))
    prob = cp.Problem(objective, constraints)

    start = time.monotonic()
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS]
    for solver in solvers:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def compute_smoothed_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) via log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n], conv, p

            break

        # Apply multiplicative update with clipping
        update = np.exp(-lr * grad)
        update = np.clip(update, 0.01, 100.0)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 1000 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}")

    return best_a


def annealed_search(a, time_budget=100.0, T_init=0.01, T_end=0.001, seed=42):
    """
    Simulated annealing with structured perturbations.
    Perturbations are drawn from a mixture of:
    - Small Gaussian noise
    - Coordinate swaps
    - Block rescaling
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    cur_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    accept_count = 0

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / max(time_budget, 1e-10))
        T = T_init * (T_end / T_init) ** progress

        # Choose perturbation type
        r = rng.random()
        a_try = a.copy()

        if r < 0.5:
            # Gaussian noise
            scale = 0.02 * (1.0 - 0.5 * progress)
            a_try = a * np.exp(rng.normal(0, scale, n))
        elif r < 0.75:
            # Block rescaling
            bs = rng.integers(2, max(3, n // 20))
            i0 = rng.integers(0, n - bs)
            a_try[i0:i0+bs] *= np.exp(rng.normal(0, 0.05))
        else:
            # Coordinate swap-like: scale up one coord, scale down another
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                factor = np.exp(rng.normal(0, 0.1))
                a_try[i] *= factor
                a_try[j] /= factor

        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        obj_try = compute_obj_normalized(a_try)
        delta = obj_try - cur_obj

        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_try
            cur_obj = obj_try
            accept_count += 1
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
        else:
            # Occasionally accept large perturbations to escape
            if rng.random() < 0.01:
                a = a_try
                cur_obj = obj_try

        it += 1

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 10, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-4, 4)
        a = a / a.sum()
    elif mode == 6:
        # Triangular with random peaks
        a = 1.0 - np.abs(2 * np.linspace(0, 1, n) - 1)
        a += 0.2 * rng.uniform(0, 1, n)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=30.0, tau_end=400.0, verbose=False):
    """Nesterov-accelerated smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress
        lr_eff = lr * (1.0 - 0.5 * progress)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        try:
            grad, _, _ = compute_smoothed_gradient(y, tau)
        except Exception:
            break

        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            lr_eff = min(max(lr_eff * 0.995 + 0.005 / (g_norm + 1e-10),
                            0.05 * lr), lr)

        v = beta * v - lr_eff * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 2000 == 0:
            print(f"    NSGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining CD, LP, MD, Nesterov,
    simulated annealing, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 12.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj_normalized(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    # === Phase 1: Coordinate Descent (main workhorse) ===
    print(f"\n=== Phase 1: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.3, 400.0)
    if p1_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget,
                                  verbose=True, seed=seed)
        obj_cd = compute_obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: LP Polishing (multiple rounds) ===
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 220.0)
    if p2_budget > 15:
        for rnd in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(p2_budget / (6 - rnd), remaining * 0.5)
            if alloc < 10:
                break
            a_lp = lp_polish(best_seq, time_budget=alloc, verbose=False,
                             seed=3000 + rnd * 13)
            obj_lp = compute_obj_normalized(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")

    # === Phase 3: CD after LP ===
    print(f"\n=== Phase 3: Post-LP CD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 120.0)
    if p3_budget > 15:
        a_cd2 = coordinate_descent(best_seq, time_budget=p3_budget,
                                   verbose=True, seed=seed + 777)
        obj_cd2 = compute_obj_normalized(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  CD2: obj={best_score:.8f} **")
        else:
            print(f"  CD2: obj={obj_cd2:.8f}")

    # === Phase 4: Nesterov Smoothed Descent ===
    print(f"\n=== Phase 4: Nesterov Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 80.0)
        a_ns = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                         tau_init=30.0, tau_end=400.0, verbose=False)
        obj_ns = compute_obj_normalized(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            # Try from a perturbed start
            pert = rng.normal(0, 0.03, len(best_seq))
            a_pert = best_seq * np.exp(pert)
            a_pert = np.maximum(a_pert, 0) / float(np.sum(a_pert))
            a_ns2 = nesterov_smoothed_descent(a_pert, time_budget=alloc,
                                              tau_init=50.0, tau_end=500.0, verbose=False)
            obj_ns2 = compute_obj_normalized(a_ns2)
            if obj_ns2 < best_score:
                best_score = obj_ns2
                best_seq = a_ns2.copy()
                print(f"  Nesterov (perturbed): obj={best_score:.8f} **")
            else:
                print(f"  Nesterov: obj={min(obj_ns, obj_ns2):.8f}")

    # === Phase 5: Simulated Annealing ===
    print(f"\n=== Phase 5: Simulated Annealing ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 80.0)
        a_sa = annealed_search(best_seq, time_budget=alloc,
                               T_init=0.005, T_end=0.0005, seed=seed + 5555)
        obj_sa = compute_obj_normalized(a_sa)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            # Follow up with CD
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_cd_sa = coordinate_descent(a_sa, time_budget=min(remaining * 0.3, 60.0),
                                             verbose=False, seed=seed + 6666)
                obj_cd_sa = compute_obj_normalized(a_cd_sa)
                if obj_cd_sa < best_score:
                    best_score = obj_cd_sa
                    best_seq = a_cd_sa.copy()
                    print(f"  SA+CD: obj={best_score:.8f} **")
            print(f"  SA: obj={best_score:.8f} **")
        else:
            print(f"  SA: obj={obj_sa:.8f}")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 50, n_best - 100, n_best + 50, n_best + 100,
        n_best + 150, n_best + 200, 600, 700, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) > 30]
    n_options = n_options[:6]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # CD on new length
        alloc_cd = min(remaining * 0.2, 50.0)
        if alloc_cd > 15:
            a_try = coordinate_descent(a_try, time_budget=alloc_cd,
                                       verbose=False, seed=n_try)

        # Brief LP
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.08, 20.0)
        if alloc_lp > 8:
            a_try = lp_polish(a_try, time_budget=alloc_lp, verbose=False,
                              seed=n_try + 2000)

        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = 1 + r
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.4 + 0.15 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / float(a_r.sum())

        alloc = min(remaining * 0.2, 60.0)
        if alloc > 20:
            a_r = coordinate_descent(a_r, time_budget=alloc, verbose=False,
                                     seed=8000 + r * 111)
            obj_r = compute_obj_normalized(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.4, 120.0)
        a_final = coordinate_descent(best_seq, time_budget=alloc,
                                     verbose=False, seed=seed + 19999)
        obj_final = compute_obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_lp_final = lp_polish(best_seq, time_budget=min(remaining * 0.5, 50.0),
                               verbose=False, seed=9999)
        obj_lp_final = compute_obj_normalized(a_lp_final)
        if obj_lp_final < best_score:
            best_score = obj_lp_final
            best_seq = a_lp_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more CD pass if time
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd_final = coordinate_descent(best_seq, time_budget=min(remaining * 0.6, 50.0),
                                        verbose=False, seed=seed + 25000)
        obj_cd_final = compute_obj_normalized(a_cd_final)
        if obj_cd_final < best_score:
            best_score = obj_cd_final
            best_seq = a_cd_final.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
