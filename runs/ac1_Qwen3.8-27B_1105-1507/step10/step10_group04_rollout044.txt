import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_lp(a, solver_list=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, **{
                'verbose': False
            })
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:
            continue
    return None, None

        max_conv = float(np.max(conv))
        
        # Compute softmax weights
        scaled = tau * (conv - max_conv)
        # Clip for numerical stability
        scaled = np.maximum(scaled, -700)
        exp_vals = np.exp(scaled)
        sum_exp = float(np.sum(exp_vals))
        
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp
        
        # Gradient: grad_j = 2 * sum_k p[k] * a_{k-j} (circular, then correct)
        # Actually, the gradient of max(conv) w.r.t. a[j] is:
        # sum_k p[k] * (conv(a,a))[k]_a[j] = sum_k p[k] * a[k-j] (for valid k)
        # Using FFT: ifft(fft(p) * conj(fft(a)))
        P = fft(p, N_fft)
        A_f = fft(y, N_fft)
        grad_raw = ifft(P * np.conj(A_f)).real[:n]
        g = 2.0 * grad_raw
        
        # Adaptive step size
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        lr = 1.0 / (g_norm + 1e-10)
        lr = min(lr, 5.0 / np.sqrt(n))
        lr = max(lr, 1e-10)
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        beta = min(0.9, beta + 0.0002)
        
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            beta = 0.5
            best_it = it
            if it > 20000:
                break
        
        if verbose and it % 2000 == 0:
            print(f"    GD-ultra it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}, t={elapsed:.0f}s")
    
    return best_a

def kkt_correction(a, time_budget, verbose=True):
    """KKT-based correction: identify active set and adjust."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Active set: entries within 1e-8 of max
        tolerance = max(1e-10, max_conv * 1e-8)
        active = np.where(conv >= max_conv - tolerance)[0]
        
        if len(active) == 0:
            break
        if len(active) == 2 * n - 1:
            break  # All active, no room for improvement via this method
        
        # The gradient at the max point: for each active k,
        # grad_j = a[k-j] (if k-j is valid, else 0)
        # We want to find a direction that decreases all active conv values
        # while maintaining sum(a) = 1 and a >= 0
        
        # Simple approach: compute the "average gradient" over active set
        avg_grad = np.zeros(n)
        for k in active:
            for j in range(n):
                idx = k - j
                if 0 <= idx <= 2*n-2:
                    # The contribution to conv[k] from a[j] is a[k-j] if 0 <= k-j < n
                    if 0 <= k - j < n:
                        avg_grad[j] += a[k-j]
        avg_grad /= len(active)
        avg_grad *= 2.0
        
        # Move opposite to average gradient
        g_norm = np.linalg.norm(avg_grad)
        if g_norm < 1e-15:
            break
        
        lr = 0.1 / g_norm
        for t in [lr, lr*0.5, lr*2]:
            a_try = a - t * avg_grad
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break
        
        if verbose and (time.monotonic() - start) % 5 < 0.1:
            print(f"    KKT: obj={best_obj:.8f}, active={len(active)}, t={time.monotonic()-start:.0f}s")
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.5
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
        a = a / a.sum()
    elif mode == 5:
        # Symmetric bell
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
        a = a / a.sum()
    elif mode == 6:
        # Two-hump
        x = np.linspace(0, 1, n)
        a = np.exp(-10*(x-0.3)**2) + np.exp(-10*(x-0.7)**2)
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a

def optimize_single(a, time_budget, verbose=False):
    """Full optimization pipeline for a single sequence."""
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    
    best = a / s
    best_obj = compute_obj(best)
    start = time.monotonic()
    
    # Phase A: LP accelerated
    t = time.monotonic() - start
    if time_budget - t > 20:
        alloc = min((time_budget - t) * 0.35, 300.0)
        res = lp_accelerated(best, time_budget=alloc, verbose=verbose)
        obj = compute_obj(res)
        if obj < best_obj:
            best_obj, best = obj, res
            if verbose:
                print(f"      LP-acc: {best_obj:.8f}")
    
    # Phase B: Ultra GD
    t = time.monotonic() - start
    if time_budget - t > 15:
        alloc = min((time_budget - t) * 0.35, 300.0)
        res = smoothed_gd_ultra(best, time_budget=alloc, tau_start=500, tau_end=50000, verbose=verbose)
        obj = compute_obj(res)
        if obj < best_obj:
            best_obj, best = obj, res
            if verbose:
                print(f"      GD-ultra: {best_obj:.8f}")
    
    # Phase C: KKT correction
    t = time.monotonic() - start
    if time_budget - t > 10:
        alloc = min((time_budget - t) * 0.2, 60.0)
        res = kkt_correction(best, time_budget=alloc, verbose=verbose)
        obj = compute_obj(res)
        if obj < best_obj:
            best_obj, best = obj, res
            if verbose:
                print(f"      KKT: {best_obj:.8f}")
    
    # Phase D: Final LP + GD
    t = time.monotonic() - start
    if time_budget - t > 10:
        alloc = min((time_budget - t) * 0.4, 60.0)
        res = lp_accelerated(best, time_budget=alloc*0.5, verbose=False)
        obj = compute_obj(res)
        if obj < best_obj:
            best_obj, best = obj, res
        t = time.monotonic() - start
        if time_budget - t > 5:
            res = smoothed_gd_ultra(best, time_budget=time_budget - t - 2, tau_start=5000, tau_end=200000, verbose=False)
            obj = compute_obj(res)
            if obj < best_obj:
                best_obj, best = obj, res
    
    return best

def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses accelerated LP best-response, ultra-high-tau
    smoothed gradient descent, KKT correction, and multi-length exploration
    to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_init = 800

    print(f"Starting: n={n_best}, obj={best_score:.8f}")
    
    # === Phase 1: Intensive optimization of initial sequence ===
    print(f"\n=== Phase 1: Initial Optimization ===")
    remaining = deadline - time.monotonic()

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        if
