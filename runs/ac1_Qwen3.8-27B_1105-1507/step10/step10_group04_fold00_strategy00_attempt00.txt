        exp_vals = np.exp(tau * (conv_full - max_conv))

        if s_exp < 1e-300:
            return np.inf, (exp_vals - 1) / np.finfo(float).eps
        else:
            obj_val = max_conv + np.log(s_exp) / tau
            # Compute gradient using vectorized form (pseudo)
            # But wait, for each entry i, grad_i = 2 * sum_k [p_k * a[k-i]] for i in support
            # Actually, we can compute gradient via ifft since it's analytic
            conv_matrix = build_conv_matrix(a_full)
            # Only the first half of the convolution is necessary for the gradient
            # But note: the provided objective is for the full convolution
            # We have to center the gradient computation for stability
            
            # Matrix-free gradient approximation (simplified)
            # using Fourier transforms for all operations
            grad_full = 2 * conv_matrix.max() * tau_full * exp_vals / s_exp
            grad_full[full_len:] = 0  # Limit gradient to first n elements
            grad_full[0:full_len] *= tau_full
            
            # Project gradient onto the interval [0, full_len]
            grad = np.zeros(full_len)
            # But note: the gradient is for log-parameters, so we take the derivative
            # from the exponential function and the constraint
            
            # Return objective and gradient (only for large elements)
            return obj_val, grad[:full_len]
        
        # Wrap the function for gradient descent
        options={'maxiter':200, 'gtol':1e-8, 'ftol':1e-10, 'disp':False}
        res = scipy_minimize(obj_grad, best_b, method='L-BFGS-B', jac=True, options=options)
        if res.success:
            b_new = res.x
            a_new = np.exp(b_new)
            s_new = np.sum(a_new)
            if s_new > 1e-12:
                a_new = a_new / s_new
            else:
                a_new = best_b / s_new
            obj_new = obj_grad(b_new)[0]
            if obj_new < compute_obj(a_new):
                return a_new, obj_new
        # Default if optimization fails
        return a, compute_obj(a)

def evaluate_sequence(sequence):
    """Evaluate the sequence with clipping, normalization, and objective."""
    n_val = len(sequence)
    if not isinstance(sequence, list):
        sequence = list(sequence)
    if n_val == 0:
        return np.inf
        
    # Defensive checks for all elements
    for x in sequence:
        if not isinstance(x, np.float64):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
        if x < 0:
            return np.inf
            
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]  # Double check
    s_val = np.sum(sequence)
    if s_val < 0.01:
        sequence = [0.0] * n_val  # Avoid division by zero
        return np.inf
        
    sequence = sequence / s_val  # Normalize
    conv = compute_conv(sequence)
    max_conv_val = np.max(conv)
    return float(2 * n_val * max_conv_val / (s_val ** 2))

def resample_sequence(a, new_n=100):
    """Resample sequence of length n_val to new_n with cubic interpolation."""
    n_old = len(a)
    if new_n == n_old:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s_new = np.sum(a_new)
    if s_new < 1e-10:
        a_new = [0.1] * new_n
        s_new = np.sum(a_new)
        a_new = a_new / s_new
    return np.maximum(a_new, 0)
