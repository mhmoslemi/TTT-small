import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_rows(a, k_indices):
    """Build rows of convolution matrix for specific k indices."""
    n = len(a)
    k_arr = np.asarray(k_indices)
    j_idx = np.arange(n)[None, :]
    k_idx = k_arr[:, None]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((len(k_arr), n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp(a, k_indices):
    """Solve LP: min M s.t. rows of conv matrix @ g <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_rows(a_norm, k_indices)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def smoothed_gradient(a, tau):
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.03):
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 6):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

Uses active-set LP-driven iteration with convergence-aware stopping,
high-tau gradient polishing, multi-n exploration, and diverse restarts.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_rows(a, k_indices):
    """Build rows of convolution matrix for specific k indices.
    Row k is a vector v such that v @ g = (a * g)[k] = sum_j a[j] * g[k-j]."""
    n = len(a)
    k_arr = np.asarray(k_indices, dtype=np.int64)
    j_idx = np.arange(n)[None, :]
    k_idx = k_arr[:, None]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((len(k_arr), n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp(a, k_indices):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for k in k_indices, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_rows(a_norm, k_indices)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

            if s_try > 1e-10:
                a_try = a_try / s_try
                best_obj = obj_at_t
                best_a = a_try.copy()
                a = a_try
                stall_count = 0

                # Update active set based on new convolution
                conv_new = compute_conv(a)
                max_new = float(np.max(conv_new))
                # Keep entries that are within 0.1% of max (or top 150)
                threshold = max_new * 0.9995
                active_set = [k for k in range(2 * n - 1) if conv_new[k] >= threshold]
                # Always include the argmax
                max_idx = int(np.argmax(conv_new))
                if max_idx not in active_set:
                    active_set.append(max_idx)
                # Limit size for speed
                if len(active_set) > 150:
                    sorted_k = np.argsort(conv_new)[::-1][:120]
                    active_set = [int(k) for k in sorted_k]

                if verbose and (iter_count % 5 == 0 or iter_count <= 5):
                    elapsed = time.monotonic() - start
                    print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.4f}, "
                          f"|A|={len(active_set)}, time={elapsed:.0f}s")
        else:
            stall_count += 1
            total_stalls += 1

            # Try progressively smaller steps
            if stall_count <= 4:
                for t_small in [0.05, 0.03, 0.01, 0.005]:
                    a_small = (1.0 - t_small) * a + t_small * g
                    a_small = np.maximum(a_small, 0)
                    s_small = float(np.sum(a_small))
                    if s_small > 1e-10:
                        a_small = a_small / s_small
                        obj_small = compute_obj(a_small)
                        if obj_small < best_obj - 1e-11:
                            best_obj = obj_small
                            best_a = a_small.copy()
                            a = a_small
                            stall_count = 0
                            if verbose:
                                print(f"    LP it{iter_count}: small step t={t_small}, obj={best_obj:.8f}")
                            break

            if stall_count >= max_stalls:
                # Escalating perturbation
                tau = 300.0 + 100.0 * (total_stalls // max_stalls)
                conv_p = compute_conv(a)
                max_p = float(np.max(conv_p))
                exp_p = np.exp(min(tau, 2000.0) * (conv_p - max_p))
                p = exp_p / np.sum(exp_p)

                N_fft = 1
                while N_fft < 3 * n:
                    N_fft <<= 1
                P = fft(p, N_fft)
                Af = fft(a, N_fft)
                grad = 2.0 * ifft(P * np.conj(Af)).real[:n]

                grad_norm = float(np.linalg.norm(grad))
                if grad_norm > 1e-12:
                    lr = 0.015 / grad_norm
                    a = a - lr * grad
                    a = np.maximum(a, 0)
                    s_new = float(np.sum(a))
                    if s_new > 1e-10:
                        a = a / s_new
                    else:
                        a = best_a.copy()

                # Add small random perturbation scaled by stall severity
                pert_scale = 0.002 + 0.001 * (total_stalls // max_stalls)
                a = a + pert_scale * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()

                # Rebuild active set
                conv_new = compute_conv(a)
                max_new = float(np.max(conv_new))
                threshold = max_new * 0.998
                active_set = [k for k in range(2 * n - 1) if conv_new[k] >= threshold]
                if max_new > 0:
                    active_set.append(int(np.argmax(conv_new)))
                if len(active_set) > 150:
                    sorted_k = np.argsort(conv_new)[::-1][:120]
                    active_set = [int(k) for k in sorted_k]

                stall_count = 0

                if verbose:
                    print(f"    LP it{iter_count}: perturbed (total_stalls={total_stalls})")

                # If we've stalled many times total, give up
                if total_stalls > max_stalls * 8:
                    if verbose:
                        print(f"    LP: giving up after {total_stalls} total stalls")
                    break

    return best_a


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def gd_polish(a, time_budget=100.0, tau_init=200.0, tau_end=5000.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * (progress ** 2.0)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        # Adaptive learning rate
        if g_norm > 1e-15:
            adapt = min(1.0 / (g_norm + 1e-10), 3.0 * base_lr)
            adapt = max(adapt, 0.1 * base_lr)
            lr = 0.8 * lr + 0.2 * adapt

        # Barzilai-Borwein step every 5 iterations
        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 3.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)

        # Reset if no improvement for too long
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it

    if verbose:
        print(f"    GD: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.04):
    """Add low-frequency Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 7):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
