"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses intensive LP polishing with parabolic line search, FFT-based smoothed gradient
descent with adaptive tau and Nesterov momentum, multi-length exploration, and
perturbation-based local minimum escape. Key improvements: tighter LP tolerances,
smoother tau scheduling, wider n sweep, and coordinate-aware bottleneck reduction.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def convolve_with_a(a, g):
    """Compute conv(a, g) = a * g (linear convolution) via FFT."""
    n_a = len(a)
    n_g = len(g)
    size = 1
    while size < n_a + n_g - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:n_a + n_g - 1]


def solve_lp_precise(a, tol=1e-12):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Uses multiple solvers with tight tolerances."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            kwargs = {}
            if solver == cp.MOSEK:
                kwargs = {'tol': 1e-11, 'max_iters': 5000}
            elif solver == cp.CLARABEL:
                kwargs = {'tol_feas': 1e-12, 'tol_gap_rel': 1e-12, 'tol_infeas': 1e-12}
            prob.solve(solver=solver, **kwargs)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:
            continue
    return None, None


def parabolic_line_search(a, direction, best_obj, n_samples=15):
    """Find optimal mixing parameter t via parabolic interpolation."""
    n = len(a)
    best_t = 0.0
    best_obj_local = best_obj
    best_a = a.copy()

    # Evaluate on a log-spaced grid
    t_vals = np.logspace(-2, 0, n_samples)  # 0.01 to 1.0
    objs = np.zeros(n_samples)

    for i, t in enumerate(t_vals):
        a_try = (1.0 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            objs[i] = best_obj + 1
            continue
        a_try = a_try / s_try
        objs[i] = compute_obj(a_try)

    # Find the best on the grid
    idx = np.argmin(objs)
    if objs[idx] >= best_obj - 1e-14:
        return a, best_obj, 0.0

    # Parabolic refinement around the best
    if 0 < idx < n_samples - 1:
        t1, t2, t3 = t_vals[idx - 1], t_vals[idx], t_vals[idx + 1]
        f1, f2, f3 = objs[idx - 1], objs[idx], objs[idx + 1]
        denom = (f1 - 2*f2 + f3)
        if abs(denom) > 1e-15:
            t_star = t2 - 0.5 * (t2 - t1)**2 * (f2 - f3) / (t2 - t3) * (t3 - t1) / (t2 - t1)
            t_star = max(t1, min(t3, t_star))
        else:
            t_star = t2

        a_try = (1.0 - t_star) * a + t_star * direction
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try > 1e-10:
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < objs[idx]:
                return a_try, obj_try, t_star

    return a / np.sum(a), objs[idx], t_vals[idx]


def lp_polish_intensive(a, time_budget, verbose=False):
    """Intensive LP polishing with parabolic line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    stall_count = 0

        if remaining < 2:

        else:

    return best_a


def fft_smoothed_grad(a, tau):
    """Compute smoothed gradient of max(conv) using FFT. Returns (obj_smooth, grad)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    obj_smooth = max_conv + np.log(sum_exp) / tau

    # Gradient: grad[j] = 2 * sum_k p[k] * conv_contrib
    # grad[j] = 2 * (p * a_flipped)_j where a_flipped is a reversed
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A_f = fft(a, N)
    g_full = ifft(P * np.conj(A_f)).real
    grad = 2.0 * g_full[:n]

    return obj_smooth, grad, max_conv, p


def nesterov_smoothed_gd(a, time_budget, tau_start=50, tau_end=10000, verbose=False):
    """Nesterov-accelerated smoothed gradient descent with adaptive tau."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.85
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Adaptive tau: start low, increase aggressively
        tau = tau_start + (tau_end - tau_start) * progress ** 0.7

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        _, grad, max_conv, p = fft_smoothed_grad(y, tau)

        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-16:
            break

        # Adaptive learning rate
        lr = min(1.0 / (g_norm + 1e-10), 3.0 / np.sqrt(n))
        lr = max(lr, 1e-10)

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        beta = min(0.95, beta + 0.00005)

        # Restart if stuck
        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it
            if it > 50000:
                break

        if verbose and it % 3000 == 0:
            print(f"      GD it{it}: obj={best_obj:.10f}, tau={tau:.0f}, t={elapsed:.0f}s")

    return best_a


def log_param_smoothed_grad(b, tau, support, full_len):
    """Smoothed objective and gradient in log-parameter space."""
    b_shifted = b - b.max()
    a_supp = np.exp(b_shifted)
    a_supp = a_supp / a_supp.sum()

    a = np.zeros(full_len)
    a[support] = a_supp

    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    obj_val = max_conv + np.log(sum_exp) / tau

    N = 1
    while N < 3 * full_len:
        N <<= 1
    P = fft(p, N)
    A_full = fft(a, N)
    g_full = ifft(P * np.conj(A_full)).real
    grad_full = 2.0 * g_full[:full_len]

    # Chain rule for softmax parameterization
    grad_b = a_supp * (grad_full[support] - np.sum(a_supp * grad_full[support]))

    return obj_val, grad_b


def log_param_lbfgs(a, time_budget, tau_min=100, tau_max=8000, verbose=False):

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.5
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
        a = a / a.sum()
    elif mode == 5:
        a = np.exp(-np.arange(n) / (n * 0.2)) + 0.3 * np.exp(-np.arange(n) / (n * 0.8))
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x ** 3 * (1 - x)
        a = a / a.sum()
    elif mode == 7:
        # Gaussian-like
        x = np.linspace(-3, 3, n)
        a = np.exp(-x**2 / 2)
        a = a / a.sum()
    elif mode == 8:
        # Two-peaked
        x = np.linspace(-3, 3, n)
        a = np.exp(-(x-1)**2 / 0.5) + np.exp(-(x+1)**2 / 0.5)
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a

def bottleneck_reduction(a, time_budget, verbose=False):
    """Try to reduce the maximum convolution entry by adjusting entries near the bottleneck."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_val = float(conv[k_star])

        # Find which entries contribute most to conv[k_star]
        # conv[k_star] = sum_i a[i] * a[k_star - i]
        contributions = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]

        # Try reducing the largest contributors
        top_idx = np.argsort(contributions)[::-1][:5]

        improved = False
        for idx in top_idx:
            for frac in [0.8, 0.7, 0.6, 0.5]:
                a_try = a.copy()
                a_try[idx] *= frac
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            if improved:
                break

        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: intensive LP polishing, Nesterov smoothed GD,
    log-param L-BFGS, multi-n exploration, perturbation restarts, and
    bottleneck reduction.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0  # safety margin

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = diverse_init(n_init, rng, mode=2)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.10f}")

    n_best = len(best_seq)

    # === Phase 1: Intensive LP Polishing (350s) ===
    print(f"\n=== Phase 1: Intensive LP Polishing ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 350.0)
    if p1_budget > 30:
        a_lp = lp_polish_intensive(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-14:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1: obj={best_score:.10f} **")
        else:
            print(f"  Phase1: obj={obj_lp:.10f} (no improvement)")

    # === Phase 2: Nesterov Smoothed GD (250s) ===
    print(f"\n=== Phase 2: Nesterov Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.18, 250.0)
    if p2_budget > 20:
        a_gd = nesterov_smoothed_gd(best_seq, time_budget=p2_budget,
                                     tau_start=50, tau_end=15000, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score - 1e-14:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase2: obj={best_score:.10f} **")
        else:
            print(f"  Phase2: obj={obj_gd:.10f}")

    # === Phase 3: Log-Param L-BFGS (100s) ===
    print(f"\n=== Phase 3: Log-Param L-BFGS ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 100.0)
    if p3_budget > 20:
        a_log = log_param_lbfgs(best_seq, time_budget=p3_budget,
                                 tau_min=200, tau_max=10000, verbose=True)
        obj_log = compute_obj(a_log)
        if obj_log < best_score - 1e-14:
            best_score = obj_log
            best_seq = a_log.copy()
            print(f"  Phase3: obj={best_score:.10f} **")
        else:
            print(f"  Phase3: obj={obj_log:.10f}")

    # === Phase 4: Bottleneck Reduction (50s) ===
    print(f"\n=== Phase 4: Bottleneck Reduction ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.05, 50.0)
    if p4_budget > 10:
        a_bot = bottleneck_reduction(best_seq, time_budget=p4_budget)
        obj_bot = compute_obj(a_bot)
        if obj_bot < best_score - 1e-14:
            best_score = obj_bot
            best_seq = a_bot.copy()
            print(f"  Phase4: obj={best_score:.10f} **")
        else:
            print(f"  Phase4: obj={obj_bot:.10f}")

    # === Phase 5: Multi-n Exploration (350s) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.25, 350.0)

    n_current = len(best_seq)
    # Generate diverse n candidates
    n_candidates = sorted(set([
        max(200, n_current - 200),
        max(200, n_current - 100),
        n_current,
        n_current + 100,
        n_current + 200,
        n_current + 400,
        n_current + 600,
        n_current + 1000,
        n_current + 2000,
        300, 400, 500, 600, 700, 900, 1000, 1200, 1500, 2000, 3000, 4000
    ]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 5000 and n != n_current]
    # Sort by distance from current (try close ones first)
    n_candidates.sort(key=lambda x: abs(x - n_current))
    n_candidates = n_candidates[:10]  # Limit to 10
    print(f"  Trying n values: {n_candidates}")

    per_n_budget = p5_budget / max(1, len(n_candidates))
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(per_n_budget * 1.5, remaining - 40)
        if alloc < 30:
            break

        # Resample from best
        a_try = resample_sequence(best_seq, n_try)

        # Quick LP polish
        lp_alloc = min(alloc * 0.4, remaining - 30)
        if lp_alloc > 15:
            a_try = lp_polish_intensive(a_try, time_budget=lp_alloc, verbose=False)

        # GD
        remaining = deadline - time.monotonic()
        gd_alloc = min(alloc * 0.4, remaining - 20)
        if gd_alloc > 10:
            a_try = nesterov_smoothed_gd(a_try, time_budget=gd_alloc,
                                          tau_start=50, tau_end=5000, verbose=False)

        # Log-param
        remaining = deadline - time.monotonic()
        log_alloc = min(alloc * 0.2, remaining - 10)
        if log_alloc > 8:
            a_try = log_param_lbfgs(a_try, time_budget=log_alloc,
                                     tau_min=100, tau_max=5000, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 6: Diverse Restarts (150s) ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p6_budget = min(remaining * 0.12, 150.0)
    n_current = len(best_seq)
    n_rest = max(1, int(p6_budget / 50))

    for r in range(n_rest):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(50, remaining - 15)
        if alloc < 20:
            break

        mode = r % 9
        a_r = diverse_init(n_current, rng, mode=mode)
        # Blend with best
        blend = 0.2 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        # Optimize
        half = alloc * 0.5
        a_r = lp_polish_intensive(a_r, time_budget=half, verbose=False)
        remaining = deadline - time.monotonic()
        gd_alloc = min(alloc * 0.35, remaining - 10)
        if gd_alloc > 5:
            a_r = nesterov_smoothed_gd(a_r, time_budget=gd_alloc,
                                        tau_start=50, tau_end=3000, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score - 1e-14:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.10f}")

    # === Phase 7: Perturbation-based exploration (80s) ===
    print(f"\n=== Phase 7: Perturbation Exploration ===")
    remaining = deadline - time.monotonic()
    p7_budget = min(remaining * 0.08, 80.0)
    if p7_budget > 20:
        per_pert = p7_budget / 10
        for p_iter in range(10):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            strength = 0.05 + 0.02 * (p_iter % 5)
            a_pert, obj_pert = perturbation_restart(best_seq, best_score, rng, strength)
            if a_pert is not None:
                # Quick polish
                polish_alloc = min(per_pert * 0.7, remaining - 10)
                if polish_alloc > 5:
                    a_pert = lp_polish_intensive(a_pert, time_budget=polish_alloc, verbose=False)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_score - 1e-14:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  Pert {p_iter} (s={strength:.2f}): obj={best_score:.10f} **")

    # === Phase 8: Final Intensive Polish (remaining) ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s")

    if remaining > 60:
        # LP
        alloc = min(remaining * 0.3, 120.0)
        a_final = lp_polish_intensive(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-14:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    # GD with high tau
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.3, 150.0)
        a_final = nesterov_smoothed_gd(best_seq, time_budget=alloc,
                                        tau_start=500, tau_end=20000, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-14:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.10f} **")

    # Log-param
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 80.0)
        a_final = log_param_lbfgs(best_seq, time_budget=alloc,
                                   tau_min=500, tau_max=15000, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-14:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Log: obj={best_score:.10f} **")

    # Another LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 60.0)
        a_final = lp_polish_intensive(best_seq, time_budget=alloc, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-14:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra-final LP: obj={best_score:.10f} **")

    # Final GD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.5, 30.0)
        a_final = nesterov_smoothed_gd(best_seq, time_budget=alloc,
                                        tau_start=2000, tau_end=30000, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-14:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra-final GD: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
