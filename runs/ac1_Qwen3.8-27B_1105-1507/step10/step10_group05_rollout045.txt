    #          = 2 * sum_{j=0}^{min(n-1, 2n-2-m)} p[j+m] * a[j]
    
    # For m in [0, n-1]: j+m ranges from m to m+n-1, and 2n-2-m >= n-1 (since m <= n-1)
    # So j ranges from 0 to n-1, and j+m ranges from m to m+n-1.
    # Since 2n-2 >= m+n-1 (i.e., n-1 >= m), this is always valid.
    
    # So grad[m] = 2 * sum_{j=0}^{n-1} a[j] * p[m+j]
    
    # This is a "reverse cross-correlation" that can be computed via FFT.
    # Let's compute p_ext = p (length 2n-1)
    # We need for each m: sum_{j=0}^{n-1} a[j] * p[m+j]
    # = <a, p[m:m+n]> (inner product of a with the segment p[m:m+n])
    
    # This is equivalent to: for the sequence p, compute the running inner product with a.
    # We can do this with a single FFT-based convolution:
    # sum_j a[j] * p[m+j] = sum_j a[j] * p[m+j]
    # Let's define a_rev[j] = a[n-1-j]. Then:
    # sum_j a[j] * p[m+j] = sum_j a_rev[n-1-j] * p[m+j]
    # = sum_i a_rev[i] * p[m+n-1-i] (with i = n-1-j)
    # = (a_rev * p)[m+n-1] (linear convolution)
    
    # So grad[m] = 2 * (a_rev * p)[m+n-1] where * denotes linear convolution.
    
    a_rev = a[::-1]
    N2 = 1
    while N2 < n + (2*n-1) - 1:
        N2 <<= 1
    A_rev = fft(a_rev, N2)
    P = fft(p, N2)
    conv_result = ifft(A_rev * P).real
    # conv_result[k] = sum_i a_rev[i] * p[k-i] for valid i,k
    # We want conv_result[m+n-1] for m = 0, ..., n-1
    grad = 2.0 * conv_result[n-1:2*n-1]
    
    return obj_smooth, grad


def lp_with_golden_line_search(a, time_budget=300.0, verbose=False, seed=42):
    """
    LP-based optimization with golden-section line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    best_it = 0
    phi = (1 + math.sqrt(5)) / 2  # golden ratio
    inv_phi = 1.0 / phi
    inv_phi2 = inv_phi * inv_phi

    momentum = 0.0
    prev_g = None

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Nesterov momentum
        if prev_g is not None:
            delta = g_lp - prev_g
            momentum = 0.5 * momentum + 0.5 * np.linalg.norm(delta)
            g_lp = g_lp + 0.2 * min(momentum, 0.5) * delta
            g_lp = np.maximum(g_lp, 0)
            s_lp = float(np.sum(g_lp))
            if s_lp > 1e-10:
                g_lp = g_lp / s_lp
        prev_g = g_lp.copy()

        # Golden section line search on f(t) = obj((1-t)*a + t*g_lp)
        # We want to minimize over t in [0, 1.5] (allowing some extrapolation)
        def line_obj(t):
            if t <= 0:
                return compute_obj(a)
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return np.inf
            a_try = a_try / s_try
            return compute_obj(a_try)

        # First evaluate at a few points to bracket
        f0 = best_obj  # f(0)
        f05 = line_obj(0.5)
        f1 = line_obj(1.0)
        
        # Determine bracket
        if f1 < f05 and f1 < f0:
            # Minimum might be beyond 1.0
            lo, hi = 0.0, 1.5
        elif f05 < f0 and f05 < f1:
            lo, hi = 0.0, 1.0
        elif f0 < f05 and f0 < f1:
            lo, hi = 0.0, 0.5
        else:
            lo, hi = 0.0, 1.0

        # Golden section search
        c = lo + inv_phi2 * (hi - lo)
        d = lo + inv_phi * (hi - lo)
        fc = line_obj(c)
        fd = line_obj(d)
        
        n_golden = 0
        while (hi - lo) > 0.01 and n_golden < 20:
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = lo + inv_phi2 * (hi - lo)
                fc = line_obj(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + inv_phi * (hi - lo)
                fd = line_obj(d)
            n_golden += 1

        t_best = (lo + hi) / 2
        a_try = (1.0 - t_best) * a + t_best * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            a_try = best_a.copy()
            t_best = 0.0
        else:
            a_try = a_try / s_try

        obj_new = compute_obj(a_try)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_try.copy()
            a = best_a
            best_it = it
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP-GS it{it}: obj={best_obj:.8f}, t={t_best:.4f}, M_lp={M_lp:.6f}, t_elapsed={elapsed:.0f}s")
        else:
            # No improvement - try perturbation
            if it - best_it > 10:
                strength = min(0.1, 0.01 * (it - best_it))
                x = np.arange(n)
                freq = rng.integers(1, max(2, n//100))
                phase = rng.uniform(0, 2*np.pi)
                pert = 1.0 + strength * np.sin(2*np.pi*freq*x/n + phase)
                a = best_a * pert
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                prev_g = None
                momentum = 0.0

        it += 1

    return best_a


def lbfgs_smoothed(a, time_budget=100.0, tau_start=50.0, tau_end=500.0, verbose=False):
    """
    Use scipy L-BFGS-B on the smoothed objective with increasing tau.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Project to simplex: we need sum(a)=1 and a>=0
    # L-BFGS-B with bounds [0, inf) and we'll handle the sum constraint by
    # working in a reduced space (drop one coordinate) or by using a projected gradient.
    
    # Actually, let's use a simpler approach: optimize in the log domain.
    # a_i = exp(x_i) / sum(exp(x_j))
    # This automatically satisfies a_i > 0 and sum(a) = 1.
    
    x0 = np.log(np.maximum(a, 1e-15))
    x0 = x0 - np.max(x0)  # for numerical stability

    def obj_and_grad(x):
        # Compute a from x
        x_shifted = x - np.max(x)
        a_x = np.exp(x_shifted)
        a_x = a_x / np.sum(a_x)
        
        # Current tau based on progress
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress * progress
        
        obj_s, grad_s = smoothed_obj_and_grad(a_x, tau)
        
        # Chain rule: d(a_x)/d(x_i) = a_x[i] * (1 - a_x[i]) (for the softmax)
        # d(obj_s)/d(x_i) = sum_j d(obj_s)/d(a_j) * d(a_j)/d(x_i)
        # = sum_j grad_s[j] * a_x[j] * (delta_ji - a_x[i])
        # = grad_s[i] * a_x[i] - a_x[i] * sum_j grad_s[j] * a_x[j]
        
        grad_x = grad_s * a_x - a_x * np.dot(grad_s, a_x)
        
        return obj_s, grad_x

    # Use L-BFGS-B
    result = minimize(obj_and_grad, x0, method='L-BFGS-B',
                      options={'maxiter': 5000, 'gtol': 1e-10, 'maxls': 50,
                               'ftol': 1e-14, 'iprint': 0})
    
    if result.x is not None:
        x_opt = result.x - np.max(result.x)
        a_opt = np.exp(x_opt)
        a_opt = a_opt / np.sum(a_opt)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    L-BFGS: obj={best_obj:.8f}")
    
    # Final check with the true objective
    elapsed = time.monotonic() - start
    if verbose:
        print(f"    L-BFGS done in {elapsed:.1f}s, best_obj={best_obj:.8f}")
    
    return best_a


def fourier_init(n, rng, n_modes=10):
    """Initialize a sequence with low-frequency Fourier structure."""
    x = np.arange(n)
    a = np.ones(n)
    for k in range(1, n_modes + 1):
        amp = rng.uniform(0.1, 0.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.sqrt(2) * np.cos(2 * np.pi * k * x / n + phase)
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    from scipy.interpolate import interp1d
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    a_new = f(x_new)
    return np.maximum(a_new, 0)
