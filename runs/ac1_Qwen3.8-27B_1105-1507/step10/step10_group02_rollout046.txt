"""
Optimizes step function coefficients using enhanced coordinate descent with
closed-form 1D subproblem initialization, spectral domain flattening,
structured perturbation escapes, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

        for k in active_lags:
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                priority[j] += conv[k]
        
        # Mix random and priority-based ordering
        if sweep % 3 == 0:
            order = rng.permutation(n)
        else:
            # Sort by priority (descending) with some randomization
            noise = rng.uniform(0, 1, n) * 0.3 * np.max(priority + 1)
            order = np.argsort(-(priority + noise))

        S = float(np.sum(a))
        improved = 0

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            # Quick check: if v is already very small, maybe skip
            if v < 1e-10 and S_other > 0.99:
                # Try setting to a small positive value
                x_try = 0.01 * S / n
                # Skip for now, let golden section handle it
                pass

            # Solve 1D subproblem
            n_local = n
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            k_quad = min(2 * i, 2 * n - 2)

            ks = np.arange(k_min, k_max + 1)
            mask = (ks != k_quad)
            ks_lin = ks[mask]
            a_ki = a[ks_lin - i]
            c_k_lin = conv[ks_lin].copy()
            c_2i = float(conv[k_quad])

            # Evaluate 1D function at several points
            scale = max(v, S / n, 1e-6)
            
            # Closed-form candidate from quadratic
            x_quad = max(0.0, (c_2i - v * v) / S_other)
            
            # Closed-form candidate from best linear
            x_lin_best = v
            if len(ks_lin) > 0:
                valid = a_ki > 1e-15
                if np.any(valid):
                    x_lin_cands = S_other - c_k_lin[valid] / a_ki[valid] + 2 * v
                    x_lin_cands = x_lin_cands[x_lin_cands > 0]
                    if len(x_lin_cands) > 0:
                        # Pick the one that minimizes the max
                        for xl in x_lin_cands[:5]:
                            lin_vals = c_k_lin + 2.0 * a_ki * (xl - v)
                            max_lin = float(np.max(lin_vals))
                            quad_val = c_2i + xl * xl - v * v
                            max_val = max(max_lin, quad_val)
                            f_val = 2.0 * n * max_val / ((S_other + xl) ** 2)
                            x_lin_best = xl  # Just take first valid for now
                        x_lin_best = float(x_lin_cands[0])

            # Gather candidates
            cands = [v, x_quad, x_lin_best, 0.0, v * 0.5, v * 1.5, v * 2.0, S / n]
            cands = sorted(set(max(0.0, float(c)) for c in cands))

            # Quick eval
            best_x = v
            best_f = 1e30
            for xc in cands:
                if len(ks_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki * (xc - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + xc * xc - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + xc) * (S_other + xc)
                if denom < 1e-30:
                    fcand = 1e30
                else:
                    fcand = 2.0 * n * max_val / denom
                if fcand < best_f:
                    best_f = fcand
                    best_x = xc

            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + 2.0 * scale
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-10

            def f_1d(xc):
                if len(ks_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki * (xc - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + xc * xc - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + xc) * (S_other + xc)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(30):
                if x_hi - x_lo < 1e-12 * max(1.0, abs(x_hi)):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            delta = x_opt - v

            if abs(delta) < 1e-14:
                continue

            # Update
            a[i] = x_opt
            S += delta
            improved += 1

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki

        # Track best
        S_new = float(np.sum(a))
        if S_new > 1e-10:
            obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s, improved={improved}")

    return best_a


def spectral_flattening(a, alpha=0.3, num_iters=20):
    """
    Spectral domain flattening: blend power spectrum with flat spectrum,
    reconstruct, project to non-negative reals.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N = 1
    while N < 2 * n:
        N <<= 1

    for _ in range(num_iters):
        A = fft(a, N)
        P = np.abs(A) ** 2
        # Blend with flat spectrum
        P_flat = np.full(N, np.mean(P))
        P_new = (1 - alpha) * P + alpha * P_flat
        # Reconstruct
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a = a_new / s_new

    return a


def structured_perturb(a, rng, mode=0):
    """Apply structured perturbation to escape local optima."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    if mode == 0:
        # Add low-frequency sinusoid
        x = np.arange(n)
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.2)
        a = a + amp * np.sin(2 * np.pi * freq * x / n + phase)
    elif mode == 1:
        # Gaussian perturbation
        a = a + rng.normal(0, 0.05, n)
    elif mode == 2:
        # Taper perturbation (modify edges)
        taper = np.ones(n)
        edge = max(1, n // 10)
        taper[:edge] = np.linspace(0.5, 1, edge)
        taper[-edge:] = np.linspace(1, 0.5, edge)
        a = a * taper
    elif mode == 3:
        # Random smooth perturbation (low-pass filtered noise)
        noise = rng.normal(0, 1, n)
        # Simple smoothing
        kernel = np.ones(5) / 5
        noise = np.convolve(noise, kernel, mode='same')
        a = a + 0.1 * noise
    elif mode == 4:
        # Bloom/anti-bloom (shift mass from center to edges or vice versa)
        x = np.arange(n)
        center = n / 2
        bloom = np.exp(-((x - center) ** 2) / (2 * (n / 6) ** 2))
        sign = rng.choice([-1, 1])
        a = a + sign * 0.1 * bloom
    else:
        a = a + rng.normal(0, 0.02, n)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def lp_polish(a, time_budget=60.0, verbose=False, seed=42):
    """LP-based polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 2:
                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 8, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a / a.sum()
    elif mode == 7:
        # Hanning window
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
