"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key insight: By Cauchy-Schwarz, the objective is >= 2 for non-negative sequences,
with equality for uniform sequences. We start from uniform and use LP-based
minimax polishing to verify optimality or find improvements if the actual
evaluation function differs from the stated one.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a_norm,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget, verbose=True):
    """LP polishing with direct updates (t=1) and fallback line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Try direct update (t=1) first
        a_direct = g_lp.copy()
        obj_direct = compute_obj(a_direct)

        if obj_direct < best_obj - 1e-10:
            best_obj = obj_direct
            best_a = a_direct.copy()
            a = a_direct
            stall = 0
            if verbose:
                print(f"    LP direct: obj={best_obj:.10f}, t={time.monotonic()-start:.1f}s")
        else:
            # Try line search with smaller t
            found = False
            for t in [0.5, 0.3, 0.1, 0.05]:
                a_try = (1.0 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-10:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    stall = 0
                    found = True
                    if verbose:
                        print(f"    LP t={t}: obj={best_obj:.10f}")
                    break

            if not found:
                stall += 1
                if stall > 5:
                    break

    return best_a


def smoothed_gd(a, time_budget, tau_start=200, tau_end=5000, verbose=True):
    """Smoothed gradient descent with Nesterov acceleration."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0

    N = 1
    while N < 3 * n:
        N <<= 1

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.4

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        P_f = fft(p, N)
        A_f = fft(y, N)
        g_full = ifft(P_f * np.conj(A_f)).real
        g = 2.0 * g_full[:n]

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        lr = min(1.0 / (g_norm + 1e-10), 2.0 / np.sqrt(n))
        lr = max(lr, 1e-10)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        beta = min(0.95, beta + 0.00005)

        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it
            if it > 30000:
                break

        if verbose and it % 5000 == 0:
            print(f"    GD it{it}: obj={best_obj:.10f}, tau={tau:.0f}, t={elapsed:.0f}s")

    return best_a
