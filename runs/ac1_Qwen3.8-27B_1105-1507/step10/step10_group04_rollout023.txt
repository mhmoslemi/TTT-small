"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based alternating minimization, high-tau smoothed gradient descent,
KKT-based refinement, multi-length exploration, and diverse structured initializations.
Key improvements: faster LP with warm starts, structured inits (cosine, raised-cosine,
water-filling), adaptive n selection, KKT direct solve for final polish.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, j)
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_lp(a, verbose=False):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    return g_val
        except Exception:
            continue
    return None


def lp_polish(a, time_budget, max_iters=50, verbose=True):
    """LP-based alternating minimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    last_t = 0.5
    stall = 0
    
    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 5:
            break
        
        g_lp = solve_lp(a)
        if g_lp is None:
            break
        
        # Line search
        t_candidates = np.unique(np.clip(np.concatenate([
            np.linspace(0.05, 1.0, 15),
            [last_t * 0.5, last_t * 0.7, last_t, last_t * 1.3, last_t * 1.6]
        ]), 0.01, 1.0))
        
        best_t_obj = best_obj
        best_t = 0
        best_a_local = best_a.copy()
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-13:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()
        
        if best_t > 0 and best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t
            stall = 0
            if verbose:
                print(f"    LP iter {it}: obj={best_obj:.8f}, t={best_t:.3f}, t={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 3:
                break
    
    return best_a


def high_tau_gd(a, time_budget, tau_start=200, tau_end=10000, verbose=False):
    """High-precision smoothed gradient descent with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.7
        
        # Nesterov
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp
        
        # Gradient via FFT
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(y, N)
        g_full = ifft(P * np.conj(A_f)).real
        g = 2.0 * g_full[:n]
        
        # Adaptive step
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        lr = min(1.0 / (g_norm + 1e-10), 3.0 / np.sqrt(n))
        lr = max(lr, 1e-8)
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        beta = min(0.98, beta + 0.0002)
        
        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            if it > 60000:
                break
    
    return best_a


def diverse_init(n, rng, mode=0):

        a = np.cos(np.pi * x / 2) ** 2
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def kkt_refine(a, time_budget, verbose=False):
    """
    KKT-based refinement: at the optimum, for all j in support,
    2 * sum_k p_k * a[k-j] = lambda (constant).
    We solve this system iteratively.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Identify active set: convolution coefficients at (or near) the max
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    tol = max_conv * 0.01
    active = np.where(conv >= max_conv - tol)[0]
    
    # p_k: uniform on active set initially
    p = np.zeros(2 * n - 1)
    p[active] = 1.0 / len(active)
    
    # Build the gradient matrix for active set
    # G[j] = 2 * sum_k p_k * a[k-j] (with a_m = 0 for invalid m)
    # We want G[j] = lambda for all j in support
    
    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        if it > 200:
            break
        
        # Compute G[j] = 2 * sum_k p_k * a[k-j]
        # This is 2 * conv(p, a) (but p has length 2n-1, a has length n)
        # Actually: sum_k p[k] * a[k-j] where a[m] = 0 for m < 0 or m >= n
        # This is the cross-correlation of p and a
        # Let's compute it directly
        G = np.zeros(n)
        for j in range(n):
            # G[j] = 2 * sum_k p[k] * a[k-j] (valid k-j in [0, n-1])
            # k ranges from j to j + 2n - 2 (but p has length 2n-1, so k in [0, 2n-2])
            # k-j in [0, n-1] means k in [j, j+n-1], intersected with [0, 2n-2]
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            if k_lo <= k_hi:
                G[j] = 2.0 * np.dot(p[k_lo:k_hi+1], a[k_lo - j : k_hi - j + 1])
        
        # KKT: G[j] should be constant (= lambda) on the support
        support = np.where(a > 1e-10)[0]
        if len(support) == 0:
            break
        
        lambda_val = float(np.mean(G[support]))
        
        # Residual: how far G is from being constant
        residual = np.max(np.abs(G[support] - lambda_val))
        if residual < 1e-10:
            break
        
        # Update: we want to adjust a to make G more constant
        # Use gradient ascent on the "flatness" of G
        # Or: directly solve the linear system G = lambda on support
        # G[j] = 2 * sum_k p[k] * a[k-j]
        # This is linear in a! So we can solve:
        # For j in support: 2 * sum_k p[k] * a[k-j] = lambda
        # Plus: sum_{j in support} a[j] = 1
        # Plus: a[j] >= 0
        
        # Build the system
        m = len(support)
        A_sys = np.zeros((m + 1, m))
        rhs = np.zeros(m + 1)
        
        for i, j in enumerate(support):
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            for j2 in support:
                if k_lo <= j2 <= k_hi:
                    A_sys[i, list(support).index(j2)] += 2.0 * p[j2 - j + j]  # a[j2-j]... 
                    # Wait, G[j] = 2 * sum_k p[k] * a[k-j]
                    # = 2 * sum_{j2 in support} p[j2 + j] * a[j2]  (where k = j2 + j, so k-j = j2)
                    # Wait, that's not right either.
                    # G[j] = 2 * sum_k p[k] * a[k-j]
                    # Let m = k-j, so k = m+j. Then G[j] = 2 * sum_m p[m+j] * a[m]
                    # = 2 * sum_{m in support} p[m+j] * a[m]
                    # So A_sys[i, idx_of_m] = 2 * p[m+j]
                    pass
        
        # Let me redo this more carefully
        # G[j] = 2 * sum_{m=0}^{n-1} p[m+j] * a[m] (where p[index] = 0 if index out of [0,2n-2])
        # For j in support: G[j] = lambda
        # So: for each j_i in support: sum_{m in support} p[m + j_i] * a[m] = lambda/2
        
        A_sys = np.zeros((m + 1, m))
        for i, j_i in enumerate(support):
            for idx_m, m_val in enumerate(support):
                p_idx = m_val + j_i
                if 0 <= p_idx <= 2 * n - 2:
                    A_sys[i, idx_m] = 2.0 * p[p_idx]
            rhs[i] = lambda_val  # We'll solve for a such that G = lambda
        
        # Sum constraint
        A_sys[m, :] = 1.0
        rhs[m] = 1.0
        
        # Solve
        try:
            a_sol = np.linalg.solve(A_sys, rhs)
            a_sol = np.maximum(a_sol, 0)
            s_sol = float(np.sum(a_sol))
            if s_sol > 1e-10:
                a_sol = a_sol / s_sol
                a_new = np.zeros(n)
                a_new[support] = a_sol
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
        except np.linalg.LinAlgError:
            # Singular system, use least squares
            try:
                a_sol, _, _, _ = np.linalg.lstsq(A_sys, rhs, rcond=None)
                a_sol = np.maximum(a_sol, 0)
                s_sol = float(np.sum(a_sol))
                if s_sol > 1e-10:
                    a_sol = a_sol / s_sol
                    a_new = np.zeros(n)
                    a_new[support] = a_sol
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj - 1e-13:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
            except Exception:
                pass
        
        # Update active set and p
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        tol = max_conv * 0.05
        active = np.where(conv >= max_conv - tol)[0]
        p = np.zeros(2 * n - 1)
        if len(active) > 0:
            p[active] = 1.0 / len(active)
    
    return best_a

    """
    Main search function. Uses LP polishing, high-tau GD, KKT refinement,
    multi-n exploration, and diverse structured initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0  # safety margin
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = diverse_init(800, rng, mode=0)
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    print(f"\n=== Phase 1: Quick LP Polish ===")

    if p1_budget > 10:
