def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_gradient_smoothed(a, tau, N_fft=None):
    """
    Compute gradient of log-sum-exp smoothed max(conv(a,a)).
    Returns (gradient, conv_vals, p_dist).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    if N_fft is None:
        N_fft = 1
        while N_fft < 3 * n:
            N_fft <<= 1
    
    P = fft(p, N_fft)
    A = fft(a, N_fft)
    g_full = ifft(P * np.conj(A)).real
    grad = 2.0 * g_full[:n]
    
    return grad, conv, p

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)

    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    best_g = None
    best_M = None
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    if best_g is None or M_val < best_M:
                        best_g = g_val
                        best_M = M_val
                    break
        except Exception:
            continue
    
    return best_g, best_M

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, kind, rng):
    """Generate structured initial sequences."""
    x = np.linspace(0, 1, n)
    if kind == 'uniform':
        a = np.ones(n)

        a = np.exp(-3 * (1 - x))

        a = np.exp(-10 * (x - 0.3)**2) + np.exp(-10 * (x - 0.7)**2)

    else:

    a = np.maximum(a, 1e-6)
    a = a / a.sum()

def duality_gap_estimate(a, tau):
    """Estimate the duality gap for the current iterate."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    expected = float(np.dot(p, conv))
    gap = max_conv - expected
    return 2.0 * n * gap, 2.0 * n * max_conv

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_gradient_smoothed(a, tau, N_fft):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*conv(a,a))) w.r.t. a.
    Returns (gradient, conv_vals, p_distribution).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    scaled = tau * (conv - max_conv)
    exp_vals = np.exp(np.maximum(scaled, -700))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    P = fft(p, N_fft)
    A = fft(a, N_fft)
    g_full = ifft(P * np.conj(A)).real
    grad = 2.0 * g_full[:n]

    return grad, conv, p

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    best_g = None
    best_M = None

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    if best_g is None or M_val < best_M:
                        best_g = g_val
                        best_M = M_val
                    break
        except Exception:
            continue

    return best_g, best_M

def resample_seq(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                                   verbose=False, seed=n_try, beta=0.75, lr_scale=5.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Diversification ===
    print(f"\n=== Phase 4: Diversification ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    # Try different initializations blended with best
    init_kinds = [0, 1, 2, 3, 4, 5, 6, 7]
    per_init = max(15, min(50, remaining / len(init_kinds)))

    for kind in init_kinds:
        remaining = deadline - time.monotonic()
        if remaining < per_init + 5:
            break

        a_init = structured_init(n_cur, kind, rng)
        a_blend = 0.6 * best_seq + 0.4 * a_init
        a_blend = a_blend / a_blend.sum()

        alloc = min(per_init, remaining - 10)
        if alloc > 10:
            tau_i = max(30, n_cur // 15)
            tau_e = n_cur * 25
            a_div = accelerated_mw(a_blend, time_budget=alloc,
                                   tau_init=tau_i, tau_end=tau_e,
                                   verbose=False, seed=1000 + kind, beta=0.75, lr_scale=5.0)
            # Also try perturbation of current best
            a_pert = fourier_perturb(best_seq, rng, strength=0.04)
            a_pert2 = accelerated_mw(a_pert, time_budget=min(alloc * 0.5, 30),
                                     tau_init=tau_i, tau_end=tau_e,
                                     verbose=False, seed=2000 + kind, beta=0.8, lr_scale=4.0)

            obj_div = compute_obj(a_div)
            obj_pert = compute_obj(a_pert2)
            best_candidate = a_div if obj_div < obj_pert else a_pert2
            best_cand_obj = min(obj_div, obj_pert)

            if best_cand_obj < best_score:
                best_score = best_cand_obj
                best_seq = best_candidate.copy()
                print(f"  Div kind={kind}: obj={best_score:.8f} **")
            else:
                print(f"  Div kind={kind}: obj={best_cand_obj:.8f}")

    # === Phase 5: Final LP + AMW polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    if remaining > 50:
        alloc = min(remaining * 0.35, 180.0)
        a_f = lp_polish(best_seq, time_budget=alloc, verbose=True, seed=7777)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_f:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.35, 150.0)
        tau_i = max(200, n_cur // 3)
        tau_e = n_cur * 150
        a_f = accelerated_mw(best_seq, time_budget=alloc,
                             tau_init=tau_i, tau_end=tau_e,
                             verbose=True, seed=8888, beta=0.85, lr_scale=3.0)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final AMW: obj={best_score:.8f} **")
        else:
            print(f"  Final AMW: obj={obj_f:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.5, 100.0)
        a_f = lp_polish(best_seq, time_budget=alloc, verbose=False, seed=9999)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Ultra LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
