def evaluate_sequence(sequence: Union[List[float], np.ndarray]) -> float:
    """
    Evaluates a sequence of non-negative coefficients for the given minimization objective.
    Normalizes the sequence to ensure sum >= 0.01, clamps values to [0, 1000], and computes the normalized max autocorrelation.
    Returns float('inf') if invalid.
    """
    arr = np.array(sequence)
    n = len(arr)
    if n == 0 or not np.all(arr >= 0) or not np.isfinite(arr).all():
        return float('inf')
    sum_a = np.sum(arr)
    if sum_a < 0.01:
        # Insufficient sum, need to scale up
        arr = np.clip(arr, 0, 1000)  # Ensure we don't have values above 1000
        scaling_factor = 0.01 / sum_a / 0.997  # Avoid numerical issues with scaling
        arr = (arr * scaling_factor).astype(float)
        sum_a = 0.01
    
    normalized_seq = arr / sum_a
    conv_result = np.convolve(normalized_seq, normalized_seq, mode='full')
    max_b = np.max(conv_result)
    return (2.0 * n * max_b / (sum_a ** 2)).item()

def create_sequence_from_weights(weights: List[float], n: int) -> np.ndarray:
    """ 
    Create a sequence of length n by distributing the initial weights across the length.
    Weights are placed at increasing intervals: the first weight gets 0.5 of the remaining amount,
    the second gets 0.25, etc. This creates a decreasing trend with early peaks.
    If fewer weights than n, leftover portion is divided by zeros.
    """
    sequence = np.zeros(n)
    remaining_sum = 0.0  # Will account for the total contribution from weights
    amount = np.array(weights) / np.sum(weights)
    # Distribute concentrated weights at specific positions
    start = 0
    for i, val in enumerate(amount):
        try:
            pos = int(start + (n / len(weights))) if len(weights) > 1 else i
            if val > 0 and pos < n:
                if np.isnan(val) or val < 0:  # Safety check
                    val = 0.0
                sequence[pos] = sum(weights) * 0.1**(i/len(weights))
                remaining_sum += val
            else:
                val = 0.0
        except (ValueError, TypeError):
            val = 0.0
        finally:
            start += 1  # Ensure we move the position
    
    final_sum = np.sum(sequence)
    if abs(final_sum - sum(weights)) > 1e-5:
        # Adjustments in case some mass was lost during computations
        sequence = (sequence * np.sqrt(sum(weights)**2 / final_sum**2)).astype(float)
    return normalized_seq = sequence / np.sum(sequence)
