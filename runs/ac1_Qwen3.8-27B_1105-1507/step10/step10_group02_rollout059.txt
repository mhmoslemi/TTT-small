    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""

def build_conv_cross_matrix(a, n):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g) (cross correlation)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        if r > 0:
            A[j:j + r, j] = a[:r]
    return A

    """

    active = np.where(conv_aa >= threshold)[0]

    A_full = build_conv_cross_matrix(a_n, n)

    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

        try:

                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

            a = np.maximum(a, 0)
            a = a / a.sum()
            continue
        
        # Try multiple blending factors
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 4:
                # Try different top_frac
                for tf in [0.2, 0.5, 0.6]:
                    g_lp2, _ = solve_lp_active(a, top_frac=tf)
                    if g_lp2 is not None:
                        for t in [0.1, 0.3, 0.5]:
                            a_try = (1.0-t)*a + t*g_lp2
                            a_try = np.maximum(a_try, 0)
                            s_try = float(np.sum(a_try))
                            if s_try > 1e-10:
                                a_try = a_try / s_try
                                obj_try = 2.0*n*float(np.max(compute_conv(a_try)))
                                if obj_try < best_obj:
                                    best_obj = obj_try
                                    best_a = a_try.copy()
                                    a = a_try
                                    stall = 0
                                    break
                        if stall == 0:
                            break
                if stall > 7:
                    pert = rng.normal(0, 0.03, n)
                    a = a * np.exp(pert)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    stall = 3
    
    return best_a


def symmetrize(a):
    """Try symmetrized version (a + reverse(a))/2."""
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0)
    s = float(np.sum(a_sym))
    if s < 1e-10:
        return a
    return a_sym / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1.0 - x**2 + 1e-8)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 10, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-4, 4)
        a = a / a.sum()
    elif mode == 6:
        # Triangular
        a = 1.0 - np.abs(np.arange(n) - n//2) / (n//2)
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def smoothed_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) via log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    from scipy.fft import fft, ifft
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def nesterov_descent(a, time_budget=100.0, verbose=False):
    """Nesterov accelerated smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    
    beta = 0.85
    v = np.zeros(n)
    lr = 2.0 / np.sqrt(n)
    tau = 30.0
    
    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = (time.monotonic() - start) / max(time_budget, 1e-10)
        tau = 30.0 + 300.0 * elapsed
        beta = 0.85 + 0.1 * elapsed
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g = smoothed_gradient(y, tau)
        g_n = float(np.linalg.norm(g))
        if g_n > 1e-15:
            lr = min(max(lr * 0.998, 0.3 * 2.0 / np.sqrt(n)), 3.0 / np.sqrt(n))
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        it += 1
        if verbose and it % 3000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}")
    
    return best_a


def propose_candidate(budget_s=1500):

    rng = np.random.default_rng(12345)

        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
