from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_matrix(a, n):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a, warm_start=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = compute_conv_matrix(a_norm, n)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            if warm_start is not None and solver in [cp.MOSEK, cp.CLARABEL]:
                prob.solve(solver=solver, warm_start=True)
            else:
                prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft

import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    lr = lr_init

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * (progress ** 0.7)
        lr = lr_init * (1.0 + 0.5 * progress)

        # Compute convolution and softmax weights
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Compute gradient
        grad = compute_grad_smoothed(a, p)

        # Mirror descent update: a_i <- a_i * exp(-lr * grad_i / (2n)) / Z
        # Scale gradient for numerical stability
        grad_scale = lr / (2.0 * n)
        logits = -grad_scale * grad
        logits -= np.max(logits)  # numerical stability
        weights = np.exp(logits)
        a_new = a * weights
        s_new = float(np.sum(a_new))
        if s_new < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        it += 1

        # Check objective (every few iterations to save time)
        if it % 5 == 0:
            obj = compute_obj_normalized(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        if verbose and it % 2000 == 0:
            print(f"      MD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}")

    # Final check
    obj = compute_obj_normalized(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a


def coordinate_descent_fast(a, time_budget, seed=42, verbose=True):
    """
    Peak-directed coordinate descent with efficient 1D subproblem solving.
    Uses gradient-based ordering and adaptive golden section.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    prev_obj = best_obj
    no_improve_sweeps = 0

    while time.monotonic() - start < time_budget:
        sweep += 1

        # Recompute conv periodically to avoid drift
        if sweep % 5 == 1 or sweep == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj_check = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a.copy()

        # Determine ordering: peak-directed
        max_val = float(np.max(conv))
        # Score coordinates by contribution to top peaks
        top_frac = 0.1
        n_top = max(1, int(len(conv) * top_frac))
        top_ks = np.argsort(-conv)[:n_top]

        scores = np.zeros(n)
        for k in top_ks:
            k = int(k)
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                scores[j] += a[k - j]

        # Mix peak-directed with random for exploration
        mix = min(0.3, 0.5 / sweep)  # less random over time
        if sweep <= 3:
            order = rng.permutation(n)
        else:
            order_peak = np.argsort(-scores)
            order_rand = rng.permutation(n)
            # Interleave
            order = np.zeros(n, dtype=int)
            n_peak = max(1, int(n * (1.0 - mix)))
            order[:n_peak] = order_peak[:n_peak]
            order[n_peak:] = order_rand[:n - n_peak]

        S_cur = S

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            i = int(i)
            v = a[i]
            S_other = S_cur - v

            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks]

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Compute inactive max once
            inactive_mask = np.ones(2 * n - 1, dtype=bool)
            inactive_mask[k_min:k_max + 1] = False
            if np.any(inactive_mask):
                inactive_max = float(np.max(conv[inactive_mask]))
            else:
                inactive_max = 0.0

            # 1D objective
            def f_1d(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val_ = max(max_lin, quad_val, inactive_max)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val_ / denom

            # Evaluate at current and a few neighbors
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.3, v * 0.7, v * 1.3, v * 2.0]
            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                if x < 0:
                    continue
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section search
            x_lo = max(0.0, best_x - scale * 0.8)
            x_hi = best_x + scale * 1.2
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(25):
                if x_hi - x_lo < 1e-11 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) > 1e-13:
                delta = x_opt - v
                a[i] = x_opt
                if 0 <= k_quad <= 2 * n - 2:
                    conv[k_quad] += x_opt * x_opt - v * v
                if len(ks_lin) > 0:
                    conv[ks_lin] += 2.0 * delta * a_ki_lin
                S_cur += delta

        # End of sweep: track best
        S_new = float(np.sum(a))
        if S_new > 1e-10:
            obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                no_improve_sweeps = 0
            else:
                no_improve_sweeps += 1

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

        # Early stop if converged
        if no_improve_sweeps > 15 and sweep > 10:
            break

    return best_a


def build_conv_matrix(a, n):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm, n)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None


def lp_polish(a, time_budget, seed=42, verbose=False):
    """LP-based polishing with blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Try multiple blending factors
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 2:
                # Perturb
                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if time.monotonic() - start > time_budget * 0.8:
                    break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine
        x = np.arange(n) / (n - 1)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a + 0.01, 0.01)
        a = a / a.sum()
    elif mode == 3:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * x - 1.0)
        a = np.maximum(a + 0.05, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Gaussian center
        x = np.arange(n)
        c = n / 2
        w = n / 6
        a = np.exp(-((x - c) ** 2) / (2 * w * w))
        a = a / a.sum()
    elif mode == 5:
        # Two-hump
        x = np.arange(n)
        a = np.exp(-((x - n * 0.25) ** 2) / (2 * (n * 0.1) ** 2)) + \
            np.exp(-((x - n * 0.75) ** 2) / (2 * (n * 0.1) ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Sine-weighted
        x = np.linspace(0.01, 1.0, n)
        a = np.sin(np.pi * x) ** rng.uniform(0.5, 2.0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def optimize_single_n(a_init, time_budget, seed=42, verbose=True):
    """Optimize a sequence of fixed length n using all methods."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    # Phase 1: Mirror descent (40% of budget)
    md_budget = time_budget * 0.40
    if md_budget > 10:
        a_md = mirror_descent(a, time_budget=md_budget, verbose=verbose)
        obj_md = compute_obj_normalized(a_md)
        if obj_md < best_obj:
            best_obj = obj_md
            best_a = a_md.copy()
        if verbose:
            print(f"      After MD: obj={best_obj:.8f}")

    # Phase 2: Coordinate descent (40% of budget)
    cd_budget = time_budget * 0.40
    remaining = time_budget - (time.monotonic() - start)
    cd_budget = min(cd_budget, remaining * 0.6)
    if cd_budget > 10:
        a_cd = coordinate_descent_fast(best_a, time_budget=cd_budget,
                                       seed=seed, verbose=verbose)
        obj_cd = compute_obj_normalized(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()
        if verbose:
            print(f"      After CD: obj={best_obj:.8f}")

    # Phase 3: LP polish (15% of budget)
    lp_budget = time_budget * 0.15
    remaining = time_budget - (time.monotonic() - start)
    lp_budget = min(lp_budget, remaining * 0.7)
    if lp_budget > 15:
        a_lp = lp_polish(best_a, time_budget=lp_budget, seed=seed + 1,
                         verbose=verbose)
        obj_lp = compute_obj_normalized(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
        if verbose:
            print(f"      After LP: obj={best_obj:.8f}")

    # Phase 4: Final CD polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a_final = coordinate_descent_fast(best_a, time_budget=remaining * 0.8,
                                          seed=seed + 99, verbose=False)
        obj_final = compute_obj_normalized(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
        if verbose:
            print(f"      Final: obj={best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines mirror descent, coordinate descent,
    LP polishing, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    rng = np.random.default_rng(seed)

    # Load parent sequence
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                n_parent = parent.size
                obj_parent = compute_obj_normalized(parent)
                print(f"Parent (n={n_parent}): obj={obj_parent:.8f}")
    except Exception:
        n_parent = 800
        parent = np.ones(n_parent) / n_parent
        obj_parent = compute_obj_normalized(parent)
        print(f"No parent available, using uniform init (n={n_parent})")

    best_seq = parent.copy()
    best_score = obj_parent

    # Define n values to try (prioritize around parent n)
    n_vals = sorted(set([
        n_parent,
        n_parent - 100, n_parent + 100, n_parent + 200,
        n_parent + 400, n_parent + 600,
        600, 800, 1000, 1200, 1500
    ]))
    n_vals = [n for n in n_vals if 400 <= n <= 2500]
    # Sort by expected quality (prefer n close to parent)
    n_vals.sort(key=lambda x: abs(x - n_parent))
    n_vals = n_vals[:7]  # Limit to 7 different n values
    print(f"\nPlanning n values: {n_vals}")

    # Allocate time: more time to n values close to parent
    # First pass: quick screen of all n values
    screen_budget_per_n = min(80.0, (budget_s * 0.3) / len(n_vals))
    print(f"\n=== Phase 1: Screening (budget={screen_budget_per_n:.0f}s per n) ===")

    screen_results = {}
    for n_try in n_vals:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Initialize
        if n_try == n_parent and parent is not None:
            a_init = parent.copy()
        else:
            a_init = resample_sequence(parent, n_try)

        # Quick optimization: just mirror descent + short CD
        alloc = min(screen_budget_per_n, remaining * 0.4)
        if alloc < 15:
            break

        a_screen, obj_screen = optimize_single_n(a_init, time_budget=alloc,
                                                  seed=seed + n_try, verbose=False)
        screen_results[n_try] = (obj_screen, a_screen)
        mark = " **" if obj_screen < best_score else ""
        print(f"  n={n_try}: obj={obj_screen:.8f}{mark}")
        if obj_screen < best_score:
            best_score = obj_screen
            best_seq = a_screen.copy()

    # Second pass: deep optimization on top candidates
    print(f"\n=== Phase 2: Deep Optimization ===")
    # Sort by screen result
    sorted_ns = sorted(screen_results.keys(), key=lambda n: screen_results[n][0])
    top_ns = sorted_ns[:4]  # Focus on top 4
    print(f"  Top n values: {top_ns}")

    deep_budget_per_n = min(250.0, (budget_s * 0.45) / max(1, len(top_ns)))

    for n_try in top_ns:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        alloc = min(deep_budget_per_n, remaining * 0.5)
        if alloc < 30:
            break

        print(f"\n  Deep optimizing n={n_try} (budget={alloc:.0f}s)")
        a_start = screen_results[n_try][1]
        a_deep, obj_deep = optimize_single_n(a_start, time_budget=alloc,
                                             seed=seed + 10000 + n_try, verbose=True)
        mark = " **" if obj_deep < best_score else ""
        print(f"  n={n_try}: obj={obj_deep:.8f}{mark}")
        if obj_deep < best_score:
            best_score = obj_deep
            best_seq = a_deep.copy()

    # Phase 3: Diverse restarts at best n
    print(f"\n=== Phase 3: Diverse Restarts ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 60:
        restart_budget = min(100.0, remaining * 0.3)
        n_restart = int(restart_budget / 3)
        if n_restart < 20:
            n_restart = 20

        for r in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            mode = 2 + r * 2  # modes 2, 4, 6
            a_r = diverse_init(n_best, rng, mode=mode)
            # Blend with best
            blend = 0.4 + 0.1 * r
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            alloc = min(restart_budget / (3 - r), remaining * 0.5)
            if alloc < 20:
                break

            a_r_opt, obj_r = optimize_single_n(a_r, time_budget=alloc,
                                               seed=seed + 50000 + r * 100, verbose=False)
            mark = " **" if obj_r < best_score else ""
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}{mark}")
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r_opt.copy()

    # Phase 4: Final polish
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # LP polish
        lp_alloc = min(60.0, remaining * 0.3)
        if lp_alloc > 15:
            a_lp = lp_polish(best_seq, time_budget=lp_alloc, seed=seed + 7777)
            obj_lp = compute_obj_normalized(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
            print(f"  LP polish: obj={obj_lp:.8f}")

        # Final CD
        remaining = deadline - time.monotonic()
        if remaining > 20:
            cd_alloc = min(remaining * 0.6, 100.0)
            a_final = coordinate_descent_fast(best_seq, time_budget=cd_alloc,
                                              seed=seed + 99999, verbose=False)
            obj_final = compute_obj_normalized(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
            print(f"  Final CD: obj={obj_final:.8f}")

    # Final validation
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(100)
        best_score = compute_obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
