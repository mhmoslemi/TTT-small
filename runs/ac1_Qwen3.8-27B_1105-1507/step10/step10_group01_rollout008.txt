def multiplicative_update(a, alpha=0.1, epsilon=0.01):
    """One step of the multiplicative flattening update."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    # Active set: entries within epsilon of max
    S = np.where(conv >= max_conv - epsilon * max_conv)[0]
    # Compute pressure: p[i] = sum_{k in S} a[k-i]
    p = np.zeros(n)
    for k in S:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                p[i] += a[j]
    # Normalize pressure
    p_mean = np.mean(p[p > 0]) if np.any(p > 0) else 1.0
    # Multiplicative update
    a_new = a * (1 - alpha * (p / p_mean - 1))
    a_new = np.maximum(a_new, 0)
    a_new = a_new / np.sum(a_new)
    return a_new

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_gradient_conv(a, beta):
    """
    Compute smoothed gradient of max(conv(a,a)) using softmax weighting.
    Returns gradient vector of length n.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = beta * (conv - max_conv)
    # Safe softmax
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] (for valid k-i)
    # This is a cross-correlation: grad = 2 * corr(p, a) truncated to length n
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g = ifft(P * A).real  # This gives cross-correlation
    # Actually we need: grad[i] = sum_k p[k] * a[k-i]
    # corr(p,a)[i] = sum_k p[k] * a[k-i] which is what we want
    # But ifft(P*A) gives conv(p,a) not corr. Let's use flip.
    p_flip = p[::-1]
    P_flip = fft(p_flip, N)
    g = ifft(P_flip * A).real[:n]
    # This gives sum_k p_flip[k]*a[i-k] = sum_k p[2n-2-k]*a[i-k]
    # Hmm, let me just do it directly for clarity with n up to ~2000
    # Actually for efficiency, let's use the direct formula:
    # grad[i] = sum_k p[k] * a[k-i] where 0 <= k-i < n, i.e., i <= k < i+n
    # For each i, the valid k range is [i, min(i+n-1, 2n-2)]
    # This is a sliding dot product - can use FFT
    # Let me just compute it as: grad = ifft(fft(p) * conj(fft(a))) gives corr
    P_fft = fft(p, N)
    A_fft = fft(a, N)
    # corr(p, a)[i] = sum_k p[k+i] * conj(a[k]) -- that's not what we want either
    # Let me think: we want g[i] = sum_k p[k] * a[k-i] for 0<=k-i<n
    # = sum_j p[j+i] * a[j] where j = k-i
    # This is the cross-correlation of p and a at lag i
    # numpy corrcoef or manual:
    # g[i] = <p[i:], a> (dot product of p starting at i with a)
    # This is conv(flip(p), a)[n-1+i] ... let me just do it with a loop for safety
    # For n up to 2000, a loop of 2000 iterations each doing O(n) work = 4M ops, fine
    # But let's use the FFT trick properly:
    # g[i] = sum_j a[j] * p[j+i] for j=0..n-1 (where j+i < 2n-1)
    # Define q = p (length 2n-1), then g[i] = <a, q[i:i+n]>
    # This is the "sliding window inner product" = cross-correlation
    # xcorr(a, p)[i] where cross-correlation at lag i = sum_j a[j] * p[j+i]
    # In FFT: xcorr = ifft(conj(fft(a)) * fft(p)) ... let me verify
    # Actually: ifft(conj(A) * P) gives sum_j conj(a[j-m]) * p[j] = sum_j a[j-m] * p[j]
    # = sum_j a[j] * p[j+m] (substituting j->j+m) = g[m] !
    g = ifft(np.conj(A_fft) * P_fft).real
    return 2.0 * g[:n]


def multiplicative_optimize(a, time_budget=100.0, beta_init=20.0, beta_end=500.0,
                            verbose=True):
    """
    Fast multiplicative (mirror descent) optimization with adaptive beta.
    Updates: a[i] *= (1 - alpha * (grad[i] - mean_grad) / (max_grad + eps))
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    alpha = 0.3

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        beta = beta_init + (beta_end - beta_init) * progress ** 0.8

        grad = softmax_gradient_conv(a, beta)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        # Adaptive step size
        max_grad = float(np.max(np.abs(grad)))
        alpha_adapt = 0.5 / (max_grad + 1e-10)
        alpha = 0.7 * alpha + 0.3 * alpha_adapt
        alpha = min(max(alpha, 0.01), 1.0)

        # Multiplicative update (mirror descent)
        grad_centered = grad - np.mean(grad)
        scale = float(np.max(np.abs(grad_centered))) + 1e-10
        a_new = a * np.exp(-alpha * grad_centered / scale)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        if verbose and it % 500 == 0:
            print(f"      Mult it{it}: obj={best_obj:.8f}, beta={beta:.0f}, "
                  f"alpha={alpha:.4f}, time={time.monotonic()-start:.0f}s")

        if it - best_it > 3000:
            a = best_a.copy()
            alpha *= 0.5
            best_it = it

    if verbose:
        print(f"      Mult done: obj={best_obj:.8f}, iters={it}")
    return best_a


def build_conv_matrix(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_minpeak(a, time_limit=5.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.MOSEK, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_optimize(a, time_budget=200.0, verbose=True):
    """LP-driven optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        g, M_lp = solve_lp_minpeak(a, time_limit=min(remaining * 0.3, 5.0))
        if g is None:
            break

        it += 1

        # Line search: try several step sizes
        best_t = 0
        best_t_obj = best_obj
        for t in [0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_t = t

        if best_t > 0 and best_t_obj < best_obj - 1e-13:
            a_new = (1 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_t_obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, "
                      f"M={M_lp:.6f}, time={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall >= 8:
                # Perturb to escape
                rng = np.random.default_rng(it * 7919)
                a = a + 0.003 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

            if verbose and it % 10 == 0:
                print(f"      LP it{it}: stalled obj={best_obj:.8f}, M={M_lp:.6f}")

            if it > 200:
                break

    if verbose:
        print(f"      LP done: obj={best_obj:.8f}, iters={it}")
    return best_a


def gd_nesterov(a, time_budget=100.0, beta_init=100.0, beta_end=2000.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = 0.85
    v = np.zeros(n)
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        beta = beta_init + (beta_end - beta_init) * progress ** 1.2

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        grad = softmax_gradient_conv(y, beta)
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break

        # Adaptive learning rate
        lr = lr * 0.999
        if g_norm > 1e-10:
            lr_adapt = 0.5 * base_lr / (g_norm + 0.1)
            lr = 0.8 * lr + 0.2 * lr_adapt
            lr = min(max(lr, 1e-6), 2.0 * base_lr)

        v = momentum * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

    if verbose:
        print(f"      GD: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def spectral_perturb(a, rng, num_modes=8, strength=0.05):
    """Perturb in frequency domain for diverse restarts."""
    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1
    A = fft(a, N)
    # Perturb magnitudes
    for _ in range(num_modes):
        k = rng.integers(1, N // 2)
        phase_shift = rng.uniform(0, 2 * np.pi)
        A[k] *= np.exp(1j * phase_shift * strength)
        A[N - k] *= np.exp(-1j * phase_shift * strength)
    a_new = ifft(A).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb_search(a, time_budget=60.0, verbose=False):
    """Try small random perturbations and keep improvements."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99999)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        # Random perturbation
        scale = rng.uniform(0.001, 0.02)
        a_try = best_a + scale * rng.standard_normal(n)
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
        else:
            # Try smaller scale
            scale2 = scale * 0.3
            a_try2 = best_a + scale2 * rng.standard_normal(n)
            a_try2 = np.maximum(a_try2, 0)
            s_try2 = float(np.sum(a_try2))
            if s_try2 > 1e-10:
                a_try2 = a_try2 / s_try2
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj - 1e-14:
                    best_obj = obj_try2
                    best_a = a_try2.copy()

    if verbose:
        print(f"      RandPerturb: obj={best_obj:.8f}")
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: combines multiplicative updates, LP refinement,
    multi-n exploration, gradient polish, and random perturbation search.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: Fast Multiplicative Optimization ===
    print(f"\n=== Phase 1: Multiplicative Optimization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a_p1 = multiplicative_optimize(best_seq, time_budget=p1_budget,
                                       beta_init=30.0, beta_end=600.0, verbose=True)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase1 improved: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_p1:.8f} (no improvement)")

    # === Phase 2: LP Refinement ===
    print(f"\n=== Phase 2: LP Refinement ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 30:
        a_p2 = lp_optimize(best_seq, time_budget=p2_budget, verbose=True)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_score:
            best_score = obj_p2
            best_seq = a_p2.copy()
            print(f"  Phase2 improved: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj_p2:.8f} (no improvement)")

    # === Phase 3: Gradient Polish ===
    print(f"\n=== Phase 3: Gradient Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 120.0)
    if p3_budget > 20:
        a_p3 = gd_nesterov(best_seq, time_budget=p3_budget,
                           beta_init=200.0, beta_end=3000.0, verbose=True)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_score:
            best_score = obj_p3
            best_seq = a_p3.copy()
            print(f"  Phase3 improved: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj_p3:.8f} (no improvement)")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(300, n_best - 200), max(300, n_best - 100),
        n_best + 100, n_best + 200, n_best + 400,
        600, 700, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_best) > 50]
    n_options = n_options[:5]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            a_try = np.ones(n_try) / n_try

        # Quick multiplicative optimization
        alloc_mult = min(remaining * 0.3, 60.0)
        if alloc_mult > 15:
            a_try = multiplicative_optimize(a_try, time_budget=alloc_mult,
                                            beta_init=50.0, beta_end=400.0, verbose=False)

            a_try = lp_optimize(a_try, time_budget=alloc_lp, verbose=False)

                                beta_init=300.0, beta_end=2000.0, verbose=False)

    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    restart_types = ["spectral", "mirror", "smooth", "asymmetric"]
    for rtype in restart_types:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        if rtype == "spectral":
            a_r = spectral_perturb(best_seq, rng, num_modes=12, strength=0.08)
        elif rtype == "mirror":
            a_r = (best_seq + best_seq[::-1]) / 2.0
            a_r = a_r / a_r.sum()

            bias = np.linspace(1.0, 0.6, n_cur)
            a_r = best_seq * bias
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

        # Optimize the restart
        alloc_mult = min(remaining * 0.25, 40.0)
        if alloc_mult > 10:
            a_r = multiplicative_optimize(a_r, time_budget=alloc_mult,
                                          beta_init=50.0, beta_end=500.0, verbose=False)
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.15, 30.0)
        if alloc_lp > 8:
            a_r = lp_optimize(a_r, time_budget=alloc_lp, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({rtype}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({rtype}): obj={obj_r:.8f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        alloc_lp = min(remaining * 0.35, 150.0)
        a_final = lp_optimize(best_seq, time_budget=alloc_lp, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_gd = min(remaining * 0.3, 80.0)
        a_final2 = gd_nesterov(best_seq, time_budget=alloc_gd,
                               beta_init=500.0, beta_end=5000.0, verbose=True)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # === Phase 7: Random Perturbation Search ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_rand = min(remaining * 0.5, 60.0)
        a_rand = random_perturb_search(best_seq, time_budget=alloc_rand, verbose=True)
        obj_rand = compute_obj(a_rand)
        if obj_rand < best_score:
            best_score = obj_rand
            best_seq = a_rand.copy()
            print(f"  Random: obj={best_score:.8f} **")

        best_score = compute_obj(best_seq)

    # Ensure all values are valid

        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
