Hybrid optimizer combining fast exact coordinate descent with peak-directed 
perturbation, 2D pair optimization, and simulated annealing escape.
Key improvements over previous version:
1. Faster CD with warm starts and adaptive precision
2. Peak-directed perturbation targeting worst autocorrelation peaks

4. Simulated annealing for generalization
5. More aggressive multi-n exploration

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def fast_cd(a, time_budget=300.0, verbose=True, seed=42, max_sweeps=10000):
    """
    Fast coordinate descent with warm starts and adaptive precision.
    Key optimizations:
    - Warm start: use previous coordinate value as initial bracket
    - Adaptive iterations: fewer golden section iterations when close to optimum
    - Early sweep termination: stop if improvement < threshold
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    prev_x = a.copy()  # warm start values

    while time.monotonic() - start < time_budget and sweep < max_sweeps:
        sweep += 1
        sweep_improvement = 0.0
        pre_obj = best_obj

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:

            if S_other < 1e-12 and v < 1e-12:

            k_min = max(0, i)

            c_k = conv[ks]

            mask = (ks != 2 * i)
            ks_lin = ks[mask]

            k_quad = 2 * i

            n_lin = len(ks_lin)

            # 1D objective: 2n * max(max_lin, quad) / (S_other + x)^2
            # Where max_lin = max(c_k_lin + 2*a_ki_lin*(x-v))

            # Quick evaluation at current point

            else:

            f_v = 2.0 * n * max_v / (S * S)

            # Try a few quick moves
            candidates = [0.0, v * 0.5, v * 1.5, v * 2.0, v * 0.75, v * 1.25]
            if v > 0:

                candidates.append(v * 0.9)

            best_x = v
            best_f = f_v

            for x in candidates:
                if x < 0:
                    x = 0.0
                if n_lin > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    fx = 1e30
                else:
                    fx = 2.0 * n * max_val / denom
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement (fewer iterations than before)
            scale = max(abs(best_x - v), S / n * 0.5, 1e-8)
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 1.5
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            def eval_1d(x):
                if n_lin > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = eval_1d(c_pt)
            fd = eval_1d(d_pt)

            # Adaptive number of iterations
            for _ in range(25):
                if x_hi - x_lo < 1e-12 * max(1.0, x_hi):

"""
Hybrid optimizer for step function coefficient minimization.
Combines fast exact coordinate descent (with precomputed structures and adaptive precision),
peak-directed perturbation, pair coordinate descent, simulated annealing escape,
and multi-length exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming a is already normalized (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return compute_obj_normalized(a / s)


def eval_1d_obj(n, S_other, c_k_lin, a_ki_lin, c_2i, v, x):
    """Evaluate the 1D subproblem objective for changing a[i] from v to x."""
    if len(c_k_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom


def golden_section_1d(n, S_other, c_k_lin, a_ki_lin, c_2i, v, x_lo, x_hi, max_iters=25):
    """Golden section search on the 1D subproblem."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = x_hi - gr * (x_hi - x_lo)
    d_pt = x_lo + gr * (x_hi - x_lo)
    fc = eval_1d_obj(n, S_other, c_k_lin, a_ki_lin, c_2i, v, c_pt)
    fd = eval_1d_obj(n, S_other, c_k_lin, a_ki_lin, c_2i, v, d_pt)

    for _ in range(max_iters):
        if x_hi - x_lo < 1e-12 * max(1.0, x_hi):
            break
        if fc < fd:
            x_hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = x_hi - gr * (x_hi - x_lo)
            fc = eval_1d_obj(n, S_other, c_k_lin, a_ki_lin, c_2i, v, c_pt)
        else:
            x_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = x_lo + gr * (x_hi - x_lo)
            fd = eval_1d_obj(n, S_other, c_k_lin, a_ki_lin, c_2i, v, d_pt)

    return max(0.0, (x_lo + x_hi) / 2.0)


def fast_coordinate_descent(a, time_budget=300.0, verbose=True, seed=42):
    """
    Fast exact coordinate descent with adaptive precision and early termination.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improvement = 0.0

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask].copy()
            c_k_lin = c_k[mask].copy()
            n_lin = len(ks_lin)

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Quick candidate scan
            f_v = eval_1d_obj(n, S_other, c_k_lin, a_ki_lin, c_2i, v, v)
            candidates = [0.0, v * 0.5, v * 1.5, v * 0.75, v * 1.25]
            if v > 1e-10:
                candidates.append(v * 0.1)
                candidates.append(v * 2.0)
            scale_est = S / n
            candidates.append(scale_est)
            candidates.append(2.0 * scale_est)

            best_x = v
            best_f = f_v
            for x in candidates:
                x = max(0.0, float(x))
                fx = eval_1d_obj(n, S_other, c_k_lin, a_ki_lin, c_2i, v, x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement with adaptive iterations
            diff = abs(best_x - v)
            if diff < 1e-10:
                continue

            bracket_scale = max(diff, S / n * 0.3, 1e-8)
            x_lo = max(0.0, best_x - bracket_scale)
            x_hi = best_x + bracket_scale * 1.5

            # Determine iteration count based on potential improvement
            potential = (f_v - best_f) / max(f_v, 1e-10)
            if potential < 0.001:
                n_iters = 10
            elif potential < 0.01:
                n_iters = 18
            else:
                n_iters = 28

            x_opt = golden_section_1d(n, S_other, c_k_lin, a_ki_lin, c_2i, v,
                                      x_lo, x_hi, max_iters=n_iters)

            if abs(x_opt - v) < 1e-14:
                continue

            # Update a and conv incrementally
            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if n_lin > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

            # Track best
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    sweep_improvement += (best_obj - obj)
                    best_obj = obj
                    best_a = a.copy()

        # Early termination
        if sweep_improvement < 1e-12:
            stall_count += 1
            if stall_count >= 3:
                break
        else:
            stall_count = 0

        if verbose and sweep % 30 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, improvement={sweep_improvement:.2e}, t={elapsed:.1f}s")

    # Final normalization
    S_final = float(np.sum(best_a))
    if S_final > 1e-10:
        best_a = best_a / S_final

    return best_a


def pair_coordinate_descent(a, time_budget=60.0, verbose=False, seed=42):
    """
    Pair coordinate descent: optimize pairs of coordinates simultaneously.
    Targets pairs that jointly contribute to the maximum convolution peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iteration = 0
    while time.monotonic() - start < time_budget:
        iteration += 1
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        max_conv = float(conv[peak_idx])

        # Find top contributing pairs to the peak
        j_min = max(0, peak_idx - (n - 1))
        j_max = min(n - 1, peak_idx)
        contributing = np.arange(j_min, j_max + 1)
        contrib_vals = a[contributing] * a[peak_idx - contributing]

        # Get top pairs
        top_k = min(8, len(contributing))
        top_order = np.argsort(contrib_vals)[-top_k:]
        top_indices = contributing[top_order]

        improved = False
        for idx in range(min(top_k, len(top_indices))):
            if time.monotonic() - start >= time_budget:
                break
            i = top_indices[idx]
            j = peak_idx - i
            if j < 0 or j >= n or i == j:
                continue
            if i >= n or j >= n:
                continue

            vi = a[i]
            vj = a[j]

            # Try 2D optimization: vary a[i] and a[j]
            # Use a small grid search in 2D
            best_local_obj = best_obj
            best_xi = vi
            best_xj = vj

            # Grid: 5x5 around current values
            scale_i = max(vi * 0.5, S / n * 0.3)
            scale_j = max(vj * 0.5, S / n * 0.3)

            xi_candidates = [max(0.0, vi + d) for d in [-scale_i, -scale_i/2, 0, scale_i/2, scale_i]]
            xj_candidates = [max(0.0, vj + d) for d in [-scale_j, -scale_j/2, 0, scale_j/2, scale_j]]

            for xi in xi_candidates:
                for xj in xj_candidates:
                    a_try = a.copy()
                    a_try[i] = xi
                    a_try[j] = xj
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj_normalized(a_try)
                    if obj_try < best_local_obj:
                        best_local_obj = obj_try
                        best_xi = xi
                        best_xj = xj

            if best_local_obj < best_obj - 1e-12:
                a[i] = best_xi
                a[j] = best_xj
                S = float(np.sum(a))
                a = a / S
                best_obj = best_local_obj
                best_a = a.copy()
                improved = True

        if not improved and iteration > 10:
            # Try random pairs
            pairs_tried = 0
            for _ in range(20):
                if pairs_trieved >= 5:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue

                vi = a[i]
                vj = a[j]
                scale_i = max(vi * 0.3, S / n * 0.2)
                scale_j = max(vj * 0.3, S / n * 0.2)

                for xi_frac in [0.5, 0.75, 1.0, 1.25, 1.5]:
                    for xj_frac in [0.5, 0.75, 1.0, 1.25, 1.5]:
                        xi = vi * xi_frac
                        xj = vj * xj_frac
                        a_try = a.copy()
                        a_try[i] = xi
                        a_try[j] = xj
                        s_try = float(np.sum(a_try))
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj_normalized(a_try)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try.copy()
                            improved = True
                            pairs_tried += 1
                            break
                    if improved:
                        break

            if not improved:
                break

        if verbose and iteration % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    PairCD iter {iteration}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def peak_directed_optimization(a, time_budget=60.0, verbose=False, seed=42):
    """
    Directly target the worst peaks by finding optimal mass transfers
    from peak-contributing coordinates to non-contributing ones.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for iteration in range(50):
        if time.monotonic() - start >= time_budget:
            break

        conv = compute_conv(a)
        sorted_conv = np.sort(conv)[::-1]
        # How many peaks are near the maximum?
        threshold = sorted_conv[0] * 0.995
        peak_indices = np.where(conv >= threshold)[0]

        for pk in peak_indices[:3]:  # Focus on top 3 peaks
            if time.monotonic() - start >= time_budget:
                break

            j_min = max(0, pk - (n - 1))
            j_max = min(n - 1, pk)
            contrib_idx = np.arange(j_min, j_max + 1)
            contrib_vals = a[contrib_idx] * a[pk - contrib_idx]

            # Total contribution
            total_contrib = float(np.sum(contrib_vals))

            # Non-contributing indices
            contrib_set = set(contrib_idx.tolist())
            non_contrib = [j for j in range(n) if j not in contrib_set]

            if len(non_contrib) == 0:
                continue

            # Try transferring mass from top contributors to non-contributors
            top_contrib_order = np.argsort(contrib_vals)[::-1]
            for tc_idx in top_contrib_order[:5]:
                src = contrib_idx[tc_idx]
                src_val = a[src]
                if src_val < 1e-8:
                    continue

                # Try different transfer fractions and targets
                for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                    transfer = src_val * frac
                    # Pick a random target from non-contributors
                    for _ in range(3):
                        if time.monotonic() - start >= time_budget:
                            break
                        target = non_contrib[rng.integers(0, len(non_contrib))]

                        a_try = a.copy()
                        a_try[src] -= transfer
                        a_try[target] += transfer
                        a_try = np.maximum(a_try, 0)
                        s_try = float(np.sum(a_try))
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try

                        # Quick check: does this reduce the peak?

                        obj_try = 2.0 * n * float(np.max(conv_try))

                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try.copy()

    return best_a


def simulated_annealing(a, time_budget=60.0, verbose=False, seed=42):
    """
    Simulated annealing with adaptive temperature and move types.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    current_obj = best_obj

    start = time.monotonic()
    rng = np.random.default_rng(seed)

    T_init = 0.05
    T_min = 0.0001
    total_time = time_budget
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(total_time, 1e-10))
        T = T_init * (T_min / T_init) ** progress

        it += 1

        # Choose move type
        move_type = rng.integers(0, 4)

        if move_type == 0:
            # Single coordinate perturbation
            i = rng.integers(0, n)
            noise = rng.normal(0, 0.1 * a[i] + 0.01 / n)
            a_try = a.copy()
            a_try[i] = max(0.0, a[i] + noise)

        elif move_type == 1:
            # Pair swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                a_try = a.copy()
                t = rng.uniform(0.1, 0.5)
                val = t * abs(a[i] - a[j])
                if a[i] >= a[j]:
                    a_try[i] -= val
                    a_try[j] += val
                else:
                    a_try[i] += val
                    a_try[j] -= val
                a_try = np.maximum(a_try, 0)
            else:
                continue

        elif move_type == 2:
            # Block redistribution
            block_size = rng.integers(3, min(10, n))
            start_idx = rng.integers(0, n - block_size + 1)
            block = a[start_idx:start_idx + block_size].copy()
            block_sum = float(np.sum(block))
            if block_sum > 1e-10:
                # Random redistribution
                weights = rng.dirichlet(np.ones(block_size))
                a_try = a.copy()
                a_try[start_idx:start_idx + block_size] = weights * block_sum
            else:
                continue

        else:
            # Multiplicative perturbation on random subset
            subset_size = rng.integers(5, min(20, n))
            subset = rng.choice(n, size=subset_size, replace=False)
            a_try = a.copy()
            perturbations = rng.normal(0, 0.05, subset_size)
            a_try[subset] = np.maximum(0, a_try[subset] * np.exp(perturbations))

        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        conv_try = compute_conv(a_try)
        obj_try = 2.0 * n * float(np.max(conv_try))

        delta = obj_try - current_obj
        if delta < 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / max(T, 1e-10))

        if accept:
            a = a_try
            current_obj = obj_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    SA it{it}: best={best_obj:.8f}, curr={current_obj:.8f}, T={T:.5f}, t={elapsed:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x ** 2 + 1e-6)
    elif mode == 4:
        freqs = rng.integers(1, 10, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
    elif mode == 6:
        # Cosine-based
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif mode == 7:
        # Bump functions
        a = np.ones(n) * 0.1
        for _ in range(rng.integers(2, 5)):
            c = rng.integers(0, n)
            w = rng.integers(max(2, n // 10), n // 3)
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
    else:
        a = np.ones(n)
    return np.maximum(a, 0.0)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def smoothed_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1

    from scipy.fft import fft, ifft
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def accelerated_smoothed_descent(a, time_budget=80.0, tau_init=30.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated smoothed gradient descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)

    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-15:
            # Backtracking line search
            lr_try = min(lr, 2.0 / g_norm)
            a_new = y - lr_try * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                lr *= 0.5
            else:
                a_new = a_new / s_new
                obj_new = compute_obj_normalized(a_new)
                if obj_new <= compute_obj_normalized(y) + lr_try * g_norm * 0.5:
                    pass  # Armijo condition met
                else:
                    lr = lr_try * 0.5
                    a_new = y - lr * g
                    a_new = np.maximum(a_new, 0)
                    s_new = float(np.sum(a_new))
                    if s_new > 1e-10:
                        a_new = a_new / s_new
                    else:
                        a_new = best_a.copy()

            # Update momentum
            v = beta * v - lr * g
            a = y + v
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            if s_a < 1e-10:
                a = best_a.copy()
                v = np.zeros(n)
            else:
                a = a / s_a

            obj_new = compute_obj_normalized(a)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()

            # Gradually increase learning rate
            lr = min(lr * 1.001, 5.0 / np.sqrt(n))
        else:
            break

        it += 1
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.4f}, t={elapsed:.1f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple optimization strategies:
    1. Fast coordinate descent (main workhorse)
    2. Peak-directed optimization
    3. Pair coordinate descent
    4. Multi-n exploration
    5. Simulated annealing escape
    6. Smoothed gradient descent
    7. Final CD polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj_normalized(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default)
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    # === Phase 1: Fast Coordinate Descent (main workhorse) ===
    print(f"\n=== Phase 1: Fast Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.40, 550.0)
    if p1_budget > 30:
        a_cd = fast_coordinate_descent(best_seq, time_budget=p1_budget,
                                       verbose=True, seed=seed)
        obj_cd = compute_obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  ** CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: Peak-Directed Optimization ===
    print(f"\n=== Phase 2: Peak-Directed Optimization ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 120.0)
    if p2_budget > 20:
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(p2_budget / (3 - rnd), remaining * 0.5)
            if alloc < 10:
                break
            a_pd = peak_directed_optimization(best_seq, time_budget=alloc,
                                              verbose=False, seed=1000 + rnd * 111)
            obj_pd = compute_obj_normalized(a_pd)
            if obj_pd < best_score:
                best_score = obj_pd
                best_seq = a_pd.copy()
                print(f"  ** PeakDir round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  PeakDir round {rnd}: obj={obj_pd:.8f}")

            # Quick CD after peak-directed
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_cd_quick = fast_coordinate_descent(a_pd, time_budget=min(30, remaining * 0.15),
                                                     verbose=False, seed=2000 + rnd * 100)
                obj_cd_q = compute_obj_normalized(a_cd_quick)
                if obj_cd_q < best_score:
                    best_score = obj_cd_q
                    best_seq = a_cd_quick.copy()
                    print(f"  ** CD after PeakDir {rnd}: obj={best_score:.8f} **")

    # === Phase 3: Pair Coordinate Descent ===
    print(f"\n=== Phase 3: Pair Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.10, 80.0)
    if p3_budget > 15:
        a_pair = pair_coordinate_descent(best_seq, time_budget=p3_budget,
                                         verbose=False, seed=3000)
        obj_pair = compute_obj_normalized(a_pair)
        if obj_pair < best_score:
            best_score = obj_pair
            best_seq = a_pair.copy()
            print(f"  ** PairCD: obj={best_score:.8f} **")
        else:
            print(f"  PairCD: obj={obj_pair:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 200),
        max(400, n_best - 100),
        n_best + 100,
        n_best + 200,
        n_best + 400,
        600, 700, 800, 900, 1000, 1200, 1400, 1600
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) > 50]
    n_options = n_options[:6]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.20, 60.0)
        if alloc > 15:
            a_try = fast_coordinate_descent(a_try, time_budget=alloc,
                                            verbose=False, seed=n_try)

        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts with CD ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = 1 + r
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        alloc = min(remaining * 0.20, 50.0)
        if alloc > 15:
            a_r = fast_coordinate_descent(a_r, time_budget=alloc,
                                          verbose=False, seed=5000 + r * 100)
            obj_r = compute_obj_normalized(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  ** Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 6: Simulated Annealing Escape ===
    print(f"\n=== Phase 6: Simulated Annealing ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Perturb current best slightly
        pert = rng.normal(0, 0.02, len(best_seq))
        a_pert = best_seq * np.exp(pert)
        a_pert = np.maximum(a_pert, 0)
        s_pert = float(np.sum(a_pert))
        if s_pert > 1e-10:
            a_pert = a_pert / s_pert

        alloc = min(remaining * 0.20, 60.0)
        a_sa = simulated_annealing(a_pert, time_budget=alloc,
                                   verbose=False, seed=6000)
        obj_sa = compute_obj_normalized(a_sa)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  ** SA: obj={best_score:.8f} **")
        else:
            print(f"  SA: obj={obj_sa:.8f}")

        # Quick CD after SA
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_cd_sa = fast_coordinate_descent(a_sa, time_budget=min(30, remaining * 0.2),
                                              verbose=False, seed=6100)
            obj_cd_sa = compute_obj_normalized(a_cd_sa)
            if obj_cd_sa < best_score:
                best_score = obj_cd_sa
                best_seq = a_cd_sa.copy()
                print(f"  ** CD after SA: obj={best_score:.8f} **")

    # === Phase 7: Smoothed Gradient Descent ===
    print(f"\n=== Phase 7: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        srv = rng.normal(0, 0.03, len(best_seq))
        a_gd_init = best_seq * np.exp(srv)
        a_gd_init = np.maximum(a_gd_init, 0)
        s_gd = float(np.sum(a_gd_init))
        if s_gd > 1e-10:
            a_gd_init = a_gd_init / s_gd

        alloc = min(remaining * 0.15, 50.0)
        a_gd = accelerated_smoothed_descent(a_gd_init, time_budget=alloc,
                                            tau_init=20.0, tau_end=300.0,
                                            verbose=False)
        obj_gd = compute_obj_normalized(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  ** GD: obj={best_score:.8f} **")
        else:
            print(f"  GD: obj={obj_gd:.8f}")

    # === Phase 8: Final CD Polish ===
    print(f"\n=== Phase 8: Final CD Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.6, 200.0)
        a_final = fast_coordinate_descent(best_seq, time_budget=alloc,
                                          verbose=True, seed=seed + 9999)
        obj_final = compute_obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # === Phase 9: One more peak-directed pass ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_pd_final = peak_directed_optimization(best_seq, time_budget=min(remaining * 0.4, 40.0),
                                                verbose=False, seed=9999)
        obj_pd_final = compute_obj_normalized(a_pd_final)
        if obj_pd_final < best_score:
            best_score = obj_pd_final
            best_seq = a_pd_final.copy()
            print(f"  ** Final PeakDir: obj={best_score:.8f} **")
            # Quick CD
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_cd_final = fast_coordinate_descent(best_seq, time_budget=min(remaining * 0.5, 30.0),
                                                     verbose=False, seed=8888)
                obj_cd_final = compute_obj_normalized(a_cd_final)
                if obj_cd_final < best_score:
                    best_score = obj_cd_final
                    best_seq = a_cd_final.copy()
                    print(f"  ** Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    # Ensure all values are valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
