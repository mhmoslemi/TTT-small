from scipy.optimize import minimize as scipy_minimize
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ab(a, b):
    """Full linear convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_obj(a):

        return np.inf

    return A

            d = c
            fd = fc
            c = hi - phi * (hi - lo)
            fc = f_of_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + phi * (hi - lo)
            fd = f_of_t(d)
    
    t_opt = (lo + hi) / 2.0
    return t_opt, f_of_t(t_opt)


def lp_polish(a, time_budget, verbose=True):
    """LP polishing with exact line search and dual-guided updates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    warm_g = a.copy()

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp, dual_y = solve_lp(a, warm_g=warm_g)
        if g_lp is None:
            break

        warm_g = g_lp.copy()

        # Exact line search
        t_opt, f_opt = exact_line_search(best_a, g_lp)
        
        # Apply the step
        a_new = (1.0 - t_opt) * best_a + t_opt * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            if verbose and time.monotonic() - start < 60:
                elapsed = time.monotonic() - start
                print(f"    LP: obj={best_obj:.8f}, t={t_opt:.4f}, M_lp={M_lp:.6f}, elapsed={elapsed:.0f}s")
        else:
            # Try smaller step
            for frac in [0.5, 0.3, 0.1]:
                a_try = (1.0 - t_opt * frac) * best_a + t_opt * frac * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    stall = 0
                    break
            else:
                stall += 1
                if stall > 5:
                    break

    return best_a


def dual_guided_gradient(a, dual_y, time_budget, verbose=True):
    """Gradient descent using dual weights from LP as softmax weights."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Use dual weights to compute gradient
    # grad_j = 2 * sum_k dual_y[k] * a_{k-j}
    # This is 2 * conv(dual_y, a) but truncated
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(dual_y, N)
    A_f = fft(a, N)
    g_full = ifft(P * np.conj(A_f)).real
    grad = 2.0 * g_full[:n]

    # Gradient descent with adaptive step
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        # Recompute gradient with current a
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(dual_y, N)
        A_f = fft(a, N)
        g_full = ifft(P * np.conj(A_f)).real
        grad = 2.0 * g_full[:n]

        # Project gradient to null space of sum constraint
        grad_centered = grad - np.mean(grad)
        
        g_norm = float(np.linalg.norm(grad_centered))
        if g_norm < 1e-12:
            break

        # Adaptive step size
        lr = 0.1 / (g_norm + 1e-10)
        lr = min(lr, 1.0 / np.sqrt(n))

        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    return best_a


def high_tau_gd(a, time_budget, tau_start=100, tau_end=10000, verbose=True):
    """High-precision smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    beta = 0.9
    v = np.zeros(n)

    N = 1
    while N < 3 * n:
        N <<= 1

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.4

        # Nesterov momentum
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Smoothed gradient via softmax
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient: grad_j = 2 * sum_k p_k * y_{k-j}
        P = fft(p, N)
        A_f = fft(y, N)
        g_full = ifft(P * np.conj(A_f)).real
        g = 2.0 * g_full[:n]

        # Center gradient (remove component along all-ones)
        g = g - np.mean(g)

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-14:
            break

        lr = min(0.5 / (g_norm + 1e-10), 1.0 / np.sqrt(n))
        lr = max(lr, 1e-10)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        beta = min(0.98, beta + 0.0002)

        if it - best_it > 50000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it
            if it > 100000:
                break

        if verbose and it % 20000 == 0:
            print(f"    HTGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.5
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(6 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a


def log_param_optimize(a, time_budget, tau_min=200, tau_max=8000, verbose=True):
    """Optimize using log-parameterization with L-BFGS."""
    full_len = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    threshold = 1e-8
    support = np.where(a > threshold)[0]
    if len(support) < 2:
        support = np.where(a > a.max() * 1e-10)[0]
    if len(support) < 2:
        return a

    b0 = np.log(np.maximum(a[support], 1e-30))
    b0 = b0 - b0.mean()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    n_stages = 4
    tau_stages = [tau_min + (tau_max - tau_min) * (i / max(1, n_stages - 1)) ** 1.5 
                  for i in range(n_stages)]

    current_b = b0.copy()
    for stage_idx, tau in enumerate(tau_stages):
        if time.monotonic() - start > time_budget:
            break

        def obj_grad(b_vec):
            b_shifted = b_vec - b_vec.max()
            a_supp = np.exp(b_shifted)
            a_supp = a_supp / a_supp.sum()
            a_full = np.zeros(full_len)
            a_full[support] = a_supp

            conv = compute_conv(a_full)
            max_conv = float(np.max(conv))
            exp_vals = np.exp(tau * (conv - max_conv))
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp

            obj_val = max_conv + np.log(sum_exp) / tau

            N = 1
            while N < 3 * full_len:
                N <<= 1
            P = fft(p, N)
            A_full = fft(a_full, N)
            g_full = ifft(P * np.conj(A_full)).real
            grad_full = 2.0 * g_full[:full_len]

            grad_b = a_supp * (grad_full[support] - np.sum(a_supp * grad_full[support]))
            return obj_val, grad_b

        try:
            res = scipy_minimize(obj_grad, current_b, method='L-BFGS-B',
                                 jac=True, options={'maxiter': 300, 'gtol': 1e-12,
                                                    'ftol': 1e-15, 'maxfun': 1000})
            if res.x is not None:
                b_shifted = res.x - res.x.max()
                a_supp = np.exp(b_shifted)
                a_supp = a_supp / a_supp.sum()
                a_new = np.zeros(full_len)
                a_new[support] = a_supp
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    current_b = res.x.copy()
                    if verbose:
                        print(f"    LogOpt stage{stage_idx}: tau={tau:.0f}, obj={best_obj:.8f}")
        except Exception:
            pass

    return best_a


def support_shrink(a, time_budget, verbose=True):
    """Try zeroing small entries and re-optimizing."""
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    sorted_vals = np.sort(a)[::-1]
    for frac in [0.5, 0.7, 0.8, 0.9, 0.95]:
        if time.monotonic() - start > time_budget:
            break
        k = max(2, int(len(a) * frac))
        threshold = sorted_vals[k - 1] if k < len(sorted_vals) else 0
        support = np.where(a >= threshold)[0]
        if len(support) < 2:
            continue
        a_shrunk = np.zeros(len(a))
        a_shrunk[support] = a[support]
        a_shrunk = a_shrunk / a_shrunk.sum()

        alloc = min(20, time_budget - 5 - (time.monotonic() - start))
        if alloc > 5:
            a_shrunk = lp_polish(a_shrunk, time_budget=alloc, verbose=False)
            remaining = time_budget - (time.monotonic() - start)
            if remaining > 5:
                a_shrunk = high_tau_gd(a_shrunk, time_budget=min(10, remaining),
                                       tau_start=200, tau_end=2000, verbose=False)

        obj_shrunk = compute_obj(a_shrunk)
        if obj_shrunk < best_obj - 1e-12:
            best_obj = obj_shrunk
            best_a = a_shrunk.copy()
            if verbose:
                print(f"    Shrink frac={frac}: obj={best_obj:.8f}, support={len(support)}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines LP polishing with exact line search,
    dual-guided gradient descent, high-tau smoothed GD, log-param optimization,
    support shrinkage, multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting optimization with n={n_best}, obj={best_score:.8f}")

    # === Phase 1: LP Polishing with Exact Line Search (250s) ===
    print(f"\n=== Phase 1: LP + Exact Line Search ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.22, 250.0)
    if p1_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-12:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 improved: obj={best_score:.8f}")
        else:
            print(f"  Phase1: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: High-Tau GD (150s) ===
    print(f"\n=== Phase 2: High-Tau GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.14, 150.0)
    if p2_budget > 20:
        a_gd = high_tau_gd(best_seq, time_budget=p2_budget,
                           tau_start=200, tau_end=8000, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score - 1e-12:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase2 improved: obj={best_score:.8f}")
        else:
            print(f"  Phase2: obj={obj_gd:.8f}")

    # === Phase 3: LP again after GD (100s) ===
    print(f"\n=== Phase 3: LP Refinement ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 100.0)
    if p3_budget > 20:
        a_lp2 = lp_polish(best_seq, time_budget=p3_budget, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score - 1e-12:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase3 improved: obj={best_score:.8f}")
        else:
            print(f"  Phase3: obj={obj_lp2:.8f}")

    # === Phase 4: Log-Param Optimization (80s) ===
    print(f"\n=== Phase 4: Log-Param Optimization ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 80.0)
    if p4_budget > 15:
        a_log = log_param_optimize(best_seq, time_budget=p4_budget,
                                   tau_min=500, tau_max=8000, verbose=True)
        obj_log = compute_obj(a_log)
        if obj_log < best_score - 1e-12:
            best_score = obj_log
            best_seq = a_log.copy()
            print(f"  Phase4 improved: obj={best_score:.8f}")
        else:
            print(f"  Phase4: obj={obj_log:.8f}")

    # === Phase 5: Support Shrink (60s) ===
    print(f"\n=== Phase 5: Support Shrink ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.06, 60.0)
    if p5_budget > 15:
        a_shrunk = support_shrink(best_seq, time_budget=p5_budget, verbose=True)
        obj_shrunk = compute_obj(a_shrunk)
        if obj_shrunk < best_score - 1e-12:
            best_score = obj_shrunk
            best_seq = a_shrunk.copy()
            print(f"  Phase5 improved: obj={best_score:.8f}")
        else:
            print(f"  Phase5: obj={obj_shrunk:.8f}")

    # === Phase 6: Multi-n Exploration (250s) ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p6_budget = min(remaining * 0.25, 250.0)
    n_best = len(best_seq)

    n_options = sorted(set([
        max(200, n_best - 200), max(200, n_best - 100), n_best,
        n_best + 100, n_best + 200, n_best + 400,
        n_best + 800, n_best + 1500,
        400, 600, 1000, 1200, 1600, 2000, 3000
    )))
    n_options = [n for n in n_options if 200 <= n <= 5000 and n != n_best]
    n_options = n_options[:10]
    print(f"  Trying n values: {n_options}")

    per_n_budget = p6_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_n_budget, remaining - 20)
        if alloc < 20:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try > 0:
            a_try = a_try / s_try

        # Quick optimization
        if alloc > 15:
            a_try = lp_polish(a_try, time_budget=alloc * 0.5, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 15:
            gd_alloc = min(alloc * 0.35, remaining - 10)
            if gd_alloc > 5:
                a_try = high_tau_gd(a_try, time_budget=gd_alloc,
                                    tau_start=100, tau_end=3000, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_try = lp_polish(a_try, time_budget=min(alloc * 0.15, remaining - 5), verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score - 1e-12:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Restarts (120s) ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p7_budget = min(remaining * 0.12, 120.0)
    n_current = len(best_seq)

    n_restarts = max(1, int(p7_budget / 40))
    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(45, remaining - 10)
        if alloc < 15:
            break

        mode = r % 7
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r
            a_r = lp_polish(a_r, time_budget=alloc * 0.5, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_r = high_tau_gd(a_r, time_budget=min(alloc * 0.35, remaining - 5),
                                  tau_start=100, tau_end=2000, verbose=False)
            obj_r = compute_obj(a_r)
            if obj_r < best_score - 1e-12:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final Intensive Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s")

    # LP
    if remaining > 20:
        alloc = min(remaining * 0.3, 80.0)
        a_f = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score - 1e-12:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f}")

    # High-tau GD with very high tau
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining *
