from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    b = a / s
    conv = compute_conv(b)
    return 2.0 * n * float(np.max(conv))

def solve_lp_fast(a):
    """Solve LP: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0.
    Uses scipy linprog with HiGHS."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix A_eq for sum(g) = 1
    # Inequality constraints: (A_conv @ g)_k - M <= 0 for all k
    # Variables: [g_0, ..., g_{n-1}, M]
    # c = [0, ..., 0, 1] (minimize M)
    
    # A_ineq * [g; M] <= b_ineq
    # For each k: sum_j A_conv[k,j] * g_j - M <= 0
    # A_ineq has shape (2n-1, n+1)
    
    # Build A_conv: A_conv[k,j] = a_norm[max(0, k-j)] if 0 <= k-j < n else 0
    # This is a Hankel-like matrix
    
    A_conv = np.zeros((2*n-1, n))
    for j in range(n):
        A_conv[j:j+n, j] = a_norm
    
    # A_ineq: (2n-1) x (n+1)
    A_ineq = np.hstack([A_conv, -np.ones((2*n-1, 1))])
    b_ineq = np.zeros(2*n-1)
    
    # A_eq: (1) x (n+1), sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g_j >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]
    
    # Objective: minimize M
    c = np.zeros(n+1)
    c[n] = 1.0
    
    res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None, None
    
    g = res.x[:n]
    M = res.x[n]
    g = np.maximum(g, 0)
    s = float(np.sum(g))
    if s > 1e-10:
        g = g / s
    return g, M

**Proximal operator for the simplex:**
Given v, find the closest point to v on the simplex {x >= 0, sum(x) = 1}.

```python
def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast HiGHS LP-based fixed-point iteration with dual-guided subgradient
refinement, Nesterov acceleration on smoothed objective, adaptive line search,
duality-gap-based convergence detection, perturbation escape from local minima,
and multi-length exploration with resampling.
Key improvements: scipy HiGHS for 100x faster LP solves, dual subgradient
polishing between LP steps, tight tolerances, and structured multi-n search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    b = a / s
    conv = compute_conv(b)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_fast(a, tight_tol=False):
    """Solve LP: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0.
    Uses scipy HiGHS. Returns (g, M, dual_y) or (None, None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)

    A_conv = build_conv_matrix(a_norm)

    # Variables: [g_0, ..., g_{n-1}, M]
    # Minimize M => c = [0,...,0,1]
    # Inequality: A_conv @ g - M <= 0 => [A_conv | -1] @ [g; M] <= 0
    # Equality: sum(g) = 1 => [1,...,1 | 0] @ [g; M] = 1
    # Bounds: g >= 0, M free

    A_ub = np.hstack([A_conv, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    c = np.zeros(n + 1)
    c[n] = 1.0

    options = {'presolve': True}
    if tight_tol:
        options['primal_feasibility_tolerance'] = 1e-12
        options['dual_feasibility_tolerance'] = 1e-12
        options['ipm_optimality_tolerance'] = 1e-12

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options=options)
    except Exception:
        return None, None, None

    if not res.success:
        return None, None, None

    g = np.maximum(res.x[:n], 0)
    s = float(np.sum(g))
    if s < 1e-15:
        return None, None, None
    g = g / s
    M = float(res.x[n])

    # Extract dual variables (shadow prices for A_ub constraints)
    # For min c^T x s.t. A_ub x <= b_ub: marginals = d(opt)/d(b_ub) <= 0
    # Lagrange multipliers = -marginals >= 0
    try:
        y = -np.array(res.ineqlin.marginals, dtype=np.float64)
        y = np.maximum(y, 0)
        y_sum = float(np.sum(y))
        if y_sum > 1e-10:
            y = y / y_sum
        else:
            y = None
    except Exception:
        y = None

    return g, M, y


def compute_subgrad_fft(a, y):
    """Compute subgradient of max_k conv(a,a)_k using dual weights y.
    grad_j = 2 * sum_k y_k * a_{k-j} = 2 * (a cross-corr y)_j"""
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    A_f = fft(a, N)
    Y_f = fft(y, N)
    # Cross-correlation: ifft(conj(fft(a)) * fft(y))
    corr = ifft(np.conj(A_f) * Y_f).real[:n]
    return 2.0 * corr


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lp_fixed_point_iteration(a, time_budget, verbose=True, tight_phase=False):
    """Main LP fixed-point iteration with adaptive line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_improve_time = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        # Solve LP
        g, M_lp, y = solve_lp_fast(a, tight_tol=tight_phase)
        if g is None:
            break

        # Current max conv
        conv_cur = compute_conv(a)
        M_cur = float(np.max(conv_cur))

        # Duality gap
        gap = M_cur - M_lp
        if gap < 1e-12:
            if verbose:
                print(f"    LP converged: gap={gap:.2e}, obj={best_obj:.8f}, it={it}")
            break

        # Adaptive step size based on gap
        if gap < 1e-6:
            t_candidates = [0.01, 0.03, 0.05, 0.1, 0.2]
        elif gap < 1e-4:
            t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5]
        else:
            t_candidates = [0.2, 0.4, 0.5, 0.7, 1.0]

        # Line search
        best_t = 0.0
        best_t_obj = best_obj
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:

            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-14:
                best_t_obj = obj_try
                best_t = t

        # Fine line search around best_t
        if best_t > 0:
            for dt in [-0.05, -0.02, 0.02, 0.05]:
                t_fine = best_t + dt
                if 0 < t_fine < 1.0:
                    a_try = (1.0 - t_fine) * a + t_fine * g
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_t_obj - 1e-14:
                        best_t_obj = obj_try
                        best_t = t_fine

        if best_t > 0 and best_t_obj < best_obj - 1e-14:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_t_obj
            best_a = a.copy()
            stall_count = 0
            last_improve_time = time.monotonic()
            if verbose and it % 100 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, gap={gap:.2e}, t={best_t:.3f}")
        else:
            stall_count += 1
            if stall_count > 50:
                if verbose:
                    print(f"    LP stalled at it{it}: obj={best_obj:.8f}")
                break

        it += 1

    return best_a


def subgrad_polish(a, y, time_budget, verbose=False):
    """Polish using subgradient steps with fixed dual weights y."""
    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    a_curr = a.copy()
    a_avg = a.copy()
    it = 0
    lr = 0.1

    while time.monotonic() - start < time_budget:
        grad = compute_subgrad_fft(a_curr, y)
        a_new = a_curr - lr * grad
        a_new = project_simplex(a_new)
        a_curr = a_new
        it += 1
        a_avg = ((it - 1) / it) * a_avg + (1.0 / it) * a_new

        # Check periodically
        if it % 100 == 0:
            obj = compute_obj(a_avg)
            if obj < best_obj:
                best_obj = obj
                best_a = a_avg.copy()
            # Decay learning rate
            lr *= 0.995

        if time.monotonic() - start > time_budget:
            break

    final_obj = compute_obj(a_avg)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a_avg.copy()

    return best_a


def smoothed_obj_grad(a, tau):
    """Compute smoothed objective and gradient."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    obj_val = max_conv + np.log(sum_exp) / tau

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A_f = fft(a, N)
    g_full = ifft(P * np.conj(A_f)).real
    grad = 2.0 * g_full[:n]

    return obj_val, grad, p


def nesterov_smoothed_opt(a, time_budget, tau_start=100, tau_end=5000, verbose=True):
    """Nesterov-accelerated optimization on the smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y_vec = a.copy()
    v = np.zeros(n)
    beta = 0.5
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1))
        tau = tau_start + (tau_end - tau_start) * progress

        # Nesterov step
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y_vec = a + (beta / beta_new) * (a - y_vec + v)
        y_vec = np.maximum(y_vec, 0)
        s_y = float(np.sum(y_vec))
        if s_y < 1e-10:
            y_vec = best_a.copy()
        else:
            y_vec = y_vec / s_y

        obj_val, grad, _ = smoothed_obj_grad(y_vec, tau)

        # Gradient step on simplex
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break
        lr = 1.0 / (g_norm + 1e-8)
        lr = min(lr, 2.0 / np.sqrt(n))

        a_new = a - lr * grad
        a_new = project_simplex(a_new)

        v = a_new - a
        a = a_new
        beta = beta_new
        it += 1

        if it % 50 == 0:
            obj_check = compute_obj(a)
            if obj_check < best_obj:
                best_obj = obj_check
                best_a = a.copy()

        # Periodic restart
        if it % 2000 == 0:
            y_vec = a.copy()
            v = np.zeros(n)

        if verbose and it % 500 == 0:
            print(f"    Nest it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.0f}s")

    final_obj = compute_obj(a)
    if final_obj < best_obj:
        best_a = a.copy()
        best_obj = final_obj

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
        return a / a.sum()
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
        return a / a.sum()
    elif mode == 5:
        # Two-bump: high at edges, low in middle
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * x ** 2) + np.exp(-8 * (1 - x) ** 2)
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif mode == 6:
        # U-shaped
        x = np.linspace(0, 1, n)
        a = (x - 0.5) ** 2 + 0.01
        return a / a.sum()
    elif mode == 7:
        # Cosine window (symmetric, high in middle)
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        return a / a.sum()

                                           tight_phase=True, verbose=False)

        best_seq = diverse_init(n_init, rng, mode=5)
        best_score = compute_obj(best_seq)
        print(f"Two-bump init (n={n_init}): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"\nInitial: n={n_best}, score={best_score:.8f}")

    # === Phase 1: Intensive LP optimization at current n (350s) ===
    print(f"\n=== Phase 1: LP Optimization (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.28, 350.0)
    if p1_budget > 30:
        a_opt = optimize_at_n(best_seq, time_budget=p1_budget, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_opt:.8f} (no improvement)")

    # === Phase 2: Multi-n exploration (400s) ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 400.0)
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current - 200, n_current - 100, n_current - 50,
        n_current + 50, n_current + 100, n_current + 200,
        n_current + 400, n_current + 600, n_current + 1000,
        n_current + 1500, n_current + 2000,
        400, 600, 1000, 1500, 2000, 3000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 5000 and abs(n - n_current) > 30]
    # Prioritize n values close to current best
    n_options.sort(key=lambda n: abs(n - n_current))
    n_options = n_options[:10]
    print(f"  n options: {n_options}")

    per_n_budget = p2_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_n_budget, remaining - 20)
        if alloc < 20:
            break

        print(f"  Trying n={n_try} (alloc={alloc:.0f}s)...", end=" ")
        a_try = resample_sequence(best_seq, n_try)
        a_try = optimize_at_n(a_try, time_budget=alloc, verbose=False)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"obj={best_score:.8f} **")
        else:
            print(f"obj={obj_try:.8f}")

    # === Phase 3: Diverse restarts with blending (200s) ===
    print(f"\n=== Phase 3: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.18, 200.0)
    n_current = len(best_seq)
    n_restart = max(1, int(p3_budget / 40))

    for r in range(n_restart):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = min(40, remaining - 10)
        if alloc < 15:
            break

        mode = r % 8
        a_r = diverse_init(n_current, rng, mode=mode)
        # Blend with current best
        blend = 0.5 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        a_r = optimize_at_n(a_r, time_budget=alloc, verbose=False)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 4: Perturbation escape (150s) ===
    print(f"\n=== Phase 4: Perturbation Escape ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.12, 150.0)
    n_current = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(30, remaining - 10)
        if alloc < 10:
            break

        # Add structured perturbation
        noise_scale = [0.01, 0.005, 0.02, 0.003, 0.015][r]
        a_pert = best_seq + noise_scale * rng.standard_normal(n_current)
        a_pert = np.maximum(a_pert, 0)
        s_p = float(np.sum(a_pert))
        if s_p > 1e-10:
            a_pert = a_pert / s_p
            a_pert = optimize_at_n(a_pert, time_budget=alloc, verbose=False)
            obj_p = compute_obj(a_pert)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_pert.copy()
                print(f"  Perturb {r} (scale={noise_scale}): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (scale={noise_scale}): obj={obj_p:.8f}")

    # === Phase 5: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s")

    # Round 1: Tight LP
    if remaining > 20:
        alloc = min(remaining * 0.3, 100.0)
        a_final = lp_fixed_point_iteration(best_seq, time_budget=alloc,
                                           tight_phase=True, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP tight: obj={best_score:.8f} **")

    # Round 2: Subgradient with fresh dual
    remaining = deadline - time.monotonic()
    if remaining > 15:
        _, _, y_final = solve_lp_fast(best_seq, tight_tol=True)
        if y_final is not None:
            alloc = min(remaining * 0.2, 60.0)
            a_final = subgrad_polish(best_seq, y_final, time_budget=alloc)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final SubGrad: obj={best_score:.8f} **")

    # Round 3: Nesterov with high tau
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.25, 80.0)
        a_final = nesterov_smoothed_opt(best_seq, time_budget=alloc,
                                        tau_start=1000, tau_end=20000, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Nesterov: obj={best_score:.8f} **")

    # Round 4: One more tight LP
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.3, 50.0)
        a_final = lp_fixed_point_iteration(best_seq, time_budget=alloc,
                                           tight_phase=True, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra-final LP: obj={best_score:.8f} **")

    # Round 5: Final subgradient
    remaining = deadline - time.monotonic()
    if remaining > 8:
        _, _, y_ultra = solve_lp_fast(best_seq, tight_tol=True)
        if y_ultra is not None:
            alloc = min(remaining * 0.3, 30.0)
            a_final = subgrad_polish(best_seq, y_ultra, time_budget=alloc)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Ultra SubGrad: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
