"""
Hybrid optimizer combining 2D block coordinate descent, LP polishing with line search,
targeted peak-reduction restarts, and multi-n exploration.
Key innovation over previous approach: pairwise coordinate updates capture interactions
between coordinates that single-coordinate CD misses, enabling escape from local minima.
Additionally uses targeted perturbation (toward reducing the peak lag) instead of random noise.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 with normalized a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    return 2.0 * n * float(np.max(compute_conv(a_n)))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_limit=60):
    """Solve LP: min M s.t. conv(a_norm,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_n = a / s
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except Exception:
            continue
    return None, None


def _f_1d_coord(a, conv, n, i, x, S):
    """Compute objective value if we set a[i]=x, given current a, conv, S."""
    v = a[i]
    if abs(x - v) < 1e-15:
        max_val = float(np.max(conv))
        return 2.0 * n * max_val / (S * S)

    k_min = max(0, i)
    k_max = min(2 * n - 2, i + n - 1)
    ks = np.arange(k_min, k_max + 1)
    a_ki = a[ks - i]
    c_k = conv[ks]

    mask = (ks != 2 * i)
    if np.any(mask):
        lin_vals = c_k[mask] + 2.0 * a_ki[mask] * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0

    k_quad = 2 * i
    if 0 <= k_quad <= 2 * n - 2:
        quad_val = c_k[k_quad] + x * x - v * v
    else:
        quad_val = 0.0

    max_val = max(max_lin, quad_val)
    S_new = S - v + x
    if S_new < 1e-15:
        return 1e30
    return 2.0 * n * max_val / (S_new * S_new)


def _golden_1d(a, conv, n, i, S, x_init, scale):
    """Golden section search for coordinate i."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    v = a[i]

    # Quick scan to find good bracket
    candidates = [0.0, v * 0.3, v * 0.7, v, v * 1.3, v * 1.7,
                  v + scale * 0.5, v + scale, v + 2 * scale]
    candidates = [max(0.0, float(c)) for c in candidates]

    best_x = v
    best_f = _f_1d_coord(a, conv, n, i, v, S)
    for x in candidates:
        fx = _f_1d_coord(a, conv, n, i, x, S)
        if fx < best_f:
            best_f = fx
            best_x = x

    # Golden section refinement
    x_lo = max(0.0, best_x - scale)
    x_hi = best_x + scale * 2.0
    if x_hi - x_lo < 1e-14:
        x_hi = x_lo + 1e-10

    c_pt = x_hi - gr * (x_hi - x_lo)
    d_pt = x_lo + gr * (x_hi - x_lo)
    fc = _f_1d_coord(a, conv, n, i, c_pt, S)
    fd = _f_1d_coord(a, conv, n, i, d_pt, S)

    for _ in range(40):
        if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
            break
        if fc < fd:
            x_hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = x_hi - gr * (x_hi - x_lo)
            fc = _f_1d_coord(a, conv, n, i, c_pt, S)
        else:
            x_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = x_lo + gr * (x_hi - x_lo)
            fd = _f_1d_coord(a, conv, n, i, d_pt, S)

    return max(0.0, (x_lo + x_hi) / 2.0)


def _update_conv_one(a, conv, i, old_val, new_val, n):
    """Update conv when a[i] changes from old_val to new_val."""
    delta = new_val - old_val
    if abs(delta) < 1e-16:
        return
    k_min = max(0, i)
    k_max = min(2 * n - 2, i + n - 1)
    ks = np.arange(k_min, k_max + 1)
    a_ki = a[ks - i].copy()
    # For k = 2i, the term is a[i]^2, not 2*a[i]*a[i]
    mask = (ks != 2 * i)
    if np.any(mask):
        conv[ks[mask]] += 2.0 * delta * a_ki[mask]
    if 0 <= 2 * i <= 2 * n - 2:
        conv[2 * i] += new_val * new_val - old_val * old_val


def block_coordinate_descent(a, time_budget=300.0, verbose=False, seed=42,
                             pair_alpha=0.3):
    """
    Block coordinate descent with random pairs.
    Each pair (i,j) is updated by alternating 1D minimization (3 sweeps).
    With probability pair_alpha, use random pairs; otherwise use ordered sweep.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        S = float(np.sum(a))
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed

        # Determine pair list
        if sweep % 4 < 2:
            # Ordered sweep with pairs (i, i+n//2) for spread
            pairs = []
            for i in range(n):
                j = (i + n // 2) % n
                if j != i:
                    pairs.append((i, j))
            rng.shuffle(pairs)
        else:
            # Random pairs
            pairs = []
            for _ in range(n // 2):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if j != i:
                    pairs.append((i, j))

        for (i, j) in pairs:
            if time.monotonic() - start >= time_budget:
                break

            scale = max(a[i], a[j], S / n, 1e-6)

            # Alternate 1D minimization 3 times
            for inner in range(3):
                # Optimize i
                xi = _golden_1d(a, conv, n, i, S, a[i], scale)
                if abs(xi - a[i]) > 1e-14:
                    _update_conv_one(a, conv, i, a[i], xi, n)
                    a[i] = xi
                    S = float(np.sum(a))

                # Optimize j
                xj = _golden_1d(a, conv, n, j, S, a[j], scale)
                if abs(xj - a[j]) > 1e-14:
                    _update_conv_one(a, conv, j, a[j], xj, n)
                    a[j] = xj
                    S = float(np.sum(a))

        # Track best
        if sweep % 2 == 0:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 20 == 0:
            elapsed2 = time.monotonic() - start
            print(f"    BCD sweep {sweep}: obj={best_obj:.8f}, t={elapsed2:.1f}s")

    return best_a

def lp_polish_with_linesearch(a, time_budget=100.0, verbose=False, seed=42):
    """LP polishing with proper line search over blending factor."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)

        g_lp, M_lp = solve_lp_minpeak(a, time_limit=min(remaining * 0.5, 45))
        if g_lp is None:
            break

        # Line search over t
        best_t_obj = best_obj
        best_t = 0.0
        best_a_ls = best_a.copy()

        # Coarse search
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_ls = a_try.copy()

        # Fine search around best_t
        if best_t > 0 and best_t < 1.0:
            for t in np.linspace(max(0, best_t - 0.1), min(1, best_t + 0.1), 11):
                a_try = (1.0 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_t = t
                    best_a_ls = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_ls.copy()
            a = best_a.copy()
            stall = 0
        else:
            stall += 1

                peak_lag = int(np.argmax(a_conv))
                # Find positions contributing to peak

def targeted_perturbation(a, rng, strength=0.03):
    """Perturb toward reducing the peak autocorrelation."""
    n = len(a)
    a = a.copy()
    conv = compute_conv(a)
    peak_lag = int(np.argmax(conv))

    # Find positions contributing to peak
    contrib = np.zeros(n)
    for i in range(n):
        j = peak_lag - i
        if 0 <= j < n:
            contrib[i] = a[i] * a[j]

    # Reduce high contributors, increase low contributors
    threshold = np.percentile(contrib, 70)
    high_mask = contrib > threshold
    low_mask = contrib < np.percentile(contrib, 30)

    total_reduce = 0.0
    for i in np.where(high_mask)[0]:
        reduction = strength * a[i]
        a[i] -= reduction
        total_reduce += reduction

    a = np.maximum(a, 0)

    # Redistribute to low contributors
    if total_reduce > 0 and np.any(low_mask):
        low_idx = np.where(low_mask)[0]
        for i in low_idx:
            a[i] += total_reduce / len(low_idx)

    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a / a.sum()
    elif mode == 3:
        # Chirp
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(np.pi * x * (1 + 4 * x))) + 0.1
        a = a / a.sum()
    elif mode == 4:
        # Two bumps
        centers = [n // 3, 2 * n // 3]
        width = n // 6
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * width ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Near-binary
        a = (rng.random(n) > 0.5).astype(float)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Sinusoidal
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * rng.uniform(2, 8) * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                print(f"  LP round {rnd}: obj={obj_lp:.8f}")

            # Short CD after LP
            remaining = deadline - time.monotonic()
            if remaining > 30 and obj_lp >= best_score - 1e-9:
                alloc_cd = min(30, remaining * 0.15)
                if alloc_cd > 10:
                    a_cd = single_cd_pass(a_lp, time_budget=alloc_cd, seed=2000 + rnd)
                    obj_cd = compute_obj_norm(a_cd)
                    if obj_cd < best_score:
                        best_score = obj_cd
                        best_seq = a_cd.copy()
                        print(f"  CD after LP {rnd}: obj={best_score:.8f} **")

    # === Phase 3: Targeted perturbation + BCD restart ===
    print(f"\n=== Phase 3: Targeted Restart (budget ~150s) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 150.0)
    if p3_budget > 30:
        for attempt in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = p3_budget / (3 - attempt)
            alloc = min(alloc, remaining * 0.7)
            if alloc < 15:
                break

            a_pert = targeted_perturbation(best_seq, rng,
                                           strength=0.02 + 0.015 * attempt)
            a_pert = block_coordinate_descent(a_pert, time_budget=alloc,
                                              verbose=False, seed=3000 + attempt * 100)
            obj_pert = compute_obj_norm(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Targeted restart {attempt}: obj={best_score:.8f} **")
            else:
                print(f"  Targeted restart {attempt}: obj={obj_pert:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current - 50, n_current + 50, n_current + 100, n_current - 150,
        n_current + 200, n_current - 200, 600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_current) > 30]
    n_options = n_options[:6]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.2, 70.0)
        if alloc > 15:
            a_try = block_coordinate_descent(a_try, time_budget=alloc,
                                             verbose=False, seed=n_try)
        obj_try = compute_obj_norm(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        mode = (r * 2) % 8
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 60.0)
        if alloc > 15:
            a_r = block_coordinate_descent(a_r, time_budget=alloc,
                                           verbose=False, seed=5000 + r * 100)
            obj_r = compute_obj_norm(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 6: Final refinement ===
    print(f"\n=== Phase 6: Final Refinement ===")
    remaining = deadline - time.monotonic()

    if remaining > 60:
        # BCD
        alloc1 = min(remaining * 0.3, 120.0)
        if alloc1 > 20:
            a_final = block_coordinate_descent(best_seq, time_budget=alloc1,
                                               verbose=False, seed=seed + 7777)
            obj_final = compute_obj_norm(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final BCD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 40:
        # LP
        alloc2 = min(remaining * 0.4, 80.0)
        if alloc2 > 20:
            a_lp_f = lp_polish_with_linesearch(best_seq, time_budget=alloc2,
                                               verbose=False, seed=9999)
            obj_lp_f = compute_obj_norm(a_lp_f)
            if obj_lp_f < best_score:
                best_score = obj_lp_f
                best_seq = a_lp_f.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # One more CD pass
        alloc3 = min(remaining * 0.5, 60.0)
        if alloc3 > 10:
            a_cd_f = single_cd_pass(best_seq, time_budget=alloc3, seed=12345)
            obj_cd_f = compute_obj_norm(a_cd_f)
            if obj_cd_f < best_score:
                best_score = obj_cd_f
                best_seq = a_cd_f.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
