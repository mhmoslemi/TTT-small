import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def compute_conv_fft(a, b=None):
    n = len(a)
    if b is None:
        b = a
    m = len(b)
    if n == 0 or m == 0:
        return np.array([0.0])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    if b is a:
        result = irfft(fa * fa, size)[:n + m - 1]
    else:
        fb = rfft(b, size)
        result = irfft(fa * fb, size)[:n + m - 1]
    return result

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv_fft(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def convolve_matrix(a, g):
    """Compute conv(a, g) efficiently using FFT."""
    return compute_conv_fft(a, g)

def solve_lp_best_response(a, active_indices=None, max_constraints=500):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns g_opt or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix T where T[k,i] = a_norm[k-i] if 0<=k-i<n
    # T is (2n-1) x n
    # But for large n, this might be too big. Use only active constraints.
    
    if active_indices is None:
        # Compute current conv to find active constraints
        conv_current = compute_conv_fft(a_norm, a_norm)
        max_val = float(np.max(conv_current))
        # Keep constraints within 95% of max
        threshold = 0.95 * max_val
        active_indices = np.where(conv_current >= threshold)[0]
        if len(active_indices) > max_constraints:
            # Keep only the top max_constraints
            top_idx = np.argsort(conv_current)[::-1][:max_constraints]
            active_indices = top_idx
    
    n_constraints = len(active_indices)
    T = np.zeros((n_constraints, n))
    for idx, k in enumerate(active_indices):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                T[idx, i] = a_norm[j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [T @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid approach: LP-based alternating minimization for coarse optimization,
followed by FFT-accelerated Nesterov gradient descent with log-sum-exp smoothing
for fine polishing, combined with multi-restart exploration and adaptive length search.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a, b=None):
    """Full linear convolution via FFT."""
    n = len(a)
    if b is None:
        b = a
    m = len(b)
    if n == 0 or m == 0:
        return np.array([0.0])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_obj_normalized(a):
    """Compute objective assuming a is already normalized (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a, indices):
    """Build constraint matrix rows for given indices."""
    n = len(a)
    T = np.zeros((len(indices), n))
    for idx, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            T[idx, i] = a[k - i]
    return T


def solve_lp(a, target_constraints=400, solver_list=None):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for active k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_s = np.sum(a)
    if a_s < 1e-10:
        return None, None
    a_norm = a / a_s
    
    # Find active constraints
    conv_self = compute_conv(a_norm)
    max_val = float(np.max(conv_self))
    
    # Select constraints: those above 90% of max, plus a spread of others
    threshold = 0.90 * max_val
    active_mask = conv_self >= threshold
    indices = np.where(active_mask)[0]
    
    if len(indices) > target_constraints:
        # Prioritize the highest ones
        order = np.argsort(conv_self)[::-1]
        indices = order[:target_constraints]
    elif len(indices) < 50:
        # Add some random constraints for diversity
        all_idx = np.arange(len(conv_self))
        remaining = all_idx[~active_mask]
        if len(remaining) > 0:
            extra = np.random.default_rng(42).choice(remaining, size=min(50, len(remaining)))
            indices = np.concatenate([indices, extra])
    
    T = build_conv_matrix(a_norm, indices)
    
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.ECOS, cp.SCS]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [T @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.asarray(g.value).flatten()
                g_val = np.maximum(g_val, 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    return g_val / s_g, float(M.value)
        except Exception:
            continue
    
    return None, None

def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = float(np.sum(w))
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def nesterov_polish(a, time_budget, tau_start=50.0, tau_end=5000.0, verbose=False):
    """
    Nesterov accelerated gradient descent on the smoothed objective.
    Works on the simplex (sum a = 1, a >= 0).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 1.0 / np.sqrt(n)
    v = np.zeros(n)
    tau = tau_start
    patience = 0
    max_patience = 5000
    
    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = a + beta * v
        y = project_simplex(y)
        
        # Compute gradient
        g = smoothed_max_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))
        
        if g_norm < 1e-12:
            break
        
        # Adaptive step size
        step = lr / (g_norm + 1e-10)
        step = min(max(step, 0.01 * lr), 5.0 * lr)
        
        # Update
        v_new = beta * v - step * g
        a_new = y + v_new
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        v = v_new
        beta = min(beta * 1.0001, 0.999)
        tau = min(tau * 1.05, tau_end)
        lr *= 0.9999
        
        if patience > max_patience:
            # Reset and restart from best
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            beta = 0.9
            tau = tau_start
            patience = 0
    
    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 8
    for k in range(1, n_modes + 1):

        amp = strength * rng.uniform(0.3, 1.0) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new
