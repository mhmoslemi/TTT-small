"""
Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast dual LP + subgradient/FISTA hybrid approach with multi-n search.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time
from scipy.ndimage import uniform_filter1d


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_dual_lp(a):
    """
    Solve dual LP: max lambda s.t. sum(p)=1, (A^T p)_j >= lambda, p>=0.
    Returns (lambda_star, p_star) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)  # normalize
    
    # Variables: [p_0, ..., p_{2n-2}, lambda]  total 2n variables
    num_p = 2 * n - 1
    num_vars = num_p + 1  # +1 for lambda
    lambda_idx = num_p
    
    # Objective: minimize -lambda
    c = np.zeros(num_vars)
    c[lambda_idx] = -1.0
    
    # Inequality constraints: A_ub @ vars <= b_ub
    # n constraints: -sum_{m=0}^{n-1} a[m]*p[j+m] + lambda <= 0
    # 2 constraints: sum(p) <= 1, -sum(p) <= -1
    
    # Build the constraint matrix
    # For efficiency, use the Toeplitz structure
    A_ub = np.zeros((n + 2, num_vars))
    b_ub = np.zeros(n + 2)
    
    # Fill the n convolution constraints
    # Row j: -a[0]*p[j] - a[1]*p[j+1] - ... - a[n-1]*p[j+n-1] + lambda <= 0
    for j in range(n):
        for m in range(n):
            p_idx = j + m
            if p_idx < num_p:
                A_ub[j, p_idx] = -a[m]
        A_ub[j, lambda_idx] = 1.0
        b_ub[j] = 0.0
    
    # sum(p) <= 1
    A_ub[n, :num_p] = 1.0
    b_ub[n] = 1.0
    
    # -sum(p) <= -1
    A_ub[n+1, :num_p] = -1.0
    b_ub[n+1] = -1.0
    
    # Bounds: p_k >= 0, lambda free
    bounds = [(0, None)] * num_p + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            p_star = res.x[:num_p]
            lambda_star = res.x[lambda_idx]
            p_star = np.maximum(p_star, 0)
            p_sum = p_star.sum()
            if p_sum > 1e-15:
                p_star /= p_sum
            return float(lambda_star), p_star
    except Exception as e:
        pass
    
    return None, None


def solve_primal_lp(a):
    """
    Solve primal LP: min M s.t. (a*x)_k <= M, sum(x)=1, x>=0.
    Returns (x_star, M_star) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Variables: [x_0, ..., x_{n-1}, M]
    num_vars = n + 1
    M_idx = n
    
    # Objective: minimize M
    c = np.zeros(num_vars)
    c[M_idx] = 1.0

    except Exception:
        pass

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x) = target_sum, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fista_smooth(a, time_budget=100.0, tau_init=30.0, tau_end=500.0, verbose=False):
    """
    FISTA on the smoothed objective on the simplex.
    Minimizes f_smooth(a, tau) = (1/tau)*log(sum exp(tau * conv_k))
    subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    it = 0
    best_it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.2
        
        y = a + beta * v
        y = project_simplex(y)
        
        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))
        
        # Adaptive LR
        if g_norm > 1e-15:
            lr = max(0.5 * base_lr, min(2.0 * base_lr, base_lr / (g_norm / base_lr + 1)))
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 6):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast scipy linprog LP-driven iteration + FISTA smoothing + systematic n exploration.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def precompute_conv_structure(n):
    """Precompute the mask and indices for the convolution matrix."""
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    mask = (d >= 0) & (d < n)
    d_vals = d[mask]
    return mask, d_vals


def build_A(a, mask, d_vals):
    """Build convolution matrix using precomputed structure."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    A[mask] = a[d_vals]
    return A


def solve_primal_lp(a, mask, d_vals):
    """
    Solve primal LP: min M s.t. (A@x)[k] <= M, sum(x)=1, x>=0.
    Returns (x_star, M_star) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    
    A = build_A(a, mask, d_vals)
    num_constraints = 2 * n - 1
    num_vars = n + 1
    M_idx = n
    
    # Objective: minimize M
    c = np.zeros(num_vars)
    c[M_idx] = 1.0
    
    # Inequality: A@x - M <= 0
    A_ub = np.hstack([A, -np.ones((num_constraints, 1))])
    b_ub = np.zeros(num_constraints)
    
    # Equality: sum(x) = 1
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: x >= 0, M >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            x_star = np.maximum(res.x[:n], 0)
            M_star = max(res.x[M_idx], 0)
            x_sum = float(np.sum(x_star))
            if x_sum > 1e-15:
                x_star /= x_sum
            return x_star, float(M_star)
    except Exception:
        pass
    return None, None


def smoothed_gradient(a, tau):
    """
    Compute gradient of smoothed max of conv(a,a).
    Returns (A^T p) where p_k = softmax(tau * conv_k).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # Compute (A^T p)_j = sum_{m=0}^{n-1} a[m] * p[j+m]
    # Using FFT: conv(a_reversed, p)[n-1+j]
    a_rev = a[::-1]
    N = 1
    while N < n + (2 * n - 1) - 1:
        N <<= 1
    ar = fft(a_rev, N)
    pr = fft(p, N)
    conv_ap = ifft(ar * pr).real
    grad = conv_ap[n - 1:n - 1 + n]
    return grad


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x) = target_sum, x >= 0}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def fista_smooth(a, time_budget=100.0, tau_init=30.0, tau_end=1000.0, verbose=False):
    """
    Nesterov-accelerated gradient descent on smoothed objective over the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    t_k = 1.0
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    y = a.copy()
    it = 0
    best_it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.3
        
        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))
        
        # Adaptive learning rate
        if g_norm > 1e-15:
            lr_eff = lr / max(g_norm / base_lr, 1.0)
        else:
            lr_eff = lr
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        
        x_new = y - lr_eff * g
        x_new = project_simplex(x_new)
        
        y = x_new + beta * (x_new - a)
        y = project_simplex(y)
        
        obj_new = compute_obj(x_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = x_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
        
        a = x_new
        t_k = t_new
        it += 1
        lr *= 0.9998
        
        if stall > 8000:
            a = best_a.copy()
            y = best_a.copy()
            t_k = 1.0
            lr = base_lr * 0.4
            stall = 0
        
        if time.monotonic() - start > time_budget * 0.95 and (it - best_it) > 2000:
            break
    
    if verbose:
        print(f"    FISTA: obj={best_obj:.8f}, iters={it}")
    return best_a


def lp_driven_optimize(a, time_budget=300.0, verbose=True, mask=None, d_vals=None):
    """
    LP-driven optimization: solve LP for direction, line search, update.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    if mask is None or d_vals is None:
        mask, d_vals = precompute_conv_structure(n)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    stall = 0
    it = 0
    consecutive_bad = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        
        # Solve LP
        g, M = solve_primal_lp(a, mask, d_vals)
        if g is None:
            # Fallback: use exact subgradient
            conv_curr = compute_conv(a)
            max_curr = np.max(conv_curr)
            p_exact = np.zeros(len(conv_curr))
            p_exact[np.argmax(conv_curr)] = 1.0
            p_exact += 1e-8
            p_exact /= p_exact.sum()
            g = smoothed_gradient(a, 1000.0)
            g = project_simplex(a - 0.1 * g / (np.linalg.norm(g) + 1e-10))
            M = float(max_curr)
        
        # Current state
        conv_curr = compute_conv(a)
        max_curr = float(np.max(conv_curr))
        gap = max_curr - M
        
        # Adaptive step size candidates
        if gap > 1e-6:
            t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5]
        else:
            t_candidates = [0.01, 0.02, 0.05, 0.1, 0.2]
        
        # Line search
        best_t = 0.0
        best_obj_t = best_obj
        for t in t_candidates:
            a_try = project_simplex((1 - t) * a + t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t
        
        it += 1
        if best_t > 0 and best_obj_t < best_obj - 1e-12:
            a = project_simplex((1 - best_t) * a + best_t * g)
            best_obj = best_obj_t
            best_a = a.copy()
            stall = 0
            consecutive_bad = 0
            if verbose and (it % 30 == 0 or it == 1):
                elapsed = time.monotonic() - start
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, "
                      f"gap={gap:.6f}, time={elapsed:.0f}s")
        else:
            consecutive_bad += 1
            stall += 1
            
            if verbose and it % 50 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{it}: stalled obj={best_obj:.8f}, gap={gap:.6f}, time={elapsed:.0f}s")
            
            # Try smaller steps
            if consecutive_bad <= 3:
                for t_small in [0.02, 0.01, 0.005]:
                    a_small = project_simplex((1 - t_small) * a + t_small * g)
                    obj_small = compute_obj(a_small)
                    if obj_small < best_obj - 1e-12:
                        a = a_small
                        best_obj = obj_small
                        best_a = a.copy()
                        stall = 0
                        consecutive_bad = 0
                        if verbose:
                            print(f"    LP it{it}: small step t={t_small}, obj={best_obj:.8f}")
                        break
            
            if stall >= 3:
                # Perturb to escape
                tau_p = 300.0
                conv_p = compute_conv(a)
                max_p = np.max(conv_p)
                exp_p = np.exp(tau_p * (conv_p - max_p))
                p_p = exp_p / exp_p.sum()
                grad_p = smoothed_gradient(a, tau_p)
                grad_norm = np.linalg.norm(grad_p)
                if grad_norm > 1e-15:
                    a = project_simplex(a - 0.08 * grad_p / grad_norm)
                a = a + 0.008 * rng.standard_normal(n)
                a = project_simplex(a)
                stall = 0
                consecutive_bad = 0
            
            if stall >= 25:
                if verbose:
                    print(f"    LP it{it}: giving up after {it} iterations")
                break
    
    return best_a


def subgradient_method(a, time_budget=100.0, verbose=False):
    """
    Subgradient method using LP dual solutions.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    mask, d_vals = precompute_conv_structure(n)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    base_lr = 3.0 / np.sqrt(n)
    lr = base_lr
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1:
            break
        
        # Get direction from LP
        g, M = solve_primal_lp(a, mask, d_vals)
        if g is None:
            break
        
        # Subgradient-like direction: g - a (points from a toward g)
        direction = g - a
        dir_norm = np.linalg.norm(direction)
        
        if dir_norm < 1e-12:
            stall += 1
            if stall > 10:
                break
        else:
            # Adaptive step
            step = lr / (dir_norm + 1e-10)
            step = min(step, 0.5)
            
            a_new = project_simplex(a + step * direction)
            obj_new = compute_obj(a_new)
            
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                lr *= 0.95
            
            it += 1
            lr = max(lr * 0.999, base_lr * 0.3)
    
    if verbose:
        print(f"    Subgrad: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 7):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_walk_perturb(a, rng, num_steps=50, step_size=0.02):
    """Random walk perturbation in Fourier domain."""
    n = len(a)
    a_f = fft(a, n)
    for _ in range(num_steps):
        mask_freq = np.zeros(n)
        idx = rng.integers(0, n // 4)
        mask_freq[idx] = 1.0
        if idx > 0:
            mask_freq[n - idx] = 1.0
        delta = step_size * rng.standard_normal(n) * mask_freq
        a_f += delta
    a_new = ifft(a_f).real
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: fast LP-driven optimization + FISTA polishing + multi-n exploration.
    Strategy:
    1. Quick multi-n scan to find promising n
    2. Intensive LP optimization on best candidate
    3. FISTA polishing with increasing tau
    4. Second LP round (different starting point via perturbation)
    5. Focused n refinement
    6. Diverse restarts
    7. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")
    
    n_current = len(best_seq)
    
    # === Phase 1: Quick multi-n scan ===
    print(f"\n=== Phase 1: Quick multi-n scan ===")
    remaining = deadline - time.monotonic()
    
    # Generate n candidates around current and well-known good ranges
    n_scan = sorted(set([
        400, 500, 600, 700, 750, 800, 850, 900, 950, 1000,
        1100, 1200, 1300, 1500, 1700, 2000
    ]))
    # Add n near current
    n_scan = sorted(set(n_scan) | set([int(n_current * f) for f in [0.7, 0.8, 0.9, 1.1, 1.2, 1.4]]))
    n_scan = [n for n in n_scan if 300 <= n <= 2500]
    n_scan = n_scan[:12]
    
    print(f"  Scanning n: {n_scan}")
    for n_try in n_scan:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        
        a_try = resample_sequence(best_seq, n_try)
        mask_t, dvals_t = precompute_conv_structure(n_try)
        a_try = lp_driven_optimize(a_try, time_budget=min(remaining * 0.06, 12.0), verbose=False, mask=mask_t, d_vals=dvals_t)
        obj_try = compute_obj(a_try)
        
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
        
        marker = " **" if obj_try < best_score + 1e-8 else ""
        print(f"  n={n_try}: obj={obj_try:.8f}{marker}")
    
    n_best = len(best_seq)
    print(f"  Best after scan: n={n_best}, obj={best_score:.8f}")
    
    # === Phase 2: Intensive LP on best ===
    print(f"\n=== Phase 2: Intensive LP (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.30, 450.0)
    if p2_budget > 20:
        mask_n, dvals_n = precompute_conv_structure(n_best)
        a_lp = lp_driven_optimize(best_seq, time_budget=p2_budget, verbose=True, mask=mask_n, d_vals=dvals_n)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
    
    # === Phase 3: FISTA polish ===
    print(f"\n=== Phase 3: FISTA Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 120.0)
    if p3_budget > 10:
        a_fista = fista_smooth(best_seq, time_budget=p3_budget, tau_init=50.0, tau_end=800.0, verbose=True)
        obj_fista = compute_obj(a_fista)
        if obj_fista < best_score:
            best_score = obj_fista
            best_seq = a_fista.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
    
    # === Phase 4: Second LP with perturbed start ===
    print(f"\n=== Phase 4: Second LP (perturbed) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.20, 280.0)
    if p4_budget > 15:
        a_pert = fourier_perturb(best_seq, rng, strength=0.025)
        mask_n, dvals_n = precompute_conv_structure(len(a_pert))
        a_lp2 = lp_driven_optimize(a_pert, time_budget=p4_budget * 0.7, verbose=True, mask=mask_n, d_vals=dvals_n)
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_lp2 = fista_smooth(a_lp2, time_budget=min(remaining * 0.3, 40.0), tau_init=100.0, tau_end=600.0)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase4: obj={best_score:.8f} **")
        else:
            print(f"  Phase4: obj={obj_lp2:.8f} (no improvement)")
    
    # === Phase 5: Focused n refinement ===
    print(f"\n=== Phase 5: Focused n refinement ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    n_focus = sorted(set([
        n_best - 50, n_best - 30, n_best - 20, n_best + 20,
        n_best + 30, n_best + 50, n_best + 100, n_best - 100
    ]))
    n_focus = [n for n in n_focus if 300 <= n <= 3000 and abs(n - n_best) >= 15]
    n_focus = n_focus[:5]
    print(f"  Trying n: {n_focus}")
    
    for n_try in n_focus:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        a_try = resample_sequence(best_seq, n_try)
        mask_t, dvals_t = precompute_conv_structure(n_try)
        
        # LP
        a_try = lp_driven_optimize(a_try, time_budget=min(remaining * 0.35, 35.0), verbose=False, mask=mask_t, d_vals=dvals_t)
        remaining = deadline - time.monotonic()
        # FISTA
        if remaining > 10:
            a_try = fista_smooth(a_try, time_budget=min(remaining * 0.25, 20.0), tau_init=80.0, tau_end=500.0)
        
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={obj_try:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 6: More diverse restarts ===
    print(f"\n=== Phase 6: Diverse restarts ===")
    remaining =
