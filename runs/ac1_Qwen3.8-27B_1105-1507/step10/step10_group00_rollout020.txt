Hybrid gradient-LP optimization for minimizing peak auto-convolution.
Combines fast multiplicative gradient descent with periodic LP polishing,

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
import math

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def fast_gradient_descent(a, time_budget, lr_init=0.05, verbose=False):

    if s < 1e-10:
        return a

    it = 0
    stall = 0

    beta = 0.7

        k_max = int(np.argmax(conv))

        grad = np.zeros(n)

        # Projected gradient step

        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        
        # Nesterov update
        y = (1 - beta) * a_new + beta * (a_new - a)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y > 1e-10:
            y = y / s_y
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        lr *= 0.9997
        it += 1
        
        if stall > 500:
            a = best_a.copy()
            y = a.copy()
            momentum = np.zeros(n)
            lr = lr_init * 0.5
            stall = 0
        
        if verbose and it % 1000 == 0:
            print(f"      FD it{it}: obj={best_obj:.8f}, lr={lr:.6f}, t={time.monotonic()-start:.1f}s")

def diverse_init(n, rng, mode=0):

        a[:n//10] *= 0.3

        a = np.cos(np.pi * (x - 0.5)) ** 2 + 0.01
    elif mode == 2:
        # Bimodal
        a = np.exp(-((idx - n//4) ** 2) / (2 * (n//8) ** 2)) + \
            np.exp(-((idx - 3*n//4) ** 2) / (2 * (n//8) ** 2)) + 0.01

        a = 1.0 - 2 * np.abs(x - 0.5) + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_subgrad(a, k_max):
    """Vectorized subgradient of conv(a,a)[k_max] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    # Term 1: a[k_max - i] for valid i
    lo1 = max(0, k_max - n + 1)
    hi1 = min(n, k_max + 1)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[idx] += a[k_max - idx]
    # Term 2: a[i - k_max] for valid i
    lo2 = max(0, k_max)
    hi2 = min(n, k_max + n)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[idx] += a[idx - k_max]
    return grad


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):

    n = len(a)

    A = build_conv_matrix(a_norm)

    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

        conv = compute_conv(y)
        k_max = int(np.argmax(conv))
        grad = compute_subgrad(y, k_max)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        y = a_new + beta * (a_new - a)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y > 1e-10:
            y = y / s_y
        else:
            y = a_new.copy()

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        lr *= 0.9998
        it += 1

        if stall > 300:
            a = best_a.copy()
            y = a.copy()
            lr = lr_init * 0.7
            stall = 0

        if verbose and it % 2000 == 0:
            print(f"      PGD it{it}: obj={best_obj:.8f}, lr={lr:.6f}, t={time.monotonic()-start:.1f}s")

    return best_a


def line_search_blend(a, direction, time_budget=2.5):
    """Find optimal t in [0, 1.5] minimizing obj((1-t)*a + t*direction)."""
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    lo, hi = 0.0, 1.5

    def obj_at(t):
        if t < 0:
            t = 0
        a_try = (1 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try)

    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    start = time.monotonic()
    for _ in range(20):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)

    for t_try in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.2]:
        obj_try = obj_at(t_try)
        if obj_try < obj_best:
            obj_best = obj_try
            t_best = t_try

    return t_best, obj_best


def lp_polish(a, time_budget, max_iters=40, verbose=False):
    """LP-based polishing with line search and adaptive restarts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(555 + n)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=min(2.0, remaining * 0.1))

        if t_opt > 0 and obj_opt < best_obj - 1e-14:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1

        if stall >= 2:
            if stall <= 4:
                noise = rng.normal(0, 0.01 * stall, n)
                a = best_a * np.exp(noise)
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
            else:
                x = np.linspace(0, 1, n)
                mode = stall % 4
                if mode == 0:
                    a_div = np.cos(np.pi * (x - 0.5)) ** 2 + 0.01
                elif mode == 1:
                    a_div = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.01
                elif mode == 2:
                    a_div = rng.uniform(0.3, 1.0, n)
                    a_div = np.convolve(a_div, np.ones(5) / 5, mode='same')
                else:
                    a_div = 1.0 - 2 * np.abs(x - 0.5) + 0.02
                a_div = np.maximum(a_div, 0)
                a_div = a_div / a_div.sum()
                a = 0.3 * a_div + 0.7 * best_a
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
            stall = 0

        if verbose and it % 5 == 0:
            print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, elapsed={time.monotonic()-start:.1f}s")

    return best_a


def hybrid_optimize(a, time_budget, verbose=False):
    """
    Hybrid: MD -> LP -> PGD -> LP pipeline.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Step 1: Multiplicative descent
    alloc1 = min(time_budget * 0.35, 50.0)
    if alloc1 > 5:
        if verbose:
            print(f"    [1] MD ({alloc1:.0f}s)")
        a1 = multiplicative_descent(a, alloc1)
        o1 = compute_obj(a1)
        if o1 < best_obj:
            best_obj = o1
            best_a = a1.copy()
            if verbose:
                print(f"      -> {o1:.8f}")

    # Step 2: LP polish
    remaining = time_budget - (time.monotonic() - start)
    alloc2 = min(remaining * 0.3, 60.0)
    if alloc2 > 8:
        if verbose:
            print(f"    [2] LP ({alloc2:.0f}s)")
        a2 = lp_polish(best_a, alloc2, max_iters=25)
        o2 = compute_obj(a2)
        if o2 < best_obj:
            best_obj = o2
            best_a = a2.copy()
            if verbose:
                print(f"      -> {o2:.8f}")

    # Step 3: PGD from LP result
    remaining = time_budget - (time.monotonic() - start)
    alloc3 = min(remaining * 0.25, 30.0)
    if alloc3 > 5:
        if verbose:
            print(f"    [3] PGD ({alloc3:.0f}s)")
        a3 = projected_gradient_descent(best_a, alloc3)
        o3 = compute_obj(a3)
        if o3 < best_obj:
            best_obj = o3
            best_a = a3.copy()
            if verbose:
                print(f"      -> {o3:.8f}")

    # Step 4: Final LP
    remaining = time_budget - (time.monotonic() - start)
    alloc4 = min(remaining * 0.4, 40.0)
    if alloc4 > 8:
        if verbose:
            print(f"    [4] LP2 ({alloc4:.0f}s)")
        a4 = lp_polish(best_a, alloc4, max_iters=20)
        o4 = compute_obj(a4)
        if o4 < best_obj:
            best_obj = o4
            best_a = a4.copy()
            if verbose:
                print(f"      -> {o4:.8f}")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        # Flat interior, reduced edges
        a = np.ones(n)
        edge = max(1, n // 15)
        a[:edge] *= 0.2
        a[-edge:] *= 0.2
    elif mode == 1:
        # Cosine bell
        a = np.cos(np.pi * (x - 0.5)) ** 2 + 0.01
    elif mode == 2:
        # Bimodal gaussian
        a = np.exp(-((idx - n // 4) ** 2) / (2.0 * (n // 8) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2.0 * (n // 8) ** 2)) + 0.01
    elif mode == 3:
        # Random smooth
        a = rng.uniform(0.2, 1.0, n)
        a = np.convolve(a, np.ones(9) / 9, mode='same')
    elif mode == 4:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.01
    elif mode == 5:
        # Inverted triangle
        a = 1.0 - 2.0 * np.abs(x - 0.5) + 0.02
    elif mode == 6:
        # Sech
        a = 1.0 / np.cosh(3.0 * (2.0 * x - 1.0)) + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def try_new_length(a_best, n_new, time_budget, rng):
    """Interpolate to new length and optimize."""
    n_old = len(a_best)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        return None, np.inf
    a_try = a_try / s
    a_opt = hybrid_optimize(a_try, time_budget)
    obj = compute_obj(a_opt)
    return a_opt, obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: hybrid multiplicative gradient + LP with multi-start,
    adaptive length, and intensive final polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_base = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_base = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        print("No parent, using default init")

    if best_seq is None:
        best_seq = diverse_init(n_base, rng, mode=0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_base}): obj={best_score:.8f}")

    # === Phase 1: Optimize parent ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 1: Optimize Parent (budget={min(remaining*0.2, 200):.0f}s) ===")
        alloc = min(remaining * 0.2, 200.0)
        a_opt = hybrid_optimize(best_seq, alloc, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** Parent optimized: {best_score:.8f} **")
        else:
            print(f"  Parent: no improvement ({obj:.8f})")

    # === Phase 2: Multi-start diverse ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 2: Multi-Start ===")
        n_cur = len(best_seq)
        for mode in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            a_init = diverse_init(n_cur, rng, mode=mode)
            # Blend with current best for warm start
            a_init = 0.4 * a_init + 0.6 * best_seq
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()

            alloc = min(remaining * 0.12, 100.0)
            if alloc < 20:
                break

            a_try = hybrid_optimize(a_init, alloc)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** Mode {mode}: {best_score:.8f} **")
            else:
                print(f"  Mode {mode}: {obj_try:.8f}")

    # === Phase 3: Intensive LP on best ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: Intensive LP (budget={min(remaining*0.25, 250):.0f}s) ===")
        alloc = min(remaining * 0.25, 250.0)
        a_lp = lp_polish(best_seq, alloc, max_iters=50, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP: {best_score:.8f} **")
        else:
            print(f"  LP: {obj_lp:.8f}")

    # === Phase 4: MD refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 4: MD Refinement (budget={min(remaining*0.15, 80):.0f}s) ===")
        alloc = min(remaining * 0.15, 80.0)
        a_md = multiplicative_descent(best_seq, alloc, lr_init=0.01)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  ** MD: {best_score:.8f} **")
        else:
            print(f"  MD: {obj_md:.8f}")

    # === Phase 5: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 5: Length Exploration ===")
        n_cur = len(best_seq)
        n_options = sorted(set([
            n_cur - 100, n_cur + 100, n_cur - 200, n_cur + 200,
            n_cur + 400, 600, 700, 900, 1000, 1200, 1500
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2500 and abs(n - n_cur) >= 50]
        n_options = n_options[:4]
        print(f"  Trying n: {n_options} (current n={n_cur})")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            alloc = min(remaining * 0.18, 100.0)
            if alloc < 15:
                break
            a_try, obj_try = try_new_length(best_seq, n_try, alloc, rng)
            if a_try is not None and obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: {best_score:.8f} **")
            else:
                print(f"  n={n_try}: {obj_try:.8f}")

    # === Phase 6: Final hybrid polish ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 6: Final Polish (budget={min(remaining-15, 200):.0f}s) ===")
        alloc = min(remaining - 15, 200.0)
        a_fin = hybrid_optimize(best_seq, alloc, verbose=True)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  ** Final: {best_score:.8f} **")
        else:
            print(f"  Final: {obj_fin:.8f}")

    # === Phase 7: Last LP ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        a_last = lp_polish(best_seq, min(remaining - 8, 50.0), max_iters=15)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  ** Last LP: {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
