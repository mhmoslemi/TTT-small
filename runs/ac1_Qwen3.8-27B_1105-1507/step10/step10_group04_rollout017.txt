def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_g=None):

    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    g_val = None
    M_val = None

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:

    if s < 1e-10:
        return a

    best_obj = compute_obj(a)

    it = 0
    best_it = 0
    eta = 1.0

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        eta = 1.0 / (1.0 + 0.1 * it)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
New approach: Multiplicative Peak Shaving (MPS) with adaptive targeting, improved
LP polishing with better line search, Nesterov-accelerated smoothed gradient,
frequency-guided resampling, and aggressive multi-n exploration.
Key novelty: MPS uses multiplicative weight updates targeting top-k autocorrelation
peaks via FFT-based cross-correlation, providing a fast alternative to LP that

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    g_val = None
    M_val = None

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                break
        except Exception:
            continue

    return g_val, M_val


def multiplicative_peak_shave(a, time_budget, k_peaks=20, eta_decay=0.999, verbose=True):
    """
    Multiplicative weight updates targeting top-k autocorrelation peaks.
    Uses FFT-based cross-correlation for efficient pressure computation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    eta = 1.0
    floor = 1e-10

    # Precompute FFT size
    N = 1
    while N < 3 * n:
        N <<= 1

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 1:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Find top-k peaks
        len_conv = len(conv)
        k_eff = min(k_peaks, len_conv)
        if k_eff >= len_conv:
            peak_indices = np.arange(len_conv)
        else:
            peak_indices = np.argpartition(conv, -k_eff)[-k_eff:]
            peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        # Build peak indicator with soft weighting (top peaks get more weight)
        peak_weights = np.zeros(len_conv)
        for i, k in enumerate(peak_indices):
            # Exponential weighting: top peaks matter more
            peak_weights[k] = np.exp(-0.5 * i / k_eff)
        peak_weights /= peak_weights.sum()

        # Compute pressure via cross-correlation
        # pressure_j = sum_k peak_weights[k] * a[k-j] for valid k-j in [0,n-1]
        # = sum_m a[m] * peak_weights[j+m] for valid j+m
        A_f = fft(a, N)
        P_f = fft(peak_weights, N)
        pressure_full = ifft(np.conj(A_f) * P_f).real
        pressure = pressure_full[:n]
        pressure = np.maximum(pressure, 0)

        # Normalized pressure for stable updates
        p_max = float(np.max(pressure))
        if p_max < 1e-15:
            break
        pressure_norm = pressure / p_max

        # Adaptive eta: start large, decay
        eta = eta * eta_decay
        eta = max(eta, 0.05)

        # Multiplicative update
        a_new = a * np.exp(-eta * pressure_norm)
        a_new = np.maximum(a_new, floor)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            a = a_new
        else:
            # Don't update a if no improvement (avoid divergence)
            # But occasionally "kick" to escape
            if it % 500 == 0:
                n_kick = max(1, n // 50)
                kick_idx = np.random.default_rng(it).choice(n, n_kick, replace=False)
                a_new[kick_idx] *= (0.5 + np.random.default_rng(it + 1).均匀 if False else 0.5 + 0.5 * np.random.default_rng(it + 1000).uniform(size=n_kick))
                a_new = np.maximum(a_new, floor)
                a_new = a_new / a_new.sum()
                a = a_new
            # else keep a unchanged

        it += 1

        if it - best_it > 3000:
            a = best_a.copy()
            best_it = it
            if it > 15000:
                break

        if verbose and it % 2000 == 0:
            print(f"    MPS it{it}: obj={best_obj:.8f}, eta={eta:.4f}, max_c={max_conv:.6f}, t={elapsed:.0f}s")

    return best_a


def lp_polish_linesearch(a, time_budget, verbose=True):
    """LP polishing with golden-section line search for optimal step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.3

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Golden-section search over t in [0.01, 1.0]
        def obj_at_t(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return np.inf
            a_try = a_try / s_try
            return compute_obj(a_try)

        # Evaluate at several points
        t_candidates = np.array([
            0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5,
            0.6, 0.7, 0.8, 0.9, 1.0
        ])
        # Add points around last best
        t_candidates = np.concatenate([t_candidates, [
            max(0.02, last_t * 0.5),
            max(0.02, last_t * 0.7),
            max(0.02, last_t * 0.85),
            last_t,
            min(1.0, last_t * 1.15),
            min(1.0, last_t * 1.3),
            min(1.0, last_t * 1.5),
        ]])
        t_candidates = np.unique(np.clip(t_candidates, 0.02, 1.0))

        best_t = last_t
        best_t_obj = np.inf
        best_a_local = best_a.copy()

        for t in t_candidates:
            obj_t = obj_at_t(t)
            if obj_t < best_t_obj:
                best_t_obj = obj_t
                best_t = t
                best_a_local = ((1.0 - t) * a + t * g_lp)
                best_a_local = np.maximum(best_a_local, 0)
                best_a_local = best_a_local / best_a_local.sum()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t
            stall = 0
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.4f}, elapsed={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 4:
                break

    return best_a


def high_tau_gd(a, time_budget, tau_start=100, tau_end=8000, verbose=True):
    """High-precision smoothed gradient descent with Nesterov and adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.0
    v = np.zeros(n)
    it = 0
    best_it = 0

    N = 1
    while N < 3 * n:
        N <<= 1

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 1:
            break
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.4

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Smoothed max gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        P_f = fft(p, N)
        A_f = fft(y, N)
        g_full = ifft(P_f * np.conj(A_f)).real
        g = 2.0 * g_full[:n]

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        lr = min(1.0 / (g_norm + 1e-10), 3.0 / np.sqrt(n))
        lr = max(lr, 1e-8)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        beta = 2.0 / (1.0 + np.sqrt(1.0 + 4.0 * beta * beta))

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            beta = 0.0
            best_it = it
            if it > 50000:
                break

        if verbose and it % 5000 == 0:
            print(f"    HTGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.0f}s")

    return best_a


def log_param_optimize(a, time_budget, tau_min=200, tau_max=8000, verbose=True):
    """Optimize using log-parameterization with L-BFGS."""
    full_len = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    threshold = 1e-6
    support = np.where(a > threshold)[0]
    if len(support) < 2:
        support = np.where(a > a.max() * 1e-8)[0]
    if len(support) < 2:
        return a

    b0 = np.log(np.maximum(a[support], 1e-30))
    b0 = b0 - b0.mean()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    n_stages = 4
    tau_stages = [tau_min + (tau_max - tau_min) * (i / max(1, n_stages - 1)) ** 2
                  for i in range(n_stages)]

    current_b = b0.copy()
    for stage_idx, tau in enumerate(tau_stages):
        if time.monotonic() - start > time_budget:
            break

        def obj_grad_smooth(b_vec):
            b_shifted = b_vec - b_vec.max()
            a_supp = np.exp(b_shifted)
            a_supp = a_supp / a_supp.sum()
            a_full = np.zeros(full_len)
            a_full[support] = a_supp
            conv = compute_conv(a_full)
            max_conv = float(np.max(conv))
            exp_vals = np.exp(tau * (conv - max_conv))
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp
            obj_val = max_conv + np.log(sum_exp) / tau
            N = 1
            while N < 3 * full_len:
                N <<= 1
            P = fft(p, N)
            A_full_f = fft(a_full, N)
            g_full = ifft(P * np.conj(A_full_f)).real
            grad_full = 2.0 * g_full[:full_len]
            grad_b = a_supp * (grad_full[support] - np.sum(a_supp * grad_full[support]))
            return obj_val, grad_b

        try:
            res = scipy_minimize(obj_grad_smooth, current_b, method='L-BFGS-B',
                                 jac=True, options={'maxiter': 300, 'gtol': 1e-10,
                                                    'ftol': 1e-14, 'maxfun': 800})
            if res.x is not None:
                b_shifted = res.x - res.x.max()
                a_supp = np.exp(b_shifted)
                a_supp = a_supp / a_supp.sum()
                a_new = np.zeros(full_len)
                a_new[support] = a_supp
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    current_b = res.x.copy()
                    if verbose:
                        print(f"    LogOpt stage{stage_idx}: tau={tau:.0f}, obj={best_obj:.8f}")
        except Exception:
            pass

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation with edge handling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        else:
            print(f"  LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: MPS Refinement (150s) ===
    print(f"\n=== Phase 3: MPS Refinement ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 150.0)
    if p3_budget > 15:
        a_mps2 = multiplicative_peak_shave(best_seq, time_budget=p3_budget, k_peaks=10,
                                           eta_decay=0.9995, verbose=True)
        obj_mps2 = compute_obj(a_mps2)
        if obj_mps2 < best_score:
            best_score = obj_mps2
            best_seq = a_mps2.copy()
            print(f"  MPS2 improved: obj={best_score:.8f} **")
        else:
            print(f"  MPS2: obj={obj_mps2:.8f}")

    # === Phase 4: High-tau GD (150s) ===
    print(f"\n=== Phase 4: High-tau GD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.12, 150.0)
    if p4_budget > 15:
        a_gd = high_tau_gd(best_seq, time_budget=p4_budget,
                           tau_start=200, tau_end=6000, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  GD improved: obj={best_score:.8f} **")
        else:
            print(f"  GD: obj={obj_gd:.8f}")

    # === Phase 5: Multi-n Exploration (250s) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.22, 250.0)
    n_best = len(best_seq)

    n_options = sorted(set([
        max(200, n_best - 200),
        n_best,
        n_best + 100,
        n_best + 300,
        n_best + 600,
        n_best + 1000,
        1200,
        1600,
        2000,
        2500,
        3000,
    ]))
    n_options = [n for n in n_options if 300 <= n <= 5000 and n != n_best]
    # Add some purely for exploration
    rng_n = [400, 600, 900, 1100, 1400, 1800, 2200, 2800, 3500]
    for nn in rng_n:
        if nn not in n_options and 300 <= nn <= 5000:
            n_options.append(nn)
    n_options = sorted(set(n_options))[:10]
    print(f"  n options: {n_options}")

    per_n_budget = p5_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_n_budget, remaining - 20)
        if alloc < 25:
            break

        # Resample from best
        a_try = resample_sequence(best_seq, n_try)

        # Quick MPS
        mps_alloc = alloc * 0.4
        if mps_alloc > 8:
            a_try = multiplicative_peak_shave(a_try, time_budget=mps_alloc, k_peaks=15, verbose=False)

        # Quick LP
        remaining = deadline - time.monotonic()
        lp_alloc = min(alloc * 0.4, remaining - 15)
        if lp_alloc > 10:
            a_try = lp_polish_linesearch(a_try, time_budget=lp_alloc, verbose=False)

        # Quick GD
        remaining = deadline - time.monotonic()
        gd_alloc = min(alloc * 0.2, remaining - 10)
        if gd_alloc > 5:
            a_try = high_tau_gd(a_try, time_budget=gd_alloc,
                                tau_start=100, tau_end=3000, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts (100s) ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p6_budget = min(remaining * 0.08, 100.0)
    n_current = len(best_seq)
    n_restarts = max(2, int(p6_budget / 40))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(40, remaining - 10)
        if alloc < 15:
            break

        mode = (r * 3) % 9
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        # Optimize: MPS + LP
        half = alloc * 0.45
        a_r = multiplicative_peak_shave(a_r, time_budget=half, k_peaks=20, verbose=False)
        remaining = deadline - time.monotonic()
        lp_alloc = min(alloc * 0.45, remaining - 8)
        if lp_alloc > 8:
            a_r = lp_polish_linesearch(a_r, time_budget=lp_alloc, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish (remaining) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s")

    # LP round 1
    if remaining > 40:
        alloc = min(remaining * 0.3, 120.0)
        a_final = lp_polish_linesearch(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP1: obj={best_score:.8f} **")

    # MPS round
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 80.0)
        a_final = multiplicative_peak_shave(best_seq, time_budget=alloc, k_peaks=5,
                                            eta_decay=0.9998, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final MPS: obj={best_score:.8f} **")

    # GD round
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 100.0)
        a_final = high_tau_gd(best_seq, time_budget=alloc,
                              tau_start=500, tau_end=12000, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Log-param round
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.2, 60.0)
        a_final = log_param_optimize(best_seq, time_budget=alloc,
                                     tau_min=500, tau_max=10000, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Log: obj={best_score:.8f} **")

    # LP round 2 (final)
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.3, 60.0)
        a_final = lp_polish_linesearch(best_seq, time_budget=alloc, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Ultra-fine GD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 40.0)
        a_final = high_tau_gd(best_seq, time_budget=alloc,
                              tau_start=2000, tau_end=20000, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
