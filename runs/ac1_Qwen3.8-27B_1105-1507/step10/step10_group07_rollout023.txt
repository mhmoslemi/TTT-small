from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smoothed_grad(a, tau):
    """Compute gradient of smoothed max of convolution using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    p = np.exp(tau * (conv - max_conv))
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(conv)
        np.unravel_index(np.argmax(conv), conv.shape)[0] and (p[np.argmax(conv)] = 1)
    else:
        p /= p_sum
    
    # cross-correlation: grad_j = (4/tau) * sum_m p_{m+j} * a_m
    flip_a = a[::-1]
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    F = rfft(flip_a, N)
    P = rfft(p, N)
    conv_result = irfft(F * P, N)
    cross_corr = conv_result[n-1 : 2*n-1]
    
    return (4.0 / tau) * cross_corr

"""
FISTA + Spectral GD + LP Hybrid for Minimum Peak Autocorrelation.
Uses Nesterov-accelerated proximal gradient (FISTA) with log-sum-exp smoothing,
FFT-based gradient computation, spectral initialization via Gerchberg-Saxton,
annealing schedule for the smoothing parameter, and multi-length exploration.
Finishes with a brief LP polish using cvxpy.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_val(a, tau):
    """Compute (1/tau)*log(sum exp(tau*C_k)) where C = conv(a,a)."""
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    log_sum = float(np.sum(np.exp(shifted)))
    if log_sum < 1e-300:
        return max_c
    return max_c + np.log(log_sum) / tau


def compute_smoothed_grad(a, tau):
    """Compute gradient of smoothed max of convolution using FFT cross-correlation."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    max_shifted = float(np.max(shifted))
    exp_vals = np.exp(shifted - max_shifted)
    p = exp_vals / float(np.sum(exp_vals))

    # Cross-correlation: grad_j = 2 * sum_m p_{j+m} * a_m
    flip_a = a[::-1]
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    F = rfft(flip_a, N)
    P = rfft(p, N)
    conv_result = irfft(F * P, N)
    cross_corr = conv_result[n - 1:2 * n - 1]
    return 2.0 * cross_corr


def fista_smoothed(a, tau, steps=20000, L_init=None, verbose=False):
    """
    FISTA (Nesterov accelerated proximal gradient) for minimizing
    smoothed max of convolution over the simplex {x>=0, sum(x)=1}.
    """
    n = len(a)
    x = project_simplex(a.copy())
    x_prev = x.copy()
    t = 1.0

    if L_init is None:
        # Estimate L from Hessian bound
        conv = compute_conv(x)
        max_c = float(np.max(conv))
        L = 4.0 * tau * max_c * n * 4.0
        L = max(L, 10.0)
    else:
        L = L_init

    for k in range(steps):
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y > 0:
            y = y / s_y
        else:
            y = np.ones(n) / n

        g = compute_smoothed_grad(y, tau)
        f_y = smooth_max_val(y, tau)

        # Backtracking
        L_trial = L
        for _ in range(30):
            x_new = project_simplex(y - g / L_trial)
            f_new = smooth_max_val(x_new, tau)
            diff = x_new - y
            f_bound = f_y + np.dot(g, diff) + 0.5 * np.dot(diff, diff) / L_trial
            if f_new <= f_bound + 1e-9:
                break
            L_trial *= 2.0
        else:
            x_new = project_simplex(y - g / L_trial)

        L = max(L_trial * 0.7, 1.0)

        x_prev = x
        x = x_new
        t = t_new

        if verbose and (k + 1) % 5000 == 0:
            obj = compute_obj(x)
            print(f"      FISTA tau={tau:.0f} iter={k+1}: obj={obj:.8f}")

    return x


def fista_with_annealing(a, tau_start=20.0, tau_end=800.0, total_steps=100000,
                          time_budget=300.0, verbose=False):
    """FISTA with annealing schedule for tau."""
    n = len(a)
    x = project_simplex(a.copy())
    n_phases = 8
    steps_per_phase = total_steps // n_phases
    start_time = time.monotonic()

    for phase in range(n_phases):
        remaining = time_budget - (time.monotonic() - start_time)
        if remaining < 5:
            break
        progress = phase / max(n_phases - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** (progress ** 1.2)
        steps = min(steps_per_phase, int(remaining * 3000))
        if steps < 100:
            break
        x = fista_smoothed(x, tau, steps=steps, verbose=verbose)

    return x


def spectral_init(n, seed=42, iterations=200):
    """Gerchberg-Saxton style spectral initialization."""
    rng = np.random.default_rng(seed)
    a = rng.dirichlet(np.ones(n))
    N = n
    for it in range(iterations):
        A = fft(a, N)
        target_mag = 1.0 / np.sqrt(n)
        A_new = target_mag * np.exp(1j * np.angle(A))
        A_new[0] = A[0]  # preserve DC
        a_new = ifft(A_new, N).real
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
        else:
            a_new = np.ones(n) / n
        a = a_new
    return a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based polishing: solve min M s.t. A@g <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = project_simplex(a.copy())
    best_arr = a.copy()
    best_obj = compute_obj(best_arr)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        A = build_conv_matrix(a)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        prob = cp.Problem(cp.Minimize(M), [A @ g <= M, cp.sum(g) == 1])
        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = g_val.sum()
                    if s > 1e-10:
                        g_val = g_val / s
                        # Line search
                        best_t = 0
                        best_t_obj = best_obj
                        for t_val in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]:
                            a_try = (1 - t_val) * best_arr + t_val * g_val
                            a_try = project_simplex(a_try)
                            obj_try = compute_obj(a_try)
                            if obj_try < best_t_obj - 1e-14:
                                best_t_obj = obj_try
                                best_t = t_val
                        if best_t > 0:
                            best_obj = best_t_obj
                            best_arr = (1 - best_t) * best_arr + best_t * g_val
                            best_arr = project_simplex(best_arr)
                            a = best_arr.copy()
                        solved = True
                        break
            except Exception:
                continue
        if not solved:
            break
        it += 1

    return best_arr


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
