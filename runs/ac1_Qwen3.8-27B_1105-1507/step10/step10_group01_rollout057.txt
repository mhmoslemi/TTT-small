"""

Uses a novel combination of:

3. Spectral shaping (frequency-domain equalization of the power spectrum)
4. Nesterov accelerated gradient descent with adaptive smoothed max

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a.astype(np.float64), size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_maxpeak(a, top_k=1):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for top_k largest positions, sum(g)=1, g>=0.
    Actually: min M s.t. (a*g)[k] <= M for ALL k, sum(g)=1, g>=0.
    Returns (g, M_normalized) or (None, None).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    # Solve: min M s.t. M >= A[g] for all g, sum(g)=1, g>=0
    # Equivalent to: min sum_i c_i * g_i where c_i = max_k A[k,i]... no.
    # We need: minimize M subject to A @ g <= M * ones, sum(g) = 1, g >= 0.
    # This is a standard LP. Let's use scipy linprog.
    from scipy.optimize import linprog

    # Variables: g[0..n-1], M
    # Minimize: M
    # Constraints:
    #   A @ g - M <= 0  (2n-1 constraints)
    #   sum(g) = 1
    #   g >= 0, M free (but will be positive)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize M

    # Inequality constraints: A @ g - M <= 0
    # In linprog form: A_ub @ x <= b_ub
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status == 0 and res.x is not None:
        g_val = np.maximum(res.x[:n], 0)
        sg = float(np.sum(g_val))
        if sg > 1e-10:
            return g_val / sg, float(res.x[-1])
    return None, None


def solve_lp_topk(a, top_k=10):
    """
    Solve a relaxed LP: minimize sum of top_k peaks of (a*g).
    Approximation: minimize M s.t. sum of top_k of (a*g) <= top_k * M.
    We approximate by penalizing the top_k positions.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    # Find the top_k positions of the current a*a
    conv_curr = compute_conv(a_norm)
    top_positions = np.argsort(conv_curr)[-top_k:]

    # Solve: min mean of (a*g) at top positions + small * max
    # Approximate: min sum_{k in top} (a*g)[k] / top_k
    # This is: min sum_i (sum_{k in top} A[k,i]) / top_k * g_i
    c = A[top_positions, :].sum(axis=0) / top_k

    from scipy.optimize import linprog
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n

    res = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

    if res.status == 0 and res.x is not None:
        g_val = np.maximum(res.x, 0)
        sg = float(np.sum(g_val))
        if sg > 1e-10:
            return g_val / sg
    return None


def line_search(a, g, n_evals=20):
    """Line search: find best t in [0,1] to minimize obj((1-t)*a + t*g)."""
    n = len(a)
    ts = np.linspace(0.01, 0.99, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try / s)

    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]

    # Parabolic refinement
    if 0 < best_idx < n_evals - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.01, 0.99))
            a_par = (1 - t_par) * a + t_par * g
            a_par = np.maximum(a_par, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                f_par = compute_obj(a_par / s_par)
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par

    return best_t, best_obj

def spectral_shaping(a, time_budget=100.0, alpha=0.3, n_outer=100):
    """
    Spectral shaping: iteratively modify |FFT(a)|^2 to flatten the autocorrelation.
    This is a frequency-domain optimization that directly targets flattening conv(a,a).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 4 * n:
        N <<= 1

    # Target: flat autocorrelation means |A|^2 should be proportional to
    # |FFT(rectangular_window(2n-1, N))|^2
    rect = np.zeros(N)
    rect[:2 * n - 1] = 1.0
    target_spec = np.abs(fft(rect)) ** 2
    target_spec = target_spec / (np.sum(target_spec) + 1e-30)

    iter_count = 0
    while time.monotonic() - start < time_budget and iter_count < n_outer:
        iter_count += 1
        A = fft(a, N)
        power = np.abs(A) ** 2
        power = power / (np.sum(power) + 1e-30)

        # Blend current power spectrum toward target
        new_power = (1.0 - alpha) * power + alpha * target_spec
        # Preserve phase
        phase = np.angle(A)
        A_new = np.sqrt(np.maximum(new_power, 0)) * np.exp(1j * phase)
        a_new = ifft(A_new, N).real[:n]

        # Enforce non-negativity and normalize
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a = best_a.copy()
            break

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    if time_budget > 0:
        elapsed = time.monotonic() - start
        print(f"    Spectral shaping: obj={best_obj:.8f}, iters={iter_count}, time={elapsed:.0f}s")

    return best_a


def spectral_shaping_v2(a, time_budget=80.0):
    """
    Alternative spectral shaping: use iterative water-filling on the spectrum.
    At each iteration, reduce power at frequencies that contribute most to the peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 4 * n:
        N <<= 1

    iter_count = 0
    step_size = 0.5
    while time.monotonic() - start < time_budget and iter_count < 200:
        iter_count += 1
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        k_peak = int(np.argmax(conv))

        A = fft(a, N)
        power = np.abs(A) ** 2

        # Compute which frequencies contribute to the peak
        # (a*a)[k_peak] = sum_m A[m] * conj(A[m]) * exp(2j*pi*m*k_peak/N) / N
        # = sum_m |A[m]|^2 * exp(2j*pi*m*k_peak/N) / N
        # The real part: sum_m |A[m]|^2 * cos(2*pi*m*k_peak/N) / N
        # To reduce the peak, reduce power where cos(2*pi*m*k_peak/N) > 0

        phase_factor = np.cos(2.0 * np.pi * np.arange(N) * k_peak / N)
        # Reduce power where phase_factor is positive (contributes to peak)
        reduction = np.where(phase_factor > 0, phase_factor * step_size, 0.0)
        power_new = power * (1.0 - reduction)
        power_new = np.maximum(power_new, 1e-20)

        # Also add small power where phase_factor is negative (to maintain total energy)
        power_new += 0.01 * step_size * np.maximum(-phase_factor, 0) * power.max() / N

        A_new = np.sqrt(power_new) * np.exp(1j * np.angle(A))
        a_new = ifft(A_new, N).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step size
            step_size *= 0.9

        if iter_count % 50 == 0:
            step_size = 0.5  # Reset step size periodically

    return best_a


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def gd_polish(a, time_budget=150.0, tau_init=50.0, tau_end=1500.0, verbose=False):

        return a

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.2

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        # Adaptive LR
        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.3 * base_lr), 3.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adapt

        # Barzilai-Borwein every few iterations
        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 4.0 / np.sqrt(n))
                lr = 0.6 * lr + 0.4 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        # Decay learning rate slowly
        lr = max(1e-10, lr * 0.9999)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

    if verbose:
        print(f"    GD: obj={best_obj:.8f}, iters={it}")
    return best_a


def coordinate_descent(a, time_budget=60.0, verbose=False):
    """
    Coordinate descent: try small perturbations of individual coordinates.
    This can find improvements that gradient methods miss.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999 + n)

    # Precompute convolution for fast updates
    # When we change a[i] by delta, the convolution changes by:
    # delta * (a shifted by i) + delta * (a shifted by -(i)) ... actually it's more complex
    # For speed, just recompute when needed but try many coordinates per pass

    it = 0
    while time.monotonic() - start < time_budget:
        # Random coordinate
        i = rng.integers(0, n)
        # Try increasing and decreasing
        for delta_frac in [0.01, 0.005, -0.01, -0.005, 0.02, -0.02]:
            a_try = a.copy()
            a_try[i] = max(0, a_try[i] + delta_frac * best_a.mean())
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break

        it += 1
        if it % 5000 == 0 and verbose:
            print(f"    CD: obj={best_obj:.8f}, iters={it}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, method="random"):
    """Generate diverse initializations."""
    if method == "uniform":
        return np.ones(n) / n
    elif method == "exponential":
        a = np.exp(-np.arange(n) * 0.01)
        return a / a.sum()
    elif method == "sinusoidal":
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * np.arange(n) / n)
        a = np.maximum(a, 0.1)
        return a / a.sum()
    elif method == "ramp":
        a = np.linspace(0.1, 2.0, n)
        return a / a.sum()
    elif method == "bell":
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
        return a / a.sum()
    elif method == "asymmetric_bell":
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * ((x + 0.3) ** 2))
        return a / a.sum()
    elif method == "two_peak":
        x = np.linspace(-1, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x + 0.4) ** 2)
        return a / a.sum()
    else:  # random
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines multi-n exploration, LP optimization, spectral shaping,
    gradient descent, and coordinate descent for comprehensive optimization.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # === Initialize from parent ===
    parent_n = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                parent_n = parent.size
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        parent = np.ones(800) / 800.0
        best_score = compute_obj(parent)
        best_seq = parent.copy()
        parent_n = 800
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: Multi-n Coarse Optimization (300s) ===
    print("\n=== Phase 1: Multi-n Coarse Optimization ===")
    phase1_budget = min(300.0, (deadline - time.monotonic()) * 0.25)
    n_candidates = sorted(set([
        600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600
    ]))
    # Prioritize n values close to parent
    n_candidates.sort(key=lambda x: abs(x - parent_n))
    n_to_try = n_candidates[:5]

    per_n_budget = phase1_budget / len(n_to_try)
    init_methods = ["uniform", "sinusoidal", "asymmetric_bell", "two_peak", "random"]

    for idx, n_try in enumerate(n_to_try):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_n_budget, remaining - 10)
        if alloc < 15:
            continue

        # Try a couple of initializations
        for m_idx in range(min(2, len(init_methods))):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            method = init_methods[(idx + m_idx) % len(init_methods)]

            if n_try == parent_n and m_idx == 0:
                a_try = best_seq.copy()  # Use parent for original n
            else:
                if n_try == parent_n:
                    a_try = resample_sequence(best_seq, n_try)
                else:
                    a_try = diverse_init(n_try, rng, method)
                    # Blend with resampled parent for better starting point
                    parent_resampled = resample_sequence(best_seq, n_try)
                    a_try = 0.5 * a_try + 0.5 * parent_resampled
                    a_try = a_try / a_try.sum()

            # Quick LP optimization
            alloc_lp = alloc * 0.6
            if alloc_lp > 5:
                a_try = lp_driven_optimize(a_try, time_budget=alloc_lp, verbose=False)

            # Quick spectral shaping
            remaining = deadline - time.monotonic()
            alloc_spec = min(alloc * 0.2, remaining - 5)
            if alloc_spec > 5:
                a_try = spectral_shaping(a_try, time_budget=alloc_spec, alpha=0.3)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Phase1: n={n_try}, init={method}: obj={best_score:.8f} **")
            else:
                print(f"  Phase1: n={n_try}, init={method}: obj={obj_try:.8f}")

    # === Phase 2: Fine LP Optimization (350s) ===
    print(f"\n=== Phase 2: Fine LP (n={len(best_seq)}, score={best_score:.8f}) ===")
    phase2_budget = min(350.0, (deadline - time.monotonic()) * 0.28)
    if phase2_budget > 30:
        a_lp = lp_driven_optimize(best_seq, time_budget=phase2_budget, verbose=True, top_k=1)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Spectral Shaping (100s) ===
    print(f"\n=== Phase 3: Spectral Shaping (score={best_score:.8f}) ===")
    phase3_budget = min(100.0, (deadline - time.monotonic()) * 0.08)
    if phase3_budget > 15:
        a_spec = spectral_shaping(best_seq, time_budget=phase3_budget * 0.5, alpha=0.3)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Phase3 spec_v1: obj={best_score:.8f} **")

        a_spec2 = spectral_shaping_v2(best_seq, time_budget=phase3_budget * 0.5)
        obj_spec2 = compute_obj(a_spec2)
        if obj_spec2 < best_score:
            best_score = obj_spec2
            best_seq = a_spec2.copy()
            print(f"  Phase3 spec_v2: obj={best_score:.8f} **")

    # === Phase 4: Gradient Descent Polish (300s) ===
    print(f"\n=== Phase 4: GD Polish (score={best_score:.8f}) ===")
    phase4_budget = min(300.0, (deadline - time.monotonic()) * 0.22)
    if phase4_budget > 20:
        # First pass: moderate tau
        alloc1 = phase4_budget * 0.4
        a_gd1 = gd_polish(best_seq, time_budget=alloc1, tau_init=50.0, tau_end=500.0, verbose=True)
        obj_gd1 = compute_obj(a_gd1)
        if obj_gd1 < best_score:
            best_score = obj_gd1
            best_seq = a_gd1.copy()
            print(f"  Phase4 GD1: obj={best_score:.8f} **")

        # Second pass: high tau for fine polish
        remaining = deadline - time.monotonic()
        alloc2 = min(phase4_budget * 0.4, remaining - 10)
        if alloc2 > 15:
            a_gd2 = gd_polish(best_seq, time_budget=alloc2, tau_init=200.0, tau_end=2000.0, verbose=True)
            obj_gd2 = compute_obj(a_gd2)
            if obj_gd2 < best_score:
                best_score = obj_gd2
                best_seq = a_gd2.copy()
                print(f"  Phase4 GD2: obj={best_score:.8f} **")

    # === Phase 5: Second LP Round + GD (150s) ===
    print(f"\n=== Phase 5: Second LP + GD (score={best_score:.8f}) ===")
    phase5_budget = min(150.0, (deadline - time.monotonic()) * 0.15)
    if phase5_budget > 20:
        alloc_lp = phase5_budget * 0.5
        a_lp2 = lp_driven_optimize(best_seq, time_budget=alloc_lp, verbose=True, top_k=5)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase5 LP: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        alloc_gd = min(phase5_budget * 0.3, remaining - 10)
        if alloc_gd > 15:
            a_gd3 = gd_polish(best_seq, time_budget=alloc_gd, tau_init=100.0, tau_end=1000.0, verbose=True)
            obj_gd3 = compute_obj(a_gd3)
            if obj_gd3 < best_score:
                best_score = obj_gd3
                best_seq = a_gd3.copy()
                print(f"  Phase5 GD: obj={best_score:.8f} **")

    # === Phase 6: Coordinate Descent + Final Polish (100s) ===
    print(f"\n=== Phase 6: Coordinate Descent + Final (score={best_score:.8f}) ===")
    phase6_budget = min(100.0, (deadline - time.monotonic()) * 0.1)
    if phase6_budget > 15:
        alloc_cd = phase6_budget * 0.5
        a_cd = coordinate_descent(best_seq, time_budget=alloc_cd, verbose=False)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase6 CD: obj={best_score:.8f} **")

        # Final high-tau GD
        remaining = deadline - time.monotonic()
        alloc_final = min(phase6_budget * 0.4, remaining - 5)
        if alloc_final > 10:
            a_final = gd_polish(best_seq, time_budget=alloc_final, tau_init=500.0, tau_end=5000.0, verbose=True)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Phase6 Final: obj={best_score:.8f} **")

    # === Phase 7: Try slightly different n values (if time) ===
    print(f"\n=== Phase 7: n Exploration (score={best_score:.8f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n_cur = len(best_seq)
        n_options = [n_cur + 50, n_cur - 50, n_cur + 100, n_cur - 100]
        n_options = [n for n in n_options if 400 <= n <= 2000]
        per_n = min(remaining * 0.25, 25.0)

        for n_try in n_options[:3]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_try = resample_sequence(best_seq, n_try)
            a_try = gp_direct_optimize(a_try, time_budget=per_n)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Phase7: n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  Phase7: n={n_try}: obj={obj_try:.8f}")

    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result


def gp_direct_optimize(a, time_budget=20.0):
    """Quick combined LP + GD optimization for n-exploration."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    t1 = time_budget * 0.6
    t2 = time_budget * 0.4
    a = lp_driven_optimize(a, time_budget=max(t1, 5.0), verbose=False)
    if t2 > 5.0:
        a = gd_polish(a, time_budget=t2, tau_init=100.0, tau_end=500.0)
    return a
