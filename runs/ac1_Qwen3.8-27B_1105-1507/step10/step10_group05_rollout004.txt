    for step in range(n_steps):
        # Choose coordinate: bias toward those that contribute to active constraints
        # Find active constraints (where conv is close to max)
        max_conv = float(np.max(conv))
        active_tol = max_conv * 0.9
        active_indices = np.where(conv >= active_tol)[0]
        
        # For each active index k, the contributing coordinates are j where 0 <= k-j < n
        # Pick a random active index, then a random contributing coordinate
        if len(active_indices) > 0:
            k_star = rng.choice(active_indices)
            valid_j = np.arange(max(0, k_star - n + 1), min(k_star + 1, n))
            if len(valid_j) > 0:
                j = rng.choice(valid_j)
            else:
                j = rng.integers(0, n)
        else:
            j = rng.integers(0, n)
        
        # Golden section search for optimal alpha at coordinate j
        # We want to find alpha that minimizes the objective when a[j] += alpha
        lo = -a[j] * 0.999
        hi = a[j] * 3.0 + 0.01
        
        # Quick grid search first
        grid = np.linspace(lo, hi, 10)
        best_alpha = 0.0
        grid_best_obj = best_obj
        
        for alpha in grid:
            a_try = a.copy()
            a_try[j] += alpha
            if a_try[j] < 0:
                continue
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < grid_best_obj:
                grid_best_obj = obj_try
                best_alpha = alpha
        
        # Refine with golden section if we found improvement
        if grid_best_obj < best_obj:
            # Narrow down the range
            for i in range(len(grid) - 1):
                if grid[i] <= best_alpha <= grid[i+1]:
                    lo = grid[max(0, i-1)]
                    hi = grid[min(len(grid)-1, i+1)]
                    break
            else:
                lo = -a[j] * 0.999
                hi = a[j] * 3.0 + 0.01
            
            invphi = (np.sqrt(5) - 1) / 2
            c = hi - invphi * (hi - lo)
            d = lo + invphi * (hi - lo)
            fc = compute_obj(np.maximum(a + c * np.eye(n)[:, j], 0) / np.sum(np.maximum(a + c * np.eye(n)[:, j], 0)))
            fd = compute_obj(np.maximum(a + d * np.eye(n)[:, j], 0) / np.sum(np.maximum(a + d * np.eye(n)[:, j], 0)))
            
            for _ in range(30):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - invphi * (hi - lo)
                    ac = a.copy()
                    ac[j] += c
                    ac = np.maximum(ac, 0)
                    sc = float(np.sum(ac))
                    if sc > 1e-10:
                        fc = compute_obj(ac / sc)
                    else:
                        fc = np.inf
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + invphi * (hi - lo)
                    ad = a.copy()
                    ad[j] += d
                    ad = np.maximum(ad, 0)
                    sd = float(np.sum(ad))
                    if sd > 1e-10:
                        fd = compute_obj(ad / sd)
                    else:
                        fd = np.inf
            
            best_alpha = (lo + hi) / 2
            a_try = a.copy()
            a_try[j] += best_alpha
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    conv = compute_conv(a)
        
        if verbose and step % 200 == 0:
            print(f"    CD step {step}: obj={best_obj:.8f}")
        
        # Periodically recompute conv
        if step % 100 == 0:
            conv = compute_conv(a)
    
    return best_a


def lbfgs_smoothed(a, time_budget=100.0, tau=50.0, verbose=False):
    """
    L-BFGS-B optimization of the log-sum-exp smoothed objective.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    def objective_and_grad(x):
        x = np.maximum(x, 0)
        s_x = float(np.sum(x))
        if s_x < 1e-10:
            return 1e10, np.zeros(n)
        x = x / s_x
        conv = compute_conv(x)
        max_c = float(np.max(conv))
        shifted = tau * (conv - max_c)
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        lse = max_c + np.log(sum_exp) / tau
        obj = 2.0 * n * lse
        
        # Gradient
        weights = exp_vals / sum_exp
        # grad conv[k] wrt x[j] = 2*x[k-j] if 0<=k-j<n
        # grad_x[j] = 2 * sum_k weights[k] * x[k-j] (for 0<=k-j<n)
        # = 2 * (flip(weights) convolved with x) ... 
        # Actually: sum_k weights[k] * x[k-j] = sum_k weights[k] * x[k-j]
        # Let m = k-j, so k = m+j: sum_m weights[m+j] * x[m] (for 0<=m<n, 0<=m+j<2n-1)
        # This is like a cross-correlation of weights and x
        
        # Use FFT for efficiency
        N = 1
        while N < 3 * n:
            N <<= 1
        W = fft(weights, N)
        X = fft(x, N)
        # We need: grad[j] = 2 * sum_k weights[k] * x[k-j] (0 <= k-j < n)
        # Let's compute h = ifft(W * conj(X)) which gives circular cross-correlation
        # h[j] = sum_k W_k * conj(X_k) * exp(2πijk/N) / N
        # = sum_k weights[k] * x[k-j mod N] (approximately, for our range)
        h = ifft(W * np.conj(X)).real
        grad = 2.0 * h[:n]
        
        # Adjust for the normalization constraint (sum x = 1)
        # Using the "projected gradient" idea: subtract the mean
        grad = grad - np.sum(grad * x) * x  # This isn't quite right
        
        # Actually, for the constrained problem min f(x) s.t. sum(x)=1, x>=0:
        # The projected gradient is: grad - (sum(grad*x)/sum(x^2)) * x (projection onto simplex)
        # But for L-BFGS with bounds, we just return the gradient and let the solver handle bounds
        # The constraint sum(x)=1 is handled by working in the normalized space
        
        # Since we normalize x inside the function, the effective gradient wrt the unnormalized x is:
        # d f(x/s) / d x = (1/s) * grad_f(x/s) - f(x/s)/s^2 * x
        # = (1/s) * (2n * sum_k w_k * 2*x_norm[k-j]) - (2n*lse)/s^2 * x
        # But this is complicated. Let's just use the simple gradient and let L-BFGS work with it.
        
        return obj, grad

    best_a = a.copy()
    best_obj = compute_obj(a)
    
    try:
        result = minimize(
            objective_and_grad,
            a,
            method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': 5000,
                'gtol': 1e-8,
                'maxls': 50,
                'callback': None
            }
        )
        
        if result.x is not None:
            a_result = np.maximum(result.x, 0)
            s_r = float(np.sum(a_result))
            if s_r > 1e-10:
                a_result = a_result / s_r
                obj_result = compute_obj(a_result)
                if obj_result < best_obj:
                    best_obj = obj_result
                    best_a = a_result.copy()
