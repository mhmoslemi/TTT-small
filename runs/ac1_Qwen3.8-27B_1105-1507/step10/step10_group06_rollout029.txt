```python
def solve_lp_mw(a, T_inner=500, eta=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0
    Using multiplicative weights on the dual.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    K = 2 * n - 1
    
    if eta is None:
        eta = 0.5
    
    # Precompute FFT constants
    N = 1
    while N < n + K:
        N <<= 1
    A_fft = fft(a_norm, N)
    
    # MW for dual: max_y min_j (A^T y)_j
    w = np.ones(K)
    y = w / K
    y_avg = np.zeros(K)
    
    for t in range(T_inner):
        # Compute (A^T y)_j = sum_i a_norm[i] * y[i+j] for j=0,...,n-1
        Y_fft = fft(y, N)
        R = ifft(np.conj(A_fft) * Y_fft).real[:n]
        
        # Row player's best response
        j_star = np.argmin(R)
        
        # Update weights: y_k *= exp(-eta * a_{k - j_star})
        # a_{k - j_star} is non-zero for k in [j_star, j_star + n)
        # The log-update: log(w_k) -= eta * a_{k - j_star}
        # For efficiency, just multiply the slice
        lo = j_star
        hi = j_star + n
        w[lo:hi] *= np.exp(-eta * a_norm)
        
        y = w / w.sum()
        y_avg += y
    
    y_avg /= T_inner
    
    # Recover primal solution from dual
    # g_j > 0 only where (A^T y_avg)_j is close to min
    Y_avg_fft = fft(y_avg, N)
    R_avg = ifft(np.conj(A_fft) * Y_avg_fft).real[:n]
    M_opt = float(np.min(R_avg))
    
    # Heuristic: g proportional to exp(-alpha * (R_avg - M_opt))
    alpha = 10.0
    g = np.exp(-alpha * (R_avg - M_opt))
    g = g / g.sum()
    
    return g, M_opt
```

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a fast MW-based LP solver (replacing cvxpy) combined with adaptive blending

import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))

def next_pow2(x):
    return 1 << (x - 1).bit_length()

def solve_lp_mw(a, T_inner=300, eta=0.5, verbose=False):
    """
    Fast LP solver using multiplicative weights on the dual.
    Solves: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0
    """
    n = len(a)
    s = float(np.sum(a))
    a_norm = a / s
    K = 2 * n - 1
    N = next_pow2(n + K)  # >= 3n-1
    
    A_fft = fft(a_norm, N)
    
    w = np.ones(K)
    y = w / K
    y_sum = K
    y_avg = np.zeros(K)
    
    for t in range(T_inner):
        Y_fft = fft(y, N)
        R = ifft(np.conj(A_fft) * Y_fft).real[:n]
        j_star = int(np.argmin(R))
        
        # Update: w[k] *= exp(-eta * a_norm[k - j_star]) for valid k
        lo = j_star
        hi = j_star + n
        w[lo:hi] *= np.exp(-eta * a_norm)
        
        y_sum = w.sum()
        y = w / y_sum
        y_avg += y
    
    y_avg /= T_inner
    
    # Recover primal from dual
    Y_avg_fft = fft(y_avg, N)
    R_avg = ifft(np.conj(A_fft) * Y_avg_fft).real[:n]
    M_opt = float(np.min(R_avg))
    
    # g_j proportional to exp(-alpha * (R_avg[j] - M_opt))
    gap = R_avg - M_opt
    alpha = 5.0
    g = np.exp(-alpha * gap)
    g_sum = g.sum()
    if g_sum < 1e-300:
        g = np.full(n, 1.0 / n)
    else:
        g = g / g_sum
    
    return g, M_opt
