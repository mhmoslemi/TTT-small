from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:

def gradient_conv(a, p):
    """grad of sum_k p_k (a*a)_k wrt a. p has len 2n-1, a has len n."""
    n = len(a)
    N = 1
    while N < 3*n:
        N <<= 1

    A = fft(a, N)

def build_conv_matrix(a):
    n = len(a)
    a_norm = a / np.sum(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a_norm
    return A

def solve_conv_lp(a):
    n = len(a)
    a_n = a / np.sum(a)
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None
