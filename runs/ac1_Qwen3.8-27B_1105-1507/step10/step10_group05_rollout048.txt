        # g[m] = 2 * sum_j a[j] * active[m+j]

        # g[m] = 2 * sum_{j=0}^{n-1} a[j] * active[m+j]
        # = 2 * sum_{j=0}^{n-1} a[j] * active[m+j]

        # g[m] = 2 * <a, h[m:m+n]>  (dot product of a with the slice h[m:m+n])

        g_half = np.fft.irfft(np.conj(fa) * fav, N)[:n]
        g = 2.0 * g_half
        
        # Update: multiplicative weight
        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 1e-30)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)

        it += 1

        if it - best_it > 3000:

            lr = 1.0 / np.sqrt(n) * 0.5

    return best_a

def lp_iteration(a, time_budget=100.0, verbose=False, seed=42):
    """LP-based iteration with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        it += 1
        
        # Line search
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.2, 0.4, 0.6, 0.8, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            best_it = it
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, M_lp={M_lp:.6f}")
        else:
            if it - best_it > 10:
                # Perturbation
                strength = min(0.1, 0.01 * (it - best_it))
                a = a * np.exp(rng.normal(0, strength, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_it = it
    
    return best_a

def smoothed_gd_descent(a, time_budget=60.0, tau_init=20.0, tau_end=500.0, verbose=False):
    """Smoothed gradient descent with soft-max approximation of max."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 2.0 / np.sqrt(n)
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp
        
        # Gradient: g[m] = 2 * sum_k p[k] * a[k-m] (with bounds)
        # Use FFT
        N = 1
        while N < 4 * n:
            N <<= 1
        ap = np.zeros(2 * n - 1)
        ap[:n] = a
        fa = np.fft.rfft(ap, N)
        fp = np.fft.rfft(p.astype(np.float64), N)
        g = 2.0 * np.fft.irfft(np.conj(fa) * fp, N)[:n]
        
        # Nesterov update
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        lr = max(1e-6, lr * 0.999)
        
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
