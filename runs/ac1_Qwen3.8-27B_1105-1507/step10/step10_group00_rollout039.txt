def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.001, 0.999, n)
    a = 1.0 / np.sqrt(1 - (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(3.0 * (2 * x - 1))
    elif mode == 7:
        # Power U-shape
        a = np.abs(2 * x - 1) ** 0.5 + 0.02
    elif mode == 8:
        # Very sharp U
        a = 1.0 / (0.01 + 10.0 * (2 * x - 1) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def line_search_blend(a, g_lp, time_budget=4.0):
    """Find optimal blend factor t that minimizes obj((1-t)*a + t*g_lp)."""
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    
    lo, hi = 0.0, 1.5
    
    def obj_at(t):
        if t < 0:
            t = 0
        a_try = (1 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)
    
    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)
    
    start = time.monotonic()
    for _ in range(30):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)
    
    t_best = c if fc < fd else d
    obj_best = min(fc, fd)
    
    for t_try in [0.01, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0, 1.2, 1.4]:
        obj_try = obj_at(t_try)
        if obj_try < obj_best:
            obj_best = obj_try
            t_best = t_try
    
    return t_best, obj_best


def lp_polish(a, time_budget=120.0, verbose=False, max_iters=60):
    """LP polishing with golden-section line search and adaptive restarts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(54321 + it)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        T0 = time.monotonic()
        g_lp, M_lp = solve_lp_minpeak(a)
        T1 = time.monotonic()
        
        if g_lp is None:
            break
        
        t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=min(3.0, remaining * 0.06))
        
        if t_opt > 0 and obj_opt < best_obj - 1e-14:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, M={M_lp:.6f}, "
                  f"lp_t={T1-T0:.2f}s, elapsed={time.monotonic()-start:.1f}s")
        
        if stall >= 3:
            if stall <= 5:
                freq = rng.integers(1, max(2, n // 30))
                phase = rng.uniform(0, 2 * np.pi)
                x = np.arange(n) / max(1, n - 1)
                pert = 0.03 * np.cos(2 * np.pi * freq * x + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                else:
                    a = best_a.copy()
            elif stall <= 8:
                noise = rng.normal(0, 0.02 * stall, n)
                a = np.maximum(best_a + noise, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                else:
                    a = best_a.copy()
            else:
                mode = it % 9
                a_div = diverse_init(n, rng, mode=mode)
                a = 0.3 * a_div + 0.7 * best_a
                a = np.maximum(a, 0)
                a = a / max(float(np.sum(a)), 1e-10)
            stall = 0
    
    return best_a


def md_lp_hybrid(a, time_budget=300.0, verbose=False, lp_every=30000):
    """
    Hybrid: mirror descent main loop with periodic LP direction injection.
    This is the key new algorithm.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 0.03
    it = 0
    patience = 0
    lp_count = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Main mirror descent step
        grad, max_val = get_subgradient(a)
        
        log_update = -eta * grad
        log_update -= np.max(log_update)
        a_new = a * np.exp(log_update)
        s_new = float(np.sum(a_new))
        
        if s_new < 1e-30:
            eta *= 0.5
            it += 1
            continue
        
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
            if patience % 200 == 0:
                eta *= 0.92
        
        a = a_new
        it += 1
        
        # Periodic LP injection
        if it % lp_every == 0 and remaining > 20:
            T0 = time.monotonic()
            g_lp, M_lp = solve_lp_minpeak(a)
            lp_time = time.monotonic() - T0
            lp_count += 1
            
            if g_lp is not None:
                # Blend with LP solution
                t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=min(2.0, remaining * 0.03))
                if t_opt > 0 and obj_opt < best_obj - 1e-14:
                    a_new = (1 - t_opt) * a + t_opt * g_lp
                    a_new = np.maximum(a_new, 0)
                    s_new = float(np.sum(a_new))
                    if s_new > 0.01:
                        a_new = a_new / s_new
                        obj_new = compute_obj(a_new)
                        if obj_new < best_obj:
                            best_obj = obj_new
                            best_a = a_new.copy()
                        a = a_new
                        patience = 0
                
                if verbose:
                    print(f"    MD-LP it{it} lp#{lp_count}: obj={best_obj:.8f}, "
                          f"t={t_opt:.4f}, M={M_lp:.6f}, lp_t={lp_time:.2f}s, "
                          f"elapsed={time.monotonic()-start:.1f}s")
        
        # Restart if stuck
        if patience > 3000:
            # Try LP restart
            if remaining > 15:
                g_lp, _ = solve_lp_minpeak(best_a)
                if g_lp is not None:
                    a = 0.5 * best_a + 0.5 * g_lp
                    a = np.maximum(a, 0)
                    a = a / max(float(np.sum(a)), 1e-10)
                else:
                    a = best_a.copy()
            else:
                a = best_a.copy()
            eta = 0.02
            patience = 0
        
        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD-LP it{it}: obj={best_obj:.8f}, eta={eta:.4f}, pat={patience}, t={elapsed:.1f}s")
    
    return best_a

"""
Mirror descent + LP hybrid optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses exponentiated gradient (mirror descent with entropy) as the primary optimizer,
combined with periodic LP-based direction finding, shape parameter sweeps,
and systematic n exploration. Key improvement over subgradient descent:
mirror descent avoids expensive simplex projections and handles the
non-negativity constraint naturally via multiplicative updates.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def get_subgradient(a):
    """
    Compute subgradient of max_k (a*a)_k at point a.
    Uses vectorized slice operations for speed.
    Returns (grad, max_val).
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    tol = 1e-8 * max_val + 1e-15
    peaks = np.where(conv >= max_val - tol)[0]

    grad = np.zeros(n)
    for k in peaks:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # grad[lo..hi] += 2 * a[k-lo : k-hi-1 : -1]
        grad[lo:hi + 1] += 2.0 * a[k - lo:k - hi - 1:-1]

    grad /= len(peaks)
    return grad, max_val


def mirror_descent(a, time_budget=60.0, verbose=False, eta_init=0.05):
    """
    Mirror descent (exponentiated gradient) for minimizing max_k (a*a)_k over simplex.
    Uses adaptive step size with geometric decay on stall and slow increase on progress.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = eta_init
    it = 0
    patience = 0
    consecutive_improve = 0

    while time.monotonic() - start < time_budget:
        grad, max_val = get_subgradient(a)

        # Mirror descent: a_j <- a_j * exp(-eta * grad_j) / Z
        log_update = -eta * grad
        max_lu = np.max(log_update)
        if max_lu > 50:  # extra safety for very large eta*grad
            log_update -= max_lu
        a_new = a * np.exp(log_update)
        s_new = float(np.sum(a_new))

        if s_new < 1e-30:
            eta *= 0.5
            it += 1
            continue

        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
            consecutive_improve += 1
        else:
            patience += 1
            consecutive_improve = 0

        a = a_new
        it += 1

        # Adaptive step size
        if consecutive_improve > 2000:
            eta = min(eta * 1.01, 0.3)
            consecutive_improve = 0
        if patience > 500:
            eta *= 0.85
        if patience > 2000:
            eta *= 0.7

        # Hard restart
        if patience > 8000:
            a = best_a.copy()
            eta = max(eta_init * 0.3, 0.005)
            patience = 0

        if it % 10000 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.5f}, pat={patience}, t={elapsed:.1f}s")

    return best_a


def subgradient_descent(a, time_budget=30.0, verbose=False):
    """Standard subgradient descent with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.025
    it = 0
    momentum = np.zeros(n)
    beta = 0.5
    best_it = 0

    while time.monotonic() - start < time_budget:
        grad, _ = get_subgradient(a)
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        lr *= 0.999
        it += 1

        if it - best_it > 4000:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = 0.015
            best_it = it

    return best_a


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def diverse_init(n, rng, mode=0):

    x = np.linspace(0, 1, n)
    idx = np.arange(n)

    elif mode == 2:

    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(3.0 * (2 * x - 1))
    elif mode == 7:
        a = np.abs(2 * x - 1) ** 0.5 + 0.02
    elif mode == 8:
        a = 1.0 / (0.01 + 10.0 * (2 * x - 1) ** 2)
    elif mode == 9:
        # Double U (peaks at edges and one-quarter marks)
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 2) + 0.3 * np.exp(-8 * (2 * x - 1) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def line_search_blend(a, g_lp, time_budget=3.0):
    """Find optimal blend factor t minimizing obj((1-t)*a + t*g_lp)."""
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi

    lo, hi = 0.0, 1.5

    def obj_at(t):
        if t < 0:
            t = 0
        a_try = (1 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    start = time.monotonic()
    for _ in range(30):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)

    for t_try in [0.01, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0, 1.3]:
        obj_try = obj_at(t_try)
        if obj_try < obj_best:
            obj_best = obj_try
            t_best = t_try

    return t_best, obj_best


def lp_polish(a, time_budget=120.0, verbose=False, max_iters=60):
    """LP polishing with golden-section line search and adaptive restarts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777 + n)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        T0 = time.monotonic()
        g_lp, M_lp = solve_lp_minpeak(a)
        lp_time = time.monotonic() - T0

        if g_lp is None:
            break

        t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=min(2.5, remaining * 0.05))

        if t_opt > 0 and obj_opt < best_obj - 1e-14:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 3 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, M={M_lp:.6f}, "
                  f"lp_t={lp_time:.2f}s, elapsed={time.monotonic()-start:.1f}s")

        if stall >= 2:
            if stall <= 4:
                # Fourier perturbation
                freq = rng.integers(1, max(2, n // 20))
                phase = rng.uniform(0, 2 * np.pi)
                x = np.arange(n) / max(1, n - 1)
                pert = 0.04 * np.cos(2 * np.pi * freq * x + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                else:
                    a = best_a.copy()
            elif stall <= 7:
                # Gaussian perturbation
                noise = rng.normal(0, 0.015 * stall, n)
                a = np.maximum(best_a + noise, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                else:
                    a = best_a.copy()
            else:
                # LP solution blend as restart
                if g_lp is not None:
                    a = 0.4 * best_a + 0.6 * g_lp
                    a = np.maximum(a, 0)
                    a = a / max(float(np.sum(a)), 1e-10)
                else:
                    mode = it % 10
                    a_div = diverse_init(n, rng, mode=mode)
                    a = 0.3 * a_div + 0.7 * best_a
                    a = np.maximum(a, 0)
                    a = a / max(float(np.sum(a)), 1e-10)
            stall = 0

    return best_a

                        a = a_new
                        patience = 0

            if verbose:
                print(f"    MD-LP it{it} lp#{lp_count}: obj={best_obj:.8f}, "
                      f"eta={eta:.5f}, pat={patience}, lp_t={lp_time:.2f}s, "
                      f"elapsed={time.monotonic()-start:.1f}s")

        # Adaptive step size
        if patience == 0 and it % 5000 == 0:
            eta = min(eta * 1.015, 0.25)

        # Hard restart
        if patience > 5000:
            if remaining > 15:
                g_lp, _ = solve_lp_minpeak(best_a)
                if g_lp is not None:
                    a = 0.5 * best_a + 0.5 * g_lp
                    a = np.maximum(a, 0)
                    a = a / max(float(np.sum(a)), 1e-10)
                else:
                    a = best_a.copy()
            else:
                a = best_a.copy()
            eta = max(eta_init * 0.5, 0.01)
            patience = 0

        if verbose and it % 50000 == 0 and it % lp_every != 0:
            elapsed = time.monotonic() - start
            print(f"    MD-LP it{it}: obj={best_obj:.8f}, eta={eta:.5f}, pat={patience}, t={elapsed:.1f}s")

    return best_a


def shape_sweep(n, time_budget=60.0, verbose=False):
    """
    Sweep over U-shape parameters to find a good initial shape.
    Tries different powers and bases for the parametric U-shape.
    """
    best_a = None
    best_obj = np.inf
    start = time.monotonic()

    params = []
    for power in [0.5, 0.75, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 6.0]:
        for base in [0.01, 0.02, 0.05, 0.1, 0.2]:
            params.append((power, base))

    rng = np.random.default_rng(999)
    # Add some random parameters
    for _ in range(20):
        power = rng.uniform(0.3, 8.0)
        base = rng.uniform(0.005, 0.3)
        params.append((power, base))

    for power, base in params:
        if time.monotonic() - start > time_budget * 0.6:
            break
        a = parametric_u_init(n, power=power, base=base)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose:
                print(f"    Shape (p={power:.2f}, b={base:.3f}): obj={obj:.8f}")

    # Also try arcsine and standard U
    a_arc = arcsine_init(n)
    obj_arc = compute_obj(a_arc)
    if obj_arc < best_obj:
        best_obj = obj_arc
        best_a = a_arc.copy()

    a_std = u_shaped_init(n, sharpness=3.0)
    obj_std = compute_obj(a_std)
    if obj_std < best_obj:
        best_obj = obj_std
        best_a = a_std.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Mirror descent + LP hybrid optimization.
    Strategy:
    1. Shape parameter sweep for good initialization
    2. Mirror descent + LP hybrid (main optimization)
    3. Intensive LP polishing
    4. Second MD round
    5. n-exploration with interpolation
    6. Diverse restarts
    7. Final polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_default = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_default = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = u_shaped_init(n_default, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default U-shaped init (n={n_default}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Shape sweep + quick warm-up ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 1: Shape Sweep ===")
        a_shape, obj_shape = shape_sweep(n_current, time_budget=min(30.0, remaining * 0.15), verbose=True)
        if a_shape is not None and obj_shape < best_score:
            best_score = obj_shape
            best_seq = a_shape.copy()
            print(f"  Shape sweep: obj={best_score:.8f} **")
        else:
            print(f"  Shape sweep best: {obj_shape:.8f} (parent={best_score:.8f})")

        # Quick MD warm-up
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_warm = mirror_descent(best_seq, time_budget=min(15.0, remaining * 0.1), eta_init=0.05)
            obj_warm = compute_obj(a_warm)
            print(f"  Warm-up MD: obj={obj_warm:.8f}")
            if obj_warm < best_score:
                best_score = obj_warm
                best_seq = a_warm.copy()
                print(f"  ** Improvement: {best_score:.8f}")

    # === Phase 2: Main MD-LP hybrid ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 2: MD-LP Hybrid (Main) ===")
        alloc = min(remaining * 0.38, 500.0)
        a_main = md_lp_hybrid(best_seq, time_budget=alloc, verbose=True, lp_every=20000, eta_init=0.04)
        obj_main = compute_obj(a_main)
        print(f"  MD-LP: obj={obj_main:.8f} (best={best_score:.8f})")
        if obj_main < best_score:
            best_score = obj_main
            best_seq = a_main.copy()
            print(f"  ** Improvement: {best_score:.8f}")

    # === Phase 3: Intensive LP polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print("\n=== Phase 3: LP Polishing ===")
        alloc = min(remaining * 0.22, 280.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True, max_iters=70)
        obj_lp = compute_obj(a_lp)
        print(f"  LP: obj={obj_lp:.8f} (best={best_score:.8f})")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** Improvement: {best_score:.8f}")

    # === Phase 4: Second MD round ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 4: MD Round 2 ===")
        alloc = min(remaining * 0.18, 200.0)
        a_md2 = md_lp_hybrid(best_seq, time_budget=alloc, verbose=True, lp_every=15000, eta_init=0.03)
        obj_md2 = compute_obj(a_md2)
        print(f"  MD2: obj={obj_md2:.8f} (best={best_score:.8f})")
        if obj_md2 < best_score:
            best_score = obj_md2
            best_seq = a_md2.copy()
            print(f"  ** Improvement: {best_score:.8f}")

    # === Phase 5: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 5: n Exploration ===")
        n_best = len(best_seq)

        n_options = set()
        for delta in [-200, -100, -50, 50, 100, 200, 400, 600]:
            n_try = n_best + delta
            if 300 <= n_try <= 2500:
                n_options.add(n_try)
        for n_fixed in [400, 500, 600, 700, 900, 1000, 1200, 1500, 1800, 2000]:
            if abs(n_fixed - n_best) > 80:
                n_options.add(n_fixed)

        n_options = sorted(n_options)[:7]
        print(f"  Trying n: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s

            # Quick optimization for this n
            if remaining > 40:
                a_try = mirror_descent(a_try, time_budget=min(12.0, remaining * 0.15), eta_init=0.05)
            if remaining > 25:
                a_try = lp_polish(a_try, time_budget=min(20.0, remaining * 0.3), max_iters=15)

            obj_try = compute_obj(a_try)
            print(f"  n={n_try}: obj={obj_try:.8f}")
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** Improvement: {best_score:.8f}")

    # === Phase 6: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print("\n=== Phase 6: Diverse Restarts ===")
        n_cur = len(best_seq)

        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            mode = (r * 3) % 10
            a_r = diverse_init(n_cur, rng, mode=mode)
            a_r = 0.2 * a_r + 0.8 * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / max(float(np.sum(a_r)), 1e-10)

            alloc = min(remaining * 0.35, 60.0)
            a_r = md_lp_hybrid(a_r, time_budget=alloc, lp_every=10000, eta_init=0.04)

            obj_r = compute_obj(a_r)
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  ** Improvement: {best_score:.8f}")

    # === Phase 7: Final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        print("\n=== Phase 7: Final Polish ===")
        alloc = min(remaining - 10, 100.0)
        if alloc > 15:
            a_final = lp_polish(best_seq, time_budget=alloc, verbose=True, max_iters=40)
            obj_final = compute_obj(a_final)
            print(f"  Final: obj={obj_final:.8f} (best={best_score:.8f})")
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  ** Improvement: {best_score:.8f}")

    # Extra LP if time remains
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_extra = lp_polish(best_seq, time_budget=min(remaining - 5, 50.0), max_iters=20)
        obj_extra = compute_obj(a_extra)
        if obj_extra < best_score:
            best_score = obj_extra
            best_seq = a_extra.copy()
            print(f"  Extra LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
