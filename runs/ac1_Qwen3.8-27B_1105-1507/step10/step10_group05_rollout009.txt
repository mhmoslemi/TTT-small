Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set LP (reduced constraints), spectral initialization,

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def sparse_gradient_step(a, step_size, top_k=20):
    """
    Take a gradient step using only the top-k convolution positions.
    The subgradient of max_k c[k] (where c = conv(a,a)) at a is:
    2 * (delta_{k*} * a) for k* = argmax, truncated to length n.
    For a smoothed version using top-k: weighted average of subgradients.
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Get top k positions
    top_idx = np.argsort(conv)[::-1][:top_k]
    top_vals = conv[top_idx]
    
    # Softmax weights
    max_val = top_vals[0]
    weights = np.exp(10 * (top_vals - max_val))
    weights = weights / weights.sum()
    
    # Compute gradient (weighted sum of subgradients)
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        # Subgradient at position k: (delta_k * a)[:n] = a[max(0, k-n+1):min(n, k+1)] shifted
        # Actually: (delta_k * a)[j] = a[k-j] if 0 <= k-j < n, else 0
        # So grad[j] += w * a[k-j] for valid j
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += w * a[idx]
    
    grad *= 2.0  # factor of 2 from the square

            t_vals = [0.3, 0.5, 0.7, 1.0, 1.2]

        best_t_obj = best_obj + 1e-14
        best_a_new = None

        for t in t_vals:
            if t <= 1.0:
                a_try = (1 - t) * a + t * g_lp
            else:
                a_try = t * g_lp - (t - 1) * a
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_new = a_try

        if best_a_new is not None and best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a_new.copy()
            stagnant = 0
            if verbose and it % 10 == 0:
                print(f"  LP it{it}: obj={best_obj:.8f}, M_lp={M_lp:.6f}")
        else:
            stagnant += 1
            a = best_a.copy()
            if stagnant > 8:
                # Diversification: blend with perturbed version
                strength = 0.02 + 0.01 * min(stagnant - 8, 10)
                x = np.linspace(0, 1, n)
                pert = 1 + strength * np.sin(2 * np.pi * rng.uniform(1, 5) * x + rng.uniform(0, 2 * np.pi))
                a = best_a * pert
                a = np.maximum(a, 0)
                a = a / a.sum()
                prev_g = None
                stagnant = 0

        prev_g = g_lp.copy()

    return best_a


def spectral_init(n, rng):
    """Generate initial sequence using spectral flattening idea."""
    # Start with random positive sequence
    a = rng exponential(1.0, n)
    a = a / a.sum()
    
    # Apply a few spectral flattening steps
    for _ in range(5):
        N = 1
        while N < 2 * n - 1:
            N <<= 1
        F = fft(a, N)
        mag = np.abs(F)
        # Flatten: set all magnitudes to mean (except DC)
        mean_mag = np.mean(mag[1:])
        new_mag = mag.copy()
        new_mag[1:] = mean_mag
        # Also set the negative frequency part
        for k in range(n // 2 + 1, N):
            new_mag[k] = new_mag[N - k]
        # Reconstruct with original phase
        phase = np.angle(F)
        F_new = new_mag * np.exp(1j * phase)
        a_new = ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
    
    return a


def resample_seq(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    from scipy.interpolate import interp1d
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    a_new = f(x_new)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def mirror_sym(a):
    """Make sequence mirror symmetric."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = float(np.sum(a_sym))
    if s < 1e-10:
        return a.copy()
    return a_sym / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: spectral init + Nesterov GD + active-set LP + multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    rng = np.random.default_rng(seed)

    # Get parent
    best_seq = None
    best_score = np.inf
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={best_seq.size}): obj={best_score:.8f}")
    except Exception:
        pass

    n_default = 800

    # If no parent, generate initial candidates
    if best_seq is None:
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default uniform (n={n_default}): obj={best_score:.8f}")

    # === Try spectral initialization to potentially find a better start ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        for n_try in [600, 800, 1000]:
            a_spectral = spectral_init(n_try, rng)
            obj_sp = compute_obj(a_spectral)
            # Quick GD polish
            if remaining > 50:
                a_sp = nesterov_smoothed_descent(a_spectral, time_budget=min(30, remaining * 0.2),
                                                  tau_init=30, tau_end=300)
                obj_sp = compute_obj(a_sp)
            if obj_sp < best_score:
                best_score = obj_sp
                best_seq = a_sp.copy()
                print(f"  Spectral init (n={n_try}): obj={best_score:.8f} **")
            else:
                print(f"  Spectral init (n={n_try}): obj={obj_sp:.8f}")
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

    # === Phase 1: Nesterov smoothed gradient descent ===
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.3, 250.0)
    if p1_budget > 30:
        print(f"\n=== Phase 1: Nesterov GD (budget={p1_budget:.0f}s) ===")
        a_gd = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                         tau_init=20.0, tau_end=800.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase 1 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 result: obj={obj_gd:.8f} (no improvement)")

    # === Phase 2: Active-set LP optimization ===
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.3, 300.0)
    if p2_budget > 30:
        print(f"\n=== Phase 2: Active-set LP (budget={p2_budget:.0f}s) ===")
        a_lp = lp_optimize(best_seq, time_budget=p2_budget, verbose=True, seed=seed,
                           n_constraints=250)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 2 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 result: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: GD + LP alternation ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: GD+LP alternation (budget={remaining:.0f}s) ===")
        # Alternating: GD for a bit, then LP for a bit
        cycles = 3
        for c in range(cycles):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            gd_b = min(remaining * 0.3, 60.0)
            lp_b = min(remaining * 0.4, 100.0)
            if gd_b > 10:
                best_seq = nesterov_smoothed_descent(best_seq, time_budget=gd_b,
                                                     tau_init=50.0, tau_end=1000.0)
                obj = compute_obj(best_seq)
                if obj < best_score:
                    best_score = obj
                    print(f"  P3 cycle {c} GD: obj={best_score:.8f}")
            if remaining > lp_b + 10 and lp_b > 10:
                best_seq = lp_optimize(best_seq, time_budget=lp_b, seed=seed + 100 * c,
                                       n_constraints=200)
                obj = compute_obj(best_seq)
                if obj < best_score:
                    best_score = obj
                    print(f"  P3 cycle {c} LP: obj={best_score:.8f} **")

    # === Phase 4: Multi-n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: Multi-n exploration ===")
        n_current = len(best_seq)
        n_candidates = sorted(set([
            n_current - 100, n_current - 50, n_current + 50, n_current + 100,
            n_current + 200, n_current - 200,
            500, 700, 900, 1100, 1300
        ]))
        n_candidates = [n for n in n_candidates if 300 <= n <= 2000]

        for n_try in n_candidates:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            a_try = resample_seq(best_seq, n_try)
            obj_try = compute_obj(a_try)

            # Quick optimization
            if remaining > 20 and obj_try < best_score + 0.05:
                alloc = min(remaining * 0.15, 50.0)
                if alloc > 10:
                    a_try = nesterov_smoothed_descent(a_try, time_budget=alloc * 0.5,
                                                      tau_init=40, tau_end=400)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_score and remaining > 30:
                        alloc2 = min(remaining * 0.1, 30.0)
                        if alloc2 > 10:
                            a_try = lp_optimize(a_try, time_budget=alloc2, seed=n_try,
                                                n_constraints=200)
                            obj_try = compute_obj(a_try)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Final intensive LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 5: Final LP polish ===")
        alloc = min(remaining * 0.6, 150.0)
        best_seq = lp_optimize(best_seq, time_budget=alloc, verbose=True, seed=9999,
                               n_constraints=300)
        obj = compute_obj(best_seq)
        if obj < best_score:
            best_score = obj
            print(f"  Final LP: obj={best_score:.8f} **")

    # === Phase 6: Final GD with high tau ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 60.0)
        best_seq = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                             tau_init=100.0, tau_end=2000.0)
        obj = compute_obj(best_seq)
        if obj < best_score:
            best_score = obj
            print(f"  Final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
