def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):

    n = len(v)

        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_tr_sscp(a, r, M_target=None):
    """Solve trust region SOCP."""
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)
    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta
    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
        cp.norm(delta, 2) <= r,
    ]
    if M_target is not None:
        constraints.append(M <= M_target)
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_direction(a, M_target=None):
    """Solve LP: min M s.t. a*g[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

        constraints.append(M <= M_target)
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def brent_line_search(a, direction, best_obj, max_iter=50):
    """Brent's line search along direction, projecting to simplex at each step."""
    n = len(a)

    def obj_func(t):
        a_try = a + t * direction
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            return 1e10
        a_try = a_try / s
        return compute_obj(a_try)

    res = minimize_scalar(obj_func, bounds=(0.0, 2.5), method='bounded',
                          options={'xatol': 0.01, 'maxiter': max_iter})
    if res.success and res.fun < best_obj:
        return res.x, float(res.fun)
    return 0.0, best_obj

            a = np.maximum(a, 0)
            s_n = a.sum()
            if s_n > 1e-10:
                a = a / s_n
            best_obj = best_alpha_obj
            best_a = a.copy()
            no_improve = 0
            r = min(r * 1.1, 2.0)
        else:
            no_improve += 1
            r *= 0.6
            if no_improve >= 6:
                if r < 0.02:
                    a = best_a * np.exp(rng.normal(0, 0.01, n))
                    a = np.maximum(a, 0)
                    s_p = a.sum()
                    if s_p > 1e-10:
                        a = a / s_p
                    r = r_init
                    no_improve = 0
                    if it > 100:
                        break
                else:
                    r = max(r * 0.5, 0.02)
        it += 1
        if verbose and it % 10 == 0:
            print(f"    TR it{it}: obj={best_obj:.8f}, r={r:.4f}, t={time.monotonic()-start:.0f}s")
    return best_a


def spectral_shaping(a, n_iters=50, verbose=False):
    """Flatten the power spectrum to reduce peak autocorrelation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    N = 1
    while N < 4 * n:
        N <<= 1

    for it in range(n_iters):
        A = fft(a, N)
        P = np.abs(A) ** 2
        # Target: reduce peaks in autocorrelation by flattening spectrum
        # Use iterative amplitude modification
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        # Reduce high-power spectral components
        threshold = np.percentile(P[1:], 70)
        P_new = P.copy()
        P_new[1:] = np.minimum(P_new[1:], threshold * 1.5)
        # Keep phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try softer reduction
            P_new2 = P.copy()
            P_new2[1:] = np.minimum(P_new2[1:], np.percentile(P[1:], 85) * 1.3)
            A_new2 = np.sqrt(P_new2) * np.exp(1j * phase)
            a_new2 = ifft(A_new2).real[:n]
            a_new2 = np.maximum(a_new2, 0)
            s_new2 = a_new2.sum()
            if s_new2 > 1e-10:
                a_new2 = a_new2 / s_new2
                obj_new2 = compute_obj(a_new2)
                if obj_new2 < best_obj:
                    best_obj = obj_new2
                    best_a = a_new2.copy()
                    a = a_new2
                else:
                    break
    return best_a


def smoothed_gradient_descent(a, time_budget=100.0, tau_init=15.0, tau_end=300.0):
    """Smoothed gradient descent with increasing temperature."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 4.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.8)
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        lr = max(1e-10, lr * 0.999)
        it += 1
    return best_a


def subgradient_refine(a, time_budget=80.0, K=10):
    """Subgradient method with top-K peak weighting."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 3.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 10.0 / max_pv)
        weights /= weights.sum()
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 150:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.998
        it += 1
    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def coordinate_descent(a, time_budget=100.0, verbose=False):
    """Cyclic coordinate descent with exact line search on each coordinate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    for epoch in range(1000):
        if time.monotonic() - start > time_budget:
            break
        # Random permutation of coordinates
        perm = rng.permutation(n)
        for j in perm:
            if time.monotonic() - start > time_budget:
                break
            # Try a few candidate values for a[j]
            old_val = a[j]
            if old_val < 1e-10:
                candidates = [old_val * 0.5, old_val + 1e-5, 1e-4]
            else:
                candidates = [old_val * 0.7, old_val * 0.85, old_val * 1.15,
                              old_val * 1.3, old_val * 1.5, old_val * 0.5]
            best_cand = old_val
            best_cand_obj = best_obj
            for cv in candidates:
                if cv < 0:
                    continue
                a[j] = cv
                s_tmp = a.sum()
                if s_tmp < 1e-10:
                    a[j] = old_val
                    continue
                a_norm = a / s_tmp
                obj_tmp = compute_obj(a_norm)
                if obj_tmp < best_cand_obj - 1e-14:
                    best_cand_obj = obj_tmp
                    best_cand = cv
            a[j] = best_cand
            s_tmp = a.sum()
            if s_tmp > 1e-10:
                a = a / s_tmp
            if best_cand_obj < best_obj - 1e-14:
                best_obj = best_cand_obj
                best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Hybrid S-PROX + TR-SOCP + Spectral + Coordinate Descent.
    Multi-phase approach with adaptive time allocation and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Diversified Initialization (10% time) ===
    print(f"\n=== Phase 1: Diversified Initialization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.10, 120.0)
    print(f"  Budget: {p1_budget:.0f}s")

    candidates = [best_seq.copy()]

    # Spectral shaping on parent
    if p1_budget > 20:
        a_spec = spectral_shaping(best_seq, n_iters=30)
        obj_spec = compute_obj(a_spec)
        candidates.append(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()

    # Smoothed GD warmup
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 40.0)
        a_sgd = smoothed_gradient_descent(best_seq, time_budget=alloc,
                                          tau_init=10.0, tau_end=200.0)
        obj_sgd = compute_obj(a_sgd)
        candidates.append(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  GD warmup: obj={best_score:.8f} **")
        else:
            print(f"  GD warmup: obj={obj_sgd:.8f}")

    # Perturbed versions
    remaining = deadline - time.monotonic()
    for p_idx in range(3):
        if time.monotonic() > deadline - p1_budget * 0.3:
            break
        scale = 0.02 + 0.015 * p_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        candidates.append(a_pert)
        obj_pert = compute_obj(a_pert)
        print(f"  Pert{p_idx} (scale={scale:.3f}): obj={obj_pert:.8f}")

    # Sort candidates by objective, keep top 4
    candidates.sort(key=lambda x: compute_obj(x))
    top_candidates = candidates[:4]
    print(f"  Top candidate obj: {compute_obj(top_candidates[0]):.8f}")

    # === Phase 2: Main TR-SOCP optimization on best candidates ===
    print(f"\n=== Phase 2: TR-SOCP Main Optimization ===")

    print(f"  Budget: {p2_budget:.0f}s")

    # Optimize top 2 candidates with TR-SOCP

        if remaining < 40:
            break

        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Candidate {cand_idx} TR: obj={best_score:.8f} **")
        else:
