"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy:
1. Active-set LP for fast direction finding (only top-K constraints)
2. Nesterov-accelerated projected gradient with adaptive smoothing
3. Multi-start exploration with diverse initializations (symmetric, asymmetric, spiky)
4. Coarse-to-fine: optimize a shortened version first, then upsample
5. Length exploration (try n=700, 750, 800, 850, 900, 1000)
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix_banded(a, top_k_indices):
    """Build only the rows of the convolution matrix corresponding to top_k_indices."""
    n = len(a)
    A = np.zeros((len(top_k_indices), n))
    for i, k in enumerate(top_k_indices):
        # (a*g)[k] = sum_j a_j * g_{k-j}
        # So the k-th row of A has a_j at position (k-j) for valid j
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[i, j] = a[j]
    return A

def solve_active_lp(a, K=200):
    """Solve LP with only top-K active constraints."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a_norm = a / s
    conv = compute_conv(a_norm)
    
    # Get top K indices by convolution value
    top_k_indices = np.argsort(conv)[::-1][:K]
    
    A = build_conv_matrix_banded(a_norm, top_k_indices)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    return g_val / s_g
        except Exception:
            continue
    return None

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated projected gradient on the smoothed autocorrelation peak,
with periodic inner-subproblem direction finding via multiplicative weights,
multi-start exploration, and adaptive hyperparameter scheduling.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.maximum(v, 0)
    s = float(np.sum(v))
    if s < 1e-300:
        return np.ones(n) / n
    return v / s


def smoothed_obj_and_grad(a, tau):
    """
    Compute smoothed objective and gradient.
    Objective: sum_k p_k * (a*a)[k] where p = softmax(tau*(conv - max))
    Gradient: 2 * cross_correlate(p, a)
    Returns (obj, grad)
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    # Softmax weights
    shifted = tau * (conv - max_conv)
    # Clip to avoid underflow
    shifted = np.clip(shifted, -50, 0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # Smoothed objective
    obj = float(np.sum(p * conv))
    
    # Gradient: grad_j = 2 * sum_k p_k * a[k-j] (with a[m]=0 for invalid m)

    # = (p_rev * a)[j] where p_rev is p reversed and padded
    N = 1

    P = fft(p, N)
    A = fft(a, N)
    # cross_correlate(p, a)[j] = ifft(P * conj(A))[j]
    corr = ifft(P * np.conj(A)).real[:n]
    grad = 2.0 * corr
    
    return obj, grad, conv

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated projected gradient with adaptive smoothing,
fast inner-direction finding via multiplicative weights, multi-start
exploration, and final LP polish.

import time
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = float(np.sum(v))
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return v / s


def smoothed_grad(a, tau):
    """
    Compute gradient of smoothed max of (a*a).
    Returns (smoothed_obj, gradient_vector).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -50.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    smoothed_obj = float(np.sum(p * conv))
    
    # Gradient: grad_j = 2 * sum_k p[k] * a[k-j]
    # = 2 * cross_correlate(p, a)[j]
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    corr = ifft(P * np.conj(A)).real[:n]
    grad = 2.0 * corr
    
    return smoothed_obj, grad

def cross_convolve(a, b):
    """Cross-convolution a*b via FFT. Returns array of length 2n-1 (assumes same length)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]


def inner_multiplicative_weights(a, n_mw_iters=30, eta=None):
    """
    Compute approximate best-response direction using multiplicative weights.
    Solves: min_g max_k (a_norm * g)[k] s.t. g >= 0, sum(g) = 1
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    if eta is None:
        eta = 0.5 / np.sqrt(n)
    
    g = np.ones(n) / n
    
    for _ in range(n_mw_iters):
        # Compute (a_norm * g)
        cross = cross_convolve(a_norm, g)
        k_star = int(np.argmax(cross))
        
        # Subgradient: sg[j] = a_norm[k_star - j]
        indices = k_star - np.arange(n)
        sg = np.zeros(n)
        valid = (indices >= 0) & (indices < n)
        sg[valid] = a_norm[indices[valid]]
        
        # Multiplicative weights update
        # g[j] *= exp(-eta * sg[j])
        max_sg = float(np.max(sg))
        exponents = -eta * (sg - max_sg)  # Shift for stability
        weights = np.exp(exponents)
        g = g * weights
        g = project_simplex(g)
    
    return g

            valid = (indices >= 0) & (indices < n)
            A[i, valid] = a_norm[indices[valid]]
        
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        solved = False
        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_dir = g_val / s_g
                        solved = True
                        break
            except Exception:
                continue
        
        if not solved:
            break
        
        it += 1
        best_t = 0.1
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            a_try = (1.0 - t) * a + t * g_dir
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                best_t = t
                break
        
        if verbose := False:
            pass
    
    return best_a
