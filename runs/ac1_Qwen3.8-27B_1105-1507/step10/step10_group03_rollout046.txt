import numpy as np

from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """Gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def staircase_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9, lr_scale=2.0):
    """Nesterov accelerated gradient descent with staircase tau schedule."""

def solve_lp(a, solver_pref=None):
    """
    Solve the LP: min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    if n > 3000:
        return None, None

    # Build convolution matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if solver_pref is None:
        solver_pref = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]

    for solver in solver_pref:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue

    return None, None

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_init(n, mode, rng=None):
    """Create diverse initial sequences of length n."""
    if rng is None:

    if mode == 0:

    elif mode == 1:
        x = np.linspace(0, 1, n)

        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.0001))
        return a / a.sum()
    elif mode == 3:
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 4:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, np.pi, n)
        a = np.maximum(0.5 + 0.5 * np.cos(x), 0.001)
        return a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [2, 3, 5]:
            a += 0.2 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 4))
        return a / a.sum()
    elif mode == 8:
        x = np.linspace(0, 1, n)
        a = np.where((x > 0.2) & (x < 0.8), 1.0,
                     np.maximum(0.5 - np.abs(x - 0.5), 0.01))
        return a / a.sum()
    else:
        return np.ones(n) / n

def population_de(n, pop_size, time_budget, verbose=False, seed=42):
    """Differential evolution with GD refinement."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    pop = np.zeros((pop_size, n))
    for i in range(pop_size):
        pop[i] = make_init(n, mode=i % 9, rng=rng)

    objs = np.array([compute_obj(pop[i]) for i in range(pop_size)])
    best_idx = np.argmin(objs)
    best_a = pop[best_idx].copy()
    best_obj = objs[best_idx]

    gen = 0
    per_trial_budget = 2.0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < per_trial_budget * 2:
            break

        candidates = list(range(pop_size))
        r1, r2, r3 = rng.choice(candidates, 3, replace=False)
        F = 0.5 + 0.5 * rng.random()

        trial = pop[r1] + F * (pop[r2] - pop[r3])
        trial = np.maximum(trial, 0)
        st = float(np.sum(trial))
        if st < 1e-10:
            gen += 1
            continue
        trial = trial / st

        trial = staircase_gd(trial, time_budget=min(per_trial_budget, remaining * 0.3),
                             tau_levels=[20, 100, 500], verbose=False)
        obj_trial = compute_obj(trial)

        worst_idx = np.argmax(objs)
        if obj_trial < objs[worst_idx]:
            pop[worst_idx] = trial
            objs[worst_idx] = obj_trial

        if obj_trial < best_obj:
            best_obj = obj_trial
            best_a = trial.copy()

        gen += 1

    return best_a, best_obj
