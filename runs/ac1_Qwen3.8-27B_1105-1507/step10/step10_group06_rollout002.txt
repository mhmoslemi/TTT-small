"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based alternating minimization with adaptive line search,
Nesterov gradient polishing, and multi-restart exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_clarabel(a):
    """Solve the LP using CLARABEL. Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(g.value, 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(M.value)
    except Exception:
        pass
    
    # Fallback to other solvers
    for solver in [cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None


def line_search_blend(a, g, max_evals=20):
    """Find best t in [0,1] to blend a and g. Returns (best_a, best_t, best_obj)."""
    best_a = a.copy()
    best_obj = compute_obj(a)
    best_t = 0.0
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    lo, hi = 0.0, 1.0
    c = lo + (1 - gr) * (hi - lo)
    d = lo + gr * (hi - lo)
    
    a_c = (1 - c) * a + c * g
    a_c = np.maximum(a_c, 0)
    s_c = float(np.sum(a_c))
    obj_c = compute_obj(a_c / s_c) if s_c > 0 else np.inf
    
    a_d = (1 - d) * a + d * g
    a_d = np.maximum(a_d, 0)
    s_d = float(np.sum(a_d))
    obj_d = compute_obj(a_d / s_d) if s_d > 0 else np.inf
    
    evals = 2
    
    while hi - lo > 1e-6 and evals < max_evals:
        if obj_c < obj_d:
            hi = d
            d = c
            obj_d = obj_c
            c = lo + (1 - gr) * (hi - lo)
            a_c = (1 - c) * a + c * g
            a_c = np.maximum(a_c, 0)
            s_c = float(np.sum(a_c))
            obj_c = compute_obj(a_c / s_c) if s_c > 0 else np.inf
        else:
            lo = c
            c = d
            obj_c = obj_d
            d = lo + gr * (hi - lo)
            a_d = (1 - d) * a + d * g
            a_d = np.maximum(a_d, 0)
            s_d = float(np.sum(a_d))
            obj_d = compute_obj(a_d / s_d) if s_d > 0 else np.inf
        evals += 1
    
    t_best = (lo + hi) / 2
    a_best = (1 - t_best) * a + t_best * g
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))
    if s_best > 0:
        a_best = a_best / s_best
        obj_best = compute_obj(a_best)
        if obj_best < best_obj:
            best_obj = obj_best
            best_a = a_best
            best_t = t_best
    
    # Also check t=1
    a_1 = g.copy()
    obj_1 = compute_obj(a_1)
    if obj_1 < best_obj:
        best_obj = obj_1
        best_a = a_1
        best_t = 1.0
    
    return best_a, best_t, best_obj


def smoothed_gradient(a, tau):
    """Smoothed gradient of the objective using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 4 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def nesterov_polish(a, time_budget, tau_init=50.0, tau_max=5000.0):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    base_lr = 1.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))
        
        if g_norm > 1e-15:
            adaptive_lr = 1.0 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.05 * base_lr), 2.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adaptive_lr
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9999

def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 8
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
