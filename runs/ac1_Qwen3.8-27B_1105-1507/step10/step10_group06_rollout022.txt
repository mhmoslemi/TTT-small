```python
def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_lp_fast(a):
    """
    Solve: min M s.t. (a*g)_k <= M for all k, sum(g)=1, g>=0.
    Using scipy linprog with HiGHS.
    Variables: [g_0, ..., g_{n-1}, M]
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Objective: minimize M (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A @ g - M <= 0 for each k
    # A is (2n-1) x n
    A_ineq = np.zeros((2 * n - 1, n + 1))

    A_ineq[:, -1] = -1.0
    b_ineq = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]

                  bounds=bounds, method='highs')
    
    if res.success:
```

```python
def fw_best_response(a, max_iter=100, tol=1e-8):
    """
    Frank-Wolfe method for: min_g max_k (a@g)_k s.t. g in simplex.
    Returns g_opt.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Initialize g as uniform
    g = np.ones(n) / n
    
    for it in range(max_iter):
        conv = np.convolve(a_norm, g)  # length 2n-1
        max_val = np.max(conv)
        
        # Find active set (within tolerance)
        active_tol = 1e-6
        active = conv >= max_val - active_tol
        p = active.astype(float) / active.sum()
        
        # Compute subgradient
        # grad_j = sum_k p_k * a_{k-j} (for valid k-j)
        # This is essentially a weighted sum of shifted a's
        grad = np.zeros(n)
        for k in range(len(conv)):
            if p[k] > 0:
                # (a@g)_k = sum_j a[k-j] * g_j for 0 <= k-j < n
                # d/dg_j (a@g)_k = a[k-j] if 0 <= k-j < n
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                # j ranges from lo to hi, and a[k-j] is valid
                indices = np.arange(lo, hi + 1)
                a_vals = a_norm[k - indices]
                grad[indices] += p[k] * a_vals
        
        # FW step: find j* = argmin_j grad_j
        j_star = np.argmin(grad)
        e_j = np.zeros(n)
        e_j[j_star] = 1.0
        
        # Step size (exact line search or diminishing)
        # Exact: minimize max_k (a @ ((1-t)*g + t*e_j))
        # For simplicity, use diminishing step size
        t = 2.0 / (it + 2.0)
        
        g_new = (1 - t) * g + t * e_j
        g_new = np.maximum(g_new, 0)
        g_new = g_new / np.sum(g_new)
        
        conv_new = np.convolve(a_norm, g_new)
        if abs(np.max(conv_new) - max_val) < tol:
            g = g_new
            break
        g = g_new
    
    return g
```

```python
def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def simplex_gd(a, time_budget, tau_init=50.0, tau_max=5000.0):
    """GD with exact line search on the simplex for minimizing max(a*a)."""
    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    tau = tau_init
    lr = 1.0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Smoothed softmax
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / np.sum(exp_vals)
        
        # Gradient of smoothed max: 2 * conv(a, p_reversed)
        # (a*a)_k gradient w.r.t. a_j is a_{k-j} + a_{k-j} = 2*a_{k-j}
        # Smoothed: 2 * sum_k p_k * a_{k-j}
        N = 1
        while N < 3 * n:
            N <<= 1
        A_fft = fft(a, N)
        P_fft = fft(p, N)
        grad = 2.0 * np.real(ifft(P_fft * np.conj(A_fft)))[:n]
        
        # Gradient step
        a_new = project_simplex(a - lr * grad)
        
        # Exact line search (backtracking)
        obj_new = compute_obj(a_new)
        while obj_new > best_obj and lr > 1e-10:
            lr *= 0.5
            a_new = project_simplex(a - lr * grad)
            obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        
        # Increase tau (less smoothing) over time
        tau = min(tau * 1.05, tau_max)
        lr *= 1.01  # Slightly increase lr
    
    return best_a
```

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast LP-based alternating minimization (scipy HiGHS), dual-guided perturbation,
multi-start diverse initialization, adaptive line search, simplex-projected gradient
polishing, and multi-length exploration to escape local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_lp_matrices(a_norm):
    """Build LP constraint matrices for the best-response problem."""
    n = len(a_norm)
    # A_ineq: (2n-1) x (n+1), each col j has a in rows j..j+n-1, last col is -1
    A_ineq = np.zeros((2 * n - 1, n + 1))
    for j in range(n):
        A_ineq[j:j + n, j] = a_norm
    A_ineq[:, -1] = -1.0
    b_ineq = np.zeros(2 * n - 1)
    # A_eq: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    return A_ineq, b_ineq, A_eq, b_eq


def solve_lp_fast(a):
    """
    Solve LP: min M s.t. (a_norm*g)_k <= M for all k, sum(g)=1, g>=0.
    Uses scipy linprog with HiGHS. Returns (g_opt, dual_y) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A_ineq, b_ineq, A_eq, b_eq = build_lp_matrices(a_norm)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
                # Extract dual variables for the inequality constraints (they give y)
                dual_y = None
                if res.ineqlin is not None and res.ineqlin.marginals is not None:
                    dual_y = np.maximum(-res.ineqlin.marginals, 0)
                    dy_sum = float(np.sum(dual_y))
                    if dy_sum > 1e-10:
                        dual_y = dual_y / dy_sum
                return g, dual_y
    except Exception:
        pass
    return None, None


def line_search_blend(a, g, max_evals=30):
    """
    Find best blending factor t in [0,1] minimizing obj((1-t)*a + t*g).
    Uses ternary search followed by golden section refinement.
    Returns (best_t, best_a, best_obj).
    """
    n = len(a)
    best_t = 0.0
    best_obj = compute_obj(a)
    best_a = a.copy()

    def obj_at_t(t):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return np.inf
        a_try = a_try / s
        return compute_obj(a_try)

    # Coarse ternary search
    lo, hi = 0.0, 1.0
    for _ in range(12):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
        if best_obj > f1:
            best_obj = f1
        if best_obj > f2:
            best_obj = f2

    # Fine search around the best region
    mid = (lo + hi) / 2.0
    step = (hi - lo) / 10.0
    for i in range(11):
        t = lo + i * step
        f = obj_at_t(t)
        if f < best_obj:
            best_obj = f
            best_t = t

    a_best = (1.0 - best_t) * a + best_t * g
    a_best = np.maximum(a_best, 0)
    s = float(np.sum(a_best))
    if s > 1e-10:
        a_best = a_best / s
        best_obj = compute_obj(a_best)

    return best_t, a_best, best_obj


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smoothed_subgradient(a, p):
    """
    Compute subgradient of sum_k p_k * (a*a)_k with respect to a.
    Result: 2 * cross_correlation(p, a)[:n]
    p has length 2n-1 (the convolution indices).
    """
    n = len(a)
    N = 1
    while N < 2 * (2 * n - 1) - 1:
        N <<= 1
    A_fft = fft(a, N)
    P_fft = fft(p, N)
    # Cross-correlation: sum_k p_k * a_{k-j}
    # = ifft(fft(p) * conj(fft(a)))
    corr = np.real(ifft(P_fft * np.conj(A_fft)))[:n]
    return 2.0 * corr


def simplex_gd_polish(a, time_budget, tau_init=30.0, tau_max=3000.0, use_nesterov=True):
    """
    Projected gradient descent with adaptive smoothing on the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = tau_init
    lr_base = 4.0 / np.sqrt(n)
    lr = lr_base
    v = np.zeros(n)
    beta = 0.9
    patience = 0

    while time.monotonic() - start < time_budget:
        if use_nesterov:
            y = a + beta * v
            y = project_simplex(y)
        else:
            y = a.copy()

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        grad = smoothed_subgradient(y, p)
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break

        # Adaptive step size
        eff_lr = min(lr, 1.0 / (g_norm + 1e-10))

        a_new = project_simplex(y - eff_lr * grad)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
            if use_nesterov:
                v = a_new - y
                a = a_new
            else:
                a = a_new
            lr = min(lr * 1.02, 2.0 * lr_base)
        else:
            patience += 1
            lr *= 0.9
            if use_nesterov:
                v = 0.9 * v - eff_lr * grad
                a = project_simplex(y + v)
            else:
                a = y

        tau = min(tau * 1.03, tau_max)

        if patience > 200:
            # Reset
            a = best_a.copy()
            v = np.zeros(n)
            lr = lr_base
            tau = max(tau_init, tau * 0.5)
            patience = 0

    return best_a


def dual_guided_perturb(a, dual_y, rng, strength=0.05):
    """
    Perturb a in a direction guided by dual variables.
    The dual tells us which lags are active; we try to reduce them.
    """
    n = len(a)
    if dual_y is None:
        # Random Fourier perturbation
        pert = rng.standard_normal(n) * strength
        a_new = a * np.exp(pert)
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            return a.copy()
        return a_new / s

    # Compute gradient direction: increase a where it reduces active lags
    # grad_j = 2 * sum_k dual_y_k * a_{k-j}
    grad = smoothed_subgradient(a, dual_y)
    # Move in the negative gradient direction (reduces the active lags)
    a_new = a - strength * grad / (np.linalg.norm(grad) + 1e-10)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def fourier_perturb(a, rng, strength=0.03):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = max(3, n // 100)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / (k ** 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_alternate(a, time_budget, seed=42, verbose=False, use_line_search=True):
    """
    Main LP-based alternating minimization with adaptive blending.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    stall_count = 0
    t_blend = 0.5

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break

        g_lp, dual_y = solve_lp_fast(a)
        if g_lp is None:
            # Fallback: use uniform
            g_lp = np.ones(n) / n

        iter_count += 1

        if use_line_search:
            t_best, a_try, obj_try = line_search_blend(a, g_lp)
        else:
            a_try = (1.0 - t_blend) * a + t_blend * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            stall_count = 0
            if use_line_search:
                t_blend = max(0.1, min(0.9, t_best))
            if verbose and iter_count % 50 == 0:
                print(f"    LP alt {iter_count}: obj={best_obj:.10f}, t={t_blend:.4f}")
        else:
            stall_count += 1
            if stall_count >= 4:
                # Perturb to escape
                strength = 0.02 + 0.01 * min(stall_count - 4, 5)
                a = dual_guided_perturb(best_a, dual_y, rng, strength=strength)
                t_blend = max(0.2, min(0.8, t_blend * 1.1))
                stall_count = 0
            elif stall_count >= 2:
                a = fourier_perturb(best_a, rng, strength=0.02)
                stall_count = 1

    if verbose:
        print(f"    LP alternating: {iter_count} iterations, best_obj={best_obj:.10f}")
    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    from scipy.interpolate import PchipInterpolator
    try:
        x_old = np.arange(n, dtype=float)
        x_new = np.linspace(0, n - 1, new_n)
        interp = PchipInterpolator(x_old, a)
        a_new = interp(x_new)
    except Exception:
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, new_n)
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, seed):
    """Generate a random non-negative sequence."""
    rng = np.random.default_rng(seed)
    a = rng.exponential(1.0, n)
    return a / a.sum()


def structured_init(n, seed, mode=0):
    """Generate structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    if mode == 0:
        # Cosine band
        a = np.exp(-0.5 * ((x - n / 2) / (n / 4)) ** 2)
        a += 0.1 * rng.standard_normal(n)
    elif mode == 1:
        # Comb pattern
        a = np.ones(n)
        period = max(2, n // 20)
        a[period::period] *= 2.0
        a += 0.05 * rng.standard_normal(n)
    elif mode == 2:
        # Random low-frequency
        freqs = rng.uniform(1, 5, 3)
        phases = rng.uniform(0, 2 * np.pi, 3)
        a = np.ones(n)
        for f, ph in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x / n + ph)
    else:
        a = rng.uniform(0.5, 1.5, n)
    a = np.maximum(a, 0)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating minimization,
    gradient polishing, multi-start, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    # Load parent if available
    try:
        a_parent = np.asarray(height_sequence_1, dtype=np.float64)
        a_parent = np.maximum(a_parent, 0)
        if a_parent.size == 0 or a_parent.sum() < 0.01:
            raise ValueError("Invalid parent")
        n_init = a_parent.size
        a_parent = a_parent / a_parent.sum()
        obj_parent = compute_obj(a_parent)
        print(f"Parent loaded: n={n_init}, obj={obj_parent:.10f}")
    except Exception:
        n_init = 800
        a_parent = np.ones(n_init) / n_init
        obj_parent = compute_obj(a_parent)
        print(f"Using default init: n={n_init}, obj={obj_parent:.10f}")

    best_a = a_parent.copy()
    best_obj = obj_parent
    rng = np.random.default_rng(seed)

    # === Phase 1: Multi-start initialization (60s) ===
    print(f"\n=== Phase 1: Multi-start Init ===")
    p1_budget = min(60.0, budget_s * 0.04)
    n_starts = 6
    per_start = p1_budget / n_starts

    inits = [a_parent.copy()]
    for i in range(1, n_starts):
        if i == 1:
            a_s = random_init(n_init, seed + i * 100)
        else:
            a_s = structured_init(n_init, seed + i * 100, mode=(i - 1) % 4)
        a_s = lp_alternate(a_s, time_budget=per_start * 0.6, seed=seed + i,
                           verbose=False, use_line_search=False)
        inits.append(a_s)

    for i, a_s in enumerate(inits):
        obj_s = compute_obj(a_s)
        if obj_s < best_obj:
            best_obj = obj_s
            best_a = a_s.copy()
            print(f"  Start {i}: obj={obj_s:.10f} **")
        else:
            print(f"  Start {i}: obj={obj_s:.10f}")

    # === Phase 2: Main LP alternating with line search (primary workhorse) ===
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.45, 500.0)
    a_p2 = lp_alternate(best_a, time_budget=p2_budget, seed=seed,
                        verbose=True, use_line_search=True)
    obj_p2 = compute_obj(a_p2)
    if obj_p2 < best_obj:
        best_obj = obj_p2
        best_a = a_p2.copy()
        print(f"  Phase 2 result: obj={best_obj:.10f} **")
    else:
        print(f"  Phase 2 result: obj={obj_p2:.10f}")

    # === Phase 3: Gradient polishing ===
    print(f"\n=== Phase 3: Gradient Polishing ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = min(remaining * 0.15, 150.0)
        a_p3 = simplex_gd_polish(best_a, time_budget=p3_budget,
                                  tau_init=20.0, tau_max=2000.0)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3.copy()
            print(f"  Phase 3 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 3 result: obj={obj_p3:.10f}")

    # === Phase 4: Second LP pass with perturbation ===
    print(f"\n=== Phase 4: Second LP Pass ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        p4_budget = min(remaining * 0.2, 200.0)
        a_pert = fourier_perturb(best_a, rng, strength=0.015)
        a_p4 = lp_alternate(a_pert, time_budget=p4_budget, seed=seed + 777,
                            verbose=True, use_line_search=True)
        obj_p4 = compute_obj(a_p4)
        if obj_p4 < best_obj:
            best_obj = obj_p4
            best_a = a_p4.copy()
            print(f"  Phase 4 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 4 result: obj={obj_p4:.10f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    n_options = sorted(set([
        n_best - 50, n_best + 50,
        n_best - 100, n_best + 100,
        max(300, n_best - 200), min(2000, n_best + 200),
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500 and n != n_best]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_to_length(best_a, n_try)
        alloc = min(remaining * 0.2, 100.0)
        if alloc < 30:
            break

        a_try = lp_alternate(a_try, time_budget=alloc * 0.6, seed=n_try,
                             verbose=False, use_line_search=True)
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_try = simplex_gd_polish(a_try, time_budget=min(remaining * 0.3, 30.0),
                                       tau_init=50.0, tau_max=1000.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: obj={best_obj:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 6: Final LP refinement ===
    print(f"\n=== Phase 6: Final LP Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p6_budget = min(remaining * 0.5, 120.0)
        a_p6 = lp_alternate(best_a, time_budget=p6_budget, seed=seed + 12345,
                            verbose=False, use_line_search=True)
        obj_p6 = compute_obj(a_p6)
        if obj_p6 < best_obj:
            best_obj = obj_p6
            best_a = a_p6.copy()
            print(f"  Final LP: obj={best_obj:.10f} **")
        else:
            print(f"  Final LP: obj={obj_p6:.10f}")

    # === Phase 7: Ultra-final gradient polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        p7_budget = min(remaining * 0.6, 40.0)
        a_p7 = simplex_gd_polish(best_a, time_budget=p7_budget,
                                  tau_init=100.0, tau_max=5000.0)
        obj_p7 = compute_obj(a_p7)
        if obj_p7 < best_obj:
            best_obj = obj_p7
            best_a = a_p7.copy()
            print(f"  Final GD: obj={best_obj:.10f} **")
        else:
            print(f"  Final GD: obj={obj_p7:.10f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_obj:
        best_obj = val
    elif val < np.inf:
        # Use evaluated value
        best_obj = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
```
