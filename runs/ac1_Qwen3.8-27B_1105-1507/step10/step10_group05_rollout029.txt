Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
 Uses FFT-based subgradient descent with Nesterov acceleration,
 LP refinement, spectral initialization, and multi-length exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def subgradient_of_conv_peak(a, peak_indices):
    """
    Compute subgradient of max_k b[k] where b = conv(a,a).
    grad[j] = 2 * sum_{k in peak_indices} a[k-j] (where out-of-range = 0)
    """
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        # grad[j] += 2 * a[k-j] if 0 <= k-j <= n-1
        # This is equivalent to: grad += 2 * flip_shift(a, k)
        # flip_shift(a, k)[j] = a[k-j] if valid
        lo = max(0, k - (n - 1))

        # a[k-j] for j from lo to hi means a[k-lo] down to a[k-hi]
        # So grad[lo:hi+1] += 2 * a[k-hi : k-lo+1][::-1]
        a_slice = a[k - hi : k - lo + 1]  # This gives a[k-hi], ..., a[k-lo]
        grad[lo:hi + 1] += 2.0 * a_slice[::-1]
    return grad

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based mirror descent with Nesterov acceleration, LP refinement,
spectral initialization, and multi-length exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time
import math

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return float(2.0 * n * np.max(conv))

def compute_grad_smoothed(a, p, N=None):
    """
    Compute gradient of smoothed max of conv(a,a).
    p is the smoothing weight distribution over convolution indices.
    Returns gradient of length len(a).
    """
    n = len(a)
    if N is None:
        N = 1
        while N < 2 * n:
            N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad = 2.0 * irfft(P * np.conj(A), N)[:n]
    return grad

def mirror_descent(a_init, time_budget, seed=42, verbose=False):

    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Hyperparameters
    eta = 2.0 / math.sqrt(n)
    tau = 20.0
    momentum = 0.5
    v = np.zeros(n)
    
    N = 1
    while N < 2 * n:
        N <<= 1
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        # Compute convolution
        b = compute_conv(a)
        max_b = float(np.max(b))
        
        # Smoothing weights
        t = min(tau, 300.0)
        shifted = t * (b - max_b)
        shifted = np.clip(shifted, -50, 0)
        p = np.exp(shifted)
        ps = p.sum()
        if ps < 1e-300:
            idx = int(np.argmax(b))
            p = np.zeros(len(b))
            p[idx] = 1.0
        else:

        y = np.maximum(y, 1e-15)

        pys = py.sum()
        if pys < 1e-300:

            py = np.zeros(len(by))
            py[idx] = 1.0

def solve_lp_minpeak(a, warm_start=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix: A @ g = conv(a_norm, g)
    # Shape: (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # Column j: a_norm contributes to rows j, j+1, ..., j+n-1
        # A[k, j] = a_norm[k-j] if 0 <= k-j <= n-1
        end = min(2 * n - 1, j + n)
        A[j:end, j] = a_norm[:end - j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val /= s
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None

        if g_lp is None:
            break
        
        # Try different blend factors
        best_t_obj = best_obj
        best_t = 0.5

def spectral_init(n, mode=0, seed=42):

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
