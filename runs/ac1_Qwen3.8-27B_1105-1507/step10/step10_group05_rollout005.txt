"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based fixed-point iteration with adaptive step size, 
coordinate-wise polishing, spectral initialization, and multi-length search.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build convolution matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] for k in [j, j+n-1]
        A[j:j + n, j] = a
    return A


def solve_lp(a, solver_list=None):
    """
    Solve: min M s.t. conv(a_normalized, g) <= M, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.ECOS, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()

    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    elif mode == 1:
        a = np.exp(-2.0 * x * (1.0 - x))
    elif mode == 2:
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x)
    else:
        a = 1.0
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def fourier_perturb(a, rng, strength=0.04):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    x = np.arange(n)
    n_modes = min(6, n // 20 + 1)
    for k in range(1, n_modes + 1):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * x + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(a_new.sum())
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def optimize_n(n, init=None, lp_budget=200, poly_budget=80, verbose=False, seed=0):
    """Optimize for a specific n. Returns (best_seq, best_obj)."""
    if init is not None:
        a = resample(init, n)
    else:
        a = spectral_init(n, mode=n % 4)
    a = a / a.sum()

    obj_initial = compute_obj(a)
    if verbose:
        print(f"    n={n}: init obj={obj_initial:.8f}")

    # LP iteration
    a = lp_iteration(a, budget=lp_budget, verbose=verbose, seed=seed)
    obj_lp = compute_obj(a)
    if verbose:
        print(f"    n={n}: after LP obj={obj_lp:.8f}")

    # Coordinate polish
    a = coordinate_polish(a, budget=poly_budget, verbose=verbose)
    obj_final = compute_obj(a)
    if verbose:
        print(f"    n={n}: after Poly obj={obj_final:.8f}")

    return a, obj_final


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP fixed-point + coordinate polish + multi-n exploration.
    """
    start_time = _time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    n_list = [600, 650, 700, 750, 800, 850, 900, 950, 1000, 1050, 1100, 1200]
    np.random.seed(12345)

    # --- Phase 1: Try multiple n with initial LP ---
    print("=== Phase 1: Multi-n LP optimization ===")
    phase1_results = []

    for n in n_list:
        remaining = deadline - _time.monotonic()
        if remaining < 60:
            break

        lp_b = min(remaining * 0.12, 120.0)
        poly_b = min(remaining * 0.03, 30.0)
        if lp_b < 15:
            break

        a_try, obj_try = optimize_n(
            n, init=None,
            lp_budget=lp_b, poly_budget=poly_b,
            verbose=False, seed=seed + n
        )
        phase1_results.append((n, a_try, obj_try))

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"    n={n}: {obj_try:.8f} **")
        else:
            print(f"    n={n}: {obj_try:.8f}")

    if best_seq is None:
        best_seq = spectral_init(800, 0)
        best_score = compute_obj(best_seq)
        print(f"    Default init: {best_score:.8f}")

    n_best = len(best_seq)
    print(f"  Phase 1 best: n={n_best}, obj={best_score:.8f}")

    # --- Phase 2: Intensify best n ---
    print("\n=== Phase 2: Intensify best n ===")
    remaining = deadline - _time.monotonic()
    lp_b2 = min(remaining * 0.30, 300.0)
    poly_b2 = min(remaining * 0.05, 50.0)
    if lp_b2 > 30:
        a2, obj2 = optimize_n(
            n_best, init=best_seq,
            lp_budget=lp_b2, poly_budget=poly_b2,
            verbose=True, seed=seed + 1000
        )
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"    Phase 2: {best_score:.8f} **")

    # --- Phase 3: Fine n sweep around best ---
    print("\n=== Phase 3: Fine n sweep ===")
    remaining = deadline - _time.monotonic()
    fine_ns = []
    for dn in range(-100, 101, 25):
        nc = n_best + dn
        if 400 <= nc <= 1400 and nc != n_best:
            fine_ns.append(nc)

    for n in fine_ns:
        remaining = deadline - _time.monotonic()
        if remaining < 50:
            break
        lp_b3 = min(remaining * 0.08, 80.0)
        poly_b3 = min(remaining * 0.02, 20.0)
        if lp_b3 < 15:
            break

        a3, obj3 = optimize_n(
            n, init=best_seq,
            lp_budget=lp_b3, poly_budget=poly_b3,
            verbose=False, seed=seed + n + 2000
        )
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            n_best = n
            print(f"    n={n}: {best_score:.8f} **")
        else:
            print(f"    n={n}: {obj3:.8f}")

    # --- Phase 4: Diversification restarts ---
    print("\n=== Phase 4: Diversification ===")
    remaining = deadline - _time.monotonic()
    for r in range(3):
        if remaining < 60:
            break
        if r == 0:
            a4 = fourier_perturb(best_seq, rng, strength=0.05)
        elif r == 1:
            a4 = best_seq * (1.0 + 0.08 * rng.normal(0, 1, len(best_seq)))
            a4 = np.maximum(a4, 0)
            a4 = a4 / a4.sum()
        else:
            a4 = 0.6 * best_seq + 0.4 * spectral_init(len(best_seq), r + 1)
            a4 = a4 / a4.sum()

        lp_b4 = min(remaining * 0.12, 100.0)
        poly_b4 = min(remaining * 0.03, 30.0)
        if lp_b4 > 20:
            a4, obj4 = optimize_n(
                len(a4), init=a4,
                lp_budget=lp_b4, poly_budget=poly_b4,
                verbose=False, seed=seed + 3000 + r * 100
            )
            if obj4 < best_score:
                best_score = obj4
                best_seq = a4.copy()
                n_best = len(a4)
                print(f"    Restart {r}: {best_score:.8f} **")
            else:
                print(f"    Restart {r}: {obj4:.8f}")

    # --- Phase 5: Final intensive polish ---
    print("\n=== Phase 5: Final Polish ===")
    remaining = deadline - _time.monotonic()

    if remaining > 100:
        lp_b5 = min(remaining * 0.50, 300.0)
        poly_b5 = min(remaining * 0.15, 100.0)
        a5, obj5 = optimize_n(
            n_best, init=best_seq,
            lp_budget=lp_b5, poly_budget=poly_b5,
            verbose=True, seed=seed + 9999
        )
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"    Final: {best_score:.8f} **")

    # Deep coordinate polish with remaining time
    remaining = deadline - _time.monotonic()
    if remaining > 20:
        poly_b6 = min(remaining * 0.70, 150.0)
        a6 = coordinate_polish(best_seq, budget=poly_b6, verbose=True)
        obj6 = compute_obj(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"    Deep Poly: {best_score:.8f} **")

    # --- Validation ---
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if np.isinf(val):
            result = [1.0] * 100
            best_score = 2.0
        else:
            best_score = val
    except Exception:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = _time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
