import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
from scipy.ndimage import uniform_filter1d

def compute_conv_fft(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_conv_with(a, b):
    """Compute conv(a, b) via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M). Uses scipy HiGHS.
    """

    a_norm = a / float(np.sum(a))
    A_conv = build_conv_matrix(a_norm)

    # Variables: x = [g_0, ..., g_{n-1}, M] (n+1 variables)
    # Minimize: M -> c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality constraints: conv(a,g)[k] - M <= 0 for each k

    A_ineq = np.zeros((2 * n - 1, n + 1))

    A_ineq[:, n] = -1.0
    b_ineq = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,

        s = float(np.sum(g))
        if s > 1e-10:

    b_arr[n:2 * n - 1] = conv[n - 2::-1]    # negative lags -1..-(n-1)

    A_fft = fft(a, N)
    B_fft = fft(b_arr, N)
    grad = 2.0 * ifft(A_fft * np.conj(B_fft)).real[:n]

    return grad


def exact_subgradient(a, active_set=None, eps=0.01):
    """
    Compute exact subgradient using active set (lags near max).
    """
    n = len(a)
    conv = compute_conv_fft(a)
    max_conv = float(np.max(conv))

    if active_set is None:
        threshold = max_conv * (1.0 - eps)
        active_set = np.where(conv >= threshold)[0]

    # For each active lag k (in conv indexing), the contribution to grad[m] is:
    # a[m - (k - (n-1))] + a[m + (k - (n-1))]
    # lag = k - (n-1)
    lags = active_set.astype(int) - (n - 1)

    # Average over active set
    p = np.zeros(len(conv))
    p[active_set] = 1.0 / len(active_set)

    N = 1
    while N < 3 * n:
        N <<= 1

    b_arr = np.zeros(N)
    b_arr[:n] = p[n - 1:2 * n - 1]
    b_arr[n:2 * n - 1] = p[n - 2::-1]

    A_fft = fft(a, N)
    B_fft = fft(b_arr, N)
    grad = 2.0 * ifft(A_fft * np.conj(B_fft)).real[:n]

    return grad, max_conv


def nesterov_gd(a, time_budget=100.0, tau_start=20.0, tau_end=2000.0,
                 verbose=False, rng=None):
    """
    Nesterov accelerated gradient descent on smoothed objective.
    a is normalized to sum=1.
    """
    if rng is None:
        rng = np.random.default_rng(42)

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 1.5

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_subgradient(y, tau)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Adaptive step size (simplified BB)
        if prev_g is not None and it > 2:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-10), 5.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb

        lr = max(1e-12, min(lr, 10.0 / np.sqrt(n)))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        # Decay LR slightly
        lr = max(1e-12, lr * 0.9995)

        # Restart if stuck
        if stall_count > 2000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            beta = 0.8
            stall_count = 0

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    Nesterov it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.0f}s")

    if verbose:
        print(f"    Nesterov done: obj={best_obj:.8f}, iters={it}")
    return best_a


def frank_wolfe(a, time_budget=200.0, verbose=False, max_iters=500):
    """
    Frank-Wolfe (conditional gradient) method.
    At each step, solve LP to get vertex, then line search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 3.0:
            break

        # Solve LP for direction
        g, M_val = solve_lp_direction(a)
        if g is None:
            break

        # Line search: minimize f((1-t)*a + t*g) for t in [0,1]
        # Coarse grid
        ts = np.linspace(0.01, 0.99, 20)
        best_t = 0.0
        best_t_obj = best_obj

        for t in ts:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t > 0 and best_t_obj < best_obj - 1e-13:
            a_new = (1 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / float(np.sum(a_new))
            best_obj = best_t_obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and (it % 20 == 0 or it == 0):
                print(f"    FW it{it}: obj={best_obj:.8f}, t={best_t:.3f}, M={M_val:.6f}")
        else:
            stall += 1
            if verbose and it % 50 == 0:
                print(f"    FW it{it}: stalled, obj={best_obj:.8f}, M={M_val:.6f}")

            if stall > 30:
                # Try smaller steps
                improved = False
                for t_small in [0.02, 0.01, 0.005]:
                    a_small = (1 - t_small) * a + t_small * g
                    a_small = np.maximum(a_small, 0)
                    s_small = float(np.sum(a_small))
                    if s_small > 1e-10:
                        a_small = a_small / s_small
                        obj_small = compute_obj(a_small)
                        if obj_small < best_obj - 1e-13:
                            best_obj = obj_small
                            best_a = a_small.copy()
                            a = a_small
                            stall = 0
                            improved = True
                            break
                if not improved and stall > 50:
                    # Escape: perturb and restart
                    rng = np.random.default_rng(it)
                    a = a + 0.01 * rng.standard_normal(n)
                    a = np.maximum(a, 0)
                    a = a / float(np.sum(a))
                    stall = 0
                    if it > 200:
                        break

    if verbose:
        print(f"    FW done: obj={best_obj:.8f}")
    return best_a


def mirror_descent(a, time_budget=100.0, eta_start=0.1, eta_end=5.0, verbose=False):
    """
    Mirror descent with entropy kernel.
    g[i] = a[i] * exp(-eta * subgrad[i]) / sum(...)
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        eta = eta_start + (eta_end - eta_start) * progress ** 0.8

        # Compute subgradient (using smoothed version for stability)
        tau = 50.0 + 500.0 * progress
        g = smoothed_subgradient(a, tau)

        # Mirror descent step
        log_a = np.log(np.maximum(a, 1e-300))
        new_log_a = log_a - eta * g
        new_log_a -= np.max(new_log_a)  # numerical stability
        a_new = np.exp(new_log_a)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        it += 1

        if verbose and it % 200 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.3f}")

    return best_a


def coord_descent(a, time_budget=60.0, verbose=False):
    """
    Cyclic coordinate descent on the normalized simplex.
    For each coordinate, optimize its value given others fixed.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(0)

    # Precompute conv for reuse
    conv = compute_conv_fft(a)
    max_conv = float(np.max(conv))

    stall = 0
    epoch = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 2:
            break

        # Random order for diversity in later epochs
        if epoch < 5:
            order = np.arange(n)
        else:
            order = rng.permutation(n)

        epoch_improved = False
        epoch += 1

        for idx in order:
            # Try small perturbations of a[idx]
            # The objective depends on a[idx] through conv(a,a)
            # For a given a[idx] = v, the change in conv is:
            # conv_new[k] = conv_old[k] + v*(a[idx+(k-(n-1))]) + v*(a[idx-(k-(n-1))]) - old_contrib
            # This is complex, so just do a simple 1D search

            old_val = a[idx]
            # Try a few candidate values
            candidates = [0.0, old_val * 0.5, old_val * 0.75, old_val, old_val * 1.25, old_val * 1.5,
                          old_val + 0.001, old_val - 0.001, 0.001, 0.01]
            candidates = [max(0.0, c) for c in candidates]

            best_c = old_val
            best_c_obj = best_obj

            # We need to recompute conv for each candidate - expensive
            # Instead, do a quick approximation:
            # Only update if the change is likely to help
            # The subgradient at this point tells us the direction
            # For now, just try a few values with full recompute

            if elapsed > time_budget - 5:
                break

            # Quick obj evaluation with modified a
            for c in candidates:
                if c == old_val:
                    continue
                a[idx] = c
                s_try = float(np.sum(a))
                if s_try < 1e-10:
                    a[idx] = old_val
                    continue
                a_norm = a / s_try
                conv_try = compute_conv_fft(a_norm)
                obj_try = 2.0 * n * float(np.max(conv_try))
                if obj_try < best_c_obj - 1e-13:
                    best_c_obj = obj_try
                    best_c = c
            a[idx] = best_c

            if best_c != old_val:
                epoch_improved = True
                # Update conv
                s_try = float(np.sum(a))
                a = a / s_try
                conv = compute_conv_fft(a)
                max_conv = float(np.max(conv))
                if max_conv * 2.0 * n < best_obj - 1e-14:
                    best_obj = 2.0 * n * max_conv
                    best_a = a.copy()

        if not epoch_improved:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, best_seq_parent=None):
    """Generate diverse initializations."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine shape
    x = np.linspace(0, np.pi, n)
    c = np.cos(x / 2) ** 2
    inits.append(c / c.sum())

    # Gaussian shape
    x = np.linspace(-3, 3, n)
    g = np.exp(-x ** 2 / 2)
    inits.append(g / g.sum())

    # Two peaks
    x = np.linspace(0, 1, n)
    two = np.exp(-40 * (x - 0.25) ** 2) + np.exp(-40 * (x - 0.75) ** 2)
    inits.append(two / two.sum())

    # Random
    r = rng.uniform(0.1, 1.0, n)
    inits.append(r / r.sum())

    # From parent if available
    if best_seq_parent is not None:
        inits.append(resample_sequence(best_seq_parent, n))

    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines Frank-Wolfe, Nesterov GD, mirror descent,
    coordinate descent, multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: Nesterov GD on parent length ===
    print(f"\n=== Phase 1: Nesterov GD (parent n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 20:
        a1 = nesterov_gd(best_seq, time_budget=p1_budget,
                         tau_start=10.0, tau_end=1500.0,
                         verbose=True, rng=rng)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1 Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 Nesterov: obj={obj1:.8f} (no improvement)")

    # === Phase 2: Frank-Wolfe ===
    print(f"\n=== Phase 2: Frank-Wolfe ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.3, 350.0)
    if p2_budget > 30:
        a2 = frank_wolfe(best_seq, time_budget=p2_budget, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 FW: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 FW: obj={obj2:.8f} (no improvement)")

    # === Phase 3: Mirror Descent ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 100.0)
    if p3_budget > 15:
        a3 = mirror_descent(best_seq, time_budget=p3_budget,
                            eta_start=0.05, eta_end=3.0)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 MD: obj={obj3:.8f} (no improvement)")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Try various n values around the best
    n_candidates = sorted(set([
        n_best - 200, n_best - 100, n_best - 50,
        n_best + 50, n_best + 100, n_best + 200, n_best + 400,
        600, 700, 850, 900, 1000, 1100, 1200, 1500
    ]))
    n_candidates = [n for n in n_candidates if 500 <= n <= 2000 and abs(n - n_best) > 40]
    n_candidates = n_candidates[:5]
    print(f"  Trying n values: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        # Resample from best
        a_try = resample_sequence(best_seq, n_try)

        # Nesterov GD
        alloc = min(remaining * 0.25, 60.0)
        if alloc > 15:
            a_try = nesterov_gd(a_try, time_budget=alloc,
                                tau_start=10.0, tau_end=800.0,
                                verbose=False, rng=rng)

        # FW
        remaining = deadline - time.monotonic()
        alloc_fw = min(remaining * 0.15, 40.0)
        if alloc_fw > 10:
            a_try = frank_wolfe(a_try, time_budget=alloc_fw, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts from different inits ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    restart_configs = [
        (n_cur, "cosine"),
        (n_cur, "gaussian"),
        (n_cur, "random"),
        (n_cur - 50, "uniform"),
        (n_cur + 100, "cosine"),
    ]

    for n_r, init_type in restart_configs:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_r < 400:
            continue

        # Generate init
        if init_type == "uniform":
            a_r = np.ones(n_r) / n_r
        elif init_type == "cosine":
            x = np.linspace(0, np.pi, n_r)
            a_r = np.cos(x / 2) ** 2
            a_r = a_r / a_r.sum()
        elif init_type == "gaussian":
            x = np.linspace(-3, 3, n_r)
            a_r = np.exp(-x ** 2 / 2)
            a_r = a_r / a_r.sum()
        elif init_type == "random":
            a_r = rng.uniform(0.1, 1.0, n_r)
            a_r = a_r / a_r.sum()
        else:
            a_r = resample_sequence(best_seq, n_r)

        # Optimize
        alloc_gd = min(remaining * 0.2, 40.0)
        if alloc_gd > 10:
            a_r = nesterov_gd(a_r, time_budget=alloc_gd,
                              tau_start=5.0, tau_end=500.0,
                              verbose=False, rng=rng)

        remaining = deadline - time.monotonic()
        alloc_fw = min(remaining * 0.12, 30.0)
        if alloc_fw > 8:
            a_r = frank_wolfe(a_r, time_budget=alloc_fw, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({init_type}, n={n_r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({init_type}, n={n_r}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Nesterov with high tau
    if remaining > 20:
        alloc = min(remaining * 0.35, 150.0)
        a_final = nesterov_gd(best_seq, time_budget=alloc,
                              tau_start=100.0, tau_end=3000.0,
                              verbose=True, rng=rng)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Nesterov: obj={best_score:.8f} **")

    # FW
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 100.0)
        a_final2 = frank_wolfe(best_seq, time_budget=alloc, verbose=True)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final FW: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 50.0)
        a_final3 = coord_descent(best_seq, time_budget=alloc)
        obj_final3 = compute_obj(a_final3)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()
            print(f"  Coord Descent: obj={best_score:.8f} **")

    # One more Nesterov round
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.5, 40.0)
        a_final4 = nesterov_gd(best_seq, time_budget=alloc,
                               tau_start=200.0, tau_end=5000.0,
                               verbose=False, rng=rng)
        obj_final4 = compute_obj(a_final4)
        if obj_final4 < best_score:
            best_score = obj_final4
            best_seq = a_final4.copy()
            print(f"  Final Nesterov2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
