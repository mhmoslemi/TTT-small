def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smooth_max_grad(a, beta=30.0):
    """
    Compute the smoothed objective and its gradient using log-sum-exp.
    Objective: 2*n*logsumexp(beta * conv / s^2) / beta (for normalized a, s=1)
    For normalized a (sum=1): obj = 2*n*logsumexp(beta * conv) / beta
    Gradient: 2*n * beta * sum_k exp(beta*c_k) * dc_k/da_j / (beta * sum_k exp(beta*c_k))
            = 2*n * sum_k softmax_k * dc_k/da_j
    """
    n = len(a)
    conv = compute_conv(a)
    # Softmax weights
    bconv = beta * conv
    bconv_shifted = bconv - np.max(bconv)
    exp_bconv = np.exp(bconv_shifted)
    weights = exp_bconv / np.sum(exp_bconv)  # softmax weights
    
    # Smoothed objective (for normalized a)
    smooth_obj = 2.0 * n * (np.log(np.sum(exp_bconv)) / beta + np.max(bconv) / beta)

    # So smoothed_obj = 2*n * logsumexp(beta*conv) / beta
    # = 2*n * (max(conv) + log(sum(exp(beta*(conv-max(conv))))) / beta)

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_obj_and_grad(a, beta):
    """
    Compute smoothed objective and gradient for normalized a (sum=1).
    Uses log-sum-exp smoothing of the max convolution value.
    Returns (smoothed_obj, gradient_vector).
    """
    n = len(a)
    conv = compute_conv(a)
    bconv = beta * conv
    max_b = np.max(bconv)
    exp_b = np.exp(bconv - max_b)
    sum_exp = np.sum(exp_b)
    weights = exp_b / sum_exp
    
    # Smoothed objective: 2*n * (max_con + log(sum_exp) / beta) / beta
    # = 2*n * (max_b + log(sum_exp)) / beta
    smooth_obj = 2.0 * n * (max_b + np.log(sum_exp)) / beta
    
    # Gradient: 4*n * correlate(weights, a)[n-1 : n-1+n]
    # g[j] = sum_k weights[k] * a[k-j] [0<=k-j<n]
    g = correlate(weights, a, mode='full')[n - 1:n - 1 + n]
    grad = 4.0 * n * g
    
    return smooth_obj, grad


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1} using clip and normalize."""
    v = np.maximum(v, 0)
    s = np.sum(v)
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s


def exact_project_simplex(v):
    """Exact Euclidean projection onto simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_gradient_descent(a, time_budget=60.0, beta_start=15.0, beta_end=100.0,
                            lr_start=0.5, verbose=False):
    """
    Projected gradient descent with AdaGrad and adaptive beta.
    a is normalized (sum=1, non-negative).
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    G = np.zeros(n)  # AdaGrad accumulator
    start = time.monotonic()
    lr = lr_start
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        # Adaptive beta: increase as we converge
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        
        obj_s, grad = smooth_obj_and_grad(a, beta)
        
        # AdaGrad step
        G += grad * grad
        step = lr / np.sqrt(G + 1e-10)
        a_new = a - step * grad
        a_new = project_simplex(a_new)
        
        # Evaluate actual objective
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1

                lr *= 0.5

                    noise = rng_local.normal(0, 0.01, n)
                    a = project_simplex(best_a + noise)
                    G = np.zeros(n)
                    lr = lr_start * 0.3
                    no_improve = 0
            elif no_improve > 50:
                lr *= 0.8
        
        it += 1
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    SGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.4f}, t={elapsed:.1f}s")
    
    return best_a


def build_conv_matrix(a, use_symmetry=True):
    """Build matrix A (m x n) such that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    if use_symmetry:
        # Only use k = 0, ..., n-1 (exploit c_k = c_{2n-2-k})
        m = n
    else:
        m = 2 * n - 1
    
    A = np.zeros((m, n))
    for k in range(m):
        for j in range(n):
            k_j = k - j
            if 0 <= k_j < n:
                A[k, j] = a[k_j]
    return A


def solve_lp(a, use_symmetry=True, max_constraints=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Uses symmetry to halve constraints.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    if use_symmetry:
        m = n  # Only k=0,...,n-1
        A = np.zeros((m, n))
        for k in range(m):
            for j in range(n):
                k_j = k - j
                if 0 <= k_j < n:
                    A[k, j] = a_norm[k_j]
    else:
        A = build_conv_matrix(a_norm, use_symmetry=False)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def arcsine_init(n):
    """Arcsine distribution."""
    x = np.linspace(0.001, 0.999, n)
    a = 1.0 / np.sqrt(1.0 - (2 * x - 1) ** 2 + 1e-10)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""

"""
Hybrid smoothed-gradient + cutting-plane LP optimization for step function coefficients.
Uses log-sum-exp smoothing for stable O(n log n) gradients, AdaGrad adaptive steps,
cutting-plane LP with symmetry exploitation, adaptive beta scheduling, and n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.signal import correlate
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_obj_and_grad(a, beta):
    """
    Compute smoothed objective and gradient for normalized a (sum=1).
    F_smooth = 2*n*logsumexp(beta*c)/beta where c=conv(a,a)
    grad_j = 4*n * sum_k softmax_k * a[k-j] [0<=k-j<n]
    """
    n = len(a)
    conv = compute_conv(a)
    bconv = beta * conv
    max_b = np.max(bconv)
    exp_b = np.exp(bconv - max_b)
    sum_exp = np.sum(exp_b)
    weights = exp_b / sum_exp
    smooth_obj = 2.0 * n * (max_b + np.log(sum_exp)) / beta
    g = correlate(weights, a, mode='full')[0:n]
    grad = 4.0 * n * g
    return smooth_obj, grad


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0)
    s = np.sum(v)
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s


def smooth_gradient_descent(a, time_budget=60.0, beta_start=15.0, beta_end=100.0,
                            lr_start=0.5, verbose=False, seed=0):
    """Projected gradient descent with AdaGrad and adaptive beta."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    G = np.zeros(n)
    start = time.monotonic()
    lr = lr_start
    it = 0
    no_improve = 0
    rng_local = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / max(1.0, time_budget))
        beta = beta_start + (beta_end - beta_start) * progress

        obj_s, grad = smooth_obj_and_grad(a, beta)
        G += grad * grad
        step = lr / np.sqrt(G + 1e-10)
        a_new = a - step * grad
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.5
                if lr < 0.02:
                    noise = rng_local.normal(0, 0.02, n)
                    a = project_simplex(best_a + noise)
                    G = np.zeros(n)
                    lr = lr_start * 0.5
                    no_improve = 0
            elif no_improve > 100:
                lr *= 0.8

        it += 1
        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - start
            print(f"    SGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.4f}, t={elapsed:.1f}s")

    return best_a

    if len(active_idx) > max_constraints:
        active_idx = np.argsort(conv_half)[::-1][:max_constraints]

    g_best = None

                    g_val = np.maximum(g.value, 0)

        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g_dir, time_budget=3.0):
    """Golden section search for optimal blend factor t in (1-t)*a + t*g_dir."""
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    lo, hi = 0.0, 1.5

    def obj_at(t):
        if t < 0:
            t = 0.0
        a_try = (1 - t) * a + t * g_dir
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    start = time.monotonic()
    for _ in range(30):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)
    return t_best, obj_best


def lp_polish(a, time_budget=60.0, verbose=False, max_iters=40, use_cutting_plane=True):
    """LP polishing with line search, using cutting-plane or full LP."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        if use_cutting_plane:
            g_lp, M_lp = solve_lp_cutting_plane(a, max_constraints=400, time_per_lp=min(4.0, remaining * 0.05))
        else:
            g_lp, M_lp = solve_lp_full(a, use_symmetry=True)

        if g_lp is None:
            break

        t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=min(2.0, remaining * 0.04))

        if t_opt > 0 and obj_opt < best_obj - 1e-15:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = project_simplex(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 3 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, stall={stall}, t_elapsed={elapsed:.1f}s")

        if stall >= 5:
            break

    return best_a


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.001, 0.999, n)
    a = 1.0 / np.sqrt(1.0 - (2 * x - 1) ** 2 + 1e-10)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def try_new_length(a_best, n_try, time_budget, rng):
    """Try optimizing at a different sequence length via interpolation."""
    x_old = np.linspace(0, 1, len(a_best))
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = project_simplex(a_try)

    remaining = time_budget
    if remaining > 30:
        a_try = smooth_gradient_descent(a_try, time_budget=min(20.0, remaining * 0.3),
                                         beta_start=20.0, beta_end=100.0,
                                         lr_start=0.4, seed=int(rng.integers(0, 10000)))
    remaining = time_budget - (time.monotonic() - time.monotonic())  # approximate
    if time_budget > 50:
        a_try = lp_polish(a_try, time_budget=min(30.0, time_budget * 0.5),
                          max_iters=10, use_cutting_plane=True)

    obj_try = compute_obj(a_try)
    return a_try, obj_try


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Hybrid smoothed-gradient + cutting-plane LP optimization.
    Uses log-sum-exp smoothing for fast stable gradients, AdaGrad for adaptive steps,
    cutting-plane LP with symmetry, adaptive beta, and n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default U-shaped init (n={n_current}): obj={best_score:.8f}")

    # Try alternative initializations
    init_modes = [
        ("arcsine", arcsine_init(n_current)),
        ("U5", u_shaped_init(n_current, 5.0)),
        ("U2", u_shaped_init(n_current, 2.0)),
        ("U8", u_shaped_init(n_current, 8.0)),
    ]
    for name, a_try in init_modes:
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"Init {name} (n={n_current}): obj={best_score:.8f} **")

    # === Phase 1: Smoothed Gradient Descent (fast warm-up) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 1: SGD warm-up (budget={min(70.0, remaining*0.1):.0f}s) ===")
        t1 = min(70.0, remaining * 0.1)
        a_sgd = smooth_gradient_descent(best_seq, time_budget=t1,
                                         beta_start=10.0, beta_end=60.0,
                                         lr_start=0.5, verbose=True, seed=seed)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 GD: obj={obj_sgd:.8f} (no improvement)")

    # === Phase 2: Cutting-plane LP polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 2: LP Polishing (budget={min(250.0, remaining*0.25):.0f}s) ===")
        t2 = min(250.0, remaining * 0.25)
        a_lp = lp_polish(best_seq, time_budget=t2, verbose=True, max_iters=60,
                         use_cutting_plane=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: High-beta GD refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 3: High-beta GD (budget={min(60.0, remaining*0.08):.0f}s) ===")
        t3 = min(60.0, remaining * 0.08)
        a_gd2 = smooth_gradient_descent(best_seq, time_budget=t3,
                                         beta_start=40.0, beta_end=300.0,
                                         lr_start=0.3, verbose=True, seed=seed + 1)
        obj_gd2 = compute_obj(a_gd2)
        if obj_gd2 < best_score:
            best_score = obj_gd2
            best_seq = a_gd2.copy()
            print(f"  Phase3 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 GD: obj={obj_gd2:.8f} (no improvement)")

    # === Phase 4: Second LP round (full LP for precision) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 4: LP Round 2 (budget={min(150.0, remaining*0.18):.0f}s) ===")
        t4 = min(150.0, remaining * 0.18)
        a_lp2 = lp_polish(best_seq, time_budget=t4, verbose=True, max_iters=30,
                          use_cutting_plane=False)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase4 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 LP: obj={obj_lp2:.8f} (no improvement)")

    # === Phase 5: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 5: n Exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best + 100, n_best + 200, n_best + 400, n_best + 600,
            n_best - 100, 1000, 1200, 1400
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) >= 50]
        n_options = n_options[:4]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            t_alloc = min(100.0, remaining * 0.6)
            a_try, obj_try = try_new_length(best_seq, n_try, t_alloc, rng)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 6: Diverse Restarts ===")
        n_cur = len(best_seq)
        for r in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            a_r = diverse_init(n_cur, rng, mode=r)
            blend = 0.4
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = project_simplex(a_r)
            if remaining > 50:
                a_r = smooth_gradient_descent(a_r, time_budget=min(20.0, remaining * 0.2),
                                               beta_start=20.0, beta_end=100.0,
                                               lr_start=0.4, seed=seed + 100 + r)
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_r = lp_polish(a_r, time_budget=min(30.0, remaining * 0.4),
                                max_iters=15, use_cutting_plane=True)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 7: Final Polish ===")
        t7 = min(remaining - 10, 120.0)
        if t7 > 15:
            a_final = lp_polish(best_seq, time_budget=t7, verbose=True, max_iters=25,
                                use_cutting_plane=True)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_gd_f = smooth_gradient_descent(best_seq, time_budget=min(remaining - 5, 30.0),
                                              beta_start=80.0, beta_end=500.0,
                                              lr_start=0.2, seed=seed + 999)
            obj_gd_f = compute_obj(a_gd_f)
            if obj_gd_f < best_score:
                best_score = obj_gd_f
                best_seq = a_gd_f.copy()
                print(f"  Final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
