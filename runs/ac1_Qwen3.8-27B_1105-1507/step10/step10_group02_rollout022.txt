"""
Optimizes step function coefficients using exact coordinate descent, LP polishing,
KKT-guided fixed-point iteration, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming a is normalized (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    # Verify
                    conv_check = A @ g_val
                    M_actual = float(np.max(conv_check))
                    return g_val, M_actual
        except Exception:
            continue

    return None, None

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming a is normalized (sum=1). Returns 2n*max(conv)."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    conv_check = A @ g_val
                    M_actual = float(np.max(conv_check))
                    return g_val, M_actual
        except Exception:
            continue
    return None, None

def kkt_subgradient_descent(a, time_budget=200.0, verbose=False, seed=42):
    """
    KKT-guided subgradient descent. Uses the exact subgradient of max(conv(a,a))
    to guide updates, with adaptive step size and projection onto the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = 1.0 / np.sqrt(n)
    momentum = 0.5
    v_mom = np.zeros(n)
    it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining_frac = 1.0 - elapsed / max(time_budget, 1e-10)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        argmax_set = np.where(conv >= max_conv * (1.0 - 1e-10))[0]

        # Compute subgradient: v[i] = sum_{k in argmax} a[k-i] * 1{0<=k-i<n} / |argmax|
        v_sub = np.zeros(n)
        for k in argmax_set:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                v_sub[lo:hi + 1] += a[k - hi:k - lo + 1]
        v_sub /= max(len(argmax_set), 1)

        # Target: v_sub[i] should equal max_conv/2 (from KKT: 2*v_sub[i] = lambda)
        target = max_conv / 2.0

        # Gradient direction
        grad = 2.0 * v_sub - target * np.ones(n) * 2.0 / n

        # Momentum
        v_mom = momentum * v_mom - eta * grad
        a_new = a + v_mom
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            v_mom = np.zeros(n)
            eta *= 0.5
        else:
            a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience > 20:
                # Reduce step size and add noise
                eta *= 0.7
                if eta < 1e-4:
                    # Add random perturbation
                    pert = rng.normal(0, 0.01, n)
                    a_pert = a * np.exp(pert)
                    a_pert = np.maximum(a_pert, 0)
                    a_pert = a_pert / a_pert.sum()
                    obj_pert = 2.0 * n * float(np.max(compute_conv(a_pert)))
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                        a = a_pert
                    eta = 1.0 / np.sqrt(n)
                    v_mom = np.zeros(n)
                    patience = 0

        it += 1
        if verbose and it % 500 == 0:
            print(f"    KKT-GD it{it}: obj={best_obj:.8f}, eta={eta:.6f}, t={elapsed:.1f}s")

    return best_a

                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

    return best_a


def smoothed_gradient_descent(a, time_budget=100.0, tau_init=30.0, tau_end=500.0, verbose=False):
    """Nesterov-accelerated smoothed gradient descent with exponential surrogate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = 1.0
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        N = 1
        while N < 3 * n:
            N <<= 1

        P = fft(p, N)
        A_fft = fft(y, N)
        g_full = ifft(P * np.conj(A_fft)).real
        grad = 2.0 * g_full[:n]

        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            lr_eff = min(lr, 3.0 / (np.sqrt(n) * (g_norm / np.sqrt(n) + 1e-10)))
        else:
            lr_eff = lr

        v = beta * v - lr_eff * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        it += 1

        if verbose and it % 2000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x * x + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 10, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Two-bump with gap
        a = np.zeros(n)
        w = n // 5
        a[:w] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * (w // 4) ** 2))
        a[-w:] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * (w // 4) ** 2))
        a = a / a.sum()
    elif mode == 7:
        # Cosine bump
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining exact CD, KKT-subgradient
    descent, LP polishing, smoothed GD, and systematic n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_default = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(n_default) / n_default
        best_score = 2.0 * n_default * float(np.max(compute_conv(best_seq)))
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(800) / 800.0
        best_score = 2.0

    n_best = len(best_seq)

    # === Phase 1: Extended Coordinate Descent ===
    print(f"\n=== Phase 1: Exact Coordinate Descent (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 500.0)
    if p1_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_cd = 2.0 * a_cd.size * float(np.max(compute_conv(a_cd / a_cd.sum())))
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: KKT-Subgradient Descent ===
    print(f"\n=== Phase 2: KKT-Subgradient Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 180.0)
    if p2_budget > 20:
        a_kkt = kkt_subgradient_descent(best_seq, time_budget=p2_budget,
                                        verbose=False, seed=seed + 1)
        obj_kkt = 2.0 * a_kkt.size * float(np.max(compute_conv(a_kkt / a_kkt.sum())))
        if obj_kkt < best_score:
            best_score = obj_kkt
            best_seq = a_kkt.copy()
            print(f"  KKT-GD: obj={best_score:.8f} **")
        else:
            print(f"  KKT-GD: obj={obj_kkt:.8f}")
            # Try CD from KKT result
            if remaining - time.monotonic() > 30:
                a_cd2 = coordinate_descent(a_kkt, time_budget=min(30, remaining * 0.05),
                                           verbose=False, seed=seed + 2)
                obj_cd2 = 2.0 * a_cd2.size * float(np.max(compute_conv(a_cd2 / a_cd2.sum())))
                if obj_cd2 < best_score:
                    best_score = obj_cd2
                    best_seq = a_cd2.copy()
                    print(f"  CD after KKT: obj={best_score:.8f} **")

    # === Phase 3: LP Polishing ===
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.18, 250.0)
    if p3_budget > 20:
        for rnd in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(p3_budget / (5 - rnd), remaining * 0.5)
            if alloc < 10:
                break
            a_lp = lp_polish(best_seq, time_budget=alloc, verbose=False,
                             seed=1000 + rnd * 7)
            obj_lp = 2.0 * a_lp.size * float(np.max(compute_conv(a_lp / a_lp.sum())))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
            # Follow-up CD after LP
            remaining = deadline - time.monotonic()
            if remaining > 20 and rnd < 4:
                a_cd3 = coordinate_descent(best_seq, time_budget=min(25, remaining * 0.08),
                                           verbose=False, seed=2000 + rnd)
                obj_cd3 = 2.0 * a_cd3.size * float(np.max(compute_conv(a_cd3 / a_cd3.sum())))
                if obj_cd3 < best_score:
                    best_score = obj_cd3
                    best_seq = a_cd3.copy()
                    print(f"  CD after LP{rnd}: obj={best_score:.8f} **")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Generate n candidates: around current, plus some spread
    n_candidates = sorted(set([
        max(200, n_current - 200), max(200, n_current - 100),
        n_current + 100, n_current + 200, n_current + 400, n_current + 600,
        400, 500, 600, 700, 900, 1000, 1200, 1500, 2000
    ]))
    # Filter to reasonable range and deduplicate close values
    n_filtered = []
    for nc in n_candidates:
        if 300 <= nc <= 2500:
            if all(abs(nc - nf) > 50 for nf in n_filtered):
                n_filtered.append(nc)
    n_filtered = n_filtered[:6]
    print(f"  n candidates: {n_filtered}")

    for n_try in n_filtered:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick CD
        alloc = min(remaining * 0.15, 60.0)
        if alloc > 15:
            a_try = coordinate_descent(a_try, time_budget=alloc, verbose=False,
                                       seed=n_try)

        obj_try = 2.0 * a_try.size * float(np.max(compute_conv(a_try / a_try.sum())))
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = 1 + r
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 70.0)
        if alloc > 20:
            a_r = coordinate_descent(a_r, time_budget=alloc, verbose=False,
                                     seed=5000 + r * 100)
            obj_r = 2.0 * a_r.size * float(np.max(compute_conv(a_r / a_r.sum())))
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 6: Smoothed GD Escape ===
    print(f"\n=== Phase 6: Smoothed GD Escape ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        pert = rng.normal(0, 0.04, len(best_seq))
        a_pert = best_seq * np.exp(pert)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.2, 80.0)
        a_gd = smoothed_gradient_descent(a_pert, time_budget=alloc,
                                         tau_init=40.0, tau_end=600.0)
        obj_gd = 2.0 * a_gd.size * float(np.max(compute_conv(a_gd / a_gd.sum())))
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  GD escape: obj={best_score:.8f} **")
        else:
            print(f"  GD escape: obj={obj_gd:.8f}")

    # === Phase 7: Final Polish (CD + LP alternating) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for final_round in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            # CD
            cd_alloc = min(remaining * 0.4, 100.0)
            if cd_alloc > 10:
                a_f = coordinate_descent(best_seq, time_budget=cd_alloc,
                                         verbose=False, seed=9000 + final_round)
                obj_f = 2.0 * a_f.size * float(np.max(compute_conv(a_f / a_f.sum())))
                if obj_f < best_score:
                    best_score = obj_f
                    best_seq = a_f.copy()
                    print(f"  Final CD {final_round}: obj={best_score:.8f} **")
            # LP
            remaining = deadline - time.monotonic()
            if remaining > 10:
                lp_alloc = min(remaining * 0.3, 50.0)
                if lp_alloc > 8:
                    a_f2 = lp_polish(best_seq, time_budget=lp_alloc,
                                     verbose=False, seed=9500 + final_round)
                    obj_f2 = 2.0 * a_f2.size * float(np.max(compute_conv(a_f2 / a_f2.sum())))
                    if obj_f2 < best_score:
                        best_score = obj_f2
                        best_seq = a_f2.copy()
                        print(f"  Final LP {final_round}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    # Ensure valid output
    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, x)) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
