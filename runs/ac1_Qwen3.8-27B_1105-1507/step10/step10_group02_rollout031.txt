The projection onto the simplex `{x: sum(x)=1, x>=0}`:

```python
def project_simplex(v, S=1.0):
    """Project v onto the simplex {x: sum(x)=S, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

    """Gerchberg-Saxton for flat convolution."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    a = a / S  # normalize
    
    if N is None:
        N = 1
        while N < 2 * n - 1:
            N *= 2
    
    # Target spectrum magnitude
    # C_target[k] = 1/(2n-1) for k=0,...,2n-2 (since S=1 after normalization)
    C_target = np.zeros(N)
    C_target[:2*n-1] = 1.0 / (2*n - 1)
    target_mag = np.abs(fft(C_target, N))
    target_mag[0] = 1.0  # DC: |A[0]| = S = 1
    
    for it in range(n_iter):
        A = fft(a, N)
        phase = A / np.abs(A + 1e-30)
        A_new = target_mag * phase
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new
        a = a_new
    
    return a
