- `conv(2i)`: `c_2i + x^2 - v^2` (quadratic in x)
- Denominator: `(S_other + x)^2` (convex in x for x > 0)

The objective is `2n * max(linear terms, quadratic term) / (S_other + x)^2`.

The numerator is convex (max of convex functions). The denominator is convex. The ratio of two convex functions is not necessarily convex or unimodal.

However, in practice, for the range of x we care about (near the current value), it's likely unimodal. The golden section search should work if we bracket well.

Actually, that's a great idea! The function `max_k c_k(x)` where each c_k is linear or quadratic changes identity at specific breakpoints. At each breakpoint, we can analytically compute the optimum of the single-k subproblem.

For a linear term `c_k + 2*a[k-i]*(x-v)`, the subproblem is:
minimize `(c_k + 2*a[k-i]*(x-v)) / (S_other + x)^2`

Taking derivative and setting to 0:
`2*a[k-i]*(S_other + x)^2 - 2*(c_k + 2*a[k-i]*(x-v))*(S_other + x) = 0`
`2*a[k-i]*(S_other + x) - 2*(c_k + 2*a[k-i]*(x-v)) = 0`
`a[k-i]*(S_other + x) = c_k + 2*a[k-i]*(x-v)`
`a[k-i]*S_other + a[k-i]*x = c_k + 2*a[k-i]*x - 2*a[k-i]*v`
`a[k-i]*S_other - c_k + 2*a[k-i]*v = a[k-i]*x`
`x = S_other - c_k/a[k-i] + 2*v` (if a[k-i] ≠ 0)

For the quadratic term `c_2i + x^2 - v^2`:
minimize `(c_2i + x^2 - v^2) / (S_other + x)^2`

This is more complex but still solvable.

So the exact 1D optimum is at one of:
- x = 0 (boundary)
- x where two linear terms cross
- x where a linear term and the quadratic cross
- x that minimizes a single linear term
- x that minimizes the quadratic term

This gives a finite set of candidates, and we can evaluate the objective at each to find the exact 1D minimum!

This is a significant improvement over golden section search, which is approximate.

```python
def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)  # ensure normalized
    
    # A matrix: (2n-1) x n, A[k,i] = a[i] if 0 <= k-i < n
    # conv(a,g)[k] = sum_i A[k,i] * g[i]
    
    # We want: minimize M
    # s.t.: A@g - M <= 0 (2n-1 constraints)
    #       sum(g) = 1
    #       g >= 0
    #       M free
    
    # Using linprog: minimize c^T x
    # x = [g_0, ..., g_{n-1}, M]
    # c = [0, ..., 0, 1]
    # A_ub: [A, -I] (each row: A[k,:] @ g - M <= 0)
    # A_eq: [ones(1,n), 0]
    # b_ub: zeros(2n-1)
    # b_eq: [1]
    # bounds: g[i] in [0, inf], M in (-inf, inf)
    
    # Build A_ub: (2n-1) x (n+1)
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        A_ub[k, i_start:i_end+1] = a[i_start:i_end+1]
        A_ub[k, n] = -1  # coefficient of M
    
    c = np.zeros(n + 1)
    c[n] = 1  # minimize M
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1  # sum(g) = 1
    b_eq = [1.0]
    b_ub = np.zeros(2 * n - 1)
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        g = np.maximum(result.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            g = g / s
        M = float(result.x[n])
        return g, M
    return None, None
```

```python
"""
Optimizes step function coefficients using LP-guided coordinate descent with
exact line search, adaptive restarts, and multiscale exploration.
Key improvements over prior: (1) LP with exact line search instead of fixed blending,
(2) perturbation-based restarts when stalled, (3) tighter CD with better bracketing,
(4) adaptive n exploration around the best-performing length.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a[i]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M) or (None, None)."""
    n = len(a)
    a = a / float(np.sum(a))

    # Build constraint matrix
    A = build_conv_matrix(a)  # (2n-1) x n

    # LP: minimize M
    # x = [g_0..g_{n-1}, M], size n+1
    # s.t.: A@g - M <= 0  (2n-1 constraints)
    #       sum(g) = 1   (1 equality)
    #       g >= 0

    n_cons = 2 * n - 1
    A_ub = np.zeros((n_cons, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0

    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    b_ub = np.zeros(n_cons)

    bounds = [(0.0, None)] * n + [(-np.inf, np.inf)]

    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(g))
            if s > 1e-12:
                g = g / s
            M_val = float(res.x[n])
            return g, M_val
    except Exception:
        pass

    return None, None


def line_search_blend(a, g, n_points=30):
    """Find optimal t in [0,1] minimizing obj((1-t)*a + t*g)."""
    best_t = 0.0
    best_obj = compute_obj_normalized(a)

    # Coarse grid
    ts = np.linspace(0.0, 1.0, n_points)
    obj_vals = np.zeros(n_points)
    for idx in range(n_points):
        t = ts[idx]
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0.0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            obj_vals[idx] = 1e30
        else:
            obj_vals[idx] = compute_obj_normalized(a_try / s_try)

    min_idx = int(np.argmin(obj_vals))
    best_obj = obj_vals[min_idx]
    best_t = ts[min_idx]

    # Refine with golden section in neighborhood
    if min_idx > 0 and min_idx < n_points - 1:
        x_lo = ts[min_idx - 1]
        x_hi = ts[min_idx + 1]
    elif min_idx == 0:
        x_lo = 0.0
        x_hi = ts[1]
    else:
        x_lo = ts[min_idx - 1]
        x_hi = 1.0

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = x_hi - gr * (x_hi - x_lo)
    d_pt = x_lo + gr * (x_hi - x_lo)

    def eval_at_t(t):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0.0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            return 1e30
        return compute_obj_normalized(a_try / s_try)

    fc = eval_at_t(c_pt)
    fd = eval_at_t(d_pt)

    for _ in range(40):
        if x_hi - x_lo < 1e-12:
            break
        if fc < fd:
            x_hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = x_hi - gr * (x_hi - x_lo)
            fc = eval_at_t(c_pt)
        else:
            x_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = x_lo + gr * (x_hi - x_lo)
            fd = eval_at_t(d_pt)

    t_opt = (x_lo + x_hi) / 2.0
    obj_opt = eval_at_t(t_opt)
    if obj_opt < best_obj:
        best_t = t_opt
        best_obj = obj_opt

    return best_t, best_obj


def cd_sweep(a, max_time=None, rng=None, start_time=None):
    """One sweep of exact coordinate descent with golden section 1D subproblems."""
    n = len(a)
    a = a.copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    if rng is None:
        rng = np.random.default_rng(0)

    order = rng.permutation(n)

    for i in order:
        if max_time is not None and start_time is not None:
            if time.monotonic() - start_time >= max_time:
                break

        v = a[i]
        S_other = S - v

        if S_other < 1e-14:
            continue

        k_min = max(0, i)
        k_max = min(2 * n - 2, i + n - 1)
        ks = np.arange(k_min, k_max + 1)
        a_ki = a[ks - i]
        c_k = conv[ks].copy()

        mask = (ks != 2 * i)
        ks_lin = ks[mask]
        a_ki_lin = a_ki[mask]
        c_k_lin = c_k[mask]

        k_quad = 2 * i
        c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

        def f_1d(x):
            if len(a_ki_lin) > 0:
                lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                max_lin = float(np.max(lin_vals))
            else:
                max_lin = 0.0
            quad_val = c_2i + x * x - v * v
            max_val = max(max_lin, quad_val)
            denom = (S_other + x) * (S_other + x)
            if denom < 1e-30:
                return 1e30
            return 2.0 * n * max_val / denom

        # Candidate points for initial bracket
        scale = max(v, S / n, 1e-6)
        candidates = [0.0, v * 0.3, v * 0.5, v * 0.7, v,
                      v * 1.2, v * 1.5, v * 2.0, v * 3.0,
                      S_other * 0.3, S_other * 0.7, S_other, S_other + v]
        candidates = sorted(set(max(0.0, float(c)) for c in candidates))

        best_x = v
        best_f = f_1d(v)
        for x in candidates:
            fx = f_1d(x)
            if fx < best_f:
                best_f = fx
                best_x = x

        # Golden section refine
        x_lo = max(0.0, best_x - scale)
        x_hi = best_x + scale * 1.5
        if x_hi - x_lo < 1e-12:
            x_hi = x_lo + 1e-8

        c_pt = x_hi - gr * (x_hi - x_lo)
        d_pt = x_lo + gr * (x_hi - x_lo)
        fc = f_1d(c_pt)
        fd = f_1d(d_pt)

        for _ in range(45):
            if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                break
            if fc < fd:
                x_hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = x_hi - gr * (x_hi - x_lo)
                fc = f_1d(c_pt)
            else:
                x_lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = x_lo + gr * (x_hi - x_lo)
                fd = f_1d(d_pt)

        x_opt = max(0.0, (x_lo + x_hi) / 2.0)

        if abs(x_opt - v) < 1e-15:
            continue

        delta = x_opt - v
        a[i] = x_opt

        if 0 <= k_quad <= 2 * n - 2:
            conv[k_quad] += x_opt * x_opt - v * v
        if len(ks_lin) > 0:
            conv[ks_lin] += 2.0 * delta * a_ki_lin

        S += delta

    # Renormalize
    S_final = float(np.sum(a))
    if S_final > 1e-10:
        a = a / S_final
    return a


def lp_cd_alternate(a, time_budget, rng, verbose=True):
    """Alternate between LP directions and CD sweeps with exact line search."""
    n = len(a)
    a = a.copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 10:
            break

        # LP step
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is not None:
            # Line search for optimal blend
            t_opt, obj_blend = line_search_blend(a, g_lp, n_points=20)
            if obj_blend < best_obj - 1e-12:
                a_new = (1.0 - t_opt) * a + t_opt * g_lp
                a_new = np.maximum(a_new, 0.0)
                s_new = float(np.sum(a_new))
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    best_obj = obj_blend
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        # CD step
        cd_time = min(remaining * 0.3, 30.0)
        if cd_time > 3:
            a_cd = cd_sweep(a, max_time=cd_time, rng=rng,
                            start_time=time.monotonic())
            obj_cd = compute_obj_normalized(a_cd)
            if obj_cd < best_obj - 1e-12:
                best_obj = obj_cd
                best_a = a_cd.copy()
                a = a_cd
                stall = 0
            else:
                stall += 1

        # Perturbation if stalled
        if stall >= 3:
            pert_size = 0.02 + 0.01 * (stall // 3)
            pert = rng.normal(0, pert_size, n)
            a = a * np.exp(pert)
            a = np.maximum(a, 0.0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
            stall = 0
            # Quick CD after perturbation
            if remaining > 20:
                a = cd_sweep(a, max_time=min(10, remaining * 0.2), rng=rng,
                             start_time=time.monotonic())
                obj_pert = compute_obj_normalized(a)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a.copy()

        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP-CD it{it}: obj={best_obj:.8f}, t={elapsed:.1f}s, stall={stall}")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x ** 2 + 1e-8)
        a = a / a.sum()
    elif mode == 4:
        # Bimodal asymmetric
        a = np.zeros(n)
        a[:n // 3] = 1.0
        a[2 * n // 3:] = 0.5
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 4)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
    elif mode == 6:
        # Comb pattern
        a = np.zeros(n)
        period = max(2, n // 10)
        for i in range(0, n, period):
            a[i] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP-CD alternation with exact line search, perturbative
    restarts, multi-n exploration, and diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj_normalized(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    # === Phase 1: LP-CD Alternation (main workhorse) ===
    print("\n=== Phase 1: LP-CD Alternation ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.5, 700.0)
    if p1_budget > 30:
        a_p1 = lp_cd_alternate(best_seq, time_budget=p1_budget, rng=rng, verbose=True)
        obj_p1 = compute_obj_normalized(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_p1:.8f} (no improvement)")

    # === Phase 2: Second LP-CD round with perturbation ===
    print("\n=== Phase 2: Perturbed LP-CD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 30:
        # Perturb and try again
        for attempt in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            pert = rng.normal(0, 0.015 + 0.005 * attempt, len(best_seq))
            a_pert = best_seq * np.exp(pert)
            a_pert = np.maximum(a_pert, 0.0)
            s_pert = float(np.sum(a_pert))
            if s_pert < 1e-10:
                continue
            a_pert = a_pert / s_pert

            alloc = min(p2_budget / (3 - attempt), remaining * 0.6)
            if alloc < 20:
                break
            a_p2 = lp_cd_alternate(a_pert, time_budget=alloc,
                                   rng=np.random.default_rng(seed + 100 * attempt),
                                   verbose=False)
            obj_p2 = compute_obj_normalized(a_p2)
            if obj_p2 < best_score:
                best_score = obj_p2
                best_seq = a_p2.copy()
                print(f"  Perturb {attempt}: obj={best_score:.8f} **")
            else:
                print(f"  Perturb {attempt}: obj={obj_p2:.8f}")

    # === Phase 3: Pure CD refinement ===
    print("\n=== Phase 3: CD Refinement ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 15:
        n_sweeps = int(p3_budget / 5)
        n_sweeps = max(3, min(n_sweeps, 30))
        for sweep_i in range(n_sweeps):
            if time.monotonic() > deadline:
                break
            a_cd = cd_sweep(best_seq, max_time=min(15, remaining * 0.1),
                            rng=np.random.default_rng(seed + 500 + sweep_i),
                            start_time=time.monotonic())
            obj_cd = compute_obj_normalized(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                if verbose_check(sweep_i):
                    print(f"  CD sweep {sweep_i}: obj={best_score:.8f} **")
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

    # === Phase 4: Multi-n Exploration ===
    print("\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 100), min(2000, n_best + 100),
        min(2000, n_best + 200),
        600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) > 50]
    n_options = n_options[:6]

    print(f"  Best n={n_best}, trying: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Quick LP-CD on new length
        alloc = min(remaining * 0.2, 60.0)
        if alloc > 15:
            a_try = lp_cd_alternate(a_try, time_budget=alloc,
                                    rng=np.random.default_rng(n_try),
                                    verbose=False)
        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts ===
    print("\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = (r + 1) % 7
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r < 1e-10:
            continue
        a_r = a_r / s_r

        alloc = min(remaining * 0.2, 60.0)
        if alloc > 15:
            a_r = lp_cd_alternate(a_r, time_budget=alloc,
                                  rng=np.random.default_rng(5000 + r * 100),
                                  verbose=False)
            obj_r = compute_obj_normalized(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final LP Polish ===
    print("\n=== Phase 6: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 120.0)
        a_final = lp_cd_alternate(best_seq, time_budget=alloc,
                                  rng=np.random.default_rng(seed + 9999),
                                  verbose=True)
        obj_final = compute_obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # === Final CD polish if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_last = cd_sweep(best_seq, max_time=min(remaining * 0.5, 50),
                          rng=np.random.default_rng(seed + 7777),
                          start_time=time.monotonic())
        obj_last = compute_obj_normalized(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result


def verbose_check(sweep_i):
    """Helper for conditional printing."""
    return sweep_i % 5 == 0
```
