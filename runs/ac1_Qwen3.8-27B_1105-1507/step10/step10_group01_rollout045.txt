        return np.inf
    a_norm = a / s

def build_conv_matrix(a):

    n = len(a)

        return 0, f_0
    d_clipped_d = d_clipped_d / s_d
    f_1 = compute_obj(d_clipped_d)

    if f_1 >= f_0:
        # Minimize is near 0, use golden section in [0, 0.5]
        lo, hi = 0.0, 0.5
    else:
        # Minimize is somewhere in [0, 1]
        lo, hi = 0.0, 1.0

    if hi <= lo:
        return 0, f_0

    gr = (np.sqrt(5) - 1) / 2
    c = lo + (1 - gr) * (hi - lo)
    d_pt = lo + gr * (hi - lo)

    # Ensure valid
    c = max(c, 1e-6)
    d_pt = max(d_pt, 1e-6)

    a_c = (1 - c) * a + c * d
    a_c = np.maximum(a_c, 0)
    sc = float(np.sum(a_c))
    f_c = compute_obj(a_c) if sc > 1e-10 else np.inf

    a_d = (1 - d_pt) * a + d_pt * d
    a_d = np.maximum(a_d, 0)
    sd = float(np.sum(a_d))
    f_d = compute_obj(a_d) if sd > 1e-10 else np.inf

    for _ in range(max_evals - 2):
        if f_c <= f_d:
            hi = d_pt
            d_pt = c
            f_d = f_c
            c = lo + (1 - gr) * (hi - lo)
            c = max(c, 1e-8)
            a_c = (1 - c) * a + c * d
            a_c = np.maximum(a_c, 0)
            sc = float(np.sum(a_c))
            f_c = compute_obj(a_c) if sc > 1e-10 else np.inf
        else:
            lo = c
            c = d_pt
            f_c = f_d
            d_pt = lo + gr * (hi - lo)
            d_pt = min(d_pt, 1.0 - 1e-8)
            a_d = (1 - d_pt) * a + d_pt * d
            a_d = np.maximum(a_d, 0)
            sd = float(np.sum(a_d))
            f_d = compute_obj(a_d) if sd > 1e-10 else np.inf

        if hi - lo < 1e-7:
            break

    t_best = (lo + hi) / 2
    t_best = min(max(t_best, 0.0), 1.0)
    a_best = (1 - t_best) * a + t_best * d
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))
    if s_best < 1e-10:
        return 0, f_0
    f_best = compute_obj(a_best)
    return t_best, f_best


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A_f = fft(a, N)
    g_full = ifft(P * np.conj(A_f)).real
    return 2.0 * g_full[:n]


def nesterov_gd(a, time_budget, tau_start, tau_end, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    it = 0
    best_it = 0
    patience = 8000

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 1.2

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        # Adaptive LR based on gradient norm
        if g_norm > 1e-15:
            lr = 0.8 * lr + 0.2 * (base_lr / (g_norm + base_lr))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-12, lr * 0.99995)

        if it - best_it > patience:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            beta = 0.8
            patience = int(patience * 0.7)
            if patience < 500:
                break

    if verbose:
        print(f"    Nesterov GD: obj={best_obj:.10f}, iters={it}, final_tau~{tau:.0f}")
    return best_a


def coordinate_descent(a, time_budget, tau=500.0, max_sweeps=200):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, the update that minimizes the smoothed objective is found
    by solving a small 1D problem.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Precompute the "off-diagonal" contribution for each coordinate
    # conv(a,a)[k] includes terms a[i]*a[k-i] for each valid i
    # For coordinate i, the contribution to conv[k] is a[i]*a[k-i] + a[k-i]*a[i]
    # (unless i == k-i, in which case it's a[i]^2)
    # When we change a[i] to a[i]+delta, conv[k] changes by:
    #   delta * a[k-i] + delta * a[i-k] (for k != 2i)
    #   or 2*delta * a[i] + delta^2 (for k = 2i)
    # This is complex, so we use a simpler approach:
    # For each i, try a few values and pick the best.

    rng = np.random.default_rng(999)

    sweep = 0
    while time.monotonic() - start < time_budget and sweep < max_sweeps:
        sweep += 1
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break

            # Try a few multipliers for coordinate i
            old_val = a[i]
            if old_val < 1e-12:
                # Try small positive values
                candidates = [0.0, 1e-4, 1e-3]
            else:
                candidates = [0.0, old_val * 0.5, old_val * 0.75, old_val,
                              old_val * 1.25, old_val * 1.5, old_val * 2.0]

            best_obj_i = best_obj
            best_val_i = old_val

            for cv in candidates:
                a[i] = cv
                s_try = float(np.sum(a))
                if s_try < 1e-10:
                    a[i] = old_val
                    continue
                a_try = a / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj_i - 1e-14:
                    best_obj_i = obj_try
                    best_val_i = cv

            if best_val_i != old_val:
                a[i] = best_val_i
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                improved = True
                if best_obj_i < best_obj - 1e-14:
                    best_obj = best_obj_i
                    best_a = a.copy()
            else:
                a[i] = old_val

        if not improved and sweep > 5:
            break

    if best_obj < compute_obj(a):
        return best_a
    return a


def lp_iterate(a, time_budget, verbose=False, stall_limit=40):
    """
    Main LP-driven optimization loop.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    consecutive_no_improve = 0
    iter_count = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        a_norm = a / float(np.sum(a))
        g, M_lp = solve_lp_direction(a_norm)
        if g is None:
            break

        # Line search
        t_best, obj_at_t = line_search(a, g, max_evals=35)

        iter_count += 1

        if obj_at_t < best_obj - 1e-13:
            a_try = (1 - t_best) * a + t_best * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                best_obj = obj_at_t
                best_a = a_try.copy()
                a = a_try
                consecutive_no_improve = 0
                if verbose and (iter_count % 20 == 0 or iter_count <= 3):
                    elapsed = time.monotonic() - start
                    print(f"    LP it{iter_count}: obj={best_obj:.10f}, t={t_best:.4f}, "
                          f"LP_M={M_lp:.8f}, time={elapsed:.0f}s")
        else:
            consecutive_no_improve += 1

            # Try small steps
            if consecutive_no_improve <= 3:
                for t_small in [0.1, 0.05, 0.02]:
                    a_small = (1 - t_small) * a + t_small * g
                    a_small = np.maximum(a_small, 0)
                    s_small = float(np.sum(a_small))
                    if s_small > 1e-10:
                        a_small = a_small / s_small
                        obj_small = compute_obj(a_small)
                        if obj_small < best_obj - 1e-13:
                            best_obj = obj_small
                            best_a = a_small.copy()
                            a = a_small
                            consecutive_no_improve = 0
                            if verbose:
                                print(f"    LP it{iter_count}: small step t={t_small}, obj={best_obj:.10f}")
                            break

            if consecutive_no_improve >= stall_limit:
                # Perturb to escape
                tau_pert = 300.0
                conv_p = compute_conv(a)
                max_p = float(np.max(conv_p))
                shifted_p = tau_pert * (conv_p - max_p)
                shifted_p = np.clip(shifted_p, -700, 700)
                exp_p = np.exp(shifted_p)
                p = exp_p / exp_p.sum()
                N_fft = 1
                while N_fft < 3 * n:
                    N_fft <<= 1
                P = fft(p, N_fft)
                Af = fft(a, N_fft)
                grad = 2.0 * ifft(P * np.conj(Af)).real[:n]

                # Move against gradient
                lr_p = 0.01 / (float(np.linalg.norm(grad)) + 1e-10)
                a = a - lr_p * grad
                a = np.maximum(a, 0)
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                else:
                    a = best_a.copy()

                # Add random perturbation
                a = a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()

                consecutive_no_improve = 0

                if iter_count > 500:
                    if verbose:
                        print(f"    LP: stopping after {iter_count} iterations, obj={best_obj:.10f}")
                    break

    if verbose:
        elapsed = time.monotonic() - start
        print(f"    LP complete: obj={best_obj:.10f}, iters={iter_count}, time={elapsed:.0f}s")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, kind='uniform'):
    """Create a structured initial sequence."""
    x = np.linspace(0, 1, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'half_cos':
        a = np.cos(np.pi * x / 2.0)
        a = np.maximum(a, 0.01)
    elif kind == 'trapezoid':
        a = np.minimum(x * 4, 1.0)
        a = np.minimum(a, (1 - x) * 4)
        a = np.maximum(a, 0.01)
    elif kind == 'gaussian':
        a = np.exp(-20 * (x - 0.5) ** 2)
        a = np.maximum(a, 0.01)
    elif kind == 'bump':
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1500):

            s = float(np.sum(parent_arr))

                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()

                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = structured_init(n_init, 'uniform')

        a_lp = lp_iterate(best_seq, time_budget=p1_budget, verbose=True, stall_limit=50)

        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()

            print(f"  Phase 2 result: obj={best_score:.10f} **")

    print(f"Phase 3: Coordinate Descent Polish")

        a_r = structured_init(n_cur, rk)

        a_r = np.maximum(a_r, 0)
