import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_reduced(a, top_k=200):
    """Solve LP with only top-k active constraints for speed."""

        all_indices = np.concatenate([top_indices, random_extra])
    else:

    A = A_full[all_indices, :]
    
    g = cp.Variable(n, nonneg=True)

def line_search_bisection(a, g, t_max=1.5, n_coarse=30, n_fine=30):
    """
    Smart line search using coarse grid + subgradient bisection.
    Minimizes h(t) = max_k conv((1-t)a + t*g, (1-t)a + t*g)[k]
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    g_norm = g / float(np.sum(g))
    
    # Precompute convolutions
    conv_aa = compute_conv(a_norm)
    conv_ag = np.convolve(a_norm, g_norm)  # cross-convolution
    conv_gg = compute_conv(g_norm)
    
    def h(t):
        """h(t) = max_k P_k(t) where P_k(t) = (1-t)^2*aa[k] + 2t(1-t)*ag[k] + t^2*gg[k]"""
        if t < 0:
            t = 0
        val = (1-t)**2 * conv_aa + 2*t*(1-t) * conv_ag + t**2 * conv_gg
        return float(np.max(val)), val
    
    def h_subgrad(t):
        """Subgradient of h at t (gradient of the active quadratic)."""
        val = (1-t)**2 * conv_aa + 2*t*(1-t) * conv_ag + t**2 * conv_gg
        k_max = int(np.argmax(val))
        # P_k(t) = (1-t)^2 * aa_k + 2t(1-t) * ag_k + t^2 * gg_k
        # P_k'(t) = 2(1-t)(-1)*aa_k + 2(1-2t)*ag_k + 2t*gg_k
        #         = -2(1-t)*aa_k + 2(1-2t)*ag_k + 2t*gg_k
        return 2 * (-(1-t) * conv_aa[k_max] + (1-2*t) * conv_ag[k_max] + t * conv_gg[k_max])
    
    # Coarse grid search
    ts = np.linspace(0, t_max, n_coarse)
    hs = [h(t)[0] for t in ts]
    h_min_coarse = min(hs)
    t_coarse = ts[np.argmin(hs)]
    
    if h_min_coarse >= h(0.0)[0] - 1e-15:
        return 0.0, h(0.0)[0]
    
    # Fine bisection around the coarse minimum
    # Find a bracket [lo, hi] where the subgradient changes sign
    bracket_ts = np.linspace(max(0, t_coarse - 0.3), min(t_max, t_coarse + 0.3), 50)
    best_t = t_coarse
    best_h = h_min_coarse
    
    for t in bracket_ts:
        h_val, _ = h(t)
        if h_val < best_h:
            best_h = h_val
            best_t = t
    
    # Subgradient bisection
    lo, hi = max(0.0, best_t - 0.2), min(t_max, best_t + 0.2)
    for _ in range(n_fine):
        mid = (lo + hi) / 2
        sg = h_subgrad(mid)
        if abs(sg) < 1e-10:
            best_t = mid
            break
        if sg < 0:
            lo = mid
        else:
            hi = mid
        # Update best
        h_val, _ = h((lo+hi)/2)
        if h_val < best_h:
            best_h = h_val
            best_t = (lo+hi)/2
    
    # Final check at a few points
    for t_check in [0.0, 0.05, 0.1, 0.2, 0.3, 0.5, best_t, best_t*0.9, best_t*1.1, t_max*0.5]:
        if 0 <= t_check <= t_max:
            h_val, _ = h(t_check)
            if h_val < best_h:
                best_h = h_val
                best_t = t_check
    
    return best_t, best_h

def smooth_obj(a, epsilon):
    """Smooth approximation: -epsilon * log(sum(exp(conv/epsilon)))"""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    shifted = conv / epsilon - np.max(conv) / epsilon
    return 2.0 * n * epsilon * (np.max(conv) / epsilon + np.log(np.sum(np.exp(shifted))))


def smooth_grad(a, epsilon):
    """Gradient of the smooth objective."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf, np.zeros(n)
    a = a / s
    conv = compute_conv(a)
    shifted = conv - np.max(conv)
    p = np.exp(shifted / epsilon)
    p = p / p.sum()
    
    # Gradient of sum_k p_k * conv_k with respect to a
    # But we need the gradient w.r.t. the unnormalized a, then handle the 1/s factor
    # For simplicity, work with normalized a
    
    grad = np.zeros(n)
    for k in range(len(conv)):

            continue
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * p[k] * a[j]
    
    return 2.0 * n * (np.max(conv) + epsilon * np.log(p.sum())), grad


def fista_smooth(a, time_budget=30.0, epsilon_init=0.01, epsilon_final=0.001):
    """FISTA on the smooth approximation."""
    n = len(a)

    t_k = 1.0
    epsilon = epsilon_init
    lr = 0.5
    
    while time.monotonic() - start < time_budget:
        # Animate
        a_curr = z + ((t_k - 1) / (t_k + 1)) * (z - a)
        a_curr = np.maximum(a_curr, 0)
        s_curr = float(np.sum(a_curr))
        if s_curr < 0.01:
            a_curr = best_a.copy()
        else:
            a_curr = a_curr / s_curr
        
        # Gradient step
        _, grad = smooth_grad(a_curr, epsilon)
        a_new = a_curr - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 0.01:
            a_new = a_new / s_new
        
        # Update t
        t_new = (1 + np.sqrt(1 + 4 * t_k**2)) / 2
        
        # Project onto simplex (approximately via normalization)
        # The gradient step already maintains non-negativity
        z = a
        a = a_new
        t_k = t_new
        
        # Decay epsilon

        # Compute actual objective periodically

            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(3 * (2 * x - 1))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def lp_polish(a, time_budget=120.0, verbose=False, max_iters=100, use_reduced=True):
    """
    LP polishing with smart line search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(99999)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Solve LP (reduced or full)
        T0 = time.monotonic()
        if use_reduced and n > 200:
            g_lp, M_lp = solve_lp_reduced(a, top_k=min(200, 2*n-1))
        else:
            g_lp, M_lp = solve_lp_full(a)
        T1 = time.monotonic()
        lp_time = T1 - T0

        if g_lp is None:
            break

        # Smart line search
        t_opt, obj_opt = line_search_bisection(a, g_lp)

        if t_opt > 0.001 and obj_opt < best_obj - 1e-14:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, M_lp={M_lp:.6f}, "
                  f"lp_time={lp_time:.2f}s, elapsed={time.monotonic()-start:.1f}s")

        # Adaptive restart
        if stall >= 3:
            if stall <= 5:
                # Fourier perturbation
                freq = rng.integers(1, max(2, n // 100))
                phase = rng.uniform(0, 2 * np.pi)
                x = np.arange(n, dtype=np.float64) / n
                pert = 0.02 * stall * np.cos(2 * np.pi * freq * x + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                else:
                    a = best_a.copy()
            elif stall <= 8:
                # Add noise
                noise = rng.normal(0, 0.02 * stall, n)
                a = best_a + noise
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                else:
                    a = best_a.copy()
            else:
                # Major restart
                mode = it % 7
                a_div = diverse_init(n, rng, mode=mode)
                a = 0.3 * a_div + 0.7 * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()

            stall = 0

    return best_a


def block_coordinate_descent(a, time_budget=30.0, block_size=4):
    """
    Block coordinate descent: optimize blocks of the sequence.
    For each block, use the LP to find the best update.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    n_blocks = n // block_size
    
    while time.monotonic() - start < time_budget:
        improved = False
        for block in range(n_blocks):
            if time.monotonic() - start > time_budget:
                break
            
            i_start = block * block_size
            i_end = min(i_start + block_size, n)
            block_idx = list(range(i_start, i_end))
            
            # Create a reduced problem: optimize only these coordinates
            # Fix all other coordinates, optimize the block subject to sum(a)=1
            a_fixed = a.copy()
            a_fixed[block_idx] = 0
            sum_fixed = float(np.sum(a_fixed))
            
            if sum_fixed >= 1.0:
                continue  # Can't add any mass to this block
            
            mass_available = 1.0 - sum_fixed
            
            # Solve a small LP: min M s.t. conv(a, g)[k] <= M, sum(g)=mass_available for g in block
            # This is complex, so just do a simple gradient step on the block
            conv = compute_conv(a)
            k_max = int(np.argmax(conv))
            
            # Gradient for the block coordinates
            grad_block = np.zeros(block_size)
            for ii, i in enumerate(block_idx):
                j = k_max - i
                if 0 <= j < n:
                    grad_block[ii] = a[j]
            
            # Simple update: move mass from high-gradient to low-gradient within block
            # Actually, just do a simple projected gradient step
            lr = 0.01
            a_new_block = a[block_idx] - lr * grad_block
            a_new_block = np.maximum(a_new_block, 0)
            
            # Renormalize: adjust the block so total sum = 1
            s_new = sum_fixed + float(np.sum(a_new_block))
            if s_new > 0.01:
                a[block_idx] = a_new_block * (1.0 / s_new)
                a_fixed_block = a.copy()
                a_fixed_block[block_idx] = 0
                # Scale fixed part
                a = a_fixed_block + a[block_idx]
                # Actually this is getting messy. Let me just normalize the whole thing.
                a = np.maximum(a, 0)
                a = a / a.sum()
            
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a


def idx_func(n):
    """Helper: return arange(n) normalized to [0,1]."""
    return np.arange(n, dtype=np.float64) / max(1, n - 1)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP polishing, smart line search,
    smooth optimization, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default U-shaped init (n={n_current}): obj={best_score:.8f}")

    # Try several initializations
    for sharp in [1.0, 2.0, 3.0, 5.0, 8.0, 15.0]:
        a_init = u_shaped_init(n_current, sharpness=sharp)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  U-sharp={sharp}: obj={obj_init:.8f} **")

    for mode in range(7):
        a_init = diverse_init(n_current, rng, mode=mode)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Diverse mode={mode}: obj={obj_init:.8f} **")

    # === Phase 1: Subgradient warm-up ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 1: Subgradient warm-up ({remaining:.0f}s remaining) ===")
        a_sgd = subgradient_descent(best_seq, time_budget=min(40.0, remaining * 0.1))
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_sgd:.8f} (no improvement)")

    # === Phase 2: Main LP polishing (reduced constraints) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: LP Polish (reduced) ({remaining:.0f}s remaining) ===")
        alloc = min(remaining * 0.35, 300.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True, max_iters=150, use_reduced=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: LP polishing (full constraints) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 3: LP Polish (full) ({remaining:.0f}s remaining) ===")
        alloc = min(remaining * 0.3, 200.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True, max_iters=80, use_reduced=False)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj_lp:.8f} (no improvement)")

    # === Phase 4: Smooth optimization polish ===
    remaining = deadline - time.monotonic()
