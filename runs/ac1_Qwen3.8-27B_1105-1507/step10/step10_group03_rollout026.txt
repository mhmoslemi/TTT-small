"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key insight: asymmetric (ramp-like) initializations give much better starting
points than symmetric ones. The linear ramp alone achieves ~4/3 for large n.
Uses Nesterov accelerated gradient descent with adaptive tau, active-set LP
polishing, and multi-n exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smoothed_grad(a, tau):
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max

def nesterov_gd(a, time_budget, tau_levels=None, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    if tau_levels is None:

    beta = 0.9
    for tau in tau_levels:
        lr = 3.0 / np.sqrt(n)

        it_level = 0

        local_best = best_obj

            if it_level >= max_iters:

            y = np.maximum(y, 0)
            sy = float(np.sum(y))

                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy
            
            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-14:
                break
            
            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)

def active_set_lp(a, max_constraints=50, tolerance=0.01):
    """
    Solve min M s.t. (a*g)(k) <= M for active k, g >= 0, sum(g) = 1.
    Active k are those where (a*a)(k) is within tolerance of max.
    """
    n = len(a)
    if n > 3000:
        return None
    
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    
    # Find active constraints
    active = np.where(conv >= max_val * (1 - tolerance))[0]
    if len(active) > max_constraints:
        # Keep the ones closest to max
        active_sorted = active[np.argsort(conv[active])[::-1][:max_constraints]]
    else:
        active_sorted = active
    
    # Build constraint matrix for active constraints only
    # (a*g)(k) = sum_j a_j * g_{k-j} (where indices are valid)
    m = len(active_sorted)
    A = np.zeros((m, n))
    for idx, k in enumerate(active_sorted):
        # (a*g)(k) = sum_{j: 0<=j<n, 0<=k-j<n} a_j * g_{k-j}
        # = sum_j a_{k-j} * g_j (reindexing: let i = k-j, j = k-i)
        # = sum_i a_i * g_{k-i} where 0<=k-i<n => k-n+1<=i<=k
        # So for each i in [max(0,k-n+1), min(k,n-1)]: A[idx, k-i] += a_i
        for i in range(max(0, k - n + 1), min(k, n - 1) + 1):
            A[idx, k - i] += a[i]
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg
        except Exception:
            continue
    return None
