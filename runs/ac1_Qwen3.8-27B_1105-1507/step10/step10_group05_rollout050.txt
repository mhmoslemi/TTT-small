"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based exact subgradient descent with multiplicative weights,
log-sum-exp smoothed gradient with Adam, and LP polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def subgradient_fft(p, active_set):
    """
    Compute subgradient of max_k (p*p)[k] using active set S.
    g[j] = (1/|S|) sum_{k in S} p[k-j] (masked to valid range)
    Uses FFT for O(n log n) computation.
    """
    n = len(p)
    m = len(active_set)
    if m == 0:
        return np.zeros(n)
    
    # Create indicator of active set
    s = np.zeros(2 * n - 1)
    s[active_set] = 1.0 / m
    
    # We need g[j] = sum_i p[i] * s[j+i] for j in [0, n-1]
    # This equals (q * s)[j + n - 1] where q = p reversed
    N = 1
    while N < (n - 1) + (2 * n - 1):  # need >= 3n - 3 + 1 = 3n-2
        N <<= 1
    
    q = p[::-1]
    Q = fft(q, N)
    S = fft(s, N)
    conv_qs = ifft(Q * S).real
    
    g = conv_qs[n - 1: 2 * n - 1]
    return g


def logsumexp_gradient(p, tau):
    """
    Compute smoothed gradient of logsumexp(tau * (p*p)) / tau.
    Returns (gradient, smoothed_value, active_probs).
    """
    n = len(p)
    conv = compute_conv(p)
    max_conv = float(np.max(conv))
    
    # Compute softmax weights
    shifted = tau * (conv - max_conv)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        # Degenerate case: exact max
        active = np.where(conv >= max_conv - 1e-10)[0]
        s = np.zeros(len(conv))
        s[active] = 1.0 / len(active)
    else:
        s = exp_vals / sum_exp
    
    # Smoothed value
    smooth_val = max_conv - math.log(sum_exp) / tau if sum_exp > 0 else max_conv
    
    # Gradient: same as subgradient but with soft weights s instead of hard indicator
    # g[j] = sum_k s[k] * p[k-j] (masked) = (q * s)[j + n - 1]
    N = 1
    while N < (n - 1) + (2 * n - 1):
        N <<= 1
    
    q = p[::-1]
    Q = fft(q, N)
    S = fft(s, N)
    conv_qs = ifft(Q * S).real
    
    g = conv_qs[n - 1: 2 * n - 1]
    
    return g, smooth_val, s


def mwd_descent(p_init, time_budget=100.0, verbose=False, tol=1e-6):
    """
    Multiplicative Weights Descent with exact subgradient.
    Minimizes max_k (p*p)[k] subject to sum(p)=1, p>=0.
    """
    n = len(p_init)
    p = np.maximum(p_init, 0).copy()
    s = float(np.sum(p))
    if s < 1e-15:
        return p
    p = p / s
    
    best_p = p.copy()
    best_obj = float(np.max(compute_conv(p)))
    start = time.monotonic()
    
    eta = 1.0 / math.sqrt(n)
    consecutive_no_improve = 0
    
    while time.monotonic() - start < time_budget:
        # Compute convolution and find active set
        conv = compute_conv(p)
        max_conv = float(np.max(conv))
        
        if max_conv < best_obj * (1 - 1e-12):
            best_obj = max_conv
            best_p = p.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        # Active set: indices within tol of max
        threshold = max_conv - tol * max_conv
        active_set = np.where(conv >= threshold)[0]
        
        if len(active_set) == 0:
            active_set = np.array([np.argmax(conv)])
        
        # Compute subgradient
        g = subgradient_fft(p, active_set)
        
        # Normalize gradient (subtract mean for stability)
        g_centered = g - np.mean(g)
        
        # Multiplicative weights update
        # p_new = p * exp(-eta * g_centered) / Z
        log_p_new = np.log(np.maximum(p, 1e-300)) - eta * g_centered
        log_p_new -= np.max(log_p_new)  # numerical stability
        p_new = np.exp(log_p_new)
        p_new = np.maximum(p_new, 0)
        s_new = float(np.sum(p_new))
        if s_new < 1e-15:

        p_new = p_new / s_new

        max_new = float(np.max(conv_new))

            p = p_new

            eta *= 1.01
        else:

            # Try smaller step

            log_p_new2 -= np.max(log_p_new2)
            p_new2 = np.exp(log_p_new2)
            p_new2 = np.maximum(p_new2, 0)
            s_new2 = float(np.sum(p_new2))
            if s_new2 > 1e-15:

                if float(np.max(conv_new2)) < max_conv:
                    p = p_new2
                    eta *= 0.8

                break

        if consecutive_no_improve > 200:
            break
        if eta < 1e-8:
            break
    
    return best_p

Optimizes step function to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated multiplicative weights descent with adaptive active sets,
log-sum-exp smoothed refinement, and targeted LP polishing.
Key: exact subgradient via FFT for O(n log n) per iteration.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    """
    Compute weighted subgradient: g[j] = sum_k s_weights[k] * p[k-j] (masked).
    s_weights should sum to 1 over the active set.
    Uses FFT for O(n log n) computation.

    Q = fft(q, N)
    S = fft(s_weights, N)
    conv_qs = ifft(Q * S).real

    p = normalize_nonneg(p_init)
    
    best_p = p.copy()
    best_peak = float(np.max(compute_conv(p)))
    start = time.monotonic()
    
    eta = 2.0 / math.sqrt(n)
    active_tol = 0.01  # fraction of max to include in active set
    eta_min = 1e-10
    no_improve = 0
    it = 0
    
    while time.monotonic() - start < time_budget and eta > eta_min:
        conv = compute_conv(p)
        peak = float(np.max(conv))
        
        if peak < best_peak * (1 - 1e-14):
            best_peak = peak
            best_p = p.copy()
            no_improve = 0
            # Tighten active set as we converge
            active_tol = max(0.001, active_tol * 0.95)
        else:
            no_improve += 1
        
        # Find active set
        threshold = peak * (1 - active_tol)
        active_mask = conv >= threshold
        active_count = int(np.sum(active_mask))
        if active_count == 0:
            active_mask[np.argmax(conv)] = True
            active_count = 1
        
        # Weighted subgradient (uniform weights over active set)
        s_weights = active_mask.astype(np.float64) / active_count
        g = subgradient_fft(p, s_weights)
        g_centered = g - np.mean(g)
        
        # MWD update
        log_p = np.log(np.maximum(p, 1e-30))
        log_p_new = log_p - eta * g_centered
        log_p_new -= np.max(log_p_new)
        p_new = np.exp(log_p_new)
        p_new = np.maximum(p_new, 0)
        s_new = float(np.sum(p_new))
        if s_new < 1e-15:
            break
        p_new = p_new / s_new
        
        # Evaluate
        conv_new = compute_conv(p_new)
        peak_new = float(np.max(conv_new))
        
        if peak_new < peak:
            p = p_new
            # Slightly increase eta if making good progress
            if no_improve == 0:
                eta = min(eta * 1.02, 5.0 / math.sqrt(n))
        else:
            # Backtrack
            eta *= 0.5
            if eta > eta_min / 10:
                # Retry with smaller step
                log_p_new2 = log_p - 0.5 * eta * g_centered
                log_p_new2 -= np.max(log_p_new2)
                p_new2 = np.exp(log_p_new2)
                p_new2 = np.maximum(p_new2, 0)
                s_new2 = float(np.sum(p_new2))
                if s_new2 > 1e-15:
                    p_new2 = p_new2 / s_new2
                    if float(np.max(compute_conv(p_new2))) < peak:
                        p = p_new2
                        eta *= 0.8
        
        it += 1
        if no_improve > 500:
            # Restart with perturbation
            rng = np.random.default_rng(it)
            p = p * (1 + 0.01 * rng.standard_normal(n))
            p = normalize_nonneg(p)
            no_improve = 0
            active_tol = 0.01
            eta = 2.0 / math.sqrt(n)
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MWD it{it}: peak={best_peak:.10f}, obj={2*n*best_peak:.8f}, "
                  f"eta={eta:.6f}, |S|={active_count}, tol={active_tol:.5f}, t={elapsed:.0f}s")
    
    return best_p


def smoothed_adam_descent(p_init, time_budget=100.0, tau_init=5.0, tau_end=200.0, verbose=False):
    """
    Adam optimizer on log-sum-exp smoothed max of convolution.
    """
    n = len(p_init)
    p = normalize_nonneg(p_init)
    
    best_p = p.copy()
    best_peak = float(np.max(compute_conv(p)))
    start = time.monotonic()
    
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = 0.05
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * (1 - math.cos(math.pi * progress)) / 2
        
        # Compute smoothed gradient
        conv = compute_conv(p)
        max_c = float(np.max(conv))
        shifted = tau * (conv - max_c)
        shifted = np.clip(shifted, -100, 0)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        
        if sum_exp > 1e-300:
            s_weights = exp_vals / sum_exp
        else:
            s_weights = np.zeros(len(conv))
            s_weights[np.argmax(conv)] = 1.0
        
        g = subgradient_fft(p, s_weights)
        g -= np.mean(g)
        
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        p_new = p - lr * m_hat / (np.sqrt(v_hat) + eps)
        p_new = normalize_nonneg(p_new)
        
        conv_new = compute_conv(p_new)
        peak_new = float(np.max(conv_new))
        
        if peak_new < best_peak * (1 - 1e-14):
            best_peak = peak_new
            best_p = p_new.copy()
        
        p = p_new
        lr *= 0.99995
        
        if verbose and t % 10000 == 0:
            print(f"    Adam it{t}: peak={best_peak:.10f}, obj={2*n*best_peak:.8f}, "
                  f"tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")
    
    return best_p


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def projected_subgradient_step(p, g, lr):
    """Take a projected subgradient step."""
    p_new = p - lr * g
    p_new = project_simplex(p_new)
    return p_new


def hybrid_descent(p_init, time_budget=100.0, verbose=False):
    """
    Hybrid: alternate between MWD and projected subgradient steps.
    """
    n = len(p_init)
    p = normalize_nonneg(p_init)
    best_p = p.copy()
    best_peak = float(np.max(compute_conv(p)))
    start = time.monotonic()
    
    # Phase A: MWD
    remaining = time_budget - (time.monotonic() - start)
    alloc_a = remaining * 0.6
    if alloc_a > 10:
        p = mwd_descent(p, time_budget=alloc_a, verbose=verbose)
    
    # Phase B: Smoothed Adam
    remaining = time_budget - (time.monotonic() - start)
    alloc_b = remaining * 0.4
    if alloc_b > 10:
        p = smoothed_adam_descent(p, time_budget=alloc_b, tau_init=20.0, tau_end=500.0, verbose=verbose)
    
    obj = compute_obj(p)
    return p


def fourier_init(n, rng):
    """Create a structured initialization using Fourier modes."""
    x = np.arange(n)
    a = np.ones(n)
    for k in range(1, 6):
        a += 0.1 * np.cos(2 * np.pi * k * x / n + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
