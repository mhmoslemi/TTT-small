Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses KKT-based water-filling, FFT-accelerated gradient methods, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    if n == 0:

    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    s = abs(s)
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_weighted(a, y):
    """
    Compute gradient of sum_k y_k * conv(a,a)_k with respect to a.
    grad_j = 2 * sum_{i=0}^{n-1} a_i * y_{i+j}
    Uses FFT for O(n log n) computation.

    n = len(a)
    m = 2 * n - 1
    N = 1
    while N < n + m:

    A_pad = fft(a, N)
    Y_pad = fft(y, N)
    cross = ifft(Y_pad * np.conj(A_pad)).real[:n]
    return 2.0 * cross


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s > 0:
        w = w / s
    else:
        w = np.ones(n) / n
    return w


def softmax_weights(conv, tau):
    """Compute softmax weights from convolution values."""
    max_c = np.max(conv)
    exp_vals = np.exp(tau * (conv - max_c))
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / s
    return p

        obj_curr = best_obj
        a_try = a + alpha * direction
        a_try = project_simplex(a_try)
        obj_try = compute_obj(a_try)

        backtracks = 0
        while obj_try >= obj_curr - 1e-14 and backtracks < 10:
            alpha *= 0.7
            a_try = a + alpha * direction
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            backtracks += 1

        if obj_try < best_obj - 1e-14:
            a = a_try
            best_obj = obj_try
            best_a = a.copy()
            stall = 0
            # Adaptively increase alpha if no backtracks
            if backtracks == 0:
                alpha = min(alpha * 1.05, 1.0)
            # Adaptively increase tau
            tau = min(tau * 1.02, tau_max)
        else:
            stall += 1
            # Reduce tau if stuck (restore exploration)
            if stall > 10:
                tau = max(tau * 0.5, tau_init)
                alpha = alpha * 0.5
            if stall > 50:
                # Random perturbation to escape
                pert = rng.standard_normal(n) * 0.01
                a = a + pert
                a = project_simplex(a)
                tau = tau_init
                alpha = 0.05 / np.sqrt(n)
                stall = 0

        it += 1
        if verbose and (it % 500 == 0 or it == 1):
            elapsed = time.monotonic() - start
            print(f"    KKT it{it}: obj={best_obj:.8f}, tau={tau:.1f}, "
                  f"alpha={alpha:.4f}, stall={stall}, time={elapsed:.0f}s")

        # Early stopping if gradient is very uniform
        if it > 100:
            grad_var = float(np.std(grad[support]) / (abs(lam) + 1e-10))
            if grad_var < 0.001 and it > 500:
                if verbose:
                    print(f"    KKT: converged at it{it}, grad_var={grad_var:.2e}")
                break

    return best_a


def build_conv_matrix_fast(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_fast(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_refine(a, time_budget=100.0, max_iters=50, verbose=False):
    """LP-driven refinement with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        g, M_lp = solve_lp_minpeak(a)
        if g is None:
            break
        it += 1

        best_t = 0.0
        best_obj_t = best_obj
        ts = np.linspace(0.01, 0.95, 20)
        for t in ts:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj_t:
                    best_obj_t = obj_try
                    best_t = t

        if best_t > 0 and best_obj_t < best_obj - 1e-13:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_obj = best_obj_t
            best_a = a.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            stall += 1
            if stall >= 8:
                break

    return best_a


def gd_polish(a, time_budget=60.0, tau_init=100.0, tau_end=3000.0, verbose=False):
    """Adaptive-tau gradient descent polishing with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0
    tau = tau_init

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.2

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        grad = gradient_weighted(y, softmax_weights(compute_conv(y), tau))
        g_norm = float(np.linalg.norm(grad))

        lr = 2.0 / (g_norm * np.sqrt(n) + 1e-10)
        lr = min(lr, 50.0 / n)

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it

    if verbose:
        print(f"    GD polish: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use scipy's interpolation for better quality
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(x_old, a, bc_type='natural')
        a_new = cs(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 7):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: KKT water-filling + LP refinement + multi-n exploration + GD polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 12.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: KKT Water-Filling (main workhorse) ===
    print(f"\n=== Phase 1: KKT Water-Filling ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 480.0)
    if p1_budget > 30:
        a_kkt = kkt_waterfilling(best_seq, time_budget=p1_budget,
                                  tau_init=20.0, tau_max=5000.0, verbose=True)
        obj_kkt = compute_obj(a_kkt)
        if obj_kkt < best_score:
            best_score = obj_kkt
            best_seq = a_kkt.copy()
            print(f"  Phase1 KKT: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 KKT: obj={obj_kkt:.8f} (no improvement)")

    # === Phase 2: LP Refinement ===
    print(f"\n=== Phase 2: LP Refinement ===")
    remaining = deadline - time.monotonic()

    if p2_budget > 15:
        a_lp = lp_refine(best_seq, time_budget=p2_budget, max_iters=40, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:

            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Second KKT pass with different parameters ===
    print(f"\n=== Phase 3: KKT Round 2 ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 15:
        a_kkt2 = kkt_waterfilling(best_seq, time_budget=p3_budget,
                                   tau_init=50.0, tau_max=8000.0, verbose=True, seed=999)
        obj_kkt2 = compute_obj(a_kkt2)
        if obj_kkt2 < best_score:
            best_score = obj_kkt2
            best_seq = a_kkt2.copy()
            print(f"  Phase3 KKT2: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 KKT2: obj={obj_kkt2:.8f} (no improvement)")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 100, n_best - 50,
        n_best + 50, n_best + 100, n_best + 200,
        n_best + 400, n_best + 600,
        500, 600, 700, 900, 1000, 1200, 1500, 2000
    ]))
    # Filter: keep reasonable sizes, prefer ones different from current
    n_options = [n for n in n_options if 300 <= n <= 2500 and abs(n - n_best) > 40]
    # Sort by closeness to current (try nearby first)
    n_options.sort(key=lambda x: abs(x - n_best))
    n_options = n_options[:7]
    print(f"  Trying n values: {n_options}")

        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()
        obj_init = compute_obj(a_try)
        print(f"  n={n_try}: init obj={obj_init:.8f}")

        # KKT optimization

    remaining = deadline - time.monotonic()
    if remaining > 50:
        a_r = (best_seq + best_seq[::-1]) / 2.0
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        alloc = min(remaining * 0.25, 80.0)
        if alloc > 20:
