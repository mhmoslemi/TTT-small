from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_tr_sscp(a, r):
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)
    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta
    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
    ]
    if r is not None:
        constraints.append(cp.norm(delta, 2) <= r)
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None

def solve_lp_direction(a):
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

"""
Hybrid optimizer: Spectral Equalization + Trust-Region SOCP + LP + Smoothed GD.
Combines frequency-domain flattening (spectral equalization) with time-domain
trust-region SOCP and LP direction finding to minimize peak linear autocorrelation
of non-negative sequences. Key innovations: spectral-domain initialization,
adaptive alpha scheduling, multi-length exploration with resampling, and
aggressive diversification via modulated initializations.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def spectral_equalize(a, time_budget=60.0, alpha_init=0.8, alpha_end=0.35, seed=0):
    """
    Iterative spectral equalization: flatten the power spectrum to reduce
    autocorrelation peak. Uses P^alpha compression with adaptive alpha.
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        progress = min(1.0, it / 300)
        alpha = alpha_init + (alpha_end - alpha_init) * (progress ** 0.6)
        if alpha < 0.15:
            break

        A = fft(a, L)
        P = np.abs(A) ** 2
        P_min = np.min(P)
        P_shifted = np.maximum(P - P_min + 1e-14, 1e-14)
        P_new = P_shifted ** alpha
        # Renormalize to preserve DC
        dc_ratio = P_new[0] / P[0] if P[0] > 1e-15 else 1.0
        P_new = P_new / dc_ratio
        # Preserve phase
        phase = np.exp(1j * np.angle(A))
        A_new = np.sqrt(P_new) * phase
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            alpha = min(alpha * 1.15, 0.95)
            it += 5
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a


def solve_tr_sscp(a, r):
    """
    Solve trust region SOCP:
    min M s.t. C_old[k] + 2*(a*delta)[k] <= M, ||delta||_2 <= r,
               sum(a+delta)=1, a+delta >= 0
    """
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)
    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta
    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
    ]
    if r is not None:
        constraints.append(cp.norm(delta, 2) <= r)
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_direction(a, M_target=None):
    """
    Solve LP: min M s.t. A @ g <= M, sum(g)=1, g>=0.
    Finds the best g that minimizes peak of conv(a, g).
    """
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

def resample(a, new_n):

def project_simplex(v):
    n = len(v)
    if n == 0:

    u = np.sort(v)[::-1]

    rho = rho[-1]

    return np.maximum(v - theta, 0.0)

        else:
            nd = np.linalg.norm(delta_dir)
            if nd > r:
                delta_dir = delta_dir * (r / nd)
        delta_dir = np.where(np.isfinite(delta_dir), delta_dir, 0.0)
        best_alpha, best_alpha_obj = 0, best_obj

            at = a + ta * delta_dir
            at = np.maximum(at, 0)
            st = at.sum()
            if st < 1e-10:
                continue
            at = at / st
            ot = compute_obj(at)
            if ot < best_alpha_obj - 1e-15:
                best_alpha_obj, best_alpha = ot, ta
        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-15:
            a = np.maximum(a + best_alpha * delta_dir, 0)
            sn = a.sum()
            a = a / sn if sn > 1e-10 else a
            best_obj = best_alpha_obj
            best_a = a.copy()

        for tv in [0.03, 0.08, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0, 1.5]:
            at = (1 - tv) * best_a + tv * g_lp
            at = np.maximum(at, 0)
            st = at.sum()
            if st < 1e-10:
                continue
            at = at / st
            ot = compute_obj(at)
            if ot < best_t_obj - 1e-15:
                best_t_obj, best_t = ot, tv
        if best_t > 0 and best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = np.maximum((1 - best_t) * best_a + best_t * g_lp, 0)
            sb = best_a.sum()

                    a = a / sa

        it += 1
        if verbose and it % 8 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
    return best_a


def smoothed_gd(a, time_budget=70.0, tau_init=20.0, tau_end=500.0):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 4.0 / np.sqrt(n)
    mom = 0.82
    v = np.zeros(n)
    N = 1
    while N < 3 * n:
        N <<= 1
    it = 0
    patience = 0
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.65)
        conv = compute_conv(a)
        mc = float(np.max(conv))
        ev = np.exp(tau * (conv - mc))
        p = ev / ev.sum()
        P = fft(p, N)
        A = fft(a, N)
        g = 2.0 * ifft(P * np.conj(A)).real[:n]
        v = mom * v + g
        an = project_simplex(a - lr * v)
        on = compute_obj(an)
        if on < best_obj - 1e-14:
            best_obj, best_a, patience = on, an.copy(), 0
        else:
            patience += 1
            if patience > 200:
                lr *= 0.7
                v *= 0.5
                patience = 0
                if lr < 1e-12:
                    break
        a = an
        lr = max(1e-10, lr * 0.9994)
        it += 1
    return best_a


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
