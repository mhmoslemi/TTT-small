"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of LP-based alternating minimization, pairwise coordinate descent
for fine-grained polish, and aggressive smoothed gradient descent.
Key innovation: pairwise coordinate descent exploits the bilinear structure of the
convolution to make very precise local adjustments that gradient methods miss.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of (a*a)_k using FFT.
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if np.all(v <= 0):
        return np.full(n, target_sum / n)
    
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def nesterov_gd_polish(a, time_budget, tau_init=30.0, tau_max=3000.0):
    """
    Nesterov accelerated gradient descent on the smoothed objective.
    Projects onto the simplex {x >= 0, sum(x) = 1} after each step.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 1.5 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 1:
            break

        # Nesterov lookahead
        y = a + beta * v
        y = project_simplex(y, 1.0)

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        # Adaptive learning rate
        if g_norm > 1e-15:
            adaptive_lr = 0.5 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.05 * base_lr), 5.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adaptive_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = project_simplex(a_new, 1.0)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr *= 0.9999
        tau = min(tau * 1.015, tau_max)
        it += 1

        # Reset if stuck
        if no_improve > 2000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            no_improve = 0
            tau = max(tau_init, tau * 0.5)

    return best_a

    """
    For pair (i,j), find optimal t = a[i] (with a[j] = c - t) 
    that minimizes max_k (a*a)[k].
    Returns new (t, c-t) values.
    """
    n = len(a)
    c = a[i] + a[j]
    if c < 1e-15:
        return a[i], a[j]

    # Bisection on the subgradient
    lo = 0.0
    hi = c

    for _ in range(45):
        mid = (lo + hi) / 2.0
        # Compute convolution with a[i]=mid, a[j]=c-mid
        a_tmp = a.copy()
        a_tmp[i] = mid
        a_tmp[j] = c - mid
        conv = compute_conv(a_tmp)
        k_star = int(np.argmax(conv))

        # Compute subgradient of max_conv w.r.t. t at this point
        # d(a*a)[k]/dt = 2*a[k-i] (if valid) - 2*a[k-j] (if valid)
        # Special cases: if k-i==i, contributes 2*mid (derivative of mid^2)
        #                if k-j==j, contributes -2*(c-mid) (derivative of (c-mid)^2)
        #                if k-i==j OR k-j==i, cross term 2*(c-2*mid) or similar
        diag = 0.0

        # Contribution from a[i] = t:
        ki = k_star - i
        if 0 <= ki < n:
            if ki == i:
                diag += 2.0 * mid
            elif ki == j:
                diag += 2.0 * (c - mid)
            else:
                diag += 2.0 * a[ki]

        # Contribution from a[j] = c-t:
        kj = k_star - j
        if 0 <= kj < n:
            if kj == j:
                diag -= 2.0 * (c - mid)
            elif kj == i:
                diag -= 2.0 * mid
            else:
                diag -= 2.0 * a[kj]

        if diag > 1e-14:
            hi = mid
        elif diag < -1e-14:
            lo = mid
        else:
            break

    t_opt = (lo + hi) / 2.0
    return t_opt, c - t_opt


def pairwise_cd_polish(a, time_budget, rng, pairs_per_sweep=300, verbose=False):
    """
    Pairwise coordinate descent: repeatedly pick random pairs and optimize
    the 1D subproblem for that pair.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    sweep = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        sweep_improved = False

        for _ in range(pairs_per_sweep):
            if time.monotonic() - start > time_budget - 0.5:
                break

            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            if abs(i - j) < 3:  # Avoid very close pairs (correlated)
                continue

            t_new, tj_new = pairwise_cd_step(a, i, j)
            if abs(t_new - a[i]) < 1e-12:
                continue

            a[i] = t_new
            a[j] = tj_new
            # Sum is preserved by construction

            obj = compute_obj(a)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                sweep_improved = True
            else:
                # Revert if it didn't help (can happen due to numerical issues)
                a[i] = best_a[i]
                a[j] = best_a[j]

        sweep += 1
        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if verbose and sweep % 50 == 0:
                print(f"    Pairwise CD sweep {sweep}: obj={best_obj:.10f}")

        if no_improve > 200:
            # Perturb to escape
            a = best_a + 0.002 * rng.standard_normal(n) * best_a
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0

        if no_improve > 500:
            break

    if verbose:
        print(f"    Pairwise CD: {sweep} sweeps, best_obj={best_obj:.10f}")
    return best_a


def fourier_perturb(a, rng, strength=0.03):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 8
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_alternate(a, time_budget, seed=42, verbose=True):
    """
    Main LP-based alternating minimization with adaptive blending.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    t_blend = 0.5
    stall_count = 0
    last_obj = best_obj

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_conv_lp(a)
        lp_sol_time = time.monotonic() - t0

        if g_lp is None:
            break

        iter_count += 1

        # Adaptive blend factor candidates
        candidates_t = [t_blend, t_blend * 0.5, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if iter_count > 100:
            candidates_t.extend([0.05, 0.02, 0.01, 0.03])
        if iter_count > 300:
            candidates_t.extend([0.005, 0.002])

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend

        for t in candidates_t:
            t = min(max(t, 0.001), 1.0)
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            t_blend = max(0.005, best_local_t * 0.98)
            stall_count = 0
            if verbose and iter_count % 50 == 0:
                print(f"    LP alt {iter_count}: obj={best_obj:.10f}, t={t_blend:.5f}, "
                      f"M_lp={M_lp:.6f}, time={lp_sol_time:.3f}s")
        else:
            stall_count += 1
            if stall_count >= 5:
                strength = 0.005 + 0.003 * min(stall_count, 15)
                a = fourier_perturb(best_a, rng, strength=strength)
                t_blend = max(0.1, min(t_blend * 1.3, 0.6))
                stall_count = 0
            elif stall_count >= 2:
                a = best_a + 0.003 * rng.standard_normal(n) * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()

        # Convergence check
        if best_obj > last_obj - 1e-12 and iter_count > 50:
            if iter_count % 200 == 0:
                break
        last_obj = best_obj

    if verbose:
        print(f"    LP alternating: {iter_count} iterations, best_obj={best_obj:.10f}")
    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def aggressive_gradient_polish(a, time_budget, tau_start=50.0, tau_end=5000.0):
    """
    Very aggressive gradient polishing with rapidly increasing tau.
    Uses backtracking line search instead of fixed step.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    tau = tau_start
    lr = 1.0 / np.sqrt(n)
    prev_obj = best_obj

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 0.5:
            break

        g = smoothed_gradient(a, tau)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Backtracking line search
        best_trial_obj = best_obj
        best_trial_a = a.copy()
        trial_lr = lr

        for bt in range(10):
            a_try = a - trial_lr * g
            a_try = project_simplex(a_try, 1.0)
            obj_try = compute_obj(a_try)
            if obj_try < best_trial_obj:
                best_trial_obj = obj_try
                best_trial_a = a_try.copy()
            trial_lr *= 0.5

        if best_trial_obj < best_obj - 1e-15:
            best_obj = best_trial_obj
            best_a = best_trial_a.copy()
            a = best_trial_a
            lr = min(lr * 1.1, 5.0 / np.sqrt(n))
        else:
            lr *= 0.7
            a = best_a.copy()

        tau = min(tau * 1.03, tau_end)
        it += 1

        if it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    AgGrad it={it}: obj={best_obj:.10f}, tau={tau:.0f}, lr={lr:.4f}, t={elapsed:.1f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: LP alternating -> pairwise CD -> gradient polish -> final refinement.
    Tries multiple n values and returns the best sequence found.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    # Load initial sequence
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0)
        if a.size == 0 or a.sum() < 0.01:
            raise ValueError("Invalid parent")
        a = a / a.sum()
        n_init = a.size
        print(f"Parent loaded: n={n_init}, obj={compute_obj(a):.10f}")
    except Exception:
        n_init = 800
        a = np.ones(n_init) / n_init
        print(f"Using default init: n={n_init}, obj={compute_obj(a):.10f}")

    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)

    # === Phase 1: LP Alternating (main workhorse, 45% of time) ===
    print(f"\n=== Phase 1: LP Alternating ===")
    p1_budget = min(budget_s * 0.45, deadline - time.monotonic() - 10)
    p1_budget = max(60, p1_budget)
    a_p1 = lp_alternate(best_a, time_budget=p1_budget, seed=seed, verbose=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1.copy()
        print(f"  Phase 1 result: obj={best_obj:.10f} **")
    else:
        print(f"  Phase 1 result: obj={obj_p1:.10f} (no improvement)")

    # === Phase 2: Pairwise Coordinate Descent (25% of time) ===
    print(f"\n=== Phase 2: Pairwise Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        p2_budget = min(remaining * 0.5, budget_s * 0.25)
        p2_budget = max(60, p2_budget)
        print(f"  Allocating {p2_budget:.0f}s for pairwise CD")
        a_p2 = pairwise_cd_polish(best_a, time_budget=p2_budget, rng=rng,
                                   pairs_per_sweep=400, verbose=True)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_obj:
            best_obj = obj_p2
            best_a = a_p2.copy()
            print(f"  Phase 2 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 2 result: obj={obj_p2:.10f} (no improvement)")

    # === Phase 3: Aggressive Gradient Polish (15% of time) ===
    print(f"\n=== Phase 3: Aggressive Gradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = min(remaining * 0.4, budget_s * 0.15)
        p3_budget = max(40, p3_budget)
        print(f"  Allocating {p3_budget:.0f}s for gradient polish")
        a_p3 = aggressive_gradient_polish(best_a, time_budget=p3_budget,
                                           tau_start=50.0, tau_end=5000.0)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3.copy()
            print(f"  Phase 3 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 3 result: obj={obj_p3:.10f} (no improvement)")

    # === Phase 4: Second LP pass (10% of time) ===
    print(f"\n=== Phase 4: Second LP Pass ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p4_budget = min(remaining * 0.4, budget_s * 0.10)
        p4_budget = max(30, p4_budget)
        a_pert = fourier_perturb(best_a, rng, strength=0.008)
        a_p4 = lp_alternate(a_pert, time_budget=p4_budget, seed=seed + 1234, verbose=True)
        obj_p4 = compute_obj(a_p4)
        if obj_p4 < best_obj:
            best_obj = obj_p4
            best_a = a_p4.copy()
            print(f"  Phase 4 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 4 result: obj={obj_p4:.10f} (no improvement)")

    # === Phase 5: Final pairwise CD + gradient (remaining time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        half = remaining * 0.5
        a_final1 = pairwise_cd_polish(best_a, time_budget=half, rng=rng,
                                       pairs_per_sweep=500, verbose=False)
        obj_f1 = compute_obj(a_final1)
        if obj_f1 < best_obj:
            best_obj = obj_f1
            best_a = a_final1.copy()
            print(f"  Phase 5a (CD): obj={best_obj:.10f} **")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_final2 = aggressive_gradient_polish(best_a, time_budget=remaining * 0.7,
                                                   tau_start=100.0, tau_end=8000.0)
            obj_f2 = compute_obj(a_final2)
            if obj_f2 < best_obj:
                best_obj = obj_f2
                best_a = a_final2.copy()
                print(f"  Phase 5b (Grad): obj={best_obj:.10f} **")

    # === Phase 6: Try nearby n values ===
    print(f"\n=== Phase 6: Nearby n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([
        n_best - 20, n_best + 20, n_best - 50, n_best + 50,
        n_best - 100, n_best + 100
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000]

    for n_try in n_options[:3]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_to_length(best_a, n_try)
        a_try = a_try / a_try.sum()

        # Quick LP + CD on new length
        alloc_lp = min(remaining * 0.4, 80.0)
        alloc_cd = min(remaining * 0.3, 60.0)

        if alloc_lp > 20:
            a_try = lp_alternate(a_try, time_budget=alloc_lp, seed=n_try, verbose=False)

        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_try = pairwise_cd_polish(a_try, time_budget=min(alloc_cd, remaining * 0.5),
                                        rng=rng, pairs_per_sweep=300, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: obj={best_obj:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # Final validation
    if best_a is None or best_a.size == 0:
        best_a = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
