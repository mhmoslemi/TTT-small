"""
Gradient-enhanced optimization for step function coefficients.
Combines smoothed log-sum-exp gradient descent (FFT-based analytic gradients),
LP-based direction finding with parabolic line search, parametric shape families,
and multi-scale n exploration. Targets improvement over 1.5063 by finding
better global shapes via frequency-domain initialization and smoother optimization.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_obj_and_grad(a, beta):
    """
    Compute smoothed objective 2n*logsumexp(beta*conv(a,a)) and its gradient.
    Returns (obj_value, gradient_array).
    """
    n = len(a)
    conv = compute_conv(a)
    m = np.max(conv)
    exp_vals = np.exp(beta * (conv - m))
    s_exp = np.sum(exp_vals)
    weights = exp_vals / s_exp
    smooth_val = m + np.log(s_exp) / beta
    obj = 2.0 * n * smooth_val

    # Gradient: 4n * g where g[i] = sum_k weights[k] * a[k-i]
    # Using FFT: g[i] = conv(w_rev, a)[2n-2-i]
    w_rev = weights[::-1]
    size = 1
    while size < (2 * n - 1) + n:
        size <<= 1
    Wr = rfft(w_rev, size)
    A = rfft(a, size)
    conv_ha = irfft(Wr * A, size)
    g = conv_ha[n - 1:2 * n - 1][::-1]
    grad = 4.0 * n * g
    return obj, grad


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def parabolic_line_search(a, direction, t_init=0.3, max_iters=20):
    """
    Find optimal step size t for a + t*direction using parabolic interpolation.
    direction should be normalized so that sum(direction)=1.
    """
    a_norm = a / float(np.sum(a))
    d_norm = direction / float(np.sum(direction))

    def obj_at(t):
        if t < -0.5:
            t = -0.5
        a_try = a_norm + t * (d_norm - a_norm)
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    # Start with a coarse grid to find the basin
    ts = [0.0, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.2]
    vals = [obj_at(t) for t in ts]
    best_i = int(np.argmin(vals))

    # Parabolic refinement around the best point
    lo = max(0.0, ts[max(0, best_i - 1)])
    hi = min(1.5, ts[min(len(ts) - 1, best_i + 1)])
    t_mid = (lo + hi) / 2

    f_lo, f_hi, f_mid = obj_at(lo), obj_at(hi), obj_at(t_mid)

    for _ in range(max_iters):
        if hi - lo < 1e-6:
            break
        # Parabolic interpolation
        denom = (f_lo - 2 * f_mid + f_hi)
        if abs(denom) < 1e-14:
            t_new = (lo + hi) / 2
        else:
            t_new = t_mid - 0.5 * denom * (hi - lo) ** 2 / \
                (2.0 * (f_hi - 2 * f_mid + f_lo) * (hi - t_mid) + \
                 2.0 * (f_lo - 2 * f_mid) * (t_mid - lo))
            if not (lo < t_new < hi):
                t_new = (lo + hi) / 2
        f_new = obj_at(t_new)
        if f_new < min(f_lo, f_hi):
            if t_new < t_mid:
                hi, f_hi = t_new, f_new
            else:
                lo, f_lo = t_new, f_new
        else:
            if (t_new - t_mid) * (t_mid - lo) < 0:
                lo, f_lo = t_new, f_new
            else:
                hi, f_hi = t_new, f_new
        t_mid = (lo + hi) / 2
        f_mid = obj_at(t_mid)

    t_best = min(lo, hi, t_mid, key=lambda t: obj_at(t))
    return t_best, obj_at(t_best)


def smoothed_gradient_descent(a, time_budget, beta_start=15, beta_end=300, verbose=True):
    """
    Gradient descent with smoothed objective and adaptive beta.
    Uses Nesterov momentum and adaptive learning rate.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.08
    momentum = np.zeros(n)
    beta_m = 0.7
    it = 0
    best_it = 0
    lr_min = 0.001

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / (time_budget * 0.95))
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)

        _, grad = smooth_obj_and_grad(a, beta)

        # Nesterov lookahead
        a_look = a + beta_m * momentum
        _, grad_look = smooth_obj_and_grad(a_look, beta)

        momentum = beta_m * momentum - lr * grad_look
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            lr = min(lr * 1.005, 0.15)
        else:
            lr *= 0.98
            if lr < lr_min:
                lr = lr_min

        a = a_new
        it += 1

        if it - best_it > 3000:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = 0.03
            best_it = it

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a


def lp_polish(a, time_budget, max_iters=100, verbose=True):
    """
    LP-based polishing with parabolic line search and adaptive restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(999 + it)
    x_grid = np.linspace(0, 1, n)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        T0 = time.monotonic()
        g_lp, M_lp = solve_lp_minpeak(a)
        T1 = time.monotonic()

        if g_lp is None:
            break

        t_opt, obj_opt = parabolic_line_search(a, g_lp)

        if obj_opt < best_obj - 1e-12:
            a_new = a + t_opt * (g_lp - a)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 3 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"lp_time={T1-T0:.2f}s, elapsed={time.monotonic()-start:.1f}s")

        if stall >= 3:
            if stall <= 5:
                freq = rng.integers(1, max(2, n // 40))
                phase = rng.uniform(0, 2 * np.pi)
                amp = 0.015 * stall
                a = best_a * np.exp(amp * np.sin(2 * np.pi * freq * x_grid + phase))
            elif stall <= 8:
                noise_scale = 0.02 * (stall - 4)
                a = best_a + rng.normal(0, noise_scale, n)
            else:
                # Major perturbation
                idx_swap = rng.choice(n, size=max(2, n // 50), replace=False)
                a = best_a.copy()
                for i in idx_swap:
                    a[i] *= rng.uniform(0.5, 1.5)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 0.01:
                a = a / s
            else:
                a = best_a.copy()
            stall = 0

    return best_a


def parametric_init(n, rng):
    """
    Generate a parametric family of shapes and return the best one.
    Explores frequency-domain parameterizations.
    """
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    candidates = []

    # Cosine-based shapes
    for k in [1, 2, 3]:
        for alpha in [0.3, 0.5, 0.8, 1.2, 1.5, 2.0, 3.0]:
            a = np.exp(-alpha * (1 + np.cos(2 * np.pi * k * (x - 0.5))))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                candidates.append(a / s)

    # Power U-shapes
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = (2 * x - 1) ** (2 * p) + 0.01
        candidates.append(a / a.sum())

    # Gaussian edge peaks
    for sigma in [0.05, 0.08, 0.12, 0.15, 0.2]:
        a = np.exp(-((x - 0.0) / sigma) ** 2) + np.exp(-((x - 1.0) / sigma) ** 2) + 0.001
        candidates.append(a / a.sum())

    # Fourier synthesis: a_i = exp(sum_j c_j cos(2pi j (x_i - 0.5)))
    for _ in range(10):
        coeffs = rng.normal(0, 1.0, 4)
        a = np.ones(n)
        for j in range(1, 5):
            if j - 1 < len(coeffs):
                a *= np.exp(coeffs[j - 1] * np.cos(2 * np.pi * j * (x - 0.5)))
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s > 1e-10:
            candidates.append(a / s)

    # Trimodal
    for c in [0.15, 0.2, 0.25]:
        a = np.exp(-((x - 0.5) / 0.1) ** 2) + \
            np.exp(-((x - c) / 0.08) ** 2) + \
            np.exp(-((x - (1 - c)) / 0.08) ** 2)
        candidates.append(a / a.sum())

    best_a = None
    best_obj = np.inf
    for a in candidates:
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def diverse_init(n, rng, mode=0):

                print(f"  Parametric n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  Parametric n={n_try}: obj={obj_param:.8f}")

    # === Phase 1: Smoothed gradient descent ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print("\n=== Phase 1: Smoothed Gradient Descent ===")
        alloc = min(80.0, remaining * 0.12)
        a_sgd = smoothed_gradient_descent(best_seq, time_budget=alloc,
                                          beta_start=10, beta_end=200, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_sgd:.8f} (no improvement)")

    # === Phase 2: LP Polishing (main) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 2: LP Polishing ===")
        alloc = min(remaining * 0.3, 350.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, max_iters=120, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Gradient refinement after LP ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 3: Gradient Refinement ===")
        alloc = min(40.0, remaining * 0.1)
        a_ref = smoothed_gradient_descent(best_seq, time_budget=alloc,
                                          beta_start=50, beta_end=500, verbose=False)
        obj_ref = compute_obj(a_ref)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj_ref:.8f}")

    # === Phase 4: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 4: n Exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 150, n_best - 80, n_best + 80, n_best + 150,
            n_best + 300, n_best + 500,
            500, 600, 700, 900, 1000, 1200, 1500
        ]))
        n_options = [n for n in n_options if 300 <= n <= 2500 and abs(n - n_best) >= 50]
        n_options = n_options[:6]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            alloc_n = min(remaining * 0.35, 150.0)
            a_try, obj_try = try_different_n(best_seq, n_try, alloc_n, rng)
            if a_try is not None:
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    print(f"  ** n={n_try}: obj={best_score:.8f} **")
                else:
                    print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Multi-start with diverse inits ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 5: Diverse Restarts ===")
        n_cur = len(best_seq)

        for r in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            mode = r % 8
            a_div = diverse_init(n_cur, rng, mode=mode)
            blend = 0.3 + 0.05 * r
            a_r = (1 - blend) * a_div + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            # Quick gradient
            t_g = min(10.0, remaining * 0.15)
            if t_g > 3:
                a_r = smoothed_gradient_descent(a_r, time_budget=t_g,
                                                beta_start=20, beta_end=80, verbose=False)

            # LP polish
            t_lp = min(remaining * 0.25, 60.0)
            if t_lp > 8:
                a_r = lp_polish(a_r, time_budget=t_lp, max_iters=25, verbose=False)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive LP ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 6: Final Polish ===")
        alloc = min(remaining - 10, 200.0)
        if alloc > 20:
            a_final = lp_polish(best_seq, time_budget=alloc, max_iters=80, verbose=True)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final: obj={best_score:.8f} **")
            else:
                print(f"  Final: obj={obj_final:.8f}")

    # === Phase 7: Extra gradient pass ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining - 5, 40.0)
        if alloc > 5:
            a_extra = smoothed_gradient_descent(best_seq, time_budget=alloc,
                                                beta_start=100, beta_end=1000, verbose=False)
            obj_extra = compute_obj(a_extra)
            if obj_extra < best_score:
                best_score = obj_extra
                best_seq = a_extra.copy()
                print(f"  Extra grad: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
