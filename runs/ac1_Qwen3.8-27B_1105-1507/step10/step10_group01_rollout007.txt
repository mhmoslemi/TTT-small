import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import csr_matrix
import time

def compute_conv(a):
    """Fast convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj_fast(a):
    """Compute objective without allocating new array for normalized a."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    # max(conv(a/s, a/s)) = max(conv(a,a))/s^2
    # So obj = 2*n*max(conv(a,a))/s^2
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv) / (s * s)

def solve_lp(a, warm_start=None):
    """Solve LP: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build sparse convolution matrix

    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a_norm[k - i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    best_g = None
    best_M = None
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS]:
        try:

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = np.sum(g_val)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-driven optimization with dual guidance, Nesterov accelerated gradient 
descent with adaptive smoothing, multi-scale exploration, and coordinate descent.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def smooth_grad(a, tau):
    """Gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    s_exp = float(np.sum(exp_vals))
    if s_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / s_exp
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad = 2.0 * ifft(P * np.conj(A)).real[:n]
    return grad


def build_conv_matrix(a):
    """Dense matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_end = min(i + n - 1, 2 * n - 2)
        A[i:k_end + 1, i] = a[:k_end - i + 1]
    return A


def solve_lp(a, verbose=False):
    """LP: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    if verbose:
                        print(f"      LP solved with {solver}, M={M.value:.8f}")
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, direction, n_evals=30):
    """Find best t in [0,1] minimizing obj((1-t)*a + t*direction)."""
    n = len(a)
    best_t = 0.0
    best_obj = compute_obj(a)
    
    # Coarse grid
    ts = np.logspace(-3, -0.05, n_evals)  # from 0.001 to ~0.89
    ts = np.clip(ts, 0.001, 0.999)
    objs = np.full(n_evals, np.inf)
    
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try)
    
    best_idx = int(np.argmin(objs))
    if objs[best_idx] < best_obj:
        best_t = ts[best_idx]
        best_obj = objs[best_idx]
    
    # Parabolic refinement
    if 0 < best_idx < n_evals - 1 and best_t > 0:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.001, 0.999))
            a_par = (1 - t_par) * a + t_par * direction
            a_par = np.maximum(a_par, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                f_par = compute_obj(a_par)
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par
    
    return best_t, best_obj

        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.2
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g = smooth_grad(y, tau)
        g_norm = float(np.linalg.norm(g))
        
        # Adaptive learning rate
        if g_norm > 1e-15:
            adapt_lr = 1.0 / (g_norm + 1e-10)
            adapt_lr = min(max(adapt_lr, 0.3 * base_lr), 3.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adapt_lr
        
        # Barzilai-Borwein every few iterations
        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                bb_lr = min(max(bb_lr, 1e-8), 5.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb_lr
