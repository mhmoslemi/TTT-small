"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP fixed-point iteration with Anderson acceleration, geometric blending,
multi-n exploration, and high-tau smoothed gradient descent for final polish.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Returns (gradient of length n, exact_max of conv).
    """
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def lp_solve(a, solver_list=None):
    """
    Solve: min M s.t. (a*g)[k] <= M for all k, g >= 0, sum(g) = 1.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    if n > 3000:
        return None, None

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue

    return None, None


def anderson_extrapolate(history_X, history_R, m=5):
    """
    Anderson acceleration: given history of iterates X and residuals R,
    compute extrapolated point minimizing ||sum w_k R_k|| subject to sum w = 1.
    history_X: list of arrays (each of length n)
    history_R: list of arrays (residuals, each of length n)
    Returns the extrapolated point.
    """
    k = len(history_X)
    if k < 2:
        return history_X[-1].copy()
    
    k = min(k, m)
    X = np.array(history_X[-k:])  # k x n
    R = np.array(history_R[-k:])  # k x n
    
    # Solve min ||R^T w||^2 s.t. sum(w) = 1
    # Using KKT: [2R R^T, 1; 1^T, 0] [w; lambda] = [0; 1]
    # Simplified: use least squares with constraint
    # Minimize ||R^T w||^2 s.t. 1^T w = 1
    # KKT system:
    # 2 R R^T w + 1 * lambda = 0
    # 1^T w = 1
    
    RR_T = R @ R.T  # k x k
    ones = np.ones(k)
    
    # Build KKT matrix
    KT = np.zeros((k + 1, k + 1))
    KT[:k, :k] = 2 * RR_T
    KT[:k, k] = 1.0
    KT[k, :k] = 1.0
    KT[k, k] = 0.0
    
    rhs = np.zeros(k + 1)
    rhs[k] = 1.0
    
    try:
        sol = np.linalg.solve(KT + 1e-12 * np.eye(k + 1), rhs)
        w = sol[:k]
    except np.linalg.LinAlgError:
        w = np.ones(k) / k
    
    # Don't clamp w to be non-negative (Anderson doesn't require it)
    # But ensure numerical stability
    w = np.clip(w, -10, 10)
    w = w / w.sum()
    
    return X.T @ w

def staircase_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9, lr_scale=2.0):
    """
    Nesterov accelerated gradient descent with staircase tau schedule.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [3, 8, 20, 50, 120, 300, 700, 1500, 3000]

    it = 0
    best_it = 0
    stall_patience = 5000

    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0
        max_iters_per_level = 5000
        local_best = best_obj

        while time.monotonic() - start < time_budget:
            if tau_iters >= max_iters_per_level:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it

            a = a_new
            it += 1
            tau_iters += 1
            lr = max(1e-7, lr * 0.99998)

            if it - best_it > stall_patience:
                a = best_a.copy()
                v = np.zeros(n)
                lr = lr_scale / np.sqrt(n)

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_init(n, mode, rng=None):

        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.0001))
        return a / a.sum()
    elif mode == 3:
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 4:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, np.pi, n)
        a = np.maximum(0.5 + 0.5 * np.cos(x), 0.001)
        return a / a.sum()
    elif mode == 6:
        # Ramp with smooth edges
        x = np.linspace(0, 1, n)
        a = x * (1 - x) * 4  # parabola
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 7:
        # Arithmetic progression (ramp)
        a = np.arange(1, n + 1, dtype=np.float64)
        return a / a.sum()
    elif mode == 8:
        # Bump with oscillation
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2) * (1 + 0.3 * np.sin(20 * np.pi * x))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n

    n_candidates = [n for n in n_candidates if 500 <= n <= 2000]
    # Sort by proximity to current
    n_candidates.sort(key=lambda x: abs(x - n_current))
    n_candidates = n_candidates[:8]
    print(f"  n candidates: {n_candidates}")
    
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == n_current:
            continue
        
        a_try = resample(best_seq, n_try)
        
        # Quick LP (just a few iterations)
        alloc = min(remaining * 0.08, 40.0)
        if alloc > 10:
            a_try = lp_polish_anderson(a_try, time_budget=alloc,
                                        verbose=False, seed=n_try, max_iters=20)
        
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 2: Intensive LP polishing with Anderson at best n ===
    print(f"\n=== Phase 2: Intensive LP+Anderson (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.3, 400.0)
    if p2_budget > 50:
        # Long LP with Anderson acceleration
        a2 = lp_polish_anderson(best_seq, time_budget=p2_budget,
                                 verbose=True, seed=42, max_iters=300,
                                 use_geometric=True, anderson_m=6,
                                 base_t=0.3)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  LP-And: obj={best_score:.10f} **")
        else:
            print(f"  LP-And: obj={obj2:.10f}")

    # === Phase 3: GD refinement ===
    print(f"\n=== Phase 3: Smoothed GD Refinement ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 30:
        a3 = staircase_gd(best_seq, time_budget=p3_budget,
                           tau_levels=[10, 50, 200, 600, 1500, 3000],
                           verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  GD: obj={best_score:.10f} **")
        else:
            print(f"  GD: obj={obj3:.10f}")

    # === Phase 4: LP + GD alternation ===
    print(f"\n=== Phase 4: LP+GD Alternation ===")
    remaining = deadline - time.monotonic()
    
    for rnd in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        # LP step
        alloc_lp = min(remaining * 0.25, 100.0)
        if alloc_lp > 20:
            a4 = lp_polish_anderson(best_seq, time_budget=alloc_lp,
                                     verbose=False, seed=100 + rnd * 7,
                                     max_iters=50, use_geometric=(rnd % 2 == 0),
                                     anderson_m=5)
            obj4 = compute_obj(a4)
            if obj4 < best_score:
                best_score = obj4
                best_seq = a4.copy()
                print(f"  Alt{rnd} LP: obj={best_score:.10f} **")
        
        # GD step
        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.15, 60.0)
        if alloc_gd > 15:
            a5 = staircase_gd(best_seq, time_budget=alloc_gd,
                               tau_levels=[100, 400, 1000],
                               verbose=False)
            obj5 = compute_obj(a5)
            if obj5 < best_score:
                best_score = obj5
                best_seq = a5.copy()
                print(f"  Alt{rnd} GD: obj={best_score:.10f} **")

    # === Phase 5: Try different n after optimization ===
    print(f"\n=== Phase 5: Post-opt Multi-n ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    
    n_post = sorted(set([
        n_current - 50, n_current + 50,
        n_current - 100, n_current + 100,
        n_current + 150
    ]))
    n_post = [n for n in n_post if 500 <= n <= 2000]
    
    for n_try in n_post:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        a_try = resample(best_seq, n_try)
        
        # LP polish
        alloc = min(remaining * 0.2, 80.0)
        if alloc > 20:
            a_try = lp_polish_anderson(a_try, time_budget=alloc,
                                        verbose=False, seed=n_try + 500,
                                        max_iters=30, use_geometric=True)
        
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    # Ultra-high tau GD
    if remaining > 40:
        alloc = min(remaining * 0.2, 80.0)
        a_f1 = staircase_gd(best_seq, time_budget=alloc,
                             tau_levels=[500, 1500, 3000, 5000],
                             verbose=True)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final GD: obj={best_score:.10f} **")

    # LP with very small steps (fine-tuning)
    remaining = deadline - time.monotonic()
    if remaining > 30 and len(best_seq) <= 2000:
        alloc = min(remaining * 0.25, 80.0)
        a_f2 = lp_polish_anderson(best_seq, time_budget=alloc,
                                   verbose=True, seed=9999,
                                   max_iters=100, use_geometric=False,
                                   anderson_m=5, base_t=0.1)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    # One more GD
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.3, 50.0)
        a_f3 = staircase_gd(best_seq, time_budget=alloc,
                             tau_levels=[1000, 3000, 6000],
                             verbose=False)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Ultra GD: obj={best_score:.10f} **")

    # === Phase 7: Diverse initialization attempts ===
    print(f"\n=== Phase 7: Diverse Inits ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    for mode in [6, 7, 8, 2, 3]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        a_init = make_init(n_best, mode, rng)
        alloc = min(remaining * 0.15, 60.0)
        if alloc > 15:
            a_init = staircase_gd(a_init, time_budget=min(alloc * 0.5, 30),
                                   tau_levels=[10, 100, 500], verbose=False)
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20 and n_best <= 1500:
                a_init = lp_polish_anderson(a_init, time_budget=min(alloc * 0.5, remaining2 * 0.5),
                                             verbose=False, seed=mode * 100, max_iters=30)
        
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Mode {mode}: obj={best_score:.10f} **")
        else:
            print(f"  Mode {mode}: obj={obj_init:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
