"""
Optimizes step function coefficients using mirror descent, LP polishing,
and peak-reduction strategies to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.1:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smoothed_subgrad(a, tau):
    """
    Compute smoothed subgradient of max(conv(a,a)) using log-sum-exp.
    Returns (gradient, exact_max).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    
    # Softmax weights
    exp_vals = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # Compute g[i] = 2 * sum_k p[k] * a[k-i] * 1_{0<=k-i<n}
    # Using FFT cross-correlation
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    h = ifft(P * np.conj(A), N).real[:n]
    g = 2.0 * h
    return g, max_c

def mirror_descent(a, tau, lr):
    """One step of mirror descent (multiplicative update)."""
    n = len(a)
    g, _ = smoothed_subgrad(a, tau)
    a_new = a * np.exp(-lr * g)
    a_new = np.maximum(a_new, 1e-30)
    s = float(np.sum(a_new))
    if s < 1e-30:
        return a
    return a_new / s
