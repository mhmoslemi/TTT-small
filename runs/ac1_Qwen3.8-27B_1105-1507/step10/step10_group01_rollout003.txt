def compute_conv_full(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_full(a_norm)
    return 2.0 * n * float(np.max(conv))


def topk_gradient(a, k=20):
    """
    Gradient of top-k average of conv(a,a).
    Returns gradient vector of length n.
    """
    n = len(a)
    conv = compute_conv_full(a)
    # Get top-k indices
    k_use = min(k, len(conv))
    top_idx = np.argsort(conv)[-k_use:]
    top_vals = conv[top_idx]
    max_val = top_vals[-1]
    # Soft weights among top-k
    tau_local = 10.0
    weights = np.exp(tau_local * (top_vals - max_val))
    weights /= weights.sum()
    # p is a distribution over the top-k positions
    p = np.zeros(2 * n - 1)
    p[top_idx] = weights
    # Gradient: grad_j = 2 * sum_k p[k] * a[k-j] (cross-correlation)
    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1
    P = fft(p, N_fft)
    A = fft(a, N_fft)
    # cross-correlation: ifft(conj(P) * A) gives sum_k p[k] * a[k-j]...

            max_cv = float(np.max(conv_y))
            exp_vals = np.exp(tau * (conv_y - max_cv))
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(2 * n - 1)
                p[int(np.argmax(conv_y))] = 1.0
            else:
                p = exp_vals / sum_exp
            N_fft = 1
            while N_fft < 3 * n:
                N_fft <<= 1
            P = fft(p, N_fft)
            Y = fft(y, N_fft)
            g = 2.0 * ifft(np.conj(P) * Y).real[:n]

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Adaptive learning rate
        lr = base_lr / (g_norm ** 0.5 + 1.0)
        lr = min(max(lr, 0.1 * base_lr), 3.0 * base_lr)

        # Backtracking line search
        a_candidate = y - lr * g
        a_candidate = np.maximum(a_candidate, 0)
        s_cand = float(np.sum(a_candidate))
        if s_cand > 1e-10:
            a_candidate = a_candidate / s_cand
            obj_cand = compute_obj(a_candidate)
            backtrack_count = 0
            while obj_cand > compute_obj(y) + 1e-15 and backtrack_count < 5:
                lr *= 0.5
                a_candidate = y - lr * g
                a_candidate = np.maximum(a_candidate, 0)
                s_cand = float(np.sum(a_candidate))
                if s_cand > 1e-10:
                    a_candidate = a_candidate / s_cand
                    obj_cand = compute_obj(a_candidate)
                else:
                    break
                backtrack_count += 1
            if s_cand <= 1e-10:
                a_candidate = y.copy()
                obj_cand = compute_obj(a_candidate)
        else:
            a_candidate = y.copy()
            obj_cand = compute_obj(a_candidate)

        if obj_cand < best_obj - 1e-14:
            best_obj = obj_cand
            best_a = a_candidate.copy()
            best_it = it

        # Update momentum
        new_v = beta * v + (y - a_candidate)
        a = a_candidate

        prev_y = y.copy()
        prev_g = g.copy()
        it += 1

        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it

    if verbose:
        print(f"    GradPolish: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def prolate_init(n, bandwidth=0.5):
    """
    Initialize with a prolate-like sequence (concentrated in time and frequency).
    Uses raised cosine window as approximation.
    """
    x = np.arange(n)
    t = 2.0 * x / (n - 1) - 1.0  # [-1, 1]
    a = 0.5 * (1.0 + np.cos(np.pi * t))  # Raised cosine
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian-like initialization."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = a / a.sum()
    return a


def optimize_n(a, time_budget=120.0):
    """Optimize a sequence of fixed length using LP + gradient."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Phase A: LP optimization
    t_remaining = time_budget
    lp_budget = min(t_remaining * 0.6, 70.0)
    if lp_budget > 10:
        a = lp_optimize_reduced(a, time_budget=lp_budget, verbose=False)

    # Phase B: Gradient polish
    t_remaining = time_budget - (70.0 if lp_budget > 10 else 0)
    gd_budget = min(t_remaining * 0.5, 40.0)
    if gd_budget > 8:
        a = gradient_polish(a, time_budget=gd_budget, tau_init=100.0, tau_end=2000.0)

    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines constraint-reduced LP, top-k gradient,
    multi-n exploration with diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Aggressive LP on current n ===
    print(f"\n=== Phase 1: LP-Reduced on n={n_current} ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.3, 400.0)
    if p1_budget > 30:
        a_lp = lp_optimize_reduced(best_seq, time_budget=p1_budget, verbose=True,
                                   max_constraints=max(30, n_current // 5))
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Gradient Polish ===
    print(f"\n=== Phase 2: Gradient Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 200.0)
    if p2_budget > 20:
        a_gd = gradient_polish(best_seq, time_budget=p2_budget,
                               tau_init=50.0, tau_end=3000.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase2 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 GD: obj={obj_gd:.8f} (no improvement)")

    # === Phase 3: Second LP round (refined) ===
    print(f"\n=== Phase 3: Refined LP ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.25, 300.0)
    if p3_budget > 30:
        a_lp2 = lp_optimize_reduced(best_seq, time_budget=p3_budget, verbose=True,
                                    max_constraints=max(20, n_current // 8))
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase3 LP2: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 LP2: obj={obj_lp2:.8f} (no improvement)")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Try a range of n values with different inits
    n_candidates = sorted(set([
        n_best - 200, n_best - 100, n_best + 100, n_best + 200,
        n_best + 400, n_best + 600,
        500, 600, 700, 900, 1000, 1200, 1500, 1800, 2000
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 2500 and abs(n - n_best) > 50]
    n_candidates = n_candidates[:8]
    print(f"  n candidates: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Try multiple initializations
        inits = [
            resample_sequence(best_seq, n_try),
            prolate_init(n_try, bandwidth=0.4),
            gaussian_init(n_try, sigma_frac=0.25),
        ]

        best_for_n = None
        best_obj_for_n = np.inf

        for init in inits:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            per_init_budget = min(remaining * 0.3, 50.0)
            if per_init_budget < 15:
                break

            a_opt = optimize_n(init, time_budget=per_init_budget)
            obj_opt = compute_obj(a_opt)

            if obj_opt < best_obj_for_n:
                best_obj_for_n = obj_opt
                best_for_n = a_opt

        if best_for_n is not None:
            remaining = deadline - time.monotonic()
            if best_obj_for_n < best_score and remaining > 30:
                # Polish the best one more
                extra_budget = min(remaining * 0.2, 40.0)
                best_for_n = gradient_polish(best_for_n, time_budget=extra_budget,
                                             tau_init=200.0, tau_end=2000.0)
                best_obj_for_n = compute_obj(best_for_n)

            if best_obj_for_n < best_score:
                best_score = best_obj_for_n
                best_seq = best_for_n.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: best obj={best_obj_for_n:.8f}")

    # === Phase 5: Diverse Restarts on best n ===
    print(f"\n=== Phase 5: Diverse Restarts (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    rng2 = np.random.default_rng(seed + 100)

    restarts = []
    # Fourier perturbation
    pert = np.zeros(n_cur)
    for k in range(1, 6):
        freq = k * 2.0 * np.pi / n_cur
        amp = 0.03 * rng2.uniform(0.5, 1.5) / k
        phase = rng2.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n_cur) + phase)
    a_r1 = best_seq * np.exp(pert)
    a_r1 = np.maximum(a_r1, 0)
    a_r1 = a_r1 / a_r1.sum()
    restarts.append(("fourier", a_r1))

    # Smoothing blend
    ks = max(5, n_cur // 20)
    if ks % 2 == 0:
        ks += 1
    a_smooth = uniform_filter1d(best_seq, size=ks)
    a_r2 = 0.7 * best_seq + 0.3 * a_smooth
    a_r2 = np.maximum(a_r2, 0)
    a_r2 = a_r2 / a_r2.sum()
    restarts.append(("smooth", a_r2))

    # Random blend
    a_r3 = 0.6 * best_seq + 0.4 * (rng2.uniform(0, 1, n_cur) / n_cur)
    a_r3 = a_r3 / a_r3.sum()
    restarts.append(("random", a_r3))

    # Mirror
    a_r4 = (best_seq + best_seq[::-1]) / 2.0
    a_r4 = np.maximum(a_r4, 0)
    a_r4 = a_r4 / a_r4.sum()
    restarts.append(("mirror", a_r4))

    for name, a_r in restarts:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_opt = optimize_n(a_r, time_budget=min(remaining * 0.2, 60.0))
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_opt:.8f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        alloc = min(remaining * 0.5, 250.0)
        a_final = lp_optimize_reduced(best_seq, time_budget=alloc, verbose=True,
                                      max_constraints=max(15, len(best_seq) // 10))
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.7, 120.0)
        a_final2 = gradient_polish(best_seq, time_budget=alloc,
                                   tau_init=500.0, tau_end=5000.0, verbose=True)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
