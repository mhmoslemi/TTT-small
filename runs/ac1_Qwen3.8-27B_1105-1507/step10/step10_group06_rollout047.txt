    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        idx = np.arange(j, j + n)
        A[idx, j] = a
    return A


def solve_conv_lp(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of (a*a)_k using FFT.
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = conv - max_conv
    # Clip for numerical stability
    shifted = np.clip(shifted, -50.0 / tau, 0.0)
    exp_vals = np.exp(tau * shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def golden_section_line_search(a, direction, obj_fn, tol=1e-8, max_iter=30):
    """
    Golden section search along a + t*direction for t in [0, 1].
    Returns best t found.
    """
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    fc = obj_fn(a + c * direction)
    fd = obj_fn(a + d * direction)
    fa = obj_fn(a)

    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - phi * (hi - lo)
            fc = obj_fn(a + c * direction)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + phi * (hi - lo)
            fd = obj_fn(a + d * direction)

    # Evaluate candidates
    candidates = [0.0, 1.0, c, d, (lo + hi) / 2.0]
    best_t = 0.0
    best_val = fa
    for t in candidates:
        t = max(0.0, min(1.0, t))
        val = obj_fn(a + t * direction)
        if val < best_val:
            best_val = val
            best_t = t
    return best_t


def gradient_descent_step(a, tau, lr_max=0.5):
    """Single gradient descent step with projection onto probability simplex."""
    n = len(a)
    g = smoothed_gradient(a, tau)
    g_norm = float(np.linalg.norm(g))
    if g_norm < 1e-15:
        return a.copy()
    # Adaptive step size
    lr = min(lr_max, 0.5 / (g_norm + 1e-10))
    a_new = a - lr * g
    # Project onto simplex {x >= 0, sum(x) = 1}
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def nesterov_polish(a, time_budget, tau_init=15.0, tau_max=3000.0, verbose=False):
    """
    Nesterov accelerated gradient descent on smoothed objective.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    base_lr = 1.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adaptive_lr = 0.5 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.05 * base_lr), 2.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adaptive_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        lr *= 0.9995
        tau = min(tau * 1.03, tau_max)
        it += 1

        if it % 100 == 0 and verbose:
            print(f"    GD iter {it}: obj={compute_obj(a):.10f}, best={best_obj:.10f}, tau={tau:.0f}")

        if stall > 200:
            a = best_a.copy()
            v = np.zeros(n)
            stall = 0
            lr = base_lr * 0.5
            tau = max(10.0, tau * 0.5)

    return best_a


def fourier_perturb(a, rng, strength=0.03):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 8
    for k in range(1, n_modes + 1):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_alternate(a, time_budget, seed=42, verbose=True, use_line_search=True):
    """
    LP-based alternating minimization with adaptive blending.
    Uses golden-section line search when enabled.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    t_blend = 0.4
    stall_count = 0
    stagnation = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_conv_lp(a)
        lp_time = time.monotonic() - t0

        if g_lp is None:
            # Fall back to gradient step
            a = gradient_descent_step(a, tau=50.0)
            continue

        iter_count += 1

        if use_line_search:
            # Golden section line search along a -> g_lp
            direction = g_lp - a

            def obj_at_t(t):
                a_try = a + t * direction
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    return np.inf
                a_try = a_try / s_try
                return compute_obj(a_try)

            best_t = golden_section_line_search(a, direction, obj_at_t, max_iter=20)

            a_new = a + best_t * direction
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
        else:
            # Fixed set of blending factors
            candidates_t = [t_blend, t_blend * 0.5, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
            if iter_count > 100:
                candidates_t.extend([0.05, 0.02])

            best_local_obj = best_obj
            best_local_a = best_a.copy()
            for t in candidates_t:
                t = min(max(t, 0.001), 1.0)
                a_try = (1.0 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_local_a = a_try.copy()

            a_new = best_local_a
            obj_new = best_local_obj

        if obj_new < best_obj - 1e-13:
            improvement = best_obj - obj_new
            best_obj = obj_new
            best_a = a_new.copy()
            a = best_a.copy()
            stall_count = 0
            stagnation = 0
            # Decrease blend factor slowly
            t_blend = max(0.05, t_blend * 0.98)
            if verbose and iter_count % 50 == 0:
                print(f"    LP alt {iter_count}: obj={best_obj:.10f}, t={t_blend:.4f}, "
                      f"improve={improvement:.2e}, lp_t={lp_time:.3f}s")
        else:
            stall_count += 1
            stagnation += 1

            if stall_count >= 5:
                # Perturbation to escape
                strength = 0.01 + 0.003 * min(stall_count, 15)
                if stall_count >= 15:
                    # More aggressive restart
                    a = fourier_perturb(best_a, rng, strength=strength * 2)
                else:
                    a = fourier_perturb(best_a, rng, strength=strength)
                t_blend = max(0.2, min(t_blend * 1.5, 0.8))
                stall_count = 0
            elif stall_count >= 2:
                a = best_a + 0.003 * rng.standard_normal(n) * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()

            # If stuck for a long time, try gradient steps
            if stagnation >= 20:
                for _ in range(5):
                    a = gradient_descent_step(a, tau=30.0)
                obj_grad = compute_obj(a)
                if obj_grad < best_obj:
                    best_obj = obj_grad
                    best_a = a.copy()
                else:
                    a = best_a.copy()
                stagnation = 0

    if verbose:
        print(f"    LP alternating: {iter_count} iterations, best_obj={best_obj:.10f}")

    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_ramp_init(n, direction=1):
    """Create a ramp-shaped initialization."""
    if direction == 1:
        a = np.arange(1, n + 1, dtype=np.float64)
    else:
        a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def make_cosine_init(n):
    """Create a cosine bell initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def make_periodic_init(n, freq=2):
    """Create a periodic initialization."""
    x = np.arange(n)
    a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x / n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def make_bimodal_init(n):
    """Create a bimodal initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-10 * (x - 0.33) ** 2) + np.exp(-10 * (x - 0.67) ** 2)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def make_random_smooth_init(n, rng, smoothness=3):
    """Create a random smooth initialization."""
    a = rng.standard_normal(n) + 1.0
    # Smooth via moving average
    kernel = np.ones(smoothness) / smoothness
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.01)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating minimization,
    gradient polishing, multi-start with diverse inits, and length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    rng = np.random.default_rng(seed)

    # Load parent
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0)
        if a.size == 0 or a.sum() < 0.01:
            raise ValueError("Invalid parent")
        a = a / a.sum()
        n_init = a.size
        print(f"Parent loaded: n={n_init}, obj={compute_obj(a):.10f}")
    except Exception:
        n_init = 800
        a = np.ones(n_init) / n_init
        print(f"Using default init: n={n_init}, obj={compute_obj(a):.10f}")

    best_a = a.copy()
    best_obj = compute_obj(a)

    # === Phase 1: LP Alternating from parent (35% of time) ===
    print(f"\n=== Phase 1: LP Alternating (from parent) ===")
    p1_deadline = start_time + budget_s * 0.35
    p1_budget = max(30, p1_deadline - time.monotonic())
    a_p1 = lp_alternate(best_a, time_budget=p1_budget, seed=seed, verbose=True, use_line_search=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1.copy()
        print(f"  Phase 1 result: obj={best_obj:.10f} **")
    else:
        print(f"  Phase 1 result: obj={obj_p1:.10f}")

    # === Phase 2: Gradient Polish (15% of time) ===
    print(f"\n=== Phase 2: Gradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        p2_budget = min(remaining * 0.2, 200.0)
        a_p2 = nesterov_polish(best_a, time_budget=p2_budget,
                               tau_init=10.0, tau_max=5000.0, verbose=False)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_obj:
            best_obj = obj_p2
            best_a = a_p2.copy()
            print(f"  Phase 2 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 2 result: obj={obj_p2:.10f}")

    # === Phase 3: LP from perturbed start (20% of time) ===
    print(f"\n=== Phase 3: LP from Perturbed Start ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = min(remaining * 0.35, 300.0)
        a_pert = fourier_perturb(best_a, rng, strength=0.015)
        a_p3 = lp_alternate(a_pert, time_budget=p3_budget, seed=seed + 777,
                            verbose=True, use_line_search=True)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3.copy()
            print(f"  Phase 3 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 3 result: obj={obj_p3:.10f}")

    # === Phase 4: Multi-start with diverse initializations (25% of time) ===
    print(f"\n=== Phase 4: Multi-start Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    # Generate diverse initializations at current best length and nearby lengths
    inits = []
    init_names = []

    # Structured inits at best length
    inits.append(make_ramp_init(n_best, 1))
    init_names.append(f"ramp_forward_n{n_best}")
    inits.append(make_ramp_init(n_best, -1))
    init_names.append(f"ramp_backward_n{n_best}")
    inits.append(make_cosine_init(n_best))
    init_names.append(f"cosine_n{n_best}")
    inits.append(make_periodic_init(n_best, freq=3))
    init_names.append(f"periodic3_n{n_best}")
    inits.append(make_bimodal_init(n_best))

    inits.append(make_random_smooth_init(n_best, rng, smoothness=5))
    init_names.append(f"random_smooth_n{n_best}")

    # Different lengths
    for n_try in [n_best - 50, n_best + 50, 600, 1000]:
        if 300 <= n_try <= 2000 and n_try != n_best:

        a_p5 = lp_alternate(best_a, time_budget=p5_budget, seed=seed + 9999,
                            verbose=True, use_line_search=True)
        obj_p5 = compute_obj(a_p5)
        if obj_p5 < best_obj:
            best_obj = obj_p5
            best_a = a_p5.copy()
            print(f"  Phase 5 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 5 result: obj={obj_p5:.10f}")

    # === Phase 6: Ultra-final gradient polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 6: Ultra-final Polish ===")
        alloc = min(remaining * 0.6, 60.0)
        a_final = nesterov_polish(best_a, time_budget=alloc,
                                  tau_init=5.0, tau_max=10000.0, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final: obj={best_obj:.10f} **")
        else:
            print(f"  Final: obj={obj_final:.10f}")

    # Final validation
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:

        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
