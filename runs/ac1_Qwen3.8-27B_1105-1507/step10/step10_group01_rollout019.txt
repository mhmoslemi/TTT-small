"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses exponentiated gradient (mirror descent) with adaptive log-sum-exp smoothing,
combined with structured initialization, multi-n search, and occasional LP polishing.
Key improvements: multiplicative updates for natural non-negativity, faster convergence,
better initialization from window functions, and efficient use of LP for final polish.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smoothed_grad(a, tau):
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]

def exponentiated_gradient_optimize(a, time_budget=200.0, tau_init=100.0, tau_end=5000.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta_base = 1.0 / np.sqrt(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 0.7
        
        g = smoothed_grad(a, tau)
        g_mean = float(np.mean(g))
        g_centered = g - g_mean
        
        # Adaptive step size
        g_norm = float(np.linalg.norm(g_centered))
        if g_norm < 1e-15:
            break
        eta = eta_base / (g_norm + 1e-10)

def build_conv_matrix(a):
    n = len(a)
    a_norm = a / np.sum(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a_norm[diff[valid]]
    return A

def solve_lp(a, max_time=30.0):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    start = time.monotonic()
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        if time.monotonic() - start > max_time:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val
        except Exception:
            continue
    return a

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def make_raised_cosine(n, beta=0.5):
    x = np.linspace(0, 1, n)
    # Raised cosine window
    w = np.ones(n)
    edge = int(beta * n)
    for i in range(edge):
        t = i / edge
        w[i] = 0.5 * (1 + np.cos(np.pi * (2*t - 1)))
        w[n-1-i] = w[i]
    return w / w.sum()

def make_gaussian(n, sigma_frac=0.3):
    x = np.linspace(0, 1, n)
    center = 0.5
    sigma = sigma_frac
    w = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return w / w.sum()

def make_tapered(n, power=2.0):
    """Sequence that's 1 in the middle and tapers to 0 at edges."""

    return w / s

def fourier_perturb(a, rng, strength=0.05):
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FISTA (fast projected gradient with Nesterov acceleration) combined with
exponentiated gradient, structured initialization from window functions,
multi-n exploration, and a single LP polish at the end.
Key improvements over previous: much faster inner loop (no per-iteration LP),
better initialization, adaptive tau scheduling, and momentum-based acceleration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """Gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def proj_simplex(v):
    """Project v onto the non-negative simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def fista_optimize(a, time_budget=200.0, tau_init=50.0, tau_end=10000.0, verbose=False):
    """
    FISTA (Fast Iterative Shrinkage-Thresholding Algorithm) for minimizing
    the smoothed max autocorrelation with simplex constraint.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    a_prev = a.copy()
    lr_base = 2.0 / np.sqrt(n)
    it = 0
    best_it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 0.8

        # Nesterov momentum
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a + (t_k - 1.0) / t_next * (a - a_prev)
        y = proj_simplex(y)

        # Gradient
        g = smoothed_grad(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm < 1e-15:
            break

        # Adaptive step size
        lr = lr_base / (g_norm + 1e-10)
        lr = min(lr, 1.0 / (tau * g_norm + 1e-10))
        lr = max(lr, 1e-8)

        a_new = proj_simplex(y - lr * g)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1

        a_prev = a.copy()
        a = a_new
        t_k = t_next
        it += 1

        # Reset if stuck
        if stall_count > 3000:
            a = best_a.copy()
            a_prev = a.copy()
            t_k = 1.0
            lr_base *= 0.7
            stall_count = 0

        if verbose and it % 2000 == 0:
            print(f"      FISTA it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}")

    if verbose:
        print(f"      FISTA done: obj={best_obj:.8f}, iters={it}")
    return best_a


def expo_grad_optimize(a, time_budget=200.0, tau_init=50.0, tau_end=10000.0, verbose=False):
    """
    Exponentiated gradient (multiplicative weights) with adaptive step size.
    Naturally maintains non-negativity.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta_base = 3.0 / np.sqrt(n)
    it = 0
    best_it = 0
    momentum = 0.0
    g_mom = np.zeros(n)
    beta = 0.9

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 0.7

        g = smoothed_grad(a, tau)
        g_mean = float(np.mean(g))
        g_centered = g - g_mean

        # Momentum
        g_mom = beta * g_mom + (1 - beta) * g_centered

        g_norm = float(np.linalg.norm(g_mom))
        if g_norm < 1e-15:
            break

        eta = eta_base / (g_norm + 1e-10)

        # Exponentiated update
        log_a = np.log(a + 1e-30)
        update = log_a - eta * g_mom
        update -= np.max(update)
        a_new = np.exp(update)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        if it - best_it > 5000:
            a = best_a.copy()
            g_mom = np.zeros(n)
            eta_base *= 0.6

        if verbose and it % 2000 == 0:
            print(f"      EG it{it}: obj={best_obj:.8f}, tau={tau:.0f}, eta={eta:.6f}")

    if verbose:
        print(f"      EG done: obj={best_obj:.8f}, iters={it}")
    return best_a


def build_conv_matrix(a):
    """Build convolution matrix for LP."""
    n = len(a)
    a_norm = a / np.sum(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a_norm[diff[valid]]
    return A


def solve_lp(a, max_time=60.0):
    """Solve LP to find g minimizing max conv(a,g) with sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        if time.monotonic() - start > max_time:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val
        except Exception:
            continue
    return a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_raised_cosine(n, beta=0.4):
    """Raised cosine window normalized to sum=1."""
    x = np.linspace(0, 1, n)
    w = np.ones(n)
    edge = max(1, int(beta * n / 2))
    for i in range(edge):
        t = i / edge
        w[i] = 0.5 * (1 + np.cos(np.pi * (2 * t / beta - 1)))
        w[n - 1 - i] = w[i]
    s = w.sum()
    return w / s


def make_tapered(n, power=2.5):
    """Tapered sequence: 1 in middle, smooth decay to edges."""
    x = np.linspace(0, 1, n)
    d = np.abs(2 * x - 1)
    w = np.maximum(0, 1 - d ** power)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def make_gaussian_window(n, sigma_frac=0.35):
    """Gaussian window normalized to sum=1."""
    x = np.linspace(0, 1, n)
    w = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    return w / w.sum()


def fourier_perturb(a, rng, strength=0.05):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: FISTA + exponentiated gradient + multi-n + LP polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                print(f"Parent loaded: n={parent.size}")
    except Exception:
        pass

    # Generate initial candidates
    print("=== Generating initial candidates ===")
    candidates = []

    if parent is not None:
        for n_try in [750, 850, 1000, 1100]:
            a = resample(parent, n_try)
            a = a / a.sum()
            candidates.append((a.copy(), f"parent_n{n_try}"))

    # Structured windows at various n
    for n_try in [800, 1000, 1200]:
        candidates.append((make_raised_cosine(n_try, 0.35), f"rc_{n_try}"))
        candidates.append((make_tapered(n_try, 3.0), f"tap_{n_try}"))
        candidates.append((make_gaussian_window(n_try, 0.3), f"gauss_{n_try}"))

    # Evaluate
    for a, name in candidates:
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  {name}: obj={obj:.8f} **")
        else:
            print(f"  {name}: obj={obj:.8f}")

    print(f"Best initial: obj={best_score:.8f}, n={len(best_seq)}")

    # Phase 1: FISTA (main workhorse)
    print(f"\n=== Phase 1: FISTA ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a_opt = fista_optimize(best_seq, time_budget=p1_budget, tau_init=50.0, tau_end=5000.0, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Phase 1 FISTA: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 FISTA: obj={obj:.8f}")

    # Phase 2: Exponentiated gradient
    print(f"\n=== Phase 2: Exponentiated Gradient ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 30:
        a_opt = expo_grad_optimize(best_seq, time_budget=p2_budget, tau_init=100.0, tau_end=8000.0, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Phase 2 EG: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 EG: obj={obj:.8f}")

    # Phase 3: Second FISTA round with higher tau
    print(f"\n=== Phase 3: FISTA (high tau) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 180.0)
    if p3_budget > 20:
        a_opt = fista_optimize(best_seq, time_budget=p3_budget, tau_init=500.0, tau_end=20000.0, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Phase 3 FISTA: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 FISTA: obj={obj:.8f}")

    # Phase 4: Multi-n exploration
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(500, n_best - 150), max(500, n_best - 50),
        n_best + 100, n_best + 200, n_best + 350,
        700, 850, 950, 1050, 1200, 1400
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1800 and abs(n - n_best) >= 40]
    n_options = n_options[:5]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample(best_seq, n_try)

        alloc = min(remaining * 0.2, 100.0)
        if alloc > 15:
            a_try = fista_optimize(a_try, time_budget=alloc, tau_init=50.0, tau_end=5000.0, verbose=False)

        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_try = expo_grad_optimize(a_try, time_budget=min(remaining * 0.15, 40.0), tau_init=200.0, tau_end=5000.0, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 5: Restarts
    print(f"\n=== Phase 5: Restarts ===")
    remaining = deadline - time.monotonic()

    restarts = [
        fourier_perturb(best_seq, rng, strength=0.03),
        fourier_perturb(best_seq, rng, strength=0.07),
        (best_seq + best_seq[::-1]) / 2.0,
    ]
    # Smooth blend
    ks = max(5, len(best_seq) // 8)
    if ks % 2 == 0:
        ks += 1
    a_smooth = uniform_filter1d(best_seq, size=ks)
    a_blend = 0.75 * best_seq + 0.25 * a_smooth
    a_blend = np.maximum(a_blend, 0)
    restarts.append(a_blend / a_blend.sum())

    names = ["fourier_3%", "fourier_7%", "mirror", "smooth"]
    for i, (a_r, name) in enumerate(zip(restarts, names)):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_r = a_r / a_r.sum()
        alloc = min(remaining * 0.2, 80.0)
        if alloc > 15:
            a_r = fista_optimize(a_r, time_budget=alloc, tau_init=100.0, tau_end=5000.0, verbose=False)

        remaining = deadline - time.monotonic()
        if remaining > 25:
            a_r = expo_grad_optimize(a_r, time_budget=min(remaining * 0.12, 30.0), tau_init=500.0, tau_end=5000.0, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # Phase 6: Final LP polish
    print(f"\n=== Phase 6: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        lp_budget = min(remaining * 0.35, 80.0)
        a_lp = solve_lp(best_seq, max_time=lp_budget)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

        # Follow-up with FISTA after LP
        remaining = deadline - time.monotonic()
        if remaining > 25:
            a_post = fista_optimize(best_seq, time_budget=min(remaining * 0.4, 60.0), tau_init=1000.0, tau_end=30000.0, verbose=True)
            obj_post = compute_obj(a_post)
            if obj_post < best_score:
                best_score = obj_post
                best_seq = a_post.copy()
                print(f"  Post-LP FISTA: obj={best_score:.8f} **")

    # Phase 7: Final high-tau polish
    print(f"\n=== Phase 7: Final High-Tau Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = fista_optimize(best_seq, time_budget=min(remaining * 0.6, 80.0), tau_init=2000.0, tau_end=50000.0, verbose=True)
        obj = compute_obj(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
            print(f"  Final: obj={best_score:.8f} **")

    # Validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
