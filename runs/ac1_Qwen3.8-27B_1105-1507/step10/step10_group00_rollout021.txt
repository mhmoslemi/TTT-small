Let me also think about whether I can use the structure of the problem more. The optimal solution for minimizing max(c) subject to sum(a)=1, a>=0 is known in the literature as related to "minimum peak" or "flat-top" designs.

One key observation: at the optimum, the constraint is "tight" at multiple points. That is, multiple c_k values equal the maximum. This is the "equal ripple" condition.

Another observation: the Lagrangian of the problem is:
L = max_k c_k - λ(sum(a) - 1) - sum_i μ_i * a_i

At the optimum (KKT conditions):
- For the active max (c_k = M for k in K): the gradient of c_k (averaged over ties) minus λ * 1 - μ = 0
- For inactive elements (a_i = 0): μ_i >= 0
- For active elements (a_i > 0): μ_i = 0

This suggests that at the optimum, for all active i (a_i > 0):
2 * sum_{k in K} (1/|K|) * a_{k-i} = λ

This is a linear system that can be solved! Given the set K of active peaks and the set of active variables, we can solve for a and λ.

This is essentially what the LP is doing, but we can exploit the structure more directly.

   This determines a up to the choice of support.
4. Update the support and active set
5. Iterate

"""
Smooth gradient descent with soft-argin, symmetry enforcement, and LP polishing.
Uses temperature annealing for the soft max approximation.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    c = fft_convolve(a_norm, a_norm)
    return 2.0 * n * float(np.max(c))

def build_hankel(p, n):
    """Build n x n Hankel matrix H where H[i,j] = p[i+j]."""
    idx = np.arange(n)
    return p[idx[:, None] + idx[None, :]]

def smooth_gradient(a, tau):
    """Compute smooth gradient of max(conv(a,a)) using soft-argin."""
    n = len(a)
    c = fft_convolve(a, a)
    tc = tau * c
    tc -= np.max(tc)
    p = np.exp(tc)
    p /= np.sum(p)
    H = build_hankel(p, n)
    return 2.0 * (H @ a)

def symmetric_project(a):
    """Enforce palindromic symmetry and non-negativity."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return np.maximum(a_sym, 0)

def normalize(a):
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = symmetric_project(a)
    return normalize(a)

def smooth_grad_descent(a0, time_budget, tau_start=50, tau_end=500, use_symmetry=True, verbose=False):
    n = len(a0)
    a = normalize(a0.copy())
    if use_symmetry:
        a = normalize(symmetric_project(a))
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    velocity = np.zeros(n)
    beta = 0.7
    lr = 0.05
    step = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        # Temperature schedule
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Compute gradient
        c = fft_convolve(a, a)
        tc = tau * c
        tc -= np.max(tc)
        p = np.exp(tc)
        p /= np.sum(p)
        H = build_hankel(p, n)
        grad = 2.0 * (H @ a)
        
        # Nesterov momentum
        velocity = beta * velocity - lr * grad
        a_new = a + velocity
        
        if use_symmetry:
            a_new = symmetric_project(a_new)
        else:
            a_new = np.maximum(a_new, 0)
        
        a_new = normalize(a_new)

        a = a_new

            no_improve = 0

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Build convolution matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except:
            continue
    return None

    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget and it < 50:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Try several blend factors
        best_t = 0
        best_obj_t = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t
        
        if best_t > 0 and best_obj_t < best_obj - 1e-14:
            a = (1 - best_t) * a + best_t * g_lp
            a = np.maximum(a, 0)
            a = normalize(a)
            best_obj = best_obj_t
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
        
        it += 1
    
    return best_a
