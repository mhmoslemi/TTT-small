import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val, float(M.value)
        except Exception:
            continue
    return None, None


def fast_cd(a, time_budget=300.0, verbose=True, seed=42):

    Optimizes each coordinate via golden section search on the exact 1D subproblem.
    Uses sensitivity-based ordering to prioritize high-impact coordinates.

    best_obj = 2.0 * n * float(np.max(conv))

                else:

    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 5 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        # Find peak convolution index
        k_peak = int(np.argmax(conv))
        S = float(np.sum(a))

        # Generate candidate pairs (i, j) where i + j is near k_peak
        pairs = []
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n and j != i:
                pairs.append((i, j))

        # Also add some random pairs
        for _ in range(min(20, n // 2)):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.append((int(i), int(j)))

        rng.shuffle(pairs)

        for (i, j) in pairs[:50]:  # Limit per sweep
            if time.monotonic() - start >= time_budget:
                break

            v_i = a[i]
            v_j = a[j]
            S_others = S - v_i - v_j

            if S_others < 1e-12:
                continue

            # Precompute the "background" conv values
            # conv[k] without contributions from a[i] and a[j]
            # This is complex; instead, use a simpler approach:
            # Evaluate the 2D objective at grid points

            def f_2d(xi, xj, v_i=v_i, v_j=v_j, S_others=S_others):
                # Compute what conv looks like with a[i]=xi, a[j]=xj
                # Only need to track changes from (v_i, v_j)
                # conv[2i]: changes by xi^2 - v_i^2
                # conv[2j]: changes by xj^2 - v_j^2
                # conv[i+j]: changes by 2*xi*xj - 2*v_i*v_j
                # conv[i+m] for m!=j: changes by 2*(xi-v_i)*a[m]
                # conv[j+m] for m!=i: changes by 2*(xj-v_j)*a[m]

                new_conv_vals = []
                delta_i = xi - v_i
                delta_j = xj - v_j

                # conv[2i]
                if 0 <= 2 * i <= 2 * n - 2:
                    new_conv_vals.append(conv[2 * i] + xi * xi - v_i * v_i)
                # conv[2j]
                if 0 <= 2 * j <= 2 * n - 2:
                    new_conv_vals.append(conv[2 * j] + xj * xj - v_j * v_j)
                # conv[i+j]
                if 0 <= i + j <= 2 * n - 2:
                    new_conv_vals.append(conv[i + j] + 2 * xi * xj - 2 * v_i * v_j)

                # Current max (conservative: include current max)
                current_max = float(np.max(conv))

                # Check linear contributions that could exceed current max
                # For conv[i+m], m != j: change is 2*delta_i*a[m]
                # We need to check if any of these exceed current_max
                # For efficiency, check only the top few
                top_ks = np.argsort(-conv)[:5]
                for k in top_ks:
                    k = int(k)
                    # Does coordinate i contribute to conv[k]?
                    m = k - i
                    if 0 <= m < n and m != j and m != i:
                        new_val = conv[k] + 2 * delta_i * a[m]
                        new_conv_vals.append(new_val)
                    # Does coordinate j contribute to conv[k]?
                    m2 = k - j
                    if 0 <= m2 < n and m2 != i and m2 != j:
                        new_val = conv[k] + 2 * delta_j * a[m2]
                        new_conv_vals.append(new_val)

                if new_conv_vals:
                    max_new = max(current_max, max(new_conv_vals))
                else:
                    max_new = current_max

                denom = (S_others + xi + xj) ** 2
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_new / denom

            # 2D grid search
            scale_i = max(v_i, S / n, 1e-6)
            scale_j = max(v_j, S / n, 1e-6)

            best_xi = v_i
            best_xj = v_j
            best_f = f_2d(v_i, v_j)

            # Coarse grid
            for xi in [0.0, v_i * 0.5, v_i, v_i * 1.5, v_i * 2.0]:
                for xj in [0.0, v_j * 0.5, v_j, v_j * 1.5, v_j * 2.0]:
                    fx = f_2d(xi, xj)
                    if fx < best_f:
                        best_f = fx
                        best_xi = xi
                        best_xj = xj

            # 1D refinement in x direction
            lo = max(0.0, best_xi - scale_i)
            hi = best_xi + scale_i
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = f_2d(c_pt, best_xj)
            fd = f_2d(d_pt, best_xj)
            for _ in range(25):
                if hi - lo < 1e-12:
                    break
                if fc < fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = f_2d(c_pt, best_xj)
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = f_2d(d_pt, best_xj)
            best_xi = (lo + hi) / 2.0

            # Then refine in y direction
            lo = max(0.0, best_xj - scale_j)
            hi = best_xj + scale_j
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = f_2d(best_xi, c_pt)
            fd = f_2d(best_xi, d_pt)
            for _ in range(25):
                if hi - lo < 1e-12:
                    break
                if fc < fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = f_2d(best_xi, c_pt)
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = f_2d(best_xi, d_pt)
            best_xj = (lo + hi) / 2.0

            # Apply update
            if abs(best_xi - v_i) > 1e-12 or abs(best_xj - v_j) > 1e-12:
                a[i] = best_xi
                a[j] = best_xj
                S_new = float(np.sum(a))
                if S_new > 1e-10:
                    conv = compute_conv(a)
                    obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
                    S = S_new

        if verbose and sweep % 3 == 0:
            elapsed = time.monotonic() - start
            print(f"    BlockCD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def lp_polish_enhanced(a, time_budget=60.0, verbose=False, seed=42):
    """Enhanced LP polishing with adaptive blending and multiple iterations."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    it = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        it += 1
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Try a range of blending factors with adaptive selection
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        best_t = 0.0

        t_values = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
                best_t = t

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 4:
                # Perturb to escape
                pert = rng.normal(0, 0.02 * (1 + stall * 0.5), n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s < 1e-10:
                    a = best_a.copy()
                    stall = 0
                    break
                a = a / s
                stall = 0
                if time.monotonic() - start > time_budget * 0.85:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    # Use higher-order interpolation for smoother resampling
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Cubic interpolation using PCHIP-like approach
    from scipy.interpolate import PchipInterpolator
    try:
        interp = PchipInterpolator(x_old, a)
        a_new = interp(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two-peaked
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        # Multi-frequency
        freqs = rng.integers(1, 10, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Hyperbola-like (concentrated in middle)
        x = np.linspace(-1, 1, n)
        a = 1.0 / (1 + 10 * x**2)
        a = a / a.sum()
    elif mode == 7:
        # Cosine raised to power
        x = np.linspace(0, 1, n)
        p = rng.uniform(0.5, 3.0)
        a = np.cos(np.pi * (x - 0.5)) ** (2 * p)
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: combines fast CD with peak-sensitive ordering,
    block 2D optimization, enhanced LP polishing, and adaptive n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = 2.0 * n_default * float(np.max(compute_conv(best_seq)))
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Fast CD with peak-sensitive ordering (45% time) ===
    print(f"\n=== Phase 1: Fast CD (peak-sensitive) ===")
    p1_budget = min((deadline - time.monotonic()) * 0.45, 600.0)
    if p1_budget > 30:
        a_cd = fast_cd(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_cd = 2.0 * a_cd.size * float(np.max(compute_conv(a_cd / a_cd.sum())))
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: Block 2D CD (15% time) ===
    print(f"\n=== Phase 2: Block 2D CD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 180.0)
    if p2_budget > 20:
        a_block = block_cd_2d(best_seq, time_budget=p2_budget, verbose=True, seed=seed + 100)
        obj_block = 2.0 * a_block.size * float(np.max(compute_conv(a_block / a_block.sum())))
        if obj_block < best_score:
            best_score = obj_block
            best_seq = a_block.copy()
            print(f"  BlockCD: obj={best_score:.8f} **")
        else:
            print(f"  BlockCD: obj={obj_block:.8f}")
            # If block CD didn't help, try CD on block CD result anyway
            a_cd2 = fast_cd(a_block, time_budget=min(30, remaining * 0.05),
                            verbose=False, seed=seed + 200)
            obj_cd2 = 2.0 * a_cd2.size * float(np.max(compute_conv(a_cd2 / a_cd2.sum())))
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  CD after Block: obj={best_score:.8f} **")

    # === Phase 3: Enhanced LP Polishing (20% time) ===
    print(f"\n=== Phase 3: Enhanced LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 250.0)
    if p3_budget > 30:
        # Multiple LP rounds with CD in between
        for rnd in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(p3_budget / (5 - rnd), remaining * 0.6)
            if alloc < 15:
                break
            a_lp = lp_polish_enhanced(best_seq, time_budget=alloc, verbose=False,
                                      seed=2000 + rnd * 7)
            obj_lp = 2.0 * a_lp.size * float(np.max(compute_conv(a_lp / a_lp.sum())))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
                # Quick CD after successful LP
                remaining = deadline - time.monotonic()
                if remaining > 20:
                    a_cd3 = fast_cd(best_seq, time_budget=min(30, remaining * 0.1),
                                    verbose=False, seed=3000 + rnd)
                    obj_cd3 = 2.0 * a_cd3.size * float(np.max(compute_conv(a_cd3 / a_cd3.sum())))
                    if obj_cd3 < best_score:
                        best_score = obj_cd3
                        best_seq = a_cd3.copy()
                        print(f"    CD after LP{rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 4: n Exploration (15% time) ===
    print(f"\n=== Phase 4: n Exploration ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 200.0)

    # Explore n values around current best
    n_candidates = sorted(set([
        n_best - 50, n_best - 100, n_best + 50, n_best + 100,
        n_best + 200, n_best + 300,
        600, 700, 900, 1000, 1200
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 2000]
    # Prioritize near current best
    n_candidates.sort(key=lambda x: abs(x - n_best))
    n_candidates = n_candidates[:6]

    print(f"  Trying n values: {n_candidates}")
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(p4_budget / len(n_candidates) * 2, remaining * 0.4)
        if alloc < 15:
            break

        a_try = fast_cd(a_try, time_budget=alloc, verbose=False, seed=n_try)
        obj_try = 2.0 * a_try.size * float(np.max(compute_conv(a_try / a_try.sum())))
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Final CD + LP polish (5% time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_cd = min(remaining * 0.5, 100.0)
        a_final = fast_cd(best_seq, time_budget=alloc_cd, verbose=False, seed=seed + 9999)
        obj_final = 2.0 * a_final.size * float(np.max(compute_conv(a_final / a_final.sum())))
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # Final LP if time remains
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_lp_f = lp_polish_enhanced(best_seq, time_budget=min(remaining * 0.6, 50.0),
                                    verbose=False, seed=7777)
        obj_lp_f = 2.0 * a_lp_f.size * float(np.max(compute_conv(a_lp_f / a_lp_f.sum())))
        if obj_lp_f < best_score:
            best_score = obj_lp_f
            best_seq = a_lp_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_norm(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
