    best_a = a.copy()
    best_obj = compute_obj(a)
    tau = 1.0  # temperature for log-sum-exp
    lr = 0.1
    momentum = 0.7

    while time < budget:
        # Compute p from a
        conv = compute_conv(a)
        p = softmax(tau * (conv - max(conv)))
        
        # Compute gradient of a^T P a
        p_rev = p[::-1]
        # b[j] = sum_l a[l] * p[j+l] = conv(a, p_rev)[2n-2-j]

        A = fft(a, N)

        # But I need conv_ap[2n-2-j] for j=0..n-1, which is indices 2n-2, 2n-3, ..., n-1.

        # That's conv_ap[n-1:2n-1] reversed. Yes.

        # Momentum update
        v = momentum * v + grad
        a_new = project_simplex(a - lr * v)
        
        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            a = a_new
            best_a = a_new.copy()
            best_obj = obj_new
            # Adaptively increase tau
            tau = min(tau * 1.005, 100.0)
        else:
