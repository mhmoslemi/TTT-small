def coordinate_descent(a_init, time_budget, n_sweeps=100, verbose=False):
    """
    Coordinate descent: optimize one coordinate at a time.
    For each coordinate j, the objective as a function of a[j] is:
    2n * max_k (conv_k - a[j]*(a[k-j] + a[j-k]) + a[j]^2 * delta_{2j,k})
    This is a piecewise quadratic, so we can find the optimal a[j] analytically.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    sweep = 0
    while time.monotonic() - start < time_budget and sweep < n_sweeps:
        sweep += 1
        improved = False
        
        # Process coordinates in random order
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            
            # Current convolution
            conv = compute_conv(a)
            max_conv = float(np.max(conv))
            
            # If we change a[j] by delta, conv changes by:
            # conv_k += delta * (a[k-j] + a[j-k]) for k != 2j
            # conv_{2j} += 2*delta*a[j] + delta^2 (wait, let me be more careful)
            # 
            # conv_k = sum_i a[i]*a[k-i]
            # The terms involving a[j] are: a[j]*a[k-j] (when i=j) and a[i]*a[j] (when k-i=j, i.e., i=k-j)
            # So if j != k-j (i.e., 2j != k): conv_k has terms a[j]*a[k-j] + a[k-j]*a[j] = 2*a[j]*a[k-j]
            # If j == k-j (i.e., 2j == k): conv_k has term a[j]*a[j] = a[j]^2
            #
            # So if we increase a[j] by delta:
            # For k where 2j != k and 0 <= k-j < n: conv_k increases by 2*delta*a[k-j]
            # For k = 2j (if 0 <= 2j < 2n-1): conv_{2j} increases by 2*delta*a[j] + delta^2
            #   (because a[j]^2 -> (a[j]+delta)^2 = a[j]^2 + 2*delta*a[j] + delta^2)
            #   Wait, but conv_{2j} also has the cross terms from other i values.
            #   Let me be more careful.
            #
            # conv_k = sum_i a[i]*a[k-i]
            # Terms with a[j]: when i=j: a[j]*a[k-j]; when k-i=j, i.e., i=k-j: a[k-j]*a[j]
            # If j != k-j: both terms present, total contribution = 2*a[j]*a[k-j]
            # If j == k-j (k=2j): only one term a[j]*a[j] = a[j]^2
            #
            # So d(conv_k)/da[j] = 2*a[k-j] if k != 2j and 0<=k-j<n, else a[j] if k=2j
            # (with a[i]=0 for i outside [0,n-1])
            
            # For the purpose of coordinate descent, let's just try a few values of a[j]
            # and pick the best one.
            current_val = a[j]
            
            # Try: 0, current/2, current, current*2, and a few others
            candidates = [0.0, current_val * 0.5, current_val, current_val * 1.5, current_val * 2.0]
            if current_val > 0:
                candidates.append(current_val * 0.25)
                candidates.append(current_val * 0.75)
            
            best_candidate = current_val
            best_candidate_obj = best_obj
            
            for cv in candidates:
                a[j] = cv
                s_tmp = float(np.sum(a))
                if s_tmp < 1e-10:
                    a[j] = current_val
                    continue
                a_norm_tmp = a / s_tmp
                obj_tmp = compute_obj_normalized(a_norm_tmp)
                if obj_tmp < best_candidate_obj:
                    best_candidate_obj = obj_tmp
                    best_candidate = cv
            
            a[j] = best_candidate
            s = float(np.sum(a))
            if s < 1e-10:
                a[j] = current_val
                s = float(np.sum(a))
            a = a / s
            
            if best_candidate != current_val:
                improved = True
                if compute_obj_normalized(a) < best_obj:
                    best_obj = compute_obj_normalized(a)
                    best_a = a.copy()
        
        if not improved:
            break
    
    if verbose:
        print(f"    CoordDescent: obj={best_obj:.8f}, sweeps={sweep}")
    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A

def lp_direction(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns direction g or None.
    """
    n = len(a)
    try:
        import cvxpy as cp
        a_norm = a / np.sum(a)
        A = build_conv_matrix(a_norm)
        
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except Exception:
        pass
    return None

def fourier_perturb(a, rng, strength=0.05):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

def resample(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def diverse_init(n, rng):
    """Create diverse initial sequences of length n."""
    strategies = []
    
    # Cosine bell
    x = np.linspace(0, 1, n)
    strategies.append(np.maximum(np.cos(np.pi * x), 0))
    
    # Gaussian
    x = np.linspace(-3, 3, n)
    strategies.append(np.exp(-x**2))
    
    # Sinc-like
    x = np.linspace(-2, 2, n)
    strategies.append(np.maximum(np.sinc(x / 2), 0))
    
    # Power law
    x = np.linspace(0.01, 1, n)
    strategies.append(x ** rng.uniform(0.5, 3.0))
    
    # Two peaks
    x = np.linspace(0, 1, n)
    c1 = rng.uniform(0.2, 0.4)
    c2 = rng.uniform(0.6, 0.8)
    w = rng.uniform(0.05, 0.15)
    strategies.append(np.maximum(3, 0) * (np.exp(-((x-c1)/w)**2) + np.exp(-((x-c2)/w)**2)))
    
    # Uniform with noise
    strategies.append(1.0 + 0.3 * rng.standard_normal(n))
    
    # Triangle
    x = np.linspace(0, 1, n)
    strategies.append(np.maximum(1 - np.abs(2*x - 1), 0))
    
    for s in strategies:
        s = np.maximum(s, 0)
        sm = float(np.sum(s))
        if sm > 1e-10:
            strategies[-1 if s is strategies[-1] else strategies.index(s)] = s / sm
        # Fix: normalize all
    result = []
    for s in strategies:
        s = np.maximum(s, 0)
        sm = float(np.sum(s))
        if sm > 1e-10:
            result.append(s / sm)
        else:
            result.append(np.ones(n) / n)
    return result

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Multi-n exploration with diverse initializations, Nesterov accelerated
gradient descent on log-sum-exp smoothed max, L-BFGS-B polishing, and coordinate
descent for fine-tuning. Uses higher tau values and tangent-space projection
to achieve precise convergence near the optimum.
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def smoothed_max_gradient(a, tau):
    """
    Compute (smoothed_max_value, gradient) of (1/tau)*log(sum exp(tau*conv_k)).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad = 2.0 * ifft(P * np.conj(A)).real[:n]
    smoothed_val = max_conv + np.log(sum_exp) / tau
    return smoothed_val, grad


def nesterov_gd(a_init, time_budget, tau_start, tau_end, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    v = np.zeros(n)
    it = 0
    best_it = 0
    patience = 30000
    base_lr = 2.0 / np.sqrt(n)
    beta = 0.9

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start * (tau_end / tau_start) ** progress

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        _, g = smoothed_max_gradient(y, tau)
        g_proj = g - np.mean(g)
        g_norm = float(np.linalg.norm(g_proj))

        if g_norm > 1e-15:
            lr = base_lr / (g_norm + 0.05)
            lr = min(lr, 5.0 / np.sqrt(n))
        else:
            break

        v = beta * v - lr * g_proj
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        beta = min(0.995, 0.9 + 0.0095 * it / 10000)

        if it - best_it > patience:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it
            patience = max(2000, patience // 2)
            if patience < 1000:
                break

    if verbose:
        print(f"      Nesterov: obj={best_obj:.8f}, iters={it}")
    return best_a


def lbfgs_objective(x, tau):
    """Objective function for L-BFGS with log-sum-exp smoothing."""
    if np.any(x < -1e-8):
        return 1e10
    x_c = np.maximum(x, 0)
    s = float(np.sum(x_c))
    if s < 1e-10:
        return 1e10
    x_n = x_c / s
    return compute_obj_normalized(x_n)


def lbfgs_gradient(x, tau):
    """Gradient for L-BFGS on the simplex."""
    x_c = np.maximum(x, 0)
    s = float(np.sum(x_c))
    if s < 1e-10:
        n = len(x)
        return np.zeros(n)
    x_n = x_c / s
    _, g = smoothed_max_gradient(x_n, tau)
    return g - np.mean(g)


def lbfgs_polish(a_init, time_budget, tau, maxiter, verbose=False):
    """L-BFGS-B polishing on the simplex."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    try:
        result = minimize(
            lbfgs_objective, a, args=(tau,),
            method='L-BFGS-B',
            jac=lbfgs_gradient,
            bounds=[(0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': maxiter * 4,
                     'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 60}
        )
        if result.x is not None:
            a_r = np.maximum(result.x, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
                obj_r = compute_obj_normalized(a_r)
                if obj_r < best_obj:
                    best_obj = obj_r
                    best_a = a_r.copy()
    except Exception:
        pass

    if verbose:
        print(f"      L-BFGS(tau={tau:.0f}): obj={best_obj:.8f}, time={time.monotonic()-start:.1f}s")
    return best_a


def coordinate_descent(a_init, time_budget, verbose=False):
    """Coordinate descent for precise fine-tuning."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    sweep = 0
    max_sweeps = 200

    while time.monotonic() - start < time_budget and sweep < max_sweeps:
        sweep += 1
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break
            current_val = a[j]
            if current_val < 1e-15:
                candidates = [1e-6, current_val + 0.01 / n]
            else:
                candidates = [0.0, current_val * 0.5, current_val * 1.5,
                              current_val * 0.8, current_val * 1.2,
                              current_val * 0.3, current_val * 2.0]

            best_cv = current_val
            best_co = best_obj

            for cv in candidates:
                a[j] = cv
                s_tmp = float(np.sum(a))
                if s_tmp < 1e-10:
                    a[j] = current_val
                    continue
                a_norm = a / s_tmp
                obj_tmp = compute_obj_normalized(a_norm)
                if obj_tmp < best_co:
                    best_co = obj_tmp
                    best_cv = cv

            a[j] = best_cv
            s = float(np.sum(a))
            if s < 1e-10:
                a[j] = current_val
                s = float(np.sum(a))
            a = a / s

            if best_cv != current_val and best_co < best_obj - 1e-14:
                best_obj = best_co
                best_a = a.copy()
                improved = True

        if not improved:
            break

    if verbose:
        print(f"      CoordDescent: obj={best_obj:.8f}, sweeps={sweep}")
    return best_a


def fourier_perturb(a, rng, strength):
    """Add low-frequency Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def make_diverse_inits(n, rng):
    """Create diverse initial sequences of length n."""
    inits = []

    x = np.linspace(0, 1, n)

    # Cosine bell
    a = np.maximum(np.cos(np.pi * x), 0)
    s = float(np.sum(a))
    if s > 1e-10:
        inits.append(a / s)

    # Gaussian
    xg = np.linspace(-3, 3, n)
    a = np.exp(-xg ** 2)
    s = float(np.sum(a))
    if s > 1e-10:
        inits.append(a / s)

    # Power law
    xp = np.linspace(0.01, 1, n)
    a = xp ** rng.uniform(0.5, 2.0)
    s = float(np.sum(a))
    if s > 1e-10:
        inits.append(a / s)

    # Two peaks
    c1 = rng.uniform(0.25, 0.35)
    c2 = rng.uniform(0.65, 0.75)
    w = rng.uniform(0.08, 0.18)
    a = np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / w) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        inits.append(a / s)

    # Uniform with noise
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        inits.append(a / s)

    # Triangle
    a = np.maximum(1.0 - np.abs(2 * x - 1), 0)
    s = float(np.sum(a))
    if s > 1e-10:
        inits.append(a / s)

    # Wide Gaussian
    xg2 = np.linspace(-1.5, 1.5, n)
    a = np.exp(-xg2 ** 2)
    s = float(np.sum(a))
    if s > 1e-10:
        inits.append(a / s)

    if not inits:
        inits.append(np.ones(n) / n)
    return inits


def optimize_pipeline(a_init, time_budget, verbose=False):
    """Full optimization pipeline: Nesterov -> L-BFGS -> Nesterov2 -> CoordDescent -> L-BFGS."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    # Step 1: Nesterov GD (medium tau)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 25:
        alloc = remaining * 0.35
        a1 = nesterov_gd(a, alloc, 50.0, 2000.0)
        obj1 = compute_obj_normalized(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
        if verbose:
            print(f"        Nesterov1: {obj1:.8f} (best={best_obj:.8f})")

    # Step 2: L-BFGS-B
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        alloc = remaining * 0.2
        a2 = lbfgs_polish(best_a, alloc, tau=300.0, maxiter=min(3000, int(alloc * 8)),
                          verbose=verbose)
        obj2 = compute_obj_normalized(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()

        # Step 2b: L-BFGS again with higher tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            alloc2 = remaining * 0.2
            a2b = lbfgs_polish(best_a, alloc2, tau=2000.0, maxiter=min(2000, int(alloc2 * 8)),
                               verbose=verbose)
            obj2b = compute_obj_normalized(a2b)
            if obj2b < best_obj:
                best_obj = obj2b
                best_a = a2b.copy()

    # Step 3: High-tau Nesterov
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        alloc = remaining * 0.25
        a3 = nesterov_gd(best_a, alloc, 500.0, 10000.0)
        obj3 = compute_obj_normalized(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
        if verbose:
            print(f"        Nesterov2: {obj3:.8f} (best={best_obj:.8f})")

    # Step 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        alloc = remaining * 0.2
        a4 = coordinate_descent(best_a, alloc)
        obj4 = compute_obj_normalized(a4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
        if verbose:
            print(f"        CoordDescent: {obj4:.8f} (best={best_obj:.8f})")

    # Step 5: Final L-BFGS with very high tau
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = remaining * 0.3
        a5 = lbfgs_polish(best_a, alloc, tau=5000.0, maxiter=min(3000, int(alloc * 6)),
                          verbose=verbose)
        obj5 = compute_obj_normalized(a5)
        if obj5 < best_obj:
            best_obj = obj5
            best_a = a5.copy()

    return best_a, best_obj
