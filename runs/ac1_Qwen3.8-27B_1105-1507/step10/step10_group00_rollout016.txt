import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_fast(a, active_k=None, max_constraints=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for (selected) k, sum(g)=1, g>=0.
    Uses scipy.optimize.linprog with HiGHS.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Determine which constraints to use
    if active_k is None or max_constraints is None or max_constraints >= 2*n-1:
        ks = list(range(2*n-1))
    else:
        # Sort all k by conv(a, a)[k] and take top ones
        conv_aa = compute_conv(a)
        ks = list(range(2*n-1))
        if active_k is not None:
            # Include previously active plus top-k
            top_ks = np.argsort(conv_aa)[::-1][:max_constraints]
            ks = list(set(list(active_k) + list(top_ks)))[:max_constraints]
    
    num_k = len(ks)
    
    # Build constraint matrix: A_eq @ g = 1, A_ub @ g <= M
    # Variables: [g_0, ..., g_{n-1}, M]
    # Constraints: sum(g_i * a_{k-i}) - M <= 0 for each k
    #              sum(g) = 1
    #              g >= 0 (M is free)
    
    # A_ub: num_k x (n+1)
    A_ub = np.zeros((num_k, n + 1))
    b_ub = np.zeros(num_k)
    for idx, k in enumerate(ks):
        # conv(a_norm, g)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a_norm[i] * g[k-i]
        # = sum_j g[j] * a_norm[k-j] (valid j)
        # So row idx of A_ub: A_ub[idx, j] = a_norm[k-j] if 0<=k-j<n, else 0
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_ub[idx, j] = a_norm[k - j]
        A_ub[idx, n] = -1.0  # coefficient of M
    
    # A_eq: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    # Objective: minimize M
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Bounds: g_i >= 0, M is free
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n]), ks
    return None, None, ks


def subgradient_descent(a, time_budget=30.0, top_k=5, verbose=False):
    """
    Subgradient descent on the simplex for min max_k conv(a,a)[k].
    Uses top-k smoothing for a better gradient estimate.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = float(np.max(compute_conv(a)))
    start = time.monotonic()
    lr = 0.1 / np.sqrt(n)
    it = 0
    best_it = 0
    
    # AdaGrad
    grad_sq = np.zeros(n)
    lr_const = 0.5
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Get top-k indices
        top_indices = np.argsort(conv)[::-1][:top_k]
        weights = np.zeros(2*n-1)
        for idx in top_indices:
            weights[idx] = conv[idx]
        w_sum = weights[top_indices].sum()
        if w_sum > 0:
            weights[top_indices] = weights[top_indices] / w_sum
        else:
            weights[top_indices[0]] = 1.0
        
        # Compute gradient: weighted sum of subgradients
        grad = np.zeros(n)
        for k in top_indices:
            w = weights[k]
            if w < 1e-10:
                continue
            for i in range(n):
                j1 = k - i
                if 0 <= j1 < n:
                    grad[i] += w * a[j1]
                j2 = i - k
                if 0 <= j2 < n:
                    grad[i] += w * a[j2]
        
        # AdaGrad update
        grad_sq += grad ** 2
        a = np.maximum(a - lr_const * grad / (np.sqrt(grad_sq) + 1e-8), 0)
        s_new = float(np.sum(a))
        if s_new < 1e-10:
            a = best_a.copy()
        else:
            a = a / s_new
        
        obj = float(np.max(compute_conv(a)))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            best_it = it
        
        it += 1
        if it - best_it > 3000:
            a = best_a.copy()
            grad_sq = np.zeros(n)
            best_it = it
        
        if it % 1000 == 0 and verbose:
            print(f"    SGD it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    
    return best_a


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s


def ramp_init(n, slope=0.3):
    """Linear ramp initialization (asymmetric)."""
    a = 1.0 + slope * (np.arange(n) / (n - 1) - 0.5) * 2
    a = np.maximum(a, 0.01)
    s = float(np.sum(a))
    return a / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 + 0.5 * np.sin(np.pi * (2 * x - 1) ** 2)
    elif mode == 6:
        a = 1.0 / np.cosh(2 * (2 * x - 1) - np.log(2))
    elif mode == 7:
        # Ramp with noise
        a = 1.0 + 0.3 * (x - 0.5) * 2 + rng.normal(0, 0.05, n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    s = float(np.sum(a))
    return a / s
