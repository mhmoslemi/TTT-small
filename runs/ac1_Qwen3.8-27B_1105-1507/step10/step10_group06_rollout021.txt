import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix_rows(a, rows):
    """Build only specific rows of the convolution matrix.
    rows: array of indices into the (2n-1) convolution output
    Returns A (len(rows) x n) such that A[i] @ g = (a*g)[rows[i]]
    """
    n = len(a)
    A = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        # (a*g)[k] = sum_j a[k-j] * g[j] for valid j
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[idx, j] = a[k - j]
    return A

def solve_sparse_lp(a, constraint_lags, g_warm=None):
    """Solve LP: min M s.t. (a*g)[k] <= M for k in constraint_lags, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_rows(a_norm, constraint_lags)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    if g_warm is not None:
        g.value = g_warm
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, warm_start=True)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def build_conv_matrix_selected(a, lags):
    n = len(a)
    K = len(lags)
    A = np.zeros((K, n))
    for idx, k in enumerate(lags):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[idx, j] = a[k - j]
    return A


def solve_lp_selected(a, lags, g_warm=None):

    A = build_conv_matrix_selected(a_norm, lags)

                g_val = np.maximum(g.value, 0)

        return a

    it = 0

    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses accelerated LP alternating with adaptive constraint selection,
mirror descent polishing, and multi-length exploration.
Key improvements: Nesterov momentum on LP alternating, top-K constraint
selection for faster LPs, multiplicative mirror descent polish, and
systematic n-sweep with resampling.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_full(a):
    """Build full convolution matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def build_conv_matrix_selected(a, lags):
    """Build selected rows of convolution matrix for given lags."""
    n = len(a)
    K = len(lags)
    A = np.zeros((K, n))
    for idx in range(K):
        k = lags[idx]
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        A[idx, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_with_lags(a, lags, g_warm=None):
    """Solve LP: min M s.t. (a*g)[k] <= M for k in lags, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_selected(a_norm, lags)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    if g_warm is not None:
        g.value = g_warm

    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, warm_start=True)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0).astype(np.float64)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_full_constraints(a, g_warm=None):
    """Solve LP with all 2n-1 convolution constraints."""
    n = len(a)
    lags = np.arange(2 * n - 1)
    return solve_lp_with_lags(a, lags, g_warm=g_warm)


def lp_alternate(a, time_budget, k_lags=600, use_nesterov=True, full_every=200, seed=42):
    """
    LP alternating minimization with Nesterov acceleration and adaptive constraint selection.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    g_warm = a.copy()
    t_blend = 0.5
    stall = 0
    it = 0
    beta = 0.6
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.3:
            break

        # Nesterov lookahead
        if use_nesterov and it > 10:
            y = a + beta * v
            y = np.maximum(y, 0)
            s_y = float(np.sum(y))
            if s_y < 1e-10:
                y = a.copy()
            else:
                y = y / s_y
        else:
            y = a

        # Determine which lags to use

            lags = np.arange(2 * n - 1)
        else:
            conv_y = compute_conv(y)
            top_k = min(k_lags, 2 * n - 1)
            lags = np.argsort(conv_y)[-top_k:]

        # Solve LP
        g_lp, M_lp = solve_lp_with_lags(y, lags, g_warm=g_warm)
        if g_lp is None:
            break

        g_warm = g_lp
        it += 1

        best_local_obj = best_obj
        best_local_a = best_a.copy()

        for t in candidates_t:
            a_try = (1.0 - t) * y + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:

            a_try = a_try / s_try

                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                v = np.zeros(n)
                t_blend = 0.3
                stall = 0

    return best_a


def mirror_descent_polish(a, time_budget, tau_init=50.0, tau_max=10000.0):
    """
    Multiplicative mirror descent on the log-sum-exp smoothed objective.
    Naturally maintains positivity. Projects onto simplex after each step.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    tau = tau_init
    it = 0
    stall = 0

    N = 1
    while N < 3 * n:
        N <<= 1

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Log-sum-exp weights
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient via FFT: grad[i] = 2 * sum_k p[k] * a[k-i] (bounded)
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        grad = 2.0 * g_full[:n]

        # Multiplicative update (mirror descent with entropic regularization)
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-30)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        # Adaptive parameters
        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            target_lr = 2.0 / (g_norm * np.sqrt(n) + 1e-10)
            lr = 0.7 * lr + 0.3 * min(max(target_lr, 0.05 * lr), 3.0 * lr)
        tau = min(tau * 1.03, tau_max)
        lr *= 0.9995

        it += 1

        # Reset if stuck
        if stall > 200:
            a = best_a.copy()
            lr = 1.0 / np.sqrt(n)
            tau = max(tau_init, tau * 0.5)
            stall = 0

    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def cosine_init(n, seed=0):
    """Create a cosine-bell shaped initial sequence."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * (x - 0.5)) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()
