"""
Optimizes step function coefficients using enhanced coordinate descent with pair moves,
LP polishing with line search, systematic n exploration, and diverse initializations.
Key improvements over previous approach: (1) pair coordinate moves for more powerful
steps, (2) exact line search in LP polishing, (3) broader n sweep, (4) subgradient
refinement pass.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective for normalized (sum=1) sequence: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M.value)
        except Exception:
            continue
    return None, None


def golden_section_min(f, lo, hi, max_iter=50, tol=1e-14):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi - lo < tol:
        return (lo + hi) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol * max(1.0, abs(hi)):
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0


def cd_single_and_pair(a, time_budget=300.0, verbose=False, seed=42):

    Coordinate descent with both single-coordinate and pair moves.
    Single: golden section on a[i] (others fixed).
    Pair: golden section on t in [0,1] where a[i]=c*t, a[j]=c*(1-t).

    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1

        # Periodically recompute conv
        if sweep % 4 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        order = rng.permutation(n)
        S = float(np.sum(a))

            if time.monotonic() - start >= time_budget:
                break

            if do_pairs and idx + 1 < n and idx % 2 == 0:

                c_ij = a[i] + a[j]

                v_i = a[i]
                v_j = a[j]

                # Determine which conv indices are affected by i or j

                k_max_i = min(2 * n - 2, i + n - 1)
                k_min_j = max(0, j)
                k_max_j = min(2 * n - 2, j + n - 1)

                k_lo = min(k_min_i, k_min_j)
                k_hi = max(k_max_i, k_max_j)

                ks = np.arange(k_lo, k_hi + 1)
                conv_ks = conv[ks].copy()

                # conv_new[k] = conv_ks[k] + delta_i[k]*t + delta_j[k]*(1-t) + cross terms
                # Actually: a[i] changes from v_i to c_ij*t, a[j] changes from v_j to c_ij*(1-t)
                # delta_a_i = c_ij*t - v_i
                # delta_a_j = c_ij*(1-t) - v_j = c_ij - c_ij*t - v_j
                # Note: delta_a_i + delta_a_j = c_ij - v_i - v_j = 0 (sum preserved)

                # conv change for index k:
                # If i contributes to conv[k]: 2*a[k-i]*delta_a_i (plus a[i]^2 term if k=2i)
                # If j contributes to conv[k]: 2*a[k-j]*delta_a_j (plus a[j]^2 term if k=2j)

                # Build linear and quadratic coefficients in t
                # conv_new[k] = conv[k] + 2*a[k-i]*(c_ij*t - v_i) [if valid]
                #                          + 2*a[k-j]*(c_ij*(1-t) - v_j) [if valid]
                #                          + (c_ij*t)^2 - v_i^2 [if k=2i]
                #                          + (c_ij*(1-t))^2 - v_j^2 [if k=2j]

                # For each k, conv_new[k] is a quadratic in t: A_k*t^2 + B_k*t + C_k
                # We need max_k conv_new[k], which is the max of quadratics in t.
                # This is convex in t (max of convex functions), so golden section works.

                def pair_obj(t, c_ij=c_ij, v_i=v_i, v_j=v_j, i=i, j=j, n=n,
                            conv=conv, a=a, ks=ks, conv_ks=conv_ks):
                    da_i = c_ij * t - v_i
                    da_j = c_ij * (1.0 - t) - v_j

                    new_conv_vals = conv_ks.copy()

                    # Contribution from i
                    ks_i = ks
                    # For k in range, i contributes if 0 <= k-i < n
                    mask_i = (ks_i >= i) & (ks_i < i + n)
                    if np.any(mask_i):
                        # new_conv[k] += 2 * a[k-i] * da_i for k != 2i
                        # new_conv[k] += (c_ij*t)^2 - v_i^2 for k == 2i

                        a_vals = a[valid_i - i]
                        contrib_i[mask_i] = 2.0 * a_vals * da_i
                        # Handle k = 2i (quadratic term)
                        if 2 * i >= k_lo and 2 * i <= k_hi:
                            pos_2i = 2 * i - k_lo
                            contrib_i[pos_2i] = (c_ij * t) ** 2 - v_i * v_i
                        new_conv_vals += contrib_i

                    mask_j = (ks >= j) & (ks < j + n)
                    if np.any(mask_j):
                        contrib_j = np.zeros(len(ks))
                        valid_j = ks[mask_j]
                        a_vals_j = a[valid_j - j]
                        contrib_j[mask_j] = 2.0 * a_vals_j * da_j
                        if 2 * j >= k_lo and 2 * j <= k_hi:
                            pos_2j = 2 * j - k_lo
                            contrib_j[pos_2j] = (c_ij * (1.0 - t)) ** 2 - v_j * v_j
                        new_conv_vals += contrib_j

                    return float(np.max(new_conv_vals))

                # Golden section on t in [0, 1]
                t_opt = golden_section_min(lambda t: pair_obj(t), 0.0, 1.0, max_iter=40)
                t_opt = max(0.0, min(1.0, t_opt))

                new_ai = c_ij * t_opt
                new_aj = c_ij * (1.0 - t_opt)

                if abs(new_ai - v_i) < 1e-14 and abs(new_aj - v_j) < 1e-14:
                    continue

                # Update a and conv
                a[i] = new_ai
                a[j] = new_aj
                conv = compute_conv(a)  # Recompute for correctness

                S = float(np.sum(a))
                if S > 1e-10:
                    obj = 2.0 * n * float(np.max(conv)) / (S * S)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()

            else:
                # Single coordinate move
                v = a[i]
                S_other = S - v
                if S_other < 1e-12:
                    continue

                k_min = max(0, i)
                k_max = min(2 * n - 2, i + n - 1)
                ks = np.arange(k_min, k_max + 1)
                a_ki = a[ks - i]
                c_k = conv[ks]

                mask = (ks != 2 * i)
                ks_lin = ks[mask]
                a_ki_lin = a_ki[mask]
                c_k_lin = c_k[mask]

                k_quad = 2 * i
                c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

                def f_1d(x, v=v, S_other=S_other, n=n, c_k_lin=c_k_lin,
                        a_ki_lin=a_ki_lin, c_2i=c_2i):
                    if len(a_ki_lin) > 0:
                        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                        max_lin = float(np.max(lin_vals))
                    else:
                        max_lin = 0.0
                    quad_val = c_2i + x * x - v * v
                    max_val = max(max_lin, quad_val)
                    denom = (S_other + x) * (S_other + x)
                    if denom < 1e-30:
                        return 1e30
                    return 2.0 * n * max_val / denom

                # Bracket
                scale = max(v, S / n, 1e-6)
                candidates = [0.0, v * 0.5, v, v * 1.5, v * 2.0,
                              min(S_other * 0.3, S), min(S_other * 0.5, S * 0.5)]
                candidates = sorted(set(max(0.0, float(c2)) for c2 in candidates))

                best_x = v
                best_f = f_1d(v)
                for x in candidates:
                    fx = f_1d(x)
                    if fx < best_f:
                        best_f = fx
                        best_x = x

                x_lo = max(0.0, best_x - scale)
                x_hi = best_x + scale * 1.5
                if x_hi - x_lo < 1e-12:
                    x_hi = x_lo + 1e-8

                x_opt = golden_section_min(f_1d, x_lo, x_hi, max_iter=40)
                x_opt = max(0.0, x_opt)

                if abs(x_opt - v) < 1e-14:
                    continue

                delta = x_opt - v
                a[i] = x_opt

                # Update conv incrementally
                if 0 <= k_quad <= 2 * n - 2:
                    conv[k_quad] += x_opt * x_opt - v * v
                if len(ks_lin) > 0:
                    conv[ks_lin] += 2.0 * delta * a_ki_lin

                S += delta

                if S > 1e-10:
                    obj = 2.0 * n * float(np.max(conv)) / (S * S)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def lp_polish_linesearch(a, time_budget=60.0, verbose=False, seed=42):
    """LP-based polishing with exact golden section line search on step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Line search: find best t in [0, 1] for a_new = (1-t)*a + t*g_lp
        def step_obj(t, a=a, g_lp=g_lp, n=n):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return 1e30
            a_try = a_try / s_try
            return 2.0 * n * float(np.max(compute_conv(a_try)))

        t_opt = golden_section_min(step_obj, 0.0, 1.0, max_iter=30)
        t_opt = max(0.0, min(1.0, t_opt))

        a_new = (1.0 - t_opt) * a + t_opt * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 4:
                # Perturb
                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if time.monotonic() - start > time_budget * 0.85:
                    break

        if verbose and it % 3 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def subgradient_polish(a, time_budget=30.0, verbose=False):
    """Projected subgradient method for final refinement."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    lr_init = 1.0 / np.sqrt(n)
    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient of max_k (a*a)[k] at a, assuming k* is the argmax
        # g[j] = 2*a[k*-j] if 0 <= k*-j < n, else 0
        g = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        for j in range(j_min, j_max + 1):
            g[j] = 2.0 * a[k_star - j]

        lr = lr_init / np.sqrt(1.0 + it * 0.1)
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        obj = 2.0 * n * float(np.max(compute_conv(a)))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if verbose and it % 1000 == 0:
            print(f"    Subgrad it{it}: obj={best_obj:.8f}")

    return best_a


def diverse_init(n, rng, mode=0, parent=None):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Blackman-Harris like
        x = np.arange(n) / max(n - 1, 1)
        a = 0.35875 - 0.48829 * np.cos(2 * np.pi * x) + 0.14128 * np.cos(4 * np.pi * x) \
            - 0.01168 * np.cos(6 * np.pi * x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 3:
        # Raised cosine
        x = np.arange(n) / max(n - 1, 1)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a / a.sum()
    elif mode == 4:
        # Gaussian centered
        c = n / 2.0
        w = n / 4.0
        x = np.arange(n)
        a = np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Two bumps
        c1, c2 = n * 0.3, n * 0.7
        w = n * 0.1
        x = np.arange(n)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Triangular
        x = np.arange(n)
        c = n / 2.0
        a = 1.0 - np.abs(x - c) / c
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n

    if parent is not None and len(parent) == n:
        blend = 0.3
        a = (1 - blend) * a + blend * parent
        a = np.maximum(a, 0)
        a = a / a.sum()

    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining CD with pair moves, LP polishing with line search,
    systematic n exploration, and subgradient refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent_arr = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Quick n sweep with diverse initializations ===
    print(f"\n=== Phase 1: N Sweep ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 350.0)

    # Generate n values to try
    n_options = []
    for n_try in range(400, 1801, 50):
        n_options.append(n_try)
    # Add parent n and nearby
    n_options.append(n_parent)
    n_options.append(max(400, n_parent - 100))
    n_options.append(min(1800, n_parent + 100))
    n_options = sorted(set(n for n in n_options if 400 <= n <= 1800))

    # Limit to reasonable number based on budget
    n_per_try = max(10, int(p1_budget / len(n_options)))
    n_per_try = min(n_per_try, 25)
    if n_per_try * len(n_options) > p1_budget:
        # Subsample
        step = max(1, len(n_options) * n_per_try // int(p1_budget))
        n_options = n_options[::step]

    print(f"  Trying {len(n_options)} n values, ~{n_per_try}s each")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < n_per_try + 10:
            break

        alloc = min(n_per_try, remaining * 0.9)
        if alloc < 10:
            break

        # Try a couple of initializations
        best_for_n = np.inf
        best_a_for_n = None
        for mode in [0, 2, 4]:
            a_try = diverse_init(n_try, rng, mode=mode, parent=best_seq if len(best_seq) == n_try else None)
            a_cd = cd_single_and_pair(a_try, time_budget=alloc / 3.0, verbose=False,
                                      seed=seed + n_try + mode)
            obj_try = compute_obj_normalized(a_cd)
            if obj_try < best_for_n:
                best_for_n = obj_try
                best_a_for_n = a_cd

        if best_a_for_n is not None and best_for_n < best_score:
            best_score = best_for_n
            best_seq = best_a_for_n.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f}")
        elif n_try == n_parent or n_options.index(n_try) % 5 == 0:
            print(f"  n={n_try}: obj={best_for_n:.8f}")

    # === Phase 2: Main CD on best n ===
    print(f"\n=== Phase 2: Main CD (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 500.0)
    if p2_budget > 30:
        a_cd = cd_single_and_pair(best_seq, time_budget=p2_budget,
                                  verbose=True, seed=seed)
        obj_cd = compute_obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Main CD: obj={best_score:.8f}")
        else:
            print(f"  Main CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 3: LP Polish ===
    print(f"\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 250.0)
    if p3_budget > 30:
        for rnd in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(p3_budget / (5 - rnd), remaining * 0.6)
            if alloc < 15:
                break
            a_lp = lp_polish_linesearch(best_seq, time_budget=alloc, verbose=False,
                                        seed=1000 + rnd * 111)
            obj_lp = compute_obj_normalized(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f}")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                # Short CD after LP
                if remaining > 20:
                    a_cd2 = cd_single_and_pair(a_lp, time_budget=min(30, remaining * 0.15),
                                               verbose=False, seed=3000 + rnd * 77)
                    obj_cd2 = compute_obj_normalized(a_cd2)
                    if obj_cd2 < best_score:
                        best_score = obj_cd2
                        best_seq = a_cd2.copy()
                        print(f"  CD after LP{rnd}: obj={best_score:.8f}")

    # === Phase 4: Final CD refinement ===
    print(f"\n=== Phase 4: Final CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 200.0)
        a_final = cd_single_and_pair(best_seq, time_budget=alloc,
                                     verbose=False, seed=seed + 5000)
        obj_final = compute_obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f}")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # === Phase 5: Subgradient polish ===
    print(f"\n=== Phase 5: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.2, 60.0)
        a_sg = subgradient_polish(best_seq, time_budget=alloc, verbose=False)
        obj_sg = compute_obj_normalized(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f}")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    # === Phase 6: One more LP if time ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 80.0)
        a_lp2 = lp_polish_linesearch(best_seq, time_budget=alloc, verbose=False, seed=9999)
        obj_lp2 = compute_obj_normalized(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Extra LP: obj={best_score:.8f}")
        # Quick CD after
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_cd3 = cd_single_and_pair(best_seq, time_budget=min(remaining * 0.5, 60.0),
                                       verbose=False, seed=7777)
            obj_cd3 = compute_obj_normalized(a_cd3)
            if obj_cd3 < best_score:
                best_score = obj_cd3
                best_seq = a_cd3.copy()
                print(f"  CD after extra LP: obj={best_score:.8f}")

    # === Phase 7: Try a few different n near the best one ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n_cur = len(best_seq)
        for dn in [-50, 50, -100, 100]:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            n_try = n_cur + dn
            if n_try < 400 or n_try > 1800:
                continue
            a_res = resample_sequence(best_seq, n_try)
            alloc = min(remaining * 0.3, 50.0)
            a_res = cd_single_and_pair(a_res, time_budget=alloc, verbose=False,
                                       seed=n_try)
            obj_res = compute_obj_normalized(a_res)
            if obj_res < best_score:
                best_score = obj_res
                best_seq = a_res.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_res:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
