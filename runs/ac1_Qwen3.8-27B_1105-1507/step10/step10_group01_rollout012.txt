import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a_norm):
    """Build convolution matrix A (2n-1 x n) for (a_norm @ g)."""
    n = len(a_norm)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a_norm[diff[valid]]
    return A

def solve_lp(a_norm):
    """Solve LP: min M s.t. conv(a_norm,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a_norm)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def smoothed_grad(a, tau):
    """Compute gradient of soft-max smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_norm):
    """Build convolution matrix A (2n-1 x n) for (a_norm*g)[k]."""
    n = len(a_norm)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a_norm[diff[valid]]
    return A


def solve_lp(a_norm):
    """Solve LP: min M s.t. conv(a_norm,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a_norm)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_grad(a, tau):
    """Compute gradient of soft-max smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]

def palindromic_project(a):

    for j in range(half):
        a_pal[j] = (a[j] + a[n - 1 - j]) / 2.0
        a_pal[n - 1 - j] = a_pal[j]
    a_pal = np.maximum(a_pal, 0)
    s = float(np.sum(a_pal))
    if s < 1e-10:
        return np.ones(n) / n
    return a_pal / s

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f}")

    # === Phase 3: GD polish ===
    print(f"\n=== Phase 3: Adam GD polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 100.0)
    if p3_budget > 15:
        a_gd = adam_smoothed_gd(best_seq, time_budget=p3_budget,
                                tau_init=200, tau_end=5000, lr_base=0.005)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase3 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 GD: obj={obj_gd:.8f}")

    # === Phase 4: Second LP round ===
    print(f"\n=== Phase 4: Second LP round ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.18, 200.0)
    if p4_budget > 20:
        a_lp2 = lp_driven_with_avg(best_seq, time_budget=p4_budget, max_iters=100)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase4 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 LP: obj={obj_lp2:.8f}")

    # === Phase 5: Expanded multi-n exploration ===
    print(f"\n=== Phase 5: Expanded multi-n exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = set()
    for delta in [-300, -200, -150, -100, -50, 50, 100, 150, 200, 300, 400, 500]:
        n_try = n_best + delta
        if 300 <= n_try <= 2000:
            n_options.add(n_try)
    for n_fix in [500, 700, 900, 1100, 1300, 1500]:
        if abs(n_fix - n_best) > 30:
            n_options.add(n_fix)

    n_options = sorted(n_options - {n_best})[:7]
    print(f"  n options: {n_options}")

    time_per_n = min(remaining * 0.35 / max(len(n_options), 1), 50.0)

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        a_try = resample_sequence(best_seq, n_try)
        a_try = optimize_at_n(a_try, time_budget=time_per_n, max_lp_iters=30)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    n_best = len(best_seq)

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    restarts = []

    # Palindromic projection
    a_pal = palindromic_project(best_seq)
    restarts.append(("palindromic", a_pal))

    # Mirror blend
    a_mirror = (best_seq + best_seq[::-1]) / 2.0
    a_mirror = np.maximum(a_mirror, 0)
    a_mirror = a_mirror / a_mirror.sum()
    restarts.append(("mirror", a_mirror))

    # Gaussian window blend
    x = np.arange(n_cur).astype(float)
    a_gauss = np.exp(-0.5 * ((x - n_cur / 2) / (n_cur / 3.5)) ** 2)
    a_gauss = 0.7 * best_seq + 0.3 * (a_gauss / a_gauss.sum())
    a_gauss = np.maximum(a_gauss, 0)
    a_gauss = a_gauss / a_gauss.sum()
    restarts.append(("gauss", a_gauss))

    # Fourier perturbation
    a_four = fourier_perturb(best_seq, rng, strength=0.05)
    restarts.append(("fourier", a_four))

    # Smooth random blend
    a_rand = rng.uniform(0.3, 1.7, n_cur)
    ks = max(5, n_cur // 8)
    if ks % 2 == 0:
        ks += 1
    a_rand = uniform_filter1d(a_rand, size=ks)
    a_rand = 0.6 * best_seq + 0.4 * (a_rand / a_rand.sum())
    a_rand = np.maximum(a_rand, 0)
    a_rand = a_rand / a_rand.sum()
    restarts.append(("rand_smooth", a_rand))

    # Triangle
    tri = np.minimum(x, n_cur - 1 - x) + 1.0
    tri /= tri.sum()
    a_tri = 0.5 * best_seq + 0.5 * tri
    a_tri = a_tri / a_tri.sum()
    restarts.append(("triangle", a_tri))

    time_per_restart = min(remaining * 0.4 / max(len(restarts), 1), 35.0)

    for name, a_r in restarts:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        a_r = optimize_at_n(a_r, time_budget=time_per_restart, max_lp_iters=25)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 25:
        alloc = min(remaining * 0.5, 120.0)
        a_final = lp_driven_with_avg(best_seq, time_budget=alloc, max_iters=80)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.55, 80.0)
        a_final2 = adam_smoothed_gd(best_seq, time_budget=alloc,
                                    tau_init=500, tau_end=8000, lr_base=0.003)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
