def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)

        return np.array([])
    size = 1
    while size < 2 * n - 1:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""

    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, max_constraints=None, verbose=False):

                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) ** 2
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Golden section search
            scale = max(v, S / n, 1e-6)
            
            # Quick scan
            candidates = [0.0, v * 0.3, v * 0.7, v, v * 1.3, v * 1.7, 2.0 * S / n]
            candidates = sorted(set(max(0.0, min(10.0 * S, float(c))) for c in candidates))
            
            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Refine
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 1.5
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(40):
                if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            # Update
            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta
            improved = True

        if not improved:
            # Try a random perturbation to escape
            if rng.random() < 0.3:
                idx = rng.integers(n)
                a[idx] = max(0, a[idx] + rng.normal(0, S/n))
                S = float(np.sum(a))
                a = a / S
                conv = compute_conv(a)

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            obj_now = 2.0 * n * float(np.max(conv)) / (float(np.sum(a)) ** 2)
            print(f"    CD sweep {sweep}: obj={best_obj:.8f} (current={obj_now:.8f}), t={elapsed:.1f}s")

    return best_a


def successive_linearization(a, time_budget=400.0, verbose=True, seed=42):
    """
    Successive linearization method: iteratively solve LP relaxation
    with trust region, blending with current solution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    trust_radius = 1.0  # blending factor
    stall = 0

    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 10:
            break

        it += 1

        # Solve LP with top constraints (adaptive number)
        conv = compute_conv(a)
        top_k = min(2 * n - 1, max(20, int(0.1 * n)))
        
        g_lp, M_lp = solve_lp_minpeak(a, max_constraints=top_k)
        if g_lp is None:
            # Fallback: full LP
            g_lp, M_lp = solve_lp_minpeak(a)
            if g_lp is None:
                stall += 1
                if stall > 5:
                    break
                # Perturb
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                continue

        # Line search along direction a -> g_lp
        best_t = 0.0
        best_t_obj = best_obj
        
        # Try a range of t values
        t_candidates = np.linspace(0.05, min(1.0, trust_radius), 20)
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t > 0 and best_t_obj < best_obj - 1e-12:
            a = (1.0 - best_t) * a + best_t * g_lp
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_obj = best_t_obj
            best_a = a.copy()
            stall = 0
            # Increase trust region if we made good progress
            if best_t >= 0.8:
                trust_radius = min(1.0, trust_radius * 1.2)
        else:
            stall += 1
            if stall == 1:
                trust_radius = max(0.1, trust_radius * 0.5)
            elif stall > 3:
                # Perturb to escape
                pert_scale = 0.05 * stall
                a = a * np.exp(rng.normal(0, pert_scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                trust_radius = 0.5

            if stall > 10:
                # Larger perturbation
                a = a * np.exp(rng.normal(0, 0.2, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                trust_radius = 1.0

        if verbose and it % 5 == 0:
            print(f"    SL it{it}: obj={best_obj:.8f}, t={best_t:.3f}, trust={trust_radius:.2f}, t_elapsed={elapsed:.1f}s")

    return best_a


def peak_balancing(a, time_budget=100.0, verbose=False):
    """
    Directly target the top peaks and try to reduce them by adjusting
    the coordinates that contribute most.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(123)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        obj = 2.0 * n * max_val / (float(np.sum(a)) ** 2)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Find top peaks
        threshold = 0.95 * max_val
        peak_ks = np.where(conv >= threshold)[0]

        if len(peak_ks) == 0:
            break

        # For each peak, find the coordinates that contribute most
        for k_star in peak_ks[:5]:
            # Contribution of coordinate i to peak k_star: a[i] * a[k_star - i]
            contributions = np.zeros(n)
            for i in range(n):
                ki = k_star - i
                if 0 <= ki < n:
                    contributions[i] = a[i] * a[ki]
            
            # Find high-contributing coordinates and try to reduce them
            top_contrib = np.argsort(contributions)[::-1][:min(10, n)]
            
            for i in top_contrib[:3]:
                # Try reducing a[i]
                reduction = rng.uniform(0.01, 0.1) * a[i]
                a_new = a.copy()
                a_new[i] = max(0, a[i] - reduction)
                # Redistribute to coordinates with low contribution to this peak
                low_contrib = np.argsort(contributions)[:min(5, n)]
                for j in low_contrib:
                    if j != i:
                        a_new[j] += reduction / len(low_contrib)
                
                a_new = np.maximum(a_new, 0)
                s_new = float(np.sum(a_new))
                if s_new < 1e-10:
                    continue
                a_new = a_new / s_new
                obj_new = 2.0 * n * float(np.max(compute_conv(a_new))) / (s_new ** 2)
                # Wait, I already normalized, so s_new should be 1
                obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
                
                if obj_new < obj:
                    a = a_new

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps
        c1, c2 = n // 4, 3 * n // 4
        w = n // 6
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x**2)
        a = a / a.sum()
    elif mode == 4:
        # One-sided
        a = np.exp(-np.arange(n) * 0.5)
        a = a / a.sum()
    elif mode == 5:
        # Chirp-like (using sine)
        x = np.arange(n) / n
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 3 * x + 1)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1 - np.abs(2 * x - 1)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines successive linearization, priority coordinate descent,
    peak balancing, and multi-n exploration for optimal step function coefficients.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0  # Safety margin

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except (NameError, Exception):
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = 2.0 * n_default * float(np.max(compute_conv(best_seq)))
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    # === Phase 1: Quick Coordinate Descent ===
    print(f"\n=== Phase 1: Quick CD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.15, 150.0)
    if p1_budget > 20:
        a_cd = coordinate_descent_priority(best_seq, time_budget=p1_budget,
                                           verbose=True, seed=seed)
        obj_cd = 2.0 * a_cd.size * float(np.max(compute_conv(a_cd / a_cd.sum())))
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase 1 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: Successive Linearization (main workhorse) ===
    print(f"\n=== Phase 2: Successive Linearization ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.4, 500.0)
    if p2_budget > 30:
        # Run multiple rounds with different seeds
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = p2_budget / (3 - rnd) if rnd < 2 else remaining * 0.7
            alloc = min(alloc, remaining * 0.8)
            if alloc < 20:
                break
            a_sl = successive_linearization(best_seq, time_budget=alloc,
                                            verbose=True, seed=seed + rnd * 100)
            obj_sl = 2.0 * a_sl.size * float(np.max(compute_conv(a_sl / a_sl.sum())))
            if obj_sl < best_score:
                best_score = obj_sl
                best_seq = a_sl.copy()
                print(f"  SL round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  SL round {rnd}: obj={obj_sl:.8f}")
            
            # Quick CD polish after each SL round
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_polish = coordinate_descent_priority(best_seq, time_budget=min(30, remaining * 0.1),
                                                       verbose=False, seed=7000 + rnd)
                obj_polish = 2.0 * a_polish.size * float(np.max(compute_conv(a_polish / a_polish.sum())))
                if obj_polish < best_score:
                    best_score = obj_polish
                    best_seq = a_polish.copy()
                    print(f"  CD after SL {rnd}: obj={best_score:.8f} **")

    # === Phase 3: Peak Balancing ===
    print(f"\n=== Phase 3: Peak Balancing ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        pb_budget = min(remaining * 0.15, 150.0)
        a_pb = peak_balancing(best_seq, time_budget=pb_budget, verbose=True)
        obj_pb = 2.0 * a_pb.size * float(np.max(compute_conv(a_pb / a_pb.sum())))
        if obj_pb < best_score:
            best_score = obj_pb
            best_seq = a_pb.copy()
            print(f"  Peak balancing: obj={best_score:.8f} **")
        else:
            print(f"  Peak balancing: obj={obj_pb:.8f}")

    # === Phase 4: More CD Refinement ===
    print(f"\n=== Phase 4: CD Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        cd_budget = min(remaining * 0.2, 200.0)
        a_cd2 = coordinate_descent_priority(best_seq, time_budget=cd_budget,
                                            verbose=True, seed=seed + 500)
        obj_cd2 = 2.0 * a_cd2.size * float(np.max(compute_conv(a_cd2 / a_cd2.sum())))
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Phase 4 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4 CD: obj={obj_cd2:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = []
    # Nearby options
    for delta in [-100, -50, 50, 100, 200, 300, 400]:
        n_try = n_best + delta
        if 300 <= n_try <= 2000:
            n_options.append(n_try)
    # Some fixed options
    for n_fixed in [500, 600, 700, 800, 900, 1000, 1100, 1200, 1500, 2000]:
        if 300 <= n_fixed <= 2000 and abs(n_fixed - n_best) > 30:
            n_options.append(n_fixed)

    n_options = sorted(set(n_options))
    # Limit to 6 options
    n_options = n_options[:6]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        
        # Quick CD
        alloc = min(remaining * 0.2, 60.0)
        if alloc > 15:
            a_try = coordinate_descent_priority(a_try, time_budget=alloc,
                                                verbose=False, seed=n_try)

        obj_try = 2.0 * a_try.size * float(np.max(compute_conv(a_try / a_try.sum())))
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = r + 1
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 60.0)
        if alloc > 20:
            a_r = coordinate_descent_priority(a_r, time_budget=alloc,
                                              verbose=False, seed=9000 + r * 100)
            obj_r = 2.0 * a_r.size * float(np.max(compute_conv(a_r / a_r.sum())))
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Final CD
        alloc = min(remaining * 0.5, 120.0)
        a_final = coordinate_descent_priority(best_seq, time_budget=alloc,
                                              verbose=False, seed=seed + 9999)
        obj_final = 2.0 * a_final.size * float(np.max(compute_conv(a_final / a_final.sum())))
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

        # Final SL
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_sl_final = successive_linearization(best_seq, time_budget=min(remaining * 0.5, 60.0),
                                                  verbose=False, seed=7777)
            obj_sl_final = 2.0 * a_sl_final.size * float(
