from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""

    The 1D subproblem has closed-form solution: x_i* = S2_other / S_other.
    This converges to uniform in one sweep.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return np.ones(n) / n
    a = a / S

    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()

    for sweep in range(50):
        if time.monotonic() - start >= time_budget:
            break

        S2 = float(np.sum(a * a))
        S = float(np.sum(a))

        for i in range(n):
            v = a[i]
            S_other = S - v
            S2_other = S2 - v * v

            if S_other < 1e-14:
                continue

            # Closed-form optimal for minimizing sum(a^2)/(sum(a))^2
            x_opt = S2_other / S_other
            x_opt = max(0.0, x_opt)

            if abs(x_opt - v) > 1e-15:
                S2 = S2 - v * v + x_opt * x_opt
                S = S - v + x_opt
                a[i] = x_opt

        # Normalize
        S = float(np.sum(a))
        if S > 1e-10:
            a = a / S
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        if verbose:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}")

        # Check convergence
        if best_obj < 2.0 + 1e-10:
            break

    return best_a


def smoothed_gradient_descent(a, time_budget=200.0, verbose=False):
    """
    Smoothed gradient descent using log-sum-exp approximation of max.
    Works for general (not just non-negative) objectives.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1

    tau = 50.0
    lr = 5.0 / np.sqrt(n)
    beta = 0.85
    v = np.zeros(n)

    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = 30.0 + 300.0 * progress

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(np.minimum(tau * (conv - max_conv), 500))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        P = fft(p, N_fft)
        A = fft(y, N_fft)
        g_full = ifft(P * np.conj(A)).real
        grad = 2.0 * g_full[:n]

        # Also subtract the mean gradient to maintain zero-sum-like property
        grad -= np.sum(grad) * y / max(float(np.sum(y)), 1e-10)

        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            lr = np.clip(lr * 0.9995, 1.0 / np.sqrt(n), 10.0 / np.sqrt(n))
            step = lr / g_norm
        else:
            step = lr

        v = beta * v - step * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 10000 == 0:
            print(f"    GD it{it}: obj={best_obj:.10f}, tau={tau:.0f}")

    return best_a


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based polishing: find g minimizing max(a*g) under sum(g)=1, g>=0."""
    try:
        import cvxpy as cp
    except ImportError:
        return a

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Build A matrix where (A @ g)[k] = conv(a, g)[k]
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]

    while time.monotonic() - start < time_budget:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        g_val = None
        M_val = None
        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    M_val = float(M.value)
                    break
            except Exception:
                continue

        if g_val is None:
            break

        s_g = float(np.sum(g_val))
        if s_g < 1e-10:
            break
        g_val = g_val / s_g

        # Try blending
        improved = False
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        return a / a.sum()
    elif mode == 3:
        x = np.arange(n) / n
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x) + 0.2 * np.sin(7 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 4:
        centers = [n // 4, n // 2, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (n * n / 16))
        return a / a.sum()
    elif mode == 5:
        # Chirp-like
        x = np.arange(n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x / n * 5)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines:
    1. Fast coordinate descent (exploits non-negativity structure)
    2. Smoothed gradient descent (general purpose)
    3. LP polishing
    4. Multi-n exploration
    5. Diverse restarts
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        parent = None

    # Try uniform sequence first
    if best_seq is None:
        for n_try in [400, 600, 800, 1000, 1200, 1600, 2000]:
            uni = np.ones(n_try) / n_try
            obj = compute_obj(uni)
            if obj < best_score:
                best_score = obj
                best_seq = uni.copy()
        print(f"Uniform best: n={best_seq.size}, obj={best_score:.10f}")
    else:
        # Also check if uniform is better
        n_p = len(best_seq)
        uni = np.ones(n_p) / n_p
        obj_uni = compute_obj(uni)
        if obj_uni < best_score:
            best_score = obj_uni
            best_seq = uni.copy()
            print(f"Uniform better: obj={best_score:.10f}")

    # === Phase 1: Fast Coordinate Descent ===
    print(f"\n=== Phase 1: Fast Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 200.0)
        a_cd = fast_coordinate_descent(best_seq, time_budget=alloc, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.10f} **")
        else:
            print(f"  CD: obj={obj_cd:.10f} (no improvement)")

    # === Phase 2: Smoothed Gradient Descent ===
    print(f"\n=== Phase 2: Smoothed Gradient Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        # Perturb slightly to escape potential flat regions
        pert = rng.normal(0, 0.01, len(best_seq))
        a_pert = best_seq * np.exp(pert)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.25, 300.0)
        a_gd = smoothed_gradient_descent(a_pert, time_budget=alloc, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  GD: obj={best_score:.10f} **")
        else:
            print(f"  GD: obj={obj_gd:.10f} (no improvement)")

        # Follow up with CD
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_cd2 = fast_coordinate_descent(a_gd, time_budget=min(remaining * 0.1, 100.0))
            obj_cd2 = compute_obj(a_cd2)
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  CD after GD: obj={best_score:.10f} **")

    # === Phase 3: LP Polishing ===
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 120.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.10f} **")
        else:
            print(f"  LP: obj={obj_lp:.10f}")

        # CD after LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_cd3 = fast_coordinate_descent(a_lp, time_budget=min(remaining * 0.1, 60.0))
            obj_cd3 = compute_obj(a_cd3)
            if obj_cd3 < best_score:
                best_score = obj_cd3
                best_seq = a_cd3.copy()
                print(f"  CD after LP: obj={best_score:.10f} **")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(200, n_best - 200), max(200, n_best - 100),
        n_best + 100, n_best + 200, n_best + 500,
        300, 500, 700, 900, 1100, 1400, 1800
    ]))
    n_options = [n for n in n_options if 200 <= n <= 3000 and abs(n - n_best) > 50]
    n_options = n_options[:8]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.15, 80.0)
        if alloc > 10:
            a_try = fast_coordinate_descent(a_try, time_budget=alloc)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 5: Diverse Restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = r
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.15, 60.0)
        if alloc > 10:
            a_r = fast_coordinate_descent(a_r, time_budget=alloc * 0.6)
            obj_after_cd = compute_obj(a_r)

            # Also try GD on this
            if obj_after_cd > best_score and remaining > 60:
                a_r2 = smoothed_gradient_descent(a_r, time_budget=alloc * 0.4)
                obj_r2 = compute_obj(a_r2)
                if obj_r2 < obj_after_cd:
                    a_r = a_r2
                    obj_after_cd = obj_r2

            obj_r = obj_after_cd
        else:
            obj_r = compute_obj(a_r)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.10f}")

    # === Phase 6: Iterative refinement ===
    print(f"\n=== Phase 6: Iterative Refinement ===")
    remaining = deadline - time.monotonic()
    for it in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # GD with perturbation
        if remaining > 40:
            scale = 0.02 * (it + 1)
            pert = rng.normal(0, scale, len(best_seq))
            a_p = best_seq * np.exp(pert)
            a_p = np.maximum(a_p, 0)
            a_p = a_p / a_p.sum()
            alloc = min(remaining * 0.2, 80.0)
            a_p = smoothed_gradient_descent(a_p, time_budget=alloc)
            obj_p = compute_obj(a_p)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Refine {it} GD: obj={best_score:.10f} **")

        # CD polish
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_c = fast_coordinate_descent(best_seq, time_budget=min(remaining * 0.2, 40.0))
            obj_c = compute_obj(a_c)
            if obj_c < best_score:
                best_score = obj_c
                best_seq = a_c.copy()
                print(f"  Refine {it} CD: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(800)
        best_score = compute_obj(best_seq)

    # Ensure all values are valid for evaluate_sequence
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback to uniform
        result = [1.0 / 800.0] * 800
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
