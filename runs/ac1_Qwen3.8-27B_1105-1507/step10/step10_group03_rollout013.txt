"""

import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """Gradient of log-sum-exp smoothed max of conv(a,a). Returns (grad, exact_max)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = conv - exact_max
    max_shift = float(np.max(shifted))
    if max_shift > 50 / tau:
        exp_vals = np.exp(tau * (shifted - max_shift))
    else:
        exp_vals = np.exp(tau * shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def normalize_simplex(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def build_lp_data(n):
    """Precompute row/col indices for the convolution LP matrix (shape (2n-1, n))."""
    row_ind = []
    col_ind = []
    for m in range(n):
        col_ind.extend([m] * n)
        row_ind.extend(range(m, m + n))
    return np.array(row_ind, dtype=np.int32), np.array(col_ind, dtype=np.int32)


def lp_solve(a, lp_indices=None, solver_list=None):
    """Solve min M s.t. conv(a,g) <= M, g>=0, sum(g)=1. Returns (g, M) or (None, None)."""
    n = len(a)
    if n > 3500:
        return None, None
    if lp_indices is None:
        lp_indices = build_lp_data(n)
    row_ind, col_ind = lp_indices
    data = np.tile(a, n)
    A = sp.csr_matrix((data, (row_ind, col_ind)), shape=(2 * n - 1, n))
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

def anderson_lp(a, time_budget, m=5, verbose=False, lp_indices=None, full_lp_frac=0.6):
    """Anderson-accelerated LP fixed-point iteration."""
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    # History for Anderson
    X_hist = [a.copy()]
    F_hist = []
    step = 0
    while time.monotonic() - start < time_budget:
        # Solve LP at current a
        g, M = lp_solve(a, lp_indices=lp_indices)
        if g is None:
            break
        F_hist.append(g)
        # Check fixed-point residual
        residual = np.linalg.norm(g - a)
        step += 1
        if step <= m:
            # Simple iteration for first m steps
            a = g
        else:
            # Anderson acceleration
            m_act = min(m, len(X_hist))
            G = np.zeros((m_act, m_act))
            diffs = np.zeros((m_act, n))
            base = len(X_hist) - m_act
            for i in range(m_act):
                diffs[i] = F_hist[base + i] - X_hist[base + i]
                G[i, i] = 1.0
                for j in range(i):
                    G[i, j] = np.dot(diffs[i], diffs[j])
            e1 = np.zeros(m_act)
            e1[0] = 1.0
            try:
                gamma = np.linalg.solve(G, e1)
            except np.linalg.LinAlgError:
                gamma = e1
            a_new = np.zeros(n)
            for i in range(m_act):
                a_new += gamma[i] * F_hist[base + i]
            a = normalize_simplex(a_new)
        X_hist.append(a.copy())
        if len(X_hist) > m + 2:
            X_hist.pop(0)
            F_hist.pop(0)
        obj_now = compute_obj(a)
        if obj_now < best_obj - 1e-14:
            best_obj = obj_now
            best_a = a.copy()
            if verbose:
                print(f"    AA-LP step {step}: obj={best_obj:.10f}, res={residual:.6f}")
        if residual < 1e-8:
            if verbose:
                print(f"    AA-LP converged at step {step}, residual={residual:.2e}")
            break
    return best_a, best_obj

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize_simplex(a_new)


def make_init(n, mode, rng=None):
    """Create diverse initial sequences of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.0001))
        return a / a.sum()
    elif mode == 3:
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 4:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, np.pi, n)
        a = np.maximum(0.5 + 0.5 * np.cos(x), 0.001)
        return a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [2, 3, 5, 7]:
            a += 0.15 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 4))
        return a / a.sum()
    elif mode == 8:
        x = np.linspace(0, 1, n)
        a = np.where((x > 0.2) & (x < 0.8), 1.0, np.maximum(0.3 - np.abs(x - 0.5), 0.01))
        return a / a.sum()
    elif mode == 9:
        # Power law taper
        x = np.arange(n) / (n - 1)
        a = (1 - np.abs(2 * x - 1)) ** 2.5 + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def high_tau_gd(a, time_budget, verbose=False):
    """Very high tau GD for final polishing."""
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    tau_levels = [500, 2000, 5000, 15000]
    for tau in tau_levels:
        lr = 0.8 / np.sqrt(n)
        v = np.zeros(n)
        it_level = 0
        while time.monotonic() - start < time_budget:
            if it_level >= 2000:
                break
            y = a + 0.8 * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                continue
            y = y / sy
            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-14:
                break
            # Backtracking
            trial_lr = lr
            a_new = y.copy()
            best_trial = compute_obj(y)
            improved = False
            for _ in range(10):
                v_new = 0.8 * v - trial_lr * g
                cand = np.maximum(y + v_new, 0)
                sc = float(np.sum(cand))
                if sc < 1e-10:
                    break
                cand = cand / sc
                obj_c = compute_obj(cand)
                if obj_c < best_trial - 1e-15:
                    a_new = cand
                    v = v_new
                    best_trial = obj_c
                    improved = True
                    break
                trial_lr *= 0.4
            if not improved:
                a_new = y
                v = 0.8 * v - 0.01 * lr * g
            a = a_new
            it_level += 1
            obj_now = compute_obj(a)
            if obj_now < best_obj - 1e-15:
                best_obj = obj_now
                best_a = a.copy()
            lr = max(1e-9, lr * 0.9998)
        if verbose:
            print(f"    HiTau GD tau={tau:5d}: obj={best_obj:.10f}")
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining Anderson-accelerated LP,
    high-tau GD with line search, active-set LP, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 1000
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.10f}")

    n_current = len(best_seq)
    print(f"Starting: n={n_current}, obj={best_score:.10f}")

    # === Phase 1: Quick diverse screening (40s) ===
    print("\n=== Phase 1: Diverse Init Screening ===")
    alloc_total = min(40, (deadline - time.monotonic()) * 0.05)
    init_modes = [0, 2, 5, 6, 9]
    for mode in init_modes:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        a_init = make_init(n_current, mode, rng)
        alloc = min(alloc_total / len(init_modes), 8.0)
        a_screen = gd_line_search(a_init, time_budget=alloc,
                                  tau_levels=[20, 100, 500], verbose=False)
        obj_screen = compute_obj(a_screen)
        if obj_screen < best_score:
            best_score = obj_screen
            best_seq = a_screen.copy()
            print(f"  Init mode {mode}: obj={best_score:.10f} **")
        else:
            print(f"  Init mode {mode}: obj={obj_screen:.10f}")
    print(f"  After screening: best={best_score:.10f}")

    # === Phase 2: Initial GD to refine (150s) ===
    print("\n=== Phase 2: Initial GD Refinement ===")
    remaining = deadline - time.monotonic()
    alloc = min(150, remaining * 0.15)
    if alloc > 30:
        a2 = gd_line_search(best_seq, time_budget=alloc,
                            tau_levels=[10, 30, 100, 300, 800], verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 GD: obj={best_score:.10f} **")

    # === Phase 3: Anderson-Accelerated LP (main workhorse, 500s) ===
    print("\n=== Phase 3: Anderson-Accelerated LP ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    if n_cur <= 3000:
        lp_indices = build_lp_data(n_cur)
        alloc = min(500, remaining * 0.45)
        if alloc > 30:
            a3, obj3 = anderson_lp(best_seq, time_budget=alloc, m=5,
                                    verbose=True, lp_indices=lp_indices)
            if obj3 < best_score:
                best_score = obj3
                best_seq = a3.copy()
                print(f"  AA-LP: obj={best_score:.10f} **")

    # === Phase 4: High-tau GD polish (200s) ===
    print("\n=== Phase 4: High-Tau GD Polish ===")
    remaining = deadline - time.monotonic()
    alloc = min(200, remaining * 0.2)
    if alloc > 20:
        a4 = high_tau_gd(best_seq, time_budget=alloc, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  HiTau GD: obj={best_score:.10f} **")

    # === Phase 5: Active-set LP refinement (200s) ===
    print("\n=== Phase 5: Active-Set LP ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    if n_cur <= 3000:
        lp_indices = build_lp_data(n_cur)
        alloc = min(200, remaining * 0.2)
        if alloc > 20:
            a5, obj5 = active_set_lp(best_seq, time_budget=alloc,
                                      verbose=True, lp_indices=lp_indices)
            if obj5 < best_score:
                best_score = obj5
                best_seq = a5.copy()
                print(f"  ActLP: obj={best_score:.10f} **")

    # === Phase 6: Multi-n exploration (200s) ===
    print("\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    n_options = sorted(set([
        n_cur, n_cur - 100, n_cur + 100, n_cur + 200,
        n_cur + 400, 800, 1000, 1200, 1500, 2000
    ]))
    n_options = [n for n in n_options if 500 <= n <= 2500]
    n_options.sort(key=lambda x: abs(x - n_cur))
    n_options = n_options[:5]
    print(f"  n candidates: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try == len(best_seq):
            continue
        a_try = resample_seq(best_seq, n_try)
        # Quick GD
        alloc1 = min(remaining * 0.3, 40.0)
        if alloc1 > 10:
            a_try = gd_line_search(a_try, time_budget=alloc1,
                                    tau_levels=[20, 100, 400], verbose=False)
        # Quick LP if small enough
        remaining = deadline - time.monotonic()
        if n_try <= 2000 and remaining > 30:
            lp_idx = build_lp_data(n_try)
            a_try2, obj_try2 = anderson_lp(a_try, time_budget=min(remaining * 0.3, 40.0),
                                            m=5, verbose=False, lp_indices=lp_idx)
            if obj_try2 < compute_obj(a_try):
                a_try = a_try2
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 7: Final intensive polish (100s) ===
    print("\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    # More AA-LP
    if n_cur <= 3000 and remaining > 40:
        lp_indices = build_lp_data(n_cur)
        alloc = min(remaining * 0.4, 60.0)
        a7, obj7 = anderson_lp(best_seq, time_budget=alloc, m=5,
                                verbose=True, lp_indices=lp_indices)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Final AA-LP: obj={best_score:.10f} **")

    # Ultra high-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 50.0)
        a8 = high_tau_gd(best_seq, time_budget=alloc, verbose=False)
        obj8 = compute_obj(a8)
        if obj8 < best_score:
            best_score = obj8
            best_seq = a8.copy()
            print(f"  Ultra GD: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
