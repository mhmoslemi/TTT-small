import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(a, M_target=None, warm_start=None):
    """
    Solve LP: min M s.t. A_a @ g <= M, sum(g)=1, g>=0.
    Where A_a[k,j] = a[k-j] (valid indices).
    Returns (g, M_value) or (None, None).
    """
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    
    if warm_start is not None:
        g.value = warm_start
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.GLPK]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def lp_alternation(a, time_budget=300.0, verbose=False, seed=42):

    LP alternation: alternate between fixing a and optimizing g,

    best_obj = compute_obj(a)
    start = time.monotonic()

    it = 0
    no_improve = 0
    warm_g = a.copy()

    while time.monotonic() - start < time_budget:
        # Solve LP with current a as the fixed factor
        g_lp, M_lp = solve_lp_direction(a, warm_start=warm_g)
        if g_lp is None:
            no_improve += 1
            if no_improve >= 5:
                break
            a = best_a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            continue

        warm_g = g_lp.copy()

def coordinate_descent(a, time_budget=80.0, verbose=False):

    Actually, for a fixed a[i], the contribution to conv[k] is:
    - conv[k] gets a[i]*a[k-i] added (from j=i term)
    - conv[k] gets a[i]*a[k-i] added (from k-j=i term, i.e., j=k-i)
    - If k=2i: conv[k] gets a[i]^2 added
    - If k=i: conv[k] gets a[i]*a[0]... 
    
    This is complex. Let's use a simpler approach: for each coordinate i,
    try a few candidate values and pick the best.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    it = 0

    while time.monotonic() - start < time_budget:
        improved = False
        # Random order for diversity
        order = rng.permutation(n) if it % 3 == 0 else np.arange(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            # Try scaling a[i] by various factors
            # After changing a[i], renormalize
            candidates = [old_val * f for f in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0]]
            candidates.append(0.0)
            
            best_candidate = old_val
            best_candidate_obj = best_obj
            
            for cv in candidates:
                a[i] = cv
                s_new = a.sum()
                if s_new < 1e-10:
                    a[i] = old_val
                    continue
                a_norm = a / s_new
                obj_candidate = compute_obj(a_norm)
                if obj_candidate < best_candidate_obj - 1e-13:
                    best_candidate_obj = obj_candidate
                    best_candidate = cv
            
            if best_candidate != old_val:
                a[i] = best_candidate
                s_new = a.sum()
                if s_new > 1e-10:
                    a = a / s_new
                best_obj = best_candidate_obj
                best_a = a.copy()
                improved = True
        
        it += 1
        if not improved and it > 5:
            break
        if verbose and it % 3 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a

def smoothed_gradient_descent(a, time_budget=80.0, tau_init=15.0, tau_end=300.0):
    """Smoothed gradient descent with softmax smoothing of the max."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 3.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        exp_sum = float(np.sum(exp_vals))
        if exp_sum < 1e-300:
            p = np.zeros(2*n-1)
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / exp_sum

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        # Gradient of sum_k p[k]*conv[k] wrt a
        # conv[k] = sum_j a[j]*a[k-j]
        # d/d(a[i]) sum_k p[k]*conv[k] = sum_k p[k]*(a[k-i] + a[i-k] (if valid))
        # = (p * a_reversed)[i] + (p * a)[i] convolved appropriately
        # Using FFT: the gradient is ifft(fft(p) * conj(fft(a))) * 2 (approximately)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr = max(1e-10, lr * 0.999)
        it += 1

    return best_a

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

def solve_lp_bisect(a, M_target, warm_start=None):
    """Check feasibility of LP with M <= M_target."""
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M_target, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(0), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                if g.value is not None:
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
                return None
            elif prob.status == 'infeasible':
                return 'infeasible'
        except Exception:
            continue
    return None

"""
Multi-strategy optimizer for minimum peak autocorrelation sequences.
Uses LP alternation with warm-starting, water-filling inspired peak targeting,
coordinate descent, smoothed gradient, and multi-length exploration.
Key improvements: larger n (1500-2000), aggressive peak targeting,
warm-started LP with fine line search, and diverse perturbation strategies.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(a, warm_start=None):
    """
    Solve LP: min M s.t. A_a @ g <= M, sum(g)=1, g>=0.
    Where A_a[k,j] = a[k-j] (valid indices).
    Returns (g, M_value) or (None, None).
    """
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    if warm_start is not None:
        g.value = np.asarray(warm_start, dtype=np.float64)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.GLPK]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternation_fast(a, time_budget=200.0, verbose=False, seed=42):
    """
    Fast LP alternation with warm-starting and adaptive fine line search.
    The key improvement: very fine line search near the optimal step,
    and warm-starting the LP solver for speed.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0
    warm_g = a.copy()
    last_best_t = 0.3

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_direction(a, warm_start=warm_g)
        if g_lp is None:
            no_improve += 1
            if no_improve >= 4:
                scale = 0.004 + 0.003 * min(no_improve, 8)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                warm_g = a.copy()
                no_improve = 0
                if it > 80:
                    break
            it += 1
            continue

        warm_g = g_lp.copy()

        # Fine line search centered around last best step
        best_t = 0
        best_t_obj = best_obj

        # Coarse search
        coarse_ts = [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for t_val in coarse_ts:
            a_try = (1 - t_val) * best_a + t_val * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-14:
                best_t_obj = obj_try
                best_t = t_val

        # Fine search around best coarse t
        if best_t > 0:
            fine_ts = [best_t * 0.7, best_t * 0.85, best_t * 1.0,
                       best_t * 1.15, best_t * 1.3, best_t * 1.5]
            for t_val in fine_ts:
                if t_val < 0 or t_val > 2.0:
                    continue
                a_try = (1 - t_val) * best_a + t_val * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj - 1e-14:
                    best_t_obj = obj_try
                    best_t = t_val

        if best_t > 0 and best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            last_best_t = best_t
            best_a = (1 - best_t) * best_a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            s_ba = best_a.sum()
            if s_ba > 1e-10:
                best_a = best_a / s_ba
            a = best_a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                scale = 0.003 + 0.002 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                warm_g = a.copy()
                no_improve = 0
                if it > 100:
                    break

        it += 1
        if verbose and it % 15 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def water_filling_subgradient(a, time_budget=100.0, K=12, verbose=False):
    """
    Water-filling inspired subgradient: target the top K peaks and
    push them down towards the mean correlation level.
    Uses the exact gradient of the max-correlation term.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 2.0 / np.sqrt(n)
    momentum = 0.82
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        mean_conv = float(np.mean(conv))

        # Target: push peaks towards the mean (water-filling)
        # Weight by how far above mean each peak is
        excess = np.maximum(conv - mean_conv, 0)
        top_K = np.argsort(excess)[::-1][:K]
        top_excess = excess[top_K]

        if top_excess[0] < 1e-15:
            break

        # Softmax weighting over top peaks
        w_raw = top_excess ** 2
        if w_raw.sum() < 1e-30:
            w = np.ones(K) / K
        else:
            w = w_raw / w_raw.sum()

        # Compute gradient: d(conv[k])/d(a[i]) for each top peak
        g = np.zeros(n)
        for wi, k in zip(w, top_K):
            k = int(k)
            # d(conv[k])/d(a[i]) = a[k-i] if 0 <= k-i < n
            # (using subgradient; true gradient would be 2*a[k-i] except at i=k/2)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                g[idx] += wi * a[k - idx]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 150:
                lr *= 0.4
                v *= 0.2
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.997

        it += 1

    return best_a


def smoothed_gradient_descent(a, time_budget=80.0, tau_init=12.0, tau_end=350.0):
    """Smoothed gradient descent with softmax-approximated max."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 4.0 / np.sqrt(n)
    momentum = 0.82
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        exp_sum = float(np.sum(exp_vals))
        if exp_sum < 1e-300:
            p = np.zeros(2 * n - 1)
            p[int(np.argmax(conv))] = 1.0
        else:
            p = exp_vals / exp_sum

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr = max(1e-10, lr * 0.9993)
        it += 1

    return best_a


def coordinate_descent(a, time_budget=80.0, verbose=False):
    """
    Coordinate descent: for each coordinate, try multiplicative perturbations
    and keep the best. Uses random restarts within the budget.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    sweep = 0

    while time.monotonic() - start < time_budget:
        improved = False
        if sweep % 4 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            old_val = a[i]
            if old_val < 1e-14:
                continue

            # Try a few multiplicative factors
            factors = [0.4, 0.7, 1.4, 1.8]
            best_factor = 1.0
            best_f_obj = best_obj

            for f in factors:
                a[i] = old_val * f
                s_new = a.sum()
                if s_new < 1e-10:
                    a[i] = old_val
                    continue
                a_norm = a / s_new
                obj_f = compute_obj(a_norm)
                if obj_f < best_f_obj - 1e-13:
                    best_f_obj = obj_f
                    best_factor = f

            if best_factor != 1.0:
                a[i] = old_val * best_factor
                s_new = a.sum()
                if s_new > 1e-10:
                    a = a / s_new
                best_obj = best_f_obj
                best_a = a.copy()
                improved = True

        sweep += 1
        if not improved and sweep > 3:
            break
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep{sweep}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-strategy optimization for minimum peak
    autocorrelation. Uses LP alternation (warm-started), water-filling
    subgradient, smoothed GD, coordinate descent, and multi-length exploration.
    Targets n=1200-2000 for more degrees of freedom.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_target = 1500
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Multi-length exploration with quick optimization ===
    print("\n=== Phase 1: Multi-length Exploration ===")
    n_options = sorted(set([1000, 1200, 1500, 1800, 2000]))

    # Sort by proximity to current best
    n_options_sorted = sorted(n_options, key=lambda x: abs(x - n_current))

    for n_try in n_options_sorted[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        print(f"  Exploring n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)

        # Quick smoothed GD
        alloc1 = min(remaining * 0.06, 30.0)
        if alloc1 > 10:
            a_try = smoothed_gradient_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=200.0)

        # Quick LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.08, 50.0)
        if alloc2 > 15:
            a_try = lp_alternation_fast(a_try, time_budget=alloc2, seed=n_try)

        obj_try = compute_obj(a_try)
        print(f"  n={n_try}: obj={obj_try:.8f}")

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** New best at n={n_try}: {best_score:.8f}")

    # === Phase 2: Deep LP alternation (primary optimizer) ===
    print(f"\n=== Phase 2: Deep LP Alternation (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.28, 300.0)
    if p2_budget > 30:
        a_lp = lp_alternation_fast(best_seq, time_budget=p2_budget,
                                   verbose=True, seed=42)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f}")

    # === Phase 3: Water-filling subgradient ===
    print(f"\n=== Phase 3: Water-Filling Subgradient ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 120.0)
    if p3_budget > 15:
        a_wf = water_filling_subgradient(best_seq, time_budget=p3_budget, K=12)
        obj_wf = compute_obj(a_wf)
        if obj_wf < best_score:
            best_score = obj_wf
            best_seq = a_wf.copy()
            print(f"  Phase3 WF: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 WF: obj={obj_wf:.8f}")

    # === Phase 4: LP + WF alternating rounds ===
    print(f"\n=== Phase 4: LP + WF Alternating ===")
    for round_idx in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        # LP round
        alloc_lp = min(remaining * 0.18, 80.0)
        if alloc_lp > 15:
            a_r = lp_alternation_fast(best_seq, time_budget=alloc_lp,
                                      seed=200 + round_idx)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  R{round_idx} LP: obj={best_score:.8f} **")

        # WF round
        remaining = deadline - time.monotonic()
        alloc_wf = min(remaining * 0.12, 50.0)
        if alloc_wf > 10:
            a_r2 = water_filling_subgradient(best_seq, time_budget=alloc_wf, K=10)
            obj_r2 = compute_obj(a_r2)
            if obj_r2 < best_score:
                best_score = obj_r2
                best_seq = a_r2.copy()
                print(f"  R{round_idx} WF: obj={best_score:.8f} **")

    # === Phase 5: Coordinate descent refinement ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.12, 100.0)
    if p5_budget > 20:
        a_cd = coordinate_descent(best_seq, time_budget=p5_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase5 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 CD: obj={obj_cd:.8f}")

    # === Phase 6: Diversified perturbation + optimization ===
    print(f"\n=== Phase 6: Diversified Perturbation ===")
    remaining = deadline - time.monotonic()

    perturb_strategies = [
        (0.004, "small noise"),
        (0.010, "medium noise"),
        (0.020, "large noise"),
        (0.040, "very large"),
    ]

    for r_idx, (scale, desc) in enumerate(perturb_strategies):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        alloc = min(remaining * 0.18, 70.0)
        if alloc > 15:
            a_pert = lp_alternation_fast(a_pert, time_budget=alloc,
                                         seed=600 + r_idx)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Pert{r_idx} ({desc}): obj={best_score:.8f} **")
            else:
                print(f"  Pert{r_idx} ({desc}): obj={obj_pert:.8f}")

    # === Phase 7: Try different length for final push ===
    print(f"\n=== Phase 7: Final Length Adjustment ===")
    remaining = deadline - time.monotonic()
    n_alt = len(best_seq) + 200
    if remaining > 50:
        a_alt = resample(best_seq, n_alt)
        alloc_alt = min(remaining * 0.2, 60.0)
        if alloc_alt > 15:
            a_alt = lp_alternation_fast(a_alt, time_budget=alloc_alt, seed=1234)
            obj_alt = compute_obj(a_alt)
            if obj_alt < best_score:
                best_score = obj_alt
                best_seq = a_alt.copy()
                print(f"  Alt n={n_alt}: obj={best_score:.8f} **")
            else:
                print(f"  Alt n={n_alt}: obj={obj_alt:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.35, 80.0)
        a_final = lp_alternation_fast(best_seq, time_budget=alloc, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 40.0)
        a_final2 = coordinate_descent(best_seq, time_budget=alloc)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
