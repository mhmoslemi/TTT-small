"""
Multi-strategy optimizer for minimum peak autocorrelation of non-negative sequences.
Uses FFT-based subgradient with Nesterov momentum, LP direction finding,
multi-length exploration, and crossover.
"""

import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, p, N):
    """
    Compute the softmax-weighted subgradient of max_k (a*a)[k].
    g_j = 2 * sum_i a_i * p_{i+j} (cross-correlation of a with p)
    Uses FFT for efficiency.
    """
    # Cross-correlation: (a, p) -> g where g_j = sum_i a_i * p_{i+j}
    # This is ifft(fft(a) * conj(fft(p)))
    fa = fft(a, N)
    fp = fft(p, N)
    g_full = ifft(fa * np.conj(fp)).real
    return 2.0 * g_full[:len(a)]

def subgradient_optimize(a, time_budget=100.0, beta_init=5.0, beta_final=200.0,
                         verbose=False, seed=42):
    """
    Subgradient method with Nesterov momentum and adaptive beta.
    Optimizes 2n * max_k (a*a)[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # FFT size for cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    
    lr = 1.0 / np.sqrt(n)
    momentum = 0.9
    v = np.zeros(n)
    beta = beta_init
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_init * (beta_final / beta_init) ** (progress ** 0.7)
        
        # Compute convolution
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Compute softmax weights
        # Shift to avoid overflow
        shifted = conv - max_conv
        p = np.exp(beta * shifted)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros_like(conv)
            p[:n] = 1.0 / n  # fallback
        else:
            p = p / p_sum
        
        # Compute subgradient
        g = compute_subgrad(a, p, N)
        
        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        
        # Adaptive step size with backtracking
        best_t = lr
        best_val = best_obj
        for t_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            t = lr * t_mult / max(1.0, v_norm * 0.1)
            a_try = a - t * v / max(v_norm, 1e-10) * 10.0  # normalized step
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_val:
                best_val = obj_try
                best_t = t / max(v_norm, 1e-10) * 10.0
        
        # Simpler approach: just use a fixed step with the gradient
        step = lr / max(v_norm, 1.0)
        a_new = a - step * v
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.01, 5.0 / np.sqrt(n))
        else:
            a = a_new if obj_new < best_obj else a
            stall += 1
            lr *= 0.95
            if stall > 100:
                # Perturb
                scale = 0.01 + 0.005 * min(stall // 100, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_p = a.sum()
                if s_p > 1e-10:
                    a = a / s_p
                v = np.zeros(n)
                lr = 1.0 / np.sqrt(n)
                stall = 0
                if it > 200:
                    break
        
        it += 1
        if verbose and it % 20 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.4f}, t={elapsed:.0f}s")
    
    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_direction(a, time_limit=10.0):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_value) or (None, None).
    """
    import cvxpy as cp
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def build_conv_matrix(a):

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)

    if s < 1e-10:
        return a
    a = a / s

    no_improve = 0

    while time.monotonic() - start < time_budget:

        if g_lp is None:
            break

        best_t = 0
        best_t_obj = best_obj

            a_try = (1 - t_val) * best_a + t_val * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:

            if obj_try < best_t_obj - 1e-15:

        if best_t > 0 and best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = (1 - best_t) * best_a + best_t * g_lp

            s_ba = best_a.sum()

                no_improve = 0

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

import numpy as np

from scipy.fft import fft, ifft, rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, p, N):
    """
    Compute softmax-weighted subgradient of max_k (a*a)[k].
    g_j = 2 * sum_i a_i * p_{i+j} (cross-correlation of a with p).
    Uses FFT for O(n log n) computation.
    """
    fa = fft(a, N)
    fp = fft(p, N)
    g_full = ifft(fa * np.conj(fp)).real
    return 2.0 * g_full[:len(a)]

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_value) or (None, None).
    """
    import cvxpy as cp
    n = len(a)
    a = a / a.sum()
    A_mat = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def resample(a, new_n):

    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0.0)

            print(f"  n={n_try}: obj={obj_try:.8f}")

        # LP polish for shorter sequences
        remaining = deadline - time.monotonic()
        if remaining > 25 and n_try <= 1000:
            alloc2 = min(remaining * 0.08, 35.0)
            if alloc2 > 12:
                a_try2 = lp_polish(a_try, time_budget=alloc2, seed=200 + n_try)
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_score:
                    best_score = obj_try2
                    best_seq = a_try2.copy()
                    all_solutions.append(a_try2.copy())
                    all_scores.append(obj_try2)
                    n_current = n_try
                    print(f"  ** n={n_try} LP: obj={best_score:.8f} **")
                else:
                    print(f"  n={n_try} LP: obj={obj_try2:.8f}")

    # === Phase 4: Crossover (12% time) ===
    print(f"\n=== Phase 4: Crossover ===")
    if len(all_solutions) >= 2:
        idx_sorted = np.argsort(all_scores)
        top_k = min(4, len(all_solutions))
        top_idx = [idx_sorted[i] for i in range(top_k)]

        remaining = deadline - time.monotonic()
        for i in range(len(top_idx)):
            for j in range(i + 1, len(top_idx)):
                remaining = deadline - time.monotonic()
                if remaining < 20:
                    break

                a1 = all_solutions[top_idx[i]]
                a2 = all_solutions[top_idx[j]]
                if len(a1) != len(a2):
                    a2 = resample(a2, len(a1))

                for alpha in [0.3, 0.5, 0.7]:
                    remaining = deadline - time.monotonic()
                    if remaining < 15:
                        break
                    a_mix = alpha * a1 + (1.0 - alpha) * a2
                    a_mix = np.maximum(a_mix, 0)
                    s_mix = a_mix.sum()
                    if s_mix < 1e-10:
                        continue
                    a_mix = a_mix / s_mix

                    alloc = min(remaining * 0.12, 30.0)
                    if alloc > 8:
                        a_mix = subgradient_optimize(a_mix, time_budget=alloc,
                                                     beta_init=10.0, beta_final=200.0,
                                                     seed=500 + i * 10 + j)
                    obj_mix = compute_obj(a_mix)
                    if obj_mix < best_score:
                        best_score = obj_mix
                        best_seq = a_mix.copy()
                        all_solutions.append(a_mix.copy())
                        all_scores.append(obj_mix)
                        print(f"  ** Cross {i}x{j} a={alpha}: obj={best_score:.8f} **")
                    else:
                        print(f"  Cross {i}x{j} a={alpha}: obj={obj_mix:.8f}")

    # === Phase 5: Perturbation search (15% time) ===
    print(f"\n=== Phase 5: Perturbation ===")
    for r_idx in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        scale = 0.004 + 0.004 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            break
        a_pert = a_pert / s_pert

        alloc = min(remaining * 0.12, 50.0)
        if alloc > 10:
            a_pert = subgradient_optimize(a_pert, time_budget=alloc,
                                          beta_init=8.0, beta_final=300.0,
                                          seed=1000 + r_idx)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                all_solutions.append(a_pert.copy())
                all_scores.append(obj_pert)
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 6: Final intensive polish (11% time) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.5, 80.0)
        a_final = subgradient_optimize(best_seq, time_budget=alloc,
                                       beta_init=15.0, beta_final=600.0,
                                       lr_init=2.0 / np.sqrt(len(best_seq)),
                                       seed=9999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final SG: obj={best_score:.8f} **")
        else:
            print(f"  Final SG: obj={obj_final:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15 and len(best_seq) <= 1000:
        alloc = min(remaining * 0.4, 40.0)
        a_final_lp = lp_polish(best_seq, time_budget=alloc, seed=2000)
        obj_flp = compute_obj(a_final_lp)
        if obj_flp < best_score:
            best_score = obj_flp
            best_seq = a_final_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_flp:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
