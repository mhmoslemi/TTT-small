def anderson_step(history_x, history_f, T_x, m):
    """Compute Anderson acceleration step.
    history_x: list of m previous iterates
    history_f: list of m previous residuals (x - T(x))
    T_x: current T(x)
    m: order of Anderson acceleration
    """
    if len(history_x) < m:
        return T_x
    
    # Use last m points
    xs = np.array(history_x[-m:])  # (m, n)
    fs = np.array(history_f[-m:])  # (m, n)
    
    # Compute B (Jacobian approximation): B[i,:j] = (f[i] - f[0]) / (x[i] - x[0])
    # Better: use the "deep" Anderson formulation
    # G[i,j] = (x[i+1] - x[i]) . (f[i+1] - f[i]) for i=0..m-2
    # This gives a (m-1) x (m-1) system
    
    # Simpler: solve the "broyden" system
    # Let d_i = x_{i+1} - x_i, e_i = f_{i+1} - f_i for i=0..m-2
    # Solve: sum_i alpha_i * e_i = -f_m
    # Minimize: sum_i alpha_i^2
    # => min ||alpha|| s.t. E @ alpha = -f_m
    # where E is (n x (m-1))
    
    if m < 2:
        return T_x
    
    d = np.diff(xs, axis=0)  # (m-1, n)
    e = np.diff(fs, axis=0)  # (m-1, n)
    
    # Solve minimum-norm solution of E @ alpha = -f_m
    # E is (n, m-1), f_m is (n,)
    E = e.T  # (n, m-1)
    rhs = -fs[-1]  # (n,)
    
    # Minimum norm: alpha = E^T (E E^T)^{-1} rhs
    EE = E @ E.T  # (m-1, m-1)
    try:
        EE_inv_rhs = np.linalg.solve(EE + 1e-10 * np.eye(m-1), E.T @ rhs)
        alpha = E.T @ EE_inv_rhs  # (m-1,)
    except:
        return T_x
    
    # New iterate: x_new = x_m + sum_i alpha_i * d_i
    x_new = xs[-1] + np.sum(alpha[:, None] * d, axis=0)
    
    # Clip to feasible region (or just use as is and project later)
    return x_new

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Aggressive tau schedule: stay high longer
        tau = tau_start + (tau_end - tau_start) * progress ** 0.7

        # Nesterov look-ahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        shifted = tau * (conv - max_conv)
        # Clip for numerical stability
        shifted = np.clip(shifted, -700, 0)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient via FFT: grad[j] = 2 * sum_k p[k] * y[k-j] (cross conv)
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        Y = fft(y, N)
        g_full = ifft(P * np.conj(Y)).real
        g = 2.0 * g_full[:n]

        # Adaptive step size
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        lr = min(1.0 / (g_norm + 1e-10), 3.0 / np.sqrt(n))
        lr = max(lr, 1e-10)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        # Nesterov optimal schedule
        beta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        beta = min(beta, 1.0)

        # Reset if stuck
        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            beta = 0.9
            best_it = it
            if it > 100000:
                break

        if verbose and it % 10000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6e}, t={elapsed:.0f}s")

    return best_a


def log_space_optimize(a, time_budget, tau_min=500, tau_max=20000, verbose=False):
    """Optimize on log-space parameterization for better handling of small entries."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Identify support
    support = np.where(a > 1e-8)[0]
    if len(support) < 2:
        support = np.where(a > 0)[0]
    if len(support) < 2:
        return a

    b0 = np.log(np.maximum(a[support], 1e-30))
    b0 = b0 - b0.mean()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Multi-stage with increasing tau
    n_stages = 4
    for stage in range(n_stages):
        if time.monotonic() - start > time_budget:
            break
        tau = tau_min + (tau_max - tau_min) * (stage + 1) / n_stages
        stage_budget = (time_budget / n_stages) * 0.9

        def obj_grad(b_vec):
            b_shifted = b_vec - b_vec.max()
            a_supp = np.exp(b_shifted)
            a_supp = a_supp / a_supp.sum()
            a_full = np.zeros(n)
            a_full[support] = a_supp
            conv = compute_conv(a_full)
            max_c = float(np.max(conv))
            shifted = np.clip(tau * (conv - max_c), -700, 0)
            exp_v = np.exp(shifted)
            sum_e = float(np.sum(exp_v))
            if sum_e < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_v / sum_e
            obj_val = max_c + np.log(sum_e) / tau
            # Gradient
            N = 1
            while N < 3 * n:
                N <<= 1
            P = fft(p, N)
            A_f = fft(a_full, N)
            g_full = ifft(P * np.conj(A_f)).real
            grad_full = 2.0 * g_full[:n]
            # Chain rule for softmax parameterization
            grad_b = a_supp * (grad_full[support] - np.dot(a_supp, grad_full[support]))
            return obj_val, grad_b

        stage_start = time.monotonic()
        max_iter = int(stage_budget / 0.05)  # estimate max iterations
        max_iter = min(max_iter, 500)
        try:
            res = scipy_minimize(obj_grad, b0, method='L-BFGS-B',
                                 jac=True, options={'maxiter': max_iter,
                                                    'gtol': 1e-12, 'ftol': 1e-15})
            if res.x is not None and time.monotonic() - stage_start < stage_budget:
                b_shifted = res.x - res.x.max()
                a_supp = np.exp(b_shifted)
                a_supp = a_supp / a_supp.sum()
                a_new = np.zeros(n)
                a_new[support] = a_supp
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    b0 = res.x.copy()
                    if verbose:
                        print(f"    LogOpt stage{stage}: tau={tau:.0f}, obj={best_obj:.8f}")
        except Exception:
            pass

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.5
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
        a = a / a.sum()
    elif mode == 5:
        # Raised cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        a = a / a.sum()
    elif mode == 6:
        # Gaussian-like
        x = np.linspace(-2, 2, n)
        a = np.exp(-x ** 2)
        a = a / a.sum()
    elif mode == 7:
        # Two-hump
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) / 0.15) ** 2) + np.exp(-((x - 0.7) / 0.15) ** 2)
        a = a / a.sum()
    elif mode == 8:
        # Triangle
        x = np.linspace(0, 1, n)
        a = 1 - 2 * np.abs(x - 0.5)
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    return a


def support_refine(a, time_budget, verbose=False):
    """Try reducing support and re-optimizing."""
    a = normalize(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    sorted_vals = np.sort(a)[::-1]
    for frac in [0.4, 0.5, 0.6, 0.7, 0.8, 0.9]:
        if time.monotonic() - start > time_budget:
            break
        k = max(2, int(len(a) * frac))
        threshold = sorted_vals[min(k, len(sorted_vals) - 1)]
        support = np.where(a >= threshold)[0]
        if len(support) < 2:
            continue
        a_shrunk = np.zeros(len(a))
        a_shrunk[support] = a[support]
        a_shrunk = normalize(a_shrunk)

        alloc = min(30, time_budget - (time.monotonic() - start) - 5)
        if alloc > 5:
            a_shrunk = nesterov_gd(a_shrunk, time_budget=alloc,
                                    tau_start=500, tau_end=10000, verbose=False)

        obj_shrunk = compute_obj(a_shrunk)
        if obj_shrunk < best_obj - 1e-14:
            best_obj = obj_shrunk
            best_a = a_shrunk.copy()
            if verbose:
                print(f"    Shrink frac={frac}: obj={best_obj:.8f}, support={len(support)}")

    return best_a


def perturbation_escape(a, best_obj, rng, time_budget, verbose=False):
    """Try random perturbations to escape local optima."""
    n = len(a)
    best_a = a.copy()
    start = time.monotonic()

    for attempt in range(10):
        if time.monotonic() - start > time_budget:
            break
        # Random perturbation
        scale = 0.05 + 0.03 * attempt
        noise = rng.normal(0, scale, n)
        a_pert = a + noise
        a_pert = normalize(a_pert)

        # Quick optimization
        alloc = min(30, time_budget - (time.monotonic() - start) - 10)
        if alloc < 5:
            break
        a_pert = nesterov_gd(a_pert, time_budget=alloc,
                             tau_start=200, tau_end=5000, verbose=False)
        obj_pert = compute_obj(a_pert)
        if obj_pert < best_obj - 1e-14:
            best_obj = obj_pert
            best_a = a_pert.copy()
            if verbose:
                print(f"    Perturb {attempt}: obj={best_obj:.8f} **")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP fixed-point, Nesterov GD, multi-n exploration,
    perturbation escape, and ultra-precise final polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0  # safety margin

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent loaded (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = diverse_init(n_init, rng, mode=8)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting with n={n_best}, obj={best_score:.8f}")

    # === Phase 1: Extended LP Fixed-Point (300s) ===
    print(f"\n=== Phase 1: LP Fixed-Point (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.22, 300.0)
    if p1_budget > 30 and n_best <= 1500:
        a_lp = lp_fixed_point(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-14:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: High-Tau Nesterov GD (250s) ===
    print(f"\n=== Phase 2: Nesterov GD (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.17, 250.0)
    if p2_budget > 20:
        a_gd = nesterov_gd(best_seq, time_budget=p2_budget,
                           tau_start=500, tau_end=80000, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score - 1e-14:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase2 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 GD: obj={obj_gd:.8f}")

    # === Phase 3: Log-space optimization (100s) ===
    print(f"\n=== Phase 3: Log-space Opt (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.08, 100.0)
    if p3_budget > 20:
        a_log = log_space_optimize(best_seq, time_budget=p3_budget,
                                   tau_min=500, tau_max=30000, verbose=True)
        obj_log = compute_obj(a_log)
        if obj_log < best_score - 1e-14:
            best_score = obj_log
            best_seq = a_log.copy()
            print(f"  Phase3 Log: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 Log: obj={obj_log:.8f}")

    # === Phase 4: Support Refinement (80s) ===
    print(f"\n=== Phase 4: Support Refine (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.06, 80.0)
    if p4_budget > 15:
        a_shrunk = support_refine(best_seq, time_budget=p4_budget, verbose=True)
        obj_shrunk = compute_obj(a_shrunk)
        if obj_shrunk < best_score - 1e-14:
            best_score = obj_shrunk
            best_seq = a_shrunk.copy()
            n_best = len(best_seq)
            print(f"  Phase4 Shrink: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 Shrink: obj={obj_shrunk:.8f}")

    # === Phase 5: Multi-n Exploration (350s) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.28, 350.0)

    # Generate n options centered around current best
    base_n = n_best
    n_options = sorted(set([
        max(200, base_n - 200),
        max(200, base_n - 100),
        base_n + 100,
        base_n + 200,
        base_n + 400,
        base_n + 600,
        base_n + 1000,
        base_n + 1500,
        500, 700, 900, 1100, 1300, 1600, 2000, 2500,
    ]))
    # Filter valid and remove current
    n_options = [n for n in n_options if 200 <= n <= 5000 and n != base_n]
    # Sort by proximity to current best (try close ones first)
    n_options.sort(key=lambda x: abs(x - base_n))
    n_options = n_options[:10]  # Limit to 10
    print(f"  Trying n values: {n_options}")

    per_n_budget = p5_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_n_budget, remaining - 20)
        if alloc < 20:
            break

        print(f"  n={n_try}: budget={alloc:.0f}s")
        a_try = resample_sequence(best_seq, n_try)

        # For small n, use LP; for large n, use GD
        if n_try <= 1200 and alloc > 40:
            lp_alloc = alloc * 0.35
            a_try = lp_fixed_point(a_try, time_budget=lp_alloc, verbose=False)
            remaining = deadline - time.monotonic()
            gd_alloc = min(alloc * 0.45, remaining - 15)
            if gd_alloc > 10:
                a_try = nesterov_gd(a_try, time_budget=gd_alloc,
                                     tau_start=200, tau_end=20000, verbose=False)
        else:
            # Large n: GD only
            gd_alloc = alloc * 0.7
            a_try = nesterov_gd(a_try, time_budget=gd_alloc,
                                tau_start=100, tau_end=10000, verbose=False)
            remaining = deadline - time.monotonic()
            log_alloc = min(alloc * 0.2, remaining - 10)
            if log_alloc > 8:
                a_try = log_space_optimize(a_try, time_budget=log_alloc,
                                           tau_min=200, tau_max=10000, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score - 1e-14:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Perturbation Escape (100s) ===
    print(f"\n=== Phase 6: Perturbation Escape (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p6_budget = min(remaining * 0.08, 100.0)
    if p6_budget > 20:
        best_seq, best_score = perturbation_escape(
            best_seq, best_score, rng, time_budget=p6_budget, verbose=True)
        n_best = len(best_seq)
        print(f"  After perturbation: obj={best_score:.8f}")

    # === Phase 7: Final Intensive Polish (remaining time) ===
    print(f"\n=== Phase 7: Final Polish (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s")

    # 7a: LP polish
    if remaining > 40 and n_best <= 1500:
        alloc = min(remaining * 0.25, 150.0)
        a_final = lp_fixed_point(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-14:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # 7b: Ultra high-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 200.0)
        a_final = nesterov_gd(best_seq, time_budget=alloc,
                              tau_start=5000, tau_end=200000, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-14:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # 7c: Log-space
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 80.0)
        a_final = log_space_optimize(best_seq, time_budget=alloc,
                                     tau_min=2000, tau_max=100000, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-14:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Log: obj={best_score:.8f} **")

    # 7d: Support shrink
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.15, 40.0)
        a_final = support_refine(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-14:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Shrink: obj={best_score:.8f} **")

    # 7e: One more LP round
    remaining = deadline - time.monotonic()
    if remaining > 20 and n_best <= 1500:
        alloc = min(remaining * 0.3, 60.0)
        a_final = lp_fixed_point(best_seq, time_budget=alloc, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-14:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra LP: obj={best_score:.8f} **")

    # 7f: Final GD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 30.0)
        a_final = nesterov_gd(best_seq, time_budget=alloc,
                              tau_start=10000, tau_end=300000, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-14:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
