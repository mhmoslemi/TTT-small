        3. Project `a` onto the set `{a : a[j]=0 for j≥n, a[j]≥0 for j<n, sum a = 1}` (simple projection).

    n = len(a)
    L = 2 * n - 1
    # Find next power of 2 for FFT efficiency
    L_fft = 1
    while L_fft < L:
        L_fft <<= 1
    
    a = a.copy()
    for it in range(n_iter):
        # Pad and FFT
        a_pad = np.zeros(L_fft)
        a_pad[:n] = a
        A = fft(a_pad)
        
        # Autocorrelation
        power = np.abs(A)**2
        r = ifft(power).real[:L]  # linear autocorrelation
        
        # Clip
        r_clipped = np.minimum(r, M_target)
        
        # New power spectrum
        new_power = np.maximum(0, fft(r_clipped).real)

        a_new = a_new / s
        
        a = a_new

import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * np.max(conv)

def spectral_shaping(a, M_target, max_iter=200, tol=1e-10, verbose=False):
    """Gerchberg-Saxton style spectral shaping."""
    n = len(a)
    L = 2 * n - 1
    L_fft = 1
    while L_fft < L:
        L_fft <<= 1
    
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    prev_obj = np.inf
    for it in range(max_iter):
        a_pad = np.zeros(L_fft)
        a_pad[:n] = a
        A = fft(a_pad)
        
        power = np.abs(A) ** 2
        r = ifft(power).real[:L]
        
        # Clip autocorrelation
        r_clipped = np.minimum(r, M_target)
        
        # New power spectrum
        new_power = fft(r_clipped).real
        new_power = np.maximum(new_power, 0)
        
        # Check convergence
        delta = np.max(np.abs(r_clipped - r))
        
        # Preserve phases
        phase = np.angle(A)
        A_new = np.sqrt(new_power) * np.exp(1j * phase)
        
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        a = a_new
        
        if delta < tol:
            break
    
    return a

def bisection_spectral(a, time_budget=60, verbose=True):
    """Bisection on M_target for spectral shaping."""
    n = len(a)
    M_low = 1.0 / (2 * n - 1) * 1.001  # slightly above theoretical bound
    conv_init = compute_conv(a / a.sum())
    M_high = np.max(conv_init)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Bisection iterations
    n_bisect = min(30, max(5, int(time_budget / 5)))
    iters_per_bisect = 200
    
    for bi in range(n_bisect):
        if time.monotonic() - start > time_budget - 5:
            break
        M_mid = (M_low + M_high) / 2
        a_try = spectral_shaping(a, M_mid, max_iter=iters_per_bisect)
        obj_try = compute_obj(a_try)
        
        if obj_try < 2 * n * M_mid * 1.01:
            # Converged below target

            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
        else:
            M_low = M_mid
        
        if verbose:
            elapsed = time.monotonic() - start
            print(f"    Bisect {bi}: M_range=[{M_low:.6f}, {M_high:.6f}], obj={best_obj:.6f}, t={elapsed:.0f}s")
    
    return best_a

Actually, I want to add one more idea: **iterative reweighting** (similar to the "iterative reweighted least squares" in compressed sensing).

Idea: assign weights `w[k]` to each lag `k`, with higher weight for lags where `c[k]` is large. Then solve a weighted least-squares problem to find `a` that minimizes `sum_k w[k] * (c[k] - target)^2`.

This is a QP:
minimize `sum_k w[k] * (c[k] - target)^2`
subject to `sum a = 1`, `a ≥ 0`.

Since `c[k] = a^T B_k a`, the objective is `sum_k w[k] * (a^T B_k a - target)^2`, which is a sum of quartic terms. That's not a QP.

But we can linearize: at the current `a`, the gradient of `c[k]` w.r.t. `a` is `2 * B_k * a` (approximately). So we can use a first-order approximation:
`c[k](a + Δ) ≈ c[k](a) + 2 * (B_k * a)^T * Δ`.

Then the problem becomes:
minimize `sum_k w[k] * (c[k](a) + 2 * (B_k * a)^T * Δ - target)^2`
subject to `sum Δ + (sum a - 1) = 0`, `a + Δ ≥ 0`.

This is a QP in `Δ`, which is solvable. But it's a large QP for `n = 800`.

Alternatively, we can use gradient descent on the weighted objective:
`grad = 2 * sum_k w[k] * (c[k] - target) * B_k * a`.

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_g=None):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    g_val = None
    M_val = None
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                break
        except Exception:
            continue
    
    return g_val, M_val
