"""
Hybrid smooth (p-norm Adam) + LP polishing optimization for step function coefficients.
Uses p-norm approximation of the max-convolution objective with Adam optimizer,
followed by LP-based directional polishing with parabolic line search.
Explores multiple sequence lengths and uses diverse restarts.
"""

import numpy as np
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via numpy (uses FFT internally)."""
    return np.convolve(a, a)


def compute_score(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def pnorm_obj_grad(a, p):
    """Compute p-norm of convolution and its gradient.
    Returns (obj, grad) where obj = (sum conv_k^p)^(1/p)."""
    n = len(a)
    conv = compute_conv(a)
    conv = np.maximum(conv, 0)  # numerical safety
    conv_p = conv ** p
    S = float(np.sum(conv_p))
    if S < 1e-300:
        return 0.0, np.zeros(n)
    obj = S ** (1.0 / p)
    
    # Gradient: grad[i] = S^(1/p - 1) * sum_k conv_k^(p-1) * (a[k-i] + a[i-k])
    w = conv ** (p - 1)
    w_rev = w[::-1]
    # term1[i] = sum_k w[k]*a[k-i] = conv(a, w_rev)[2n-2-i]
    term1 = np.convolve(a, w_rev, 'full')[(2 * n - 2) - np.arange(n)]
    # term2[i] = sum_k w[k]*a[i-k] = conv(w, a)[i]
    term2 = np.convolve(w, a, 'full')[:n]
    
    scale = S ** (1.0 / p - 1.0)
    grad = scale * (term1 + term2)
    return obj, grad


def adam_pnorm_optimize(a, time_budget=200.0, p_start=20, p_end=500, verbose=False):
    """Adam optimization of p-norm approximation with p-annealing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    start = time.monotonic()
    
    # Adam parameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    
    # p-annealing schedule
    total_steps_target = 3000
    step = 0
    
    while time.monotonic() - start < time_budget:
        # Anneal p
        frac = min(1.0, step / total_steps_target)
        p = p_start * (p_end / p_start) ** frac
        
        obj, grad = pnorm_obj_grad(a, p)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 0.01:
            a_new = a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        a = a_new
        step += 1
        
        # Decay learning rate
        if step % 500 == 0:
            lr *= 0.8
        
        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    Adam step {step}: p={p:.0f}, obj={obj:.8f}, score={best_score:.8f}, t={elapsed:.1f}s")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g_lp, time_budget=3.0):
    """Find optimal blend t for (1-t)*a + t*g_lp.
    Uses parabolic + golden section hybrid."""
    def obj_at(t):
        if t <= 0:
            return compute_score(a)
        a_try = (1 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_score(a_try / s)
    
    # First, try a grid to find a good region
    candidates = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5]
    best_t = 0.0
    best_obj = obj_at(0.0)
    start = time.monotonic()
    
    for t in candidates:
        if time.monotonic() - start > time_budget * 0.4:
            break

Alternative to LP: **Projected subgradient with step size schedule**. The subgradient of `max_k conv_k` at `a` is:
- Let `K* = argmax_k conv_k` (the set of k achieving the max)
- The subgradient is the average of gradients at active k values
- For each active k: `grad_k[i] = a[k-i] + a[i-k]` (with valid indices)

So the subgradient of `max_k conv_k` is:
`subgrad[i] = (1/|K*|) * sum_{k in K*} (a[k-i] + a[i-k])`

This can be computed as a convolution!

"""
Hybrid smooth p-norm optimization + LP polishing for step function coefficient minimization.
Uses Adam with p-norm annealing for finding good basins, LP-based directional polishing
for fine-tuning, multi-length exploration, and direct subgradient as auxiliary method.
"""

import numpy as np
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a."""
    return np.convolve(a, a)


def compute_score(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def pnorm_obj_grad(a, p):
    """Compute p-norm of convolution and its gradient."""
    n = len(a)
    conv = compute_conv(a)
    conv = np.maximum(conv, 0)
    conv_p = conv ** p
    S = float(np.sum(conv_p))
    if S < 1e-300:
        return 0.0, np.zeros(n)
    obj = S ** (1.0 / p)
    w = conv ** (p - 1)
    w_rev = w[::-1]
    term1 = np.convolve(a, w_rev, 'full')[(2 * n - 2) - np.arange(n)]
    term2 = np.convolve(w, a, 'full')[:n]
    scale = S ** (1.0 / p - 1.0)
    grad = scale * (term1 + term2)
    return obj, grad


def adam_pnorm_optimize(a, time_budget=200.0, p_start=10.0, p_end=2000.0, verbose=False):
    """Adam optimization of p-norm approximation with p-annealing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    start = time.monotonic()

    lr = 0.03
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    total_steps_target = 5000

    while time.monotonic() - start < time_budget:
        frac = min(1.0, step / total_steps_target)
        p = p_start * (p_end / p_start) ** frac

        obj, grad = pnorm_obj_grad(a, p)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (step + 1))
        v_hat = v / (1.0 - beta2 ** (step + 1))

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 0.01:
            a_new = a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new
        step += 1

        if step % 1000 == 0:
            lr *= 0.85

        if verbose and step % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    Adam step {step}: p={p:.0f}, score={best_score:.8f}, t={elapsed:.1f}s")

    return best_a


def subgradient_descent(a, time_budget=30.0, verbose=False):
    """Projected subgradient descent on the max-convolution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    start = time.monotonic()
    step = 0
    lr = 0.01
    momentum = np.zeros(n)
    beta = 0.7

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))

        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]

        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            momentum = np.zeros(n)
            lr = 0.005
        else:
            a_new = a_new / s_new

        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new
        step += 1
        lr *= 0.9998

        if step % 5000 == 0 and lr < 0.001:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = 0.005

        if verbose and step % 10000 == 0:

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g_lp, time_budget=3.0):
    """Find optimal blend factor t minimizing score((1-t)*a + t*g_lp)."""
    start = time.monotonic()

    def obj_at(t):
        if t <= 0:
            return compute_score(a)
        a_try = (1.0 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_score(a_try / s)

    best_t = 0.0
    best_obj = obj_at(0.0)

    # Grid search
    grid = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5]
    for t in grid:
        if time.monotonic() - start > time_budget * 0.3:
            break
        obj_t = obj_at(t)
        if obj_t < best_obj:
            best_obj = obj_t
            best_t = t

    # Golden section refinement
    if best_t > 0:
        span = 0.15
        lo = max(0.0, best_t - span)
        hi = best_t + span
        phi = (1 + np.sqrt(5)) / 2
        gr = 2 - phi

        c = lo + gr * (hi - lo)
        d = lo + (1 - gr) * (hi - lo)
        fc = obj_at(c)
        fd = obj_at(d)

        for _ in range(20):
            if time.monotonic() - start > time_budget:
                break
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = lo + gr * (hi - lo)
                fc = obj_at(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = obj_at(d)

        t_best = c if fc < fd else d
        obj_best = min(fc, fd)
        if obj_best < best_obj:
            best_obj = obj_best
            best_t = t_best

    return best_t, best_obj

def u_shaped_init(n, sharpness=3.0):

def try_new_length(a_best, n_try, time_budget, rng):

    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = np.maximum(a_try, 0)

        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_score(best_seq)
        print(f"Default init (n={n_current}): score={best_score:.8f}")

    # Try alternative initializations
    for mode in [4, 7]:
        a_alt = diverse_init(n_current, rng, mode=mode)
        score_alt = compute_score(a_alt)
        if score_alt < best_score:
            best_score = score_alt
            best_seq = a_alt.copy()
            print(f"Diverse init mode {mode}: score={best_score:.8f} **")

    # === Phase 1: Smooth p-norm Adam optimization ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 1: p-norm Adam (budget={min(250, remaining*0.2):.0f}s) ===")
        alloc = min(250.0, remaining * 0.2)
        a_smooth = adam_pnorm_optimize(best_seq, time_budget=alloc,
                                       p_start=10, p_end=2000, verbose=True)
        score_smooth = compute_score(a_smooth)
        if score_smooth < best_score:
            best_score = score_smooth
            best_seq = a_smooth.copy()
            print(f"  Phase1: score={best_score:.8f} **")
        else:
            print(f"  Phase1: score={score_smooth:.8f} (no improvement)")

    # === Phase 2: Subgradient for different basin ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 2: Subgradient (budget={min(60, remaining*0.08):.0f}s) ===")
        alloc = min(60.0, remaining * 0.08)
        a_sg = subgradient_descent(best_seq, time_budget=alloc, verbose=True)
        score_sg = compute_score(a_sg)
        if score_sg < best_score:
            best_score = score_sg
            best_seq = a_sg.copy()
            print(f"  Phase2: score={best_score:.8f} **")
        else:
            print(f"  Phase2: score={score_sg:.8f} (no improvement)")

    # === Phase 3: Main LP polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: LP Polishing (budget={min(400, remaining*0.35):.0f}s) ===")
        alloc = min(400.0, remaining * 0.35)
        a_lp = lp_polish(best_seq, time_budget=alloc, max_iters=80, verbose=True)
        score_lp = compute_score(a_lp)
        if score_lp < best_score:
            best_score = score_lp
            best_seq = a_lp.copy()
            print(f"  Phase3: score={best_score:.8f} **")
        else:
            print(f"  Phase3: score={score_lp:.8f} (no improvement)")

    # === Phase 4: Second round - smooth then LP ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 4: Smooth + LP Round 2 ===")
        # Perturb and re-optimize
        noise = rng.normal(0, 0.003, len(best_seq))
        a_pert = best_seq + noise
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc_smooth = min(80.0, remaining * 0.15)
        if alloc_smooth > 10:
            a_pert = adam_pnorm_optimize(a_pert, time_budget=alloc_smooth,
                                         p_start=50, p_end=2000, verbose=False)

        alloc_lp = min(150.0, remaining * 0.25)
        if alloc_lp > 10:
            a_pert = lp_polish(a_pert, time_budget=alloc_lp, max_iters=40, verbose=False)

        score_pert = compute_score(a_pert)
        if score_pert < best_score:
            best_score = score_pert
            best_seq = a_pert.copy()
            print(f"  Phase4: score={best_score:.8f} **")
        else:
            print(f"  Phase4: score={score_pert:.8f} (no improvement)")

    # === Phase 5: n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 5: n Exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 100, n_best - 50,
            n_best + 50, n_best + 100, n_best + 200, n_best + 400,
            600, 700, 900, 1000, 1200
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) >= 30]
        n_options = n_options[:4]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            t_budget = min(remaining * 0.5, 100.0)
            a_try = try_new_length(best_seq, n_try, t_budget, rng)
            if a_try is not None:
                score_try = compute_score(a_try)
                if score_try < best_score:
                    best_score = score_try
                    best_seq = a_try.copy()
                    print(f"  ** n={n_try}: score={best_score:.8f} **")
                else:
                    print(f"  n={n_try}: score={score_try:.8f}")

    # === Phase 6: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 6: Diverse Restarts ===")
        n_cur = len(best_seq)
        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            mode = (r * 3 + 1) % 8
            a_r = diverse_init(n_cur, rng, mode=mode)
            blend = 0.4
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            alloc = min(remaining * 0.3, 60.0)
            if alloc > 15:
                a_r = adam_pnorm_optimize(a_r, time_budget=alloc * 0.4,
                                          p_start=30, p_end=500, verbose=False)
                a_r = lp_polish(a_r, time_budget=alloc * 0.5, max_iters=20, verbose=False)

            score_r = compute_score(a_r)
            if score_r < best_score:
                best_score = score_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): score={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): score={score_r:.8f}")

    # === Phase 7: Final intensive LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: Final Polish (budget={min(remaining-10, 200):.0f}s) ===")
        alloc = min(remaining - 10, 200.0)
        if alloc > 20:
            a_final = lp_polish(best_seq, time_budget=alloc, max_iters=80, verbose=True)
            score_final = compute_score(a_final)
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(f"  Final: score={best_score:.8f} **")
            else:
                print(f"  Final: score={score_final:.8f}")

    # === Phase 8: Extra time - more LP ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining - 5, 60.0)
        if alloc > 10:
            a_extra = lp_polish(best_seq, time_budget=alloc, max_iters=30, verbose=False)
            score_extra = compute_score(a_extra)
            if score_extra < best_score:
                best_score = score_extra
                best_seq = a_extra.copy()
                print(f"  Extra: score={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_score(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
