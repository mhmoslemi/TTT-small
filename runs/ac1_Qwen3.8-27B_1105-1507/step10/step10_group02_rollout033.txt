Optimizes step function coefficients using active-set Newton method for the minimax

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective for normalized sequence (sum=1): 2n * max(conv)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_full(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

                denom = (S_other + x) ** 2
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Quick candidates
            scale = max(v, 1.0 / n, 1e-6)
            best_x = v
            best_f = eval_1d(v)
            for mult in [0.3, 0.5, 0.7, 1.3, 1.7, 2.5]:
                x_test = max(0.0, v * mult)
                f_test = eval_1d(x_test)
                if f_test < best_f:
                    best_f = f_test
                    best_x = x_test

            # Golden section refinement
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + 2.0 * scale
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = eval_1d(c_pt)
            fd = eval_1d(d_pt)

            for _ in range(30):
                if x_hi - x_lo < 1e-12 * max(1.0, abs(x_hi)):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = eval_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = eval_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-13 * max(1.0, v):
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

            obj = 2.0 * n * float(np.max(conv)) / (S * S)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a


def active_set_newton(a, time_budget=200.0, verbose=True, seed=42):
    """
    Active-set Newton method: iteratively identify active constraints and
    solve linearized system to equalize peak convolution values.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    M = float(np.max(conv))
    best_obj = 2.0 * n * M
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    max_iter = 500
    for it in range(max_iter):
        if time.monotonic() - start >= time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        obj = 2.0 * n * M
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Identify active set: conv values within tolerance of max
        tol = max(1e-6 * M, 1e-10)
        active_mask = conv >= M - tol
        ks_active = np.where(active_mask)[0]

        if len(ks_active) == 0:
            break

        # Build the linear system: 2 * conv_matrix(a, ks_active) @ da = rhs
        # where rhs[k] = M_target - conv[k] for k in active set
        # We want to equalize all active conv values to their mean (or to M)
        
        # Target: equalize active values
        active_vals = conv[ks_active]
        target_M = float(np.mean(active_vals))  # or use max
        
        # Build matrix: A[row, j] = 2 * a[k - j] if valid
        A = np.zeros((len(ks_active), n))
        rhs = np.zeros(len(ks_active))
        for row, k in enumerate(ks_active):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    A[row, j] = 2.0 * a[idx]
            rhs[row] = target_M - conv[k]

        # Add normalization constraint: sum(da) = 0
        A_full = np.vstack([A, np.ones((1, n))])
        rhs_full = np.append(rhs, 0.0)

        # Solve least squares (minimum norm solution)
        try:
            # Use SVD for minimum norm solution
            U, sigma, Vt = np.linalg.svd(A_full, full_matrices=False)
            # Regularize
            sigma_reg = sigma + 1e-10
            da = Vt.T @ np.diag(1.0 / sigma_reg) @ U.T @ rhs_full
        except np.linalg.LinAlgError:
            da = np.linalg.lstsq(A_full, rhs_full, rcond=None)[0]

        # Ensure non-negativity: a + da >= 0
        max_neg = 0.0
        if np.any(a + da < 0):
            max_neg = float(np.min(a + da))
            # Scale down
            scale_factor = -a[max_neg < 0].min() / (-max_neg) if max_neg < 0 else 1.0
            # Actually, let's just clip
            da = da * 0.5  # Dampen

        # Line search
        best_alpha = 0.01
        best_alpha_obj = obj
        for alpha in [0.5, 0.3, 0.2, 0.1, 0.05, 0.01, 0.005]:
            a_try = a + alpha * da
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            conv_try = compute_conv(a_try)
            obj_try = 2.0 * n * float(np.max(conv_try))
            if obj_try < best_alpha_obj:
                best_alpha_obj = obj_try
                best_alpha = alpha

        if best_alpha < 0.001 and stall < 10:
            # Try a random perturbation to escape
            pert = rng.normal(0, 0.001, n)
            a = a * np.exp(pert)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall += 1
            continue

        a = a + best_alpha * da
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        if best_alpha_obj >= obj - 1e-12:
            stall += 1
            if stall > 20:
                # Perturb to escape
                pert = rng.normal(0, 0.005, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
        else:
            stall = 0

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    AS-Newton it{it}: obj={best_obj:.8f}, M={M:.8f}, |S|={len(ks_active)}, t={elapsed:.1f}s")

    return best_a


def lp_polish(a, time_budget=60.0, seed=42):
    """LP-based polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # Build LP
        conv = compute_conv(a)
        M = float(np.max(conv))
        # Active set for tighter LP
        tol = 0.1 * M
        active = np.where(conv >= M - tol)[0]

        A_sub = np.zeros((len(active), n))
        for row, k in enumerate(active):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    A_sub[row, j] = a[idx]

        g = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A_sub @ g <= M_var, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                        solved = True
                        break
            except Exception:
                continue

        if not solved:
            break

        # Try blending
        improved = False
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            stall += 1
            if stall > 5:
                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if time.monotonic() - start > time_budget * 0.85:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 4:
        centers = [n // 3, 2 * n // 3]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return a_sym

"""
Optimizes step function coefficients using active-set Newton method for the minimax
convolution problem, combined with exact coordinate descent and LP polishing.
Key innovation: linearized active-set Newton steps that equalize peak convolution
values via minimum-norm solutions of small linear systems, achieving faster and
more precise convergence than pure coordinate descent or LP blending alone.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective for normalized sequence (sum=1): 2n * max(conv)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def eval_1d_coord(v, S_other, n, c_k_lin, a_ki_lin, c_2i, x):
    """Evaluate 1D coordinate subproblem objective at value x."""
    if len(a_ki_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) ** 2
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom


def golden_section_1d(f, x_lo, x_hi, x_init, n_iter=30):
    """Golden section search on [x_lo, x_hi] with initial point x_init."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    # First evaluate candidates around x_init
    best_x = x_init
    best_f = f(x_init)
    scale = max(abs(x_init), 1e-6)
    for mult in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 3.0]:
        x_test = max(0.0, x_init * mult)
        ft = f(x_test)
        if ft < best_f:
            best_f = ft
            best_x = x_test
    # Golden section refinement
    x_lo = max(0.0, best_x - scale)
    x_hi = best_x + 2.0 * scale
    if x_hi - x_lo < 1e-14:
        x_hi = x_lo + 1e-8
    c_pt = x_hi - gr * (x_hi - x_lo)
    d_pt = x_lo + gr * (x_hi - x_lo)
    fc = f(c_pt)
    fd = f(d_pt)
    for _ in range(n_iter):
        if x_hi - x_lo < 1e-12 * max(1.0, abs(x_hi)):
            break
        if fc < fd:
            x_hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = x_hi - gr * (x_hi - x_lo)
            fc = f(c_pt)
        else:
            x_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = x_lo + gr * (x_hi - x_lo)
            fd = f(d_pt)
    return max(0.0, (x_lo + x_hi) / 2.0)


def coordinate_descent_fast(a, time_budget=100.0, seed=42):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i].copy()
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Golden section search
            x_lo = 0.0
            x_hi = max(3.0 * v, S_other + v, S / n * 3.0)
            x_opt = golden_section_1d(
                lambda x: eval_1d_coord(v, S_other, n, c_k_lin, a_ki_lin, c_2i, x),
                x_lo, x_hi, v, n_iter=28
            )

            if abs(x_opt - v) < 1e-13 * max(1.0, v):
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

            obj = 2.0 * n * float(np.max(conv)) / (S * S)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a


def active_set_newton(a, time_budget=200.0, verbose=True, seed=42):
    """
    Active-set Newton method: iteratively identify active constraints (peak
    convolution positions) and solve a small linear system to equalize them.
    Uses minimum-norm solution for the underdetermined system.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    M = float(np.max(conv))
    best_obj = 2.0 * n * M
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    max_iter = 1000
    for it in range(max_iter):
        if time.monotonic() - start >= time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        obj = 2.0 * n * M
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Identify active set
        tol = max(1e-5 * M, 1e-12)
        active_mask = conv >= M - tol
        ks_active = np.where(active_mask)[0]

        if len(ks_active) == 0 or len(ks_active) >= n:
            # Fallback to coordinate-style update
            break

        # Build linear system: 2 * A @ da = rhs
        # A[row, j] = a[k-row - j] if valid
        n_active = len(ks_active)
        A_mat = np.zeros((n_active, n))
        rhs = np.zeros(n_active)
        for row, k in enumerate(ks_active):
            # conv(a, da)[k] = sum_j a[k-j] * da[j] for valid j
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            if j_end >= j_start:
                a_idx = k - np.arange(j_start, j_end + 1)
                A_mat[row, j_start:j_end + 1] = 2.0 * a[a_idx]
            rhs[row] = float(np.mean(conv[ks_active])) - conv[k]

        # Add normalization: sum(da) = 0
        B = np.vstack([A_mat, np.ones((1, n))])
        rhs_full = np.append(rhs, 0.0)

        # Minimum norm solution: da = B^T (B B^T + eps*I)^{-1} rhs_full
        m = B.shape[0]
        BBt = B @ B.T
        eps = 1e-8 * np.trace(BBt) / m
        BBt += eps * np.eye(m)
        try:
            sol = np.linalg.solve(BBt, rhs_full)
            da = B.T @ sol
        except np.linalg.LinAlgError:
            da = np.linalg.lstsq(B, rhs_full, rcond=None)[0]

        # Scale da to reasonable magnitude
        da_norm = float(np.linalg.norm(da))
        if da_norm < 1e-15:
            stall += 1
            if stall > 5:
                pert = rng.normal(0, 0.002, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
            continue

        # Line search with backtracking
        best_alpha = 0.0
        best_alpha_obj = obj
        for alpha_candidates in [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_try = a + alpha_candidates * da
            if np.any(a_try < 0):
                # Find max step that keeps non-negativity
                neg = a_try < 0
                if np.any(neg):
                    max_step = float(np.min(-a[neg] / da[neg]))
                    if max_step <= 0:
                        continue
                    alpha_candidates = max_step * 0.9
                else:
                    continue
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            conv_try = compute_conv(a_try)
            obj_try = 2.0 * n * float(np.max(conv_try))
            if obj_try < best_alpha_obj:
                best_alpha_obj = obj_try
                best_alpha = alpha_candidates

        if best_alpha < 1e-4:
            stall += 1
            if stall > 10:
                pert = rng.normal(0, 0.005, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if time.monotonic() - start > time_budget * 0.8:
                    break
            continue

        a = a + best_alpha * da
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s
        stall = 0

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    AS-Newton it{it}: obj={best_obj:.8f}, |S|={n_active}, "
                  f"alpha={best_alpha:.4f}, t={elapsed:.1f}s")

    return best_a


def lp_polish(a, time_budget=60.0, seed=42):
    """LP-based polishing with active set reduction."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Use active set + a few nearby for the LP
        tol_f = 0.05
        active = np.where(conv >= M * (1 - tol_f))[0]
        # Also always include center and edges
        center = n - 1
        for extra_k in [0, n - 1, 2 * n - 2]:
            if extra_k not in active:
                active = np.append(active, extra_k)
        active = np.unique(active)

        # Build LP matrix
        A_sub = np.zeros((len(active), n))
        for row, k in enumerate(active):
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            if j_end >= j_start:
                a_idx = k - np.arange(j_start, j_end + 1)
                A_sub[row, j_start:j_end + 1] = a[a_idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A_sub @ g_var <= M_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                        solved = True
                        break
            except Exception:
                continue

        if not solved:
            break

        improved = False
        for t in [0.5, 0.7, 1.0, 0.3]:
            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            stall += 1
            if stall > 3:
                pert = rng.normal(0, 0.03, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if time.monotonic() - start > time_budget * 0.85:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 4:
        centers = [n // 3, 2 * n // 3]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2.0 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

        if alloc > 20:
            a_r = coordinate_descent_fast(a_r, time_budget=alloc * 0.5, seed=5000 + r * 100)
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_r = active_set_newton(a_r, time_budget=min(alloc * 0.4, remaining * 0.3),
                                        verbose=False, seed=6000 + r * 100)
            obj_r = 2.0 * a_r.size * float(np.max(compute_conv(a_r / a_r.sum())))
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final Polish ===
    print("\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 100.0)
        a_f = coordinate_descent_fast(best_seq, time_budget=alloc, seed=seed + 9999)
        obj_f = 2.0 * a_f.size * float(np.max(compute_conv(a_f / a_f.sum())))
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.3, 80.0)
        a_f2 = active_set_newton(best_seq, time_budget=alloc, verbose=False, seed=seed + 8888)
        obj_f2 = 2.0 * a_f2.size * float(np.max(compute_conv(a_f2 / a_f2.sum())))
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final AS: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 50.0)
        a_f3 = lp_polish(best_seq, time_budget=alloc, seed=7777)
        obj_f3 = 2.0 * a_f3.size * float(np.max(compute_conv(a_f3 / a_f3.sum())))
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
