import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_conv(a, k_star):
    """Compute 2 * Q_{k*} @ a, which is a shifted/reversed version of a.
    (2 Q_k a)[i] = a[k - i] if 0 <= k - i < n, else 0."""
    n = len(a)
    grad = np.zeros(n)
    # For i in range(n): grad[i] = a[k_star - i] if 0 <= k_star - i < n
    # This means: when k_star - i >= 0 and k_star - i < n
    # i.e., k_star - n < i <= k_star
    # i.e., max(0, k_star - n + 1) <= i <= min(n-1, k_star)
    i_start = max(0, k_star - n + 1)
    i_end = min(n - 1, k_star)
    if i_start <= i_end:
        grad[i_start:i_end+1] = a[k_star:i_start-1:-1]  # a[k_star], a[k_star-1], ..., a[k_star-(i_end-i_start)]
    return grad

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FISTA (accelerated projected gradient) with log-sum-exp smoothed gradient.
Key insight: the gradient of the smoothed max-convolution objective is a weighted
cross-correlation of the softmax weights with the sequence, computable in O(n log n).
Multiple initializations (ramp, uniform, random, parent) are tried with adaptive
smoothing and restart. LP polishing provides final refinement.
"""

import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, p, N):
    """Compute gradient of log-sum-exp smoothed max conv.
    grad[i] = 2 * sum_k p[k] * a[k-i] (if valid)
    Computed via FFT cross-correlation."""
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(np.conj(P) * A)[:len(a)]
    return 2.0 * cross.real


def fista_optimize(a, time_budget=100.0, verbose=False, seed=42):
    """FISTA optimization of max conv peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    
    # FFT size for cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    
    t = 30.0  # smoothing parameter
    lr = 0.5
    momentum_t = 1.0
    y = a.copy()
    
    it = 0
    best_it = 0
    stagnant = 0
    
    while time.monotonic() - start < time_budget:
        if time_budget - (time.monotonic() - start) < 0.1:
            break
            
        # Compute convolution
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        
        # Softmax weights
        shifted = conv - max_conv
        exp_vals = np.exp(t * shifted)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp
        
        # Compute gradient via FFT
        grad = smoothed_gradient(y, p, N)
        
        # FISTA step
        z = y - lr * grad
        a_new = project_simplex(z)
        
        # Update momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * momentum_t * momentum_t)) / 2.0
        y = a_new + ((momentum_t / t_new) - 1.0) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = float(np.sum(y))
        if sy > 1e-10:
            y = y / sy
        
        a = a_new
        momentum_t = t_new
        
        # Track best
        obj = 2.0 * n * float(np.max(compute_conv(a)))
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            best_it = it
            stagnant = 0
        else:
            stagnant += 1
        
        it += 1
        
        # Adaptive smoothing: increase t over time
        if it % 1000 == 0:
            t = min(t * 1.05, 500.0)
            # Reset momentum if stagnant
            if stagnant > 2000:
                y = best_a.copy()
                momentum_t = 1.0
                stagnant = 0
                lr = max(0.1, lr * 0.8)
        
        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - start
            print(f"    FISTA it{it}: obj={best_obj:.8f}, t={t:.1f}, lr={lr:.4f}, stagnant={stagnant}, t={elapsed:.1f}s")
    
    return best_a


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based polishing: solve min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build convolution matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    
    import cvxpy as cp
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    best_g = None
    best_M = None
    
    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    if best_g is None or M_val < best_M:
                        best_g = g_val
                        best_M = M_val
                    break
        except Exception:
            continue
    
    if best_g is None:
        return a
    
    # Try different blend factors
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        a_try = (1 - t) * a + t * best_g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_a = a_try.copy()
    
    return best_a


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    
    # Uniform
    inits.append(('uniform', np.ones(n) / n))
    
    # Linear ramp (increasing)
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(('ramp_inc', ramp / ramp.sum()))
    
    # Linear ramp (decreasing)
    inits.append(('ramp_dec', ramp[::-1] / ramp.sum()))
    
    # Quadratic ramp
    quad = np.arange(1, n + 1, dtype=np.float64) ** 2
    inits.append(('quad_inc', quad / quad.sum()))
    
    # Exponential ramp
    exp_ramp = np.exp(0.5 * np.linspace(0, 1, n))
    inits.append(('exp_ramp', exp_ramp / exp_ramp.sum()))
    
    # Bimodal (high at edges)
    x = np.linspace(0, 1, n)
    bimodal = 0.1 + np.abs(2 * x - 1)
    inits.append(('bimodal', bimodal / bimodal.sum()))
    
    # Cosine (high in middle)
    cosine = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    cosine = np.maximum(cosine, 0.01)
    inits.append(('cosine', cosine / cosine.sum()))
    
    # Random
    rand = rng.random(n) + 0.1
    inits.append(('random', rand / rand.sum()))
    
    # Smooth random (low-pass filtered)
    rand_smooth = np.cumsum(rng.random(n))
    rand_smooth = np.maximum(rand_smooth - 0.5, 0.01)
    inits.append(('rand_smooth', rand_smooth / rand_smooth.sum()))
    
    return inits
