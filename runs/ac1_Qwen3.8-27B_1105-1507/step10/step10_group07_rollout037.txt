def project_simplex(v):
    n = len(v)
    if n == 0:
        return v

    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)

    rho = np.where(cond)[0]

    rho = rho[-1]

        # = sum_j a_{k-j} * δ_j where 0 <= k-j < n, i.e., max(0,k-n+1) <= j <= min(k, n-1)
        row = np.zeros(n)
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)

        A_parts.append(row)

    b = conv_a[active_set]

        cp.sum(a + delta) == 1,
        a + delta >= 0,
        cp.norm(delta, 2) <= r,

"""
Multi-strategy optimizer for minimum peak autocorrelation of non-negative sequences.
Uses smoothed gradient descent (FFT-based), peak-specific SOCP, and multi-length exploration.
Key innovations: adaptive temperature scheduling, top-K peak weighting, two-level initialization,
and aggressive length exploration.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gd_v2(a, time_budget, tau_start=5.0, tau_end=5000.0, 
                 lr_init=None, momentum_init=0.85, top_k=0, verbose=False, seed=42):
    """
    Improved smooth gradient descent.
    top_k=0 means use full softmax; top_k>0 means use top-K uniform weighting.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    if lr_init is None:
        lr_init = 2.0 / np.sqrt(n)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    momentum = momentum_init
    patience = 0
    max_patience = 200
    
    N = 1
    while N < 4 * n:
        N <<= 1
    
    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Adaptive tau: start low, increase over time
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.6)
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        if top_k > 0:
            # Top-K uniform weighting
            idx = np.argsort(conv)[::-1][:top_k]
            p = np.zeros_like(conv)
            p[idx] = 1.0 / top_k
        else:
            # Full soft-max
            shifted = conv - max_conv
            # Clip to avoid overflow
            shifted = np.maximum(shifted, -50.0 / tau)
            exp_vals = np.exp(tau * shifted)
            p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient via FFT: g_j = 2 * sum_m a_m * p_{j+m}
        A = fft(a, N)
        P = fft(p, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]
        
        # Normalized gradient step
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-30:
            g = g / g_norm
        
        # Momentum update
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
            # Slowly increase momentum as we converge
            momentum = min(0.99, momentum + 0.0001)
        else:
            patience += 1
            if patience == 50:
                lr *= 0.7
            elif patience == 100:
                lr *= 0.7
                v *= 0.5
            elif patience > max_patience:
                lr *= 0.5
                v = 0.5 * v
                patience = 50
                if lr < 1e-18:
                    break
        
        a = a_new
        it += 1
        
        if verbose and it % 500 == 0:
            print(f"      SGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.2e}, t={elapsed:.1f}s")
    
    return best_a


def build_conv_submatrix(a, active_indices):
    """Build submatrix A_sub such that (A_sub @ delta)[i] = (a*delta)[active_indices[i]]."""
    n = len(a)
    m = len(active_indices)
    A_sub = np.zeros((m, n))
    for i, k in enumerate(active_indices):
        k = int(k)
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A_sub[i, j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]
    return A_sub


def solve_soco_peak(a, active_indices, r):
    """
    Solve SOCP:
    min t
    s.t. conv_a[k_i] + 2*(a*delta)[k_i] <= t for i in active
         sum(a+delta) = 1
         a+delta >= 0
         ||delta||_2 <= r
    """
    n = len(a)
    a = a / a.sum()
    conv_a = compute_conv(a)
    
    delta = cp.Variable(n)
    t = cp.Variable()
    
    A_sub = build_conv_submatrix(a, active_indices)
    b = conv_a[active_indices]
    
    constraints = [
        cp.sum(a + delta) == 1,
        a + delta >= 0,
        cp.norm(delta, 2) <= r,
        b + 2.0 * (A_sub @ delta) <= t,
    ]
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten()
        except Exception:
            continue
    return None


def solve_lp_full(a):
    """Solve LP: min M s.t. (a*g)_k <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / a.sum()
    
    # Build full convolution matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

def resample_to_length(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def generate_two_level_init(n, k, rng):
    """Generate a random two-level {x, y} sequence."""
    positions = rng.choice(n, size=k, replace=False)
    a = np.ones(n)
    a[positions] = 2.0  # temporarily two-level
    a = a / a.sum()
    return a

"""
Multi-strategy optimizer for minimum peak autocorrelation of non-negative sequences.
Combines FFT-based smoothed gradient descent (adaptive temperature, top-K weighting),
peak-specific SOCP trust regions, multi-length exploration, and two-level initialization.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gd_v2(a, time_budget, tau_start=5.0, tau_end=5000.0,

    Improved smooth gradient descent with adaptive temperature.
    top_k=0: full softmax weighting; top_k>0: uniform weight on top-K peaks.

def build_conv_submatrix(a, active_indices):
    """Build submatrix A_sub where (A_sub @ delta)[i] = (a*delta)[k_i]."""
    n = len(a)
    m = len(active_indices)
    A_sub = np.zeros((m, n))
    for i, k in enumerate(active_indices):
        k = int(k)
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A_sub[i, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A_sub


def solve_soco_peak(a, active_indices, r):
    """
    SOCP: min t s.t. c_a[k_i] + 2*(a*delta)[k_i] <= t, sum(a+delta)=1,
    a+delta>=0, ||delta||_2 <= r
    """
    n = len(a)
    a = a / a.sum()
    conv_a = compute_conv(a)

    delta = cp.Variable(n)
    t = cp.Variable()
    A_sub = build_conv_submatrix(a, active_indices)
    b = conv_a[active_indices]

    constraints = [
        cp.sum(a + delta) == 1,
        a + delta >= 0,
        cp.norm(delta, 2) <= r,
        b + 2.0 * (A_sub @ delta) <= t,
    ]

    prob = cp.Problem(cp.Minimize(t), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten()
        except Exception:
            continue
    return None


def solve_lp_full(a):
    """LP: min M s.t. (a*g)_k <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / a.sum()
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

    """SOCP polishing with peak-specific constraints and adaptive trust region."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)

                break

            break

            delta = delta * (r / norm_d)

            a = np.maximum(a, 0)
            s_n = a.sum()
            if s_n > 1e-10:

def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def generate_two_level_init(n, k, rng):
    """Generate a random two-level sequence."""
    positions = rng.choice(n, size=k, replace=False)
    a = np.ones(n)
    a[positions] = 2.5
    a = a / a.sum()
    return a

    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: Smooth GD warmup (low-to-mid tau) ===
    print("\n=== Phase 1: Smooth GD Warmup ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.10, 100.0)
    if p1_budget > 15:
        a1 = smooth_gd_v2(best_seq, time_budget=p1_budget,
                          tau_start=5.0, tau_end=3000.0,
                          lr_init=3.0 / np.sqrt(n_base),
                          momentum_init=0.85)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 GD: obj={obj1:.8f}")

    # === Phase 2: SOCP polish ===
    print("\n=== Phase 2: SOCP Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.20, 220.0)
    if p2_budget > 30 and len(best_seq) <= 1600:
        a2 = peak_socp_polish(best_seq, time_budget=p2_budget,
                              r_init=0.15, verbose=False, seed=42)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 SOCP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 SOCP: obj={obj2:.8f}")

    # === Phase 3: High-tau GD (peak-focused, top-K) ===
    print("\n=== Phase 3: Peak-Focused GD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.10, 100.0)
    if p3_budget > 15:
        a3 = smooth_gd_v2(best_seq, time_budget=p3_budget,
                          tau_start=500.0, tau_end=80000.0,
                          lr_init=1.0 / np.sqrt(n_base),
                          momentum_init=0.92, top_k=5)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 HiGD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 HiGD: obj={obj3:.8f}")

    # === Phase 4: Multi-length exploration ===
    print("\n=== Phase 4: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_base = len(best_seq)

    n_options = []
    for dn in [-200, -100, 100, 200, 400, 500, 800]:
        n_try = n_base + dn
        if 300 <= n_try <= 2500 and abs(n_try - n_base) > 50:
            n_options.append(n_try)
    for n_abs in [500, 700, 1000, 1200, 1500, 2000]:
        if 300 <= n_abs <= 2500 and n_abs not in n_options and abs(n_abs - n_base) > 50:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))

    for n_try in n_options[:7]:
        remaining = deadline - time.monotonic()
        if remaining < 55:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_to_length(best_seq, n_try)

        alloc1 = min(remaining * 0.07, 35.0)
        if alloc1 > 8:
            a_try = smooth_gd_v2(a_try, time_budget=alloc1,
                                 tau_start=5.0, tau_end=5000.0,
                                 lr_init=3.0 / np.sqrt(n_try),
                                 momentum_init=0.85)

        remaining = deadline - time.monotonic()
        if n_try <= 1500:
            alloc2 = min(remaining * 0.09, 70.0)
        else:
            alloc2 = 0
        if alloc2 > 12:
            a_try = peak_socp_polish(a_try, time_budget=alloc2, r_init=0.12)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Two-level initialization ===
    print("\n=== Phase 5: Two-level Init ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for k_frac in [0.3, 0.5, 0.65]:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        k = int(k_frac * n_best)
        a_tl = generate_two_level_init(n_best, k, rng)

        alloc = min(remaining * 0.10, 45.0)
        if alloc > 8:
            a_tl = smooth_gd_v2(a_tl, time_budget=alloc,
                                tau_start=10.0, tau_end=8000.0,
                                lr_init=3.0 / np.sqrt(n_best),
                                momentum_init=0.85)
        obj_tl = compute_obj(a_tl)
        if obj_tl < best_score:
            best_score = obj_tl
            best_seq = a_tl.copy()
            print(f"  Two-level k={k}: obj={best_score:.8f} **")
        else:
            print(f"  Two-level k={k}: obj={obj_tl:.8f}")

    # === Phase 6: Perturbation + re-optimize ===
    print("\n=== Phase 6: Perturbation ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r_idx in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        scale = 0.004 + 0.004 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
        a_pert = np.maximum(a_pert, 0)
        s_p = a_pert.sum()
        if s_p < 1e-10:
            break
        a_pert = a_pert / s_p

        alloc1 = min(remaining * 0.07, 25.0)
        if alloc1 > 8:
            a_pert = smooth_gd_v2(a_pert, time_budget=alloc1,
                                  tau_start=200.0, tau_end=50000.0,
                                  lr_init=1.0 / np.sqrt(n_best),
                                  momentum_init=0.92, top_k=3)

        remaining = deadline - time.monotonic()
        if n_best <= 1400:
            alloc2 = min(remaining * 0.05, 25.0)
        else:
            alloc2 = 0
        if alloc2 > 8:
            a_pert = peak_socp_polish(a_pert, time_budget=alloc2, r_init=0.08)

        obj_pert = compute_obj(a_pert)
        if obj_pert < best_score:
            best_score = obj_pert
            best_seq = a_pert.copy()
            print(f"  Pert{r_idx} (s={scale:.4f}): obj={best_score:.8f} **")
        else:
            print(f"  Pert{r_idx} (s={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 7: Final intensive polish ===
    print("\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    if remaining > 25:
        alloc_gd = min(remaining * 0.25, 50.0)
        if alloc_gd > 8:
            a_f = smooth_gd_v2(best_seq, time_budget=alloc_gd,
                               tau_start=2000.0, tau_end=200000.0,
                               lr_init=0.3 / np.sqrt(n_best),
                               momentum_init=0.95, top_k=3)
            obj_f = compute_obj(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final GD: obj={best_score:.8f} **")
            else:
                print(f"  Final GD: obj={obj_f:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15 and n_best <= 1400:
        alloc_socp = min(remaining * 0.5, 80.0)
        if alloc_socp > 10:
            a_f2 = peak_socp_polish(best_seq, time_budget=alloc_socp,
                                    r_init=0.06, verbose=True, seed=999)
            obj_f2 = compute_obj(a_f2)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final SOCP: obj={best_score:.8f} **")
            else:
                print(f"  Final SOCP: obj={obj_f2:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
