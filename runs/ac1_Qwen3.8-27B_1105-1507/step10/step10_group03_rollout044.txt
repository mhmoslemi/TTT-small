"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set LP polishing, top-k smoothed gradient descent with Nesterov
acceleration, dual LP analysis, and multi-n exploration.

import numpy as np

from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def topk_smoothed_grad(a, k, tau):
    """Compute gradient using top-k softmax smoothing over convolution entries."""
    n = len(a)
    conv = compute_conv(a)
    m = len(conv)
    exact_max = float(np.max(conv))
    
    # Get top k indices
    if k >= m:
        top_idx = np.arange(m)
    else:
        top_idx = np.argpartition(-conv, k - 1)[:k]
    
    # Softmax over top-k entries
    c = conv[top_idx]
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    sum_exp = exp_c.sum()
    if sum_exp < 1e-300:
        p_local = np.zeros(len(c))
        p_local[c.argmax()] = 1.0
    else:
        p_local = exp_c / sum_exp
    
    # Build full p vector
    p = np.zeros(m)
    p[top_idx] = p_local
    
    # Compute gradient: g[i] = 2 * sum_k p[k] * a[k-i]
    # = 2 * (p cross-correlate a)[i]
    N = 1
    while N < m + n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    
    return g, exact_max


def build_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_primal(a, top_k=None, solver=None):
    """Solve: min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1.
    Returns (g_opt, M_opt) or (None, None)."""
    n = len(a)
    
    if top_k is not None and top_k < 2 * n - 1:
        conv_aa = compute_conv(a)
        top_idx = np.argpartition(-conv_aa, top_k - 1)[:top_k]
        # Build reduced matrix
        A = np.zeros((top_k, n))
        for idx, k in enumerate(top_idx):
            for j in range(n):
                d = k - j
                if 0 <= d < n:
                    A[idx, j] = a[d]
    else:
        A = build_conv_matrix(a)
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    if solver is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    else:
        solver_list = [solver]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    
    return None, None

    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    stall_patience = 3000
    
    for tau in tau_levels:
        tau_iters = 0
        max_iters = 3000
        
        while time.monotonic() - start < time_budget and tau_iters < max_iters:
            # Nesterov lookahead
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0

            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break
            
            # Adaptive step size
            step = lr / (1 + tau * np.max(np.abs(g)))
            v = beta * v - step * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
            else:
                a_new = a_new / s_new

                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it
            
            a = a_new
            it += 1
            tau_iters += 1
            
            if it - best_it > stall_patience:
                a = best_a.copy()
                v = np.zeros(n)
                lr = 3.0 / np.sqrt(n)
                best_it = it
        
        if verbose:
            print(f"    tau={tau:5d}, k={k}: obj={best_obj:.10f}, iters={tau_iters}")
