import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    return compute_obj_normalized(a_norm)


def smooth_grad(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using log-sum-exp + FFT.
    Returns gradient (length n) and the conv (length 2n-1)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 2 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    # grad[i] = 2 * sum_k p[k] * a[k-i] (where valid)
    # This is 2 * ifft(P * A) but we need the right indexing
    # conv(k) = sum_i a[i]*a[k-i], d/d a[x] = sum_k p[k] * d(conv[k])/d(a[x])
    # d(conv[k])/d(a[x]) = 2*a[k-x] if 0<=k-x<n else 0
    # So grad[x] = 2 * sum_k p[k] * a[k-x] = 2 * (p_reversed * a)[x]
    # = 2 * ifft(fft(p_reversed) * fft(a))[x]
    # p_reversed[x] = p[2n-2-x]
    p_rev = conv  # placeholder
    # Actually: grad[x] = 2 * sum_{k=0}^{2n-2} p[k] * a[k-x] for 0<=k-x<n
    # Let j = k-x, then k = j+x: grad[x] = 2 * sum_{j=0}^{n-1} p[j+x] * a[j]
    # This is a cross-correlation of p and a.
    # Ifft(fft(p, N) * conj(fft(a, N))) gives correlation... let me just do it via FFT
    # grad[x] = 2 * (p ⊛ a)[x] where ⊛ is cross-correlation
    # (p ⊛ a)[x] = sum_j p[j+x] * a[j]
    # = ifft(fft(p, N) * conj(fft(a, N)))[x] ... need to check
    # Actually circular cross-correlation: ifft(conj(fft(a)) * fft(p))
    # Let's just use: grad = 2 * ifft(conj(fft(a, N)) * fft(p, N)).real[:n]
    # Hmm, let me verify with a simple case.
    # For linear conv: C = ifft(fft(a,N) * fft(a,N))
    # d(C[k])/d(a[i]) = 2*a[k-i] (as derived)
    # grad[i] = sum_k p[k] * 2*a[k-i]
    # = 2 * sum_k p[k] * a[k-i]
    # Let's define b[k] = p[k] for k=0..2n-2, 0 otherwise (padded to N)
    # Then grad[i] = 2 * sum_k b[k] * a[k-i]
    # This is the cross-correlation of b and a at lag i.
    # In FFT: ifft(conj(fft(a, N)) * fft(b, N)).real gives (b ⊛ a) if we're careful.
    # Actually: circular cross-correlation (b ⊛ a)[i] = sum_k b[k] * a[(k-i) mod N]
    # We want sum_k b[k] * a[k-i] (linear, not circular). With N >= 2n-1 + n - 1 = 3n-2,
    # the circular version equals the linear version for i=0..n-1.
    # So: grad = 2 * ifft(conj(fft(a, N)) * fft(b, N)).real[:n]
    # But we need N >= 3n-2 to avoid wrap-around. Let's use N >= 3n.
    N2 = 1
    while N2 < 3 * n:
        N2 <<= 1
    B = fft(p, N2)
    A2 = fft(a, N2)
    corr = ifft(np.conj(A2) * B).real
    grad = 2.0 * corr[:n]
    return grad, conv


def fast_cd(a, time_budget=300.0, verbose=True, seed=42):
    """
    Fast coordinate descent using analytical 1D subproblem solving.
    For each coordinate i, we solve the 1D problem analytically where possible:
    - If quadratic dominates: x* = (c_{2i} - v^2) / S_o
    - If a linear dominates: x* = 0 (boundary)
    - Crossover: found via bisection
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    no_improve_sweeps = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        obj_before = 2.0 * n * float(np.max(conv))

        order = np.arange(n)
        rng.shuffle(order)

        for idx in range(n):
            i = order[idx]
            v = a[i]
            S_o = 1.0 - v  # S is 1.0

            if S_o < 1e-14:
                continue

            # Determine affected conv indices
            k_min = i
            k_max = min(2 * n - 2, i + n - 1)

            # Linear part: for k != 2i, conv[k] + 2*a[k-i]*(x-v)
            # Quadratic part: for k = 2i, conv[2i] + x^2 - v^2
            k_quad = 2 * i
            c_2i = conv[k_quad]

            # Current max excluding changes
            # Linear terms for k in [i, i+n-1], k != 2i
            # a[k-i] for k in that range

            # Quick analytical solution:
            # Case 1: Quadratic is binding -> x_q = (c_2i - v^2) / S_o
            x_q = (c_2i - v * v) / S_o
            if x_q < 0:
                x_q = 0.0

            # Evaluate f at candidate points
            # f(x) = 2n * max(L(x), Q(x)) / (S_o + x)^2
            # where Q(x) = c_2i + x^2 - v^2
            # L(x) = max over k != 2i of (conv[k] + 2*a[k-i]*(x-v))

            candidates = [0.0, x_q]
            if v > 0:
                candidates.append(v * 0.5)
                candidates.append(v)

            best_x = v
            best_f = np.inf

            # Get the linear coefficients
            ks = np.arange(k_min, k_max + 1)
            linear_coeffs = 2.0 * a[ks - i]  # 2*a[k-i] for each k
            linear_constants = conv[ks].copy()  # current conv values

            # Mask out the quadratic position
            mask_lin = (ks != k_quad)
            ks_lin = ks[mask_lin]
            lin_coeff = linear_coeffs[mask_lin]
            lin_const = linear_constants[mask_lin]

            # Evaluate at candidates
            for x in candidates:
                if x < 0:
                    x = 0.0
                denom = S_o + x
                if denom < 1e-14:
                    continue
                q_val = c_2i + x * x - v * v
                if len(ks_lin) > 0:
                    l_val = float(np.max(lin_const + lin_coeff * (x - v)))
                else:
                    l_val = 0.0
                max_val = max(q_val, l_val)
                f_val = 2.0 * n * max_val / (denom * denom)
                if f_val < best_f:
                    best_f = f_val
                    best_x = x

            # If x_q was the best, also check a few points around it
            if abs(best_x - x_q) < 1e-10 and x_q > 0:
                for dx in [0.01, 0.001, 0.1]:
                    x_try = max(0.0, x_q - dx * max(x_q, 0.01))
                    x_try2 = x_q + dx * max(x_q, 0.01)
                    for x in [x_try, x_try2]:
                        denom = S_o + x
                        if denom < 1e-14:
                            continue
                        q_val = c_2i + x * x - v * v
                        if len(ks_lin) > 0:
                            l_val = float(np.max(lin_const + lin_coeff * (x - v)))
                        else:
                            l_val = 0.0
                        max_val = max(q_val, l_val)
                        f_val = 2.0 * n * max_val / (denom * denom)
                        if f_val < best_f:
                            best_f = f_val
                            best_x = x

            # Refine with a few bisection steps if best_x is interior
            if best_x > 1e-10 and best_x < S_o * 2:
                # Check if we can improve with a small perturbation
                slope_est = 0.0
                h = max(1e-6, 0.01 * best_x)
                x_l = max(0.0, best_x - h)
                x_r = best_x + h
                denom_l = S_o + x_l
                denom_r = S_o + x_r
                if denom_l > 1e-14 and denom_r > 1e-14:
                    q_l = c_2i + x_l * x_l - v * v
                    q_r = c_2i + x_r * x_r - v * v
                    if len(ks_lin) > 0:
                        l_l = float(np.max(lin_const + lin_coeff * (x_l - v)))
                        l_r = float(np.max(lin_const + lin_coeff * (x_r - v)))
                    else:
                        l_l = l_r = 0.0
                    f_l = 2.0 * n * max(q_l, l_l) / (denom_l * denom_l)
                    f_r = 2.0 * n * max(q_r, l_r) / (denom_r * denom_r)
                    if f_l < best_f:
                        best_x = x_l
                        best_f = f_l
                    elif f_r < best_f:
                        best_x = x_r
                        best_f = f_r

            x_opt = max(0.0, best_x)
            delta = x_opt - v
            if abs(delta) < 1e-15:
                continue

            a[i] = x_opt

            # Update conv
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a[ks_lin - i]

            S += delta

        # End coordinate sweep
        # Renormalize periodically to avoid drift
        if sweep % 5 == 0:
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                a = a / S_check
                conv = compute_conv(a)
                obj_now = 2.0 * n * float(np.max(conv))
                if obj_now < best_obj:
                    best_obj = obj_now
                    best_a = a.copy()
                    no_improve_sweeps = 0
                else:
                    no_improve_sweeps += 1
            else:
                a = best_a.copy()
                conv = compute_conv(a)
        else:
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj_now = 2.0 * n * float(np.max(compute_conv(a / S_check)))
                if obj_now < best_obj:
                    best_obj = obj_now
                    best_a = (a / S_check).copy()
                    no_improve_sweeps = 0
                else:
                    no_improve_sweeps += 1

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s, stall={no_improve_sweeps}")

        # Early stop if stalled too long
        if no_improve_sweeps > 30 and sweep > 50:
            break

    return best_a


def nesterov_smoothed(a, time_budget=100.0, tau_init=30.0, tau_end=300.0,
                       verbose=False, seed=42):
    """Nesterov-accelerated smoothed gradient descent with adaptive tau."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v_mom = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * (progress ** 0.7)

        # Nesterov lookahead
        y = a + beta * v_mom
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = 1.0
        else:
            y = y / s_y

        # Compute smoothed gradient
        g, conv = smooth_grad(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm < 1e-15:
            break

        # Adaptive learning rate
        lr = min(lr * 1.001, 3.0 / np.sqrt(n))
        lr = max(lr * 0.999, 0.05 * 3.0 / np.sqrt(n))

        v_mom = beta * v_mom - lr * g
        a_new = y + v_mom
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v_mom = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 2000 == 0:
            print(f"    Nesterov it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}")

    return best_a


def active_set_polish(a, time_budget=60.0, verbose=False):
    """
    Active set polishing: identify the top convolution peaks and solve
    a constrained optimization to reduce them.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Find active set: conv values within 10% of max
        threshold = max_val * 0.9
        active_mask = conv >= threshold
        active_ks = np.where(active_mask)[0]

        if len(active_ks) < 2:
            break

        # For each active k, we want to reduce conv[k].
        # Use a simple approach: for each active k, find the coordinate
        # that most contributes to conv[k] and reduce it.
        improved = False
        for k in active_ks[:10]:  # Only top 10
            if time.monotonic() - start >= time_budget:
                break
            # conv[k] = sum_i a[i]*a[k-i]
            # To reduce it, reduce the product. Find i that maximizes a[i]*a[k-i].
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_min > i_max:
                continue
            products = a[i_min:i_max + 1] * a[k - (i_min:i_max + 1)[::-1]]
            i_star = i_min + int(np.argmax(products))
            j_star = k - i_star
            if i_star == j_star:
                # Reduce a[i_star]: conv[k] has term a[i_star]^2
                # d(conv[k])/d(a[i_star]) = 2*a[i_star]
                # But other conv values are also affected
                # Simple: reduce by small amount
                reduction = 0.01 * a[i_star]
                a[i_star] = max(0, a[i_star] - reduction)
            else:
                # Reduce the smaller of a[i_star], a[j_star]
                if a[i_star] <= a[j_star]:
                    reduction = 0.01 * a[i_star]
                    a[i_star] = max(0, a[i_star] - reduction)
                else:
                    reduction = 0.01 * a[j_star]
                    a[j_star] = max(0, a[j_star] - reduction)
            improved = True

        if not improved:
            break

        # Renormalize and check
        s_new = float(np.sum(a))
        if s_new < 1e-10:
            a = best_a.copy()
            break
        a = a / s_new
        obj_new = 2.0 * n * float(np.max(compute_conv(a)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        # Smooth periodic
        x = np.linspace(0, 2 * np.pi, n)
        a = 1.0 + 0.3 * np.sin(x) + 0.15 * np.sin(2 * x + 0.5)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Beta-like distribution
        a = (np.arange(n) + 1) ** 0.5 * (n - np.arange(n)) ** 0.5
        a = a / a.sum()
    elif mode == 6:
        # Smooth random
        a = rng.uniform(0.5, 1.5, n)
        # Smooth
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    from scipy.interpolate import interp1d
    func = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    a_new = func(x_new)
    return np.maximum(a_new, 0)

            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")
            best_seq = fast_cd(best_seq, time_budget=min(60, remaining * 0.1),
                               verbose=False, seed=seed + 1)
            obj_cd2 = 2.0 * best_seq.size * float(np.max(compute_conv(best_seq / best_seq.sum())))
            if obj_cd2 < best_score:
                best_score = obj_cd2

    # === Phase 2: Nesterov Smoothed Descent ===
    print(f"\n=== Phase 2: Nesterov Smoothed (budget: ~20%) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.20, 250.0)
    if p2_budget > 30:
        # Try from current best with perturbation
        pert = rng.normal(0, 0.02, len(best_seq))
        a_pert = best_seq * np.exp(pert)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc1 = p2_budget * 0.6
        alloc2 = p2_budget * 0.4
        a_nest = nesterov_smoothed(a_pert, time_budget=alloc1,
                                   tau_init=20.0, tau_end=250.0,
                                   verbose=True, seed=seed)
        obj_nest = 2.0 * a_nest.size * float(np.max(compute_conv(a_nest / a_nest.sum())))
        if obj_nest < best_score:
            best_score = obj_nest
            best_seq = a_nest.copy()
            print(f"  Nesterov 1: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov 1: obj={obj_nest:.8f}")

        # Try a second Nesterov with different params
        a_nest2 = nesterov_smoothed(best_seq, time_budget=alloc2,
                                    tau_init=50.0, tau_end=400.0,
                                    verbose=False, seed=seed + 100)
        obj_nest2 = 2.0 * a_nest2.size * float(np.max(compute_conv(a_nest2 / a_nest2.sum())))
        if obj_nest2 < best_score:
            best_score = obj_nest2
            best_seq = a_nest2.copy()
            print(f"  Nesterov 2: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov 2: obj={obj_nest2:.8f}")

    # === Phase 3: Active Set Polish + CD ===
    print(f"\n=== Phase 3: Active Set + CD (budget: ~15%) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)
    if p3_budget > 20:
        a_as = active_set_polish(best_seq, time_budget=min(30, p3_budget * 0.3), verbose=False)
        obj_as = 2.0 * a_as.size * float(np.max(compute_conv(a_as / a_as.sum())))
        if obj_as < best_score:
            best_score = obj_as
            best_seq = a_as.copy()
            print(f"  Active set: obj={best_score:.8f} **")

        # Follow up with CD
        a_cd3 = fast_cd(best_seq, time_budget=p3_budget * 0.7, verbose=False, seed=seed + 200)
        obj_cd3 = 2.0 * a_cd3.size * float(np.max(compute_conv(a_cd3 / a_cd3.sum())))
        if obj_cd3 < best_score:
            best_score = obj_cd3
            best_seq = a_cd3.copy()
            print(f"  CD after AS: obj={best_score:.8f} **")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration (budget: ~12%) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.12, 180.0)
    n_best = len(best_seq)

    n_options = set()
    for delta in [50, 100, 200, 300, 500]:
        n_options.add(n_best + delta)
        n_options.add(n_best - delta)
    # Also try some round numbers
    for n_try in [600, 700, 900, 1000, 1200, 1500]:
        if abs(n_try - n_best) > 30:
            n_options.add(n_try)
    n_options = sorted([n for n in n_options if 400 <= n <= 2000])
    n_options = n_options[:6]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Quick CD
        alloc = min(remaining * 0.2, 50.0)
        if alloc > 15:
            a_try = fast_cd(a_try, time_budget=alloc, verbose=False, seed=n_try)

        obj_try = 2.0 * a_try.size * float(np.max(compute_conv(a_try / a_try.sum())))
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts ===
    print(f"\n=== Phase 5: Diverse Restarts (budget: ~8%) ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.08, 100.0)
    n_current = len(best_seq)

    if p5_budget > 30:
        for r in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            mode = 2 + r
            a_r = diverse_init(n_current, rng, mode=mode)
            blend = 0.4 + 0.15 * r
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            alloc = min(p5_budget / 3, remaining * 0.3)
            if alloc > 20:
                a_r = fast_cd(a_r, time_budget=alloc, verbose=False, seed=5000 + r * 100)
                obj_r = 2.0 * a_r.size * float(np.max(compute_conv(a_r / a_r.sum())))
                if obj_r < best_score:
                    best_score = obj_r
                    best_seq = a_r.copy()
                    print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
                else:
                    print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 6: Final CD Polish ===
    print(f"\n=== Phase 6: Final CD Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.6, 200.0)
        a_final = fast_cd(best_seq, time_budget=alloc, verbose=False, seed=seed + 9999)
        obj_final = 2.0 * a_final.size * float(np.max(compute_conv(a_final / a_final.sum())))
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # === Phase 7: One more Nesterov if time ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_nest_final = nesterov_smoothed(best_seq, time_budget=min(remaining * 0.5, 60.0),
                                         tau_init=100.0, tau_end=500.0, verbose=False, seed=7777)
        obj_nf = 2.0 * a_nest_final.size * float(np.max(compute_conv(a_nest_final / a_nest_final.sum())))
        if obj_nf < best_score:
            best_score = obj_nf
            best_seq = a_nest_final.copy()
            print(f"  Final Nesterov: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
