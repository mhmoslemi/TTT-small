import numpy as np
import time

from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def multi_lag_subgradient(a, top_k=5):
    """
    Compute a weighted subgradient using top-k lags of the convolution.
    Returns gradient vector of length n.
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Get top-k lags
    if top_k >= len(conv):
        top_idx = np.arange(len(conv))
        weights = np.ones(len(top_idx))
    else:
        top_idx = np.argsort(conv)[::-1][:top_k]
        top_vals = conv[top_idx]
        # Softmax weights
        exp_vals = np.exp(top_vals - top_vals.max())
        weights = exp_vals / exp_vals.sum()
    
    # Compute gradient: for each lag k with weight w_k,
    # dc_k/da[i] = a[k-i] (if valid) + a[i-k] (if valid)
    grad = np.zeros(n)
    for k_idx, w in zip(top_idx, weights):
        k = int(k_idx)
        for i in range(n):
            j1 = k - i
            j2 = i - k
            if 0 <= j1 < n:
                grad[i] += w * a[j1]
            if 0 <= j2 < n:
                grad[i] += w * a[j2]
    
    return grad, conv

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, max_time=10.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.001, 0.999, n)
    a = 1.0 / np.sqrt(np.maximum(1e-10, 1 - (2 * x - 1) ** 2))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def power_u_init(n, p=2.0):
    """Power U-shape: (|2x-1|^p + eps)."""
    x = np.linspace(0, 1, n)
    a = np.abs(2 * x - 1) ** p + 0.01
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = np.abs(2 * x - 1) ** 3 + 0.02
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 8) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 8) ** 2)) + 0.01
    elif mode == 2:
        a = (2 * np.abs(x - 0.5)) ** 1.5 + 0.03
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(3 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / np.cosh(3.0 * (2 * x - 1)) ** 2
    elif mode == 6:
        a = (1 - np.cos(2 * np.pi * x)) ** 1.5 + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(a, n_new):
    """Interpolate sequence a to length n_new."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return a_new / s


def optimize_at_length(a, time_budget, rng, verbose=False):
    """Full optimization pipeline for a given sequence."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Phase A: Fast projected gradient (60% of budget)
    pgd_budget = time_budget * 0.5
    if pgd_budget > 5:
        a_pgd = projected_gradient_descent(best_a, time_budget=pgd_budget, verbose=verbose)
        obj_pgd = compute_obj(a_pgd)
        if obj_pgd < best_obj:
            best_obj = obj_pgd
            best_a = a_pgd.copy()
            if verbose:
                print(f"        PGD: obj={best_obj:.10f}")
    
    # Phase B: LP polishing (25% of budget)
    remaining = time_budget - (time.monotonic() - start)
    lp_budget = min(remaining * 0.5, 30.0)
    if lp_budget > 5:
        a_lp = lp_polish(best_a, time_budget=lp_budget, max_iters=10, verbose=verbose)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            if verbose:
                print(f"        LP: obj={best_obj:.10f}")
    
    # Phase C: Second PGD pass on LP result (remaining budget)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_pgd2 = projected_gradient_descent(best_a, time_budget=remaining * 0.8, verbose=False)
        obj_pgd2 = compute_obj(a_pgd2)
        if obj_pgd2 < best_obj:
            best_obj = obj_pgd2
            best_a = a_pgd2.copy()
    
    # Phase D: Final LP if time permits
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a_lp2 = lp_polish(best_a, time_budget=min(remaining * 0.7, 20.0), max_iters=5, verbose=False)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Hybrid gradient-LP optimization with systematic n exploration.
    Uses fast projected gradient for exploration, LP for polishing,
    and multi-start at different lengths.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Try to load parent
    parent_loaded = False
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                parent_loaded = True
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception:
        pass
    
    if not parent_loaded:
        best_seq = u_shaped_init(800, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default U-shaped init: obj={best_score:.10f}")
    
    # === Phase 1: Optimize at parent length ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 1: Optimize n={len(best_seq)} (budget={remaining:.0f}s) ===")
        alloc = min(remaining * 0.3, 300.0)
        a_opt, obj_opt = optimize_at_length(best_seq, alloc, rng, verbose=True)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase1 result: obj={best_score:.10f}")
    
    # === Phase 2: Try different initializations at same length ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: Alternative inits (budget={remaining:.0f}s) ===")
        n_cur = len(best_seq)
        inits = [
            ("u3", u_shaped_init(n_cur, 3.0)),
            ("u5", u_shaped_init(n_cur, 5.0)),
            ("arc", arcsine_init(n_cur)),
            ("p1", power_u_init(n_cur, 1.5)),
            ("p3", power_u_init(n_cur, 3.0)),
            ("d0", diverse_init(n_cur, rng, 0)),
            ("d4", diverse_init(n_cur, rng, 4)),
            ("d5", diverse_init(n_cur, rng, 5)),
        ]
        
        # Quick evaluate all
        init_objs = []
        for name, init in inits:
            obj_i = compute_obj(init)
            init_objs.append((obj_i, name, init))
            print(f"    {name}: obj={obj_i:.10f}")
        
        # Optimize the best 2 that aren't already the parent
        init_objs.sort()
        tried = 0
        for obj_i, name, init in init_objs:
            if tried >= 2:
                break
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            # Skip if very close to current best
            if abs(obj_i - best_score) < 1e-8:
                continue
            alloc = min(remaining * 0.3, 120.0)
            a_opt, obj_opt = optimize_at_length(init, alloc, rng, verbose=False)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"    {name} -> obj={best_score:.10f} **")
            else:
                print(f"    {name} -> obj={obj_opt:.10f}")
            tried += 1
    
    # === Phase 3: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 3: n Exploration (budget={remaining:.0f}s) ===")
        n_best = len(best_seq)
        
        # Generate candidate n values
        n_candidates = set()
        for delta in [-200, -100, -50, 50, 100, 200, 300, 400, 500]:
            n_try = n_best + delta
            if 300 <= n_try <= 3000:
                n_candidates.add(n_try)
        # Add some fixed values
        for n_fix in [500, 600, 700, 900, 1000, 1200, 1500, 2000, 2500]:
            if abs(n_fix - n_best) > 50:
                n_candidates.add(n_fix)
        
        # Sort by proximity to current best (try nearby first)
        n_candidates = sorted([n for n in n_candidates if n != n_best], 
                             key=lambda x: abs(x - n_best))
        n_candidates = n_candidates[:8]  # Limit to 8 tries
        print(f"  n candidates: {n_candidates}")
        
        for n_try in n_candidates:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            
            # Interpolate and optimize
            a_interp = interpolate_sequence(best_seq, n_try)
            alloc = min(remaining * 0.4, 100.0)
            
            a_opt, obj_opt = optimize_at_length(a_interp, alloc, rng, verbose=False)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.10f} **")
            else:
                print(f"  n={n_try}: obj={obj_opt:.10f}")
    
    # === Phase 4: Intensive final polish at best length ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 4: Final Polish (budget={remaining:.0f}s) ===")
        alloc = min(remaining * 0.6, 200.0)
        a_final, obj_final = optimize_at_length(best_seq, alloc, rng, verbose=True)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final: obj={best_score:.10f} **")
    
    # === Phase 5: Quick extra attempts ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 5: Extra Attempts ===")
        # Try a few more perturbed variants
        for attempt in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            noise_scale = 0.01 * (attempt + 1)
            a_pert = best_seq + noise_scale * rng.standard_normal(len(best_seq))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            
            alloc = min(remaining * 0.4, 40.0)
            a_opt, obj_opt = optimize_at_length(a_pert, alloc, rng, verbose=False)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Perturb {attempt}: obj={best_score:.10f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    
    return result
