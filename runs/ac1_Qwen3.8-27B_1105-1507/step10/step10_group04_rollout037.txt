Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses KKT Newton method with active set identification, Anderson-accelerated
LP iteration, symmetry exploitation, and multi-n exploration.

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_gradient(a, weights):
    """Compute gradient of sum_k weights[k] * (a*a)_k w.r.t. a.
    Returns array of length n.
    gradient_j = 2 * sum_k weights[k] * a[k-j] (with a[i]=0 for i out of range)
    """
    n = len(a)
    m = len(weights)
    # Use FFT: gradient = 2 * ifft(FFT(weights) * conj(FFT(a)))[:n]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    W = fft(weights, size)
    A = fft(a, size)
    grad = ifft(W * np.conj(A)).real[:n]
    return 2.0 * grad

    n = len(a)

        A[lo:hi, j] = a_norm[:hi - lo]

    #      sum(g) = 1
    #      g >= 0

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    g_val = None
    M_val = None

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):

    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)

def place_weights(weights, positions, total_len):
    """Place weights at given positions in a vector of length total_len."""
    out = np.zeros(total_len)
    for i, pos in enumerate(positions):
        if 0 <= pos < total_len:
            out[pos] = weights[i]
    return out

    Parameters:

    verbose: print progress
    
    Returns: optimized sequence

        return a

        S = np.argsort(conv)[::-1][:max(3, n // 20)]

    lam = np.ones(m) / m
    
    it = 0
    converged = False

        conv = compute_conv(a)
        M_curr = float(np.max(conv))

            # Use a rolling window around the current max
            top_frac = max(0.05, min(0.4, 5.0 / M_curr / n * 10))
            active_thresh = M_curr - 0.01 * M_curr / n
            S = np.where(conv >= active_thresh)[0]
            if len(S) < 3:
                S = np.argsort(conv)[::-1][:max(3, n // 20)]
            m = len(S)
            # Reinitialize lam if active set changed significantly
            lam = np.ones(m) / m
        
        # Compute residual
        # r1: (a*a)_k - M for k in S
        r1 = conv[S] - M_curr  # m x 1
        
        # r2: sum(a) - 1 (should be ~0)
        r2 = float(np.sum(a)) - 1.0

        # c_j = 2 * sum_{k in S} lam[i] * a[k-j] where S[i] = k
        p = place_weights(lam, S, 2 * n - 1)

            break

        # Variables: [a (n), lam (m), M (1)]
        # Equations: [r1 (m), r2 (1), r3 (n-1), r4 (1)]

        J[m + n, n : n + m] = 1.0

        try:
            delta = np.linalg.solve(J, -r)
        except np.linalg.LinAlgError:

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP warm-start with dual-based active set identification, then KKT Newton
method for quadratic convergence, plus multi-n exploration and gradient polishing.
Key innovation: direct dense Newton solve on the KKT system with proper active
set management from LP dual variables, achieving machine-precision solutions.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp_minimax(a, verbose=False):
    """Solve LP: min M s.t. (a_norm*g)_k <= M, sum(g)=1, g>=0.
    Returns (g, M, dual_weights) where dual_weights are normalized non-negative."""
    n = len(a)
    a_norm = a / float(np.sum(a))

    # Build A matrix: A[k,j] = a_norm[k-j] if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a_norm[:hi - lo]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    g_val = None
    M_val = None
    dual_vals = None

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                M_val = float(M.value)
                try:
                    dual_vals = np.abs(np.array(constraints[0].dual_value, dtype=np.float64))
                    dual_sum = float(np.sum(dual_vals))
                    if dual_sum > 1e-10:
                        dual_vals = dual_vals / dual_sum
                    else:
                        dual_vals = np.ones(2 * n - 1) / (2 * n - 1)
                except Exception:
                    dual_vals = np.ones(2 * n - 1) / (2 * n - 1)
                break
        except Exception:
            continue

    if g_val is None:
        g_val = np.ones(n) / n
        conv = compute_conv(g_val)
        M_val = float(np.max(conv))
        dual_vals = np.ones(2 * n - 1) / (2 * n - 1)

    return g_val, M_val, dual_vals


def lp_iteration(a, time_budget=100.0, verbose=True):
    """LP-based minimax iteration with Anderson acceleration.
    Returns (best_a, active_set, dual_weights)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.arange(n), np.ones(n)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Anderson acceleration storage
    history_x = []
    history_f = []
    AA_order = 5
    it = 0
    last_dual = None

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g, M_lp, dual = solve_lp_minimax(a, verbose=False)
        if g is None:
            break
        last_dual = dual

        f = g - a

        # Anderson acceleration
        x_new = g
        if len(history_x) >= 2:
            m = min(AA_order, len(history_x))
            X = np.array(history_x[-m:])
            F = np.array(history_f[-m:])
            # Minimize ||F @ omega||^2 s.t. sum(omega)=1
            # KKT: (F^T F + eps*I) omega + lambda*1 = 0, 1^T omega = 1
            FtF = F.T @ F + 1e-12 * np.eye(m)
            ones = np.ones(m)
            # Solve augmented system
            aug = np.zeros((m + 1, m + 1))
            aug[:m, :m] = FtF
            aug[:m, m] = ones
            aug[m, :m] = ones
            rhs = np.zeros(m + 1)
            rhs[m] = 1.0
            try:
                sol = np.linalg.solve(aug, rhs)
                omega = sol[:m]
                omega = np.maximum(omega, 0)
                s_om = float(np.sum(omega))
                if s_om > 1e-10:
                    omega = omega / s_om
                    x_new = omega @ X
            except np.linalg.LinAlgError:
                x_new = X[-1] + F[-1]

        # Project to simplex
        x_new = np.maximum(x_new, 0)
        s_new = float(np.sum(x_new))

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new

        a = x_new

        S = np.where(active_mask)[0]

    else:
        S = np.where(conv > M_max * 0.99)[0]

    if len(S) < 3:
        S = np.argsort(conv)[::-1][:max(3, n // 20)]

    return best_a, S, last_dual if last_dual is not None else np.ones(2 * n - 1) / (2 * n - 1)


def kkt_newton(a_init, active_set, time_budget=300.0, verbose=True):
    """Solve KKT system using damped Newton's method with active set management."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    S = active_set.copy()
    m = len(S)

    conv = compute_conv(a)
    M = float(np.max(conv))
    lam = np.ones(m) / m

    it = 0
    stall_count = 0
    last_improve = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        conv = compute_conv(a)
        M_curr = float(np.max(conv))

        # Periodically update active set
        if it % 15 == 0 and it > 0:
            # Use positions where conv is very close to max
            thresh = M_curr - max(1e-10, M_curr * 1e-6)
            S_new = np.where(conv >= thresh)[0]
            if len(S_new) < 3:
                S_new = np.argsort(conv)[::-1][:max(3, n // 25)]
            if len(S_new) != m:
                S = S_new
                m = len(S)
                lam = np.ones(m) / m

        # Compute residual
        # Variables: [a (n), lam (m), M (1)] → n+m+1 unknowns
        # Equations:
        #   r1: (a*a)_k - M = 0 for k in S (m eqs)
        #   r2: sum(a) - 1 = 0 (1 eq)
        #   r3: c_j - c_0 = 0 for j=1..n-1 (n-1 eqs)
        #   r4: sum(lam) - 1 = 0 (1 eq)
        # Total: m + 1 + n-1 + 1 = m+n+1 ✓

        # r1
        r1 = conv[S] - M_curr

        # r2
        r2_val = float(np.sum(a)) - 1.0

        # r3: c_j = 2 * sum_k p_k * a_{k-j} where p has lam at positions S
        p = np.zeros(2 * n - 1)
        for i, k in enumerate(S):
            if 0 <= k < 2 * n - 1:
                p[k] = lam[i]

        # Compute c via FFT
        fft_size = 1
        while fft_size < 3 * n:
            fft_size <<= 1
        P_f = fft(p, fft_size)
        A_f = fft(a, fft_size)
        c = (2.0 * ifft(P_f * np.conj(A_f)).real)[:n]
        r3 = c[1:] - c[0]

        # r4
        r4_val = float(np.sum(lam)) - 1.0

        # Full residual
        r = np.concatenate([r1, [r2_val], r3, [r4_val]])
        r_norm = np.max(np.abs(r))

        if r_norm < 1e-15:
            if verbose:
                print(f"    KKT converged at it{it}, obj={best_obj:.12f}")
            break

        # Build Jacobian
        J = np.zeros((m + n + 1, n + m + 1))

        # Block (r1, a): m x n — J[i,j] = 2*a[S[i]-j] if valid
        for i in range(m):
            k = S[i]
            for j in range(n):
                d = k - j
                if 0 <= d < n:
                    J[i, j] = 2.0 * a[d]

        # Block (r1, M): m x 1
        J[:m, n + m] = -1.0

        # Block (r2, a): 1 x n
        J[m, :n] = 1.0

        # Block (r3, a): (n-1) x n — J[row,l] = 2*(p[j+l] - p[l])
        for j in range(1, n):
            row = m + 1 + (j - 1)
            for l in range(n):
                val = 0.0
                idx1 = j + l
                if 0 <= idx1 < 2 * n - 1:
                    val += p[idx1]
                val -= p[l] if l < 2 * n - 1 else 0.0
                J[row, l] = 2.0 * val

        # Block (r3, lam): (n-1) x m — J[row,i] = 2*(a[S[i]-j] - a[S[i]])
        for j in range(1, n):
            row = m + 1 + (j - 1)
            for i in range(m):
                k = S[i]
                val = 0.0
                d1 = k - j
                if 0 <= d1 < n:
                    val += a[d1]
                if 0 <= k < n:
                    val -= a[k]
                J[row, n + i] = 2.0 * val

        # Block (r4, lam): 1 x m
        J[m + n, n:n + m] = 1.0

        # Solve linear system
        try:
            delta = np.linalg.solve(J, -r)
        except np.linalg.LinAlgError:
            try:
                delta, _, _, _ = np.linalg.lstsq(J, -r, rcond=1e-12)
            except Exception:
                break

        delta_a = delta[:n]
        delta_lam = delta[n:n + m]
        delta_M = delta[n + m]

        # Damped update
        t = 1.0
        success = False
        for bt in range(40):
            a_try = a + t * delta_a
            lam_try = lam + t * delta_lam
            M_try = M_curr + t * delta_M

            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                t *= 0.5
                continue
            a_try = a_try / s_try

            lam_try = np.maximum(lam_try, 0)
            s_lam = float(np.sum(lam_try))
            if s_lam < 1e-10:
                lam_try = np.ones(m) / m
            else:
                lam_try = lam_try / s_lam

            obj_try = compute_obj(a_try)
            if obj_try <= best_obj + 1e-13:
                a = a_try
                lam = lam_try
                M_curr = M_try
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    stall_count = 0
                else:
                    stall_count += 1
                success = True
                break
            t *= 0.5

        if not success:
            stall_count += 5
            # Try perturbation
            if stall_count > 20:
                break

        it += 1
        if verbose and it % 25 == 0:
            elapsed = time.monotonic() - start
            print(f"    KKT it{it}: obj={best_obj:.12f}, |r|={r_norm:.2e}, m={m}, t={elapsed:.1f}s")

    return best_a


def gradient_polish(a, time_budget=60.0, tau_max=5000, verbose=True):
    """Smoothed gradient descent polishing with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    v = np.zeros(n)

    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = 100 + (tau_max - 100) * progress ** 0.7

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = exp_vals / sum_exp

        fft_size = 1
        while fft_size < 3 * n:
            fft_size <<= 1
        P_f = fft(p, fft_size)
        A_f = fft(y, fft_size)
        g = (2.0 * ifft(P_f * np.conj(A_f)).real)[:n]

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        lr = min(1.0 / (g_norm + 1e-10), 3.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1
        beta = min(0.97, beta + 5e-5)

        if it % 5000 == 0 and verbose:
            print(f"    GD it{it}: obj={best_obj:.10f}, tau={tau:.0f}, t={elapsed:.0f}s")

        if it > 100000:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.5
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
        a = a / a.sum()
    else:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP warm-start → gradient polish → KKT Newton → multi-n → final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default uniform init (n={n_init}): obj={best_score:.10f}")

    n_best = len(best_seq)

    # === Phase 1: LP Iteration (150s) ===
    print(f"\n=== Phase 1: LP Iteration ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.15, 150.0)
    active_set = None
    if p1_budget > 20:
        a_lp, active_set, _ = lp_iteration(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 done: obj={best_score:.10f} **")
        else:
            print(f"  Phase1 done: obj={obj_lp:.10f}")

    # === Phase 2: Gradient Polish (80s) ===
    print(f"\n=== Phase 2: Gradient Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.08, 80.0)
    if p2_budget > 15:
        a_gp = gradient_polish(best_seq, time_budget=p2_budget, tau_max=3000, verbose=True)
        obj_gp = compute_obj(a_gp)
        if obj_gp < best_score:
            best_score = obj_gp
            best_seq = a_gp.copy()
            print(f"  Phase2 done: obj={best_score:.10f} **")

    # === Phase 3: KKT Newton (350s) ===
    print(f"\n=== Phase 3: KKT Newton ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.3, 350.0)
    if p3_budget > 30:
        if active_set is None:
            conv = compute_conv(best_seq)
            M_max = float(np.max(conv))
            active_set = np.where(conv >= M_max * 0.999)[0]
            if len(active_set) < 5:
                active_set = np.argsort(conv)[::-1][:max(5, len(best_seq) // 20)]

        a_kkt = kkt_newton(best_seq, active_set, time_budget=p3_budget, verbose=True)
        obj_kkt = compute_obj(a_kkt)
        if obj_kkt < best_score:
            best_score = obj_kkt
            best_seq = a_kkt.copy()
            print(f"  Phase3 done: obj={best_score:.10f} **")
        else:
            print(f"  Phase3 done: obj={obj_kkt:.10f}")

    # === Phase 4: Multi-n Exploration (400s) ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.35, 400.0)
    n_best = len(best_seq)

    n_opts = sorted(set([
        n_best - 50, n_best - 20, n_best,
        n_best + 20, n_best + 50, n_best + 100,
        n_best + 200, n_best + 400, n_best + 800,
        600, 700, 900, 1000, 1200, 1500, 2000
    ]))
    n_opts = [n for n in n_opts if 400 <= n <= 3000 and n != n_best]
    n_opts = sorted(n_opts, key=lambda x: abs(x - n_best))[:6]
    print(f"  n options: {n_opts}")

    per_n = p4_budget / max(1, len(n_opts))
    for n_try in n_opts:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        alloc = min(per_n, remaining - 40)
        if alloc < 40:
            break

        print(f"  ** Trying n={n_try} (alloc={alloc:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try > 1e-10:
            a_try = a_try / s_try

        # Quick LP + GD + KKT
        lp_b = min(alloc * 0.3, 60.0)
        if lp_b > 10:
            a_try, act_set, _ = lp_iteration(a_try, time_budget=lp_b, verbose=False)

        gd_b = min(alloc * 0.25, 50.0)
        if gd_b > 8:
            remaining_n = deadline - time.monotonic()
            gd_b = min(gd_b, remaining_n - 30)
            if gd_b > 5:
                a_try = gradient_polish(a_try, time_budget=gd_b, tau_max=2000, verbose=False)

        kkt_b = min(alloc * 0.4, 100.0)
        if kkt_b > 15:
            remaining_n = deadline - time.monotonic()
            kkt_b = min(kkt_b, remaining_n - 25)
            if kkt_b > 10:
                if act_set is None:
                    conv_t = compute_conv(a_try)
                    M_t = float(np.max(conv_t))
                    act_set = np.where(conv_t >= M_t * 0.999)[0]
                    if len(act_set) < 3:
                        act_set = np.argsort(conv_t)[::-1][:max(3, n_try // 20)]
                a_try = kkt_newton(a_try, act_set, time_budget=kkt_b, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 5: Final Intensive Polish (remaining) ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s")

    # KKT with refined active set
    if remaining > 30:
        conv = compute_conv(best_seq)
        M_max = float(np.max(conv))
        # Tighter active set
        active_refined = np.where(conv >= M_max - 1e-10)[0]
        if len(active_refined) < 3:
            active_refined = np.where(conv >= M_max * (1 - 1e-6))[0]
        if len(active_refined) < 3:
            active_refined = np.argsort(conv)[::-1][:max(5, len(best_seq) // 15)]

        alloc = min(remaining * 0.4, 120.0)
        a_fin = kkt_newton(best_seq, active_refined, time_budget=alloc, verbose=True)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final KKT: obj={best_score:.10f} **")

    # Gradient polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 60.0)
        a_fin = gradient_polish(best_seq, time_budget=alloc, tau_max=10000, verbose=True)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final GD: obj={best_score:.10f} **")

    # One more KKT
    remaining = deadline - time.monotonic()
    if remaining > 20:
        conv = compute_conv(best_seq)
        M_max = float(np.max(conv))
        active_final = np.where(conv >= M_max - 1e-12)[0]
        if len(active_final) < 3:
            active_final = np.argsort(conv)[::-1][:max(5, len(best_seq) // 15)]
        alloc = min(remaining * 0.5, 80.0)
        a_fin = kkt_newton(best_seq, active_final, time_budget=alloc, verbose=True)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Ultra-final KKT: obj={best_score:.10f} **")

    # Diverse restarts if time remains
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: Diverse Restarts ===")
        n_cur = len(best_seq)
        n_restarts = min(3, int(remaining / 50))
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 35:
                break
            alloc = min(45, remaining - 10)
            if alloc < 20:
                break

            mode = r % 5
            a_r = diverse_init(n_cur, rng, mode=mode)
            blend = 0.5
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r

            # Quick optimization
            half = alloc * 0.5
            a_r, act_r, _ = lp_iteration(a_r, time_budget=half, verbose=False)
            remaining_r = deadline - time.monotonic()
            kkt_alloc = min(alloc * 0.4, remaining_r - 10)
            if kkt_alloc > 8 and act_r is not None:
                a_r = kkt_newton(a_r, act_r, time_budget=kkt_alloc, verbose=False)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.10f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
