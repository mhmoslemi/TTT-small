        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def gradient_descent(a, time_budget=100.0, tau_init=50.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)

        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

    return best_a


def peak_targeting(a, time_budget=60.0, verbose=False):
    """Directly attack the convolution peak by redistributing mass."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        peak_val = float(conv[peak_idx])

        # Find contributors to the peak
        # conv[peak_idx] = sum_j a[j] * a[peak_idx - j] for valid j
        j_lo = max(0, peak_idx - n + 1)
        j_hi = min(peak_idx, n - 1)
        contributors = np.arange(j_lo, j_hi + 1)
        if len(contributors) == 0:
            break
        partner = peak_idx - contributors
        products = a[contributors] * a[partner]

        # Find the top contributors
        top_k = min(5, len(contributors))
        top_indices = np.argsort(products)[::-1][:top_k]

        improved = False
        for idx in top_indices:
            j = contributors[idx]
            p = partner[idx]
            if j == p:
                continue
            # Transfer mass from the larger to a position that doesn't hit the peak
            if a[j] > a[p]:
                src = j
                dst_val = a[p]
            else:
                src = p
                dst_val = a[j]

            # Try moving mass from src to various positions
            for frac in [0.05, 0.1, 0.15]:
                trial = a.copy()
                amount = trial[src] * frac
                trial[src] -= amount
                # Find a position that doesn't contribute to the peak
                for dst in range(n):
                    if dst == src:
                        continue
                    # Check if dst contributes to the peak
                    if j_lo <= dst <= j_hi or j_lo <= (peak_idx - dst) <= j_hi:
                        continue
                    trial[dst] += amount
                    if trial[dst] < 0 or trial[src] < 0:
                        continue
                    s_trial = float(np.sum(trial))
                    if s_trial < 1e-10:
                        continue
                    trial = trial / s_trial
                    obj_trial = compute_obj(trial)
                    if obj_trial < best_obj - 1e-10:
                        best_obj = obj_trial
                        best_a = trial.copy()
                        a = trial.copy()
                        improved = True
                        break

        if not improved:
            # Try reducing the peak contributors directly
            trial = a.copy()
            j = contributors[top_indices[0]]
            p = partner[top_indices[0]]
            # Reduce both j and p slightly, add to a non-peak position
            amount = min(a[j], a[p]) * 0.1
            trial[j] -= amount
            trial[p] -= amount
            # Add to position 0 or n-1 (edges)
            half = amount
            trial[half > 0.1 * n]  # just to avoid unused warning
            # Spread the mass
            add_pos = (peak_idx + n) % (2 * n - 1)  # position far from peak
            if add_pos >= n:
                add_pos = 0
            trial[add_pos % n] += 2 * amount
            trial = np.maximum(trial, 0)
            s_trial = float(np.sum(trial))
            if s_trial > 1e-10:
                trial = trial / s_trial
                obj_trial = compute_obj(trial)
                if obj_trial < best_obj - 1e-10:
                    best_obj = obj_trial
                    best_a = trial.copy()
                    a = trial.copy()

        if verbose and (int(time.monotonic() - start) % 10 == 0):
            print(f"      Peak targeting: obj={best_obj:.8f}, peak={peak_val:.6f}, idx={peak_idx}")

    return best_a


def tilted_symmetric_init(n, tilt=0.3, width=0.5, rng=None):
    """Create a tilted symmetric initial sequence."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.arange(n) / max(n - 1, 1)
    center = 0.5
    # Symmetric base
    base = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    # Apply tilt (breaks symmetry)
    tilt_factor = np.exp(tilt * (x - center))
    a = base * tilt_factor
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def cosine_init(n, mode=1, rng=None):
    """Initialize with cosine structure."""
    x = np.arange(n) / max(n - 1, 1)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * mode * x)
    a = np.maximum(a, 0.01)
    s = float(np.sum(a))
    a = a / s
    return a


def lp_fictitious_play(a, time_budget=400.0, verbose=False, seed=42):
    """
    LP-based fictitious play with running average.
    Iterates: solve LP for best response g, update a = blend(a, g),
    track running average as candidate solution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Running average (fictitious play)
    avg_a = a.copy()
    count = 0

    # Momentum
    momentum_a = a.copy()
    beta = 0.7

    prev_g = None
    it = 0
    best_it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        g, M_val = solve_lp(a)
        if g is None:
            break

        it += 1
        count += 1

        # Update running average
        avg_a = ((count - 1) * avg_a + g) / count

        # Check objective of various candidates
        obj_avg = compute_obj(avg_a)
        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = avg_a.copy()
            best_it = it

        # Momentum update for working iterate
        momentum_a = beta * momentum_a + (1 - beta) * g

        # Adaptive step size: start small, increase if improving
        if stall_count == 0:
            t_step = 0.3
        elif stall_count < 5:
            t_step = 0.2
        else:
            t_step = 0.1

        a_new = (1.0 - t_step) * momentum_a + t_step * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1
            # Try the LP solution directly
            obj_g = compute_obj(g)
            if obj_g < best_obj:
                best_obj = obj_g
                best_a = g.copy()
                best_it = it
                stall_count = 0

        a = a_new

        if verbose and it % 50 == 0:
            print(f"      FP it{it}: obj={best_obj:.8f}, avg_obj={compute_obj(avg_a):.8f}, M={M_val:.6f}, t={time.monotonic()-start:.0f}s")

        # Reset if stuck for too long
        if stall_count > 30:
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            momentum_a = a.copy()
            stall_count = 0
            prev_g = None

        # Also check avg_a periodically
        if it % 20 == 0:
            obj_check = compute_obj(avg_a)
            if obj_check < best_obj:
                best_obj = obj_check
                best_a = avg_a.copy()

    return best_a


def coordinate_descent(a, time_budget=60.0):
    """Simple coordinate descent: try adjusting each element slightly."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    conv = compute_conv(a)
    peak_idx = int(np.argmax(conv))

    while time.monotonic() - start < time_budget:
        improved = False
        # Focus on elements that contribute to the peak and its neighbors
        focus_range = set()
        for di in [-2, -1, 0, 1, 2]:
            k = peak_idx + di
            if 0 <= k < 2 * n - 1:
                j_lo = max(0, k - n + 1)
                j_hi = min(k, n - 1)
                focus_range.update(range(j_lo, j_hi + 1))

        # Also try random positions
        for _ in range(5):
            focus_range.add(rng.integers(0, n))

        for j in sorted(focus_range):
            for delta in [-0.01, -0.005, 0.005, 0.01]:
                trial = best_a.copy()
                old_val = trial[j]
                trial[j] = max(0, old_val + delta)
                # Redistribute: add lost mass to a random non-peak position
                lost = old_val - trial[j]
                if lost > 0:
                    # Add to a position far from peak
                    add_j = (j + n // 2) % n
                    trial[add_j] += lost
                s_trial = float(np.sum(trial))
                if s_trial < 1e-10:
                    continue
                trial = trial / s_trial
                obj_trial = compute_obj(trial)
                if obj_trial < best_obj - 1e-10:
                    best_obj = obj_trial
                    best_a = trial.copy()
                    improved = True
                    conv = compute_conv(best_a)
                    peak_idx = int(np.argmax(conv))

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
