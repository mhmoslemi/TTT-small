"""
Nesterov-Accelerated Subgradient + Peak Equalization Optimizer.
Uses Nesterov acceleration for the non-smooth max-convolution objective,
with soft peak weighting, adaptive step sizes, peak equalization,
and multi-start structured initializations.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def soft_subgradient(a, K=50, temperature=5.0):
    """Compute a soft subgradient of max_k (a*a)[k] using top-K peak weighting."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    if max_conv < 1e-15:
        return np.zeros(n)
    
    # Soft weighting: weight by exp(temperature * (conv[k] - max_conv) / max_conv)
    weights = np.exp(temperature * (conv - max_conv) / max_conv)
    weights /= weights.sum()
    
    # Subgradient: 2 * sum_k weights[k] * reverse_shift(a, k)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        # g[m] += 2 * weights[k] * a[k-m] for valid k-m
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] += 2 * weights[k] * a[j_start:j_end + 1][::-1]
    
    return g

def nesterov_subgrad(a, time_budget=200.0, K=50, temp_init=5.0, temp_end=50.0,
                      verbose=False, seed=42):
    """Nesterov-accelerated projected subgradient method."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    f0 = best_obj / (2.0 * n)  # this is max_conv at a
    # Actually, let's use the objective directly
    # The "gap" is f(x) - f* where f* is unknown. Use f0 as upper bound.
    
    t = 1
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        temp = temp_init * (temp_end / temp_init) ** progress
        K_use = max(10, int(K * (1 - 0.5 * progress)))
        
        # Nesterov extrapolation
        theta = (t - 1) / (t + 2) if t > 1 else 0
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = x.copy()
        else:
            y = y / s_y
        
        # Compute subgradient at y
        g = soft_subgradient(y, K=K_use, temperature=temp)
        
        # Step size: eta = f0 / (L * sqrt(t))
        # L is an upper bound on the Lipschitz constant of the subgradient
        # For our problem, L ≈ 2 * max(a) * (2n-1) roughly
        L = 2.0 * float(np.max(y)) * (2 * n)
        L = max(L, 1.0)
        eta = f0 / (L * np.sqrt(t))
        
        # Take step
        x_new = y - eta * g
        x_new = project_simplex(x_new)
        
        # Evaluate
        obj_new = compute_obj(x_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        x_prev = x
        x = x_new
        t += 1
        it += 1
        
        if verbose and it % 50 == 0:
            print(f"    NS it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s, temp={temp:.1f}")
    
    return best_a

def peak_equalize(a, time_budget=100.0, verbose=False):
    """Try to equalize the top peaks of the convolution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Find peaks above 0.95*M
        peak_idx = np.where(conv > 0.95 * M)[0]
        if len(peak_idx) == 0:
            break
        
        # Compute the "excess" of each peak over the mean of peaks
        peak_vals = conv[peak_idx]
        mean_peak = float(np.mean(peak_vals))
        
        # Direction: reduce high peaks, increase low peaks
        g = np.zeros(n)
        for k in peak_idx:
            excess = conv[k] - mean_peak
            if excess > 0:
                # Subtract: g -= excess * reverse_shift(a, k)
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                if i_start <= i_end:
                    j_start = k - i_end
                    j_end = k - i_start
                    g[i_start:i_end + 1] -= excess * a[j_start:j_end + 1][::-1]
        
        # Normalize direction
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Line search
        best_alpha = 0
        best_alpha_obj = best_obj
        for alpha in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
            a_try = a - alpha * g
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj - 1e-15:
                best_alpha_obj = obj_try
                best_alpha = alpha
        
        if best_alpha > 0:
            a = a - best_alpha * g
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-10:
                a = a / s_a
            if best_alpha_obj < best_obj - 1e-14:
                best_obj = best_alpha_obj
                best_a = a.copy()
        else:
            # Add small perturbation
            a_pert = a * np.exp(rng.normal(0, 0.005, n))
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-10:
                a = a_pert / s_pert
    
    return best_a

def structured_init(n, type_idx, rng):
    """Generate structured initial sequences."""
    if type_idx == 0:
        # Uniform
        return np.ones(n) / n
    elif type_idx == 1:
        # Raised cosine window
        x = np.linspace(0, 1, n)
        w = 0.5 * (1 + np.cos(2 * np.pi * x))
        return w / w.sum()
    elif type_idx == 2:
        # Gaussian window
        x = np.linspace(-3, 3, n)
        w = np.exp(-x**2 / 2)
        return w / w.sum()
    elif type_idx == 3:
        # Binary random
        b = rng.integers(0, 2, n).astype(float)
        if b.sum() < 1:
            b[0] = 1
        return b / b.sum()
    elif type_idx == 4:
        # Low-density binary
        b = (rng.random(n) < 0.3).astype(float)
        if b.sum() < 1:
            b[0] = 1
        return b / b.sum()
    elif type_idx == 5:
        # Sinusoidal
        x = np.arange(n)
        w = 1.0 + 0.8 * np.cos(2 * np.pi * x / n * rng.integers(1, 5))
        w = np.maximum(w, 0)
        if w.sum() < 1e-10:
            return np.ones(n) / n
        return w / w.sum()
    elif type_idx == 6:
        # Fejer-like (sinc squared)
        x = np.arange(n) - n / 2
        w = np.sinc(x / (n / 4))**2
        w = np.maximum(w, 0)
        if w.sum() < 1e-10:
            return np.ones(n) / n
        return w / w.sum()
    elif type_idx == 7:
        # Two-lobe
        x = np.arange(n)
        center = n / 2
        w = np.exp(-((x - center)**2) / (n/6)**2) + 0.5 * np.exp(-((x - center)**2) / (n/12)**2)
        return w / w.sum()
    else:
        return np.ones(n) / n

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

Combines fast Nesterov-accelerated smoothed subgradient (with temperature annealing)

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, tau):
    """Compute the gradient of the smoothed max: sum_k p_k * grad_k where p_k = softmax(tau*conv)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    # Gradient via FFT: grad = 2 * ifft(FFT(p) * conj(FFT(a)))[:n]
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g
