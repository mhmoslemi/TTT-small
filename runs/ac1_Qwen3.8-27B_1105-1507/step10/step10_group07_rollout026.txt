    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.5 / n  # initial learning rate
    momentum = 0.9
    v = np.zeros(n)
    
    while time.monotonic() - start < time_budget:
        # Compute convolution via FFT
        conv = compute_conv(a)
        
        # Find top peaks with soft weighting
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        peak_vals = conv[peak_indices]
        max_peak = peak_vals[0]
        
        # Soft weighting: exp((peak - max) / temperature)
        temp = max_peak * 0.1
        weights = np.exp((peak_vals - max_peak) / temp)
        weights /= weights.sum()
        
        # Compute subgradient
        g = np.zeros(n)
        for w, k in zip(weights, peak_indices):
            k = int(k)
            # g[j] += 2*w*a[k-j] for valid j
            # k-j is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                # a[k-j] for j in [j_start, j_end]
                # When j = j_start: a[k - j_start] = a[k - max(0, k-n+1)]
                # When j = j_end: a[k - j_end] = a[k - min(n-1, k)]
                # The slice of a we need: a[k-j_end : k-j_start+1] reversed
                a_slice = a[k - j_end : k - j_start + 1]  # length = j_end - j_start + 1
                g[j_start : j_end + 1] += 2 * w * a_slice[::-1]
        
        # Update with momentum
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive learning rate
        elapsed = time.monotonic() - start
        progress = elapsed / time_budget
        lr = (0.5 / n) * (1 - 0.7 * progress)

import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def subgradient_optimize(a, time_budget, n_peaks=10, verbose=True, seed=42):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    momentum = 0.85
    v = np.zeros(n)
    no_improve = 0
    stall_count = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Adaptive learning rate
        lr = 2.0 / n * (1.0 - 0.8 * progress ** 0.5)
        if no_improve > 50:
            lr *= 0.5
            no_improve = 0
        
        # Compute convolution
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Find peaks with soft weighting
        idx = np.argsort(conv)[::-1][:n_peaks]
        peak_vals = conv[idx]
        
        # Temperature for soft weighting - anneal from high to low
        temp = max_conv * (0.05 + 0.15 * (1 - progress))
        w = np.exp((peak_vals - peak_vals[0]) / temp)
        w /= w.sum()
        
        # Compute subgradient
        g = np.zeros(n)
        for wi, k in zip(w, idx):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                g[j_start:j_end+1] += 2 * wi * a[k-j_end:k-j_start+1][::-1]
        
        # Momentum update
        v = momentum * v + g
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            stall_count = 0
        else:
            no_improve += 1
            stall_count += 1
        
        a = a_new
        it += 1
        
        # Restart mechanism
        if stall_count > 200 or (no_improve > 100 and it % 50 == 0):
            perturbation_scale = 0.01 + 0.02 * min(stall_count // 200, 3)
            a = best_a * np.exp(rng.normal(0, perturbation_scale, n))
            a = np.maximum(a, 0)
            s_r = a.sum()
            if s_r > 1e-10:
                a = a / s_r
            v = np.zeros(n)
            no_improve = 0
            stall_count = 0
        
        if verbose and it % 100 == 0:
            print(f"  SG it{it}: obj={best_obj:.8f}, lr={lr:.6f}, peaks={n_peaks}, t={elapsed:.0f}s")
    
    return best_a

"""
Hybrid subgradient + LP optimizer for minimum peak autocorrelation of step functions.
Combines FFT-accelerated subgradient descent with Nesterov momentum, adaptive
temperature-scheduled peak weighting, LP-based direction polish, and multi-length
exploration to search for sequences minimizing 2n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_hankel(p):
    """Build Hankel matrix H where H[i,j] = p[i+j] for i,j in [0, n-1].
    p has length 2n-1."""
    n = (len(p) + 1) // 2
    H = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            H[i, j] = p[i + j]
    return H


def build_hankel_fast(p):
    """Build Hankel matrix H[i,j] = p[i+j] using strides."""
    n = (len(p) + 1) // 2
    # H[i,j] = p[i+j]
    # For row i, the entries are p[i], p[i+1], ..., p[i+n-1]
    H = np.zeros((n, n))
    for i in range(n):
        H[i, :] = p[i:i + n]
    return H

"""
FFT-accelerated hybrid optimizer for minimum peak autocorrelation.
Uses subgradient descent with Nesterov momentum and adaptive peak temperature,
LP direction polishing, quadratic form refinement, and multi-length search.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_optimize(a, time_budget, n_peaks=10, seed=42, verbose=False):
    """Subgradient descent with Nesterov momentum and adaptive peak selection."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    momentum = 0.85
    v = np.zeros(n)
    no_improve = 0
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Learning rate: starts moderate, decays
        base_lr = 3.0 / n
        lr = base_lr * (1.0 - 0.7 * progress ** 0.7)
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
        
        # Compute convolution
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Adaptive number of peaks
        n_peaks_adapt = n_peaks + int(10 * progress)
        
        # Select peaks
        idx = np.argsort(conv)[::-1][:n_peaks_adapt]
        peak_vals = conv[idx]
        
        # Temperature: anneal from wide to narrow
        temp = max_conv * (0.1 * (1 - progress) + 0.02 * progress)
        w = np.exp((peak_vals - peak_vals[0]) / max(temp, 1e-10))
        w /= w.sum()
        
        # Compute subgradient: g[j] = 2 * sum_k w[k] * a[k-j]
        g = np.zeros(n)
        for wi, k in zip(w, idx):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                # g[j] += 2*wi*a[k-j] for j in [j_start, j_end]
                # a[k-j] for j in [j_start, j_end]
                # = a[k-j_end : k-j_start+1] reversed
                g[j_start:j_end+1] += 2.0 * wi * a[k-j_end:k-j_start+1][::-1]
        
        # Nesterov momentum
        v = momentum * v + g
        a_new = a - lr * v
        
        # Project to simplex
        a_new = project_simplex(a_new)
        
        # Evaluate
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            stall = 0
        else:
            no_improve += 1
            stall += 1
        
        a = a_new
        it += 1
        
        # Restart
        if stall > 300:
            scale = 0.01 + 0.015 * min(stall // 300, 4)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = project_simplex(np.maximum(a, 0))
            v = np.zeros(n)
            no_improve = 0
            stall = 0
            momentum = min(momentum + 0.02, 0.95)
        
        if verbose and it % 200 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}, lr={lr:.6f}, npk={n_peaks_adapt}, t={elapsed:.0f}s")
    
    return best_a


def lp_direction(a):
    """Solve LP to find direction g that minimizes max(a*g).
    Returns (g, M) or (None, None)."""
    n = len(a)
    a = a / a.sum()
    
    # Build A matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0
    # A is (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # Row k, column j: A[k,j] = a[k-j] if 0 <= k-j < n
        # k-j >= 0 => k >= j
        # k-j < n => k < j+n
        # So for column j, valid k range is [j, j+n-1]
        k_start = j
        k_end = j + n - 1
        A[k_start:k_end+1, j] = a[:n]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget, seed=42, verbose=False):
    """LP-based polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = lp_direction(a)
        if g_lp is None:
            break
        
        # Line search
        best_t = 0
        best_t_obj = best_obj
        for t_val in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t_val) * best_a + t_val * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-12:
                best_t_obj = obj_try
                best_t = t_val
        
        if best_t > 0 and best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = (1 - best_t) * best_a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            s_b = best_a.sum()
            if s_b > 1e-10:
                best_a = best_a / s_b
            a = best_a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                scale = 0.01 + 0.01 * min(no_improve - 3, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = project_simplex(np.maximum(a, 0))
                no_improve = 0
                if it > 40:
                    break
        
        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")
    
    return best_a


def quad_form_refine(a, time_budget, T_init=0.1, T_end=0.01, seed=42, verbose=False):
    """Quadratic form refinement: alternate between peak weighting and QP solve."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = T_init * (T_end / T_init) ** progress
        
        # Compute conv
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Peak weights (softmax)
        p = np.exp((conv - max_conv) / (T * max_conv + 1e-10))
        p /= p.sum()
        
        # Build Hankel H[i,j] = p[i+j]
        H = np.zeros((n, n))
        for i in range(n):
            H[i, :] = p[i:i+n]
        
        # Solve min a^T H a s.t. sum(a)=1, a>=0
        # Use gradient descent on the simplex (fast, avoids QP solver overhead)
        gd_steps = 20
        lr_gd = 1.0 / (n * np.max(np.abs(H).sum(axis=1)) + 1e-10)
        
        a_temp = a.copy()
        for _ in range(gd_steps):
            grad = 2.0 * (H @ a_temp)
            a_temp = a_temp - lr_gd * grad
            a_temp = project_simplex(a_temp)
        
        obj_temp = compute_obj(a_temp)
        if obj_temp < best_obj - 1e-12:
            best_obj = obj_temp
            best_a = a_temp.copy()
            a = a_temp.copy()
        
        it += 1
        if verbose and it % 10 == 0:
            print(f"    QF it{it}: obj={best_obj:.8f}, T={T:.4f}, t={elapsed:.0f}s")
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
