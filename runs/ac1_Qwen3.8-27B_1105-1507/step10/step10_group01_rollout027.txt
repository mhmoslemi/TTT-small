Uses Adam-based smoothed gradient descent, Frank-Wolfe method, coordinate descent,

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_gradient_fft(a, tau):
    """Compute gradient of smoothed objective 2*n*logsumexp(tau*conv/sum^2) via FFT."""
    n = len(a)
    s = float(np.sum(a))
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_conv = float(np.max(conv))
    scaled = tau * (conv - max_conv)
    # Clip for numerical stability
    scaled = np.clip(scaled, -50, 50)
    exp_vals = np.exp(scaled)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # Gradient of max(conv) w.r.t. a_norm is: 2 * (P * conj(Af)) inverse FFT
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a_norm, N)
    grad_full = ifft(P * np.conj(A)).real
    grad = 2.0 * grad_full[:n]
    
    # Account for normalization: a_norm = a / s

    grad = grad / s - (a / s**2) * float(np.sum(grad))
    return grad

    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_init
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.2
        
        grad = smoothed_gradient_fft(a, tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(it + 1))
        v_hat = v / (1 - beta2**(it + 1))
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        
        # Learning rate decay

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Adam-based smoothed gradient descent, exact coordinate descent with incremental
convolution updates, subgradient Frank-Wolfe, multi-n exploration, and structured restarts.
Key improvements over previous: faster coordinate descent (O(n) per step), Adam optimizer,
incremental conv updates, and better time allocation.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_gradient_fft(a, tau):
    """Compute gradient of smoothed objective via FFT. Returns gradient w.r.t. a (normalized)."""
    n = len(a)
    s = float(np.sum(a))
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_conv = float(np.max(conv))
    scaled = tau * (conv - max_conv)
    scaled = np.clip(scaled, -60.0, 60.0)
    exp_vals = np.exp(scaled)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a_norm, N)
    grad_full = ifft(P * np.conj(A)).real
    grad = 2.0 * grad_full[:n]

    # Chain rule for normalization: d/d a [f(a/s)] where f depends on a/s
    # g = grad/s - a/s^2 * sum(grad)
    grad = grad / s - (a / (s * s)) * float(np.sum(grad))
    return grad


def adam_smoothed_gd(a, time_budget, tau_init=20.0, tau_end=1500.0, lr_init=0.005, verbose=False):
    """Adam-based gradient descent on smoothed objective with simplex projection."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_init
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Smooth tau schedule: start low, end high
        tau = tau_init + (tau_end - tau_init) * (progress ** 1.3)

        grad = smoothed_gradient_fft(a, tau)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        # LR schedule: initial decay then very slow
        lr = lr_init / (1.0 + 0.0005 * it)

        # Restart if stuck too long
        if it - best_it > 30000:
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            lr = lr_init * 0.5
            best_it = it

        if verbose and it % 100000 == 0:
            print(f"        Adam it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

    if verbose:
        print(f"        Adam done: obj={best_obj:.8f}, iters={it}")
    return best_a


def coord_step(a, i, conv, n):
    """
    Optimize coordinate i using golden section search with O(n) evaluations.
    Returns optimal value for a[i].
    """
    ai = a[i]
    if ai < 1e-14:
        # Try a small positive value
        return 0.0

    k_lo = max(0, i)
    k_hi = min(i + n - 1, 2 * n - 2)

    if k_lo > k_hi:
        return ai

    n_affected = k_hi - k_lo + 1
    base = np.empty(n_affected)
    coeffs = np.empty(n_affected)
    is_quad = np.zeros(n_affected, dtype=bool)

    for idx in range(n_affected):
        k = k_lo + idx
        if k == 2 * i:
            base[idx] = conv[k] - ai * ai
            coeffs[idx] = 1.0
            is_quad[idx] = True
        else:
            base[idx] = conv[k] - 2.0 * ai * a[k - i]
            coeffs[idx] = 2.0 * a[k - i]
            is_quad[idx] = False

    # Max of unaffected
    C = 0.0
    if k_lo > 0:
        C = max(C, float(np.max(conv[:k_lo])))
    if k_hi < 2 * n - 2:
        C = max(C, float(np.max(conv[k_hi + 1:])))

    # Find quad index
    quad_idx_arr = np.where(is_quad)[0]

    # Golden section search
    lo = 0.0
    hi = max(ai * 2.5 + 1.0 / n, ai + 2.0 / n)

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    tmp = base + coeffs * 0.0  # pre-allocate

    def eval_f(val):
        tmp[:] = base
        tmp += coeffs * val
        for qi in quad_idx_arr:
            tmp[qi] = base[qi] + val * val
        m = float(np.max(tmp))
        if C > m:
            m = C
        return m

    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = eval_f(x1)
    f2 = eval_f(x2)

    for _ in range(60):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = eval_f(x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = eval_f(x2)

    opt_val = (lo + hi) / 2.0
    return max(0.0, opt_val)


def coordinate_descent(a, time_budget, verbose=False):
    """Coordinate-wise descent on exact objective with incremental conv updates."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(42)

    sweep = 0
    while time.monotonic() - start < time_budget:
        order = rng.permutation(n)
        for i in order:
            new_val = coord_step(a, i, conv, n)
            old_val = a[i]
            if abs(new_val - old_val) > 1e-13:
                a[i] = new_val
                # Incrementally update conv
                k_lo = max(0, i)
                k_hi = min(i + n - 1, 2 * n - 2)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        conv[k] = conv[k] - old_val * old_val + new_val * new_val
                    else:
                        conv[k] = conv[k] - 2.0 * old_val * a[k - i] + 2.0 * new_val * a[k - i]

        # Periodically recompute conv to avoid drift
        if sweep % 20 == 0:
            conv = compute_conv(a)

        # Check objective
        max_conv = float(np.max(conv))
        s_curr = float(np.sum(a))
        if s_curr > 0.01:
            obj = 2.0 * n * max_conv / (s_curr * s_curr)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()

        sweep += 1
        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"        Coord sweep {sweep}: obj={best_obj:.8f}, time={elapsed:.0f}s")

    # Return normalized best
    s_best = float(np.sum(best_a))
    if s_best > 0.01:
        best_a = best_a / s_best
    return best_a


def subgradient_fw_step(a, n):
    """
    One Frank-Wolfe step: compute subgradient, find min coordinate, return direction.
    Returns the vertex index (argmin of subgradient).
    """
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    # Find all argmax
    argmax_ks = np.where(conv >= max_val - 1e-10)[0]
    p = np.zeros(len(conv))
    p[argmax_ks] = 1.0 / len(argmax_ks)

    # Subgradient: 2 * fold of a weighted by p
    # grad_j = 2 * sum_k p_k * a_{k-j} (with bounds)
    # This is 2 * (p * a) circular conv, but we need linear
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad_full = ifft(P * np.conj(A)).real[:n]
    grad = 2.0 * grad_full

    # Account for normalization (a is already normalized to sum=1)
    # For FW on simplex, we need the gradient of the objective w.r.t. a on the simplex
    # The subgradient of f(a) = 2*n*max(conv) w.r.t. a (with sum(a)=1) projected
    # is just the subgradient of max(conv) w.r.t. a, since the constraint is handled by FW
    # Actually for FW, we just need the gradient of the objective function.
    # f(a) = 2*n*max_k (a*a)_k (since sum(a)=1)
    # subgrad = 2*n * 2 * fold(p * a) ... wait let me be more careful
    # f(a) = 2*n * max_k sum_j a_j a_{k-j}
    # d/da_i f = 2*n * sum_k p_k * (a_{k-i} + a_{k-i}) = 2*n * 2 * sum_k p_k * a_{k-i}
    # (the factor of 2 is from the two occurrences of a_i in the sum)
    # So subgrad_i = 4*n * sum_k p_k * a_{k-i}

    # Actually the factor already includes both: grad_full already has the factor of 2
    # from ifft(P * conj(A)) giving sum_k p_k * a_{k-j}
    # Then we multiply by 2 for the symmetry, giving 2 * sum_k p_k * a_{k-j}
    # Then multiply by 2*n for the objective factor
    grad = 4.0 * n * grad

    # For FW on simplex: find vertex e_i minimizing grad^T e_i = grad_i
    # But we want to MINIMIZE, so we find the vertex with the LARGEST negative component
    # i.e., argmin grad_i
    return int(np.argmin(grad))


def fw_optimize(a, time_budget=60.0, n_steps_per_phase=200, verbose=False):
    """Frank-Wolfe optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    gamma = 1.0  # FW step parameter

    while time.monotonic() - start < time_budget:
        # Get FW direction
        i_best = subgradient_fw_step(a, n)
        e = np.zeros(n)
        e[i_best] = 1.0

        # Line search: try several step sizes
        best_t = 0.0
        best_t_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 0.15, 0.25, 0.4, 0.6]:
            a_try = (1.0 - t) * a + t * e
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))
            if s_try > 0.01:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj:
                    best_t = t
                    best_t_obj = obj_try

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * e
            a = np.maximum(a, 0.0)
            s = float(np.sum(a))
            a = a / s
            if best_t_obj < best_obj:
                best_obj = best_t_obj
                best_a = a.copy()

        # Decay gamma for accelerated FW (not used here, but keeps code clean)
        gamma *= 0.999

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def make_initialization(n, kind='random'):
    """Create an initial sequence of length n."""
    rng = np.random.default_rng(123 + n)
    if kind == 'constant':
        a = np.ones(n)
    elif kind == 'triangular':
        i = np.arange(n)
        a = np.minimum(i + 1, n - i)
    elif kind == 'gaussian':
        i = np.arange(n)
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    elif kind == 'cosine':
        i = np.arange(n)
        a = np.cos(np.pi * i / (2.0 * n)) ** 2
        a = np.maximum(a, 0.1)
    elif kind == 'bimodal':
        i = np.arange(n)
        c1 = n // 4
        c2 = 3 * n // 4
        sigma = n / 8.0
        a = np.exp(-0.5 * ((i - c1) / sigma) ** 2) + np.exp(-0.5 * ((i - c2) / sigma) ** 2)
    else:  # random
        a = rng.random(n)
        a = a + 0.1  # ensure positivity
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = float(np.sum(a))
    return a / s


def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 6):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Adam GD + coordinate descent + FW + multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to use parent as starting point
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 50 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_initialization(800, 'gaussian')
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"Starting search from n={n_current}, obj={best_score:.8f}")

    # ============ Phase 1: Initialization exploration (100s) ============
    print("\n=== Phase 1: Initialization Exploration ===")
    p1_budget = 100.0
    p1_end = start_time + p1_budget

    inits_to_try = []
    if parent is not None:
        inits_to_try.append((n_current, 'constant'))
        inits_to_try.append((n_current, 'triangular'))
        inits_to_try.append((n_current, 'gaussian'))
        inits_to_try.append((n_current, 'cosine'))
        inits_to_try.append((700, 'gaussian'))
        inits_to_try.append((900, 'triangular'))
        inits_to_try.append((1000, 'gaussian'))
        inits_to_try.append((800, 'bimodal'))
    else:
        inits_to_try.append((800, 'gaussian'))
        inits_to_try.append((800, 'triangular'))
        inits_to_try.append((700, 'gaussian'))
        inits_to_try.append((900, 'gaussian'))

    per_init_time = min(12.0, p1_budget / max(len(inits_to_try), 1))

    for n_try, kind in inits_to_try:
        if time.monotonic() >= p1_end:
            break
        a_init = make_initialization(n_try, kind)
        # Quick Adam warmup
        remaining = p1_end - time.monotonic()
        t_budget = min(per_init_time, remaining * 0.8)
        if t_budget < 3:
            continue
        a_opt = adam_smoothed_gd(a_init, time_budget=t_budget, tau_init=20.0, tau_end=200.0, lr_init=0.008)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try
            print(f"  Init n={n_try} ({kind}): obj={best_score:.8f} **")
        else:
            print(f"  Init n={n_try} ({kind}): obj={obj_opt:.8f}")

    # If parent was good, keep it
    if parent is not None:
        parent_obj = compute_obj(parent)
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()
            n_current = len(parent)
            print(f"  Keeping parent: obj={best_score:.8f}")

    # ============ Phase 2: Intensive Adam GD (300s) ============
    print(f"\n=== Phase 2: Intensive Adam GD (n={n_current}) ===")
    p2_budget = 300.0
    p2_end = time.monotonic() + p2_budget
    if time.monotonic() >= deadline - 100:
        p2_budget = max(30, deadline - time.monotonic() - 100)
        p2_end = time.monotonic() + p2_budget

    stage2_time = p2_budget * 0.35
    stage3_time = p2_budget * 0.25

    if t1 > 5:
        a_cur = adam_smoothed_gd(a_cur, time_budget=t1, tau_init=15.0, tau_end=100.0, lr_init=0.006, verbose=True)
        obj_cur = compute_obj(a_cur)
        if obj_cur < best_score:
            best_score = obj_cur
            best_seq = a_cur.copy()
            print(f"  Stage1: obj={best_score:.8f} **")
        else:
            print(f"  Stage1: obj={obj_cur:.8f} (no improvement)")

    # Stage 2: medium tau
    remaining_to_p2 = p2_end - time.monotonic()
    t2 = min(stage2_time, remaining_to_p2)
    if t2 > 5:
        a_cur = adam_smoothed_gd(best_seq, time_budget=t2, tau_init=80.0, tau_end=500.0, lr_init=0.003, verbose=True)
        obj_cur = compute_obj(a_cur)

    print(f"\n=== Phase 3: Coordinate Descent (n={len(best_seq)}) ===")

    if time.monotonic() >= deadline - 150:
        p3_budget = max(50, deadline - time.monotonic() - 150)
        p3_end = time.monotonic() + p3_budget

    a_cd = coordinate_descent(best_seq, time_budget=p3_budget, verbose=True)
    obj_cd = compute_obj(a_cd)
    if obj_cd < best_score:
        best_score = obj_cd
        best_seq = a_cd.copy()
        print(f"  CoordDescent: obj={best_score:.8f} **")
    else:
        print(f"  CoordDescent: obj={obj_cd:.8f} (no improvement)")

    # ============ Phase 4: FW + Additional Coord Descent (200s) ============
    print(f"\n=== Phase 4: FW + Coord Descent ===")
    p4_budget = 200.0
    p4_end = time.monotonic() + p4_budget
    if time.monotonic() >= deadline - 80:

    if time.monotonic() >= deadline - 50:

    n_best = len(best_seq)
    n_options = []

        if 400 <= n_opt <= 2000 and n_opt not in n_options:

    for n_abs in [600, 700, 900, 1000, 1100, 1200]:

    n_options = n_options[:6]

    per_n_time = p5_budget / max(len(n_options), 1) * 0.85

    for n_try in n_options:

            break
        a_res = resample_sequence(best_seq, n_try)

        # Quick Adam
        rem = p5_end - time.monotonic()
