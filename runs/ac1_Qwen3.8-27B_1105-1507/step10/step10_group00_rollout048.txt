def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:

def exact_subgradient_step(a):
    """One step of subgradient descent on the exact (non-smooth) objective."""
    n = len(a)
    a = a / a.sum()
    conv = compute_conv(a)
    k_max = int(np.argmax(conv))
    
    # Subgradient of max_k (a*a)[k] at a
    grad = np.zeros(n)
    for i in range(n):
        j1 = k_max - i
        j2 = i - k_max
        if 0 <= j1 < n:
            grad[i] += a[j1]
        if 0 <= j2 < n:
            grad[i] += a[j2]
    
    return grad

def subgradient_descent_exact(a, n_steps=1000, lr=0.01):
    """Subgradient descent on the exact objective with averaging."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    a_avg = a.copy()
    lr0 = lr
    
    for it in range(n_steps):
        grad = exact_subgradient_step(a_avg)  # Use averaged iterate for gradient
        
        lr_t = lr0 / (1 + it * 0.01)
        a_new = a_avg - lr_t * grad
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s > 1e-10:
            a_new = a_new / s
        
        # Update average
        a_avg = (it / (it + 1)) * a_avg + (1 / (it + 1)) * a_new
        
        obj = compute_obj(a_avg)
        if obj < best_obj:
            best_obj = obj
            best_a = a_avg.copy()
    
    return best_a, best_obj

Actually, I realize that when the LP converges (a ≈ g_lp), the peak of (a*a) ≈ peak of (a*g_lp) = M_lp. So the LP gives an approximate value of the true objective.

The key question is: how close is the LP solution to the true optimum of the non-convex (in terms of the original objective) problem?

Wait, the problem IS convex! `max_k (a*a)[k]` is a convex function of a (since each (a*a)[k] is convex as a quadratic form, and max of convex functions is convex). So the LP relaxation should converge to the true optimum.

But the LP we're solving is NOT a relaxation of the original problem. It's a linearization. The true problem is:
min max_k sum_j a[j]a[k-j] s.t. sum(a)=1, a>=0

The LP we solve is:
min M s.t. sum_j a[j]g[k-j] <= M for all k, sum(g)=1, g>=0

These are different problems. The LP is a surrogate that works well when g ≈ a.

So the convergence is: iterate a_{k+1} = argmin_g max_k (a_k * g) s.t. g in simplex. This is a kind of fixed-point iteration.

For this to converge to the true optimum, we need the fixed point to be the same as the optimum of the original problem. This is true if the objective is "bi-convex" in the sense that for fixed a, the function g -> (a*g) is linear, and the fixed point satisfies a* = argmin_g max_k (a* * g).

At the optimum a*, we have that a* is a minimizer of max_k (a* * a*[k]) over the simplex. By complementary slackness, this means that the subgradient at a* contains 0, which is the KKT condition for the original problem.

So the fixed-point iteration SHOULD converge to the optimum, but the rate might be slow.

To speed up convergence, we can:
1. Use Nesterov acceleration on the fixed-point iteration
2. Use a better line search
3. Use multiple LP solves per iteration (with slightly different a)

Refined LP-based optimization with exhaustive line search, post-LP subgradient
refinement on exact objective, and aggressive n exploration.
Key improvements over previous: brute-force line search, more LP iterations,
exact subgradient polish, better time allocation.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def exhaustive_line_search(a, g_lp, n_coarse=40, n_fine=30, t_max=1.3):
    best_t = 0.0
    best_obj = compute_obj(a)
    ts = np.linspace(0.0, t_max, n_coarse)
    for t in ts:
        a_try = (1.0 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    if 0.0 < best_t < t_max:
        lo = max(0.0, best_t - 0.04)
        hi = min(t_max, best_t + 0.04)
        ts_fine = np.linspace(lo, hi, n_fine)
        for t in ts_fine:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    return best_t, best_obj


def exact_subgradient_descent(a, time_budget=30.0):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    a_avg = a.copy()
    lr0 = 0.05
    momentum = np.zeros(n)
    beta = 0.7
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a_avg)
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a_avg[j1]
            if 0 <= j2 < n:
                grad[i] += a_avg[j2]
        lr_t = lr0 / (1.0 + 0.005 * it)
        momentum = beta * momentum - lr_t * grad
        a_new = a_avg + momentum
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s
        a_avg = (it / (it + 1.0)) * a_avg + (1.0 / (it + 1.0)) * a_new
        obj = compute_obj(a_avg)
        if obj < best_obj:
            best_obj = obj
            best_a = a_avg.copy()
            best_it = it
        it += 1
        if it - best_it > 3000:
            a_avg = best_a.copy()
            momentum = np.zeros(n)
            lr0 = 0.02
            best_it = it
    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 4)
    else:
        a = 1.0 / (0.1 + 2.0 * np.abs(2 * x - 1))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def lp_polish(a, time_budget=100.0, max_iters=100, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(999 + it)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break
        T0 = time.monotonic()
        g_lp, M_lp = solve_lp(a)
        T1 = time.monotonic()
        lp_time = T1 - T0
        if g_lp is None:
            break
        t_opt, obj_opt = exhaustive_line_search(a, g_lp, n_coarse=30, n_fine=20)
        if t_opt > 1e-6 and obj_opt < best_obj - 1e-15:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            # Also try t=1 directly
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0
        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"lp_time={lp_time:.2f}s, elapsed={elapsed:.1f}s")
        if stall >= 4:
            if stall <= 6:
                freq = rng.integers(1, max(2, n // 80))
                phase = rng.uniform(0, 2 * np.pi)
                idx = np.arange(n, dtype=np.float64)
                pert = 0.01 * stall * np.cos(2 * np.pi * freq * idx / n + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                else:
                    a = best_a.copy()
            else:
                mode = it % 6
                a_div = diverse_init(n, rng, mode=mode)
                a = 0.3 * a_div + 0.7 * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()
            stall = 0
    return best_a


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)

    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s

        print(f"\n=== Phase 6: Final Polish ===")
        alloc = min(remaining - 10, 150.0)
        if alloc > 15:
            a_final = lp_polish(best_seq, time_budget=alloc, max_iters=100, verbose=True)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final: obj={best_score:.8f} **")
            else:
                print(f"  Final: obj={obj_final:.8f}")

    # === Phase 7: Squeeze with subgradient + short LP ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: Final Squeeze ===")
        a_sg2, obj_sg2 = exact_subgradient_descent(best_seq, time_budget=min(remaining * 0.3, 25.0))
        if obj_sg2 < best_score:
            best_score = obj_sg2
            best_seq = a_sg2.copy()
            print(f"  SG: obj={best_score:.8f} **")
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20 and best_seq is not None:
            a_sq = lp_polish(best_seq, time_budget=min(remaining2 - 5, 40.0), max_iters=30, verbose=False)
            obj_sq = compute_obj(a_sq)
            if obj_sq < best_score:
                best_score = obj_sq
                best_seq = a_sq.copy()
                print(f"  Squeeze: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
