    A = fft(a, N)
    corr = ifft(P * np.conj(A)).real[:n]
    
    grad = 4.0 * n * corr
    return grad, smoothed_val, max_c, p


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0)
    s = float(np.sum(v))
    if s < 1e-300:
        n = len(v)
        return np.ones(n) / n
    return v / s


def fista_optimize(a, time_budget=60.0, tau_start=20.0, tau_end=3000.0, 
                   verbose=False, seed=42):
    """
    FISTA (accelerated gradient) on the smoothed objective.
    a should be non-negative with sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    rng = np.random.default_rng(seed)
    
    # FISTA parameters
    t = 1.0
    a_prev = a.copy()
    v = np.zeros(n)  # momentum
    step = 1.0 / (4.0 * n * np.sqrt(n))  # initial step size
    
    it = 0
    best_it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * (progress ** 1.2)
        
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        
        y = a + beta * v
        y = project_simplex(y)
        
        # Compute gradient at y
        grad, smooth_val, max_c, p = compute_smoothed_grad(y, tau)
        
        # Step
        y_try = y - step * grad
        a_new = project_simplex(y_try)
        
        # Check if step is too large (objective increased)
        obj_new = compute_obj(a_new)
        if obj_new > best_obj * 1.01 + 0.01:
            # Halve step and retry
            step *= 0.5
            y_try = y - step * grad
            a_new = project_simplex(y_try)
            obj_new = compute_obj(a_new)
        else:
            # Increase step cautiously
            step = min(step * 1.05, 1.0 / (2.0 * n))
        
        v = a_new - a
        a = a_new
        t = t_new
        a_prev = a_new
        it += 1
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodic restart if stuck
        if no_improve > 2000:
            # Small random perturbation
            pert = rng.standard_normal(n) * 0.01
            a = project_simplex(best_a + pert)
            v = np.zeros(n)
            t = 1.0
            step = 1.0 / (4.0 * n * np.sqrt(n))
            no_improve = 0
        
        # Adaptive: if tau is large and we're still improving, that's good
        if verbose and it % 5000 == 0:
            print(f"      FISTA it{it}: tau={tau:.0f}, obj={best_obj:.8f}, "
                  f"step={step:.2e}, no_improve={no_improve}")
    
    return best_a


def subgradient_descent_exact(a, time_budget=30.0, top_k=5, verbose=False):
    """
    Subgradient descent using exact subgradient at top-k peaks.
    Very small steps for final polishing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0 / (10.0 * n)
    
    it = 0
    while time.monotonic() - start < time_budget:
        c = compute_conv(a)
        max_c = float(np.max(c))
        
        # Find top-k peaks
        top_indices = np.argsort(c)[-top_k:]
        
        # Subgradient: average of gradients at top-k peaks
        grad = np.zeros(n)
        for k in top_indices:
            for j in range(n):
                m = k - j
                if 0 <= m < n:
                    grad[j] += 2.0 * a[m]
        grad /= top_k
        grad *= 2.0 * n  # Scale by 2n
        
        # Step with backtracking
        for _ in range(20):
            a_try = project_simplex(a - lr * grad)
            obj_try = compute_obj(a_try)
            if obj_try <= best_obj + 1e-10:
                break
            lr *= 0.5
        
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            lr = min(lr * 1.01, 1.0 / n)
        else:
            lr *= 0.8
        
        it += 1
    
    if verbose:
        print(f"      SG-exact: obj={best_obj:.8f}, iters={it}")
    return best_a


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = conv(a_norm, g)[k]."""
    n = len(a_norm)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a_norm[diff[valid]]
    return A


def lp_frank_wolfe(a, n_iters=50, verbose=False):
    """
    Frank-Wolfe with LP oracle: at each step, solve the LP to find the
    best direction, then do a line search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Precompute (this depends on a, so we rebuild each iteration)
    for it in range(n_iters):
        a_norm = a / float(np.sum(a))
        A = build_conv_matrix(a_norm)
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        g_val = None
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                    break
            except Exception:
                continue
        
        if g_val is None:
            break
        
        # Line search
        best_t = 0.0
        best_t_obj = best_obj
        for t in np.linspace(0.01, 1.0, 30):
            a_try = (1 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_t = t
        
        if best_t > 0 and best_t_obj < best_obj - 1e-13:
            a = (1 - best_t) * a + best_t * g_val
            a = project_simplex(a)
            best_obj = best_t_obj
            best_a = a.copy()
            if verbose:
                print(f"      FW it{it}: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            if best_t == 0:
                # Try small step
                for t_small in [0.05, 0.02, 0.01]:
                    a_try = (1 - t_small) * a + t_small * g_val
                    a_try = project_simplex(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        a = a_try
                        best_obj = obj_try
                        best_a = a.copy()
                        break
                break  # Can't improve further with this direction
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent_polish(a, n_sweeps=3, verbose=False):
    """
    Coordinate descent: for each coordinate j, find optimal a[j]
    given all others fixed (with renormalization).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for sweep in range(n_sweeps):
        for j in range(n):
            # Try different values of a[j]
            a_orig_j = a[j]
            s_other = s - a_orig_j  # sum of all other elements
            
            # The objective as a function of a[j] (before normalization):
            # After setting a[j] = x, sum = s_other + x
            # conv[k] changes only for k values that involve index j
            # This is complex, so let's just try a grid of values
            candidates = [0, a_orig_j * 0.5, a_orig_j * 0.8, a_orig_j, 
                         a_orig_j * 1.2, a_orig_j * 1.5, 2.0 * a_orig_j / n]
            candidates = [max(0, c) for c in candidates]
            
            best_x = a_orig_j
            best_local_obj = best_obj
            
            for x in candidates:
                a[j] = x
                s_new = s_other + x
                if s_new < 1e-10:
                    a[j] = a_orig_j
                    continue
                a_norm = a / s_new
                obj_try = compute_obj(a_norm)
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_x = x
            
            a[j] = best_x
            a = a / float(np.sum(a))
            s = float(np.sum(a))
            
            if a[j] != a_orig_j and best_local_obj < best_obj - 1e-14:
                best_obj = best_local_obj
                best_a = a.copy()
        
        if verbose:
            print(f"      CD sweep {sweep}: obj={best_obj:.8f}")
    
    return best_a
