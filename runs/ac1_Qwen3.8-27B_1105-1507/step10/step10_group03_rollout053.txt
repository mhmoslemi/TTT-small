def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def subgradient_max_conv(a, k_star):
    """Compute subgradient of max(conv(a,a)) at lag k_star."""
    n = len(a)
    g = np.zeros(n)
    # g[i] = 2 * a[k_star - i] if 0 <= k_star - i < n
    lo = max(0, k_star - (n-1))
    hi = min(k_star, n-1)
    for i in range(lo, hi+1):
        g[i] = 2.0 * a[k_star - i]
    return g

def adam_subgradient_gd(a, time_budget, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    
    # Adaptive learning rate
    lr_base = lr
    patience = 0
    max_patience = 5000
    
    while time.monotonic() - start < time_budget:
        # Compute convolution
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient
        g = subgradient_max_conv(a, k_star)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            lr = lr_base * 0.5
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        
        # Learning rate schedule
        if patience > max_patience:
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = lr_base * (0.5 + 0.5 * np.random.random())
            patience = 0
        
        # Decay learning rate slowly
        lr = lr_base / (1 + 0.001 * t)
    
    return best_a

    n = len(a)
    conv = compute_conv(a)

        lo = max(0, ki - (n-1))

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) block-structure initialization, (2) Adam top-k subgradient,
(3) LP fixed-point polishing, (4) multi-n exploration.

import numpy as np
import time

from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0*n*float(np.max(conv))/(s*s)

def topk_subgradient(a, tau=10.0, top_k=10):
    """Compute smoothed subgradient using top-k lags with softmax weighting."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    # Get top-k indices
    if top_k >= len(conv):
        top_k = len(conv)
    top_idx = np.argsort(conv)[::-1][:top_k]
    # Softmax weights
    weights = np.exp(tau*(conv[top_idx] - max_c))
    weights /= weights.sum()
    # Weighted sum of subgradients
    g = np.zeros(n)
    for w, ki in zip(weights, top_idx):
        ki = int(ki)
        lo = max(0, ki - (n-1))
        hi = min(ki, n-1)
        # g[i] += w * 2 * a[ki - i] for i in [lo, hi]
        g[lo:hi+1] += w * 2.0 * a[ki - lo : ki - hi + 1][::-1]
    return g, max_c, conv
