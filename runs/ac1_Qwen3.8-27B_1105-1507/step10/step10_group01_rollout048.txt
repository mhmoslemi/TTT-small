"""

Combines Gerchberg-Saxton spectral flattening, LP-driven optimization, and
mirror descent (multiplicative updates) with adaptive parameter scheduling.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    return np.real(ifft(fa * fa))[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_gradient_smoothed(a, tau):
    """Compute gradient of tau-smoothed log-sum-exp of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    Af = fft(a, N)
    grad = 2.0 * np.real(ifft(P * np.conj(Af)))[:n]
    return grad


def build_conv_matrix(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A

    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    if weights is not None:
        weighted_A = A * weights[:, None]
        constraints = [weighted_A @ g <= M, cp.sum(g) == 1]
    else:
        constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_relative(a):
    """
    Solve LP with relative weights: min M s.t. conv(a_norm, g)[k] <= M * c[k]
    where c = conv(a_norm, a_norm) normalized. This prioritizes reducing large peaks.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    c = compute_conv(a_norm)
    c_max = float(np.max(c))
    if c_max < 1e-15:
        return None, None
    # Weight: inverse of current convolution value (higher where peaks are)
    # Use a smooth version to avoid division by zero
    weights = 1.0 / (c / c_max + 0.1)
    weights = weights / np.max(weights)

    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    weighted_A = A * weights[:, None]
    constraints = [weighted_A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def gs_spectral_flatten(a, beta=0.3, n_iter=50, verbose=False):
    """
    Gerchberg-Saxton inspired algorithm for spectral flattening.
    Alternates between:
    - Frequency domain: flatten |A|^2 towards geometric mean
    - Time domain: enforce a >= 0 and sum(a) = 1
    """
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    a_padded = np.zeros(N)
    a_padded[:n] = a
    a_padded = a_padded / np.sum(a_padded)

    best_obj = compute_obj(a)
    best_a = a.copy()

    for it in range(n_iter):
        # Frequency domain step
        A = fft(a_padded)
        mag = np.abs(A)
        phase = np.angle(A)
        power = mag ** 2

        # Compute target: geometric mean of power (smoothed)
        log_power = np.log(power + 1e-20)
        # Apply a slight smoothing in frequency to avoid over-correction
        log_power_smooth = np.convolve(log_power, np.ones(5) / 5.0, mode='same')
        target_log = float(np.median(log_power_smooth))

        # Flatten: bring mag towards exp(target_log / 2) = sqrt(target_power)
        target_mag = np.exp(0.5 * target_log)
        # Blend: mag_new = mag * (target_mag / mag)^beta
        ratio = target_mag / (mag + 1e-20)
        mag_new = mag * np.power(ratio, beta)
        # Clamp to reasonable range
        mag_new = np.clip(mag_new, 1e-10, 10.0 * mag + 1e-10)

        A_new = mag_new * np.exp(1j * phase)
        a_padded = np.real(ifft(A_new))

        # Time domain constraint
        a_padded = np.maximum(a_padded, 0)
        s = float(np.sum(a_padded))
        if s < 1e-15:
            a_padded = best_a.copy()
            N2 = 1
            while N2 < 2 * n - 1:
                N2 <<= 1
            a_padded = np.zeros(N2)
            a_padded[:n] = best_a
            a_padded = a_padded / np.sum(a_padded)
        else:
            a_padded = a_padded / s

        # Check improvement (every few iterations)
        if it % 10 == 0 or it == n_iter - 1:
            a_curr = a_padded[:n]
            obj_curr = compute_obj(a_curr)
            if obj_curr < best_obj:
                best_obj = obj_curr
                best_a = a_curr.copy()
                if verbose and (it % 50 == 0):
                    print(f"      GS it{it}: obj={best_obj:.8f}, beta={beta:.3f}")

    return best_a


def mirror_descent(a, time_budget=100.0, tau_init=30.0, tau_end=500.0, verbose=False):
    """
    Mirror descent (multiplicative updates) on smoothed max autocorrelation.
    Uses entropy-based Bregman divergence, naturally preserving non-negativity.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    base_lr = 1.0 / np.sqrt(n)
    momentum = 0.5
    prev_log_a = np.log(a + 1e-30)
    velocity = np.zeros(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.2

        grad = compute_gradient_smoothed(a, tau)
        grad_norm = float(np.linalg.norm(grad))

        # Adaptive learning rate
        if grad_norm > 1e-15:
            lr_adapt = 1.0 / (grad_norm * n ** 0.3)
            lr_adapt = min(max(lr_adapt, 0.1 * base_lr), 5.0 * base_lr)
        else:
            lr_adapt = base_lr

        # Momentum in log-space
        log_a = np.log(a + 1e-30)
        velocity = momentum * velocity - lr_adapt * grad / a
        velocity = np.clip(velocity, -5.0, 5.0)

        new_log_a = log_a + velocity
        a_new = np.exp(new_log_a)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            velocity = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        # Backtracking if too aggressive
        if obj_new > best_obj * (1 + 0.01):
            a_new = 0.5 * a + 0.5 * a_new
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            velocity *= 0.5

        prev_log_a = log_a
        a = a_new
        it += 1

        if verbose and it % 100 == 0:
            print(f"      Mirror it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

    return best_a


def nesterov_gd(a, time_budget=100.0, tau_init=50.0, tau_end=1000.0, verbose=False):
    """Nesterov accelerated gradient descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.3

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = 1.0
        else:
            y = y / s_y

        g = compute_gradient_smoothed(y, tau)
        g_norm = float(np.linalg.norm(g))

        # Adaptive LR with BB
        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.2 * base_lr), 3.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adapt

        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 5.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-12, lr * 0.9997)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

    if verbose:
        print(f"      Nesterov: obj={best_obj:.8f}, iters={it}")
    return best_a


def lp_optimize(a, time_budget=200.0, relative=True, verbose=False):
    """LP-driven optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    iter_count = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        if relative and iter_count % 3 == 0:
            g, M = solve_lp_relative(a)
        else:
            g, M = solve_lp_minpeak(a)

        if g is None:
            if remaining < 5:
                break
            # Try standard LP as fallback
            g, M = solve_lp_minpeak(a)
            if g is None:
                break

        # Line search
        best_t = 0.0
        best_t_obj = best_obj
        ts = np.linspace(0.01, 0.95, 20)
        for t in ts:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_t = t

        iter_count += 1

        if best_t_obj < best_obj - 1e-13 and best_t > 0:
            a_new = (1 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            a_new = a_new / s_new
            best_obj = best_t_obj
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            if verbose and (iter_count % 10 == 0):
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            stall_count += 1
            if stall_count >= 4:
                # Perturb
                a_pert = a + 0.01 * rng.standard_normal(n)
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                a = a_pert
                stall_count = 0
                if iter_count > 200:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 6):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def combined_optimize(a, time_budget=300.0, verbose=False):
    """
    Combined optimization: alternate between LP, GS, and mirror descent
    with adaptive scheduling based on progress.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    # Phase allocation: LP 40%, GS 25%, Mirror 20%, Nesterov 15%
    lp_frac = 0.35
    gs_frac = 0.25
    mirror_frac = 0.2
    nesterov_frac = 0.2

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 10:
            break

        # Determine which method to use based on progress
        progress = elapsed / time_budget
        if verbose:
            print(f"    Cycle at progress={progress:.2f}, remaining={remaining:.0f}s, obj={best_obj:.8f}")

        # LP phase
        lp_time = min(remaining * lp_frac, 80.0)
        if lp_time > 15:
            a_new = lp_optimize(a, time_budget=lp_time, relative=True, verbose=False)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                if verbose:
                    print(f"      LP: obj={best_obj:.8f}")

        # GS phase
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        gs_time = min(remaining * gs_frac, 60.0)
        if gs_time > 10:
            # Anneal beta
            beta = 0.2 + 0.4 * (elapsed / time_budget)
            n_gs = max(20, int(gs_time * 3))
            a_gs = gs_spectral_flatten(a, beta=beta, n_iter=n_gs, verbose=False)
            obj_gs = compute_obj(a_gs)
            if obj_gs < best_obj - 1e-14:
                best_obj = obj_gs
                best_a = a_gs.copy()
                a = a_gs
                if verbose:
                    print(f"      GS: obj={best_obj:.8f}")

        # Mirror descent phase
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        md_time = min(remaining * mirror_frac, 40.0)
        if md_time > 10:
            tau_init = 30.0 + 200.0 * (elapsed / time_budget)
            a_md = mirror_descent(a, time_budget=md_time, tau_init=tau_init,
                                  tau_end=tau_init + 300.0, verbose=False)
            obj_md = compute_obj(a_md)
            if obj_md < best_obj - 1e-14:
                best_obj = obj_md
                best_a = a_md.copy()
                a = a_md
                if verbose:
                    print(f"      Mirror: obj={best_obj:.8f}")

        # Nesterov phase
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        nv_time = min(remaining * nesterov_frac, 40.0)
        if nv_time > 10:
            tau_init = 50.0 + 300.0 * (elapsed / time_budget)
            a_nv = nesterov_gd(a, time_budget=nv_time, tau_init=tau_init,
                               tau_end=tau_init + 500.0, verbose=False)
            obj_nv = compute_obj(a_nv)
            if obj_nv < best_obj - 1e-14:
                best_obj = obj_nv
                best_a = a_nv.copy()
                a = a_nv
                if verbose:
                    print(f"      Nesterov: obj={best_obj:.8f}")

        # Check for stall and perturb
        if (time.monotonic() - start) / time_budget > 0.5:
            a_check = a / a.sum()
            if compute_obj(a_check) > best_obj * 1.001:
                a = fourier_perturb(best_a, rng, strength=0.02)

    return best_a

            a_r[:half] += 0.05 * rng.standard_normal(half)
            a_r[half:] += 0.05 * rng.standard_normal(n_cur - half)
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        else:  # concentrate
            # Try concentrating mass in the middle
            center = n_cur / 2.0
            weights = np.exp(-0.5 * ((np.arange(n_cur) - center) / (n_cur * 0.3)) ** 2)
            a_r = best_seq * (0.7 + 0.3 * weights / weights.max())
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

        # Optimize the restart
        alloc_comb = min(remaining * 0.3, 40.0)
        if alloc_comb > 10:
            a_r = combined_optimize(a_r, time_budget=alloc_comb, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final combined round
    if remaining > 30:
        alloc = min(remaining * 0.5, 200.0)
        a_final = combined_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final combined: obj={best_score:.8f} **")

    # Final Nesterov with very high tau
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 80.0)
        a_final2 = nesterov_gd(best_seq, time_budget=alloc, tau_init=200.0,
                               tau_end=3000.0, verbose=True)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final Nesterov: obj={best_score:.8f} **")

    # Final GS with very low beta (gentle)
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_final3 = gs_spectral_flatten(best_seq, beta=0.1, n_iter=100, verbose=True)
        obj_final3 = compute_obj(a_final3)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()
            print(f"  Final GS gentle: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < np.inf:
        best_score = final_val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
