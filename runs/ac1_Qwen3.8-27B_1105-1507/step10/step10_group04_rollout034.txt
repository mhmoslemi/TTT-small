import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize, linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp_fast(a, max_iter=50000):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix: (A@g)[k] - M <= 0
    # A is (2n-1) x n where A[k,j] = a_norm[k-j] if valid
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm
    
    # Full matrix for linprog: variables are [g_0..g_{n-1}, M]
    # min [0,...,0, 1] @ x
    # s.t. [A, -I] @ x <= 0  (2n-1 constraints)
    #      [1,...,1, 0] @ x = 1
    #      g >= 0, M >= 0 (M is implicitly >= 0 since A@g >= 0)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': max_iter, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-15:
                g = g / s
                M = float(res.x[n])
                return g, M
    except Exception:
        pass
    return None, None


def lp_polish(a, time_budget, verbose=True, max_t=1.0, min_t=0.05):
    """LP polishing with adaptive mixing parameter."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    t = 0.5
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_fast(a)
        if g_lp is None:
            break
        it += 1

        # Line search over mixing parameter
        t_candidates = np.unique(np.clip(
            np.concatenate([
                np.linspace(0.02, min(max_t, 1.0), 12),
                [max(min_t, t * 0.5), max(min_t, t * 0.7), t,
                 min(1.0, t * 1.3), min(1.0, t * 1.5), min(1.0, t * 2.0)]
            ]), 0.01, 1.0))

        best_t_obj = np.inf
        best_t_val = t
        best_a_local = a.copy()

        for tc in t_candidates:
            a_try = (1.0 - tc) * a + tc * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = tc
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            t = best_t_val
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t:.3f}, M={M_lp:.6f}, elapsed={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            t = max(min_t, t * 0.8)
            if stall > 8:
                break

    return best_a


def smoothed_gd_step(a, p_weights, lr):
    """One step of smoothed gradient: gradient of weighted sum of conv values."""
    n = len(a)
    # Gradient of sum_k p_k * (a*a)[k] w.r.t. a[j] = 2 * sum_k p_k * a[k-j]
    # This is 2 * conv(p, a)[j] ... wait, let me be careful.
    # (a*a)[k] = sum_j a[j]*a[k-j]
    # d/d(a[m]) (a*a)[k] = 2*a[k-m] (as derived earlier)
    # So grad_a[m] = 2 * sum_k p_k * a[k-m]
    # = 2 * (p * a)[m] where (p*a)[m] = sum_k p[k]*a[k-m] ... 
    # Actually (p*a)[m] = sum_k p[k]*a[m-k] (standard convolution)
    # And sum_k p[k]*a[k-m] = (a*p)[m] = sum_k a[k]*p[m-k]
    # So grad_a[m] = 2 * (a*p)[m] = 2 * conv(a, p)[m]
    
    N = 1
    while N < 3 * n:
        N <<= 1
    A_f = fft(a, N)
    P_f = fft(p_weights, N)
    grad_full = ifft(A_f * P_f).real
    grad = 2.0 * grad_full[:n]
    return grad


def high_tau_gd(a, time_budget, tau_start=100, tau_end=10000, verbose=True,
                use_nesterov=True, restart_period=5000):
    """High-precision smoothed gradient descent with Nesterov acceleration."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.7

        if use_nesterov:
            y = a + beta * v
            y = np.maximum(y, 0)
            s_y = float(np.sum(y))
            if s_y < 1e-10:
                y = best_a.copy()
            else:
                y = y / s_y
        else:
            y = a.copy()

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        
        # Softmax weights (numerically stable)
        shifted = conv - max_conv
        max_exp = float(np.max(shifted * tau))
        exp_vals = np.exp(tau * shifted - max_exp)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        grad = smoothed_gd_step(y, p, 1.0)
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break
        
        # Adaptive step size with backtracking
        lr = min(1.0 / (g_norm + 1e-10), 3.0 / np.sqrt(n))
        
        for _ in range(20):
            if use_nesterov:
                v_new = beta * v - lr * grad
                a_new = y + v_new
            else:
                a_new = y - lr * grad
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                break
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            obj_y = compute_obj(y)
            # Simple acceptance: accept if improvement
            if obj_new < obj_y + 1e-12:
                break
            lr *= 0.5
        else:
            a_new = y  # no update

        if use_nesterov:
            v = v_new if 'v_new' in dir() else np.zeros(n)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        beta = min(0.98, beta + 0.00005)

        # Periodic restart from best
        if it - best_it > restart_period:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it
            if it > restart_period * 10:
                break

        if verbose and it % 3000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def log_param_optimize(a, time_budget, tau_min=100, tau_max=8000, verbose=True):
    """Optimize using log-parameterization with L-BFGS on support."""
    full_len = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    threshold = max(1e-8, a.max() * 1e-6)
    support = np.where(a > threshold)[0]
    if len(support) < 3:
        support = np.where(a > a.max() * 1e-10)[0]
    if len(support) < 3:
        return a

    b0 = np.log(np.maximum(a[support], 1e-30))
    b0 = b0 - b0.mean()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    n_stages = 4
    tau_stages = tau_min + (tau_max - tau_min) * (np.arange(n_stages) / max(1, n_stages - 1)) ** 1.5
    tau_stages = np.maximum(tau_stages, tau_min)

    current_b = b0.copy()
    for stage_idx, tau in enumerate(tau_stages):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        def obj_grad(b_vec):
            b_shifted = b_vec - np.max(b_vec)
            a_supp = np.exp(b_shifted)
            a_supp = a_supp / a_supp.sum()
            a_full = np.zeros(full_len)
            a_full[support] = a_supp
            conv = compute_conv(a_full)
            max_conv = float(np.max(conv))
            shifted = conv - max_conv
            max_shift = float(np.max(shifted * tau))
            exp_vals = np.exp(tau * shifted - max_shift)
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp
            obj_val = max_conv + np.log(sum_exp) / tau + max_shift / tau
            
            # Gradient
            N = 1
            while N < 3 * full_len:
                N <<= 1
            P_f = fft(p, N)
            A_f = fft(a_full, N)
            g_full = ifft(P_f * np.conj(A_f)).real
            grad_full = 2.0 * g_full[:full_len]
            grad_b = a_supp * (grad_full[support] - np.sum(a_supp * grad_full[support]))
            return obj_val, grad_b

        try:
            res = scipy_minimize(obj_grad, current_b, method='L-BFGS-B',
                                 jac=True,
                                 options={'maxiter': 300, 'gtol': 1e-10,
                                          'ftol': 1e-14, 'maxfun': 800,
                                          'maxls': 50})
            if res.x is not None:
                b_shifted = res.x - np.max(res.x)
                a_supp = np.exp(b_shifted)
                a_supp = a_supp / a_supp.sum()
                a_new = np.zeros(full_len)
                a_new[support] = a_supp
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    current_b = res.x.copy()
                    if verbose:
                        print(f"    LogOpt stage{stage_idx}: tau={tau:.0f}, obj={best_obj:.8f}")
        except Exception:
            pass

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.5
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
        a = a / a.sum()
    elif mode == 6:
        a = np.exp(-((np.arange(n) - n/2) / (n * 0.25)) ** 2)
        a = a / a.sum()
    elif mode == 7:
        # Two bumps
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) / 0.1) ** 2) + np.exp(-((x - 0.7) / 0.1) ** 2)
        a = a / a.sum()
    else:
        a = rng.uniform(0.05, 1, n)
        a = a / a.sum()
    return a

                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = diverse_init(n_init, rng, mode=5)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, obj={best_score:.8f}")

    # === Phase 1: Fast LP Polish on initial (120s) ===
    print(f"\n=== Phase 1: LP Polish (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1 = min(remaining * 0.15, 120.0)
    if p1 > 30:
        a_lp = lp_polish(best_seq, time_budget=p1, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_lp:.8f}")

    # === Phase 2: Multi-n Exploration (350s) ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p2 = min(remaining * 0.30, 350.0)
    
    n_options = []
    for n_try in [n_best - 200, n_best - 100, n_best + 100, n_best + 200,
                  n_best + 400, n_best + 600, n_best + 1000, n_best + 1500,
                  600, 900, 1200, 1500, 2000, 2500, 3000]:
        if 400 <= n_try <= 3500 and n_try != n_best:
            n_options.append(n_try)
    n_options = sorted(set(n_options))[:10]
    print(f"  Trying n: {n_options}")

    per_n = p2 / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_n, remaining - 20)
        if alloc < 25:
            break

        a_try = resample_sequence(best_seq, n_try)
        
        # Quick LP
        lp_t = alloc * 0.45
        if lp_t > 10:
            a_try = lp_polish(a_try, time_budget=lp_t, verbose=False)
        
        # Quick GD
        remaining = deadline - time.monotonic()
        gd_t = min(alloc * 0.35, remaining - 15)
        if gd_t > 8:
            a_try = high_tau_gd(a_try, time_budget=gd_t,
                                tau_start=100, tau_end=3000, verbose=False)
        
        obj_try = compute_obj(a_try)
        tag = " **" if obj_try < best_score else ""
        print(f"  n={n_try}: obj={obj_try:.8f}{tag}")
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try

    # === Phase 3: Cyclic LP+GD on best (300s) ===
    print(f"\n=== Phase 3: Cyclic LP+GD (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p3 = min(remaining * 0.25, 300.0)
    if p3 > 30:
        a_cyc = cyclic_lp_gp(best_seq, time_budget=p3, verbose=True)
        obj_cyc = compute_obj(a_cyc)
        if obj_cyc < best_score:
            best_score = obj_cyc
            best_seq = a_cyc.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj_cyc:.8f}")

    # === Phase 4: Log-param optimization (150s) ===
    print(f"\n=== Phase 4: Log-Param Optimization ===")
    remaining = deadline - time.monotonic()
    p4 = min(remaining * 0.12, 150.0)
    if p4 > 20:
        a_log = log_param_optimize(best_seq, time_budget=p4,
                                   tau_min=200, tau_max=10000, verbose=True)
        obj_log = compute_obj(a_log)
        if obj_log < best_score:
            best_score = obj_log
            best_seq = a_log.copy()
            print(f"  Phase4: obj={best_score:.8f} **")
        else:
            print(f"  Phase4: obj={obj_log:.8f}")

    # === Phase 5: Diverse Restarts (200s) ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p5 = min(remaining * 0.15, 200.0)
    n_restarts = max(1, int(p5 / 50))
    
    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(55, remaining - 15)
        if alloc < 20:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with current best
        blend = 0.2 + 0.15 * (r % 3)
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        # Optimize
        half = alloc * 0.5
        a_r = lp_polish(a_r, time_budget=half, verbose=False)
        remaining = deadline - time.monotonic()
        gd_alloc = min(alloc * 0.4, remaining - 10)
        if gd_alloc > 5:
            a_r = high_tau_gd(a_r, time_budget=gd_alloc,
                              tau_start=100, tau_end=4000, verbose=False)

        obj_r = compute_obj(a_r)
        tag = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}{tag}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()

    # === Phase 6: Final Intensive Polish (remaining) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s")

    if remaining > 20:
        alloc = min(remaining * 0.3, 100.0)
        a_f = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 100.0)
        a_f = high_tau_gd(best_seq, time_budget=alloc,
                          tau_start=500, tau_end=15000, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.25, 80.0)
        a_f = log_param_optimize(best_seq, time_budget=alloc,
                                 tau_min=500, tau_max=15000, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final Log: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 60.0)
        a_f = lp_polish(best_seq, time_budget=alloc, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Ultra LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 40.0)
        a_f = high_tau_gd(best_seq, time_budget=alloc,
                          tau_start=2000, tau_end=20000, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Ultra GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
