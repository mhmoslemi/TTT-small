    n = len(a)
    a = a / a.sum()  # normalize
    best_a = a.copy()
    best_obj = 2 * n * max(conv(a, a))
    
    tau = initial_tau  # smoothing parameter
    beta = 0.9  # Nesterov momentum
    v = zeros(n)
    y = a.copy()  # Nesterov lookahead
    
    while time < time_budget:
        # Update y (Nesterov step)
        y = a + beta * v
        y = max(y, 0)
        y = y / y.sum()
        
        # Compute gradient at y
        C = conv(y, y)  # FFT-based
        p = softmax(tau * (C - max(C)))
        grad = 2 * cross_corr(p, y)[:n]
        
        # Adaptive learning rate
        lr = ... (Barzilai-Borwein or fixed)
        
        # Update
        v = beta * v - lr * grad
        a_new = y + v
        a_new = max(a_new, 0)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        obj = 2 * n * max(conv(a_new, a_new))
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated gradient descent on a smoothed (log-sum-exp) proxy
for the peak of the self-convolution, with FFT-based gradient computation.
Explores multiple sequence lengths and diverse initializations.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n]


def fft_self_convolve(a):
    """Linear self-convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    C = fft_self_convolve(a_norm)
    return 2.0 * n * float(np.max(C))


def compute_gradient_smoothed(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*(a*a)[k])) w.r.t. a.
    Uses FFT for the cross-correlation.
    """
    n = len(a)
    C = fft_self_convolve(a)
    C_max = float(np.max(C))
    exp_vals = np.exp(tau * (C - C_max))
    p = exp_vals / float(np.sum(exp_vals))
    
    # grad[i] = 2 * sum_m p[i+m] * a[m] for m=0..n-1
    # This is 2 * (cross-correlation of p and a)[i]
    # cross_corr[k] = sum_m p[k+m] * a[m]
    
    # Using FFT: cross_corr = IFFT(conj(FFT(p)) * FFT(a))
    L = 1
    while L < 3 * n:
        L <<= 1
    P = fft(p, L)
    A = fft(a, L)
    cross = ifft(np.conj(P) * A).real[:n]
    
    return 2.0 * cross


def nesterov_optimize(a, time_budget, tau_init=10.0, tau_rate=1.005, verbose=False):
    """
    Nesterov-accelerated gradient descent on the smoothed max convolution.
    a is assumed to be normalized (sum = 1) and non-negative.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(fft_self_convolve(a)))
    
    beta = 0.9
    v = np.zeros(n)
    y = a.copy()
    
    tau = tau_init
    lr = 1.0 / np.sqrt(n)
    
    prev_a = a.copy()
    prev_grad = None
    it = 0
    best_it = 0
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
        else:
            y = y / s_y
        
        # Compute gradient
        grad = compute_gradient_smoothed(y, tau)
        grad_norm = float(np.linalg.norm(grad))
        
        # Barzilai-Borwein adaptive LR
        if prev_grad is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = grad - prev_grad
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-18 and abs(sy) > 1e-15:
                bb = ss / abs(sy)
                bb = float(np.clip(bb, 0.1 * lr, 3.0 * lr))
                lr = 0.7 * lr + 0.3 * bb
        
        # Update
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj = 2.0 * n * float(np.max(fft_self_convolve(a_new)))
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        prev_grad = grad.copy()
        a = a_new
        it += 1
        tau *= tau_rate
        lr *= 0.9999
        
        # Reset if stuck
        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            beta = max(0.5, beta * 0.95)
            best_it = it
            lr = 1.0 / np.sqrt(n)
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"      it={it}, obj={best_obj:.8f}, tau={tau:.1f}, time={elapsed:.0f}s")
    
    if verbose:
        print(f"    Nesterov done: obj={best_obj:.8f}, iters={it}")
    return best_a


def fourier_perturb(a, rng, strength=0.05):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def geometric_init(n, alpha=1.0):
    """Create a geometric-like initialization: a[j] = (j+1)^alpha."""
    j = np.arange(n) + 1.0
    a = j ** alpha
    return a / a.sum()
