"""
Optimized step function coefficient search combining:
1. Diverse initialization (U-shaped, cosine-power, parent)
2. Projected subgradient with Nesterov momentum for warm-up
3. Frank-Wolfe with sparse LP (top-M active constraints) and golden-section line search
4. Coordinate descent polish (new - exploits 1D structure of each coordinate)
5. Adaptive n exploration
6. Final intensive polish

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_sparse_conv_matrix(a, top_m):
    """Build matrix A (top_m x n) for the top M convolution positions. Returns A and selected lags."""
    n = len(a)
    conv = compute_conv(a)
    # Select top M lags
    if top_m < len(conv):
        idx = np.argsort(conv)[::-1][:top_m]
    else:
        idx = np.arange(len(conv))
    idx = np.sort(idx)
    
    A = np.zeros((len(idx), n))
    a_norm = a / float(np.sum(a))
    for row, k in enumerate(idx):
        for j in range(n):
            ki = k - j
            if 0 <= ki < n:
                A[row, j] = a_norm[ki]
    return A, idx


def solve_lp_minpeak(a, top_m=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for selected k, sum(g)=1, g>=0."""
    n = len(a)
    if top_m is None or top_m >= 2 * n - 1:
        top_m = 2 * n - 1
    top_m = min(top_m, 2 * n - 1)
    
    a_norm = a / float(np.sum(a))
    A, idx = build_sparse_conv_matrix(a, top_m)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g_lp, t_max=1.5):
    """
    Find optimal blend factor t in [0, t_max] that minimizes obj((1-t)*a + t*g_lp).
    Uses golden section search.
    """
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi

    lo, hi = 0.0, t_max

    def obj_at(t):
        if t < 0:
            t = 0
        a_try = (1 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        a_try = a_try / s
        return compute_obj(a_try)

    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    for _ in range(30):
        if hi - lo < 1e-8:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)

    # Try some standard candidates
    for t_try in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.2, 1.5]:
        if t_try <= t_max:
            obj_try = obj_at(t_try)
            if obj_try < obj_best:
                obj_best = obj_try
                t_best = t_try

    return t_best, obj_best


def subgradient_descent(a, time_budget=60.0):
    """Subgradient descent on the probability simplex with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.03
    it = 0
    momentum = np.zeros(n)
    beta = 0.7
    best_it = 0
    counter = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))

        # Subgradient of max_k (a*a)[k] at a
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n and j2 != j1:
                grad[i] += a[j2]

        # Nesterov momentum step
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            counter = 0
        else:
            counter += 1

        a = a_new
        lr *= 0.999
        it += 1

        if counter > 3000:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = 0.02
            counter = 0

    return best_a


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def cosine_power_init(n, p=2.0):
    """Cosine-power initialization: high at edges."""

    a = np.cos(np.pi * x) ** (2 * p) + 0.01
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:

    return a / s


def arcsine_init(n):

    if s < 1e-10:

    return a / s

def coordinate_descent(a, time_budget=120.0, passes=10):
    """
    Coordinate descent polish. For each coordinate, do 1D golden section search.
    Exploits the structure: conv[k] as function of a[i] is linear (or quadratic at k=2i).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)
    it = 0

    while time.monotonic() - start < time_budget and it < passes * 20:
        # Random coordinate order
        order = rng.permutation(n)
        pass_done = 0

        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            pass_done += 1

            # Compute base convolution without a[i]'s contribution
            ai = a[i]
            a[i] = 0.0
            base_conv = compute_conv(a)
            s_other = float(np.sum(a))
            a[i] = ai

            # Now search for optimal a[i]
            # conv[k] = base_conv[k] + ai*(a[k-i] + a[i-k]) for k != 2i
            # conv[2i] = base_conv[2i] + ai^2
            # Objective: 2n * max_k conv[k] / (s_other + ai)^2

            # Search range for ai
            lo_ai = 0.0
            hi_ai = max(ai * 4.0, 4.0 / n)

            def obj_coord(x):
                if x < 0:
                    x = 0.0
                # Compute max conv value
                max_conv = 0.0
                for k in range(2 * n - 1):
                    val = base_conv[k]
                    ki = k - i
                    if 0 <= ki < n:
                        val += x * a[ki]
                    ik = i - k
                    if 0 <= ik < n and ik != ki:
                        val += x * a[ik]
                    # Handle k = 2i case: the term a[i]*a[i] = x^2
                    if 2 * i < 2 * n - 1 and ki == i and ik == i:
                        # Already counted once above via ki, need to adjust
                        pass
                    if val > max_conv:
                        max_conv = val
                total_s = s_other + x
                if total_s < 0.01:
                    return np.inf
                return 2.0 * n * max_conv / (total_s * total_s)

            # Golden section on [lo_ai, hi_ai]
            phi = (1 + np.sqrt(5)) / 2
            gr = 2 - phi
            c_age = lo_ai + gr * (hi_ai - lo_ai)
            d_age = lo_ai + (1 - gr) * (hi_ai - lo_ai)
            fc_age = obj_coord(c_age)
            fd_age = obj_coord(d_age)

            for _ in range(12):
                if hi_ai - lo_ai < 1e-10:
                    break
                if fc_age < fd_age:
                    hi_ai = d_age
                    d_age = c_age
                    fd_age = fc_age
                    c_age = lo_ai + gr * (hi_ai - lo_ai)
                    fc_age = obj_coord(c_age)
                else:
                    lo_ai = c_age
                    c_age = d_age
                    fc_age = fd_age
                    d_age = lo_ai + gr * (hi_ai - lo_ai)
                    fd_age = obj_coord(d_age)

            ai_opt = c_age if fc_age < fd_age else d_age
            obj_opt = min(fc_age, fd_age)
            obj_current = obj_coord(ai)

            if obj_opt < obj_current - 1e-12:
                a[i] = ai_opt
                s_new = float(np.sum(a))
                if s_new > 0.01:
                    a = a / s_new
                    new_obj = compute_obj(a)
                    if new_obj < best_obj:
                        best_obj = new_obj

        it += 1

    return a

def interpolate_sequence(a_old, n_new):
    """Interpolate a sequence from old length to new length."""
    n_old = len(a_old)
    if n_old == n_new:
        return a_old.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a_old)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 0.01:
        a_new = a_new / s
    return a_new

        lp_budget = min(400.0, remaining * 0.4)
        a_lp = lp_polish(best_seq, time_budget=lp_budget, max_iters=150, sparse=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP done: obj={best_score:.8f} **")
        else:
            print(f"  LP done: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Coordinate descent polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 3: Coordinate Descent (budget={min(150, remaining*0.15):.0f}s) ===")
        cd_budget = min(150.0, remaining * 0.15)
        a_cd = coordinate_descent(best_seq, time_budget=cd_budget, passes=15)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD done: obj={best_score:.8f} **")
        else:
            print(f"  CD done: obj={obj_cd:.8f} (no improvement)")

    # === Phase 4: Second LP round from CD result ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 4: LP Round 2 (budget={min(200, remaining*0.2):.0f}s) ===")
        lp2_budget = min(200.0, remaining * 0.2)
        # Small perturbation
        noise = rng.normal(0, 0.003, len(best_seq))
        a_pert = best_seq + noise
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_lp2 = lp_polish(a_pert, time_budget=lp2_budget, max_iters=80, sparse=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP2 done: obj={best_score:.8f} **")
        else:
            print(f"  LP2 done: obj={obj_lp2:.8f} (no improvement)")

    # === Phase 5: n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 5: n Exploration ===")
        n_best = len(best_seq)
        
        n_options = sorted(set([
            n_best - 150, n_best - 100, n_best - 50,
            n_best + 50, n_best + 100, n_best + 200, n_best + 300,
            500, 600, 700, 900, 1000, 1100, 1200, 1400, 1600
        ]))
        n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_best) >= 30]
        n_options = n_options[:6]
        print(f"  Current best n={n_best}, trying: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            a_interp = interpolate_sequence(best_seq, n_try)
            # Quick optimization
            sub_budget = min(remaining * 0.35, 60.0)
            a_opt = fast_optimize(a_interp, sub_budget)
            obj_opt = compute_obj(a_opt)
            
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                n_best = n_try
                print(f"  n={n_try}: obj={obj_opt:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 6: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 6: Diverse Restarts ===")
        n_cur = len(best_seq)
        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            mode = (r * 3) % 9
            a_r = diverse_init(n_cur, rng, mode=mode)
            blend = 0.3 + 0.1 * r
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
            
            sub_budget = min(remaining * 0.25, 50.0)
            a_r = fast_optimize(a_r, sub_budget)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: Final Polish (budget={min(remaining-10, 150):.0f}s) ===")
        final_budget = min(remaining - 10, 150.0)
        # Coordinate descent first
        cd_time = min(40.0, final_budget * 0.3)
        if cd_time > 5:
            a_final = coordinate_descent(best_seq, time_budget=cd_time, passes=8)
            obj_f = compute_obj(a_final)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_final.copy()
        
        # Then LP
        lp_time = final_budget - cd_time - 5
        if lp_time > 10:
            a_final = lp_polish(best_seq, time_budget=lp_time, max_iters=50, sparse=True)
            obj_f = compute_obj(a_final)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_final.copy()
                print(f"  Final LP: obj={obj_f:.8f} **")

        # One more CD
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_final = coordinate_descent(best_seq, time_budget=min(remaining - 5, 30.0), passes=5)
            obj_f = compute_obj(a_final)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_final.copy()
                print(f"  Final CD: obj={obj_f:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
