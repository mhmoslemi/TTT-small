"""
Fast subgradient optimizer with Nesterov acceleration, peak focusing,
multi-start exploration, and LP final polish.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def weighted_topk_gradient(a, conv, K=5, power=2.0):
    """Compute weighted combination of gradients at top-K peaks."""
    n = len(a)
    # Find top K using argpartition for speed
    if K >= 2 * n - 1:
        top_idx = np.arange(2 * n - 1)
        top_vals = conv
    else:
        part_idx = np.argpartition(conv, -K)[-K:]
        part_idx = part_idx[np.argsort(conv[part_idx])[::-1]]
        top_idx = part_idx
        top_vals = conv[top_idx]
    
    # Weights: emphasize larger peaks
    max_val = top_vals[0]
    if max_val < 1e-15:
        return np.zeros(n), top_idx, max_val
    
    w = (top_vals / max_val) ** power
    w_sum = w.sum()
    if w_sum < 1e-15:
        return np.zeros(n), top_idx, max_val
    w = w / w_sum
    
    # Combined gradient
    g = np.zeros(n)
    for wi, ki in zip(w, top_idx):
        ki = int(ki)
        m_start = max(0, ki - n + 1)
        m_end = min(ki, n - 1)
        if m_start <= m_end:
            j_hi = ki - m_start
            j_lo = ki - m_end
            g[m_start:m_end + 1] += 2.0 * wi * a[j_lo:j_hi + 1][::-1]
    
    return g, top_idx, max_val


def fast_subgradient(a, time_budget=60.0, K=5, verbose=False, seed=42):
    """Fast subgradient with momentum and adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    lr = 1.5 / np.sqrt(n)
    gamma = 0.8  # momentum
    v = np.zeros(n)
    stall = 0
    best_stall = 0
    it = 0
    no_restarts = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        g, top_idx, max_val = weighted_topk_gradient(a, conv, K=K)
        
        # Momentum
        v = gamma * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall >= 30:
                lr *= 0.85
                v *= 0.5
                stall = 0
                stall_count = getattr(fast_subgradient, '_stall_count', 0)
                if lr < 1e-10:
                    # Restart with perturbation
                    a = best_a * np.exp(rng.normal(0, 0.01, n))
                    a = np.maximum(a, 0)
                    s_r = a.sum()
                    if s_r > 1e-10:
                        a = a / s_r
                    v = np.zeros(n)
                    lr = 1.5 / np.sqrt(n)
                    no_restarts += 1
                    if no_restarts > 10:
                        break
        
        # Slowly decay lr
        lr = max(1e-12, lr * 0.9997)
        it += 1
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"      SG it{it}: obj={best_obj:.8f}, lr={lr:.2e}, t={elapsed:.0f}s")
    
    return best_a
