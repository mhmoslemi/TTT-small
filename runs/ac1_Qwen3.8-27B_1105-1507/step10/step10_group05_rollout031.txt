"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated smoothed gradient descent with adaptive tau,
LP-based iteration with backtracking line search, spectral initialization
for length changes, and diversity-preserving restarts.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_obj_fast(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT.
    Returns (gradient, conv)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    # Compute gradient: grad_i = 2 * sum_k p_k * a_{k-i} (approximately)
    # Using FFT: this is a cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    grad_full = ifft(np.conj(A) * P).real
    grad = 2.0 * grad_full[:n]
    
    return grad, conv


def nesterov_smoothed_gd(a, time_budget=200.0, tau_init=50.0, tau_end=2000.0, 
                          verbose=False, seed=42):
    """Nesterov accelerated smoothed gradient descent with cosine annealing on tau."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    beta = 0.9
    base_lr = 5.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    it = 0
    best_it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Cosine annealing for tau
        cos_val = 0.5 * (1 + np.cos(np.pi * progress))
        tau = tau_init + (tau_end - tau_init) * (1 - cos_val)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        grad, conv = smoothed_gradient_fft(y, tau)

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj_fast(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            patience = 0
        else:
            patience += 1

        a = a_new
        it += 1

        # Adaptive learning rate decay
        lr = max(1e-10, lr * 0.9997)

        # Reset if stuck too long
        if patience > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * (0.5 + 0.5 * rng.random())
            patience = 0

        if verbose and it % 5000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

    return best_a


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_minpeak(a, warm_start=None, timeout=30):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, optimal_M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start
        except:
            pass

    best_g = None
    best_M = None
    start = time.monotonic()

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.PDLP, cp.SCS]
    for solver in solvers:
        if time.monotonic() - start > timeout:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    if best_g is None or M_val < best_M:
                        best_g = g_val
                        best_M = M_val
                    break
        except Exception:
            continue

    return best_g, best_M


def lp_iteration_nesterov(a, time_budget=300.0, verbose=False, seed=42):
    """
    LP-based iteration with Nesterov momentum and backtracking line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    momentum_vec = np.zeros(n)
    prev_g = None
    t_momentum = 0.7
    it = 0
    best_it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # Solve LP
        g_lp, M_lp = solve_lp_minpeak(a, warm_start=prev_g if prev_g is not None else None,
                                       timeout=min(60, remaining // 2))
        if g_lp is None:
            break

        # Nesterov extrapolation
        if prev_g is not None:
            diff = g_lp - prev_g
            momentum_vec = t_momentum * momentum_vec + (1 - t_momentum) * diff
            g_extrap = g_lp + 0.3 * momentum_vec
            g_extrap = np.maximum(g_extrap, 0)
            s_extrap = float(np.sum(g_extrap))
            if s_extrap > 1e-10:
                g_extrap = g_extrap / s_extrap
                # Accept extrapolated if it's not too bad
                if compute_obj_fast(g_extrap) < best_obj * 1.02:
                    g_target = g_extrap
                else:
                    g_target = g_lp
            else:
                g_target = g_lp
        else:
            g_target = g_lp
            momentum_vec = np.zeros(n)

        prev_g = g_lp.copy()
        it += 1

        # Backtracking line search
        t = 1.0
        a_prev = a.copy()
        obj_prev = best_obj
        for _ in range(15):
            a_try = (1.0 - t) * a + t * g_target
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                t *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_fast(a_try)
            
            # Armijo condition (simplified: just check improvement)
            if obj_try < obj_prev - 1e-10:
                break
            t *= 0.5

        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            best_it = it
            patience = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t:.3f}, M_lp={M_lp:.8f}")
        else:
            patience += 1
            # Try a smaller fixed step
            t_fixed = max(0.01, 0.1 / (1 + patience * 0.1))
            a_try = (1.0 - t_fixed) * a + t_fixed * g_target
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj_fast(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    patience = 0

        # Perturbation if stuck
        if patience > 20:
            strength = min(0.1, 0.01 * (patience - 20))
            a = best_a * np.exp(rng.normal(0, strength, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            else:
                a = best_a.copy()
            prev_g = None
            momentum_vec = np.zeros(n)
            patience = 0

        # Convergence check via dual gap
        if it % 10 == 0:
            conv_a = compute_conv(a)
            max_conv_a = float(np.max(conv_a))
            dual_gap = max_conv_a - M_lp
            if verbose and it % 20 == 0:
                print(f"    LP it{it}: dual_gap={dual_gap:.8f}")
            if dual_gap < 1e-9:
                break

    return best_a


def fourier_perturbation(a, rng, strength=0.05):
    """Add low-frequency Fourier modes as structured perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(10, n // 20)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / math.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturbation(a, rng, strength=0.05):
    """Add random multiplicative perturbation."""
    n = len(a)
    pert = 1.0 + rng.normal(0, strength, n)
    a_new = a * np.maximum(pert, 0.01)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic spline interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use scipy's interpolation
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(x_old, a)
        a_new = cs(x_new)
    except:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def spectral_init(new_n, reference=None, rng=None):
    """
    Create a new sequence of length new_n using spectral methods.
    If reference is given, use it to guide the initialization.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    
    if reference is not None:
        # Resample from reference
        a = resample_sequence(reference, new_n)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
        return a
    
    # Default: slightly perturbed uniform
    a = np.ones(new_n) + 0.1 * rng.normal(0, 1, new_n)
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated smoothed gradient descent with adaptive tau,
LP-based iteration with backtracking line search and momentum, spectral
initialization for length changes, and diversity-preserving restarts.
Key innovations: adaptive momentum schedule, dual-gap-based convergence
detection, multi-scale tau annealing, and structured perturbation restarts.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.interpolate import CubicSpline
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_fast(a):
    """Compute objective: 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT.
    Returns (gradient vector of length n, conv array of length 2n-1).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    # Gradient of conv(a,a)_k w.r.t. a_i = 2*a_{k-i} (for valid indices)
    # So grad_i = 2 * sum_k p_k * a_{k-i}
    # This is a cross-correlation computable via FFT
    N = 1
    while N < 3 * n:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    grad_full = ifft(np.conj(A) * P).real
    grad = 2.0 * grad_full[:n]

    return grad, conv


def nesterov_smoothed_gd(a, time_budget=200.0, tau_init=50.0, tau_end=2000.0,
                          verbose=False, seed=42):
    """
    Nesterov accelerated smoothed gradient descent with cosine annealing on tau.
    Includes adaptive learning rate and patience-based restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    beta = 0.9
    base_lr = 5.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Cosine annealing for tau
        cos_val = 0.5 * (1 + np.cos(np.pi * progress))
        tau = tau_init + (tau_end - tau_init) * (1 - cos_val)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        grad, conv = smoothed_gradient_fft(y, tau)

        # Step
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj_fast(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new
        it += 1

        # Adaptive learning rate
        lr = max(1e-10, lr * 0.9997)

        # Patience-based restart
        if patience > 3000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * (0.5 + 0.5 * rng.random())
            patience = 0

        if verbose and it % 5000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

    return best_a


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_minpeak(a, warm_start=None, timeout=60):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, optimal_M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start
        except Exception:
            pass

    best_g = None
    best_M = None
    start = time.monotonic()

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.PDLP, cp.SCS]
    for solver in solvers:
        if time.monotonic() - start > timeout:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    g_val = g_val / s_val
                    M_val = float(M.value)
                    if best_g is None or M_val < best_M:
                        best_g = g_val
                        best_M = M_val
                    break
        except Exception:
            continue

    return best_g, best_M


def lp_iteration_nesterov(a, time_budget=300.0, verbose=False, seed=42):
    """
    LP-based iteration with Nesterov momentum and backtracking line search.
    Dual-gap monitoring for convergence detection.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    momentum_vec = np.zeros(n)
    prev_g = None
    t_momentum = 0.6
    it = 0
    patience = 0
    best_gap = np.inf

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # Solve LP with timeout
        lp_timeout = min(45, max(5, remaining // 3))
        g_lp, M_lp = solve_lp_minpeak(a, warm_start=prev_g, timeout=lp_timeout)
        if g_lp is None:
            break

        # Nesterov extrapolation on LP solutions
        if prev_g is not None:
            diff = g_lp - prev_g
            diff_norm = np.linalg.norm(diff)
            momentum_vec = t_momentum * momentum_vec + (1 - t_momentum) * diff
            # Adaptive extrapolation coefficient
            extrapolation = 0.25
            if diff_norm > 1e-10:
                # Scale extrapolation by recent step size
                extrapolation = min(0.4, 0.2 / (1 + np.linalg.norm(momentum_vec) * 10))
            g_extrap = g_lp + extrapolation * momentum_vec
            g_extrap = np.maximum(g_extrap, 0)
            s_extrap = float(np.sum(g_extrap))
            if s_extrap > 1e-10:
                g_extrap = g_extrap / s_extrap
                # Accept if not too much worse
                obj_extrap = compute_obj_fast(g_extrap)
                if obj_extrap < best_obj * 1.05:
                    g_target = g_extrap
                else:
                    g_target = g_lp
            else:
                g_target = g_lp
        else:
            g_target = g_lp
            momentum_vec = np.zeros(n)

        prev_g = g_lp.copy()
        it += 1

        # Backtracking line search (Armijo-like)
        t = 1.0
        a_base = a.copy()
        obj_base = compute_obj_fast(a_base)
        accepted = False
        for _ in range(12):
            a_try = (1.0 - t) * a_base + t * g_target
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                t *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_fast(a_try)

            if obj_try < obj_base - 1e-12:
                accepted = True
                break
            t *= 0.5

        if accepted and obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            patience = 0
        else:
            # Try smaller step or perturbation
            t_small = 0.1 / (1.0 + 0.05 * patience)
            a_try = (1.0 - t_small) * a + t_small * g_target
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj_fast(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    patience = 0
                else:
                    patience += 1
            else:
                patience += 1

        # Perturbation if stuck
        if patience > 15:
            strength = min(0.12, 0.005 * (patience - 15) + 0.02)
            a = best_a * np.exp(rng.normal(0, strength, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            else:
                a = best_a.copy()
            prev_g = None
            momentum_vec = np.zeros(n)
            patience = 0

        # Dual gap convergence check
        if it % 5 == 0:
            conv_a = compute_conv(a)
            max_conv_a = float(np.max(conv_a))
            dual_gap = max_conv_a - M_lp
            best_gap = min(best_gap, dual_gap)
            if verbose and it % 10 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, M_lp={M_lp:.8f}, gap={dual_gap:.2e}")
            if dual_gap < 1e-10 and patience > 5:
                break

    return best_a


def fourier_perturbation(a, rng, strength=0.05):
    """Add low-frequency Fourier modes as structured perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(10, n // 20)
    for k in range(1, n_modes + 1):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / math.sqrt(k)
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturbation(a, rng, strength=0.05):
    """Add random multiplicative perturbation."""
    n = len(a)
    pert = 1.0 + rng.normal(0, strength, n)
    a_new = a * np.maximum(pert, 0.01)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def cosine_blend_perturbation(a, n_current, rng):
    """Blend with a cosine-shaped pattern."""
    x = np.linspace(0, 1, n_current)
    k = rng.integers(1, 5)
    a_struct = 0.5 + 0.5 * np.cos(2.0 * np.pi * k * x + rng.uniform(0, 2 * np.pi))
    a_struct = np.maximum(a_struct, 0.01)
    a_struct = a_struct / a_struct.sum()
    a_new = 0.65 * a + 0.35 * a_struct
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    return a_new


def uniform_blend_perturbation(a, n_current, rng):
    """Blend with uniform."""
    a_new = 0.7 * a + 0.3 * np.ones(n_current) / n_current
    a_new = a_new / a_new.sum()
    return a_new


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic spline interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    try:
        cs = CubicSpline(x_old, a)
        a_new = cs(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                                        tau_init=500.0, tau_end=8000.0,
                                        verbose=True, seed=7777)
        obj_final = compute_obj_fast(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
        else:
            print(f"  Final GD: obj={obj_final:.8f}")

    # One more LP if time remains
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.4, 80.0)
        a_final = lp_iteration_nesterov(best_seq, time_budget=alloc,
                                         verbose=False, seed=5555)
        obj_final = compute_obj_fast(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Extra LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_fast(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
