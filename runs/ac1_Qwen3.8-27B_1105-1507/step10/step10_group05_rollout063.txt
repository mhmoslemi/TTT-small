        tau = tau_init + (tau_end - tau_init) * (1 - cos_val)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        grad, conv = smoothed_gradient_fft(y, tau)
        
        # Barzilai-Borwein step size adjustment
        if prev_grad is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = grad - prev_grad
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                bb_lr = min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))
                lr = 0.5 * lr + 0.5 * bb_lr
        
        # Gradient step
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        prev_grad = grad.copy()
        a = a_new
        it += 1
        
        lr = max(1e-8, lr * 0.999)
        
        # Stagnation handling
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it
            if it - best_it > 20000:
                break
        
        if verbose and it % 2000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")
    
    return best_a


def penalty_gd(a, time_budget=100.0, verbose=False):
    """
    Penalty-based gradient descent. Minimizes smoothed objective + penalty for
    deviation from uniform, to encourage flat convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 5.0 / np.sqrt(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 1:
            break
        
        tau = 50.0 + 10.0 * it
        
        grad, conv = smoothed_gradient_fft(a, tau)
        
        # Add regularization: penalize deviation from uniform
        uniform = np.ones(n) / n
        reg_grad = 0.01 * (a - uniform)
        
        total_grad = grad + reg_grad
        
        # Backtracking line search
        a_old = a.copy()
        obj_old = best_obj
        step = lr
        improved = False
        for _ in range(10):
            a_try = a - step * total_grad
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                step *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < obj_old - 1e-14:
                a = a_try
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a.copy()
                    best_it = it
                improved = True
                break
            step *= 0.5
        
        if not improved:
            step = lr * 0.01
            a_try = a - step * total_grad
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a = a_try / s_try
        
        it += 1
        lr *= 0.999
        
        if it - best_it > 3000:
            break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_init(n, rng, n_modes=5):
    """Initialize with low-frequency Fourier structure."""
    x = np.arange(n)
    a = np.ones(n) * 0.5
    for k in range(1, n_modes + 1):
        amp = rng.uniform(0.05, 0.2) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * x / n + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1500):

    Main search combining fast LP iteration, corrected smoothed gradient descent,
    penalty-based refinement, and systematic length exploration.
    """

    deadline = start_time + float(budget_s) - 20.0

    best_score = np.inf

    # Initialize from parent or default
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_seq = parent.copy()
                best_score = compute_obj(parent)
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")
    
    # === Phase 1: Fast LP iteration (main workhorse) ===
    print(f"\n=== Phase 1: LP Iteration (budget: 400s) ===")

    if t1_budget > 30:
        a1 = lp_iteration(best_seq, time_budget=t1_budget, verbose=True, seed=42)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj1:.8f}")
    
    # === Phase 2: Smoothed GD fine-tuning ===
    print(f"\n=== Phase 2: Smoothed GD (budget: 150s) ===")
    t2_budget = min(150.0, deadline - time.monotonic() - 20)
    if t2_budget > 20:
        a2 = smoothed_gd(best_seq, time_budget=t2_budget, tau_init=30.0, tau_end=800.0, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj2:.8f}")
    
    # === Phase 3: Penalty GD ===
    print(f"\n=== Phase 3: Penalty GD (budget: 100s) ===")
    t3_budget = min(100.0, deadline - time.monotonic() - 20)
    if t3_budget > 15:
        a3 = penalty_gd(best_seq, time_budget=t3_budget, verbose=False)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj3:.8f}")
    
    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_best = len(best_seq)
    
    n_candidates = []
    for dn in [-200, -100, 100, 200, 300, -300, 500]:
        n_try = n_best + dn
        if 400 <= n_try <= 2500:
            n_candidates.append(n_try)
    for n_abs in [600, 700, 900, 1000, 1200, 1500]:
        if n_abs not in n_candidates:
            n_candidates.append(n_abs)
    
    n_candidates = sorted(set(n_candidates))
    # Sort by distance from current best n (try closest first)
    n_candidates.sort(key=lambda x: abs(x - n_best))
    n_candidates = n_candidates[:6]  # Try at most 6 different lengths

        else:

        obj_final = compute_obj(a_final)

            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

            print(f"  Final GD: obj={obj_final:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 30:

    return result
