def compute_conv(a):
    """Linear convolution via FFT, returns an array of length 2*len(a)-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size *= 2
    # If a contains NaNs or not finites, return linear convolution with finite array
    if not a.astype(np.compleX).isfinite:
        return np.convolve(a, a, mode='full')[:2*n-1]
    fa = rfft(a.astype(np.float64), size)
    conv = irfft(fa * fa, size)
    return conv
