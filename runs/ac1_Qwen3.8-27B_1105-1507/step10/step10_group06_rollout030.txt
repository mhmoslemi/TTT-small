Optimizes step function coefficients using a combination of:
1. LP-based alternating minimization (best response dynamics)
2. Cyclic coordinate descent with exact 1D subproblem solving
3. Multiple diverse restarts and adaptive length selection
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix_sparse(a):
    """Build sparse convolution matrix."""
    from scipy.sparse import dia_matrix
    n = len(a)
    # (2n-1) x n matrix
    # Row k, col j: A[k,j] = a[j] if 0<=k-j<n, else 0
    # This is a banded matrix
    # For row k, the non-zeros are at columns j where 0<=j<n and 0<=k-j<n
    # i.e., j in [max(0, k-n+1), min(k, n-1)]
    # The value is a[j]
    # This is a "convolution matrix" - it's banded with n diagonals
    # Actually it's a Toeplitz-like matrix
    # Let me just build it as a dense matrix for simplicity (n=800 -> 1599x800 = 1.3M entries)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def solve_conv_lp(a):
    """Solve min M s.t. (a*g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_sparse(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def coord_descent(a, n_passes=10, verbose=False):
    """
    Cyclic coordinate descent with exact 1D subproblem solving.
    a is normalized to sum=1.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    conv = compute_conv(a)
    best_obj = float(np.max(conv))
    
    for p in range(n_passes):
        for i in range(n):
            old = a[i]
            if old < 1e-15:
                # Try a small positive value
                # The 1D problem: minimize max(conv_new)
                # If old=0, the change is: conv[2i] += x^2, conv[k] += 2*x*a[k-i] for k in [i,i+n-1], k≠2i
                pass
            
            # Precompute for the 1D subproblem
            # A_quad = conv[2i] - old^2 (the constant part of the quadratic)
            k_quad = 2*i
            if k_quad < 2*n-1:
                A_quad = conv[k_quad] - old*old
            else:
                A_quad = 0  # shouldn't happen for valid i
            
            # For linear parts: k in [i, min(i+n-1, 2n-2)], k != 2i
            k_start = i
            k_end = min(i + n - 1, 2*n - 2)
            ks = np.arange(k_start, k_end + 1)
            ks = ks[ks != k_quad]  # exclude the quadratic term
            
            if len(ks) > 0:
                # B[k] = conv[k] - 2*old*a[k-i]
                # m[k] = 2*a[k-i]
                offsets = ks - i  # these are the indices j such that a[j] = a[k-i]
                B_lin = conv[ks] - 2*old*a[offsets]
                m_lin = 2.0 * a[offsets]
            else:
                B_lin = np.array([0.0])
                m_lin = np.array([0.0])
            
            # Constant part: max of conv[k] for k not in [k_start, k_end] and k != k_quad
            mask = np.ones(2*n-1, dtype=bool)
            mask[k_start:k_end+1] = False
            if k_quad < 2*n-1:
                mask[k_quad] = True  # we'll handle it separately
            C_const = float(np.max(conv[mask])) if mask.any() else 0.0
            
            # Define h(x) = max(x^2 + A_quad, max(B_lin + m_lin*x), C_const)
            def h(x):
                quad_val = x*x + A_quad
                lin_vals = B_lin + m_lin * x
                lin_max = float(np.max(lin_vals)) if len(lin_vals) > 0 else 0.0
                return max(quad_val, lin_max, C_const)
            
            # Minimize h(x) for x >= 0

            x_max = max(old * 2, best_obj / (2*max(a.max(), 1e-10)) + 1)
            x_max = min(x_max, 10.0)  # cap for safety

            x_new = result.x

            # Update conv

            if len(ks) > 0:

            new_max = float(np.max(conv))
            if new_max < best_obj:

        if verbose and (p+1) % 5 == 0:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based alternating minimization (best response dynamics)
2. Cyclic coordinate descent with exact 1D subproblem solving
3. Fourier-based perturbation to escape local minima
4. Multiple diverse restarts and adaptive length selection
5. Nesterov-accelerated gradient polishing with smoothed max

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a, solver_list=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.PDLP, cp.SCS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def coord_descent(a, n_passes=20, time_budget=60.0, verbose=False):
    """
    Cyclic coordinate descent with exact 1D subproblem solving.
    Minimize max_k (a*a)_k over a >= 0 (scale-invariant, so sum not fixed).
    Returns (a_optimized, best_max_conv).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    conv = compute_conv(a)
    best_max_conv = float(np.max(conv))
    start = time.monotonic()

    for p in range(n_passes):
        if time.monotonic() - start > time_budget:
            break
        for i in range(n):
            old = a[i]
            k_quad = 2 * i
            has_quad = k_quad < 2 * n - 1

            # Quadratic part constant
            if has_quad:
                A_quad = conv[k_quad] - old * old
            else:
                A_quad = 0.0

            # Linear parts: k in [i, min(i+n-1, 2n-2)], k != 2i
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            ks = np.arange(k_start, k_end + 1)
            if has_quad:
                mask_ks = ks != k_quad
                ks = ks[mask_ks]

            if len(ks) > 0:
                offsets = ks - i
                B_lin = conv[ks] - 2.0 * old * a[offsets]
                m_lin = 2.0 * a[offsets]
            else:
                B_lin = np.array([-np.inf])
                m_lin = np.array([0.0])

            # Constant part: max of conv[k] for k NOT in affected range
            affected_mask = np.zeros(2 * n - 1, dtype=bool)
            affected_mask[k_start:k_end + 1] = True
            C_const = 0.0
            const_mask = ~affected_mask
            if const_mask.any():
                C_const = float(np.max(conv[const_mask]))

            # Define h(x) for x >= 0
            B_lin_safe = B_lin.copy()
            m_lin_safe = m_lin.copy()

            def h1d(x, A_q=A_quad, B_l=B_lin_safe, m_l=m_lin_safe, C_c=C_const):
                quad_val = x * x + A_q
                if len(B_l) > 0 and np.any(np.isfinite(B_l)):
                    lin_max = float(np.max(B_l + m_l * x))
                else:
                    lin_max = 0.0
                return max(quad_val, lin_max, C_c)

            # Find upper bound for x
            x_upper = max(old * 3.0 + 0.1, best_max_conv + 0.1)
            x_upper = min(x_upper, 2.0)

            # Minimize using bounded brent
            try:
                result = minimize_scalar(h1d, bounds=(0.0, x_upper), method='bounded',
                                         options={'xatol': 1e-12, 'maxiter': 60})
                x_new = float(result.x)
            except Exception:
                x_new = old

            # Update a and conv
            a[i] = x_new
            if has_quad:
                conv[k_quad] = A_quad + x_new * x_new
            if len(ks) > 0:
                conv[ks] = B_lin + m_lin * x_new

            current_max = float(np.max(conv))
            if current_max < best_max_conv:
                best_max_conv = current_max

        if verbose and (p + 1) % 5 == 0:
            print(f"    CD pass {p+1}/{n_passes}: max_conv={best_max_conv:.10f}, "
                  f"time={time.monotonic()-start:.1f}s")

    return a, best_max_conv


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 8
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of (a*a)_k using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def nesterov_polish(a, time_budget, tau_init=30.0, tau_max=2000.0):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adaptive_lr = 0.5 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.1 * base_lr), 3.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adaptive_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr *= 0.9999
        tau = min(tau * 1.01, tau_max)

    return best_a


def lp_alternate_cd(a, time_budget, seed=42, verbose=True,
                    lp_solvers=None, cd_passes_per_lp=3):
    """
    Combined LP alternating + coordinate descent.
    Each LP step is followed by a few CD passes for fast local improvement.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    t_blend = 0.4
    stall_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_conv_lp(a, solver_list=lp_solvers)
        lp_time = time.monotonic() - t0

        if g_lp is None:
            break

        iter_count += 1

        # Try blending
        candidates_t = [t_blend, t_blend * 0.5, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0]
        if iter_count > 50:
            candidates_t.extend([0.05, 0.02, 0.01])

        best_local_obj = best_obj
        best_local_a = best_a.copy()

        for t in candidates_t:
            t = min(max(t, 0.001), 1.0)
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall_count = 0

            # CD polishing after LP step
            cd_time = min(remaining * 0.15, 10.0)
            if cd_time > 2:
                a_cd, _ = coord_descent(a, n_passes=cd_passes_per_lp,
                                         time_budget=cd_time, verbose=False)
                a_cd = a_cd / a_cd.sum()
                obj_cd = compute_obj(a_cd)
                if obj_cd < best_obj:
                    best_obj = obj_cd
                    best_a = a_cd.copy()
                    a = best_a.copy()

            if verbose and iter_count % 20 == 0:
                print(f"    LP+CD {iter_count}: obj={best_obj:.10f}, "
                      f"t={t_blend:.3f}, lp_time={lp_time:.2f}s")
        else:
            stall_count += 1
            if stall_count >= 3:
                strength = 0.01 + 0.003 * min(stall_count, 15)
                a = fourier_perturb(best_a, rng, strength=strength)
                t_blend = max(0.1, min(t_blend * 1.3, 0.7))
                stall_count = 0
                # Also try CD from perturbed point
                cd_time = min(remaining * 0.1, 8.0)
                if cd_time > 2:
                    a_cd, _ = coord_descent(a, n_passes=5,
                                             time_budget=cd_time, verbose=False)
                    a_cd = a_cd / a_cd.sum()
                    obj_cd = compute_obj(a_cd)
                    if obj_cd < best_obj:
                        best_obj = obj_cd
                        best_a = a_cd.copy()
                        a = best_a.copy()
            elif stall_count == 1:
                a = best_a + 0.003 * rng.standard_normal(n) * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()

        t_blend *= 0.98

    if verbose:
        print(f"    LP+CD done: {iter_count} iters, best_obj={best_obj:.10f}")
    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Generate a random non-negative sequence normalized to sum=1."""
    a = rng.random(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def structured_init(n, rng):
    """Generate a structured initialization with some regularity."""
    # Mix of low-frequency structure and randomness
    a = np.ones(n) / n
    n_modes = 3 + rng.integers(0, 4)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = rng.uniform(0.05, 0.3) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(freq * np.arange(n) + phase)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

        if obj_p6 < best_obj:
            best_obj = obj_p6
            best_a = a_p6.copy()
            print(f"  Phase 6: obj={best_obj:.10f} ** IMPROVED")
        else:
            print(f"  Phase 6: obj={obj_p6:.10f}")

    # === Phase 7: Ultra-fast CD final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_fin, _ = coord_descent(best_a, n_passes=50,
                                  time_budget=min(remaining * 0.5, 30.0),
                                  verbose=False)
        a_fin = a_fin / a_fin.sum()
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()
            print(f"  Final CD: obj={best_obj:.10f} ** IMPROVED")
        else:
            print(f"  Final CD: obj={obj_fin:.10f}")

    # Validate
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
