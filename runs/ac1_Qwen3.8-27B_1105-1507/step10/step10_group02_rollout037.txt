Optimizes step function coefficients for minimum peak autoconvolution ratio.
Uses anti-symmetric initializations, fast vectorized coordinate descent with
pair-aware updates, peak-targeted refinement, LP polishing, and multi-n search.
Key innovation: exploits the anti-symmetric structure to reduce the center

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

        centers = [n // 6, n // 3]
        widths = [n // 12, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a += 0.05
        a = a / a.sum()
    elif mode == 10:
        # Geometric decay
        r = rng.uniform(0.95, 0.995)
        a = r ** x
        a = a / a.sum()
    elif mode == 11:
        # Random
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 12:
        # Exponential decay from left
        a = np.exp(-x / (n * rng.uniform(0.5, 2.0)))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def fast_coordinate_descent(a, time_budget=300.0, verbose=True, seed=42, max_sweeps=500):
    """
    Fast coordinate descent with vectorized 1D subproblem evaluation.
    Uses precomputed coefficients for fast evaluation of each coordinate's
    1D objective function.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget and sweep < max_sweeps:
        sweep += 1
        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            # Precompute coefficients for 1D evaluation
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)

            # Linear terms (k != 2i)
            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            idx_lin = ks_lin - i  # index into a for a[k-i]
            valid_lin = (idx_lin >= 0) & (idx_lin < n)
            ks_lin = ks_lin[valid_lin]
            idx_lin = idx_lin[valid_lin]
            coeff_lin = 2.0 * a[idx_lin]  # d_k
            const_lin = conv[ks_lin] - coeff_lin * v  # c_k

            # Quadratic term (k = 2i)
            k_quad = 2 * i
            if 0 <= k_quad <= 2 * n - 2:
                has_quad = True
                c_quad = float(conv[k_quad]) - v * v
            else:
                has_quad = False
                c_quad = 0.0

            # Base values (unchanged convolution entries)
            # The max over ALL k includes unchanged entries
            unchanged_mask = np.ones(2 * n - 1, dtype=bool)
            unchanged_mask[ks] = False
            if has_quad:
                unchanged_mask[k_quad] = False
            max_unchanged = float(np.max(conv[unchanged_mask])) if unchanged_mask.any() else 0.0

            # Evaluate 1D objective at candidate points
            def eval_1d(x):
                # Linear part max
                if len(ks_lin) > 0:
                    lin_vals = const_lin + coeff_lin * x
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                # Quadratic part
                if has_quad:
                    quad_val = c_quad + x * x
                else:
                    quad_val = 0.0
                max_changed = max(max_lin, quad_val)
                max_all = max(max_changed, max_unchanged)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_all / denom

            # Quick scan
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.3, v * 0.5, v * 0.7, v, v * 1.3, v * 1.5, v * 2.0,
                          v * 3.0, S_other * 0.5, S_other]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = eval_1d(v)
            for x in candidates:
                fx = eval_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement
            x_lo = max(0.0, best_x * 0.3)
            x_hi = best_x * 3.0 + scale
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = eval_1d(c_pt)
            fd = eval_1d(d_pt)

            for _ in range(30):
                if x_hi - x_lo < 1e-12 * max(1.0, abs(x_hi)):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = eval_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = eval_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-13:
                continue

            # Update a and conv
            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a[idx_lin]
            S += delta

            # Track best
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def pair_coordinate_descent(a, time_budget=200.0, verbose=False, seed=42):
    """
    Pair-based coordinate descent: optimizes pairs (j, n-1-j) simultaneously.
    This directly controls the center convolution contribution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    # Only process pairs for the first half
    pair_indices = list(range(n // 2))
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        rng.shuffle(pair_indices)
        S = float(np.sum(a))

        for j in pair_indices:
            if time.monotonic() - start >= time_budget:
                break
            if j >= n - 1 - j:
                continue

            jp = n - 1 - j  # partner index
            v1 = a[j]
            v2 = a[jp]
            S_other = S - v1 - v2

            # 2D subproblem: minimize over (x1, x2) >= 0
            # Use coordinate descent within the pair (2D golden section)
            def eval_2d(x1, x2):
                # Compute new convolution values for positions affected by j and jp
                # This is expensive, so we only check the peak
                # Positions affected by j: [j, j+n-1]
                # Positions affected by jp: [jp, jp+n-1]
                # For simplicity, compute the full convolution change
                # (approximate: just check center and a few key positions)
                center_idx = n - 1
                # Center contribution change
                # Original: a[j]*a[n-1-j] + a[jp]*a[n-1-jp] = a[j]*a[jp] + a[jp]*a[j] = 2*a[j]*a[jp]
                # New: x1*x2 + x2*x1 = 2*x1*x2
                # Other positions also change, but we approximate
                old_center = 2.0 * v1 * v2
                new_center = 2.0 * x1 * x2
                # Approximate max by checking center and nearby
                c_center = conv[center_idx] + new_center - old_center
                denom = (S_other + x1 + x2) ** 2
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max(c_center, 0.0) / denom

            # Simple 2D coordinate descent (alternate between x1 and x2)
            x1_opt = v1
            x2_opt = v2
            for _ in range(2):
                # Optimize x1 with x2 fixed
                best_x1 = x1_opt
                best_f = eval_2d(x1_opt, x2_opt)
                scale = max(x1_opt, 1.0 / n, 1e-6)
                for mult in [0.1, 0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
                    x1_try = max(0.0, x1_opt * mult)
                    f_try = eval_2d(x1_try, x2_opt)
                    if f_try < best_f:
                        best_f = f_try
                        best_x1 = x1_try
                # Golden section on x1
                lo = max(0.0, best_x1 * 0.5)
                hi = best_x1 * 2.0 + scale
                c_pt = hi - gr * (hi - lo)
                d_pt = lo + gr * (hi - lo)
                fc = eval_2d(c_pt, x2_opt)
                fd = eval_2d(d_pt, x2_opt)
                for _ in range(20):
                    if hi - lo < 1e-11:
                        break
                    if fc < fd:
                        hi = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = hi - gr * (hi - lo)
                        fc = eval_2d(c_pt, x2_opt)
                    else:
                        lo = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = lo + gr * (hi - lo)
                        fd = eval_2d(d_pt, x2_opt)
                x1_opt = max(0.0, (lo + hi) / 2.0)

                # Optimize x2 with x1 fixed
                best_x2 = x2_opt
                best_f = eval_2d(x1_opt, x2_opt)
                scale = max(x2_opt, 1.0 / n, 1e-6)
                for mult in [0.1, 0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
                    x2_try = max(0.0, x2_opt * mult)
                    f_try = eval_2d(x1_opt, x2_try)
                    if f_try < best_f:
                        best_f = f_try
                        best_x2 = x2_try
                lo = max(0.0, best_x2 * 0.5)
                hi = best_x2 * 2.0 + scale
                c_pt = hi - gr * (hi - lo)
                d_pt = lo + gr * (hi - lo)
                fc = eval_2d(x1_opt, c_pt)
                fd = eval_2d(x1_opt, d_pt)
                for _ in range(20):
                    if hi - lo < 1e-11:
                        break
                    if fc < fd:
                        hi = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = hi - gr * (hi - lo)
                        fc = eval_2d(x1_opt, c_pt)
                    else:
                        lo = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = lo + gr * (hi - lo)
                        fd = eval_2d(x1_opt, d_pt)
                x2_opt = max(0.0, (lo + hi) / 2.0)

            # Apply update if improved
            if abs(x1_opt - v1) > 1e-10 or abs(x2_opt - v2) > 1e-10:
                # Verify with full computation
                a[j] = x1_opt
                a[jp] = x2_opt
                S_new = float(np.sum(a))
                if S_new > 1e-10:
                    conv_check = compute_conv(a)
                    obj = 2.0 * n * float(np.max(conv_check)) / (S_new * S_new)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
                        conv = conv_check
                        S = S_new
                    else:
                        # Revert
                        a[j] = v1
                        a[jp] = v2
                        S = float(np.sum(a))

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    Pair CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def peak_targeted_refinement(a, time_budget=100.0, verbose=False, seed=42):
    """
    Targeted refinement: identify the peak position and directly reduce it
    by adjusting the contributing coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        conv = compute_conv(a)
        peak_val = float(np.max(conv))
        peak_positions = np.where(conv >= peak_val * 0.999)[0]

        if len(peak_positions) == 0:
            break

        k_peak = int(peak_positions[0])

        # Find contributors to this peak
        # conv[k] = sum_j a[j]*a[k-j]
        # Contributors: pairs (j, k-j) where both are in [0, n-1]
        j_min = max(0, k_peak - (n - 1))
        j_max = min(n - 1, k_peak)
        js = np.arange(j_min, j_max + 1)
        partners = k_peak - js

        # Reduce the largest contributors
        products = a[js] * a[partners]
        # Sort by product (largest first)
        order = np.argsort(products)[::-1]
        top = order[:min(10, len(order))]

        # Reduce top contributors and redistribute
        reduction = 0.05 * products[top].sum()
        total_reduction = 0.0
        for idx in top:
            j = js[idx]
            p = partners[idx]
            if j != p:
                # Reduce the larger of the pair
                if a[j] > a[p]:
                    delta = min(0.05 * a[j], reduction / max(products[idx], 1e-10) * 0.1)
                    a[j] = max(0, a[j] - delta)
                    total_reduction += delta
                else:
                    delta = min(0.05 * a[p], reduction / max(products[idx], 1e-10) * 0.1)
                    a[p] = max(0, a[p] - delta)
                    total_reduction += delta
            else:
                delta = 0.05 * a[j]
                a[j] = max(0, a[j] - delta)
                total_reduction += delta

        # Redistribute to non-contributors
        if total_reduction > 1e-10:
            non_contrib_mask = np.ones(n, dtype=bool)
            non_contrib_mask[js] = False
            non_contrib_mask[partners] = False
            if non_contrib_mask.any():
                add_per = total_reduction / float(non_contrib_mask.sum())
                a[non_contrib_mask] += add_per

        # Normalize
        s = float(np.sum(a))
        if s < 1e-10:
            a = best_a.copy()
            continue
        a = a / s

        obj = 2.0 * n * float(np.max(compute_conv(a)))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            a = best_a.copy()
        else:
            # Add random perturbation to escape
            a = best_a * (1.0 + rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and it % 100 == 0:
            print(f"    PTR it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a


def lp_polish(a, time_budget=60.0, verbose=False, seed=42):
    """LP-based polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 4:
                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if time.monotonic() - start > time_budget * 0.7:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: combines anti-symmetric initializations, fast
    coordinate descent, pair-based updates, peak-targeted refinement,
    LP polishing, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_default = 800
        best_seq = anti_symmetric_init(n_default, 1, rng)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    # === Phase 1: Fast Coordinate Descent on parent ===
    print(f"\n=== Phase 1: Fast CD on parent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 500.0)
    if p1_budget > 30:
        a_cd = fast_coordinate_descent(best_seq, time_budget=p1_budget,
                                       verbose=True, seed=seed)
        obj_cd = 2.0 * a_cd.size * float(np.max(compute_conv(a_cd / a_cd.sum())))
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: Pair-based CD ===
    print(f"\n=== Phase 2: Pair-based CD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 200.0)
    if p2_budget > 30:
        a_pair = pair_coordinate_descent(best_seq, time_budget=p2_budget,
                                         verbose=False, seed=seed)
        obj_pair = 2.0 * a_pair.size * float(np.max(compute_conv(a_pair / a_pair.sum())))
        if obj_pair < best_score:
            best_score = obj_pair
            best_seq = a_pair.copy()
            print(f"  Pair CD: obj={best_score:.8f} **")
        else:
            print(f"  Pair CD: obj={obj_pair:.8f}")

    # === Phase 3: Peak-targeted refinement ===
    print(f"\n=== Phase 3: Peak-targeted refinement ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 100.0)
    if p3_budget > 20:
        a_ptr = peak_targeted_refinement(best_seq, time_budget=p3_budget,
                                         verbose=False, seed=seed)
        obj_ptr = 2.0 * a_ptr.size * float(np.max(compute_conv(a_ptr / a_ptr.sum())))
        if obj_ptr < best_score:
            best_score = obj_ptr
            best_seq = a_ptr.copy()
            print(f"  PTR: obj={best_score:.8f} **")
        # Follow up with CD
        if remaining - time.monotonic() > 30:
            a_cd2 = fast_coordinate_descent(a_ptr, time_budget=min(60, remaining * 0.05),
                                            verbose=False, seed=seed + 1)
            obj_cd2 = 2.0 * a_cd2.size * float(np.max(compute_conv(a_cd2 / a_cd2.sum())))
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  CD after PTR: obj={best_score:.8f} **")

    # === Phase 4: Diverse anti-symmetric initializations ===
    print(f"\n=== Phase 4: Diverse initializations ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for mode in [1, 2, 3, 5, 6, 7, 10, 12]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_init = anti_symmetric_init(n_current, mode, rng)
        # Blend with current best to maintain quality
        blend = 0.6  # 60% best, 40% new
        a_try = (1 - blend) * a_init + blend * best_seq
        a_try = np.maximum(a_try, 0)
        a_try = a_try / a_try.sum()

        alloc = min(remaining * 0.2, 80.0)
        if alloc > 25:
            a_try = fast_coordinate_descent(a_try, time_budget=alloc,
                                            verbose=False, seed=seed + mode * 100)
            obj_try = 2.0 * a_try.size * float(np.max(compute_conv(a_try / a_try.sum())))
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Mode {mode}: obj={best_score:.8f} **")
            else:
                print(f"  Mode {mode}: obj={obj_try:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Try a range of n values around the current best
    n_candidates = set()
    for delta in [-200, -100, -50, 50, 100, 200, 300]:
        n_try = n_best + delta
        if 400 <= n_try <= 2000:
            n_candidates.add(n_try)
    # Also try some fixed values
    for n_fixed in [500, 600, 700, 800, 900, 1000, 1200, 1500]:
        if 400 <= n_fixed <= 2000 and abs(n_fixed - n_best) > 100:
            n_candidates.add(n_fixed)

    n_options = sorted(n_candidates)[:8]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        # Resample current best to new length
        a_resampled = resample_sequence(best_seq, n_try)
        # Also try anti-symmetric init at this n
        a_asym = anti_symmetric_init(n_try, 1, rng)
        a_blend = 0.5 * a_resampled + 0.5 * a_asym
        a_blend = np.maximum(a_blend, 0)
        a_blend = a_blend / a_blend.sum()

        alloc = min(remaining * 0.2, 100.0)
        if alloc > 30:
            a_result = fast_coordinate_descent(a_blend, time_budget=alloc,
                                               verbose=False, seed=n_try)
            obj_result = 2.0 * a_result.size * float(
                np.max(compute_conv(a_result / a_result.sum())))
            if obj_result < best_score:
                best_score = obj_result
                best_seq = a_result.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_result:.8f}")

    # === Phase 6: LP Polishing ===
    print(f"\n=== Phase 6: LP Polishing ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(remaining * 0.3, 80.0)
            a_lp = lp_polish(best_seq, time_budget=alloc, verbose=False,
                             seed=1000 + rnd)
            obj_lp = 2.0 * a_lp.size * float(np.max(compute_conv(a_lp / a_lp.sum())))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
            # CD after LP
            remaining = deadline - time.monotonic()
            if remaining > 30 and obj_lp < best_score + 0.01:
                a_cd = fast_coordinate_descent(a_lp, time_budget=min(remaining * 0.2, 60),
                                               verbose=False, seed=2000 + rnd)
                obj_cd = 2.0 * a_cd.size * float(np.max(compute_conv(a_cd / a_cd.sum())))
                if obj_cd < best_score:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    print(f"  CD after LP {rnd}: obj={best_score:.8f} **")

    # === Phase 7: Final refinement ===
    print(f"\n=== Phase 7: Final refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 150.0)
        a_final = fast_coordinate_descent(best_seq, time_budget=alloc,
                                          verbose=False, seed=seed + 9999)
        obj_final = 2.0 * a_final.size * float(np.max(compute_conv(a_final / a_final.sum())))
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # === Phase 8: One more LP if time ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_lp_final = lp_polish(best_seq, time_budget=min(remaining * 0.5, 40.0),
                               verbose=False, seed=7777)
        obj_lp_final = 2.0 * a_lp_final.size * float(
            np.max(compute_conv(a_lp_final / a_lp_final.sum())))
        if obj_lp_final < best_score:
            best_score = obj_lp_final
            best_seq = a_lp_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
