def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm
    return A


def solve_conv_lp(a):
    """
    Solve LP: min M s.t. (a_norm * g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

                s = float(np.sum(g_val))

        if g_lp is None:
            break

        it += 1

        # Line search: try multiple blending factors
        t_candidates = np.linspace(0.01, 1.0, 20)
        best_local_obj = best_obj
        best_local_a = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP alt it={it}, obj={best_obj:.10f}, M_lp={M_lp:.6f}, "
                      f"lp_time={lp_time:.2f}s")
        else:
            stall += 1
            if stall >= 3:
                # Fourier perturbation
                a = best_a.copy()
                n_modes = 5
                for k in range(1, n_modes + 1):
                    freq = k * 2 * np.pi / n
                    amp = 0.01 * rng.uniform(0.5, 1.5) / k
                    phase = rng.uniform(0, 2 * np.pi)
                    a = a * np.exp(amp * np.cos(freq * np.arange(n) + phase))
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                stall = 0

    if verbose:
        print(f"    LP alt done: {it} iterations, best_obj={best_obj:.10f}")
    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def coordinate_descent(a, time_budget, verbose=False):
    """
    Block coordinate descent: optimize small blocks of coordinates jointly.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Use coordinate-wise updates with a small set of candidates
    rng = np.random.default_rng(123)

    while time.monotonic() - start < time_budget:
        improved = False
        idx = rng.permutation(n)

        for i in idx:
            if time.monotonic() - start > time_budget:
                break

            t_old = a[i]
            if t_old < 1e-12:
                continue

            # Generate candidates around current value
            candidates = [
                t_old * 0.7, t_old * 0.85, t_old * 1.0,
                t_old * 1.15, t_old * 1.3
            ]
            # Also try 0
            candidates.append(0.0)

            best_t = t_old
            best_local = best_obj

            for t in candidates:
                if t < 0:
                    continue
                a[i] = t
                s_curr = float(np.sum(a))
                if s_curr < 1e-10:
                    a[i] = t_old
                    continue
                conv = compute_conv(a / s_curr)
                obj_try = 2.0 * n * float(np.max(conv))
                if obj_try < best_local:
                    best_local = obj_try
                    best_t = t

            a[i] = best_t

            if best_local < best_obj - 1e-13:
                best_obj = best_local
                best_a = a.copy()
                improved = True

        if not improved:
            break

    if verbose:
        print(f"    Coord descent: best_obj={best_obj:.10f}")
    return best_a
