"""
Optimized step function coefficient search using:
1. Smoothed-max subgradient descent with Nesterov acceleration (fast warm-up)
2. Active-set LP via scipy linprog for efficient direction finding
3. Backtracking line search for optimal blending
4. Asymmetric perturbation strategies and n-exploration
5. Periodic full-constraint LP for final polish
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_grad(a, beta):
    """
    Compute smoothed gradient of max(conv(a,a)) using log-sum-exp.
    Returns gradient vector and the smoothed max value.
    """
    n = len(a)
    c = compute_conv(a)
    # Shift to avoid overflow
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    p = exp_c / exp_c.sum()
    smoothed_max = c_max + np.log(exp_c.sum()) / beta
    # Gradient: grad_i = 2 * sum_k p_k * a[k-i] for 0<=k-i<n
    # This is a cross-correlation of p and a
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fp = rfft(p, size)
    fa = rfft(a, size)
    # Cross-correlation: sum_k p_k * a[k-i] = ifft(conj(fp) * fa)
    corr = irfft(np.conj(fp) * fa, size)[:n]
    grad = 2.0 * corr
    return grad, smoothed_max


def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smoothed_grad_descent(a, time_budget=60.0, beta_start=20.0, beta_end=5000.0, n_stages=8):
    """
    Multi-stage smoothed gradient descent with Nesterov acceleration.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    betas = np.geomspace(beta_start, beta_end, n_stages)
    it = 0
    momentum = np.zeros(n)
    beta_mom = 0.7

    for beta in betas:
        stage_start = time.monotonic()
        lr = 1.0 / (beta * 2.0 * n)
        # Adaptive learning rate
        step = 0
        while time.monotonic() - start < time_budget and time.monotonic() - stage_start < time_budget / n_stages:
            grad, _ = softmax_grad(a, beta)
            momentum = beta_mom * momentum + lr * grad
            a_new = a - momentum
            a_new = project_simplex(a_new)

            # Check every 100 steps
            if step % 100 == 0:
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()

            a = a_new
            step += 1
            it += 1

            # Decay momentum
            if step > 500:
                beta_mom *= 0.999

    # Final check
    obj_final = compute_obj(a)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a.copy()

    return best_a


def build_sparse_rows(a, top_indices, K):
    """
    Build constraint matrix rows for active-set LP.
    Returns A (K x n) where A[i, j] = a[top_indices[i] - j] if valid.
    """
    n = len(a)
    A = np.zeros((K, n))
    for idx in range(K):
        k = top_indices[idx]
        # Row: for each j, coefficient is a[k-j] if 0 <= k-j < n
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[idx, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def active_set_lp(a, K=50, top_indices=None):
    """
    Solve active-set LP: min M s.t. (a*g)[k_i] <= M for top K peaks,
    sum(g) = 1, g >= 0.
    Returns g (optimal direction) and M value.
    """
    n = len(a)
    c = compute_conv(a)

    if top_indices is None:
        top_indices = np.argsort(c)[::-1][:K]

    K_actual = len(top_indices)
    A = build_sparse_rows(a, top_indices, K_actual)

    # Variables: x = (g_0, ..., g_{n-1}, M)
    # Minimize: [0, ..., 0, 1] @ x
    # Inequality: [A, -I_K] @ x <= 0
    # Equality: [1, ..., 1, 0] @ x = 1
    # Bounds: g_i >= 0, M >= 0

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Inequality constraints: A@g - M <= 0 for each row
    A_ineq = np.hstack([A, -np.eye(K_actual)])
    b_ineq = np.zeros(K_actual)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    result = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

    if result.success and result.x is not None:
        g = np.maximum(result.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            g = g / s
        else:
            g = None
        M_val = float(result.x[-1])
        return g, M_val, top_indices
    return None, None, top_indices


def full_lp(a):
    """Solve the full LP with all 2n-1 constraints (for final polish)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    n_constraints = 2 * n - 1

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    # Build full constraint matrix
    A = np.zeros((n_constraints, n))
    for j in range(n):
        r = min(n, n_constraints - j)
        A[j:j + r, j] = a_norm[:r]

    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.HIGHS, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, direction, time_budget=3.0):
    """
    Find optimal t in [0, 1.5] minimizing obj((1-t)*a + t*direction).
    Uses golden section search.
    """
    a = np.maximum(a, 0)
    direction = np.maximum(direction, 0)
    s_dir = float(np.sum(direction))
    if s_dir < 1e-10:
        return 0.0, compute_obj(a)
    direction = direction / s_dir

    def obj_at(t):
        t_clipped = max(0.0, min(1.5, t))
        a_try = (1.0 - t_clipped) * a + t_clipped * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    lo, hi = 0.0, 1.5

    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    start = time.monotonic()
    for _ in range(30):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)

    # Check additional points
    for t_try in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0, 1.2, 1.4]:
        obj_try = obj_at(t_try)
        if obj_try < obj_best:
            obj_best = obj_try
            t_best = t_try

    return t_best, obj_best

                break

        # Line search
        t_line, obj_line = line_search(a, g_lp, time_budget=min(3.0, remaining * 0.05))

        if obj_line < best_obj - 1e-12:
            a_new = (1.0 - t_line) * a + t_line * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                a = a_new
                best_obj = obj_line
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 10 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_line:.4f}, "
                  f"lp_time={lp_time:.2f}s, elapsed={time.monotonic()-start:.1f}s")

        # Adaptive: increase K as we converge
        if it > 20 and K < 100:
            K = min(100, K + 10)

        # Perturbation when stalled
        if stall >= 4:
            if stall <= 8:
                # Fourier perturbation
                freq = rng.integers(1, max(2, n // 100))
                phase = rng.uniform(0, 2 * np.pi)
                x = np.arange(n) / max(1, n - 1)
                pert = 0.01 * stall * np.cos(2 * np.pi * freq * x + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                else:
                    a = best_a.copy()
            else:
                # Major restart: blend with diverse init
                x = np.linspace(0, 1, n)
                modes = [
                    (2 * x - 1) ** 4 + 0.01,
                    np.exp(-((x - 0.25) ** 2) / 0.01) + np.exp(-((x - 0.75) ** 2) / 0.01) + 0.01,
                    np.exp(3 * (2 * x - 1) ** 2),
                    1.0 / (0.02 + 5 * (2 * x - 1) ** 2),
                    np.abs(np.sin(3 * np.pi * x)) + 0.01,
                ]
                div = modes[it % len(modes)]
                a = 0.3 * div + 0.7 * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()
            stall = 0
            top_indices = None

    return best_a


def interpolate_sequence(a_old, n_new):
    """Interpolate sequence from old length to new length."""
    n_old = len(a_old)
    if n_old == n_new:
        return a_old.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a_old, left=0, right=0)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 0.01:
        a_new = a_new / s
    return a_new


def diverse_init(n, mode=0):
    """Generate diverse initializations for different n values."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.02
    elif mode == 1:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 2:
        a = 1.0 / (0.03 + 4 * (2 * x - 1) ** 2)
    elif mode == 3:
        a = np.exp(-((x - 0.2) ** 2) / 0.005) + np.exp(-((x - 0.8) ** 2) / 0.005) + 0.005
    elif mode == 4:
        a = np.abs(np.sin(2 * np.pi * x)) ** 0.5 + 0.01
    elif mode == 5:
        a = (1 + np.cos(2 * np.pi * x)) ** 2 + 0.01
    else:
        a = np.ones(n) * 0.1 + (2 * x - 1) ** 6
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 0.01:
        a = a / s
    return a

        print(f"Default init (n={n_parent}): obj={best_score:.8f}")

                                     beta_start=10.0, beta_end=2000.0, n_stages=6)

        print(f"  SG done: obj={obj_sg:.8f} (best={best_score:.8f}), time={time.monotonic()-t0:.1f}s")

            best_score = obj_sg
            best_seq = a_sg.copy()
            a_cur = a_sg.copy()
            print(f"  ** Improved to {best_score:.8f}")

    remaining = deadline - time.monotonic()

    # === Phase 2: Active-set LP on parent n ===
    if remaining > 100:
        print(f"\n=== Phase 2: Active-set LP (n={len(a_cur)}) ===")
        t0 = time.monotonic()
        a_lp = lp_optimize(a_cur, time_budget=min(150.0, remaining * 0.2), K=50, verbose=True)
        obj_lp = compute_obj(a_lp)
        print(f"  LP done: obj={obj_lp:.8f}, time={time.monotonic()-t0:.1f}s")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_cur = a_lp.copy()
            print(f"  ** Improved to {best_score:.8f}")

    remaining = deadline - time.monotonic()

    # === Phase 3: n exploration ===
    print(f"\n=== Phase 3: n Exploration ===")
    n_base = len(a_cur)
    n_candidates = []
    for dn in [200, 400, 600, 1000, 1500, 2000, 3000]:
        n_try = n_base + dn
        if n_try <= 5000:
            n_candidates.append(n_try)
    # Also try smaller
    for dn in [200, 400]:
        n_try = n_base - dn
        if n_try >= 400:
            n_candidates.append(n_try)

    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - n_base))[:6]
    print(f"  Trying n values: {n_candidates}")
