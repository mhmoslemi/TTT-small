1. The inverse DFT is non-negative
2. The sum of the inverse DFT is 1 (i.e., A(0) = 1)
3. `|A(f)|^2` is as flat as possible

This is a "flat spectral design" with a non-negativity constraint.

One approach: **iterative projection between the set of flat spectra and the set of spectra whose inverse DFT is non-negative.**

Algorithm:
1. Start with some `A(f)` (e.g., from a window function)
2. Flatten: set `|A(f)| = C` for all f (or adjust to reduce the peak of the inverse DFT of `|A|^2`)
3. Project: compute `a = IDFT(A)`, set `a = max(a, 0)`, renormalize, recompute `A = DFT(a)`
4. Repeat

This is similar to the Gerchberg-Saxton / error reduction algorithm used in optics.

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Multi-strategy optimizer for minimum peak convolution of non-negative sequences.
Uses FISTA with log-sum-exp smoothing, spectral projection initialization,
anti-palindromic starts, and SOCP trust-region polish. Key innovation: diverse
initializations targeting different basins of attraction, combined with adaptive
temperature annealing in the smooth approximation.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv2(a, b):
    """Full linear convolution a*b via FFT."""
    total_len = len(a) + len(b) - 1
    size = 1
    while size < total_len:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:total_len]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho_val = rho_arr[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_smooth(a, p):
    """
    Compute gradient of (2n/tau)*logsumexp(tau*c) w.r.t. a.
    g[i] = 2 * sum_k p[k] * a[k-i] (valid indices only)
    Uses the relation: g = 2 * conv(p, a_flip)[n-1 : n-1+n]
    """
    n = len(a)
    a_flip = a[::-1]
    conv = compute_conv2(p, a_flip)
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g


def fista_optimize(a_init, time_budget=300.0, tau_init=3.0, tau_end=800.0,
                   verbose=False, seed=42, lr_scale=1.0):
    """
    FISTA with adaptive log-sum-exp temperature for minimizing peak convolution.
    Uses heavy-ball momentum and backtracking line search.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    a_prev = a.copy()
    t_mom = 1.0
    L = 2.0 * tau_init * n * np.max(a) * lr_scale + 1e-6
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-6))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        # Heavy ball / Nesterov momentum
        t_old = t_mom
        t_mom = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        theta = (t_old - 1.0) / t_mom
        y = a + theta * (a - a_prev)
        y = np.maximum(y, 0)
        y_s = y.sum()
        if y_s < 1e-15:
            y = a.copy()
        else:
            y = y / y_s

        # Compute smooth objective gradient at y
        c_y = compute_conv(y)
        c_max_y = float(np.max(c_y))
        scaled = tau * (c_y - c_max_y)
        sum_exp = float(np.sum(np.exp(scaled)))
        if sum_exp < 1e-300:
            p = np.full(len(c_y), 1.0 / len(c_y))
        else:
            p = np.exp(scaled) / sum_exp

        g = compute_gradient_smooth(y, p)

        # Backtracking line search
        alpha = 1.0
        obj_y = 2.0 * n * c_max_y
        a_new = None
        for _ in range(15):
            candidate = project_simplex(y - alpha * g / (tau * 2.0 * np.max(a) * n + 1e-10))
            c_cand = compute_conv(candidate)
            obj_cand = 2.0 * n * float(np.max(c_cand))
            # Armijo condition (simplified)
            if obj_cand <= 2.0 * n * c_max_y + 0.1 * alpha * np.dot(g, candidate - y) / (tau * 2.0 * np.max(a) * n + 1e-10) + 1e-10:
                a_new = candidate
                break
            alpha *= 0.5

        if a_new is None:
            a_new = project_simplex(y - alpha * g / (tau * 2.0 * np.max(a) * n + 1e-10))

        a_prev = a.copy()
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive restart
        if no_improve > 150:
            scale = 0.005 + 0.005 * min(no_improve // 150, 3)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = project_simplex(a)
            y = a.copy()
            a_prev = a.copy()
            t_mom = 1.0
            no_improve = 0

        it += 1
        if verbose and it % 100 == 0:
            print(f"    FISTA it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

    return best_a


def spectral_flatten(a_init, n_iters=100, verbose=False):
    """
    Spectral projection: alternate between flattening power spectrum
    and projecting to non-negativity. Different from gradient methods.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    for it in range(n_iters):
        # Compute power spectrum
        A = rfft(a, L)
        P = np.abs(A) ** 2
        phases = np.angle(A)

        # Compute current autocorrelation
        c = irfft(P, L)[:2 * n - 1]
        c_mean = 1.0 / (2 * n - 1)

        # Find peaks and apply spectral notch
        # Use soft thresholding on the spectrum to flatten
        # Equivalent to iterative "water filling" in frequency domain
        c_peak = float(np.max(c))
        if c_peak < c_mean * 1.01:
            break

        # Reduce spectral components that contribute to peak
        peak_k = int(np.argmax(c))
        f = np.arange(L // 2 + 1)
        # Contribution of each frequency to c[peak_k]
        contrib = np.cos(2 * np.pi * f * peak_k / L)

        # Damp frequencies that positively contribute to the peak
        strength = 0.05 + 0.05 * (1.0 - c_mean / max(c_peak, 1e-10))
        damping = 1.0 - strength * np.maximum(contrib, 0)
        P_new = P * damping
        P_new = np.maximum(P_new, 1e-30)

        # Reconstruct with modified amplitude, same phase
        A_new = np.sqrt(P_new) * np.exp(1j * phases)
        a_new = irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new

        if verbose and it % 20 == 0:
            print(f"    SpecFlatten it{it}: obj={best_obj:.8f}")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def socp_polish(a_init, time_budget=100.0, r_init=0.2, verbose=False, seed=42):
    """SOCP trust-region polish (similar to previous but with adjustments)."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    r = r_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        # Try SOCP
        C_old = compute_conv(a)
        A_cross = build_conv_matrix(a)

        delta = cp.Variable(n)
        M = cp.Variable()
        a_new_var = a + delta

        constraints = [
            cp.sum(a_new_var) == 1,
            a_new_var >= 0,
            C_old + 2 * (A_cross @ delta) <= M,
            cp.norm(delta, 2) <= r,
        ]

        prob = cp.Problem(cp.Minimize(M), constraints)
        delta_val = None

        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    delta_val = np.asarray(delta.value).flatten()
                    break
            except Exception:
                continue

        # Fallback to LP
        if delta_val is None:
            g_lp, _ = solve_lp_direction(a)
            if g_lp is None:
                break
            delta_val = g_lp - a
            norm_d = np.linalg.norm(delta_val)
            if norm_d > r:
                delta_val = delta_val * (r / norm_d)

        # Line search

        best_alpha_obj = best_obj

            a_try = np.maximum(a_try, 0)

            if s_try < 1e-10:

            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj - 1e-14:
                best_alpha_obj = obj_try
                best_alpha = t_alpha

        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-14:
            a = a + best_alpha * delta_val
            a = np.maximum(a, 0)
            s_n = a.sum()
            if s_n > 1e-10:

            best_obj = best_alpha_obj

            no_improve = 0
            r = min(r * 1.1, 1.5)

                    a = best_a * np.exp(rng.normal(0, 0.01, n))
                    a = np.maximum(a, 0)
                    a = project_simplex(a)
                    r = r_init
                    no_improve = 0
                    if it > 60:
                        break
                else:
                    r = max(r * 0.5, 0.02)

        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    SOCP it{it}: obj={best_obj:.8f}, r={r:.4f}, t={elapsed:.0f}s")

    return best_a


def generate_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []

    # 1. Uniform
    inits.append(np.ones(n) / n)

    # 2. Linearly increasing (anti-palindromic)
    lin = np.arange(1, n + 1, dtype=np.float64)
    inits.append(lin / lin.sum())

    # 3. Linearly decreasing
    lin_dec = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(lin_dec / lin_dec.sum())

    # 4. Hann window
    hann = np.sin(np.pi * np.arange(n) / n) ** 2
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    # 5. Two-hump (Gaussian bumps at 1/3 and 2/3)
    x = np.arange(n) / n
    hump1 = np.exp(-20 * (x - 0.3) ** 2)
    hump2 = np.exp(-20 * (x - 0.7) ** 2)
    two_hump = hump1 + hump2
    inits.append(two_hump / two_hump.sum())

    # 6. Cubic tilt (stronger anti-palindromic)
    cubic = (np.arange(n, dtype=np.float64) / n) ** 3 + 0.01
    inits.append(cubic / cubic.sum())

    # 7. Sine-modulated (breaks symmetry)
    sine_mod = 1.0 + 0.3 * np.sin(2 * np.pi * x * 3)
    sine_mod = np.maximum(sine_mod, 0.01)
    inits.append(sine_mod / sine_mod.sum())

    # 8. Random (Dirichlet)
    rand_seq = rng.dirichlet(np.ones(n))
    inits.append(rand_seq)

    # 9. Exponential tilt
    exp_tilt = np.exp(0.5 * x)
    inits.append(exp_tilt / exp_tilt.sum())

    # 10. "S-curve" (sigmoid-like)
    sigmoid = 1.0 / (1.0 + np.exp(-8 * (x - 0.5)))
    inits.append(sigmoid / sigmoid.sum())

    return inits


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
