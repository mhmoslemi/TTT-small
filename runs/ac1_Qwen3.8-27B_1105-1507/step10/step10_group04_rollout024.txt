def high_tau_gd(a, time_budget, tau_start=100, tau_end=5000, verbose=False):
    """High-precision smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.7
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp
        
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(y, N)
        g_full = ifft(P * np.conj(A_f)).real
        g = 2.0 * g_full[:n]
        
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        lr = min(1.0 / (g_norm + 1e-10), 2.0 / np.sqrt(n))
        lr = max(lr, 1e-8)
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        beta = min(0.95, beta + 0.0001)
        
        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it
            if it > 40000:
                break
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:

    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast scipy HiGHS LP with adaptive restricted active sets, geometric/arithmetic
mixing, subgradient polishing, high-tau smoothed gradient descent, and multi-n
exploration. Key improvement: 10-50x faster LP solves via direct HiGHS call,
enabling many more minimax iterations within the time budget.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog, minimize as scipy_minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp_restricted(a, K=40):
    """
    Solve LP with restricted active set (top K positions).
    min M s.t. (a_norm * g)_k <= M for k in top-K, sum(g)=1, g>=0.
    Returns optimal g (normalized) or None on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv = compute_conv(a_norm)
    
    # Select top K positions
    if K >= len(conv):
        top_idx = np.arange(len(conv))
    else:
        top_idx = np.argsort(conv)[::-1][:K]
    
    # Build constraint matrix
    A_cst = np.zeros((len(top_idx), n))
    for idx, k in enumerate(top_idx):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_max >= i_min:
            A_cst[idx, i_min:i_max + 1] = a_norm[k - i_max:k - i_min][::-1]
    
    # LP formulation
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A_cst, -np.ones((len(top_idx), 1))])
    b_ub = np.zeros(len(top_idx))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            g = res.x[:n].copy()
            g = np.maximum(g, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
                return g
    except Exception:
        pass
    return None


def lp_minimax(a, time_budget, K_init=30, K_max=200, verbose=False):
    """
    LP-based minimax iteration with adaptive active set size.
    Alternates between solving the LP and updating a via mixing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    K = K_init
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break
        
        g = solve_lp_restricted(a, K)
        if g is None:
            K = min(K * 2, K_max, 2 * n - 1)
            if K >= 2 * n - 1:

            continue
        
        # Try multiple mixing strategies
        candidates = []
        
        # Arithmetic mixing with various alphas
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = (1.0 - alpha) * a + alpha * g
            cand = np.maximum(cand, 0)
            cs = float(np.sum(cand))
            if cs > 1e-10:

        # Evaluate candidates
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        
        for cand in candidates:
            obj_c = compute_obj(cand)
            if obj_c < best_local_obj - 1e-13:
                best_local_obj = obj_c
                best_local_a = cand.copy()
        
        if best_local_obj < best_obj - 1e-13:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a
            stall = 0
            it += 1
            if verbose and it % 50 == 0:
                print(f"    LP-iter {it}: obj={best_obj:.8f}, K={K}, t={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall >= 8:
                # Increase K to include more constraints
                K = min(K * 2, K_max, 2 * n - 1)
                stall = 0
                if K > K_max:
                    break
        
        # Occasionally try full constraint set
        if it > 0 and it % 500 == 0:
            g_full = solve_lp_restricted(a, min(2 * n - 1, 400))
            if g_full is not None:
                for alpha in [0.3, 0.5, 0.7]:
                    cand = (1.0 - alpha) * a + alpha * g_full
                    cand = np.maximum(cand, 0)
                    cs = float(np.sum(cand))
                    if cs > 1e-10:
                        cand = cand / cs
                        obj_c = compute_obj(cand)
                        if obj_c < best_obj - 1e-13:
                            best_obj = obj_c
                            best_a = cand.copy()
                            a = best_a
    
    return best_a


def subgradient_polish(a, time_budget, verbose=False):
    """
    Subgradient descent on the exact max with projection onto simplex.
    Uses normalized subgradients and Polyak-type step sizes.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < time_budget:
        if time.monotonic() - start < time_budget:
            conv = compute_conv(a)
            max_conv = float(np.max(conv))
            obj = 2.0 * n * max_conv
            
            #Find near-max positions (within 5% of max)
            threshold = max_conv * 0.95
            near_max = np.where(conv >= threshold)[0]
            
            #Compute subgradient: average over near-max positions
            subgrad = np.zeros(n)
            for k in near_max:
                i_min = max(0, k - n + 1)
                i_max = min(n - 1, k)
                subgrad[i_min:i_max + 1] += 2.0 * a[k - i_max:k - i_min][::-1]
            
            # Normalize the subgradient
            subgrad /= max(1.0, len(near_max))
            
            # Projected subgradient step
            proj = a - subgrad
            # Project onto simplex {x >= 0, sum(x) = 1}
            proj = _project_simplex(proj)
            
            # Check improvement
            obj_new = compute_obj(proj)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = proj.copy()
                a = proj
            else:
                # Try smaller step
                for frac in [0.5, 0.3, 0.1]:
                    proj2 = a - frac * subgrad
                    proj2 = _project_simplex(proj2)
                    obj2 = compute_obj(proj2)
                    if obj2 < best_obj - 1e-12:
                        best_obj = obj2
                        best_a = proj2.copy()
                        a = proj2
                        break
            
            it += 1
            if verbose and it % 1000 == 0:
                print(f"    Subgrad it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
        
        if time.monotonic() - start >= time_budget:
            break
    
    return best_a


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()


def high_tau_gd(a, time_budget, tau_start=200, tau_end=8000, verbose=False):
    """High-precision smoothed gradient descent with Nesterov acceleration."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.85
    v = np.zeros(n)
    it = 0
    best_it = 0
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.6
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Smoothed gradient via softmax
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        scaled = tau * (conv - max_conv)
        scaled = np.clip(scaled, -50, 0)  # prevent overflow
        exp_vals = np.exp(scaled)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp
        
        # FFT-based gradient
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(y, N)
        g_full = ifft(P * np.conj(A_f)).real
        g = 2.0 * g_full[:n]
        
        # Adaptive step size
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        lr = min(1.0 / (g_norm * 1.5 + 1e-10), 3.0 / np.sqrt(n))
        lr = max(lr, 1e-9)
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        it += 1
        beta = min(0.95, beta + 0.0002)
        
        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            beta = 0.85
            best_it = it
            if it > 30000:
                break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 1.5 * (1 - x)
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
        a = a / a.sum()
    elif mode == 5:
        a = np.exp(-np.abs(np.arange(n) - n * 0.3) / (n * 0.2))
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = 1 + 0.3 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses:
    1. Fast LP (HiGHS) with adaptive restricted active set
    2. Geometric/arithmetic mixing
    3. Subgradient polishing
    4. High-tau smoothed gradient descent
    5. Multi-n exploration
    6. Diverse restarts
    7. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")
    
    n_best = len(best_seq)
    
    # === Phase 1: LP Minimax (main driver, ~500s) ===
    print(f"\n=== Phase 1: LP Minimax (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.34, 500.0)
    if p1_budget > 30:
        a_lp = lp_minimax(best_seq, time_budget=p1_budget, K_init=25, K_max=200, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.8f} (no improvement)")
    
    # === Phase 2: Subgradient polish (~100s) ===
    print(f"\n=== Phase 2: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.08, 100.0)
    if p2_budget > 10:
        a_sg = subgradient_polish(best_seq, time_budget=p2_budget, verbose=False)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase2 Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 Subgrad: obj={obj_sg:.8f}")
    
    # === Phase 3: High-tau GD (~180s) ===
    print(f"\n=== Phase 3: High-Tau GD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.13, 180.0)
    if p3_budget > 15:
        a_gd = high_tau_gd(best_seq, time_budget=p3_budget,
                           tau_start=200, tau_end=6000, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase3 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 GD: obj={obj_gd:.8f}")
    
    # === Phase 4: LP Round 2 (~200s) ===
    print(f"\n=== Phase 4: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 200.0)
    if p4_budget > 20:
        a_lp2 = lp_minimax(best_seq, time_budget=p4_budget, K_init=20, K_max=150, verbose=False)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            n_best = len(best_seq)
            print(f"  Phase4 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 LP: obj={obj_lp2:.8f}")
    
    # === Phase 5: Multi-n Exploration (~250s) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.2, 250.0)
    
    n_options = sorted(set([
        max(200, n_best - 200), n_best - 100,
        n_best + 100, n_best + 200, n_best + 500,
        n_best + 1000, 600, 1000, 1500, 2000, 3000

    n_options = [n for n in n_options if 300 <= n <= 5000 and n != n_best]

                break
            alloc = min(50, remaining - 15)
            if alloc < 20:
                break
            
            mode = r % 6
            a_r = diverse_init(n_best, rng, mode=mode)
            # Blend with best
            blend = 0.4 + 0.15 * r
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r

            a_r = lp_minimax(a_r, time_budget=alloc * 0.55, K_init=25, K_max=100, verbose=False)

            if obj_r < best_score:

                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")
    
    # === Phase 7: Final Intensive Polish (remaining) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s")
    
    # LP final
    if remaining > 30:
        alloc = min(remaining * 0.25, 100.0)
        a_f = lp_minimax(best_seq, time_budget=alloc, K_init=15, K_max=100, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

        if obj_f < best_score:

            best_seq = a_f.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()

        if obj_f < best_score:
