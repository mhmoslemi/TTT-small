"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_conv_and_info(a):
    """Returns conv, max_conv, sum_a."""
    n = len(a)
    s = float(np.sum(a))
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_conv = float(np.max(conv))
    return conv, max_conv, s


def smoothed_grad(a, tau):
    """Gradient of smoothed max of conv(a/a.sum(), a/a.sum())."""
    n = len(a)
    s = float(np.sum(a))
    a_norm = a / s
    conv = compute_conv(a_norm)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    # Gradient of max(conv) w.r.t. a_norm:
    # d(conv[k])/d(a_norm[j]) = a_norm[k-j] + a_norm[j-k] (with 0 for out of range)
    # = 2 * cross-correlation
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1

    cross = ifft(P * np.conj(A)).real
    g_norm = 2.0 * cross[:n]
    # Chain rule: d(a_norm[j])/d(a[j]) = 1/s - a[j]/s^2
    # But since we work with a_norm directly, the gradient wrt a is:
    # g_a[j] = (g_norm[j] * s - exact_max * 2 * n * s / s^2 * ...)

    # or use the chain rule properly.

def mirror_descent(a, time_budget, tau_levels=None, verbose=False, 
                   eta_scale=1.0, lp_interval=200):
    """

    Update: a_i <- a_i * exp(-eta * g_i), then renormalize.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    s = float(np.sum(a))
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if tau_levels is None:
        tau_levels = [5, 15, 40, 100, 250, 600, 1500, 4000]
    
    it = 0
    for tau in tau_levels:
        eta = eta_scale / np.sqrt(n)
        it_level = 0
        max_per_level = 3000
        
        while time.monotonic() - start < time_budget:
            if it_level >= max_per_level:
                break
            
            g, _ = smoothed_grad(a, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break
            
            # Multiplicative update
            a_new = a * np.exp(-eta * g)
            a_new = np.maximum(a_new, 1e-15)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            
            a = a_new
            it += 1
            it_level += 1
            
            # Decay eta
            eta *= 0.9999
            eta = max(eta, 1e-8)
    
    return best_a

                prob.solve(solver=solver, verbose=False)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    sg = float(np.sum(g_val))
                    if sg > 1e-10:
                        g_val = g_val / sg
                        solved = True
                        break
            except Exception:
                continue
        
        if not solved:
            break
        
        # Try multiple blend factors
        t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        best_t_obj = best_obj
        best_t_a = best_a.copy()
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))

                best_t_a = a_try.copy()

            a = best_a
            stall = 0
            if verbose:
                print(f"    LP it{it}: obj={best_obj:.10f}")
        else:
            stall += 1
            if stall > 5:

    return best_a


def coordinate_descent(a, time_budget, verbose=False, max_sweeps=50):
    """
    Coordinate descent: for each coordinate, optimize its value
    given all others fixed.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start >= time_budget:
            break
        
        improved = False
        for j in range(n):
            if time.monotonic() - start >= time_budget:
                    break
            
            a_j = a[j]
            # Try a range of values around current
            # The optimal a_j is in [0, 2*a_j] typically
            candidates = np.unique(np.concatenate([
                [0.0, a_j * 0.5, a_j * 0.75, a_j, a_j * 1.25, a_j * 1.5, a_j * 2.0],
                np.linspace(0, max(a_j * 3, 2.0 / n), 8)
            ]))
            
            best_local = a_j
            best_local_obj = best_obj
            
            for x in candidates:
                a[j] = x
                obj_try = compute_obj(a)
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_local = x
            
            a[j] = best_local
            
            if best_local != a_j:
                improved = True
                if best_local_obj < best_obj:
                    best_obj = best_local_obj
                    best_a = a.copy()
        
        if not improved:
            break
    
    return best_a


def aggressive_gd(a, time_budget,tau_levels=None, verbose=False):
    """Nesterov accelerated gradient descent on the probability simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if tau_levels is None:
        tau_levels = [5, 15, 40, 100, 250, 600, 1500, 4000, 8000]
    
    beta = 0.9
    for tau in tau_levels:
        lr = 3.0 / np.sqrt(n)
        v = np.zeros(n)
        it_level = 0
        max_per_level = 5000
        local_best = best_obj
        
        while time.monotonic() - start < time_budget:
            if it_level >= max_per_level:
                break
            
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy
            
            g, _ = smoothed_grad(y, tau)
            
            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
            else:
                a_new = a_new / s_new
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            
            a = a_new
            it_level += 1
            lr = max(1e-7, lr * 0.99995)
        
        if verbose:
            print(f"    tau={tau:5d}: obj={best_obj:.10f}, iters={it_level}")
    
    return best_a


def make_init(n, mode, rng=None):
    """Create diverse initial sequences."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.arange(n)
    t = x / max(n - 1, 1)
    
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = np.sin(np.pi * t) ** 2
    elif mode == 2:
        a = np.exp(-4 * (t - 0.5) ** 2)
    elif mode == 3:
        a = np.exp(-((x - n//4)**2) / (2*(n//10)**2)) + np.exp(-((x - 3*n//4)**2) / (2*(n//10)**2))
    elif mode == 4:
        a = rng.uniform(0.1, 1.0, n)
    elif mode == 5:
        # Watermelon / raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    elif mode == 6:
        # Multi-frequency
        a = np.ones(n)
        for f in [2, 3, 5, 7]:
            a += 0.1 * np.sin(2 * np.pi * f * t + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
    elif mode == 7:
        # Bell shape
        a = np.exp(-((t - 0.5)**2) / (2 * 0.15**2))
    elif mode == 8:
        # Two peaks
        a = np.exp(-((x - n*0.3)**2) / (2*(n//8)**2)) + np.exp(-((x - n*0.7)**2) / (2*(n//8)**2))
    elif mode == 9:
        # Flat center
        center = n // 4
        a = np.ones(n)
        a[:center] = 0.3
        a[n-center:] = 0.3
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 1e-10)
    return a / a.sum()

"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def get_grad_and_info(a, tau):
    """Returns (gradient_full, smoothed_max, exact_max, p_weights)."""
    n = len(a)
    s = float(np.sum(a))
    b = a / s
    conv = compute_conv(b)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    B = fft(b, N)
    cross = ifft(P * np.conj(B)).real
    g_b = 2.0 * cross[:n]
    dot_gb = float(np.dot(g_b, b))
    g_a = (g_b - b * dot_gb) / s
    return g_a, exact_max, p


def power_sum_grad(a, p_power):
    """Gradient of (sum conv^p)^(1/p) for large p."""
    n = len(a)
    s = float(np.sum(a))
    b = a / s
    conv = compute_conv(b)
    # Power sum smoothing
    cs = conv ** p_power
    sum_cs = float(np.sum(cs))
    if sum_cs < 1e-300:
        p_w = np.zeros(len(conv))
        p_w[np.argmax(conv)] = 1.0
    else:
        p_w = cs / sum_cs
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p_w, N)
    B = fft(b, N)
    cross = ifft(P * np.conj(B)).real
    g_b = 2.0 * cross[:n]
    dot_gb = float(np.dot(g_b, b))
    g_a = (g_b - b * dot_gb) / s
    return g_a, float(np.max(conv))


def nesterov_gd(a, time_budget, tau_levels=None, verbose=False, 
                 lr_scale=3.0, beta=0.9, use_power_sum=False, p_power=50):
    """Nesterov accelerated GD with staircase tau."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if tau_levels is None:
        tau_levels = [5, 15, 40, 100, 250, 600, 1500, 4000, 8000]
    
    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        it_level = 0
        max_per_level = 5000
        
        while time.monotonic() - start < time_budget:
            if it_level >= max_per_level:
                break
            
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy
            
            if use_power_sum:
                g, _ = power_sum_grad(y, p_power)
            else:
                g, _, _ = get_grad_and_info(y, tau)
            
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break
            
            v_new = beta * v - lr * g
            # Momentum clip
            v_norm = float(np.linalg.norm(v_new))
            max_v = 2.0 / np.sqrt(n)
            if v_norm > max_v:
                v_new = v_new * max_v / v_norm
            v = v_new
            
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            
            a = a_new
            it_level += 1
            lr = max(1e-8, lr * 0.99995)
        
        if verbose:
            print(f"    tau={tau:5d}: obj={best_obj:.10f}, iters={it_level}")
    
    return best_a


def lp_solve(a):
    """Solve the LP: min M s.t. conv(a,g)<=M, g>=0, sum(g)=1.
    Returns (g_opt, M_opt, dual_vars) or (None, None, None)."""
    n = len(a)
    if n > 3000:
        return None, None, None
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    g_val = g_val / sg
                    # Get dual variables
                    dual_M = M_var.value
                    # Dual of constraints A@g <= M
                    # The constraint list is [A@g <= M, sum(g) == 1]
                    duals = None
                    try:
                        con1_dual = constraints[0].dual_value
                    except:
                        con1_dual = None
                    return g_val, float(dual_M), con1_dual
        except Exception:
            continue
    
    return None, None, None
