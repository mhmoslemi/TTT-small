def lp_polish(a, time_budget, verbose=False):
    """LP-based polishing: iteratively solve LP and mix with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = normalized_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Line search
        t_candidates = np.linspace(0.05, 1.0, 15)
        best_t_obj = best_obj
        best_t = last_t
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = normalized_obj(a_try)
            if obj_try < best_t_obj - 1e-13:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t
            stall = 0
        else:
            # Try smaller steps
            improved = False
            for t in [last_t * 0.5, last_t * 0.3, 0.1, 0.05, 0.02]:
                a_try = (1.0 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = normalized_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    last_t = t
                    stall = 0
                    improved = True
                    break
            if not improved:
                stall += 1
                if stall > 4:
                    break

    return best_a


def perturbation_restart(a, rng, time_budget, verbose=False):
    """Add random perturbation and re-optimize."""
    n = len(a)

    noise = rng.lognormal(0, 0.3, n)

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.5
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
        a = a / a.sum()
    elif mode == 5:
        a = np.exp(-np.arange(n) / (n * 0.2)) + 0.3 * np.exp(-np.arange(n) / (n * 0.8))
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x ** 3 * (1 - x)
        a = a / a.sum()
    elif mode == 7:
        # Truncated Gaussian
        x = np.linspace(-2, 2, n)
        a = np.exp(-x**2)
        a = a / a.sum()
    elif mode == 8:
        # Two peaks
        x = np.linspace(0, 1, n)
        a = np.exp(-((x-0.3)/0.1)**2) + np.exp(-((x-0.7)/0.1)**2)
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a


def support_shrink(a, time_budget, verbose=False):

We have `conv(a,a)[k] = sum_{i} a_i * a_{k-i}` where the sum is over valid indices (0 <= i < n, 0 <= k-i < n).

The derivative w.r.t. `a_j`:
- `a_j` appears in the first factor when `i = j`: contributes `a_{k-j}` if `0 <= k-j < n`
- `a_j` appears in the second factor when `k-i = j`, i.e., `i = k-j`: contributes `a_{k-j}` if `0 <= k-j < n`
- When `j = k-j` (i.e., `k = 2j`), these are the same term `a_j^2`, and the derivative is `2 * a_j = 2 * a_{k-j}` (since `k-j = j`)

So: `d/da_j conv(a,a)[k] = 2 * a_{k-j}` if `0 <= k-j < n`, else 0.

Therefore: `grad_j = d/da_j [sum_k p_k * conv(a,a)[k]] = 2 * sum_k p_k * a_{k-j} * [0 <= k-j < n]`

Let `m = k - j`, so `k = m + j`:
`grad_j = 2 * sum_{m=0}^{n-1} p_{m+j} * a_m * [m+j <= 2n-2]`

Since `m < n` and `j < n`, we have `m+j < 2n`, so `m+j <= 2n-2` is equivalent to `m+j < 2n-1`, which is always true for `m < n, j < n` (since `m+j <= 2n-2`).

So: `grad_j = 2 * sum_{m=0}^{n-1} p_{m+j} * a_m`

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Multi-n exploration with LP polishing, projected accelerated gradient
with proper simplex projection and backtracking, support reduction, and
perturbation restarts. Key improvements: correct FFT gradient via
ifft(FFT(p)*conj(FFT(a))), proper simplex projection, adaptive tau with
backtracking line search, and more aggressive large-n exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def normalized_obj(a):
    """Objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    return normalized_obj(a_norm)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g_var = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g_var <= M, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val
        except Exception:
            continue
    return None


def smoothed_grad_fft(a, tau):
    """Compute smoothed objective value and gradient using softmax smoothing.
    Returns (smoothed_max, grad) where grad is d/d a_j of sum_k p_k * conv(a,a)[k]."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    smooth_val = max_conv + np.log(sum_exp) / tau

    # Gradient: grad[j] = 2 * sum_m p[m+j] * a[m]
    # Computed as: 2 * ifft(FFT(p, N) * conj(FFT(a, N)))[j]
    N = 1
    while N < len(p) + n:
        N <<= 1
    P = fft(p, N)
    A_f = fft(a, N)
    grad_raw = 2.0 * ifft(P * np.conj(A_f)).real[:n]

    return smooth_val, grad_raw


def accelerated_gradient(a, time_budget, tau_start=50, tau_end=5000, verbose=False, max_it=None):
    """Nesterov accelerated projected gradient with adaptive tau and backtracking."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = normalized_obj(a)
    start = time.monotonic()
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0
    stall_count = 0
    lr = 1.0 / np.sqrt(n)

    while time.monotonic() - start < time_budget and (max_it is None or it < max_it):
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1))

        tau = tau_start + (tau_end - tau_start) * progress ** 0.6

        # Nesterov lookahead
        y = a + beta * v
        y = project_simplex(y)

        # Compute gradient
        _, grad = smoothed_grad_fft(y, tau)

        # Project gradient to tangent space of simplex (remove mean)
        grad_tan = grad - np.mean(grad)

        g_norm = float(np.linalg.norm(grad_tan))
        if g_norm < 1e-15:
            break

        # Backtracking line search
        a_try = project_simplex(y - lr * grad_tan)
        obj_try = normalized_obj(a_try)
        backtracks = 0
        while obj_try > normalized_obj(y) + 1e-12 and backtracks < 10 and lr > 1e-12:
            lr *= 0.5
            a_try = project_simplex(y - lr * grad_tan)
            obj_try = normalized_obj(a_try)
            backtracks += 1

        if lr < 1e-12:
            break

        v = beta * v - lr * grad_tan
        a_new = project_simplex(y + v)

        obj_new = normalized_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1

        a = a_new
        it += 1
        beta = min(0.98, (it + 1 + beta) / (it + 2))  # Nesterov beta schedule

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            beta = 0.9
            best_it = it
            lr = 1.0 / np.sqrt(n)

        if stall_count > 30000 and it > 5000:
            break

        if verbose and it % 5000 == 0:
            print(f"    AG it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def lp_polish(a, time_budget, verbose=False):
    """LP-based polishing with adaptive line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = normalized_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.3

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp = solve_lp(a)
        if g_lp is None:
            break

        # Adaptive line search around last_t
        t_base = max(0.02, min(1.0, last_t))
        t_candidates = np.unique(np.concatenate([
            np.linspace(0.02, 1.0, 12),
            [t_base * 0.5, t_base * 0.7, t_base, t_base * 1.3, t_base * 1.7]
        ]))
        t_candidates = np.clip(t_candidates, 0.01, 1.0)

        best_t_obj = best_obj
        best_t = last_t
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = project_simplex(a_try)
            obj_try = normalized_obj(a_try)
            if obj_try < best_t_obj - 1e-13:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t
            stall = 0
        else:
            # Try very small steps
            improved = False
            for t in [0.01, 0.02, 0.03, last_t * 0.3]:
                if t <= 0:
                    continue
                a_try = (1.0 - t) * a + t * g_lp
                a_try = project_simplex(a_try)
                obj_try = normalized_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    last_t = t
                    stall = 0
                    improved = True
                    break
            if not improved:
                stall += 1
                if stall > 5:
                    break

    return best_a


def support_shrink(a, time_budget, verbose=False):
    """Try reducing support and re-optimizing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = normalized_obj(a)
    start = time.monotonic()

    sorted_vals = np.sort(a)[::-1]
    for frac in [0.4, 0.6, 0.8, 0.9]:
        if time.monotonic() - start > time_budget:
            break
        k = max(3, int(n * frac))
        if k >= n - 1:
            continue
        threshold = sorted_vals[k - 1]
        support = np.where(a >= threshold * 0.5)[0]
        if len(support) < 3 or len(support) > n - 5:
            continue
        a_shrunk = np.zeros(n)
        a_shrunk[support] = a[support]
        a_shrunk = a_shrunk / a_shrunk.sum()

        alloc = min(40, time_budget - (time.monotonic() - start))
        if alloc > 15:
            half = alloc * 0.5
            a_shrunk = lp_polish(a_shrunk, time_budget=half, verbose=False) if n <= 1500 else a_shrunk
            a_shrunk = accelerated_gradient(a_shrunk, time_budget=half,
                                            tau_start=100, tau_end=3000, verbose=False)

        obj_shrunk = normalized_obj(a_shrunk)
        if obj_shrunk < best_obj - 1e-12:
            best_obj = obj_shrunk
            best_a = a_shrunk.copy()
            if verbose:
                print(f"    Shrink frac={frac}: obj={best_obj:.8f}, support={len(support)}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_simplex(a_new)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.5
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
    elif mode == 5:
        a = np.exp(-np.arange(n) / (n * 0.2)) + 0.3 * np.exp(-np.arange(n) / (n * 0.8))
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x ** 3 * (1 - x)
    elif mode == 7:
        x = np.linspace(-2, 2, n)
        a = np.exp(-x ** 2)
    elif mode == 8:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) / 0.1) ** 2) + np.exp(-((x - 0.7) / 0.1) ** 2)
    elif mode == 9:
        # Symmetric bump
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 4)
    else:
        a = rng.uniform(0, 1, n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def fast_optimize(a, time_budget, n, verbose=False):
    """Fast optimization pipeline for a given sequence."""
    start = time.monotonic()
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Phase A: Accelerated gradient
    rem = time_budget - (time.monotonic() - start)
    if rem > 10:
        ag_budget = rem * 0.5
        a = accelerated_gradient(a, time_budget=ag_budget,
                                 tau_start=50, tau_end=5000, verbose=verbose)

    # Phase B: LP polishing (only for moderate n)
    rem = time_budget - (time.monotonic() - start)
    if rem > 10 and n <= 1500:
        lp_budget = min(rem * 0.4, 100.0)
        if lp_budget > 8:
            a = lp_polish(a, time_budget=lp_budget, verbose=verbose)

    # Phase C: Final gradient polish
    rem = time_budget - (time.monotonic() - start)
    if rem > 5:
        a = accelerated_gradient(a, time_budget=rem * 0.8,
                                 tau_start=500, tau_end=10000, verbose=verbose)

    return a

            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s

                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
