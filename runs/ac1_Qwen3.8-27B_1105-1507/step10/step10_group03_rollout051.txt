    # = a[k-i] + a[k-i] = 2*a[k-i] (when 0 <= k-i < n)
    # Wait, that's only if both j=i and k-j=i contribute, which happens when i != k-i.
    # If i = k-i (i.e., k = 2i), then the term is a[i]*a[i] and the derivative is 2*a[i].
    # Actually no: if j = i, the term is a[i]*a[k-i]. d/d a[i] = a[k-i].
    # If k-j = i, i.e., j = k-i, the term is a[k-i]*a[i]. d/d a[i] = a[k-i].
    # These are the same term in the sum (when j = i and j = k-i are the same, i.e., k = 2i).
    # In that case, the term is a[i]^2, and d/d a[i] = 2*a[i].
    # If k != 2i, these are two different terms, and the total derivative is 2*a[k-i].
    
    # So in all cases: d((a*a)[k])/da[i] = 2*a[k-i] (if 0 <= k-i < n, else 0)
    
    # Therefore, the subgradient of max_k (a*a)[k] at a is:
    # sg[i] = 2 * sum_k lambda_k * a[k-i]  (for active k)
    # where lambda_k >= 0, sum lambda_k = 1
    
    # Using the dual variables y (normalized to sum to 1):
    # sg = 2 * sum_k y_k * (a shifted by k)
    # sg[i] = 2 * sum_k y_k * a[k-i]  (valid range)
    
    # This is 2 * (a * y_reversed)[i] where * denotes appropriate convolution
    # Actually, sg[i] = 2 * sum_k y[k] * a[k-i]
    # Let's define b[k] = y[k] for k in 0..2n-2
    # Then sg[i] = 2 * sum_k b[k] * a[k-i]
    # This is the cross-correlation of a and b, which can be computed via FFT.
    
    # Using the convolution theorem:
    # sg = 2 * ifft(fft(a) * conj(fft(b)))  (correlation)
    # Or: sg = 2 * conv(a, b_reversed)  (but need to be careful with indexing)
    
    # Let me just compute it directly for clarity (n is at most a few thousand)
    sg = np.zeros(n)
    for k in range(2 * n - 1):
        if abs(y_dual[k]) > 1e-15:
            # Add y[k] * (shift of a by k)
            # (P_k * a)[i] = a[k-i] if 0 <= k-i < n
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            a_shifted = np.zeros(n)
            for i in range(lo, hi + 1):
                a_shifted[i] = a[k - i]
            sg += y_dual[k] * a_shifted
    
    sg *= 2
    
    # Projected gradient step
    a_new = a - step_size * sg
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a
    return a_new / s


def lp_fixed_point_iteration(a, max_iters, time_budget, verbose=False, alpha_start=1.0, alpha_decay=0.95):
    """
    LP fixed-point iteration:
    1. Solve LP to get best response m
    2. Blend a with m
    3. Repeat
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = alpha_start
    
    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Solve LP
        m, t_val, y_dual = solve_lp(a)
        if m is None:
            break
        
        # Try multiple blending factors
        t_candidates = [
            alpha, alpha * 0.5, alpha * 0.25,
            0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0
        ]
        t_candidates = sorted(set([t for t in t_candidates if 0 < t <= 1]))
        
        best_t_obj = best_obj
        best_t_val = min(t_candidates)
        best_a_local = best_a.copy()
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * m
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-14:
            improvement = best_obj - best_t_obj
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            # Adapt alpha
            if best_t_val < alpha * 0.8:
                alpha = max(0.05, best_t_val)
            elif best_t_val > alpha * 1.2:
                alpha = min(1.0, best_t_val * 0.8)
            if verbose and (it < 20 or it % 20 == 0):
                print(f"    LP iter {it}: obj={best_obj:.12f}, t={best_t_val:.3f}, improvement={improvement:.2e}")
        else:
            # No improvement, try subgradient step
            if y_dual is not None and float(np.sum(y_dual)) > 1e-10:
                y_norm = y_dual / float(np.sum(y_dual))
                for step in [0.01, 0.005, 0.001]:
                    a_sg = projected_subgradient_step(a, y_norm, step)
                    obj_sg = compute_obj(a_sg)
                    if obj_sg < best_obj:
                        best_obj = obj_sg
                        best_a = a_sg.copy()
                        a = a_sg
                        if verbose:
                            print(f"    LP iter {it}: subgrad step improved to {best_obj:.12f}")
                        break
            # No improvement at all, break
            if verbose:
                print(f"    LP iter {it}: no improvement, stopping")
            break
    
    return best_a


def smoothed_gd(a, time_budget, tau_levels=None, verbose=False):
    """Smoothed gradient descent with staircase tau."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if tau_levels is None:
        tau_levels = [10, 50, 200, 800, 2000, 5000]
    
    beta = 0.9
    for tau in tau_levels:
        lr = 2.0 / np.sqrt(n)
        v = np.zeros(n)
        it_level = 0
        max_iters = 3000
        
        while time.monotonic() - start < time_budget:
            if it_level >= max_iters:
                break
            
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy
            
            # Compute smoothed gradient
            conv = compute_conv(y)
            exact_max = float(np.max(conv))
            shifted = tau * (conv - exact_max)
            # Clip for numerical stability
            shifted = np.clip(shifted, -500, 500)
            exp_vals = np.exp(shifted)
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp
            
            # Gradient via FFT
            from scipy.fft import fft, ifft
            N = 1
            while N < 2 * n:
                N <<= 1
            P = fft(p, N)
            A = fft(y, N)
            cross = ifft(P * np.conj(A)).real
            g = 2.0 * cross[:n]
            
            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
            else:
                a_new = a_new / s_new
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
            it_level += 1
            lr = max(1e-8, lr * 0.99999)
    
    return best_a
