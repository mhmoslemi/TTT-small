"""
Peak-shaving + Top-K LP optimization for step function coefficients.
Uses mirror descent for fast initial improvement, top-K LP for efficient polishing,
parabolic line search, and systematic length exploration.
"""
import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a, top_k=None):
    """Build matrix A (m x n) such that A @ g gives the top_k peaks of conv(a, g)."""
    n = len(a)
    if top_k is None:
        # Use all lags
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            A[j:j + r, j] = a[:r]
        return A, None
    else:
        # Compute conv(a, ones) to find which lags are most important
        # Actually, we should use the current a to determine which lags matter
        ones = np.ones(n)
        conv_a_ones = compute_conv(a)  # This is conv(a, a)
        # But for the LP, we want to know which lags of conv(a, g) will be largest
        # A good proxy: the lags where conv(a, a) is largest
        idx = np.argsort(conv_a_ones)[::-1][:top_k]
        idx = np.sort(idx)
        A = np.zeros((len(idx), n))
        for row, k in enumerate(idx):
            for j in range(n):
                i = k - j
                if 0 <= i < n:
                    A[row, j] = a[i]
        return A, None

def solve_lp_topk(a, top_k=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for top_k lags, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    if top_k is None or top_k >= 2 * n - 1:
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            A[j:j + r, j] = a_norm[:r]
    else:
        conv_aa = compute_conv(a_norm)
        idx = np.argsort(conv_aa)[::-1][:top_k]
        idx = np.sort(idx)
        A = np.zeros((len(idx), n))
        for row, k in enumerate(idx):
            for j in range(n):
                i = k - j
                if 0 <= i < n:
                    A[row, j] = a_norm[i]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def parabolic_line_search(a, g_dir, t_init=0.5, time_budget=2.0):
    """Parabolic + golden section line search for blend factor."""
    start = time.monotonic()
    
    def obj_at(t):
        if t < 0: t = 0
        if t > 2: t = 2
        a_try = (1 - t) * a + t * g_dir
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)
    
    # Try a few points to find a good region
    t_candidates = [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    best_t = t_init
    best_f = obj_at(t_init)
    
    for t in t_candidates:
        if time.monotonic() - start > time_budget * 0.3:
            break
        f = obj_at(t)
        if f < best_f:
            best_f = f
            best_t = t
    
    # Golden section around the best point
    lo = max(0, best_t - 0.3)
    hi = min(2.0, best_t + 0.3)
    
    if hi - lo > 0.01:
        phi = (1 + np.sqrt(5)) / 2
        gr = 2 - phi
        c = lo + gr * (hi - lo)
        d = lo + (1 - gr) * (hi - lo)
        fc = obj_at(c)
        fd = obj_at(d)
        
        for _ in range(15):
            if time.monotonic() - start >= time_budget:
                break
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = lo + gr * (hi - lo)
                fc = obj_at(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = obj_at(d)
        
        t_opt = c if fc < fd else d
        f_opt = min(fc, fd)
        if f_opt < best_f:
            best_f = f_opt
            best_t = t_opt
    
    return best_t, best_f

def mirror_descent(a, time_budget=30.0, lr_init=0.05, decay=0.999):
    """Mirror descent (multiplicative weights) on the probability simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        
        # Subgradient: d (a*a)_{k_max} / d a_i
        grad = np.zeros(n)
        # (a*a)_{k} = sum_i a_i * a_{k-i}
        # d/d a_j = a_{k-j} + a_{j-k} (if indices valid)
        # But since a*a is symmetric, d/d a_j = a_{k-j} + (something)
        # More carefully: (a*a)_k = sum_i a_i a_{k-i}
        # d/d a_j: term where i=j: a_{k-j}; term where k-i=j => i=k-j: a_{k-j}
        # If j != k-j (i.e., 2j != k), both terms exist: grad_j = 2 * a_{k-j}
        # If j == k-j (i.e., 2j == k), only one term: grad_j = a_{k-j} = a_j
        for j in range(n):
            val = 0.0
            if 0 <= k_max - j < n:
                val += a[k_max - j]
            if 0 <= j - k_max < n:
                val += a[j - k_max]
            grad[j] = val
        
        # Mirror descent: a_i <- a_i * exp(-lr * grad_i) / Z
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        lr *= decay
        it += 1
        
        if it - best_it > 500:
            a = best_a.copy()
            lr = lr_init * 0.5
    
    return best_a

"""
Peak-shaving + Adaptive Top-K LP optimization for step function coefficients.
Uses vectorized gradient mirror descent, adaptive top-K LP polishing with
parabolic line search, and systematic length exploration to minimize
2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_subgrad(a, k):
    """Compute subgradient of (a*a)_k w.r.t. a. Vectorized."""
    n = len(a)
    j_arr = np.arange(n)
    valid = (k - j_arr >= 0) & (k - j_arr < n)
    grad = np.zeros(n)
    grad[valid] = 2.0 * a[k - j_arr[valid]]
    return grad


def build_topk_matrix(a_norm, top_k):
    """Build constraint matrix for top-K lags of conv(a, g)."""
    n = len(a)
    conv_aa = compute_conv(a_norm)
    if top_k >= 2 * n - 1:
        idx = np.arange(2 * n - 1)
    else:
        idx = np.argsort(conv_aa)[::-1][:top_k]
    idx = np.sort(idx)
    A = np.zeros((len(idx), n))
    j_arr = np.arange(n)
    for row, k in enumerate(idx):
        valid = (k - j_arr >= 0) & (k - j_arr < n)
        A[row, valid] = a_norm[k - j_arr[valid]]
    return A


def solve_lp(a, top_k):
    """Solve LP: min M s.t. conv(a_norm, g)_k <= M for top-k lags, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_topk_matrix(a_norm, top_k)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g_dir, time_budget=3.0):
    """Parabolic + golden section line search for optimal blend factor."""
    start = time.monotonic()

    def obj_at(t):
        if t < 0:
            t = 0
        if t > 2.0:
            t = 2.0
        a_try = (1 - t) * a + t * g_dir
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    # Scan for good region
    t_scan = [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
    best_t = 0.0
    best_f = obj_at(0.0)

    for t in t_scan:
        if time.monotonic() - start > time_budget * 0.35:
            break
        f = obj_at(t)
        if f < best_f:
            best_f = f
            best_t = t

    # Golden section refinement around best point
    lo = max(0.0, best_t - 0.25)
    hi = min(2.0, best_t + 0.25)

    if hi - lo > 0.005:
        gr = 2 - (1 + np.sqrt(5)) / 2
        c = lo + gr * (hi - lo)
        d = lo + (1 - gr) * (hi - lo)
        fc = obj_at(c)
        fd = obj_at(d)

        for _ in range(18):
            if time.monotonic() - start >= time_budget:
                break
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = lo + gr * (hi - lo)
                fc = obj_at(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = obj_at(d)

        t_opt = c if fc < fd else d
        f_opt = min(fc, fd)
        if f_opt < best_f:
            best_f = f_opt
            best_t = t_opt

    return best_t, best_f


def mirror_descent(a, time_budget=30.0, lr_init=0.08, decay=0.998):
    """Mirror descent (multiplicative weights) on the probability simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    it = 0
    best_it = 0
    j_arr = np.arange(n)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))

        # Vectorized subgradient
        valid = (k_max - j_arr >= 0) & (k_max - j_arr < n)
        grad = np.zeros(n)
        grad[valid] = 2.0 * a[k_max - j_arr[valid]]

        # Mirror descent step
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        lr *= decay
        it += 1

        if it - best_it > 400:
            a = best_a.copy()
            lr = lr_init * 0.4
            best_it = it

    return best_a


def nesterov_sgd(a, time_budget=30.0, lr_init=0.04):
    """Subgradient descent with Nesterov momentum on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    it = 0
    momentum = np.zeros(n)
    beta = 0.7
    best_it = 0
    j_arr = np.arange(n)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))

        valid = (k_max - j_arr >= 0) & (k_max - j_arr < n)
        grad = np.zeros(n)
        grad[valid] = 2.0 * a[k_max - j_arr[valid]]

        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        lr *= 0.9997
        it += 1

        if it - best_it > 2000:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = 0.01
            best_it = it

    return best_a


def lp_polish(a, time_budget=100.0, top_k_mode='adaptive', verbose=False, max_iters=60):
    """
    LP polishing with adaptive top-K and parabolic line search.
    top_k_mode: 'adaptive' starts small K and grows, 'half' uses n/2, 'full' uses all.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777 + it)

    # Determine initial top_k
    total_lags = 2 * n - 1
    if top_k_mode == 'full':
        current_k = total_lags
    elif top_k_mode == 'half':
        current_k = n
    else:  # adaptive
        current_k = max(20, n // 3)

    increase_schedule = [0.3, 0.5, 0.7, 1.0]  # fractions of total lags
    increase_idx = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Solve LP
        g_lp, M_lp = solve_lp(a, top_k=current_k)

        if g_lp is None:
            break

        # Line search
        ls_budget = min(2.5, remaining * 0.06)
        t_opt, obj_opt = line_search(a, g_lp, time_budget=ls_budget)

        if t_opt > 0.001 and obj_opt < best_obj - 1e-14:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1

        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, k={current_k}, "
                  f"elapsed={time.monotonic()-start:.1f}s")

        # Adaptive: increase K when stalled or periodically
        if stall >= 3 and increase_idx < len(increase_schedule):
            current_k = min(total_lags, int(total_lags * increase_schedule[increase_idx]))
            increase_idx += 1
            # Perturb to escape
            noise = rng.normal(0, 0.015 * stall, n)
            a = best_a + noise
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 0.01:
                a = a / s
            else:
                a = best_a.copy()
            stall = 0
        elif stall >= 7:
            # Major restart
            noise = rng.normal(0, 0.05, n)
            a = best_a + noise
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 0.01:
                a = a / s
            else:
                a = best_a.copy()
            stall = 0
            # Switch to full LP
            current_k = total_lags
            increase_idx = len(increase_schedule)

    return best_a


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / (1.0 + 3.0 * (2 * x - 1) ** 4)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
