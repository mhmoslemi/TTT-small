Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp_top_m(a, m=50):
    """
    Solve LP: min M s.t. (a_norm * g)_k <= M for top-m lags k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv = compute_conv(a_norm)
    # Only consider first half due to symmetry: conv[k] = conv[2n-2-k]
    half = min(n, 2 * n - 1)
    conv_half = conv[:half]
    # Find top m lags in first half
    m_actual = min(m, half)
    top_indices = np.argsort(conv_half)[-m_actual:][::-1]
    top_indices = np.sort(top_indices)

    # Build constraint rows for these lags
    # (a_norm * g)_k = sum_j a_norm[j] * g[k-j] for valid j, k-j
    A = np.zeros((m_actual, n))

            g = np.maximum(result.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, float(result.fun)
    except Exception:
        pass
    return None, None


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    # Only use first n lags for efficiency (symmetry)
    conv_half = conv[:n]
    max_half = float(np.max(conv_half))
    exp_vals = np.exp(tau * (conv_half - max_half))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(n)
        p[np.argmax(conv_half)] = 1.0
    else:
        p = exp_vals / sum_exp

    # Gradient of sum_k p_k * conv_k with respect to a
    # d conv_k / d a_m = 2 * a_{k-m} if valid
    # So gradient = 2 * sum_k p_k * flip_shift(a, k)
    # This is equivalent to 2 * (p * a_circ_flip) where we handle the convolution structure
    # Using FFT: if P = fft(p, N) and A = fft(a, N), then
    # gradient = 2 * ifft(P * conj(A))[:n]
    N = 1
    while N < 2 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.maximum(v, 0)
    s = float(np.sum(v))
    if s < 1e-15:
        return np.ones(n) / n
    return v / s


def nesterov_polish(a, time_budget, tau_init=30.0, tau_max=3000.0):
    """Nesterov accelerated gradient descent with adaptive parameters."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 1.5 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = project_simplex(y)

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adaptive_lr = 0.3 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.05 * base_lr), 2.0 * base_lr)
            lr = 0.85 * lr + 0.15 * adaptive_lr

        v = beta * v - lr * g
        a_new = project_simplex(y + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr *= 0.9997
        tau = min(tau * 1.015, tau_max)
        it += 1

        if no_improve > 3000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            tau = tau_init
            no_improve = 0
            beta = min(0.95, beta + 0.05)

    return best_a


def fourier_perturb(a, rng, strength=0.03):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 8
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    return project_simplex(a_new)


def random_perturb(a, rng, strength=0.02):
    """Add random multiplicative perturbation."""
    n = len(a)
    pert = 1.0 + strength * rng.standard_normal(n)
    a_new = a * pert
    return project_simplex(a_new)


def lp_alternative_search(a, time_budget, seed=42, verbose=False):
    """
    LP-based alternating minimization with adaptive top-m and blending.
    Uses scipy HiGHS for speed.
    """
    n = len(a)
    a = project_simplex(a.copy())

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    t_blend = 0.4
    stall_count = 0
    m_lags = 50
    use_full_lp = False

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1:
            break

        # Choose LP variant
        if use_full_lp:
            g_lp, M_lp = solve_lp_full(a)
        else:
            g_lp, M_lp = solve_lp_top_m(a, m=m_lags)

        if g_lp is None:
            # Fall back to other variant
            if use_full_lp:
                use_full_lp = False
                continue
            else:
                use_full_lp = True
                continue

        iter_count += 1

        # Try multiple blending factors
        candidates_t = [t_blend, t_blend * 0.5, t_blend * 2.0, 0.1, 0.3, 0.5, 0.7, 1.0]
        candidates_t = [min(max(t, 0.001), 1.0) for t in candidates_t]

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend

        for t in candidates_t:
            a_try = project_simplex((1.0 - t) * a + t * g_lp)
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            t_blend = max(0.05, min(best_local_t * 1.1, 0.9))
            stall_count = 0
            # Adapt m_lags: increase if we're making good progress
            if iter_count % 50 == 0:
                m_lags = min(100, m_lags + 5)
            if verbose and iter_count % 100 == 0:
                print(f"    LP alt {iter_count}: obj={best_obj:.10f}, t={t_blend:.3f}, m={m_lags}")
        else:
            stall_count += 1
            if stall_count == 5:
                # Try with different m
                m_lags = max(20, min(200, m_lags + 30))
                use_full_lp = not use_full_lp
            elif stall_count >= 10:
                # Perturb to escape
                strength = 0.01 + 0.005 * min(stall_count - 10, 5)
                a = fourier_perturb(best_a, rng, strength=strength)
                t_blend = max(0.2, min(t_blend * 1.3, 0.8))
                stall_count = 0
            elif stall_count >= 5:
                a = random_perturb(best_a, rng, strength=0.01)

    if verbose:
        print(f"    LP alt: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_simplex(a_new)


def generate_initialization(n, seed=42):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    # Try several types and pick the best
    candidates = []

    # Uniform
    candidates.append(np.ones(n) / n)

    # Random
    candidates.append(project_simplex(rng.dirichlet(np.ones(n) * 0.5)))

    # Bimodal (concentrated at edges)
    x = np.linspace(0, 1, n)
    for center in [0.0, 1.0, 0.5]:
        width = rng.uniform(0.1, 0.3)
        c = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        candidates.append(project_simplex(c))

    # Quadratic residue inspired
    if n > 10:
        q = np.zeros(n)
        for i in range(n):
            q[i] = 1.0 if (i * i % 4) != 1 else 0.3
        candidates.append(project_simplex(q))

    # Sinusoidal
    for freq in [2, 5, 13]:
        s = np.abs(np.sin(2 * np.pi * freq * x / n * n)) + 0.1
        candidates.append(project_simplex(s))

    # Pick the best
    best = None
    best_obj = np.inf
    for c in candidates:
        o = compute_obj(c)
        if o < best_obj:
            best_obj = o
            best = c
    return best


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP alternating, gradient polishing,
    multi-length exploration, and adaptive restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    # Load parent
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0)
        if a.size == 0 or a.sum() < 0.01:
            raise ValueError("Invalid parent")
        a = a / a.sum()
        n_init = a.size
        obj_init = compute_obj(a)
        print(f"Parent: n={n_init}, obj={obj_init:.10f}")
    except Exception:
        n_init = 800
        a = np.ones(n_init) / n_init
        obj_init = compute_obj(a)
        print(f"Default init: n={n_init}, obj={obj_init:.10f}")

    best_a = a.copy()
    best_obj = obj_init
    rng = np.random.default_rng(seed)

    # === Phase 1: LP Alternating on parent (main workhorse) ===
    print(f"\n=== Phase 1: LP Alt (parent, n={n_init}) ===")
    p1_budget = min(budget_s * 0.40, 500.0)
    a_p1 = lp_alternative_search(best_a, time_budget=p1_budget, seed=seed, verbose=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1.copy()
        print(f"  ** Phase 1: obj={best_obj:.10f} **")
    else:
        print(f"  Phase 1: obj={obj_p1:.10f} (no improvement)")

    # === Phase 2: Gradient polish ===
    print(f"\n=== Phase 2: Gradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = min(remaining * 0.2, 150.0)
        a_p2 = nesterov_polish(best_a, time_budget=p2_budget, tau_init=50.0, tau_max=2000.0)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_obj:
            best_obj = obj_p2
            best_a = a_p2.copy()
            print(f"  ** Phase 2: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 2: obj={obj_p2:.10f}")

    # === Phase 3: Second LP pass with perturbation ===
    print(f"\n=== Phase 3: LP Alt (perturbed) ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        a_pert = fourier_perturb(best_a, rng, strength=0.015)
        p3_budget = min(remaining * 0.3, 200.0)
        a_p3 = lp_alternative_search(a_pert, time_budget=p3_budget, seed=seed + 999, verbose=False)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3.copy()
            print(f"  ** Phase 3: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 3: obj={obj_p3:.10f}")

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length Exploration ===")
    n_best = len(best_a)
    n_options = []
    for delta in [-100, -50, -20, 20, 50, 100, 150, 200]:
        n_try = n_best + delta
        if 400 <= n_try <= 1500 and n_try not in n_options:
            n_options.append(n_try)

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break

        a_try = resample_to_length(best_a, n_try)
        alloc = min(remaining * 0.3, 150.0)
        if alloc < 40:
            break

        # LP search on new length
        a_try = lp_alternative_search(a_try, time_budget=alloc * 0.7, seed=n_try, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_try = nesterov_polish(a_try, time_budget=min(remaining * 0.3, 30.0),
                                     tau_init=80.0, tau_max=1000.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: obj={best_obj:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 5: Fresh initializations at best length ===
    print(f"\n=== Phase 5: Fresh Inits at n={len(best_a)} ===")
    remaining = deadline - time.monotonic()
    n_curr = len(best_a)

    for trial in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        init = generate_initialization(n_curr, seed=seed + 1000 * (trial + 1))
        alloc = min(remaining * 0.3, 120.0)
        if alloc < 40:
            break

        a_try = lp_alternative_search(init, time_budget=alloc, seed=seed + 2000 * (trial + 1),
                                       verbose=False)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  ** Fresh init {trial}: obj={best_obj:.10f} **")
        else:
            print(f"  Fresh init {trial}: obj={obj_try:.10f}")

    # === Phase 6: Final refinement ===
    print(f"\n=== Phase 6: Final Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 80.0)
        a_final = lp_alternative_search(best_a, time_budget=alloc * 0.6,
                                         seed=seed + 77777, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_final = nesterov_polish(a_final, time_budget=min(remaining * 0.4, 20.0),
                                       tau_init=100.0, tau_max=1500.0)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  ** Final: obj={best_obj:.10f} **")
        else:
            print(f"  Final: obj={obj_final:.10f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
