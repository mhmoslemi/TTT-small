Hybrid optimizer combining Adam-smoothed gradient, QP peak reduction,
TR-SOCP polishing, and multi-restart strategy for minimum peak autocorrelation
of non-negative sequences. Key innovations: per-coordinate adaptive learning rates,
QP-based peak shaving with linearized constraints, and structured multi-restart.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    Computed via FFT cross-correlation.

    n = len(y)
    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1
    W = fft(weights, N_fft)
    Y_rev = fft(y[::-1], N_fft)
    c = ifft(W * Y_rev).real
    grad = 2.0 * c[n - 1:2 * n - 1]
    return grad


def adam_smoothed_descent(a, time_budget, tau_init=10.0, tau_end=400.0,
                          lr_init=None, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on smooth peak approximation with increasing temperature."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if lr_init is None:
        lr_init = 5.0 / np.sqrt(n)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        lr = lr_init / (1.0 + 0.5 * progress)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -60, 0)
        weights = np.exp(shifted)
        wsum = float(np.sum(weights))
        if wsum < 1e-30:
            weights = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            weights = weights / wsum

        grad = compute_smooth_gradient(weights, a)

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        if stall > 500:
            break

    return best_a


def momentum_smoothed_descent(a, time_budget, tau_init=15.0, tau_end=300.0,
                              lr_init=None, momentum=0.85):
    """Momentum-based smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if lr_init is None:
        lr_init = 4.0 / np.sqrt(n)

    v = np.zeros(n)
    t = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        lr = lr_init * (1.0 - 0.5 * progress ** 0.5)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -60, 0)
        weights = np.exp(shifted)
        wsum = float(np.sum(weights))
        if wsum < 1e-30:
            weights = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            weights = weights / wsum

        grad = compute_smooth_gradient(weights, a)

        v = momentum * v + grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        t += 1

    return best_a


def qp_peak_reduction(a, time_budget, M_init=None, verbose=False):
    """
    QP-based peak reduction: alternately solve a QP that minimizes the change
    in the sequence norm subject to linearized peak constraints.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if M_init is None:
        conv = compute_conv(a)
        M = float(np.max(conv))
    else:
        M = M_init

    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        current_max = float(np.max(conv))
        M = max(M * 0.9995, current_max * 0.99)

        A = build_conv_matrix(a)
        rhs = (M - conv) / 2.0

        delta = cp.Variable(n)
        constraints = [
            delta >= -a * 0.99,
            cp.sum(delta) == 0,
            A @ delta <= rhs,
        ]

        prob = cp.Problem(cp.Minimize(cp.sum_squares(delta) + 2.0 * (a @ delta)),
                          constraints)

        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    d = np.asarray(delta.value).flatten()
                    a_new = a + d
                    a_new = np.maximum(a_new, 0)
                    s_new = a_new.sum()
                    if s_new > 1e-10:
                        a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                    a = a_new
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            if no_improve >= 3:
                break
            # Perturb and retry
            rng = np.random.default_rng(it)
            a = best_a * np.exp(rng.normal(0, 0.005, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve += 1
            if no_improve > 10:
                break

        it += 1
        if verbose and it % 20 == 0:
            print(f"    QP it{it}: obj={best_obj:.8f}, M={M:.6f}, "
                  f"t={time.monotonic()-start:.0f}s")

    return best_a


def solve_tr_sscp(a, r, M_target=None):
    """
    Solve trust region SOCP:
    min M
    s.t. C_old[k] + 2*(a*delta)[k] <= M for all k
         ||delta||_2 <= r
         sum(a+delta) = 1
         a+delta >= 0
    """
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)

    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta

    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2.0 * (A_cross @ delta) <= M,
    ]

    if r is not None:
        constraints.append(cp.norm(delta, 2) <= r)

    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_direction(a, M_target=None):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None
