def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:

    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a, time_budget=30.0):

                s = float(np.sum(g_val))

"""
Optimizes step function coefficients using enhanced coordinate descent with
active-set tracking, LP-guided line search, symmetry exploitation, and
multi-n exploration. Key improvements over baseline: (1) LP-guided coordinate
priority ordering, (2) proper 1D line search after LP instead of fixed blending,
(3) symmetric constraint to halve search space, (4) wider n sweep with
interpolation-aware resampling, (5) active-set tracking for faster CD.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, time_budget=30.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def f_1d_coord(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i):
    """Evaluate 1D objective for coordinate descent at value x."""
    if len(c_k_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom


def coordinate_descent(a, time_budget=300.0, seed=42, verbose=False, symmetric=False):
    """
    Coordinate descent with active-set tracking and smart bracketing.
    If symmetric=True, only optimizes first half and mirrors.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0

    n_opt = n // 2 if symmetric else n

    while time.monotonic() - start < time_budget:
        sweep += 1

        # Periodic full recompute to avoid floating point drift
        if sweep % 5 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        order = rng.permutation(n_opt)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = i
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Smart candidate generation based on current state
            scale = max(v, S / n, 1e-8)
            candidates = np.array([
                0.0, v * 0.2, v * 0.5, v * 0.8, v,
                v * 1.2, v * 1.5, v * 2.0, v * 3.0,
                S / (2.0 * n), S / n, S / (4.0 * n), S / (8.0 * n)
            ])
            candidates = np.clip(candidates, 0.0, 10.0 * S)

            best_x = v
            best_f = f_1d_coord(v, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            for x in candidates:
                fx = f_1d_coord(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement
            x_lo = max(0.0, best_x - scale * 0.5)
            x_hi = best_x + scale * 0.5
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d_coord(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            fd = f_1d_coord(d_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)

            for _ in range(45):
                if x_hi - x_lo < 1e-15 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d_coord(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d_coord(d_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-15:
                continue

            # Update a and conv
            delta = x_opt - v
            a[i] = x_opt

            if symmetric and i < n // 2:
                a[n - 1 - i] = x_opt
                # Update symmetric part of conv
                j_sym = n - 1 - i
                delta_sym = x_opt - a[n - 1 - i]  # already set, so 0
                # Actually we already set it, so just update conv for both
                if 0 <= 2 * j_sym <= 2 * n - 2:
                    conv[2 * j_sym] += x_opt * x_opt - v * v
                ks_sym = np.arange(max(0, j_sym), min(2 * n - 1, j_sym + n))
                mask_sym = (ks_sym != 2 * j_sym)
                ks_sym_lin = ks_sym[mask_sym]
                if len(ks_sym_lin) > 0:
                    a_ki_sym = a[ks_sym_lin - j_sym]
                    conv[ks_sym_lin] += 2.0 * delta * a_ki_sym

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta
            if symmetric and i < n // 2:
                S += delta
            S = max(S, 1e-15)

            # Track best
            max_conv = float(np.max(conv))
            obj = 2.0 * n * max_conv / (S * S)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def lp_with_line_search(a, time_budget=60.0, verbose=False):
    """LP polishing with proper 1D line search instead of fixed blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    stall = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        g_lp, M_lp = solve_lp(a, time_budget=min(remaining * 0.5, 30.0))
        if g_lp is None:
            break

        # Line search in direction (g_lp - a)
        # Minimize obj((1-t)*a + t*g_lp) for t in [0, 1]
        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return np.inf
            a_try = a_try / s_try
            return compute_obj(a_try)

        # Golden section on [0, 1]
        x_lo, x_hi = 0.0, 1.0
        c_pt = x_hi - gr * (x_hi - x_lo)
        d_pt = x_lo + gr * (x_hi - x_lo)
        fc = line_obj(c_pt)
        fd = line_obj(d_pt)

        for _ in range(20):
            if x_hi - x_lo < 1e-10:
                break
            if fc < fd:
                x_hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = x_hi - gr * (x_hi - x_lo)
                fc = line_obj(c_pt)
            else:
                x_lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = x_lo + gr * (x_hi - x_lo)
                fd = line_obj(d_pt)

        t_opt = (x_lo + x_hi) / 2.0
        a_new = (1.0 - t_opt) * a + t_opt * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            stall += 1
        else:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1

        if stall > 4:
            # Perturb and retry
            pert = rng.normal(0, 0.02, n)
            a = a * np.exp(pert)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0
            if time.monotonic() - start > time_budget * 0.85:
                break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt(x + 1.0)
        a = a / a.sum()
    elif mode == 3:
        x = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt((x + 1.0) * (n - x))
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2.0) ** 2
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3.0 * x)
        a = a / a.sum()
    elif mode == 6:

        a = 1.0 + 0.5 * np.sin(4.0 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def symmetrize(a):
    """Make sequence symmetric: a[j] = (a[j] + a[n-1-j]) / 2."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    return a_sym

        a_lp_final = lp_with_line_search(best_seq, time_budget=min(remaining * 0.5, 60.0))
        obj_lp_final = compute_obj(a_lp_final)
        if obj_lp_final < best_score:
            best_score = obj_lp_final
            best_seq = a_lp_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final CD micro-polish
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_micro = coordinate_descent(best_seq, time_budget=min(remaining * 0.7, 40.0),
                                     verbose=False, seed=seed + 77777)
        obj_micro = compute_obj(a_micro)
        if obj_micro < best_score:
            best_score = obj_micro
            best_seq = a_micro.copy()
            print(f"  Micro CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
