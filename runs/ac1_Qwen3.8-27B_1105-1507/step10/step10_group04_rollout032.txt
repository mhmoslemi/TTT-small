Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative peak-shaving, smoothed gradient descent, and LP polishing.
Key innovations: multi-n exploration (400-2500), multiplicative updates with
top-T active set, and adaptive tau scheduling for smoothed GD.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def multiplicative_peak_shave(a, n_iter=200, tau=100, top_T=50, eta=0.1, verbose=False):
    """Iterative multiplicative update to reduce convolution peak."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(n_iter):
        conv = compute_conv(a)
        m = float(np.max(conv))
        
        # Compute weights
        if top_T > 0:
            # Top-T approach
            w = np.zeros(2*n-1)
            idx = np.argsort(conv)[::-1][:top_T]
            w[idx] = 1.0 / top_T
        else:
            # Softmax approach
            exp_vals = np.exp(tau * (conv - m))
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                w = np.zeros(2*n-1)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_vals / sum_exp
        
        # Compute pressure: pressure[i] = sum_k w[k] * a[k-i] (with bounds)
        # This is the cross-correlation of w with a
        # Use FFT: pressure = ifft(fft(w) * conj(fft(a)))
        N = 1
        while N < 3 * n:
            N <<= 1
        W = fft(w, N)
        A_f = fft(a, N)
        pressure = np.real(ifft(W * np.conj(A_f)))[:n]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * pressure)
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 50 == 0:
            print(f"    MPS it{it}: obj={best_obj:.8f}, max_conv={m:.8f}")
    
    return best_a

"""
Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses efficient FFT-based gradient for multiplicative peak shaving,
Adam-optimized smoothed gradient descent, LP polishing, and multi-n
exploration up to n=2500.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_gradient(a, w):
    """Compute gradient of sum_k w[k]*(a*a)[k] w.r.t. a.
    a: length n, w: length 2n-1. Returns gradient of length n."""
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    w_rev = w[::-1]
    Af = fft(a, N)
    Wf = fft(w_rev, N)
    c = np.real(ifft(np.conj(Af) * Wf))[:2*n-1]
    g = 2.0 * c[n-1:2*n-1][::-1]
    return g


def multiplicative_peak_shave(a, n_iter=500, tau=200, eta=0.5, verbose=False):
    """Multiplicative updates using smoothed gradient."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - start > 300:  # safety
            break
        
        conv = compute_conv(a)
        m = float(np.max(conv))
        
        # Compute weights (softmax)
        exp_vals = np.exp(tau * (conv - m))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            w = np.zeros(2*n-1)
            w[np.argmax(conv)] = 1.0
        else:
            w = exp_vals / sum_exp
        
        # Compute gradient
        g = compute_gradient(a, w)
        
        # Multiplicative update
        g_max = float(np.max(g))
        if g_max < 1e-15:
            break
        a_new = a * np.exp(-eta * g / g_max)
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 100 == 0:
            print(f"    MPS it{it}: obj={best_obj:.8f}, tau={tau}")
    
    return best_a


def smoothed_gd_adam(a, time_budget, tau_start=100, tau_end=5000, verbose=True):
    """Smoothed gradient descent with Adam optimizer."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adam state
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        
        tau = tau_start + (tau_end - tau_start) * progress
        
        conv = compute_conv(a)
        m_conv = float(np.max(conv))
        
        exp_vals = np.exp(tau * (conv - m_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            w = np.zeros(2*n-1)
            w[np.argmax(conv)] = 1.0
        else:
            w = exp_vals / sum_exp
        
        g = compute_gradient(a, w)
        # Normalize gradient for stability
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        
        # Adam update
        it += 1
        m_adam = beta1 * m_adam + (1 - beta1) * g
        v_adam = beta2 * v_adam + (1 - beta2) * g**2
        m_hat = m_adam / (1 - beta1**it)
        v_hat = v_adam / (1 - beta2**it)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            it = 0
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        
        # Decay learning rate
        lr *= 0.9995
        
        if it - best_it > 5000:
            a = best_a.copy()
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            it = 0
            if it > 10000:
                break
        
        if verbose and it % 2000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val
        except Exception:
            continue
    return None


def lp_polish(a, time_budget, verbose=True):
    """LP polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp = solve_lp(a)
        if g_lp is None:

        best_t_obj = best_obj

def diverse_init(n, rng, mode=0):
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x**2 * (1-x)**0.5
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x)**2
        a = a / a.sum()
    elif mode == 6:
        # Triangular
        x = np.arange(n) / (n - 1)
        a = 1 - np.abs(2*x - 1)
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses efficient FFT-based exact gradient for multiplicative (mirror descent) peak shaving,
Adam-optimized smoothed gradient descent, LP polishing, and multi-n exploration up to n=3000.
Key innovation: correct O(n log n) gradient via cross-correlation FFT, making high-iteration
optimization feasible for large n.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_gradient(a, w):
    """Compute exact gradient of sum_k w[k]*(a*a)[k] w.r.t. a using FFT cross-correlation.
    a: length n (nonneg, sum=1), w: length 2n-1 (nonneg, weights on convolution).
    Returns gradient of length n."""
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    w_rev = w[::-1]
    Af = fft(a, N)
    Wf = fft(w_rev, N)
    c = np.real(ifft(np.conj(Af) * Wf))[:2 * n - 1]
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return g


def get_weights(conv, tau=None, top_T=None):
    """Compute weights on convolution positions from softmax or top-T."""
    n_conv = len(conv)
    m = float(np.max(conv))
    if top_T is not None and top_T > 0:
        w = np.zeros(n_conv)
        idx = np.argsort(conv)[::-1][:top_T]
        w[idx] = 1.0 / top_T
        return w
    if tau is None:
        tau = 100.0
    exp_vals = np.exp(tau * (conv - m))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        w = np.zeros(n_conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_vals / sum_exp
    return w


def multiplicative_peak_shave(a, time_budget, tau_start=100, tau_end=5000,
                              eta_start=0.5, eta_end=0.05, verbose=False):
    """Mirror descent (multiplicative) peak shaving with adaptive tau and eta."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        tau = tau_start + (tau_end - tau_start) * progress
        eta = eta_start * (eta_end / eta_start) ** progress

        conv = compute_conv(a)
        w = get_weights(conv, tau=tau)
        g = compute_gradient(a, w)

        g_max = float(np.max(g))
        if g_max < 1e-15:
            break

        a_new = a * np.exp(-eta * g / g_max)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        if it - best_it > 5000:
            a = best_a.copy()
            best_it = it
            if it > 10000:
                break

        if verbose and it % 500 == 0:
            print(f"    MPS it{it}: obj={best_obj:.8f}, tau={tau:.0f}, eta={eta:.4f}")

    return best_a


def smoothed_gd_adam(a, time_budget, tau_start=100, tau_end=8000, verbose=False):
    """Smoothed gradient descent with Adam optimizer and adaptive tau."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.7

        conv = compute_conv(a)
        w = get_weights(conv, tau=tau)
        g = compute_gradient(a, w)

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        it += 1
        m_adam = beta1 * m_adam + (1 - beta1) * g
        v_adam = beta2 * v_adam + (1 - beta2) * g ** 2
        m_hat = m_adam / (1 - beta1 ** it)
        v_hat = v_adam / (1 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            it = 0
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        lr *= 0.9997

        if it - best_it > 3000:
            a = best_a.copy()
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            it = 0

        if verbose and it % 1000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = sum_j a[j]*g[k-j] (linear conv)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0. Returns g."""
    n = len(a)
    if n > 2000:
        return None
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_polish(a, time_budget, verbose=False):
    """LP polishing with line-search mixing."""
    n = len(a)
    if n > 2000:
        return a
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        if time_budget - (time.monotonic() - start) < 5:
            break

        g_lp = solve_lp(a)
        if g_lp is None:
            break

        best_t_obj = best_obj
        best_t = 0.0
        best_a_local = best_a.copy()

        for t in np.concatenate([np.linspace(0.05, 1.0, 12), [0.02, 0.01]]):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.4f}")
        else:
            stall += 1
            if stall > 3:
                break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = x**2 * (1 - x)**2
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
        a = a / a.sum()
    elif mode == 5:
        x = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        a = np.exp(-((x - center) / (n * 0.3))**2)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = (1 + 0.3 * np.cos(2 * np.pi * x)) * np.sin(np.pi * x) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = rng.uniform(0.5, 1.5, n)
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def optimize_at_n(a, time_budget, verbose=False):
    """Run the full optimization pipeline at a fixed n."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Stage 1: MPS with moderate tau
    budget_s1 = time_budget * 0.35
    a1 = multiplicative_peak_shave(best_a, time_budget=budget_s1,
                                   tau_start=50, tau_end=1000,
                                   eta_start=0.5, eta_end=0.1, verbose=verbose)
    obj1 = compute_obj(a1)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()

    # Stage 2: GD with higher tau
    remaining = time_budget - (time_budget * 0.35) - (time.monotonic() - (time.monotonic() - time_budget * 0.35) + time_budget * 0.35)
    # Just use fractions of budget
    budget_s2 = time_budget * 0.30
    a2 = smoothed_gd_adam(best_a, time_budget=budget_s2,
                          tau_start=200, tau_end=5000, verbose=verbose)
    obj2 = compute_obj(a2)
    if obj2 < best_obj:
        best_obj = obj2
        best_a = a2.copy()

    # Stage 3: LP polish (only if n is small enough)
    budget_s3 = time_budget * 0.20
    if n <= 2000:
        a3 = lp_polish(best_a, time_budget=budget_s3, verbose=verbose)
        obj3 = compute_obj(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()

    # Stage 4: Final high-tau MPS
    budget_s4 = time_budget * 0.15
    a4 = multiplicative_peak_shave(best_a, time_budget=budget_s4,
                                   tau_start=1000, tau_end=8000,
                                   eta_start=0.3, eta_end=0.05, verbose=verbose)
    obj4 = compute_obj(a4)
    if obj4 < best_obj:
        best_obj = obj4
        best_a = a4.copy()

    return best_a, best_obj

    """
    Multi-n search with mirror descent peak shaving, Adam-smoothed GD,
    and LP polishing. Uses exact O(n log n) FFT gradient.
    Explores n from 500 to 3000 with adaptive time allocation.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = diverse_init(800, rng, mode=2)
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    print(f"Starting: n={n_parent}, score={best_score:.8f}")

    print(f"\n=== Phase 1: Improve at n={n_parent} ===")
    p1_budget = min((deadline - time.monotonic()) * 0.20, 180.0)
    if p1_budget > 30:
        a_p1, obj_p1 = optimize_at_n(best_seq, time_budget=p1_budget, verbose=True)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()

    # Sort by priority: prefer larger n and closer to parent
