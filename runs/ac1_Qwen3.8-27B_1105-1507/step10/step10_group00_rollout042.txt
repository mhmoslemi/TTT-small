def subgradient_step(a, lr, momentum, beta):

    One subgradient step. Returns new a, new momentum, obj.

    x is the log-parameters, a = softmax(x).

    a /= a.sum()

def build_conv_matrix(a, subset_indices=None):
    """Build matrix A (len(subset) x n) such that A @ g = subset of conv(a, g)."""
    n = len(a)
    if subset_indices is None:
        subset_indices = range(2 * n - 1)
    L = 2 * n - 1
    A = np.zeros((len(subset_indices), n))
    for idx, k in enumerate(subset_indices):

    return A

import numpy as np
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x : x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) - (cssv - 1)
    rho = np.nonzero(cond > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    w = rho[-1]
    theta = (cssv[w] - 1) / (w + 1.0)
    return np.maximum(v - theta, 0)


def smooth_peak_grad(a, t):
    """
    Compute log-sum-exp smooth approximation of max(conv(a,a)) and its gradient.
    Returns (smooth_obj_value, gradient_vector).
    Uses FFT for O(n log n) gradient computation.
    """
    n = len(a)
    conv = compute_conv(a)
    L = len(conv)
    m_val = np.max(conv)

    # Numerically stable log-sum-exp
    tc = t * (conv - m_val)
    tc_max = np.max(tc)
    exp_tc = np.exp(tc - tc_max)
    log_sum_exp = m_val + (tc_max + np.log(np.sum(exp_tc))) / t

    # Softmax weights
    softmax = exp_tc / np.sum(exp_tc)

    # FFT-based gradient computation
    N = 1
    while N < 2 * L:
        N <<= 1

    a_pad = np.zeros(N)
    a_pad[:n] = a
    s_pad = np.zeros(N)
    s_pad[:L] = softmax

    FA = np.fft.rfft(a_pad, N)
    FS = np.fft.rfft(s_pad, N)

    convol = np.fft.irfft(FA * FS, N)[:n]

    grad = cross + convol
    return log_sum_exp, grad

    Nesterov-accelerated gradient descent on log-sum-exp objective.
    Uses simplex projection after each step.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.01
    it = 0
    stall = 0
    t_curr = t_init
    momentum = np.zeros(n)
    beta = 0.8
    y = a.copy()  # Nesterov auxiliary

        a_new = project_simplex(a_new)

def build_conv_matrix_reduced(a, top_m):

        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[ii, j] = a[i]
    return A


def solve_lp_fast(a, top_m=100):
    """Solve reduced LP: min M s.t. top-m conv constraints, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s

    A = build_conv_matrix_reduced(a_norm, top_m)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_full(a):
    """Solve full LP with all constraints."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s

    L = 2 * n - 1
    A = np.zeros((L, n))
    for k in range(L):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g, time_budget=3.0):
    """Find optimal blend factor t minimizing obj((1-t)*a + t*g). Golden section on [0, 2]."""
    lo, hi = 0.0, 2.0
    gr = (3.0 - np.sqrt(5.0)) / 2.0

    def obj_at(t_val):
        a_try = (1.0 - t_val) * a + t_val * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    c = lo + gr * (hi - lo)
    d = lo + (1.0 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    start = time.monotonic()
    for _ in range(30):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)

    # Check some standard points
    for t_try in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.2, 1.5]:
        obj_try = obj_at(t_try)
        if obj_try < obj_best:
            obj_best = obj_try
            t_best = t_try

    return t_best, obj_best


def lp_polish(a, time_budget=200.0, verbose=False, max_iters=100, use_full_lp=False):
    """
    Iterative LP polishing with adaptive constraint count.
    Starts with reduced constraints (fast), switches to full LP near convergence.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777 + n)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        # Adaptive top_m: start small, grow near convergence
        if stall < 5:
            top_m = min(n, max(30, n // 10))
        elif stall < 15:
            top_m = min(n, max(50, n // 5))
        else:
            top_m = min(n, max(100, n // 3))

        # Solve LP
        if use_full_lp and stall > 10:
            g_lp, M_lp = solve_lp_full(a)
        else:
            g_lp, M_lp = solve_lp_fast(a, top_m=top_m)

        if g_lp is None:
            break

        # Line search
        t_budget = min(2.5, remaining * 0.05)
        t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=t_budget)

        if t_opt > 0.01 and obj_opt < best_obj - 1e-14:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 5 == 0:
            print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, top_m={top_m}, elapsed={time.monotonic()-start:.1f}s")

        # Adaptive restart
        if stall >= 4:
            if stall <= 8:
                # Small perturbation
                noise = rng.normal(0, 0.02 * stall, n)
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
            elif stall <= 15:
                # Medium perturbation with U-shape
                x = np.linspace(0, 1, n)
                u_shape = 1.0 / (0.1 + 3.0 * (2 * x - 1) ** 2)
                a = 0.3 * u_shape + 0.7 * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()
            else:
                # Large restart
                x = np.linspace(0, 1, n)
                mode = it % 4
                if mode == 0:
                    new_init = 1.0 / (0.05 + 2.0 * (2 * x - 1) ** 2)
                elif mode == 1:
                    new_init = np.exp(3.0 * (2 * x - 1) ** 2)
                elif mode == 2:
                    new_init = (2 * x - 1) ** 4 + 0.02
                else:
                    new_init = 1.0 / np.sqrt(np.maximum(1e-6, 1.0 - (2 * x - 1) ** 2))
                a = 0.4 * new_init + 0.6 * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()
            stall = 0

    return best_a


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.005, 0.995, n)
    a = 1.0 / np.sqrt(np.maximum(1e-6, 1.0 - (2 * x - 1) ** 2))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def parametric_init(n, alpha=2.0, beta=2.0, eps=0.05):
    """Parametric U-shaped: 1/(eps + alpha*(2x-1)^beta)."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (eps + alpha * np.abs(2 * x - 1) ** beta)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = 1.0 / (0.05 + 3.0 * (2 * x - 1) ** 2)
    elif mode == 1:
        a = np.exp(3.0 * (2 * x - 1) ** 2)
    elif mode == 2:
        a = (2 * x - 1) ** 6 + 0.05
    elif mode == 3:
        a = 1.0 / (0.01 + 5.0 * (2 * x - 1) ** 4)
    elif mode == 4:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 8) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 8) ** 2)) + 0.005
    elif mode == 5:
        a = 4.0 * np.abs(x - 0.5) ** 1.5 + 0.02
    elif mode == 6:
        a = 1.0 / np.cosh(3.0 * (2 * x - 1) - 1.1)
    elif mode == 7:
        a = rng.uniform(0.05, 1.0, n)
        kernel = np.ones(7) / 7.0
        a = np.convolve(a, kernel, mode='same')
    else:
        a = 1.0 / (0.02 + 4.0 * np.abs(2 * x - 1) ** 1.5)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(a_old, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a_old)
    if n_old == n_new:
        return a_old.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a_old)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return a_new / s


def subgradient_warmstart(a, time_budget=30.0):
    """Quick subgradient descent for warm-starting."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.03
    momentum = np.zeros(n)
    beta = 0.7
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))

        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]

        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        lr *= 0.9995
        it += 1

        if stall > 3000:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = 0.01
            stall = 0

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: hybrid optimization combining gradient descent,
    LP polishing with constraint reduction, n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Initialization ---
    n_current = 800

    # Try to load parent
    parent_loaded = False
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                parent_loaded = True
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default U-shaped init (n={n_current}): obj={best_score:.8f}")

    # Try several initializations and keep the best
    inits = [
        ("u_shaped_3", u_shaped_init(n_current, 3.0)),
        ("u_shaped_5", u_shaped_init(n_current, 5.0)),
        ("arc", arcsine_init(n_current)),
        ("param_2_2", parametric_init(n_current, 2.0, 2.0, 0.05)),
        ("param_3_1", parametric_init(n_current, 3.0, 1.0, 0.02)),
    ]
    for name, init in inits:
        obj_init = compute_obj(init)
        print(f"  Init {name}: obj={obj_init:.8f}")
        if obj_init < best_score:
            best_score = obj_init
            best_seq = init.copy()
            print(f"  -> New best from {name}")

    print(f"\nInitial best: obj={best_score:.8f}, n={len(best_seq)}")

    # === Phase 1: Gradient descent warm-up ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 1: FFT Gradient Descent (warm-up) ===")
        alloc = min(60.0, remaining * 0.1)
        a_gd = gradient_descent_smooth(best_seq, time_budget=alloc, t_init=15.0, t_min=100.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_gd:.8f} (no improvement, keeping parent)")

    # === Phase 2: Intensive LP polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 2: LP Polishing (reduced constraints) ===")
        alloc = min(400.0, remaining * 0.4)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True, max_iters=150, use_full_lp=False)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Second LP round with gradient perturbation ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 3: LP Round 2 (gradient perturb + LP) ===")
        # Use gradient descent to perturb
        a_pert = gradient_descent_smooth(best_seq, time_budget=min(20.0, remaining * 0.1), t_init=30.0, t_min=200.0)
        # Blend with best
        a_pert = 0.5 * a_pert + 0.5 * best_seq
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(250.0, remaining * 0.35)
        a_lp2 = lp_polish(a_pert, time_budget=alloc, verbose=True, max_iters=80, use_full_lp=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj_lp2:.8f} (no improvement)")

    # === Phase 4: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print("\n=== Phase 4: n Exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            max(400, n_best - 200), max(400, n_best - 100),
            n_best + 100, n_best + 200, n_best + 400,
            600, 700, 900, 1000, 1200, 1400
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) >= 50]
        n_options = n_options[:6]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break

            a_try = interpolate_sequence(best_seq, n_try)
            # Quick gradient warm-start
            if remaining > 90:
                a_try = gradient_descent_smooth(a_try, time_budget=min(15.0, remaining * 0.15))
            # LP polish
            if remaining > 50:
                a_try = lp_polish(a_try, time_budget=min(40.0, remaining * 0.4), max_iters=30)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 5: Diverse Restarts ===")
        n_curr = len(best_seq)

        for r in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            mode = r % 8
            a_r = diverse_init(n_curr, rng, mode=mode)
            # Blend with current best
            blend = 0.3 + 0.1 * r
            a_r = (1.0 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            # Quick gradient
            if remaining > 50:
                a_r = gradient_descent_smooth(a_r, time_budget=min(12.0, remaining * 0.1))

            # LP
            if remaining > 30:
                a_r = lp_polish(a_r, time_budget=min(30.0, remaining * 0.25), max_iters=20)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 6: Final Polish ===")
        alloc = min(remaining - 10, 150.0)
        if alloc > 20:
            a_final = lp_polish(best_seq, time_budget=alloc, verbose=True, max_iters=60, use_full_lp=True)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final: obj={best_score:.8f} **")
            else:
                print(f"  Final: obj={obj_final:.8f}")

    # Extra: one more round if time permits
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_extra = lp_polish(best_seq, time_budget=min(remaining - 5, 60.0), max_iters=30)
        obj_extra = compute_obj(a_extra)
        if obj_extra < best_score:
            best_score = obj_extra
            best_seq = a_extra.copy()
            print(f"  Extra: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
