def evaluate_sequence(seq):
    """Evaluate a given sequence to compute objective."""
    if not isinstance(seq, (list, tuple)):
        return np.inf
    for x in seq:
        if isinstance(x, bool) or not isinstance(x, (int, float)) or np.isnan(x) or np.isinf(x):
            return np.inf
    try:
        seq = [float(x) for x in seq]
    except Exception:
        return np.inf
    for i, x in enumerate(seq):
        if x < 0 or x > 1000.0:
            return np.inf
    sum_a = float(np.sum(seq))
    if sum_a < 0.01:
        return np.inf
    # Computing full linear convolution (mode='full') requires two steps as per scipy.fft
    n = len(seq)
    padded_seq = np.array(seq)
    # We might need to pad to suitable power of two
    N = 1
    while N < 2*n-1:
        N *= 2
    # Pad the sequence to length N with zeros
    padded_seq = np.pad(padded_seq, (0, N-len(seq)), mode='constant')
    # Compute FFT
    fa = np.fft.rfft(padded_seq, N)
    conv_vals = np.fft.irfft(fa * fa, N)[:2*n-1]
    max_conv = np.max(conv_vals)
    doubled = 2.0 * n
    denom = (sum_a ** 2)
