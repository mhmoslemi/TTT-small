"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive smoothed gradient descent with exact line search,
fast LP-based reoptimization, multi-length exploration, and diverse initialization.
Key improvements: FFT-based gradient with near-exact max (high tau), 
adaptive temperature, Barzilai-Borwein step size, and structured restarts.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_smoothed_grad(a, tau):
    """
    Compute gradient of smoothed max of conv(a,a) with respect to a.
    Uses a_norm = a/sum(a) internally. Returns gradient w.r.t. a (before normalization).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-30:
        return a.copy()
    a_norm = a / s
    conv = compute_conv(a_norm)
    
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # FFT-based gradient
    N = 1
    while N < 4 * n:
        N <<= 1
    
    P = fft(p, N)
    A = fft(a_norm, N)
    g1 = ifft(np.conj(P) * A, N).real
    
    # grad[j] = g1[j] + g1[-j mod N]
    indices_pos = np.arange(n)
    indices_neg = (-indices_pos) % N
    grad_norm = g1[indices_pos] + g1[indices_neg]
    
    # Gradient w.r.t. a (chain rule: da_norm/da = (1/s) * (I - a_norm * 1^T / s))
    # But since we project to sum=1 anyway, just use grad_norm / s
    grad = grad_norm / s
    
    smoothed_val = max_c + (1.0 / tau) * math.log(sum_exp)
    return grad, smoothed_val


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives conv(a_norm, g) (dense, slow for large n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_limit=10.0):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns optimal g or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # For large n, building the dense matrix is expensive. Use sparse or skip.
    if n > 2000:
        return None
    
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val
        except Exception:
            continue
    return None

        # Cosine annealing: tau goes from tau_start to tau_end
        cos_val = 0.5 * (1 + np.cos(np.pi * progress))
        tau = tau_start + (tau_end - tau_start) * (1 - cos_val)
        
        # Nesterov lookahead
        y = a + beta * v
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y > 1e-10:
            y = y / s_y
        
        grad, _ = compute_smoothed_grad(y, tau)
        
        # Barzilai-Borwein step size

            s_vec = y - prev_a
            y_vec = grad - prev_grad
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-18 and abs(sy) > 1e-15:

        obj_new = compute_obj(a_new)

            best_it = it
            stall_count = 0
        else:
            stall_count += 1
        
        if verbose and it % 1000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")
        
        prev_a = a.copy()
        prev_grad = grad.copy()
        a = a_new
        it += 1
        
        # Decay learning rate slowly
        lr = max(1e-10, lr * 0.99995)
        
        # Reset if stuck
        if stall_count > 5000:
            a = best_a.copy()
            v = np.zeros(n)

def fourier_perturbation(a, rng, strength=0.03):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(8, n // 20)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / math.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
