"""
Exact Coordinate Descent + Smoothed Gradient + LP Hybrid for minimum peak autocorrelation.
Key innovation: exact 1D coordinate optimization using the structure of the convolution,
combined with multi-start strategy, diverse initializations, and adaptive length selection.
The coordinate descent solves each 1D subproblem exactly via ternary search, exploiting
the fact that changing one coordinate only affects a contiguous range of convolution terms.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def exact_coord_descent(a, max_cycles=50, time_limit=60.0, verbose=False, seed=42):
    """
    Exact coordinate descent: for each coordinate i, find optimal value via 1D search.
    The objective is scale-invariant so we work in unconstrained non-negative space.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    total_sum = float(np.sum(a))
    if total_sum < 1e-10:
        return a
    a = a / total_sum
    total_sum = 1.0

    conv = compute_conv(a)
    orig_max = float(np.max(conv))
    best_obj = 2.0 * n * orig_max
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    no_improve = 0

    # Precompute for each coordinate i: the affected k range
    # k ranges from i to i+n-1 (clamped to [0, 2n-2])
    # a[k-i] for k in that range is just a[0:min(n, 2n-1-i)]

    for cycle in range(max_cycles):
        elapsed = time.monotonic() - start
        if elapsed > time_limit:
            break

        cycle_improved = False
        # Random order for diversity
        order = rng.permutation(n)

        for i in order:
            ai = a[i]
            S = total_sum - ai

            # Affected k positions
            k_start = i
            k_end = min(2 * n - 2, i + n - 1)
            if k_start > k_end:
                k_start = 0
                k_end = 0

            ks = np.arange(k_start, k_end + 1)
            idx_into_a = ks - i  # indices into a: 0 to n-1
            a_ks = a[idx_into_a]
            conv_ks = conv[ks]

            # Linear coefficient for (x - ai) at each affected k
            lin_coef = 2.0 * a_ks.copy()

            # Handle k = 2i (quadratic term)
            has_2i = (2 * i >= k_start) and (2 * i <= k_end)
            if has_2i:
                idx_2i = 2 * i - k_start
                lin_coef[idx_2i] = 0.0

            # constant_part[j] = conv_ks[j] - lin_coef[j] * ai
            const_part = conv_ks - lin_coef * ai

            # Objective function for x >= 0:
            # 2*n * max(orig_max_unaffected, max_j(const_part[j] + lin_coef[j]*x + quad)) / (S+x)^2
            # where quad = x^2 - ai^2 at position 2i, 0 elsewhere
            # orig_max_unaffected: max of conv at positions NOT in [k_start, k_end]
            # For efficiency, just use orig_max (upper bound)

            def evaluate_x(x):
                vals = const_part + lin_coef * x
                if has_2i:
                    vals[idx_2i] = vals[idx_2i] + x * x - ai * ai
                local_max = float(np.max(vals))
                total_max = max(orig_max, local_max)
                denom = S + x
                if denom < 1e-15:
                    return np.inf
                return 2.0 * n * total_max / (denom * denom)

            # Determine search bounds
            hi = max(ai * 4.0, 4.0 / n, 0.05)
            f0 = evaluate_x(0.0)
            fhi = evaluate_x(hi)
            # Expand or shrink bounds
            while fhi < f0 and hi < S * 3:
                hi *= 2.0
                fhi = evaluate_x(hi)
            if fhi >= f0:
                hi = max(hi / 2, hi / 4, 1e-6)
                while hi > 1e-8:
                    fm = evaluate_x(hi / 2.0)
                    if fm < f0:
                        break
                    hi /= 2.0

            # Ternary search
            lo = 0.0
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = evaluate_x(m1)
                f2 = evaluate_x(m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
                if hi - lo < 1e-12 * max(1.0, hi):
                    break

            x_opt = max(0.0, (lo + hi) / 2.0)

            # Update
            if abs(x_opt - ai) > 1e-14:
                a[i] = x_opt
                total_sum += (x_opt - ai)

                # Update conv and orig_max periodically (every few coordinates)
                # For exactness, we should update after each change, but that's O(n log n)
                # Let's update conv every 20 coordinates
                pass

        # Update convolution at end of cycle
        total_sum = float(np.sum(a))
        a = a / total_sum
        total_sum = 1.0
        conv = compute_conv(a)
        orig_max = float(np.max(conv))
        obj = 2.0 * n * orig_max

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            cycle_improved = True
        else:
            no_improve += 1
            if no_improve >= 5:
                # Perturb to escape local minimum
                scale = 0.01 + 0.005 * min(no_improve - 5, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                total_sum = 1.0
                conv = compute_conv(a)
                orig_max = float(np.max(conv))
                no_improve = 0
                if cycle > max_cycles * 0.8:
                    break

        if verbose and (cycle + 1) % 5 == 0:
            print(f"    CD cycle {cycle+1}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a


def smoothed_gradient_descent(a, time_budget=100.0, tau_init=20.0, tau_end=400.0, seed=42):
    """Smoothed gradient descent with increasing temperature."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 8.0 / np.sqrt(n)
    momentum = 0.85
    v = np.zeros(n)
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1.0))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr = max(1e-12, lr * 0.999)
        it += 1

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, M_target=None):
    """Solve LP: min M s.t. a*g[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value).flatten(), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget=200.0, verbose=False, seed=42):
    """LP-based polishing with line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        best_t = 0
        best_t_obj = best_obj
        for t_val in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]:
            a_try = (1 - t_val) * best_a + t_val * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-15:
                best_t_obj = obj_try
                best_t = t_val

        if best_t > 0 and best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = (1 - best_t) * best_a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            s_ba = float(np.sum(best_a))
            if s_ba > 1e-10:
                best_a = best_a / s_ba
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                scale = 0.01 + 0.005 * min(no_improve - 4, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                no_improve = 0
                if it > 80:
                    break

        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def make_init(n, kind='uniform', seed=42):
    """Create different initial sequences."""
    rng = np.random.default_rng(seed)
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'raised_cosine':
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        return a / a.sum()
    elif kind == 'hann':
        a = np.hanning(n)
        return a / a.sum()
    elif kind == 'bell':
        t = np.linspace(0, 1, n)
        a = np.exp(-4 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == 'random':
        a = rng.random(n)
        return a / a.sum()
    elif kind == 'sine':
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * t)
        return a / a.sum()
    elif kind == 'plateau':
        a = np.ones(n)
        w = int(0.3 * n)
        a[:w] = 0.5
        a[n-w:] = 0.5
        return a / a.sum()
    elif kind == 'bipolar_smooth':
        # Start from a smoothed version of a Barker-like pattern
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(4 * np.pi * t) + 0.1 * np.sin(8 * np.pi * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Exact Coordinate Descent + Smoothed Gradient + LP Hybrid.
    Uses exact 1D coordinate optimization as the primary workhorse,
    with diverse initializations, multi-length exploration, and
    adaptive time allocation based on progress.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    def check_deadline():
        return time.monotonic() < deadline

    # === Phase 1: Quick multi-init exploration ===
    print(f"\n=== Phase 1: Multi-Init Exploration ===")
    inits_to_try = [
        ('raised_cosine', n_parent),
        ('hann', n_parent),
        ('bell', n_parent),
        ('sine', n_parent),
        ('plateau', n_parent),
        ('bipolar_smooth', n_parent),
        ('random', n_parent),
        ('uniform', n_parent + 200),
        ('raised_cosine', n_parent + 200),
    ]

    init_results = []
    for kind, n_try in inits_to_try:
        if not check_deadline():
            break
        a_init = make_init(n_try, kind, seed=int(rng.integers(0, 10000)))
        obj_init = compute_obj(a_init)
        init_results.append((obj_init, a_init.copy(), n_try, kind))
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  {kind} (n={n_try}): obj={obj_init:.8f} **")
        else:
            print(f"  {kind} (n={n_try}): obj={obj_init:.8f}")

    # Sort by objective and try top few with coordinate descent
    init_results.sort(key=lambda x: x[0])
    print(f"  Best init: obj={init_results[0][0]:.8f}")

    # === Phase 2: Coordinate descent on best initializations ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    top_inits = init_results[:4]
    # Include parent if it's good enough
    parent_obj = compute_obj(best_seq)
    if parent_obj < init_results[0][0] * 1.02:
        top_inits.append((parent_obj, best_seq.copy(), len(best_seq), 'parent'))

    for obj_init, a_init, n_try, kind in top_inits:
        if not check_deadline():
            break
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 120.0)
        if alloc < 20:
            break

        print(f"  CD on {kind} (n={n_try}), budget={alloc:.0f}s")
        a_cd = exact_coord_descent(a_init, max_cycles=100, time_limit=alloc, seed=int(rng.integers(0, 99999)))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"    ** {kind} CD: obj={best_score:.8f} **")
        else:
            print(f"    {kind} CD: obj={obj_cd:.8f}")

    # === Phase 3: Smoothed gradient + CD refinement ===
    print(f"\n=== Phase 3: Smoothed GD + CD ===")
    if check_deadline():
        remaining = deadline - time.monotonic()
        alloc_sg = min(remaining * 0.15, 100.0)
        alloc_cd = min(remaining * 0.15, 100.0)
        if alloc_sg > 15 and alloc_cd > 15:
            a_sg = smoothed_gradient_descent(best_seq, time_budget=alloc_sg,
                                             tau_init=10.0, tau_end=300.0, seed=123)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  SGD: obj={best_score:.8f} **")
            else:
                print(f"  SGD: obj={obj_sg:.8f}")

            if check_deadline():
                a_cd2 = exact_coord_descent(best_seq, max_cycles=80,
                                            time_limit=alloc_cd, seed=456)
                obj_cd2 = compute_obj(a_cd2)
                if obj_cd2 < best_score:
                    best_score = obj_cd2
                    best_seq = a_cd2.copy()
                    print(f"  CD after SGD: obj={best_score:.8f} **")
                else:
                    print(f"  CD after SGD: obj={obj_cd2:.8f}")

    # === Phase 4: LP Polish ===
    print(f"\n=== Phase 4: LP Polish ===")
    if check_deadline():
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.15, 120.0)
        if alloc_lp > 30:
            a_lp = lp_polish(best_seq, time_budget=alloc_lp, verbose=True, seed=789)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP: obj={best_score:.8f} **")
            else:
                print(f"  LP: obj={obj_lp:.8f}")

            # CD after LP
            if check_deadline():
                remaining2 = deadline - time.monotonic()
                alloc_cd3 = min(remaining2 * 0.2, 100.0)
                if alloc_cd3 > 20:
                    a_cd3 = exact_coord_descent(best_seq, max_cycles=80,
                                                time_limit=alloc_cd3, seed=101)
                    obj_cd3 = compute_obj(a_cd3)
                    if obj_cd3 < best_score:
                        best_score = obj_cd3
                        best_seq = a_cd3.copy()
                        print(f"  CD after LP: obj={best_score:.8f} **")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    n_current = len(best_seq)
    n_options = sorted(set([
        max(300, n_current - 200),
        max(300, n_current - 100),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        1000, 1200, 1500, 2000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500 and abs(n - n_current) > 50]

    for n_try in n_options[:6]:
        if not check_deadline():
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)

        # Quick SGD warmup
        alloc1 = min(remaining * 0.1, 25.0)
        if alloc1 > 8:
            a_try = smoothed_gradient_descent(a_try, time_budget=alloc1,
                                              tau_init=8.0, tau_end=200.0, seed=int(rng.integers(0, 999)))

        # Coordinate descent
        remaining2 = deadline - time.monotonic()
        alloc2 = min(remaining2 * 0.2, 80.0)
        if alloc2 > 15:
            a_try = exact_coord_descent(a_try, max_cycles=60,
                                        time_limit=alloc2, seed=int(rng.integers(0, 99999)))

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Perturbation + CD ===
    print(f"\n=== Phase 6: Perturbation + CD ===")
    for r_idx in range(8):
        if not check_deadline():
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        scale = 0.005 + 0.004 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        alloc = min(remaining * 0.15, 70.0)
        if alloc > 15:
            a_pert = exact_coord_descent(a_pert, max_cycles=60,
                                         time_limit=alloc, seed=500 + r_idx)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 7: Final intensive CD ===
    print(f"\n=== Phase 7: Final Polish ===")
    if check_deadline():
        remaining = deadline - time.monotonic()
        alloc_final = min(remaining * 0.5, 150.0)
        if alloc_final > 20:
            a_final = exact_coord_descent(best_seq, max_cycles=200,
                                          time_limit=alloc_final, verbose=True, seed=9999)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final CD: obj={best_score:.8f} **")
            else:
                print(f"  Final CD: obj={obj_final:.8f}")

        # Quick LP at the end
        if check_deadline():
            remaining2 = deadline - time.monotonic()
            alloc_lp2 = min(remaining2 * 0.3, 50.0)
            if alloc_lp2 > 20:
                a_lp2 = lp_polish(best_seq, time_budget=alloc_lp2, seed=2000)
                obj_lp2 = compute_obj(a_lp2)
                if obj_lp2 < best_score:
                    best_score = obj_lp2
                    best_seq = a_lp2.copy()
                    print(f"  Final LP: obj={best_score:.8f} **")

                # One more CD
                if check_deadline():
                    remaining3 = deadline - time.monotonic()
                    alloc_cd_final = min(remaining3 * 0.5, 40.0)
                    if alloc_cd_final > 10:
                        a_cdf = exact_coord_descent(best_seq, max_cycles=100,
                                                    time_limit=alloc_cd_final, seed=7777)
                        obj_cdf = compute_obj(a_cdf)
                        if obj_cdf < best_score:
                            best_score = obj_cdf
                            best_seq = a_cdf.copy()
                            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
