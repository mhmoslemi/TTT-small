"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid approach: spectral initialization, LP-based alternating minimization
with warm starts and line search, subgradient polishing, and multi-length exploration.
Key improvements over previous: better initializations, palindromic symmetry exploitation,
dual-based gradient computation, and more aggressive n exploration.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.MOSEK, cp.ECOS]:
        try:
            prob.solve(solver=solver, warm_start=(warm_start is not None))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_conv_lp_dual(a):
    """
    Solve the dual LP: max lambda s.t. A^T y >= lambda, sum(y)=1, y>=0.
    Returns (y_opt, lambda_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    m = 2 * n - 1  # number of rows of A

    y = cp.Variable(m, nonneg=True)
    lam = cp.Variable()
    constraints = [A.T @ y >= lam, cp.sum(y) == 1]
    prob = cp.Problem(cp.Maximize(lam), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.MOSEK, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if y.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                y_val = np.maximum(y.value, 0)
                s = float(np.sum(y_val))
                if s > 1e-10:
                    return y_val / s, float(lam.value)
        except Exception:
            continue
    return None, None


def spectrally_smoothed_grad(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of (a*a)_k.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / np.sum(w)


def nesterov_polish(a, time_budget, tau_init=50.0, tau_max=3000.0):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 1.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = project_simplex(y)

        g = spectrally_smoothed_grad(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adaptive_lr = 0.3 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.05 * base_lr), 2.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adaptive_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr *= 0.9999
        tau = min(tau * 1.01, tau_max)
        it += 1

        if no_improve > 2000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            no_improve = 0

    return best_a


def subgradient_polish(a, time_budget):
    """
    Polyak subgradient method on the exact (non-smoothed) max objective.
    Uses the known optimal value as target for step size.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    best_obj = compute_obj(a)

        max_val = float(np.max(conv))
        # Subgradient: 2 * a * 1{argmax} (averaged over all argmax positions)
        threshold = max_val - 1e-10
        active = np.where(conv >= threshold)[0]

            # (a*a)_k = sum_j a_j * a_{k-j}, gradient w.r.t. a_i is 2*(a_{k-i} + a_{i-k}) if in range
            # Actually gradient of (a*a)_k w.r.t. a_i is a_{k-i} + a_{i-k} (with appropriate bounds)
            # But we can compute it via the convolution matrix
            # Simpler: use the FFT-based approach
            pass
        
        # Actually, let's use a simpler approach: the subgradient of max_k (a*a)_k 
        # at a is 2 * (B^T p) where p is any convex combo of delta_{k} for active k
        # and B is the convolution matrix for a*a
        # For simplicity, use the first active index
        k_active = active[0]
        # Gradient of (a*a)_{k_active} w.r.t. a
        # (a*a)_k = sum_{j: 0<=j<n, 0<=k-j<n} a_j * a_{k-j}
        # d/d(a_i) = a_{k-i} (if 0<=k-i<n) + a_{i-k} (if 0<=i-k<n and i!=k-i)
        grad = np.zeros(n)
        for i in range(n):
            if 0 <= k_active - i < n:
                grad[i] += a[k_active - i]
            if 0 <= i - k_active < n and i != k_active - i:
                grad[i] += a[i - k_active]
        grad *= 2.0

        # Objective is 2*n*max(conv)/(sum(a))^2, on simplex sum=1 so it's 2*n*max(conv)
        # Subgradient of 2*n*max(conv) is 2*n*grad
        obj_grad = 2.0 * n * grad

        # Polyak step size
        gap = best_obj - target
        if gap < 1e-10:
            break
        g_norm_sq = float(np.dot(obj_grad, obj_grad))
        if g_norm_sq < 1e-30:
            break
        step = gap / g_norm_sq
        step = min(step, 1.0)

        a_new = a - step * obj_grad
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            target = max(best_obj * 0.999, target * 0.99)

    return best_a


def spectral_init(n, mode=0, rng=None):
    """Generate spectrally-structured initial sequences."""
    if rng is None:
        rng = np.random.default_rng()
    
    j = np.arange(n)
    
    if mode == 0:
        # Uniform
        a = np.ones(n)
    elif mode == 1:
        # Cosine window
        a = 0.5 + 0.5 * np.cos(2 * np.pi * j / n)
    elif mode == 2:
        # Sine squared
        a = 0.5 + 0.5 * np.sin(2 * np.pi * j / n) ** 2
    elif mode == 3:
        # Two-frequency
        f1, f2 = rng.integers(1, max(2, n // 4)), rng.integers(1, max(2, n // 3))
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * j / n) + 0.2 * np.cos(2 * np.pi * f2 * j / n)
    elif mode == 4:
        # Palindromic random
        half = n // 2
        a_half = rng.uniform(0.5, 1.5, half)
        a = np.concatenate([a_half, a_half[::-1]]) if n % 2 == 0 else np.concatenate([a_half, a_half[::-1][1:]])
    elif mode == 5:
        # Power-law
        a = (j + 1) ** rng.uniform(-0.5, 0.5)
    elif mode == 6:
        # Random with smoothing
        a = rng.uniform(0.5, 1.5, n)
        # Apply Gaussian smoothing
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=max(1, n // 20))
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a

def resample_to_length(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.001)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based alternating minimization with warm starts, multiple diverse 
initializations (spectral, palindromic, structured), subgradient polishing,
and multi-length exploration. Key improvements: faster convergence via gap-based
blending, dual LP verification, and more aggressive exploration of the n dimension.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    if warm_start is not None:
        g.value = warm_start.copy()
        M.value = float(np.max(compute_conv(a_norm * 0 + warm_start)))  # rough init

    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.MOSEK]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of (a*a)_k using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = float(np.sum(w))
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def nesterov_polish(a, time_budget, tau_init=30.0, tau_max=3000.0):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 1.5 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = project_simplex(y)

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adaptive_lr = 0.5 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.1 * base_lr), 3.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adaptive_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        tau = min(tau * 1.015, tau_max)
        it += 1

        if no_improve > 3000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            no_improve = 0

    return best_a


def subgradient_polish(a, time_budget):
    """
    Polyak subgradient method on the exact max objective.
    Uses the subgradient at the argmax of the convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.7
    base_lr = 1.0 / np.sqrt(n)
    lr = base_lr

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        threshold = max_val - 1e-12
        active = np.where(conv >= threshold)[0]

        # Compute subgradient: average of gradients at active positions
        # grad of (a*a)_k w.r.t. a_i = a_{k-i} + a_{i-k} (with bounds)
        subgrad = np.zeros(n)
        for k in active:
            for i in range(n):
                ki = k - i
                ik = i - k
                if 0 <= ki < n:
                    subgrad[i] += a[ki]
                if 0 <= ik < n and ik != ki:
                    subgrad[i] += a[ik]
        subgrad /= len(active)
        subgrad *= 2.0 * n  # scale by 2n for the objective 2n*max(conv)

        # Adaptive step size
        gap = best_obj - 1.0  # use 1.0 as lower bound aspiration
        gap = max(gap, 0.01)
        g_norm_sq = float(np.dot(subgrad, subgrad))
        if g_norm_sq < 1e-30:
            break
        step = min(gap / g_norm_sq, base_lr)

        momentum = beta * momentum + (1 - beta) * subgrad
        a_new = a - step * momentum
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        lr *= 0.999

    return best_a


def spectral_init(n, mode, rng):
    """Generate diverse initial sequences."""
    j = np.arange(n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * j / n)
    elif mode == 2:
        a = 0.5 + 0.3 * np.sin(2 * np.pi * j / n) ** 2
    elif mode == 3:
        f1 = rng.integers(1, max(2, n // 4))
        f2 = rng.integers(1, max(2, n // 3))
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * j / n + p1) + 0.3 * np.cos(2 * np.pi * f2 * j / n + p2)
    elif mode == 4:
        half = (n + 1) // 2
        a_half = rng.uniform(0.3, 1.5, half)
        if n % 2 == 0:
            a = np.concatenate([a_half, a_half[::-1]])
        else:
            a = np.concatenate([a_half, a_half[-2::-1]])
    elif mode == 5:
        # Triangle modified
        center = n / 2.0
        a = 1.0 - 0.5 * np.abs(j - center) / center
        a = a + 0.2 * np.cos(4 * np.pi * j / n)
    elif mode == 6:
        # Gaussian-like
        center = n / 2.0
        sigma = n / rng.uniform(3, 8)
        a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    elif mode == 7:
        # Multi-frequency
        a = np.ones(n)
        for _ in range(3):
            f = rng.integers(1, max(2, n // 5))
            amp = rng.uniform(0.1, 0.4)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * j / n + phase)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def lp_alternate_v2(a, time_budget, seed=42, verbose=True, use_palindrome=False):
    """
    LP-based alternating minimization with warm starts, gap-based blending,
    and adaptive perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    g_prev = None
    iter_count = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_conv_lp(a, warm_start=g_prev)
        lp_time = time.monotonic() - t0

        if g_lp is None:
            break

        if use_palindrome:
            g_lp = (g_lp + g_lp[::-1]) / 2.0
            g_lp = np.maximum(g_lp, 0)
            g_lp = g_lp / g_lp.sum()

        g_prev = g_lp.copy()
        iter_count += 1

        # Gap-based: the gap between current obj and LP lower bound tells us how far from fixed point
        # LP lower bound on max(a*g) is M_lp, so 2n*M_lp is a lower bound on objective
        lb = 2.0 * n * M_lp
        gap = best_obj - lb

        # Try multiple blending factors, biased by gap
        if gap > 0.01:
            candidates_t = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]
        elif gap > 0.001:
            candidates_t = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03]
        else:
            candidates_t = [0.3, 0.1, 0.05, 0.02, 0.01]

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = 0.5

        for t in candidates_t:
            a_try = (1.0 - t) * a + t * g_lp
            if use_palindrome:
                a_try = (a_try + a_try[::-1]) / 2.0
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall_count = 0
            if verbose and (iter_count % 25 == 0 or iter_count <= 5):
                print(f"    LP[{iter_count}]: obj={best_obj:.10f}, M_lp={M_lp:.6f}, "
                      f"gap={gap:.6f}, t={best_local_t:.3f}, time={lp_time:.3f}s")
        else:
            stall_count += 1
            if stall_count >= 8:
                # Structured perturbation
                strength = 0.003 + 0.002 * min(stall_count, 20)
                n_modes = 7
                pert = np.zeros(n)
                for k in range(1, n_modes + 1):
                    freq = k * 2 * np.pi / n
                    amp = strength * rng.uniform(0.3, 1.0) / np.sqrt(k)
                    phase = rng.uniform(0, 2 * np.pi)
                    pert += amp * np.cos(freq * np.arange(n) + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                if use_palindrome:
                    a = (a + a[::-1]) / 2.0
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                stall_count = 0
            elif stall_count >= 3:
                a = best_a + 0.002 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                if use_palindrome:
                    a = (a + a[::-1]) / 2.0
                    a = np.maximum(a, 0)
                a = a / a.sum()

    if verbose:
        print(f"    LP done: {iter_count} iters, best_obj={best_obj:.10f}")
    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.001)

            best_a = a_p4.copy()
            print(f"  ** Nesterov: obj={best_obj:.10f}")
        else:
            print(f"  Nesterov: obj={obj_p4:.10f}")

    # === Phase 5: Subgradient polish ===
    print(f"\n=== Phase 5: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        p5_budget = min(remaining * 0.15, 80)
        a_p5 = subgradient_polish(best_a, time_budget=p5_budget)
        obj_p5 = compute_obj(a_p5)
        if obj_p5 < best_obj:
            best_obj = obj_p5
            best_a = a_p5.copy()
            print(f"  ** Subgrad: obj={best_obj:.10f}")
        else:
            print(f"  Subgrad: obj={obj_p5:.10f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    # Try nearby and non-local n values
    n_options = [n_best - 50, n_best - 100, n_best + 50, n_best + 100,
                 400, 600, 1000, 1200]
    n_options = [n for n in n_options if 300 <= n <= 2000 and n != n_best]
    n_options = sorted(set(n_options))

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(remaining * 0.20, 120)
        if alloc < 25:
            break

        a_try = resample_to_length(best_a, n_try)
        # Quick LP
        a_try = lp_alternate_v2(a_try, time_budget=alloc * 0.6, seed=n_try * 3, verbose=False)

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_try = nesterov_polish(a_try, time_budget=min(remaining * 0.25, 30),
                                     tau_init=50.0, tau_max=1500.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            best_n = n_try
            print(f"  ** n={n_try}: obj={best_obj:.10f}")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 7: Final LP + polish ===
    print(f"\n=== Phase 7: Final Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.45, 150)
        a_final = lp_alternate_v2(best_a, time_budget=alloc, seed=seed + 99999, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_final = nesterov_polish(best_a, time_budget=min(remaining * 0.5, 50),
                                       tau_init=100.0, tau_max=5000.0)
            obj_final = compute_obj(a_final)
            if obj_final < best_obj:
                best_obj = obj_final
                best_a = a_final.copy()
                print(f"  ** Final: obj={best_obj:.10f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
