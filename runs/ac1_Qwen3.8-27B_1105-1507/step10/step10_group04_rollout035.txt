def kkt_polish(a, tol=1e-8):
    """Polish using KKT conditions. Find dual weights and adjust a."""
    n = len(a)
    a = a / a.sum()
    conv = compute_conv(a)
    max_conv = conv.max()
    
    # Active set: positions where conv is close to max
    active = np.where(conv >= max_conv - tol * max_conv)[0]
    
    # Support of a
    support = np.where(a > 1e-10)[0]
    
    if len(active) < 2 or len(support) < 2:
        return a
    
    # Build system: sum_k w_k * a_{k-j} = lambda for j in support
    # This is: W^T @ A_sub = lambda * 1
    # where W^T has rows indexed by j in support, columns by k in active
    # A_sub[j_idx, k_idx] = a_{active[k_idx] - support[j_idx]} if 0 <= active[k_idx]-support[j_idx] < n else 0
    
    # Actually, the gradient of (a*a)[k] w.r.t. a_j is a_{k-j} (if 0 <= k-j < n)
    # So we need: sum_{k in active} w_k * a_{k-j} = lambda for all j in support
    
    m = len(support)
    K = len(active)
    
    M = np.zeros((m, K))
    for i, j in enumerate(support):
        for k_idx, k in enumerate(active):
            idx = k - j
            if 0 <= idx < n:
                M[i, k_idx] = a[idx]
    
    # Solve: M @ w = lambda * 1, w >= 0, sum(w) = 1 (normalize)
    # This is: find w >= 0 such that M@w is as constant as possible
    # Minimize ||M@w - mean(M@w)*1||^2 subject to w >= 0, sum(w) = 1
    
    # This is a QP. Let's solve it with scipy.
    from scipy.optimize import minimize
    
    w0 = np.ones(K) / K
    
    def qpf(w):
        Mw = M @ w
        return np.sum((Mw - Mw.mean())**2)
    
    def qpf_grad(w):
        Mw = M @ w
        residual = Mw - Mw.mean()
        # d/dw [sum((Mw - mean)^2)] = 2*M^T*(residual - mean(residual)*1)
        # But mean(residual) = (sum residual)/m = (M@w).sum()/m - (M@w).sum()/m = 0
        # Wait: mean(Mw) = sum(Mw)/m, so residual_i = Mw_i - sum(Mw)/m
        # d residual_i / dw = M_i - M.sum(1)/m
        # grad = 2 * M^T @ (residual - residual.mean()*1)
        # But residual.mean() = (sum Mw - m*mean(Mw))/m = 0
        # So grad = 2 * M^T @ residual
        return 2 * M.T @ residual
    
    res = minimize(qpf, w0, jac=qpf_grad, method='L-BFGS-B',

    if res.x is not None:

        w = w / w.sum()
        
        # Check how well the KKT conditions are satisfied
        Mw = M @ w
        kkt_residual = np.max(np.abs(Mw - Mw.mean()))
        
        if kkt_residual < 1e-10:
            print(f"    KKT satisfied! residual={kkt_residual:.2e}")
            return a
    
    return a

def fast_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = fast_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def variance_obj(a):
    """Compute variance of convolution (smooth objective for pre-conditioning)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = fast_conv(a_norm)
    return float(np.var(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for i in range(n):
            A[j + i, j + i] = 0  # will handle differently

    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    return A

Key innovations over previous:

- Variance pre-conditioning phase to reach good basin before max-minimization
- Geometric tau scheduling with annealing for escaping local minima

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import cvxpy as cp
import time


def fast_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_value(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = fast_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def var_value(a):
    """Compute variance of normalized convolution."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = fast_conv(a_norm)
    return float(np.var(conv))


def smoothed_max_and_grad(a, tau):
    """
    Compute smoothed max of conv(a,a) and its gradient.
    Returns (smoothed_value, gradient_array of length n).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv = fast_conv(a_norm)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Numerical stability
    max_shifted = float(np.max(shifted))
    exp_vals = np.exp(shifted - max_shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    obj_smooth = max_conv + np.log(sum_exp) / tau

    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j] (where a[k-j]=0 if out of range)
    # = 2 * cross_correlation computation via FFT
    a_rev = a_norm[::-1]
    N_conv_total = n + (2 * n - 1) - 1  # 3n-2
    N_fft = 1
    while N_fft < N_conv_total:
        N_fft <<= 1
    F_a_rev = fft(a_rev.astype(np.complex128), N_fft)
    F_p = fft(p.astype(np.complex128), N_fft)
    conv_res = ifft(F_a_rev * F_p).real[:N_conv_total]
    c = conv_res[n - 1:2 * n - 1]
    grad = 2.0 * c

    return obj_smooth, grad


def subgradient_max(a):
    """Compute subgradient at the (smoothed) argmax positions of conv(a,a).
    Returns (conv_value, gradient using only top entries)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv = fast_conv(a_norm)
    max_conv = float(np.max(conv))

    # Use top entries (within 1% of max) for a softer subgradient
    threshold = max_conv - 0.01 * max_conv
    active = conv >= threshold
    p = active.astype(np.float64) / active.sum()

    a_rev = a_norm[::-1]
    N_conv_total = n + (2 * n - 1) - 1
    N_fft = 1
    while N_fft < N_conv_total:
        N_fft <<= 1
    F_a_rev = fft(a_rev.astype(np.complex128), N_fft)
    F_p = fft(p.astype(np.complex128), N_fft)
    conv_res = ifft(F_a_rev * F_p).real[:N_conv_total]
    c = conv_res[n - 1:2 * n - 1]
    grad = 2.0 * c

    return max_conv, grad

        residual = conv - mean_c

        # Gradient of variance: d/d a_j [sum_k (conv[k]-mean)^2]
        # = 2 * sum_k residual[k] * d(conv[k])/d(a_j)
        # = 4 * sum_k residual[k] * a_norm[k-j] (for the conv part, ignoring mean derivative)
        # More carefully: var = E[x^2] - E[x]^2
        # d var / d a_j = 2 * sum_k (conv[k] - mean) * d(conv[k])/d(a_j) / (something)
        # For simplicity, use: gradient of sum_k (conv[k] - mean)^2
        # = 2 * sum_k (conv[k] - mean) * 2 * a_norm[k-j]
        # But we also need to account for d(mean)/d(a_j)
        # d(mean)/d(a_j) = 2 * sum_k a_norm[k-j] / (2n-1) = 2 * sum_k a_norm[k] / (2n-1) * ... 
        # This gets complicated. Let's just use the gradient of sum_k conv[k]^2 (which is simpler)
        # and let the max-constraint handle the mean.
        # 
        # Actually for variance minimization, let's just minimize sum_k (conv[k] - target)^2
        # where target = mean(conv). The gradient is:
        # grad_j = 2 * sum_k 2*(conv[k]-mean) * a[k-j] - 2*mean * ... (mean derivative)
        # 
        # Simplification: just minimize sum_k conv[k]^2 which is the L2 norm squared.
        # This is equivalent to minimizing the energy of the convolution, which by Parseval
        # equals (1/N) * sum |FFT(a)|^4. This encourages flat spectrum.
        
        # Gradient of sum_k conv[k]^2 w.r.t. a_j:
        # = 2 * sum_k conv[k] * 2 * a_norm[k-j]
        # = 4 * (conv * a_norm reversed)[j] ... 
        # Actually: d(conv[k])/d(a_j) = 2*a_norm[k-j]
        # d(sum_k conv[k]^2)/d(a_j) = 2 * sum_k conv[k] * 2*a_norm[k-j]
        # = 4 * sum_k conv[k] * a_norm[k-j]
        # = 4 * cross_correlation(conv, a_norm)[j]

        conv_full = conv  # length 2n-1
        a_rev = a_norm[::-1]
        N_tot = n + len(conv_full) - 1
        N_f = 1
        while N_f < N_tot:
            N_f <<= 1
        F_conv = fft(conv_full.astype(np.complex128), N_f)
        F_arev = fft(a_rev.astype(np.complex128), N_f)
        c_res = ifft(F_conv * F_arev).real[:N_tot]
        grad = 4.0 * c_res[n - 1:2 * n - 1]

        # ADAM update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr_t = lr / (1 + 0.1 * np.log1p(t))

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = obj_value(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"      VarOpt t={t}: obj={best_obj:.8f}, var={var_value(a):.8f}, t={elapsed:.0f}s")

        if t > 100000:
            break

    return best_a


def adam_optimize_max(a, time_budget, tau_start=50, tau_end=20000,
                      lr=0.5, anneal_every=3000, anneal_mag=0.03, verbose=True):
    """
    ADAM optimization on smoothed max of convolution.
    Uses geometric tau scheduling and periodic annealing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_value(a)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    t = 0
    start = time.monotonic()
    last_anneal = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1))

        # Geometric tau schedule
        tau = tau_start * (tau_end / tau_start) ** progress

        # Compute smoothed objective and gradient
        obj_s, grad = smoothed_max_and_grad(a, tau)

        # ADAM update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # Adaptive learning rate: decay over time, also based on gradient norm
        g_norm = float(np.linalg.norm(grad))
        lr_t = lr / (np.sqrt(t) + 1.0)
        lr_t = min(lr_t, 1.0 / (g_norm + 1e-6))
        lr_t = max(lr_t, 1e-6)

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = obj_value(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        # Annealing: periodically add noise to escape local minima
        if t - last_anneal >= anneal_every and stall_count > 500:
            noise = np.random.randn(n) * anneal_mag
            a_anneal = a + noise
            a_anneal = np.maximum(a_anneal, 0)
            s_anneal = float(np.sum(a_anneal))
            if s_anneal > 1e-10:
                a = a_anneal / s_anneal
            last_anneal = t
            stall_count = 0
            m = np.zeros(n)
            v = np.zeros(n)

        # Reset if stuck too long
        if stall_count > 5000:
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            stal_count = 0
            if t > 200000:
                break

        if verbose and t % 10000 == 0:
            print(f"      MaxOpt t={t}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr_t:.6f}, t={elapsed:.0f}s")

    return best_a


def solve_lp(a, timeout=30):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))

    # Build constraint matrix A where (A @ g)[k] = conv(a, g)[k]
    # A[k, j] = a[k-j] if 0 <= k-j < n
    # For memory efficiency with large n, use a structured approach
    if n > 3000:
        return None, None

    A_rows = 2 * n - 1
    A = np.zeros((A_rows, n))
    for j in range(n):
        A[j:j + n, j] = a_norm

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.MOSEK]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(max(M.value, 0))
                    return g_val, M_val
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget, verbose=True):
    """LP polishing with mixing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_value(a)
    start = time.monotonic()
    it = 0
    last_mix = 0.5

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 5:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        it += 1

        # Try multiple mixing parameters
        t_candidates = np.concatenate([
            np.linspace(0.05, 1.0, 10),
            [max(0.02, last_mix * 0.5), max(0.02, last_mix * 0.8),
             last_mix, min(1.0, last_mix * 1.2), min(1.0, last_mix * 1.5)]
        ])
        t_candidates = np.unique(np.clip(t_candidates, 0.01, 1.0))

        best_t_obj = best_obj
        best_t = last_mix
        best_local = best_a.copy()

        for tt in t_candidates:
            a_try = (1.0 - tt) * a + tt * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = obj_value(a_try)
            if obj_try < best_t_obj - 1e-13:
                best_t_obj = obj_try
                best_t = tt
                best_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_local.copy()
            a = best_a
            last_mix = best_t
            if verbose:
                print(f"      LP it{it}: obj={best_obj:.8f}, mix={best_t:.3f}, t={time.monotonic()-start:.0f}s")
        else:
            if it > 5:
                break

    return best_a


def kkt_polish(a, verbose=True):
    """
    KKT-based polishing: find dual weights on the active set that make
    the gradient constant on the support, indicating KKT satisfaction.
    If not satisfied, use the violation to adjust a.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = fast_conv(a)
    max_conv = float(np.max(conv))

    # Active set: positions where conv is within 0.5% of max
    tol = 0.005 * max_conv
    active_mask = conv >= max_conv - tol
    active_idx = np.where(active_mask)[0]

    # Support of a
    support_thresh = max(1e-8, 1e-6 * a.max())
    support = np.where(a > support_thresh)[0]

    if len(active_idx) < 2 or len(support) < 2:
        return a

    # Build system: find w on active_idx such that sum_k w_k * a[k-j] is constant for j in support
    # This is the dual of the LP, restricted to the active set
    m = len(support)
    K = len(active_idx)

    # M[i, k_idx] = a[active_idx[k_idx] - support[i]] if 0 <= active_idx[k_idx] - support[i] < n
    M_mat = np.zeros((m, K))
    for i, j in enumerate(support):
        for k_idx, k in enumerate(active_idx):
            idx = k - j
            if 0 <= idx < n:
                M_mat[i, k_idx] = a[idx]

    # Solve: minimize ||M @ w - mean(M@w)*1||^2 s.t. w >= 0, sum(w) = 1
    w0 = np.ones(K) / K

    def kkt_obj_w(w_vec):
        Mw = M_mat @ w_vec
        return float(np.sum((Mw - Mw.mean()) ** 2))

    def kkt_grad_w(w_vec):
        Mw = M_mat @ w_vec
        res = Mw - Mw.mean()
        return 2.0 * M_mat.T @ res

    constraints = [
        {'type': 'eq', 'fun': lambda w_vec: float(w_vec.sum()) - 1.0,
         'jac': lambda w_vec: np.ones(K)}
    ]
    bounds = [(0.0, None)] * K

    try:
        res = scipy_minimize(kkt_obj_w, w0, jac=kkt_grad_w, method='SLSQP',
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 500, 'ftol': 1e-16})
        if res.x is not None:
            w = np.maximum(res.x, 0)
            w = w / w.sum()
            Mw = M_mat @ w
            kkt_res = float(np.max(np.abs(Mw - Mw.mean())))

            if verbose:
                print(f"      KKT: residual={kkt_res:.2e}, active={K}, support={m}")

            # If KKT is nearly satisfied, we're at optimum
            if kkt_res < 1e-8:
                return a

            # Use the violation to adjust a
            # The gradient of the Lagrangian should be zero on the support
            # grad_j = 2 * sum_k w_k * a[k-j] - lambda = Mw[j_idx] - mean(Mw)
            # If Mw[j] > mean, we should decrease a[j]; if Mw[j] < mean, increase a[j]
            # This gives a direction for updating a
            adjustment = Mw - Mw.mean()
            # Map back to full support
            adj_full = np.zeros(n)
            adj_full[support] = -adjustment  # negative because we want to reduce the violation

            # Small step
            step = 0.1
            a_new = a + step * adj_full * a  # multiplicative update to preserve non-negativity
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = obj_value(a_new)
                obj_old = obj_value(a)
                if obj_new < obj_old:
                    return a_new
    except Exception:
        pass

    return a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # U-shaped (high at edges, low in middle)
        x = np.linspace(0, 1, n)
        a = (x - 0.5) ** 2 + 0.1
        a = a / a.sum()
    elif mode == 3:
        # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.25) ** 2) / 0.02) + np.exp(-((x - 0.75) ** 2) / 0.02) + 0.01
        a = a / a.sum()
    elif mode == 4:
        # Asymmetric ramp
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 5:
        # Asymmetric U-shape
        x = np.linspace(0, 1, n)
        a = x ** 2 + 0.05
        a = a / a.sum()
    elif mode == 6:
        # Sinusoidal modulation
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 7:
        # Two-scale structure
        x = np.linspace(0, 1, n)
        coarse = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        fine = 0.15 * np.cos(2 * np.pi * 20 * x)
        a = coarse + fine + 0.3
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses ADAM optimization with variance pre-conditioning,
    smoothed max minimization with annealing, LP polishing, KKT refinement,
    and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0  # safety margin

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = obj_value(best_seq)
                print(f"Loaded parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        # Try multiple initializations and pick the best
        n_init = 800
        best_init = None
        best_init_score = np.inf
        for mode in range(8):
            a_try = diverse_init(n_init, rng, mode=mode)
            s_try = obj_value(a_try)
            if s_try < best_init_score:
                best_init_score = s_try
                best_init = a_try
        best_seq = best_init
        best_score = best_init_score
        print(f"Best init (n={n_init}): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting with n={n_best}, obj={best_score:.8f}")

    # === Phase 1: Variance Pre-conditioning (150s) ===
    print(f"\n=== Phase 1: Variance Pre-conditioning ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.15, 150.0)
    if p1_budget > 20:
        a_var = adam_optimize_var(best_seq, time_budget=p1_budget, lr=0.05, verbose=True)
        obj_var = obj_value(a_var)
        if obj_var < best_score:
            best_score = obj_var
            best_seq = a_var.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_var:.8f} (no improvement)")

    # === Phase 2: ADAM Max Minimization (350s) ===
    print(f"\n=== Phase 2: ADAM Max Minimization ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.28, 350.0)
    if p2_budget > 30:
        a_max = adam_optimize_max(best_seq, time_budget=p2_budget,
                                  tau_start=30, tau_end=15000,
                                  lr=0.8, anneal_every=5000, anneal_mag=0.02,
                                  verbose=True)
        obj_max = obj_value(a_max)
        if obj_max < best_score:
            best_score = obj_max
            best_seq = a_max.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj_max:.8f}")

    # === Phase 3: LP Polishing (200s) ===
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.18, 200.0)
    if p3_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p3_budget, verbose=True)
        obj_lp = obj_value(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj_lp:.8f}")

    # === Phase 4: KKT Polishing ===
    print(f"\n=== Phase 4: KKT Polishing ===")
    a_kkt = kkt_polish(best_seq, verbose=True)
    obj_kkt = obj_value(a_kkt)
    if obj_kkt < best_score:
        best_score = obj_kkt
        best_seq = a_kkt.copy()
        print(f"  Phase4: obj={best_score:.8f} **")

    # === Phase 5: Multi-length Exploration (300s) ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.25, 300.0)
    n_current = len(best_seq)

    n_options = sorted(set([
        max(200, n_current - 200),
        max(200, n_current - 100),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        n_current + 800,
        n_current + 1500,
        400, 600, 1000, 1200, 1600, 2500, 3000
    ]))
    n_options = [n for n in n_options if 200 <= n <= 3500 and n != n_current]
    n_options = n_options[:10]
    print(f"  Trying n values: {n_options}")

    per_n_budget = p5_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(per_n_budget, remaining - 40)
        if alloc < 30:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick ADAM optimization
        adam_alloc = alloc * 0.5
        if adam_alloc > 10:
            a_try = adam_optimize_max(a_try, time_budget=adam_alloc,
                                      tau_start=20, tau_end=8000,
                                      lr=0.5, anneal_every=3000, anneal_mag=0.02,
                                      verbose=False)

        # LP polish if n is moderate
        remaining = deadline - time.monotonic()
        lp_alloc = min(alloc * 0.3, remaining - 20)
        if lp_alloc > 15 and n_try <= 2000:
            a_try = lp_polish(a_try, time_budget=lp_alloc, verbose=False)

        # KKT polish
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_try = kkt_polish(a_try, verbose=False)

        obj_try = obj_value(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final Intensive Polish (remaining time) ===
    print(f"\n=== Phase 6: Final Intensive Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s")
    n_final = len(best_seq)

    # Cycle through different optimizers
    if remaining > 30:
        alloc = min(remaining * 0.25, 120.0)
        a_f = adam_optimize_max(best_seq, time_budget=alloc,
                                tau_start=100, tau_end=30000,
                                lr=0.3, anneal_every=2000, anneal_mag=0.01,
                                verbose=True)
        obj_f = obj_value(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final ADAM: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 25 and n_final <= 2000:
        alloc = min(remaining * 0.25, 100.0)
        a_f = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_f = obj_value(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_f = kkt_polish(best_seq, verbose=True)
        obj_f = obj_value(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final KKT: obj={best_score:.8f} **")

    # Another ADAM round with very high tau
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 80.0)
        a_f = adam_optimize_max(best_seq, time_budget=alloc,
                                tau_start=500, tau_end=50000,
                                lr=0.1, anneal_every=1000, anneal_mag=0.005,
                                verbose=False)
        obj_f = obj_value(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Ultra ADAM: obj={best_score:.8f} **")

    # Final KKT
    a_f = kkt_polish(best_seq, verbose=False)
    obj_f = obj_value(a_f)
    if obj_f < best_score:
        best_score = obj_f
        best_seq = a_f.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = obj_value(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
