"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Coarse-to-fine search over (n, initialization) space, then focused
adaptive-tau gradient descent with backtracking line search, LP polishing,
and Newton-type refinement. Uses raised-cosine and structured initializations
exploiting the flat-top nature of the optimal solution.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm
    return A


def hann_init(n):
    """Raised cosine (Hann window) initialization."""
    if n <= 1:
        return np.ones(max(n, 1))
    j = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * j / (n - 1)))
    a = a / a.sum()
    return a


def gaussian_init(n, sigma_frac=0.25):
    """Gaussian bell initialization."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((j - center) / (sigma_frac * n)) ** 2)
    a = a / a.sum()
    return a


def super_gaussian_init(n, power=4):
    """Super-Gaussian (flat-top) initialization."""
    j = np.arange(n, dtype=np.float64)
    x = 2.0 * j / (n - 1) - 1.0 if n > 1 else np.zeros(1)
    a = np.cos(np.pi * x / 2.0) ** power
    a = np.maximum(a, 0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    a = a / a.sum()
    return a


def triangular_init(n):
    """Triangular (Bartlett) window initialization."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 - np.abs(j - center) / center if center > 0 else np.ones(n)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def adaptive_tau_gd(a, time_budget, tau_start=30, tau_end=30000, verbose=False, label=""):
    """Adaptive tau gradient descent with backtracking line search and Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    v = np.zeros(n)
    it = 0
    best_it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Adaptive tau: gradual then aggressive increase
        if progress < 0.3:
            tau = tau_start + (100.0 - tau_start) * (progress / 0.3)
        else:
            tau = 100.0 + (tau_end - 100.0) * ((progress - 0.3) / 0.7) ** 1.5

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-15:
            y = best_a.copy()
            v = np.zeros(n)
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        if max_conv < 1e-30:
            break

        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient via FFT: grad_j = 2 * sum_k p[k] * y[k-j]
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(y, N)
        g_full = ifft(P * np.conj(A_f)).real
        g = 2.0 * g_full[:n]

        # Project to tangent space of simplex
        g = g - np.mean(g)

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Backtracking line search
        lr = min(1.0 / (g_norm + 1e-10), 2.0 / np.sqrt(n))
        found = False
        t_used = 0.0
        best_trial_obj = best_obj
        best_trial_a = None

        for bt in range(40):
            t = lr * (0.5 ** bt)
            a_new = y - t * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)

            if obj_new <= best_obj - 1e-15 * t * g_norm + 1e-16:
                found = True
                t_used = t
                best_trial_obj = obj_new
                best_trial_a = a_new.copy()
                break
            elif obj_new < best_trial_obj:
                best_trial_obj = obj_new
                best_trial_a = a_new.copy()
                t_used = t

        if best_trial_a is None:
            t = lr * (0.5 ** 39)
            a_new = y - t * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-15:
                a_new = best_a.copy()
                t_used = 0.0
                best_trial_obj = best_obj
            else:
                a_new = a_new / s_new
                best_trial_obj = compute_obj(a_new)
                t_used = t
            best_trial_a = a_new.copy()

        v = beta * v - t_used * g
        a = best_trial_a

        if best_trial_obj < best_obj - 1e-14:
            best_obj = best_trial_obj
            best_a = a.copy()
            best_it = it
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive momentum
        if no_improve > 50:
            beta = max(0.5, beta - 0.02)
            if no_improve > 200:
                v = np.zeros(n) * 0.5
        else:
            beta = min(0.95, beta + 0.002)

        it += 1

        # Staleness restart
        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            beta = 0.8
            if it - best_it > 30000:
                break

        if verbose and it % 1000 == 0:
            print(f"    {label} it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.0f}s")

    return best_a


def lp_polish(a, time_budget, verbose=False, max_iters=50):
    """LP-based minimax polishing with line search mixing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for iteration in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Solve LP: min M s.t. conv(a, g)[k] <= M, sum(g)=1, g>=0
        a_norm = a / np.sum(a)
        A_mat = build_conv_matrix(a_norm)

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A_mat @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        g_val = None
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                    break
            except Exception:
                continue

        if g_val is None:
            break

        # Line search over mixing parameter
        t_candidates = np.concatenate([
            np.linspace(0.05, 1.0, 15),
            [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
        ])
        t_candidates = np.unique(np.clip(t_candidates, 0.02, 1.0))

        best_t_obj = best_obj
        best_t = 0.0
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-13:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            if verbose:
                print(f"    LP iter{iteration}: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 4:
                break

    return best_a


def log_param_optimize(a, time_budget, tau_min=200, tau_max=5000, verbose=False):

    for stage_idx, tau in enumerate(tau_stages):

            a_supp = np.exp(b_shifted)

            a_full = np.zeros(full_len)
            a_full[support] = a_supp
            conv = compute_conv(a_full)

        def obj_grad_fn(b_vec):

            conv = compute_conv(a_full)

                a_new = np.zeros(full_len)
                a_new[support] = a_supp

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                        best_seq = a_res.copy()
                        print(f"  Resample n={new_n}: ** obj={best_score:.8f} **")
                    else:
                        print(f"  Resample n={new_n}: obj={obj_res:.8f}")
        except Exception:
            pass

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s")

    if best_seq is None:
        best_seq = hann_init(800)
        best_score = compute_obj(best_seq)

    # Round 1: Ultra-high tau GD
    if remaining > 60:
        alloc = min(remaining * 0.35, 200.0)
        a_final = adaptive_tau_gd(best_seq, time_budget=alloc,
                                   tau_start=500, tau_end=50000, verbose=True, label="Polish-GD")
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra GD: ** obj={best_score:.8f} **")

    # Round 2: LP
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.3, 120.0)
        a_final = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: ** obj={best_score:.8f} **")

    # Round 3: Log-param
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 80.0)
        a_final = log_param_optimize(best_seq, time_budget=alloc,
                                      tau_min=500, tau_max=10000, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Log: ** obj={best_score:.8f} **")

    # Round 4: One more ultra-high tau GD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 60.0)
        a_final = adaptive_tau_gd(best_seq, time_budget=alloc,
                                   tau_start=2000, tau_end=100000, verbose=False, label="Final-GD")
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Ultra GD: ** obj={best_score:.8f} **")

    # Round 5: Last LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 40.0)
        a_final = lp_polish(best_seq, time_budget=alloc, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra-final LP: ** obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
