- f'_j = 4n · [(p⊛a)_j · s - g] / s^3  (with s = 1 if normalized)
- Actually with the log-param, s = 1 always, so:
- f'_j = 4n · [(p⊛a)_j - g] (with g = log_sum_exp value)

Then:
∂f/∂x_i = f'_i · a_i - a_i · Σ_j f'_j · a_j = a_i · (f'_i - Σ_j f'_j · a_j)

Let mean_f = Σ_j f'_j · a_j (a weighted average of the gradient components)
Then ∂f/∂x_i = a_i · (f'_i - mean_f)

This is the gradient of f with respect to the log-parameters.

Now I can use L-BFGS-B or even plain BFGS on this unconstrained problem!

```python
def smooth_obj_and_grad_x(x, tau):
    """
    Compute smoothed objective and gradient w.r.t. log-parameters x.
    a = exp(x) / sum(exp(x))
    """
    # Numerically stable softmax
    x_shifted = x - np.max(x)
    a = np.exp(x_shifted)
    a /= np.sum(a)
    
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    g = max_conv + (1.0/tau) * np.log(sum_exp)
    f = 2.0 * n * g  # since sum(a) = 1
    
    # Weights
    p = exp_shifted / sum_exp
    
    # Cross-correlation (p ⊛ a)
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real[:n]
    
    # Gradient w.r.t. a (with s=1): f'_j = 4n * (cross_j - g)
    f_da = 4.0 * n * (cross - g)
    
    # Gradient w.r.t. x: dx_i = a_i * (f_da_i - mean(f_da * a))
    mean_fda = np.sum(f_da * a)
    grad_x = a * (f_da - mean_fda)
    
    return f, grad_x
```

This is clean and efficient. Each evaluation is O(n log n) due to FFTs.

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-parameterized smoothed optimization with adaptive tau scheduling,
L-BFGS-B inner solver, subgradient averaging, and multi-length exploration.
Key innovations: log-space parameterization for constraint-free optimization,
FFT-based gradient computation, adaptive smoothing schedule, and
subgradient Polyak averaging for the non-smooth objective.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_obj_grad_x(x, tau):
    """
    Compute smoothed objective and gradient w.r.t. log-parameters x.
    a = softmax(x), so sum(a)=1 and a>0 automatically.
    """
    x_shifted = x - np.max(x)
    a = np.exp(x_shifted)
    s = np.sum(a)
    a = a / s
    
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    
    if sum_exp < 1e-300:
        # All mass on the max
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
        g = max_conv
    else:
        p = exp_shifted / sum_exp
        g = max_conv + (1.0 / tau) * np.log(sum_exp)
    
    f = 2.0 * n * g
    
    # Cross-correlation: cross[j] = sum_k p[k] * a[k-j] (valid indices only)
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real[:n]
    
    # Gradient w.r.t. a: f'_j = 4n * (cross_j - g) [since s=1]
    f_da = 4.0 * n * (cross - g)
    
    # Gradient w.r.t. x: a_i * (f_da_i - sum(f_da * a))
    mean_fda = float(np.dot(f_da, a))
    grad_x = a * (f_da - mean_fda)
    
    return f, grad_x


def smoothed_lbfgs(a_init, tau_schedule, time_budget, verbose=False):
    """
    Run L-BFGS-B on the smoothed objective with a given tau schedule.
    tau_schedule is a list of (tau, max_iter) pairs.
    """
    n = len(a_init)
    x = np.log(a_init + 1e-30)  # Initialize log-parameters
    
    best_a = a_init.copy()
    best_f = compute_obj(a_init)
    start = time.monotonic()
    
    for tau, max_iter in tau_schedule:
        elapsed = time.monotonic() - start
        if elapsed >= time_budget:
            break
        
        def func_and_grad(x):
            f, g = smooth_obj_grad_x(x, tau)
            return f, g
        
        try:
            result = minimize(func_and_grad, x, method='L-BFGS-B',
                             jac=True, options={'maxiter': max_iter, 'gtol': 1e-10,
                                                'maxls': 50})
            if result.success or result.fun < best_f:
                x = np.maximum(x, result.x)  # Keep x valid
                a = np.exp(x - np.max(x))
                a = a / a.sum()
                f = compute_obj(a)
                if f < best_f:
                    best_f = f
                    best_a = a.copy()
                    if verbose:
                        print(f"    tau={tau}: obj={best_f:.8f} (iter={result.nit})")
        except Exception as e:
            if verbose:
                print(f"    tau={tau}: error {e}")
    
    return best_a, best_f


def subgradient_averaging(a_init, time_budget, verbose=False, seed=0):
    """
    Subgradient method with Polyak averaging for the non-smooth objective.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    a_sum = np.zeros(n)
    count = 0
    step_scale = 1.0 / np.sqrt(n)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        k_star = int(np.argmax(conv))
        # Random tie-breaking if multiple argmax
        max_indices = np.where(conv >= max_val - 1e-12)[0]
        if len(max_indices) > 1:
            k_star = max_indices[rng.integers(len(max_indices))]
        
        # Subgradient: 4n * (a[k_star - j] * [0<=k_star-j<n] - max_val)
        sg = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(2*n-1, k_star + n - 1)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx <= 2*n-2:
                sg[j] = 4.0 * n * (a[idx] - max_val)
        
        # Step size (diminishing)
        t = step_scale / np.sqrt(count + 1)
        a_new = a - t * sg
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s
        
        a_sum += a_new
        count += 1
        a = a_new
        
        # Check average
        if count % 100 == 0:
            a_avg = a_sum / count
            obj_avg = compute_obj(a_avg)
            if obj_avg < best_obj:
                best_obj = obj_avg
                best_a = a_avg.copy()
                if verbose:
                    print(f"    SG it{count}: avg_obj={best_obj:.8f}")
            
            # Also check current
            obj_cur = compute_obj(a)
            if obj_cur < best_obj:
                best_obj = obj_cur
                best_a = a.copy()
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def cosine_init(n, k=1):
    """Initialize with a cosine-like asymmetric shape."""
    x = np.linspace(0, 1, n)
    a = 1.0 + 0.9 * np.cos(2 * np.pi * k * x)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def random_init(n, seed=0):
    """Random positive initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n) + 0.1
    return a / a.sum()


def chirp_init(n):
    """Initialize with a chirp-like asymmetric shape."""
    x = np.linspace(0, 1, n)
    a = 1.0 + 0.5 * np.sin(2 * np.pi * (3 * x**2))
    a = np.maximum(a, 0.01)
    return a / a.sum()
```

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.

Algorithm: Log-parameterized smoothed optimization with adaptive tau,

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_obj_grad_x(x, tau):
    """Smoothed objective and gradient w.r.t. log-parameters x."""
    x_shifted = x - np.max(x)
    a = np.exp(x_shifted)
    a = a / np.sum(a)
    
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
        g = max_conv
    else:
        log_sum_exp = max_conv + (1.0 / tau) * np.log(sum_exp)
        p = exp_shifted / sum_exp
        g = log_sum_exp
    
    f = 2.0 * n * g
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
