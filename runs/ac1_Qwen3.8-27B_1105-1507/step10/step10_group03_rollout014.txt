import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def lse_grad(a, tau):
    """Compute gradient of LSE(tau * conv(a,a)) / tau w.r.t. a"""
    n = len(a)
    conv = compute_conv(a)
    m = np.max(conv)
    exp_vals = np.exp(tau * (conv - m))
    sum_exp = np.sum(exp_vals)
    p = exp_vals / sum_exp
    
    N = 1
    while N < 2*n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, g):
    """Linear convolution a*g via FFT. Returns array of length 2n-1."""
    n = len(a)
    m = len(g)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lse_vals(conv, tau):
    """Compute LSE and probability weights."""
    m = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - m))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    lse_val = m + np.log(sum_exp) / tau
    return lse_val, p


def smooth_gradient(a, tau):
    """Compute gradient of LSE(tau*conv(a,a))/tau w.r.t. a. Uses FFT."""
    n = len(a)
    conv = compute_conv(a)
    _, p = lse_vals(conv, tau)
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gd_staircase(a, time_budget, tau_levels=None, verbose=False, lr_base=2.0, beta=0.9):

def lp_solve_constraint_gen(a, n_constraints=400, max_cg_iters=15, solver_list=None):
    """
    LP polish with constraint generation.
    Solves: min M s.t. (a*g)_k <= M for selected k, g >= 0, sum(g)=1.
    Uses constraint generation to keep the LP small.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]

    # Initial constraint set: top convolution values of a*a
    conv_aa = compute_conv(a)
    active_k = np.argsort(conv_aa)[::-1][:n_constraints]

    g = a.copy()

    for cg_iter in range(max_cg_iters):
        # Build constraint matrix for active_k
        A = np.zeros((len(active_k), n))
        for idx, k in enumerate(active_k):
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for j in range(j_start, j_end):
                A[idx, j] = a[k - j]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        solved = False
        for solver in solver_list:
            try:
                prob.solve(solver=solver, verbose=False)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            return None, None

        g = np.maximum(g_var.value, 0)
        sg = float(np.sum(g))
        if sg < 1e-10:
            return None, None
        g = g / sg
        M = float(M_var.value)

        # Check which constraints are violated
        ag = compute_conv_cross(a, g)
        violated = np.where(ag > M + 1e-10)[0]

        if len(violated) == 0:
            return g, M

        # Add violated constraints, keep top n_constraints
        all_k = np.unique(np.concatenate([active_k, violated]))
        ag_all = ag[all_k]
        top_idx = np.argsort(ag_all)[::-1][:n_constraints]
        active_k = all_k[top_idx]

    return g, M


def lp_anderson(a, time_budget, verbose=False, seed=42, anderson_m=6):
    """
    LP fixed-point iteration with Anderson acceleration.
    The fixed-point map is: G(a) = normalize(LP_direction(a)).
    Anderson acceleration maintains a window of residuals and extrapolates.
    """

    a = np.maximum(a, 0).copy().astype(np.float64)

        return a

    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Anderson acceleration state
    window_x = []  # recent x values
    window_r = []  # recent residuals r = G(x) - x
    blend_t = 0.3
    stall = 0
    last_improve = 0

    for iteration in range(200):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Solve LP
        g_lp, M_lp = lp_solve_constraint_gen(a, n_constraints=min(300, n))
        if g_lp is None:
            break

        # Compute residual for Anderson
        r = g_lp - a

        # Anderson acceleration: find best extrapolation
        if len(window_x) >= 2:
            # Build the Biogen / BBBI system
            m = len(window_r)
            # Solve: min ||sum alpha_i r_i||^2 s.t. sum alpha_i = 1
            # This is equivalent to: min ||G - sum alpha_i x_i||^2 s.t. sum alpha_i = 1
            # where G = x_k + r_k
            R = np.array(window_r)  # (m, n)
            # Solve the linear system
            Gram = R @ R.T
            ones = np.ones(m)
            # Add L2 regularization for stability
            lam = 1e-8
            A_sys = np.zeros((m + 1, m + 1))
            A_sys[:m, :m] = Gram + lam * np.eye(m)
            A_sys[:m, m] = ones
            A_sys[m, :m] = ones
            b_sys = np.zeros(m + 1)
            # Objective: min ||r_k + sum alpha_i (r_i - r_k)||^2
            # = min ||sum_{i=0}^{m-1} alpha_i r_i + (1-sum alpha_i) r_k||^2

            A_sys[m, :m] = ones
            b_sys[:m] = -R @ r_k
            b_sys[m] = 0

            try:
                sol = np.linalg.solve(A_sys, b_sys)
                alpha = sol[:m]
                # x_new = x_k + alpha @ R
                x_new = a + alpha @ R
            except np.linalg.LinAlgError:
                x_new = g_lp.copy()
        else:

            x_new = (1.0 - blend_t) * a + blend_t * g_lp

        x_new = np.maximum(x_new, 0)
        sx = float(np.sum(x_new))
        if sx < 1e-10:
            x_new = best_a.copy()
            sx = 1.0
        x_new = x_new / sx

        obj_new = compute_obj(x_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = x_new.copy()
            stall = 0
            last_improve = iteration
            if verbose:
                print(f"  LP-AND it{iteration:3d}: obj={best_obj:.12f}, M_lp={M_lp:.6f}")
        else:
            stall += 1

        a = x_new

        # Update Anderson window
        window_x.append(a.copy())
        window_r.append(r.copy())
        if len(window_x) > anderson_m:
            window_x.pop(0)
            window_r.pop(0)

        # If stuck, perturb and reset

def make_init(n, mode, rng=None):
    """Create diverse initial sequences of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.0001))
        return a / a.sum()
    elif mode == 3:
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 4:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, np.pi, n)
        a = np.maximum(0.5 + 0.5 * np.cos(x), 0.001)
        return a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [2, 3, 5, 7]:
            a += 0.15 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 4))
        return a / a.sum()
    elif mode == 8:
        x = np.linspace(0, 1, n)
        a = np.where((x > 0.2) & (x < 0.8), 1.0,
                     np.maximum(0.5 - np.abs(x - 0.5), 0.01))
        return a / a.sum()
    elif mode == 9:
        # Optimal-shaped: raised cosine squared
        x = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif mode == 10:
        # Plateau with smooth edges
        x = np.linspace(0, 1, n)
        edge = 0.15
        a = np.ones(n)
        for i in range(n):
            if x[i] < edge:
                a[i] = 0.5 * (1 + np.cos(np.pi * x[i] / edge))
            elif x[i] > 1 - edge:
                a[i] = 0.5 * (1 + np.cos(np.pi * (x[i] - (1 - edge)) / edge))
        return a / a.sum()
    else:

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def lbfgsb_polish(a, time_budget, tau_levels=None, verbose=False):
    """
    Use scipy L-BFGS-B on the LSE-smoothed objective.
    Parameterize on simplex by fixing last element.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    a = a / a.sum()

    if tau_levels is None:
        tau_levels = [50, 200, 800, 2000]

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Work with n-1 free variables: a[0..n-2], with a[n-1] = 1 - sum(a[0..n-2])
    x0 = a[:-1].copy()

    for tau in tau_levels:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        def obj_grad(x):
            a_full = np.concatenate([x, [max(0, 1.0 - np.sum(x))]])
            a_full = np.maximum(a_full, 0)
            s = np.sum(a_full)
            if s < 1e-10:
                return 1e10, np.zeros(n - 1)
            a_full = a_full / s

            conv = compute_conv(a_full)
            lse_val, p = lse_vals(conv, tau)

            # Gradient w.r.t. a_full
            g_full = smooth_gradient(a_full, tau)

            # Gradient w.r.t. x (first n-1 elements, with a[n-1] = 1 - sum(x))
            # da[n-1]/dx_i = -1 for all i
            # d obj / dx_i = g_full[i] - g_full[n-1]
            g_x = g_full[:-1] - g_full[-1]

            return 2 * n * lse_val, g_x * 2 * n

        try:
            res = minimize(obj_grad, x0, method='L-BFGS-B',
                           options={'maxiter': 500, 'maxfun': 2000,
                                    'ftol': 1e-18, 'gtol': 1e-12})
            x_opt = res.x
            a_full = np.concatenate([x_opt, [max(0, 1.0 - np.sum(x_opt))]])
            a_full = np.maximum(a_full, 0)
            s = np.sum(a_full)
            if s > 1e-10:
                a_full = a_full / s
                obj_full = compute_obj(a_full)
                if obj_full < best_obj - 1e-14:
                    best_obj = obj_full
                    best_a = a_full.copy()
                    if verbose:
                        print(f"  L-BFGS-B tau={tau}: obj={best_obj:.12f}")
                x0 = best_a[:-1].copy()
        except Exception:
            break

    return best_a


def multi_n_search(best_seq, best_score, time_budget, verbose=True):
    """Try different sequence lengths and optimize each."""
    start = time.monotonic()
    n_current = len(best_seq)
    results = {}

    n_options = sorted(set([
        n_current,
        max(300, n_current - 200),
        n_current + 200,
        n_current + 400,
        n_current + 600,
        500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600, 2000,
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500]
    # Prioritize close to current, then larger
    n_options.sort(key=lambda x: abs(x - n_current))
    n_options = n_options[:8]

    if verbose:
        print(f"  Multi-n candidates: {n_options}")

    for n_try in n_options:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 40:
            break

        a_try = resample(best_seq, n_try)

        # GD
        alloc_gd = min(remaining * 0.3, 40.0)
        if alloc_gd > 10:
            a_try = gd_staircase(a_try, time_budget=alloc_gd,
                                  tau_levels=[10, 50, 200, 600], verbose=False)

        # LP if manageable
        remaining = time_budget - (time.monotonic() - start)
        if n_try <= 1500:
            alloc_lp = min(remaining * 0.2, 30.0)
            if alloc_lp > 10:
                a_try = lp_simple_iterate(a_try, time_budget=alloc_lp, verbose=False, seed=n_try)

        obj_try = compute_obj(a_try)
        results[n_try] = (a_try, obj_try)

        if verbose:
            marker = " **" if obj_try < best_score else ""
            print(f"  n={n_try:4d}: obj={obj_try:.12f}{marker}")

    return results


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining diverse initializations,
    staircase GD, constraint-generation LP with Anderson acceleration,
    L-BFGS-B polishing, multi-n exploration, and intensive final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.12f}")
    except Exception:
        parent = None

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.12f}")

    n_current = len(best_seq)

    # === Phase 1: Diverse initialization screening (30s) ===
    print(f"\n=== Phase 1: Diverse Init Screening ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.04, 40.0)

    init_modes = [0, 1, 2, 5, 6, 7, 9, 10]
    for mode in init_modes:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        a_init = make_init(n_current, mode, rng)
        alloc = min(p1_budget / len(init_modes) + 2, 8.0)
        if alloc < 3:
            break
        a_screen = gd_staircase(a_init, time_budget=alloc,
                                 tau_levels=[5, 30, 100], verbose=False)
        obj_screen = compute_obj(a_screen)
        if obj_screen < best_score:
            best_score = obj_screen
            best_seq = a_screen.copy()
            print(f"  Init mode {mode}: obj={best_score:.12f} **")
        else:
            print(f"  Init mode {mode}: obj={obj_screen:.12f}")

    print(f"  After screening: best={best_score:.12f}")

    # === Phase 2: Main GD with aggressive tau schedule (250s) ===
    print(f"\n=== Phase 2: Aggressive Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.20, 250.0)
    if p2_budget > 20:
        tau_sched = [3, 8, 20, 50, 120, 300, 700, 1500, 3000, 6000, 10000]
        a2 = gd_staircase(best_seq, time_budget=p2_budget,
                          tau_levels=tau_sched, verbose=True, lr_base=2.5, beta=0.9)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 GD: obj={best_score:.12f} **")
        else:
            print(f"  Phase2 GD: obj={obj2:.12f}")

    # === Phase 3: LP Polishing with Anderson acceleration (350s) ===
    print(f"\n=== Phase 3: LP Polishing (Anderson) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.25, 350.0)

    if p3_budget > 30 and n_current <= 2000:
        # First: simple LP iteration
        alloc1 = min(p3_budget * 0.4, 120.0)
        if alloc1 > 15:
            a3 = lp_simple_iterate(best_seq, time_budget=alloc1, verbose=True, seed=100)
            obj3 = compute_obj(a3)
            if obj3 < best_score:
                best_score = obj3
                best_seq = a3.copy()
                print(f"  LP-simple: obj={best_score:.12f} **")
            else:
                print(f"  LP-simple: obj={obj3:.12f}")

        # Then: Anderson-accelerated LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.3, 200.0)
        if alloc2 > 20:
            a3b = lp_anderson(best_seq, time_budget=alloc2, verbose=True, seed=200)
            obj3b = compute_obj(a3b)
            if obj3b < best_score:
                best_score = obj3b
                best_seq = a3b.copy()
                print(f"  LP-Anderson: obj={best_score:.12f} **")
            else:
                print(f"  LP-Anderson: obj={obj3b:.12f}")

    # === Phase 4: L-BFGS-B polish (100s) ===
    print(f"\n=== Phase 4: L-BFGS-B Polish ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 100.0)
    if p4_budget > 20:
        a4 = lbfgsb_polish(best_seq, time_budget=p4_budget,
                           tau_levels=[100, 400, 1500], verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  L-BFGS-B: obj={best_score:.12f} **")
        else:
            print(f"  L-BFGS-B: obj={obj4:.12f}")

    # === Phase 5: Multi-n exploration (150s) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.12, 150.0)
    if p5_budget > 30:
        results_n = multi_n_search(best_seq, best_score, p5_budget, verbose=True)
        for n_try, (a_try, obj_try) in results_n.items():
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_current = len(best_seq)
                print(f"  ** Updated best from n={n_try}: {best_score:.12f}")

    # === Phase 6: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 6: Final Intensive Polish ===")
    remaining = deadline - time.monotonic()

    # Alternate GD and LP
    for round_num in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # GD step
        alloc_gd = min(remaining * 0.15, 40.0)
        if alloc_gd > 10:
            a_f = gd_staircase(best_seq, time_budget=alloc_gd,
                               tau_levels=[50, 200, 800, 2000, 5000], verbose=False)
            obj_f = compute_obj(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final GD r{round_num}: obj={best_score:.12f} **")

        # LP step
        remaining = deadline - time.monotonic()
        if remaining > 20 and len(best_seq) <= 2000:
            alloc_lp = min(remaining * 0.15, 40.0)
            if alloc_lp > 10:
                a_lp = lp_simple_iterate(best_seq, time_budget=alloc_lp,
                                        verbose=False, seed=1000 + round_num * 7)
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    print(f"  Final LP r{round_num}: obj={best_score:.12f} **")

    # === Phase 7: Ultra-high tau GD (final) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 60.0)
        a_ultra = gd_staircase(best_seq, time_budget=alloc,
                               tau_levels=[500, 1500, 4000, 8000, 15000], verbose=True)
        obj_ultra = compute_obj(a_ultra)
        if obj_ultra < best_score:
            best_score = obj_ultra
            best_seq = a_ultra.copy()
            print(f"  Ultra GD: obj={best_score:.12f} **")

    # Final LP if time allows
    remaining = deadline - time.monotonic()
    if remaining > 20 and len(best_seq) <= 2000:
        alloc = min(remaining * 0.5, 50.0)
        a_final_lp = lp_anderson(best_seq, time_budget=alloc, verbose=True, seed=9999)
        obj_final_lp = compute_obj(a_final_lp)
        if obj_final_lp < best_score:
            best_score = obj_final_lp
            best_seq = a_final_lp.copy()
            print(f"  Final LP-AND: obj={best_score:.12f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.12f}, time={elapsed:.1f}s ===")

    return result
