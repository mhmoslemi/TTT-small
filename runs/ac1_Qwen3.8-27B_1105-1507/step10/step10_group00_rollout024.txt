"""
Continuation mirror descent for step function peak minimization.
Key ideas:
1. Soft peak (log-sum-exp) objective with temperature continuation
2. Mirror descent (exponential weights) with Nesterov momentum
3. Adaptive step size and periodic LP polishing
4. Multi-n exploration with diverse initializations

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fft_convolve(a):
    """Compute conv(a, a) via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_score(a):
    """Compute the full score: 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, max_iters=3):
    """LP polish: iteratively solve min M s.t. conv(a,g)<=M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    best_g = a_norm.copy()
    best_M = float(np.max(fft_convolve(a_norm)))
    
    for _ in range(max_iters):
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None:
                    g_val = np.maximum(g.value, 0)
                    s = float(g_val.sum())
                    if s > 1e-10:
                        g_val = g_val / s
                        M_val = float(M.value)
                        conv_g = A @ g_val
                        actual_M = float(conv_g.max())
                        if actual_M < best_M:
                            best_M = actual_M
                            best_g = g_val.copy()
                        a_norm = g_val.copy()  # Update for next iteration
                        break
            except Exception:
                continue
    
    return best_g


def continuation_mirror_descent(a, time_budget, verbose=False):

    start = time.monotonic()

    # Start T at about 2x the current peak value, end at about 0.1x
    current_peak = float(np.max(fft_convolve(a)))
    T_start = current_peak * 3.0
    T_end = current_peak * 0.05
    n_cont = 20
    
    # Mirror descent parameters
    beta = 0.9  # momentum
    base_lr = 1.0 / np.sqrt(n)
    
    for cont in range(n_cont):
        frac = cont / max(1, n_cont - 1)
        T = T_start * (T_end / T_start) ** frac

        momentum = np.zeros(n)

        max_inner_stall = 50000

            conv = fft_convolve(a)

                best_obj = hard_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            g = np.zeros(n)

            sig = np.where(p > thresh)[0]
            for k in sig:
                i_lo = max(0, k - n + 1)
                i_hi = min(k, n - 1)
                if i_lo <= i_hi:
                    g[i_lo:i_hi + 1] += p[k] * a[k - np.arange(i_lo, i_hi + 1)]
            g *= 2.0
            
            # Mirror descent step with momentum
            momentum = beta * momentum - lr * g
            log_a = np.log(a + 1e-30) + momentum
            log_a -= log_a.max()
            a = np.exp(log_a)
            a = a / a.sum()
            
            it += 1

            if no_improve > 10000:
                lr *= 0.8
                momentum *= 0.5
                no_improve = 0
                if it - no_improve > max_inner_stall:
                    break  # Move to next temperature
            elif no_improve > 5000:
                lr *= 0.9
                no_improve = 0
            
            if verbose and it % 100000 == 0:
                elapsed = time.monotonic() - start
                print(f"      T={T:.2e}, it={it}, obj={best_obj:.8f}, lr={lr:.4f}, t={elapsed:.1f}s")

def diverse_init(n, mode, rng):
    """Diverse initialization."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = 1.0 / (0.05 + 3.0 * (2 * x - 1) ** 2)
    elif mode == 1:
        a = np.exp(-((np.arange(n) - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((np.arange(n) - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = np.exp(4.0 * (2 * x - 1) ** 2)
    elif mode == 3:
        a = 1.0 / np.sqrt(np.maximum(0.01, 1 - (2 * x - 1) ** 2))
    elif mode == 4:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 5:
        a = 1.0 / np.cosh(3.0 * (2 * x - 1))
    elif mode == 6:
        a = (2 * x - 1) ** 6 + 0.03
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
