"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based best-response dynamics with warm-starting, adaptive blending,
diverse initializations, and spectral polishing.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    """Cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # i.e., A is a Toeplitz-like matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def solve_best_response(a, prev_g=None, solver=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_opt, M_opt).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    # Try solvers in order of preference
    solvers = [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    if solver is not None:
        solvers = [solver] + [s for s in solvers if s != solver]
    
    for sol in solvers:
        try:
            if prev_g is not None and sol == cp.CLARABEL:
                # Can't easily warm-start in cvxpy, but we can set initial value
                g.value = prev_g
            prob.solve(solver=sol)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def solve_best_response_approx(a, prev_g=None, n_iter=50, lr=None):
    """
    Approximate best response using projected gradient on the LP dual.
    This is much faster per call than solving the full LP.
    
    LP: min M s.t. A@g <= M, sum(g)=1, g>=0
    Equivalent: minimize max_k (A@g)_k subject to sum(g)=1, g>=0
    
    We use subgradient descent on this equivalent formulation.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    if prev_g is not None:
        g = prev_g.copy()
        g = np.maximum(g, 0)
        g = g / g.sum()
    else:
        g = np.ones(n) / n
    
    if lr is None:
        lr = 2.0 / (n + 1)
    
    best_g = g.copy()
    best_val = np.max(compute_conv_cross(a_norm, g))
    
    for it in range(n_iter):
        conv = compute_conv_cross(a_norm, g)
        max_val = np.max(conv)
        # Subgradient of max_k (A@g)_k w.r.t. g is A^T * e_{k*}
        # where k* = argmax
        k_star = np.argmax(conv)
        # A^T * e_{k_star} = A[:, k_star] ... wait, no.
        # (A@g)_k = sum_j A[k,j] * g[j]
        # d/dg_j (A@g)_{k*} = A[k*, j]
        # A[k*, j] = a[k*-j] if 0 <= k*-j < n
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = a_norm[idx]
        
        # Projected subgradient step
        g_new = g - lr * grad
        g_new = np.maximum(g_new, 0)
        s = g_new.sum()
        if s < 1e-10:
            g_new = g
        else:
            g_new = g_new / s
        
        # Check if improved
        conv_new = compute_conv_cross(a_norm, g_new)
        val_new = np.max(conv_new)
        if val_new < best_val:
            best_val = val_new
            best_g = g_new.copy()
        
        g = g_new
        lr *= 0.999  # slight decay
    
    return best_g, best_val

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based best-response dynamics with momentum acceleration, adaptive blending,
diverse initializations, constraint-reduced LP variants, and spectral polishing.
Key improvements over baseline: momentum on outer loop, smarter time allocation,
multiple diverse starting points, and aggressive local search at the end.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Cross-convolution a*b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_best_response(a, K=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k (or top K), sum(g)=1, g>=0.
    If K is specified, only use the K largest constraints (for speed).
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_sum = float(np.sum(a))
    if a_sum < 1e-10:
        return None, None
    a_norm = a / a_sum
    A = build_conv_matrix(a_norm)

    if K is not None and K < 2 * n - 1:
        # Use only top K constraints based on current iterate
        # Use uniform g to estimate active constraints
        conv_est = compute_conv_cross(a_norm, np.ones(n) / n)
        active = np.argsort(conv_est)[-K:]
        A_use = A[active, :]
    else:
        A_use = A

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_use @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def subgradient_best_response(a, g_init, n_iter=30):
    """
    Fast approximate best response using projected subgradient method.
    Minimizes max_k (a*g)_k over simplex.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    g = g_init.copy()
    g = np.maximum(g, 0)
    s = g.sum()
    if s < 1e-10:
        g = np.ones(n) / n
    else:
        g = g / s

    best_g = g.copy()
    best_val = float(np.max(compute_conv_cross(a_norm, g)))
    lr = 1.0 / (np.linalg.norm(a_norm, 1) + 1e-10)

    for it in range(n_iter):
        conv = compute_conv_cross(a_norm, g)
        k_star = int(np.argmax(conv))
        # Gradient: A[k_star, :]
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for j in range(lo, hi + 1):
            grad[j] = a_norm[k_star - j]

        t = 2.0 * lr / (it + 1)
        g = g - t * grad
        g = np.maximum(g, 0)
        s = g.sum()
        if s < 1e-10:
            g = best_g.copy()
        else:
            g = g / s

        val = float(np.max(compute_conv_cross(a_norm, g)))
        if val < best_val:
            best_val = val
            best_g = g.copy()

    return best_g, best_val

def gradient_polish(a, time_budget, tau_start=15.0, tau_max=2000.0):
    """Nesterov accelerated gradient on smoothed objective, projected to simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = tau_start
    v = np.zeros(n)
    beta = 0.9
    base_lr = 1.5 / np.sqrt(n)

    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_c = float(np.max(conv))
        scaled = tau * (conv - max_c)
        scaled = np.clip(scaled, -50, 50)
        exp_vals = np.exp(scaled)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient of sum_k p_k * (a*a)_k w.r.t. a
        # = 2 * cross_correlation(p, a)
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(a, N)
        g_full = ifft(P * np.conj(A_f)).real
        grad = 2.0 * g_full[:n]

        # Adaptive learning rate
        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            adaptive_lr = 1.0 / (g_norm + 1e-10)
            lr = min(max(adaptive_lr * 0.5, 0.1 * base_lr), 3.0 * base_lr)
        else:
            lr = base_lr

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        tau = min(tau * 1.03, tau_max)

        # Reset if stuck
        if (int((time.monotonic() - start) * 10) % 100 == 0) and compute_obj(a) > best_obj + 1e-5:
            a = best_a.copy()
            v = np.zeros(n)
            beta = 0.5

    return best_a


def fourier_perturb(a, rng, strength=0.03):
    """Add structured Fourier perturbation."""
    n = len(a)
    n_modes = 5
    pert = np.zeros(n)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, method='random'):
    """Generate diverse initializations."""
    if method == 'random':
        a = rng.random(n) + 0.1
    elif method == 'fourier':
        # Cosine-based initialization
        a = np.zeros(n)
        n_harmonics = rng.integers(2, 5)
        for h in range(n_harmonics):
            freq = rng.integers(1, n // 2)
            phase = rng.uniform(0, 2 * np.pi)
            a += np.abs(np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
        a += 0.1
    elif method == 'bimodal':
        # Two-peaked structure
        c1 = rng.integers(n // 4, 3 * n // 4)
        w = rng.integers(n // 10, n // 5)
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - c1) / w) ** 2) + np.exp(-0.5 * ((x - (n - c1)) / w) ** 2) + 0.05
    elif method == 'periodic':
        # Periodic structure
        period = rng.integers(3, 8)
        pattern = rng.random(period) + 0.3
        reps = (n + period - 1) // period
        a = np.tile(pattern, reps)[:n]
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    return a / a.sum()


def local_search(a, time_budget, rng):
    """Aggressive local search with coordinate descent and random kicks."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1:
            break

        # Try random coordinate updates
        n_trials = min(1000, int(remaining * 500))
        for _ in range(n_trials):
            if time.monotonic() - start > time_budget - 0.5:
                break
            # Pick 2-5 random coordinates to perturb
            n_pert = rng.integers(1, 4)
            coords = rng.integers(0, n, size=n_pert)
            a_try = best_a.copy()
            for c in coords:
                a_try[c] *= rng.uniform(0.5, 1.5)
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()

        # Occasional larger perturbation
        if rng.random() < 0.1:
            a_try = fourier_perturb(best_a, rng, strength=0.01)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating with momentum,
    diverse initializations, gradient polishing, and local search.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0)
        if parent.size == 0 or parent.sum() < 0.01:
            raise ValueError("Invalid parent")
        parent = parent / parent.sum()
        n_parent = parent.size
        print(f"Parent loaded: n={n_parent}, obj={compute_obj(parent):.10f}")
    except Exception:
        n_parent = 800
        parent = np.ones(n_parent) / n_parent
        print(f"Default init: n={n_parent}, obj={compute_obj(parent):.10f}")

    best_a = parent.copy()
    best_obj = compute_obj(parent)
    rng = np.random.default_rng(seed)

    # === Phase 1: LP Alternating with Momentum (55% of time) ===
    print("\n=== Phase 1: LP Alternating with Momentum ===")
    p1_budget = (deadline - start_time) * 0.55
    p1_budget = max(60, p1_budget)
    a_p1 = lp_alternate_momentum(best_a, time_budget=p1_budget, seed=seed,
                                  use_momentum=True, verbose=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1.copy()
        print(f"  Phase 1: obj={best_obj:.10f} **")
    else:
        print(f"  Phase 1: obj={obj_p1:.10f} (no improvement)")

    # === Phase 2: Diverse Initializations (25% of time) ===
    print("\n=== Phase 2: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    if remaining > 120:
        n_div = 3
        time_per_div = min(remaining * 0.8 / n_div, 150.0)
        methods = ['fourier', 'periodic', 'bimodal']

        # Also try perturbing the current best
        methods.append('parent_perturb')

        for i, method in enumerate(methods[:n_div]):
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            alloc = min(time_per_div, remaining * 0.4)
            if alloc < 30:
                break

            if method == 'parent_perturb':
                a_init = fourier_perturb(best_a, rng, strength=0.03)
            else:
                a_init = diverse_init(len(best_a), rng, method=method)

            # Run LP alternating from this init
            a_div = lp_alternate_momentum(a_init, time_budget=alloc * 0.8,
                                           seed=seed + i * 1000,
                                           use_momentum=True, verbose=False)
            obj_div = compute_obj(a_div)
            print(f"  Diversified [{method}]: obj={obj_div:.10f}")
            if obj_div < best_obj:
                best_obj = obj_div
                best_a = a_div.copy()
                print(f"  ** New best: obj={best_obj:.10f} **")

    # === Phase 3: Gradient Polish (12% of time) ===
    print("\n=== Phase 3: Gradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = min(remaining * 0.15, 120.0)
        a_p3 = gradient_polish(best_a, time_budget=p3_budget, tau_start=10.0)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3.copy()
            print(f"  Gradient polish: obj={best_obj:.10f} **")
        else:
            print(f"  Gradient polish: obj={obj_p3:.10f}")

    # === Phase 4: Try different n values (7% of time) ===
    print("\n=== Phase 4: Length Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        n_best = len(best_a)
        n_options = [n_best - 50, n_best + 50, n_best - 100, n_best + 100,
                     n_best + 200, n_best - 200]
        n_options = sorted(set([n for n in n_options if 200 <= n <= 2000]))

        for n_try in n_options[:4]:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            alloc = min(remaining * 0.2, 80.0)
            if alloc < 20:
                break

            a_try = resample_to_length(best_a, n_try)
            a_try = lp_alternate_momentum(a_try, time_budget=alloc * 0.7,
                                           seed=n_try, use_momentum=True, verbose=False)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                print(f"  ** n={n_try}: obj={best_obj:.10f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 5: Local Search (5% of time) ===
    print("\n=== Phase 5: Local Search ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        ls_budget = min(remaining * 0.4, 60.0)
        a_ls = local_search(best_a, time_budget=ls_budget, rng=rng)
        obj_ls = compute_obj(a_ls)
        if obj_ls < best_obj:
            best_obj = obj_ls
            best_a = a_ls.copy()
            print(f"  Local search: obj={best_obj:.10f} **")
        else:
            print(f"  Local search: obj={obj_ls:.10f}")

    # === Phase 6: Final LP refinement ===
    print("\n=== Phase 6: Final Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        final_budget = min(remaining * 0.7, 60.0)
        a_fin = lp_alternate_momentum(best_a, time_budget=final_budget,
                                       seed=seed + 99999,
                                       use_momentum=True, verbose=False)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()
            print(f"  Final refinement: obj={best_obj:.10f} **")
        else:
            print(f"  Final refinement: obj={obj_fin:.10f}")

    # Final validation
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
