Key innovations: (1) reduced LP constraint set for faster solves enabling more
iterations, (2) skewed/non-palindromic initializations that exploit the

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ab(a, b):
    """Full linear convolution a*b via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_subset(a, lag_indices):
    """Build matrix A (len(lag_indices) x n) such that A @ g = conv(a, g) at those lags."""
    n = len(a)

    for idx, k in enumerate(lag_indices):
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A[idx, j] = a[kj]
    return A

        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_full(a, time_limit=10.0):
    """Solve full LP with all 2n-1 constraints."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv_self = compute_conv(a_norm)
    lag_indices = np.arange(2 * n - 1)

    A = build_conv_matrix_subset(a_norm, lag_indices)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def exact_line_search(a, g, n_evals=40):
    """
    Find optimal t in [0, 1.5] minimizing max_k conv(h(t), h(t))[k]
    where h(t) = (1-t)*a + t*g, assuming both normalized to sum=1.
    The objective along this line is max of quadratics, so we evaluate
    at many points and refine.
    """
    n = len(a)
    conv_aa = compute_conv(a)
    conv_ag = compute_conv_ab(a, g)
    conv_gg = compute_conv(g)

    def max_conv_at_t(t):
        # conv((1-t)a + tg, (1-t)a + tg)[k]
        # = (1-t)^2 * conv_aa[k] + 2t(1-t) * conv_ag[k] + t^2 * conv_gg[k]
        vals = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return float(np.max(vals))

    # Coarse grid
    ts = np.linspace(0, 1.5, n_evals)
    best_t = 0
    best_val = max_conv_at_t(0)
    for t in ts:
        v = max_conv_at_t(t)
        if v < best_val:
            best_val = v
            best_t = t

    # Golden section refinement around best_t
    lo = max(0, best_t - 1.5 / n_evals)
    hi = min(1.5, best_t + 1.5 / n_evals)
    phi = (1 + math.sqrt(5)) / 2
    gr = 2 - phi

    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = max_conv_at_t(c)
    fd = max_conv_at_t(d)

    for _ in range(20):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = max_conv_at_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = max_conv_at_t(d)

    t_best = c if fc < fd else d
    v_best = min(fc, fd)
    if v_best > best_val:
        t_best = best_t
        v_best = best_val

    return t_best, v_best


def fast_subgradient(a, time_budget=20.0):
    """Fast subgradient descent with Nesterov momentum and adaptive lr."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.015
    momentum = np.zeros(n)
    beta = 0.6
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))

        # Subgradient of conv(a,a)[k_max] w.r.t. a
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a[j1]
            elif j1 >= n:
                pass
            if 0 <= j2 < n:
                grad[i] += a[j2]
            elif j2 < 0:
                pass

        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        lr *= 0.999
        it += 1

        if it - best_it > 3000:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = 0.008
            best_it = it

    return best_a


def skewed_init(n, skew=0.3, rng=None):
    """Initialize with a skewed (non-palindromic) distribution."""
    if rng is None:
        rng = np.random.default_rng()
    x = np.linspace(0, 1, n)
    # Skewed Beta-like distribution
    alpha = 1.0 + skew
    beta = 2.0 - skew
    a = x ** (alpha - 1) * (1 - x) ** (beta - 1)
    a = np.maximum(a, 1e-6)
    # Add some noise
    a += rng.uniform(0, 0.1, n)
    s = float(np.sum(a))
    return a / s


def diverse_init_v2(n, mode, rng):
    """Generate diverse initializations with emphasis on non-palindromic shapes."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    mid = n / 2.0

    if mode == 0:  # Left-heavy
        a = np.exp(-2.0 * (x - 0.3) ** 2) + 0.1 * np.exp(-2.0 * (x - 0.7) ** 2) + 0.01
    elif mode == 1:  # Right-heavy
        a = np.exp(-2.0 * (x - 0.7) ** 2) + 0.1 * np.exp(-2.0 * (x - 0.3) ** 2) + 0.01
    elif mode == 2:  # U-shaped (symmetric baseline)
        a = 1.0 / (0.1 + 4.0 * (2 * x - 1) ** 2)
    elif mode == 3:  # Skewed U
        a = 1.0 / (0.1 + 3.0 * (2 * x - 0.6) ** 2)
    elif mode == 4:  # Smooth random
        a = rng.uniform(0.2, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 5:  # Two bumps asymmetric
        a = np.exp(-((idx - n * 0.25) ** 2) / (2 * (n * 0.12) ** 2)) + \
            0.5 * np.exp(-((idx - n * 0.7) ** 2) / (2 * (n * 0.15) ** 2)) + 0.01
    elif mode == 6:  # Power law left
        a = (1.0 - x) ** 0.5 + x ** 2.0 + 0.01
    elif mode == 7:  # Power law right
        a = x ** 0.5 + (1.0 - x) ** 2.0 + 0.01
    elif mode == 8:  # Cosine asymmetric
        a = 1.5 + np.cos(2 * np.pi * (x - 0.3)) * 0.5 + 0.1
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def lp_polish_fast(a, time_budget=60.0, top_k_start=30, top_k_max=100, verbose=False):
    """
    Fast LP polishing with adaptive constraint set and exact line search.
    Starts with few constraints, gradually adds more.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    top_k = top_k_start
    rng = np.random.default_rng(99999 + n)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        # Solve LP with current constraint set
        T0 = time.monotonic()
        g_lp, M_lp = solve_lp_reduced(a, top_k=top_k)
        T1 = time.monotonic()

        if g_lp is None:
            # Try full LP
            g_lp, M_lp = solve_lp_full(a, time_limit=min(5.0, remaining * 0.3))
            if g_lp is None:
                break

        # Exact line search
        t_opt, obj_opt = exact_line_search(a, g_lp, n_evals=30)

        if t_opt > 0.001 and obj_opt < best_obj - 1e-12:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                    # Grow constraint set slowly
                    top_k = min(top_k + 3, top_k_max)
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
            top_k = min(top_k + 5, top_k_max)

        it += 1
        if verbose and it % 10 == 0:
            print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"top_k={top_k}, lp_time={T1-T0:.2f}s, elapsed={time.monotonic()-start:.1f}s")

        # Adaptive restart
        if stall >= 4:
            if stall <= 6:
                # Small perturbation
                noise = rng.normal(0, 0.01 * stall, n)
                a = best_a + noise
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                else:
                    a = best_a.copy()
            elif stall <= 10:
                # Medium perturbation with skew
                mode = it % 4
                a_div = diverse_init_v2(n, mode, rng)
                a = 0.3 * a_div + 0.7 * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()
            else:
                # Large restart
                mode = it % 8
                a_div = diverse_init_v2(n, mode, rng)
                a = 0.5 * a_div + 0.5 * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()
                top_k = top_k_start
            stall = 0

    return best_a


def lp_polish_intensive(a, time_budget=120.0, verbose=False):
    """
    Intensive LP polishing with full constraints for final refinement.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(77777 + n)

    while time.monotonic() - start < time_budget:
        remaining = time.monotonic() - start
        rem = time_budget - remaining
        if rem < 3:
            break

        T0 = time.monotonic()
        g_lp, M_lp = solve_lp_full(a)
        T1 = time.monotonic()

        if g_lp is None:
            g_lp, M_lp = solve_lp_reduced(a, top_k=100)
            if g_lp is None:
                break

        t_opt, obj_opt = exact_line_search(a, g_lp, n_evals=40)

        if t_opt > 0.001 and obj_opt < best_obj - 1e-12:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 5 == 0:
            print(f"      FullLP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"lp_time={T1-T0:.2f}s, elapsed={time.monotonic()-start:.1f}s")

        if stall >= 3:
            noise = rng.normal(0, 0.005 * stall, n)
            a = best_a + noise
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 0.01:
                a = a / s
            else:
                a = best_a.copy()
            stall = 0

    return best_a


def try_length(a_best, n_try, time_budget, rng):
    """Try interpolating to a different length and optimizing."""
    n_old = len(a_best)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        return None, np.inf
    a_try = a_try / s

    # Quick warm-up
    if time_budget > 30:
        a_try = fast_subgradient(a_try, time_budget=min(10.0, time_budget * 0.15))

    # LP polish
    remaining = time_budget - 10
    a_try = lp_polish_fast(a_try, time_budget=min(40.0, remaining * 0.6), top_k_start=30, top_k_max=80)

    obj_try = compute_obj(a_try)
    return a_try, obj_try


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-start with skewed initializations, fast LP polishing
    with adaptive constraints, exact quadratic line search, and intensive
    final refinement on the best candidate.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    # If no parent, use skewed init
    if best_seq is None:
        best_seq = skewed_init(n_current, skew=0.4, rng=rng)
        best_score = compute_obj(best_seq)
        print(f"Skewed init (n={n_current}): obj={best_score:.8f}")

    # === Phase 1: Rapid multi-start screening ===
    # Try several diverse initializations, do quick optimization on each,
    # keep the best one to carry forward.
    print("\n=== Phase 1: Multi-start screening ===")
    n_candidates = 8
    time_per_candidate = min(30.0, (deadline - time.monotonic()) * 0.15 / n_candidates)

    for c in range(n_candidates):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        mode = c % 9
        if c == 0:
            # Start from parent (if available) with slight perturbation
            a_c = best_seq + rng.normal(0, 0.01, n_current)
            a_c = np.maximum(a_c, 0)
            a_c = a_c / a_c.sum()
        else:
            a_c = diverse_init_v2(n_current, mode, rng)

        # Quick subgradient
        if time_per_candidate > 5:
            a_c = fast_subgradient(a_c, time_budget=min(5.0, time_per_candidate * 0.3))

        # Quick LP
        alloc = min(15.0, time_per_candidate * 0.6)
        if alloc > 3:
            a_c = lp_polish_fast(a_c, time_budget=alloc, top_k_start=20, top_k_max=60)

        obj_c = compute_obj(a_c)
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_c.copy()
            print(f"  Candidate {c} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Candidate {c} (mode={mode}): obj={obj_c:.8f}")

    # === Phase 2: Intensive fast LP polishing on best ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 2: Intensive LP (fast, n={len(best_seq)}) ===")
        alloc = min(remaining * 0.35, 400.0)
        a_lp = lp_polish_fast(best_seq, time_budget=alloc, top_k_start=30, top_k_max=120, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Full-constraint LP polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 3: Full LP polishing ===")
        alloc = min(remaining * 0.25, 300.0)
        a_full = lp_polish_intensive(best_seq, time_budget=alloc, verbose=True)
        obj_full = compute_obj(a_full)
        if obj_full < best_score:
            best_score = obj_full
            best_seq = a_full.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj_full:.8f} (no improvement)")

    # === Phase 4: n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: n exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 100, n_best - 50, n_best + 50, n_best + 100,
            n_best + 200, n_best - 200,
            700, 850, 900, 950, 1000, 1100, 1200
        ]))
        n_options = [n for n in n_options if 500 <= n <= 1500 and abs(n - n_best) >= 30]
        n_options = n_options[:6]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            a_try, obj_try = try_length(best_seq, n_try, min(50.0, remaining * 0.5), rng)
            if a_try is not None and obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts with best blending ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 5: Diverse restarts ===")
        n_cur = len(best_seq)
        for r in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            mode = (r * 3 + 1) % 9
            a_r = diverse_init_v2(n_cur, mode, rng)
            blend = 0.4
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            # Quick optimize
            if remaining > 30:
                a_r = fast_subgradient(a_r, time_budget=min(8.0, remaining * 0.15))
            alloc = min(remaining * 0.3, 50.0)
            if alloc > 5:
                a_r = lp_polish_fast(a_r, time_budget=alloc, top_k_start=25, top_k_max=80)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        print(f"\n=== Phase 6: Final polish ===")
        alloc = min(remaining - 10, 200.0)
        if alloc > 15:
            a_final = lp_polish_intensive(best_seq, time_budget=alloc, verbose=True)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final: obj={best_score:.8f} **")
            else:
                print(f"  Final: obj={obj_final:.8f}")

    # === Phase 7: Last-chance fast iterations ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_polish_fast(best_seq, time_budget=min(remaining - 5, 50.0),
                                top_k_start=50, top_k_max=150)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last chance: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
