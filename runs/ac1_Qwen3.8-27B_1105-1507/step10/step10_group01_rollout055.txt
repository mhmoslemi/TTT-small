"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set LP, multiplicative weights, and gradient polishing.
Key improvements: active-set LP for speed, multiplicative weights as alternative
descent direction, better n exploration, convex line search.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_rows(a, indices):
    """Build rows of convolution matrix for specified row indices."""
    n = len(a)
    k_idx = indices[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    row_count = len(indices)
    A = np.zeros((row_count, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_active(a, active_idx):
    """Solve LP with active set constraints. Returns (g, M, violated_idx)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A_sub = build_conv_matrix_rows(a_norm, active_idx)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_sub @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    best_g = None
    best_M = None
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    best_g = g_val / s
                    best_M = float(M.value)
                    break
        except Exception:

    if best_g is None:
        return None, None, None

    full_conv = compute_conv(best_g)
    max_full = float(np.max(full_conv))
    
    if max_full <= best_M + 1e-6:
        return best_g, best_M, None
    else:
        violated = np.where(full_conv > best_M + 1e-8)[0]
        return best_g, best_M, violated


def convex_line_search(a, g):
    """Line search using convexity: f(t) = obj((1-t)*a + t*g) is convex in t."""
    n = len(a)
    
    def f(t):
        a_t = (1 - t) * a + t * g
        a_t = np.maximum(a_t, 0)
        s = float(np.sum(a_t))
        if s < 1e-10:
            return np.inf
        a_t = a_t / s
        conv = compute_conv(a_t)
        return float(np.max(conv))
    
    # Check endpoints
    f0 = f(0.0)
    f1 = f(1.0)
    
    if f1 < f0:
        # Minimum is somewhere in [0, 1], try to bracket
        best_t = 1.0
        best_f = f1
        # Ternary search
        lo, hi = 0.0, 1.0
        for _ in range(40):
            m1 = lo + (hi - lo) / 3
            m2 = hi - (hi - lo) / 3
            f_m1 = f(m1)
            f_m2 = f(m2)

        best_t = (lo + hi) / 2
        best_f = f(best_t)

    else:

def multiplicative_weights_step(a, tau=100.0):
    """One step of multiplicative weights (mirror descent with entropy)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    # Smoothed subgradient
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / exp_vals.sum()
    
    # Gradient: g[j] = 2 * sum_k p[k] * a[k-j] (valid k,j)
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad = 2.0 * ifft(P * np.conj(A)).real[:n]
    
    # Multiplicative weights update
    eta = 0.1  # step size
    a_new = a * np.exp(-eta * grad / (np.max(np.abs(grad)) + 1e-10))
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()

def lp_optimize(a, time_budget=100.0, verbose=False):
    """LP-driven optimization with active set."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = float(np.max(compute_conv(a)))
    start = time.monotonic()
    
    # Start with all constraints
    active_idx = list(range(2 * n - 1))
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        
        g, M, violated = solve_lp_active(a, active_idx)
        if g is None:
            # Fallback: use all constraints
            g, M, violated = solve_lp_active(a, list(range(2 * n - 1)))
            if g is None:
                break
        
        # Update active set
        if violated is not None:
            active_idx = list(set(active_idx + list(violated)))
        else:
            # Prune: only keep near-active constraints
            full_conv = compute_conv(g)
            max_g = float(np.max(full_conv))
            active_idx = list(np.where(full_conv >= max_g * 0.98)[0])
            if len(active_idx) < 5:
                active_idx = list(np.argsort(full_conv)[-max(10, n//10):])
        
        # Line search
        best_t, best_t_obj = convex_line_search(a, g)
        
        it += 1
        if best_t_obj < best_obj - 1e-10:
            a_new = (1 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                best_obj = best_t_obj
                best_a = a_new.copy()
                a = a_new
                stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={2*n*best_obj:.8f}, M={M:.6f}, |active|={len(active_idx)}, t={best_t:.3f}")
        else:

            if verbose and it % 10 == 0:
                print(f"    LP it{it}: stalled obj={2*n*best_obj:.8f}, M={M:.6f}")

    return best_a


def gd_polish(a, time_budget=60.0, tau_init=100.0, tau_end=2000.0, verbose=False):
    """Gradient descent with increasing tau (approaches exact max)."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = float(np.max(compute_conv(a)))
    start = time.monotonic()
    
    v = np.zeros(n)
    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5
        
        # Nesterov
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Smoothed gradient
        conv = compute_conv(y)
        max_c = float(np.max(conv))
        exp_v = np.exp(tau * (conv - max_c))
        p = exp_v / exp_v.sum()
        
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        Y = fft(y, N)
        grad = 2.0 * ifft(P * np.conj(Y)).real[:n]
        
        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            lr = min(lr * 0.999, 5.0 / (g_norm * np.sqrt(n) + 1e-10))
        
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = float(np.max(compute_conv(a_new)))
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.3
            best_it = it
    
    if verbose:
        print(f"    GD: obj={2*n*best_obj:.8f}, iters={it}")
    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set LP for fast direction finding, multiplicative weights as
alternative descent, convex line search, and gradient polishing.
Explores multiple sequence lengths to find the global optimum.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_rows(a, indices):
    """Build rows of convolution matrix for specified row indices only."""
    n = len(a)
    k_idx = np.asarray(indices, dtype=np.int64)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((len(indices), n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_sub(a, row_indices):
    """Solve LP: min M s.t. A[k,:] @ g <= M for k in row_indices, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A_sub = build_conv_rows(a_norm, row_indices)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_sub @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_with_active_set(a, max_iters=200, verbose=False):
    """
    Solve the full LP using active set method.
    Returns optimal g.
    """
    n = len(a)
    all_idx = list(range(2 * n - 1))
    active = all_idx[:]

    for outer in range(max_iters):
        g, M = solve_lp_sub(a, active)
        if g is None:
            # Fallback to full problem
            g, M = solve_lp_sub(a, all_idx)
            if g is None:
                break

        # Check feasibility against full constraint
        conv_g = compute_conv(g)
        max_conv = float(np.max(conv_g))

        if max_conv <= M + 1e-7:
            return g, M

        # Add violated constraints
        violated = np.where(conv_g > M + 1e-9)[0]
        active = list(set(active + list(violated)))

        if len(active) >= 2 * n - 1:
            # All constraints active, one more solve should be exact
            g, M = solve_lp_sub(a, all_idx)
            if g is not None:
                return g, M
            break

    return g, M if g is not None else (None, None)


def convex_line_search(a, g, max_evals=35):
    """
    Convex line search: minimize f(t) = max_conv((1-t)*a + t*g) for t in [0,1].
    Uses ternary search exploiting convexity of f in t.
    """
    n = len(a)

    def f(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0)
        s = float(np.sum(a_t))
        if s < 1e-10:
            return np.inf
        a_t = a_t / s
        conv = compute_conv(a_t)
        return float(np.max(conv))

    f0 = f(0.0)
    f1 = f(1.0)

    if f1 >= f0:
        return 0.0, f0

    # Ternary search on [0, 1]
    lo, hi = 0.0, 1.0
    evals = 1
    while hi - lo > 1e-7 and evals < max_evals:
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f_m1 = f(m1)
        f_m2 = f(m2)
        evals += 2
        if f_m1 < f_m2:
            hi = m2
        else:
            lo = m1

    best_t = (lo + hi) / 2.0
    best_f = f(best_t)
    return best_t, best_f

                best_peak = best_f
                best_a = a_new.copy()
                a = a_new.copy()
                stall = 0
            if verbose and (it % 5 == 1 or it <= 3):
                elapsed = time.monotonic() - start
                print(f"    LP it{it}: peak={best_peak:.10f} (obj={2*n*best_peak:.8f}), "
                      f"M={M_sub:.6f}, t={best_t:.4f}, |active|={len(active)}, {elapsed:.0f}s")
        else:
            stall += 1
            if verbose and it % 15 == 0:
                print(f"    LP it{it}: stalled peak={best_peak:.10f}, M={M_sub:.6f}")
            if stall > 15:
                break

    return best_a


def mw_step(a, tau=100.0, eta=None):
    """One multiplicative weights (mirror descent) step."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))

    exp_v = np.exp(tau * (conv - max_c))
    p = exp_v / exp_v.sum()

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad = 2.0 * ifft(P * np.conj(A)).real[:n]

    if eta is None:
        g_max = float(np.max(np.abs(grad)))
        if g_max < 1e-15:
            return a.copy()
        eta = 0.05 / (g_max + 1e-10)

    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 1e-20)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def mw_optimize(a, time_budget=60.0, verbose=False):
    """Optimize using multiplicative weights with increasing tau."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_peak = float(np.max(compute_conv(a)))
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = 30.0 + 500.0 * progress ** 2

        a = mw_step(a, tau=tau)
        peak = float(np.max(compute_conv(a)))
        it += 1

        if peak < best_peak - 1e-11:
            best_peak = peak
            best_a = a.copy()
            if verbose and it % 200 == 0:
                print(f"    MW it{it}: peak={best_peak:.10f} (obj={2*n*best_peak:.8f}), tau={tau:.0f}")

    return best_a


def gd_polish(a, time_budget=60.0, tau_init=200.0, tau_end=5000.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_peak = float(np.max(compute_conv(a)))
    start = time.monotonic()

    v = np.zeros(n)
    beta = 0.9
    base_lr = 3.0 / np.sqrt(n)
    lr = base_lr
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.2

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_c = float(np.max(conv))
        exp_v = np.exp(tau * (conv - max_c))
        p = exp_v / exp_v.sum()

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        Y = fft(y, N)
        grad = 2.0 * ifft(P * np.conj(Y)).real[:n]

        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            lr = min(lr, base_lr / (g_norm / np.sqrt(n) + 0.5))
        lr = max(lr * 0.9995, 1e-10)

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
        else:
            a_new = a_new / s_new

        peak = float(np.max(compute_conv(a_new)))
        if peak < best_peak - 1e-12:
            best_peak = peak
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it

    if verbose:
        print(f"    GD: peak={best_peak:.10f} (obj={2*n*best_peak:.8f}), iters={it}")
    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def raised_cosine(n):
    """Raised cosine profile as initial guess."""
    x = np.linspace(0, np.pi, n)
    return 0.5 + 0.5 * np.cos(x)

                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}, peak={np.max(compute_conv(parent)):.10f}")
    except Exception:
        best_seq = raised_cosine(800)
        best_seq = best_seq / best_seq.sum()
        best_score = compute_obj(best_seq)
        print(f"Default raised_cosine (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Explore n values with quick optimization ===
    print(f"\n=== Phase 1: n-exploration ===")
    n_candidates = sorted(set([
        max(400, n_parent - 200), max(400, n_parent - 100),
        n_parent,
        n_parent + 100, n_parent + 200, n_parent + 400,
        600, 700, 900, 1000, 1200, 1500, 2000, 2500
    ]))
    n_candidates = [x for x in n_candidates if 400 <= x <= 3000]

    # Budget per n: aim to try 8-10 values in ~200s
    budget_per_n = 20.0
    n_tried = 0
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_tried >= 10:
            break

        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            # Use raised cosine for new n, resample for nearby n
            if abs(n_try - n_parent) <= 100:
                a_try = resample(best_seq, n_try)
            else:
                a_try = raised_cosine(n_try)
                a_try = a_try / a_try.sum()

        # Quick LP
        alloc = min(budget_per_n, remaining * 0.12)
        if alloc > 5:
            a_try = lp_optimize(a_try, time_budget=alloc, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} (peak={np.max(compute_conv(best_seq)):.10f})")
        else:
            print(f"     n={n_try}: obj={obj_try:.8f}")
        n_tried += 1

    # === Phase 2: Main intensive LP optimization ===
    print(f"\n=== Phase 2: Main LP (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 450.0)
    if p2_budget > 20:
        a_lp = lp_optimize(best_seq, time_budget=p2_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f} (no improvement over {best_score:.8f})")

    # === Phase 3: Multiplicative weights ===
    print(f"\n=== Phase 3: Multiplicative Weights ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 80.0)
    if p3_budget > 10:

            best_seq = a_lp2.copy()
            print(f"  Phase4 LP: obj={best_score:.8f} **")

            print(f"  Phase4 LP: obj={obj_lp2:.8f}")

        obj_gd = compute_obj(a_gd)

            print(f"  Phase5 GD: obj={best_score:.8f} **")

    n_cur = len(best_seq)

    n_adjusts = [x for x in n_adjusts if 300 <= x <= 3000]

    for n_adj in n_adjusts:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        a_adj = resample(best_seq, n_adj)
        alloc = min(remaining * 0.15, 30.0)
        if alloc > 8:
            a_adj = lp_optimize(a_adj, time_budget=alloc, verbose=False)
        obj_adj = compute_obj(a_adj)
        if obj_adj < best_score:

    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
