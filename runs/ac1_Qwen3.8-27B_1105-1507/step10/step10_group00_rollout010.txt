    """Subgradient of conv(a,a)_{k_star} w.r.t. a"""
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        m1 = k_star - j
        m2 = j - k_star
        if 0 <= m1 < n:
            g[j] += a[m1]
        if 0 <= m2 < n:
            g[j] += a[m2]

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = obj_norm(a)
    x = a.copy()
    prev_x = a.copy()
    t = 1.0

    lr = 1.0
    
    it = 0
    while time.monotonic() - start < time_budget:
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2

        momentum = (t - 1) / t_new
        y = x + momentum * (x - prev_x)
        y = np.maximum(y, 0)
        sy = np.sum(y)

def simplex_project(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    # Standard algorithm
    if np.sum(v) <= 0:
        return np.ones(n) / n
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.where(mu * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

"""
Optimization for step function coefficients using:
1. Smoothed gradient (log-sum-exp) with FFT-accelerated gradient computation
2. Nesterov accelerated gradient descent on the probability simplex
3. Coordinate descent refinement
4. LP polishing for final improvement
5. Adaptive length search

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv_full(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_full(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_max_grad(a, eps):
    """
    Compute smoothed max of conv(a,a) and its gradient.
    f_eps = (1/eps) * log(sum_k exp(eps * conv_k))
    grad_j = (a*p)[j] + (a*p)[j+n-1] where p_k = softmax(eps * conv)
    """
    n = len(a)
    conv = compute_conv_full(a)
    
    # Compute p = softmax(eps * conv)
    max_c = np.max(conv)
    shifted = eps * (conv - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    p = exp_shifted / sum_exp
    
    # Smoothed objective value (up to the 2n factor)
    f_eps = (max_c + np.log(sum_exp)) / eps
    
    # Compute gradient: grad[j] = (a*p)[j] + (a*p)[j+n-1]
    # where a*p is the linear convolution of a (length n) and p (length 2n-1)
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fp = rfft(p, size)
    ap = irfft(fa * fp, size)  # length size, first 3n-2 are the linear conv
    
    # ap[k] = sum_m a[m] * p[k-m] for k = 0, ..., 3n-3
    # grad[j] = ap[j] + ap[j + n - 1]
    grad = ap[:n] + ap[n-1:2*n-1]
    
    return f_eps, grad, p


def simplex_project(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    
    # Handle edge case
    if np.max(v) <= 0:
        return np.ones(n) / n
    
    # Standard simplex projection algorithm
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    cond = mu * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    return result


def nesterov_smoothed_descent(a_init, time_budget, verbose=False):
    """
    Nesterov accelerated gradient descent on the simplex
    for the smoothed objective.
    """
    n = len(a_init)
    a = simplex_project(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    x = a.copy()
    prev_x = a.copy()
    t = 1.0
    eps = 0.5  # Start with moderate smoothing
    start = time.monotonic()
    it = 0
    stall = 0
    lr_base = 1.0
    
    while time.monotonic() - start < time_budget:
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = x + momentum * (x - prev_x)
        y = simplex_project(y)
        
        # Compute smoothed gradient
        f_eps, grad, p = smoothed_max_grad(y, eps)
        
        # Gradient step
        step = lr_base / t_new
        x_new = y - step * grad
        x_new = simplex_project(x_new)
        
        # Evaluate true objective
        obj_new = compute_obj(x_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        prev_x = x
        x = x_new
        t = t_new
        
        # Adapt epsilon: decrease it over time
        if it % 200 == 0:
            eps *= 0.9
            t = 1.0  # Reset Nesterov momentum
            x = best_a.copy()
            prev_x = best_a.copy()
            lr_base = 2.0 / np.sqrt(it + 1)
        
        # Reset if stalled
        if stall > 1000:
            x = best_a.copy()
            prev_x = best_a.copy()
            t = 1.0
            eps *= 0.7
            lr_base = 1.0
            stall = 0
        
        it += 1
        if verbose and it % 2000 == 0:
            print(f"      NSD it{it}: obj={best_obj:.8f}, eps={eps:.4f}, t={time.monotonic()-start:.1f}s")
    
    return best_a

            break

                    break

            k_max = int(np.argmax(conv))
            numerator = conv[k_max] + 0 * d[k_max]  # = conv[k_max]

            phi = (1 + np.sqrt(5)) / 2

            c = lo + gr * (hi - lo)

                    return np.inf
                new_conv = conv + delta * d
                S_new = 1.0 + delta
                if S_new < 0.01:
                    return np.inf
                return 2.0 * n * np.max(new_conv) / (S_new * S_new)
            
            fc = obj_delta(c)
            fd = obj_delta(dd)

                    hi = dd
                    dd = c
                    fd = fc
                    c = lo + gr * (hi - lo)
                    fc = obj_delta(c)
                else:
                    lo = c
                    c = dd
                    fc = fd
                    dd = lo + gr * (hi - lo)
                    fd = obj_delta(dd)

                a = best_a.copy()
                s = 1.0
            a = a / s
            
            # Update conv incrementally (or recompute)
            # For simplicity, recompute every n/10 steps
            if j % max(1, n // 10) == 0:
                conv = compute_conv_full(a)
        
        # Evaluate after sweep
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if verbose:
            print(f"      CD sweep {sweep}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    
    return best_a


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_polish(a, time_budget=60.0, max_iters=30):
    """LP-based polishing with blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start >= time_budget - 5:
            break
        
        g_lp = solve_lp(a)
        if g_lp is None:
            break
        
        # Try different blend factors
        best_t = 0
        best_blend_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 1.0]:
            blend = (1 - t) * a + t * g_lp
            blend = np.maximum(blend, 0)
            s_b = float(np.sum(blend))
            if s_b > 0.01:
                blend = blend / s_b
                obj_b = compute_obj(blend)
                if obj_b < best_blend_obj:
                    best_blend_obj = obj_b
                    best_t = t
        
        if best_t > 0 and best_blend_obj < best_obj - 1e-14:
            a = (1 - best_t) * a + best_t * g_lp
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_obj = best_blend_obj
            best_a = a.copy()
        else:
            # Perturb and try again
            rng = np.random.default_rng(it + 777)
            noise = rng.normal(0, 0.01, n)
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a


def arcsine_init(n, eps=0.5):
    """Arcsine distribution initialization."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(np.maximum(i + eps, 0.1) * np.maximum(n - 1 - i + eps, 0.1))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s


def power_u_init(n, power=2.0, base=0.05):
    """Power-law U-shaped initialization."""
    i = np.arange(n, dtype=np.float64)
    x = i / max(1, n - 1)
    a = (2 * x - 1) ** power + base
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s

        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
