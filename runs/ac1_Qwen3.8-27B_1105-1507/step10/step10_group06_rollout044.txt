    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            if solver == cp.CLARABEL:
                prob.solve(solver=solver, tol_gap_abs=1e-10, tol_gap_rel=1e-10,
                          tol_feas=1e-10, tol_infeas_abs=1e-10, tol_infeas_rel=1e-10)
            else:
                prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def fft_gradient(y, tau):
    """
    Compute gradient of smoothed max of conv(y,y) using FFT.
    Returns gradient of shape (n,) and the convolution.
    """
    n = len(y)
    conv = compute_conv(y)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    exp_vals = np.exp(np.clip(shifted, -700, 0))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    Ay = fft(y, N)
    P = fft(p, N)
    g_full = ifft(Ay * np.conj(P)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_polish(a, time_budget, tau_init=30.0, tau_max=3000.0, verbose=False):
    """
    Nesterov accelerated gradient descent on smoothed objective with FFT gradients.
    Projects onto {x >= 0, sum(x) = 1}.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 3 * n:
        N <<= 1

    beta = 0.9
    base_lr = 1.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = 1.0
        else:
            y = y / s_y

        g, conv = fft_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        # Adaptive learning rate
        if g_norm > 1e-15:
            adaptive_lr = 1.0 / (g_norm * tau + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.05 * base_lr), 5.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adaptive_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        it += 1
        if it % 50 == 0:
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
                if verbose and it % 5000 == 0:
                    elapsed = time.monotonic() - start
                    print(f"    Nesterov: it={it}, obj={best_obj:.10f}, "
                          f"tau={tau:.0f}, t={elapsed:.1f}s")
            else:
                no_improve += 1

        a = a_new
        tau = min(tau * 1.01, tau_max)

        # Restart if stuck
        if no_improve > 30:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            no_improve = 0

    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 8
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_alternate_with_gradient(a, time_budget, seed=42, verbose=True,
                                gradient_steps_per_lp=200, tau_init=50.0):
    """
    LP alternating minimization with interleaved gradient steps.
    Between LP calls, performs gradient steps to make faster progress.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    t_blend = 0.4
    stall_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # --- Step 1: Solve LP for best response ---
        t0 = time.monotonic()
        g_lp, M_lp = solve_conv_lp(a)
        lp_time = time.monotonic() - t0

        if g_lp is None:
            break

        iter_count += 1

        # --- Step 2: Try blending factors ---
        candidates_t = [t_blend, t_blend * 0.5, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if iter_count > 100:
            candidates_t.extend([0.05, 0.02, 0.01])

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend

        for t in candidates_t:
            t = min(max(t, 0.001), 1.0)
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            t_blend = max(0.01, best_local_t * 0.95)
            stall_count = 0
            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP alt {iter_count}: obj={best_obj:.10f}, t={t_blend:.4f}, "
                      f"lp_t={lp_time:.2f}s, elapsed={elapsed:.0f}s")
        else:
            stall_count += 1
            if stall_count >= 5:
                strength = 0.005 + 0.003 * min(stall_count, 15)
                a = fourier_perturb(best_a, rng, strength=strength)
                t_blend = max(0.1, min(t_blend * 1.3, 0.6))
                stall_count = 0
            elif stall_count >= 2:
                a = best_a + 0.003 * rng.standard_normal(n) * np.maximum(best_a, 0.01)
                a = np.maximum(a, 0)
                a = a / a.sum()

        # --- Step 3: Interleaved gradient steps (if time allows) ---
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            grad_budget = min(remaining * 0.15, 2.0)
            if grad_budget > 0.5:
                a_grad = nesterov_polish(a, time_budget=grad_budget,
                                         tau_init=tau_init, tau_max=2000.0,
                                         verbose=False)
                obj_grad = compute_obj(a_grad)
                if obj_grad < best_obj:
                    best_obj = obj_grad
                    best_a = a_grad.copy()
                    a = a_grad.copy()
                    if verbose:
                        print(f"      Gradient improved: obj={best_obj:.10f}")

    if verbose:
        print(f"    LP+Grad: {iter_count} LP iters, best_obj={best_obj:.10f}")

    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def cosine_init(n):
    """Initialize with cosine window."""
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2) ** 2
    return a / a.sum()


def triangular_init(n):
    """Initialize with triangular (Bartlett) window."""
    mid = n // 2
    if n % 2 == 0:
        a = np.concatenate([np.linspace(0, 1, mid), np.linspace(1, 0, mid)])
    else:
        a = np.concatenate([np.linspace(0, 1, mid + 1), np.linspace(1, 0, mid)])
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def power_init(n, power=2.0):
    """Initialize with power-law distribution."""
    positions = np.arange(n)
    mid = (n - 1) / 2.0
    dist = np.abs(positions - mid) / mid
    a = (1.0 + 0.5 * (1 - dist)) ** power
    return a / a.sum()


def random_structured_init(n, rng):
    """Random initialization with low-frequency structure."""
    a = np.ones(n)
    for k in range(1, 6):
        freq = k * 2 * np.pi / n
        amp = rng.uniform(0.1, 0.4) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(freq * np.arange(n) + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()

        use_parent = True

        print(f"Default init: n={n_init}, obj={compute_obj(parent):.10f}")
        use_parent = False

    best_a = parent.copy()
    best_obj = compute_obj(parent)
    print(f"Starting best: obj={best_obj:.10f}")

    # === Phase 1: Main LP+Gradient optimization on parent ===
    print(f"\n=== Phase 1: Main LP+Gradient (n={n_init}) ===")

    if obj_p1 < best_obj:
        best_obj = obj_p1

        print(f"  Phase 1 result: obj={obj_p1:.10f} (no improvement)")

    print(f"\n=== Phase 2: Second LP Pass ===")

    # === Phase 3: Gradient polish ===

                                tau_init=100.0, tau_max=5000.0, verbose=True)

        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3.copy()
            print(f"  Phase 3 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 3 result: obj={obj_p3:.10f}")

    # === Phase 4: Try different n values ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    n_options = []
    for delta in [-100, -50, -20, 20, 50, 100, 150, 200]:
