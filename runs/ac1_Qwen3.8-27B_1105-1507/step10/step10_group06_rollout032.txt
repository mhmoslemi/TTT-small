import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a, prev_g=None):
    """
    Solve LP: min M s.t. (a_norm * g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    # If we have a previous solution, use it as a warm start hint
    obj = cp.Minimize(M)
    prob = cp.Problem(obj, constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_subgradient(a, tau):
    """
    Compute smoothed subgradient of max conv using log-sum-exp.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    scaled = tau * (conv - max_conv)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    exp_vals = np.exp(scaled)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Find the largest t such that sum(max(v - t, 0)) = 1
    # Use bisection
    lo, hi = 0.0, max(v.max(), v.sum())
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    result = np.maximum(v - t, 0)
    s = result.sum()
    if s > 0:
        result = result / s
    else:
        result = np.ones(n) / n
    return result


def nesterov_subgradient(a, time_budget, tau_init=50.0, tau_max=5000.0, 
                          momentum_init=0.5, verbose=False):
    """
    Nesterov accelerated subgradient method on the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = momentum_init
    base_lr = 1.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = a + momentum * v
        y = project_simplex(y)

        # Compute smoothed subgradient
        g = smoothed_subgradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm < 1e-15:
            break

        # Update
        v = momentum * v - lr * g
        a_new = y + v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        
        # Adaptive parameters
        lr *= 0.999
        tau = min(tau * 1.03, tau_max)
        
        # Occasionally increase momentum (for faster convergence)
        if it % 100 == 0:
            momentum = min(0.95, momentum * 1.005)

        # Anneal and restart if stuck
        if no_improve > 200:
            a = best_a + 0.01 * np.random.standard_normal(n) * best_a
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            momentum = momentum_init
            lr = base_lr
            tau = tau_init * 0.5
            no_improve = 0

        it += 1

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"      SG it={it}, obj={best_obj:.10f}, tau={tau:.0f}, lr={lr:.6f}, t={elapsed:.1f}s")

    return best_a

def support_prune_and_refine(a, time_budget, verbose=False):

def asymmetric_init(n, seed, style):
    """Generate asymmetric initializations."""
    rng = np.random.default_rng(seed)
    
    if style == 0:
        # Linear ramp
        a = np.linspace(0.1, 1.0, n)
    elif style == 1:
        # Quadratic
        x = np.linspace(0, 1, n)
        a = x ** 2
    elif style == 2:
        # Sine-based
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif style == 3:
        # Two-peak
        x = np.linspace(0, 1, n)
        a = np.exp(-20*(x-0.3)**2) + 0.7*np.exp(-20*(x-0.7)**2)
    elif style == 4:
        # Random smooth
        a = rng.dirichlet(np.ones(n) * n * 0.5)
    elif style == 5:
        # Cosine window (asymmetric)
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 1.5
    else:
        # Exponential
        a = np.exp(-np.linspace(0, 3, n))
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a multi-phase approach:
1. Parametric asymmetric initialization to find good starting points
2. LP-based alternating minimization with sparse matrices and adaptive blending
3. Nesterov subgradient polishing with adaptive smoothing
4. Support-set pruning and re-optimization
5. Multi-start diversification and length exploration

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a):
    """Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    from scipy.sparse import lil_matrix
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A.tocsr()


def solve_conv_lp(a, max_time=10.0):
    """
    Solve LP: min M s.t. (a_norm * g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt, status) or (None, None, 'fail') on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_sparse(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > max_time:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value), 'ok'
        except Exception:
            continue
    return None, None, 'fail'


def smoothed_subgradient(a, tau):
    """Compute smoothed subgradient of max conv using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    scaled = tau * (conv - max_conv)
    scaled = np.clip(scaled, -400, 400)
    exp_vals = np.exp(scaled)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A_fft = fft(a, N)
    g_full = ifft(P * np.conj(A_fft)).real
    g = 2.0 * g_full[:n]
    return g


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    lo, hi = 0.0, float(v.max())
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    result = np.maximum(v - t, 0)
    s = result.sum()
    if s > 0:
        result = result / s
    else:
        result = np.ones(n) / n
    return result


def nesterov_subgradient(a, time_budget, tau_init=50.0, tau_max=5000.0,

    v = np.zeros(n)
    tau = tau_init
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:

        y = project_simplex(y)

        g_norm = float(np.linalg.norm(g))

def parametric_init(n, p, q, r):
    """Generate parametric asymmetric sequence: (i+1)^p * (n-1-i)^q * exp(-r*i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    denom = max(n - 1, 1)
    terms = np.ones(n)
    if abs(p) > 1e-10:
        terms = terms * np.maximum(i + 1, 0.1) ** p
    if abs(q) > 1e-10:
        terms = terms * np.maximum(denom - i, 0.1) ** q
    if abs(r) > 1e-10:
        terms = terms * np.exp(-r * i / denom)
    terms = np.maximum(terms, 0)
    s = terms.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return terms / s


def optimize_parametric(n, time_budget):
    """Optimize over the parametric family (p, q, r) to find a good init."""
    best_obj = np.inf
    best_params = (0, 0, 0)

    # Grid search over a few values
    p_vals = [0.2, 0.5, 0.8, 1.0, 1.5, 2.0]
    q_vals = [-1.0, -0.5, -0.2, 0.0, 0.2, 0.5, 1.0]
    r_vals = [-0.5, 0.0, 0.5, 1.0]

    start = time.monotonic()
    for p in p_vals:
        for q in q_vals:
            for r in r_vals:
                if time.monotonic() - start > time_budget:
                    break
                a = parametric_init(n, p, q, r)
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_params = (p, q, r)

    # Local refinement
    def param_obj(params):
        p, q, r = params
        a = parametric_init(n, p, q, r)
        return compute_obj(a)

    res = scipy_minimize(param_obj, best_params, method='Nelder-Mead',
                         options={'maxiter': 200, 'xatol': 0.01, 'fatol': 1e-7})
    if res.fun < best_obj:
        best_obj = res.fun
        best_params = res.x

    a_best = parametric_init(n, *best_params)
    return a_best, best_obj, best_params

def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.03):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

        max(400, n_best - 200), min(1200, n_best + 200)
    ]))
    n_options = [n for n in n_options if n != n_best]

    for n_try in n_options[:3]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_to_length(best_a, n_try)
        alloc = min(remaining * 0.25, 150.0)

        a_try = lp_alternate(a_try, time_budget=alloc * 0.6, seed=n_try, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_try = nesterov_subgradient(a_try, time_budget=min(remaining * 0.3, 60.0),
                                          tau_init=100.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  n={n_try}: obj={best_obj:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 8: Final combined polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.5, 120.0)
        a_final = lp_alternate(best_a, time_budget=alloc * 0.5,
                                seed=seed + 99999, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_final = nesterov_subgradient(a_final, time_budget=min(remaining * 0.4, 50.0),
                                            tau_init=200.0, tau_max=3000.0)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final: obj={best_obj:.10f} **")
        else:
            print(f"  Final: obj={obj_final:.10f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
