"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based fixed-point iteration with Nesterov momentum, adaptive blending,
high-tau smoothed GD for fine-tuning, and multi-n exploration.
Key innovation: proximal-style LP iteration with momentum acceleration and
adaptive step scheduling for faster convergence to the minimax optimum.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad_fft(a, tau):
    """Compute smoothed gradient of max conv(a,a) using log-sum-exp + FFT.
    Returns (gradient, exact_max)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    # grad[m] = 2 * sum_k p[k] * a[k-m]
    # = 2 * ifft(conj(FFT(p)) * FFT(a)).real[:n]
    N = 1
    while N < 2 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad = ifft(np.conj(P) * A).real[:n] * 2.0
    return grad, exact_max


def lp_solve(a, solver_list=None):
    """Solve LP: min M s.t. conv(a,g) <= M, g>=0, sum(g)=1.
    Returns (g_optimal, M_optimal) or (None, None)."""
    n = len(a)
    if n > 3000:
        return None, None
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def staircase_gd(a, time_budget, tau_levels=None, beta=0.9, lr_scale=3.0, max_iters_total=50000):
    """Nesterov accelerated GD with staircase tau schedule."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    if tau_levels is None:
        tau_levels = [5, 15, 40, 100, 300, 800, 2000, 5000, 10000]
    it = 0
    best_it = 0
    stall_patience = 8000
    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0
        max_iters_per_level = max(2000, min(20000, max_iters_total // len(tau_levels)))
        while time.monotonic() - start < time_budget:
            if tau_iters >= max_iters_per_level:
                break
            if it >= max_iters_total:
                return best_a
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy
            g, _ = smoothed_grad_fft(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break
            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it
            a = a_new
            it += 1
            tau_iters += 1
            lr = max(1e-8, lr * 0.99999)
            if it - best_it > stall_patience:
                a = best_a.copy()
                v = np.zeros(n)
                lr = lr_scale / np.sqrt(n)
                best_it = it
    return best_a


def lp_momentum_iteration(a_init, time_budget, n_iters_max=300, verbose=True):
    """
    LP fixed-point iteration with Nesterov momentum and adaptive step size.
    This is the main optimization engine.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)

    start = time.monotonic()
    v = np.zeros(n)
    beta = 0.7  # momentum
    t_blend = 0.5  # initial blending
    t_adapt = 1.1  # adaptation factor
    stall_count = 0
    rng = np.random.default_rng(12345)

    for it in range(n_iters_max):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
            sy = 1.0
        y = y / sy

        # Solve LP at y
        t0 = time.monotonic()
        g_lp, M_lp = lp_solve(y)
        t_lp = time.monotonic() - t0
        if g_lp is None:
            break

        # Adaptive blending with line search over a few candidates
        candidates_t = [t_blend, max(0.02, t_blend * 0.5), min(1.0, t_blend * 1.5),
                        0.3, 0.7]
        best_t_obj = best_obj
        best_t = t_blend
        best_a_new = a.copy()
        for tc in candidates_t:
            a_try = (1.0 - tc) * a + tc * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = tc
                best_a_new = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            # Update momentum
            v = beta * v + (best_a_new - a)
            a = best_a_new
            best_obj = best_t_obj
            best_a = a.copy()
            stall_count = 0
            # Adapt step size (increase if improving well)
            t_blend = min(1.0, t_blend * t_adapt)
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"      LP-mom it{it}: obj={best_obj:.10f}, t={best_t:.3f}, LP_time={t_lp:.1f}s, elapsed={elapsed:.0f}s")
        else:
            # No improvement: decrease step, perturb
            t_blend = max(0.02, t_blend * 0.5)
            stall_count += 1
            if stall_count > 5:
                # Perturb to escape
                pert = rng.normal(0, 0.01, n)
                a = a + pert
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                v = np.zeros(n)
                beta = max(0.3, beta * 0.9)
                stall_count = 0
                if stall_count > 20:
                    break
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"      LP-mom it{it}: stall, obj={best_obj:.10f}, t={t_blend:.4f}, elapsed={elapsed:.0f}s")

    return best_a

def resample(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_diverse_init(n, mode, rng=None):

            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Ultra-high tau GD
    if remaining > 30:
        alloc = min(remaining * 0.2, 60.0)
        a_f1 = staircase_gd(best_seq, time_budget=alloc,
                             tau_levels=[500, 2000, 8000, 20000],
                             beta=0.94, lr_scale=5.0)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Ultra GD: obj={best_score:.10f} **")
        else:
            print(f"  Ultra GD: obj={obj_f1:.10f}")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 20 and len(best_seq) <= 2500:
        alloc = min(remaining * 0.4, 80.0)
        a_f2 = lp_momentum_iteration(best_seq, time_budget=alloc,
                                      n_iters_max=100, verbose=False)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final LP: obj={best_score:.10f} **")
        else:
            print(f"  Final LP: obj={obj_f2:.10f}")

    # One more GD pass
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 40.0)
        a_f3 = fast_subgradient_descent(best_seq, time_budget=alloc, tau=2000)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final FSD: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
