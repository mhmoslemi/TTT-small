Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set LP, entropy mirror descent with adaptive smoothing,
center-focused constraint reduction, multi-length exploration, and
symmetry exploitation for fine-tuning.

import numpy as np
import time
import math
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

            log_ratios = -eta_try * g
            log_ratios -= np.max(log_ratios)
            ratios = np.exp(log_ratios)
            a_try = a * ratios
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_a_try = a_try.copy()
                best_eta_try = eta_try

        if best_obj_try < best_obj - 1e-14:
            best_obj = best_obj_try
            best_a = best_a_try
            a = best_a_try
            eta = best_eta_try
            best_eta = eta
            stagnant = 0
        else:
            # Accept the best candidate anyway (non-monotone but helps escape)
            a = best_a_try
            eta = max(0.01, best_eta_try * 0.8)
            stagnant += 1

        if it % 200 == 0 and verbose:
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, eta={eta:.4f}, stag={stagnant}")

        if stagnant > 300:
            # Restart from best with perturbation
            a = best_a + 0.01 * np.random.default_rng(it).normal(0, 1, n)
            a = project_simplex(a)
            stagnant = 0
            eta = best_eta

        it += 1

    return best_a


def lp_iteration_active(a, time_budget, verbose=False, seed=42):
    """
    LP iteration with active set management and line search.
    Focuses on center constraints for efficiency.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Initialize active set: center 40% of the convolution
    m = 2 * n - 1
    center = n - 1
    half_width = max(30, n // 3)
    k_lo = max(0, center - half_width)
    k_hi = min(m - 1, center + half_width)
    k_indices = np.arange(k_lo, k_hi + 1)

    it = 0
    stagnant = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Solve LP with current active set
        g_lp, M_lp = solve_lp_active(a, k_indices, max_time=min(6.0, remaining * 0.4))
        if g_lp is None:
            break

        it += 1

        # Check if new peaks are active outside current set
        conv_check, M_check = get_active_indices(a, g_lp, M_tol=1e-5)
        new_active = np.setdiff1d(conv_check, k_indices)
        if len(new_active) > 0:
            k_indices = np.sort(np.concatenate([k_indices, new_active]))
            # Cap the active set size
            if len(k_indices) > m:
                k_indices = np.arange(m)

        # Line search: find best blend t
        t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        best_t_obj = best_obj
        best_t_val = 0.0
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stagnant = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.2f}, "
                      f"M_lp={M_lp:.8f}, |active|={len(k_indices)}")
        else:
            stagnant += 1
            if stagnant > 8:
                # Perturb and restart
                strength = min(0.1, 0.01 * stagnant)
                pert = 1.0 + rng.normal(0, strength, n)
                a = best_a * np.maximum(pert, 0)
                a = project_simplex(a)
                stagnant = 0
                # Reset active set
                k_indices = np.arange(k_lo, k_hi + 1)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def symmetrize(a):
    """Make sequence mirror-symmetric."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = float(np.sum(a_sym))
    if s < 1e-15:
        return a.copy()
    return a_sym / s


def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(4, n // 20)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.0) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    s = float(np.sum(a_new))
    if s < 1e-15:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines active-set LP, entropy mirror descent,
    multi-length exploration, and symmetry exploitation.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    # --- Initialize from parent ---
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={len(best_seq)}): obj={best_score:.8f}")
            else:
                raise ValueError("sum too small")
        else:
            raise ValueError("invalid parent")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Active-set LP optimization ===
    print(f"\n=== Phase 1: Active-Set LP (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.1, 100.0)
    if p1_budget > 15:
        a_lp = lp_iteration_active(best_seq, time_budget=p1_budget, verbose=True, seed=42)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 1 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 done: obj={obj_lp:.8f} (no improvement)")
    else:
        print("  Phase 1: skipping (insufficient time)")

    # === Phase 2: Entropy mirror descent ===
    print(f"\n=== Phase 2: Entropy Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 200.0)
    if p2_budget > 15:
        a_md = entropy_mirror_descent(best_seq, time_budget=p2_budget,
                                      tau_init=50.0, tau_end=1000.0,
                                      eta_init=0.3, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Phase 2 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 done: obj={obj_md:.8f} (no improvement)")

    # Try symmetry
    a_sym = symmetrize(best_seq)
    obj_sym = compute_obj(a_sym)
    if obj_sym < best_score:
        best_score = obj_sym
        best_seq = a_sym.copy()
        print(f"  Symmetrized: obj={best_score:.8f} **")

    # Quick mirror descent on symmetric version
    remaining = deadline - time.monotonic()
    if remaining > 100 and obj_sym < compute_obj(best_seq) + 1e-6:
        a_sym2 = entropy_mirror_descent(best_seq, time_budget=min(60.0, remaining * 0.1),
                                        tau_init=100.0, tau_end=500.0, eta_init=0.2)
        obj_sym2 = compute_obj(a_sym2)
        if obj_sym2 < best_score:
            best_score = obj_sym2
            best_seq = a_sym2.copy()
            print(f"  Sym MD: obj={best_score:.8f} **")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_candidates = []
    for dn in [-200, -100, -50, 50, 100, 200, 300, 500, 700]:
        n_try = n_best + dn
        if 300 <= n_try <= 2000:
            n_candidates.append(n_try)
    for n_abs in [500, 700, 900, 1100, 1300, 1600]:
        if n_abs not in n_candidates and abs(n_abs - n_best) > 100:
            n_candidates.append(n_abs)

    n_candidates = sorted(set(n_candidates))
    # Limit to 6 most promising (closest to current + spread)
    n_candidates = n_candidates[:6]
    print(f"  Exploring n values: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # Quick LP
        alloc_lp = min(remaining * 0.2, 30.0)
        if alloc_lp > 10:
            a_try = lp_iteration_active(a_try, time_budget=alloc_lp, seed=n_try)

        # Quick mirror descent
        remaining = deadline - time.monotonic()
        alloc_md = min(remaining * 0.15, 30.0)
        if alloc_md > 8:
            a_try = entropy_mirror_descent(a_try, time_budget=alloc_md,
                                           tau_init=80.0, tau_end=400.0, eta_init=0.2)

        obj_try = compute_obj(a_try)

        # Also try symmetric
        a_try_sym = symmetrize(a_try)
        obj_try_sym = compute_obj(a_try_sym)

        if obj_try_sym < obj_try:
            obj_try = obj_try_sym
            a_try = a_try_sym

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Diversification restarts ===
    print(f"\n=== Phase 4: Diversification ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        if r == 0:
            a_r = fourier_perturb(best_seq, rng, strength=0.04)
        elif r == 1:
            a_r = 0.8 * best_seq + 0.2 * np.ones(n_cur) / n_cur
            a_r = a_r / a_r.sum()
        else:
            a_r = best_seq * (1.0 + 0.05 * rng.standard_test_sequence(n_cur) if hasattr(rng, 'standard_test_sequence') else rng.random(n_cur))
            a_r = project_simplex(a_r)

        alloc = min(remaining * 0.15, 80.0)
        if alloc > 15:
            a_r = lp_iteration_active(a_r, time_budget=alloc * 0.5, seed=500 + r * 10)
            a_r = entropy_mirror_descent(a_r, time_budget=alloc * 0.5,
                                         tau_init=100.0, tau_end=600.0, eta_init=0.2)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    print(f"  Final polishing n={n_final}, remaining={remaining:.0f}s")

    # Extended LP
    if remaining > 40:
        alloc = min(remaining * 0.35, 250.0)
        a_fin = lp_iteration_active(best_seq, time_budget=alloc, verbose=True, seed=9999)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_fin:.8f}")

    # Extended mirror descent with high tau
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.35, 200.0)
        a_fin = entropy_mirror_descent(best_seq, time_budget=alloc,
                                       tau_init=200.0, tau_end=1500.0,
                                       eta_init=0.1, verbose=True)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final MD: obj={best_score:.8f} **")
        else:
            print(f"  Final MD: obj={obj_fin:.8f}")

    # One more LP if time
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 150.0)
        a_fin = lp_iteration_active(best_seq, time_budget=alloc, verbose=False, seed=7777)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Ultra-final LP: obj={best_score:.8f} **")

    # Final symmetry check
    a_sym_final = symmetrize(best_seq)
    obj_sym_final = compute_obj(a_sym_final)
    if obj_sym_final < best_score:
        best_score = obj_sym_final
        best_seq = a_sym_final.copy()
        print(f"  Final sym: obj={best_score:.8f} **")

    # Final micro-polish with very high tau
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 60.0)
        a_fin = entropy_mirror_descent(best_seq, time_budget=alloc,
                                       tau_init=500.0, tau_end=3000.0,
                                       eta_init=0.05, verbose=False)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Micro-polish: obj={best_score:.8f} **")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
