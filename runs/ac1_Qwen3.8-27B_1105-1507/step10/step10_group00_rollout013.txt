def compute_conv(a):

    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

    if s < 0.01:

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

    max_c = np.max(conv)
    exp_c = np.exp((conv - max_c) / tau)
    sum_exp = np.sum(exp_c)
    obj = 2.0 * n * tau * (np.log(sum_exp) + max_c / tau)
    softmax_w = exp_c / sum_exp
    
    # Gradient
    grad = np.zeros(n)
    for k in range(len(softmax_w)):
        w = softmax_w[k]
        if w < 1e-15:
            continue
        # First part: a[k-i] for valid i
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += w * a[k-i_hi:k-i_lo+1][::-1]
        # Second part: a[i-k] for valid i
        i_lo2 = max(0, k)
        i_hi2 = min(n - 1, k + n - 1)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2+1] += w * a[0:i_hi2-k+1]
    
    grad *= 2.0 * n  # Chain rule factor
    return obj, grad

def fista_optimize(a, time_budget=60.0):
    """FISTA optimization on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Start with a relatively large tau and decrease
    tau = 0.01
    a_curr = a.copy()
    a_prev = a.copy()
    t_cur = 1.0
    
    while time.monotonic() - start < time_budget:
        # FISTA update
        t_next = (1 + np.sqrt(1 + 4*t_cur*t_cur)) / 2
        y = a_curr + (t_cur - 1) / t_next * (a_curr - a_prev)
        y = np.maximum(y, 0)
        s_y = np.sum(y)
        if s_y < 1e-10:
            y = a_curr.copy()
            s_y = 1.0
        else:
            y = y / s_y
        
        # Compute gradient at y
        obj_y, grad_y = soft_max_conv_grad(y, tau)
        
        # Backtracking line search
        step = 1.0 / (4.0 * n * n / tau)  # Initial guess
        for _ in range(20):
            a_new = project_simplex(y - step * grad_y)
            obj_new, _ = soft_max_conv_grad(a_new, tau)
            obj_y_eval, _ = soft_max_conv_grad(y, tau)
            if obj_new <= obj_y_eval + 0.5 * step * step * np.sum(grad_y * (a_new - y)):
                break
            step *= 0.5
        
        a_prev = a_curr.copy()
        a_curr = a_new
        t_cur = t_next
        
        # Update best
        obj_curr = compute_obj(a_curr)
        if obj_curr < best_obj:
            best_obj = obj_curr
            best_a = a_curr.copy()
        
        # Decrease tau
        tau *= 0.98
        if tau < 0.001:
            tau = 0.001
    
    return best_a
