"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:
1. Weiss-Wolf style spectral peak reduction (fast, FFT-based)
2. LP-driven directional optimization with line search
3. FISTA-accelerated smoothed gradient descent
4. Aggressive multi-n exploration
5. Diverse restart strategies
"""

import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft
from scipy.ndimage import uniform_filter1d

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def weiss_wolf_optimize(a, time_budget=200.0, n_iter_max=50000, verbose=False):
    """
    Weiss-Wolf style spectral peak reduction.
    Zero-pads a, computes autocorrelation via FFT, applies spectral notches
    at peak lags, inverse transforms, and projects onto non-negative simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # FFT size
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    # Use a larger size for more zero-padding (helps with spectral resolution)
    N = N * 2  # Double zero-padding

    # Precompute frequency indices
    freq_idx = np.arange(N)

    iter_count = 0
    best_iter = 0
    stall_count = 0

    # Adaptive step size
    alpha = 0.1
    alpha_decay = 0.9995

    while time.monotonic() - start < time_budget and iter_count < n_iter_max:
        # Zero-pad and FFT
        a_padded = np.zeros(N)
        a_padded[:n] = a
        A = fft(a_padded)

        # Compute autocorrelation
        R = ifft(np.abs(A) ** 2).real

        # Extract aperiodic autocorrelation (lags 0 to n-1)
        r = R[:n]  # r[0], r[1], ..., r[n-1]

        # Find peak lags (top few)
        max_r = float(np.max(r))
        # Identify lags where r is close to max
        threshold = max_r * 0.95
        peak_lags = np.where(r >= threshold)[0]

        if len(peak_lags) == 0:
            peak_lags = np.array([int(np.argmax(r))])

        # Build spectral notch function
        # We want to reduce energy at frequencies that contribute to peak lags
        # The contribution of frequency m to lag k is: exp(2j*pi*m*k/N)
        # So we reduce A[m] in proportion to the "peak lag correlation"
        notch = np.ones(N, dtype=np.complex128)
        for k in peak_lags:
            if k == 0:
                # Lag 0 is special: it's sum a[j]^2, contributed equally by all frequencies
                notch -= alpha * 0.1
            else:
                notch -= alpha * np.exp(-2j * np.pi * freq_idx * k / N)

        A_new = A * notch
        a_new_padded = ifft(A_new).real
        a_new = a_new_padded[:n]

        # Project onto non-negative simplex
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        # Evaluate
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_iter = iter_count
            stall_count = 0
            # Increase step size slightly if improving
            alpha = min(alpha * 1.02, 0.5)
        else:
            stall_count += 1
            # Decrease step size
            alpha *= 0.9
            if alpha < 0.005:
                # Reset with perturbation
                alpha = 0.1
                a_new = a + 0.01 * np.random.standard_normal(n)
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()

        # Check for stagnation
        if iter_count - best_iter > 5000:
            # Perturb and restart
            rng = np.random.default_rng(iter_count)
            a = a * np.exp(0.05 * rng.standard_normal(n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            alpha = 0.15
            best_iter = iter_count

        a = a_new if obj_new < best_obj else a
        iter_count += 1
        alpha *= alpha_decay

        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    WX it{iter_count}: obj={best_obj:.8f}, alpha={alpha:.4f}, time={elapsed:.0f}s")

    return best_a


def lp_solve_direction(a):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    if not HAS_CVXPY:
        return None, None

    n = len(a)
    a_norm = a / np.sum(a)

    # Build convolution matrix
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A_mat = np.zeros((2 * n - 1, n))
    A_mat[valid] = a_norm[diff[valid]]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, direction, n_evals=40):
    """Line search: find best t in (0,1) to minimize obj((1-t)*a + t*direction)."""
    n = len(a)
    ts = np.linspace(0.005, 0.995, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            a_try = a_try / s
            objs[i] = compute_obj(a_try)

    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]

    # Parabolic refinement
    if 0 < best_idx < n_evals - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.005, 0.995))
            a_par = (1 - t_par) * a + t_par * direction
            a_par = np.maximum(a_par, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                a_par = a_par / s_par
                f_par = compute_obj(a_par)
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par

    return best_t, best_obj


def lp_driven_optimize(a, time_budget=300.0, verbose=False):
    """LP-driven optimization loop."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iter_count = 0
    consecutive_no_improve = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        g, M = lp_solve_direction(a)
        if g is None:
            break

        best_t, obj_at_t = line_search(a, g, n_evals=35)
        iter_count += 1

        if obj_at_t < best_obj - 1e-12:
            a_try = (1 - best_t) * a + best_t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                best_obj = obj_at_t
                best_a = a_try.copy()
                a = a_try
                stall_count = 0
                consecutive_no_improve = 0
                if verbose and (iter_count % 5 == 0 or iter_count == 1):
                    elapsed = time.monotonic() - start
                    print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, time={elapsed:.0f}s")
        else:
            consecutive_no_improve += 1

            # Try smaller steps
            for t_small in [0.1, 0.05, 0.02]:
                a_small = (1 - t_small) * a + t_small * g
                a_small = np.maximum(a_small, 0)
                s_small = float(np.sum(a_small))
                if s_small > 1e-10:
                    a_small = a_small / s_small
                    obj_small = compute_obj(a_small)
                    if obj_small < best_obj - 1e-12:
                        best_obj = obj_small
                        best_a = a_small.copy()
                        a = a_small
                        stall_count = 0
                        consecutive_no_improve = 0
                        if verbose:
                            print(f"    LP it{iter_count}: small step t={t_small}, obj={best_obj:.8f}")
                        break

            stall_count += 1
            if stall_count >= 4:
                # Perturb
                conv_p = compute_conv(a)
                max_p = float(np.max(conv_p))
                tau = 300.0
                exp_p = np.exp(tau * (conv_p - max_p))
                p = exp_p / exp_p.sum()
                N_fft = 1
                while N_fft < 3 * n:
                    N_fft <<= 1
                P = fft(p, N_fft)
                Af = fft(a, N_fft)
                grad = 2.0 * ifft(P * np.conj(Af)).real[:n]
                lr = 0.01 / (np.linalg.norm(grad) + 1e-10)
                a = a - lr * grad
                a = np.maximum(a, 0)
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                else:
                    a = best_a.copy()
                a = a + 0.003 * np.random.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0
                consecutive_no_improve = 0

                if iter_count > 50:
                    break

    return best_a


def fista_smoothed(a, time_budget=200.0, tau_init=100.0, tau_end=3000.0, verbose=False):
    """FISTA-accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    it = 0
    best_it = 0
    stall = 0

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.2

        # FISTA momentum
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = 1.0
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient via FFT
        P = fft(p, N)
        Y = fft(y, N)
        grad = 2.0 * ifft(P * np.conj(Y)).real[:n]

        # Center gradient (project onto sum=0 hyperplane)
        grad = grad - np.mean(grad)

        # Adaptive LR
        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            lr = 0.8 * lr + 0.2 * (base_lr / (g_norm / np.sqrt(n) + 1e-10))
            lr = min(max(lr, 0.1 * base_lr), 5.0 * base_lr)

        # Update
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1

        beta = (1 + np.sqrt(1 + 4 * beta ** 2)) / 2
        # Keep beta bounded
        beta = min(beta, 2.0)

        prev_a = a.copy()
        a = a_new
        it += 1
        lr *= 0.9999

        if it - best_it > 20000 or stall > 3000:
            a = best_a.copy()
            v = np.zeros(n)
            beta = 0.9
            lr = base_lr * 0.5
            best_it = it
            stall = 0

    if verbose:
        print(f"    FISTA: obj={best_obj:.8f}, iters={it}")
    return best_a


def coordinate_descent_2pt(a, time_budget=120.0, verbose=False):
    """
    2-point coordinate descent: transfer mass between pairs (i,j)
    to minimize the objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    # Precompute initial convolution
    c = compute_conv(a)

    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        improved = False

        # Random order of pairs for this sweep
        # Use a subset of pairs for efficiency
        n_pairs = min(n, 200)
        for _ in range(n_pairs):
            time_remaining = time_budget - (time.monotonic() - start)
            if time_remaining < 1.0:
                break

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # Ensure i < j for consistency
            if i > j:
                i, j = j, i

            ai, aj = a[i], a[j]
            if ai < 1e-15 or aj < 1e-15:
                continue

            # Compute the change vector for c[k] as a function of t
            # a[i] -> a[i] + t, a[j] -> a[j] - t
            # delta_c[k] = 2*t*(a[k-i] - a[k-j]) + t^2*(delta_{k,2i} - delta_{k,2j})

            # Precompute the linear coefficients
            # a[k-i] for valid k-i
            lin_i = np.zeros(2 * n - 1)
            lin_j = np.zeros(2 * n - 1)
            valid_i = np.arange(n) + i  # k values where k-i is in [0, n-1]
            lin_i[valid_i] = a[np.arange(n)]
            valid_j = np.arange(n) + j
            lin_j[valid_j] = a[np.arange(n)]

            lin = lin_i - lin_j  # coefficient of 2t

            # Quadratic terms
            quad = np.zeros(2 * n - 1)
            if 2 * i < 2 * n - 1:
                quad[2 * i] = 1.0
            if 2 * j < 2 * n - 1:
                quad[2 * j] = -1.0

            # 1D minimization: f(t) = max_k (c[k] + 2*t*lin[k] + t^2*quad[k])
            # t in (-ai, aj)
            t_lo = -ai + 1e-15
            t_hi = aj - 1e-15

            # Golden section search
            gr = (np.sqrt(5) - 1) / 2  # ~0.618
            t1 = t_hi - gr * (t_hi - t_lo)
            t2 = t_lo + gr * (t_hi - t_lo)

            for _ in range(30):
                f1 = np.max(c + 2 * t1 * lin + t1**2 * quad)
                f2 = np.max(c + 2 * t2 * lin + t2**2 * quad)
                if f1 < f2:
                    t_hi = t2
                    t2 = t1
                    t1 = t_hi - gr * (t_hi - t_lo)
                else:
                    t_lo = t1
                    t1 = t2
                    t2 = t_lo + gr * (t_hi - t_lo)

            t_opt = (t_lo + t_hi) / 2

            # Evaluate at t_opt
            c_new = c + 2 * t_opt * lin + t_opt**2 * quad
            max_new = float(np.max(c_new))
            max_old = float(np.max(c))

            if max_new < max_old * (1 - 1e-10):
                # Apply update
                a[i] += t_opt
                a[j] -= t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                c = compute_conv(a)
                best_a = a.copy()
                best_obj = max_new  # (since sum=1, obj = 2*n*max_conv)
                improved = True
                if best_obj < 2.0 * n * max_new:
                    # Actually compute properly
                    pass

        if not improved and sweep > 10:
            # Try larger random jumps
            a_pert = a + 0.02 * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            obj_pert = compute_obj(a_pert)
            if obj_pert < 2 * n * float(np.max(compute_conv(a))):
                a = a_pert
                c = compute_conv(a)
            else:
                break

    best_obj_final = 2 * n * float(np.max(compute_conv(a)))
    if best_obj_final < best_obj:
        best_obj = best_obj_final
        best_a = a.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use scipy for cubic interpolation
    from scipy.interpolate import interp1d
    f = interp1d(x_old, a, kind='cubic', bounds_error=False, fill_value=0)
    a_new = f(x_new)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.05):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
