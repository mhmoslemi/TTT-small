Optimizes step function coefficients using a hybrid approach combining:

4. Targeted perturbation at peak positions

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def mirror_descent(a, time_budget=300.0, verbose=True, seed=42):
    """
    Mirror descent with multiplicative updates for minimizing 2n*max(conv)/(sum)^2.
    Uses the exact subgradient of the max convolution peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    lr_base = 5.0 / np.sqrt(n)
    decay = 0.9999
    lr = lr_base
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        lr = lr_base * (0.3 + 0.7 * (1 - progress))
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        peak_idx = np.where(conv >= max_conv - 1e-12)[0]
        
        # Compute subgradient: g[i] = sum_{k in peak_idx} (a[k-i] + a[i-k])
        g = np.zeros(n)
        for k in peak_idx:
            for i in range(n):
                val = 0.0
                ji = k - i
                if 0 <= ji < n:
                    val += a[ji]
                ji2 = i - k
                if 0 <= ji2 < n:
                    val += a[ji2]
                g[i] += val
        
        # Multiplicative update
        g_max = float(np.max(g))
        if g_max > 1e-15:
            a = a * np.exp(-lr * g / g_max)
        
        # Enforce non-negativity (exp already ensures this)
        # Renormalize
        s = float(np.sum(a))
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s
        
        # Evaluate
        obj = compute_obj_normalized(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        it += 1
        if verbose and it % 1000 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, lr={lr:.6f}, t={elapsed:.1f}s")
        
        # If stuck, try a perturbation
        if no_improve > 5000 and time.monotonic() - start < time_budget - 10:
            # Perturb at peak-contributing positions
            pert = rng.normal(0, 0.1, n)
            a = a * np.exp(pert)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            no_improve = 0
    
    return best_a

"""
Optimizes step function coefficients minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses a hybrid approach:
1. Mirror descent (multiplicative updates) with exact subgradient of max peak
2. Exact coordinate descent with analytical 1D subproblem solving
3. LP-based polishing via cvxpy
4. Targeted perturbation at peak-contributing coordinates
5. Multi-n exploration and diverse restarts
Key innovation: mirror descent provides a fundamentally different optimization
trajectory than gradient descent, exploiting the geometry of the simplex.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def peak_subgradient(a, conv, tolerance=1e-10):
    """Compute subgradient of max(conv) w.r.t. a. O(n * |peak_set|)."""
    n = len(a)
    max_conv = float(np.max(conv))
    peak_idx = np.where(conv >= max_conv - tolerance * max_conv)[0]
    
    g = np.zeros(n)
    for k in peak_idx:
        k = int(k)
        # a[k-i] for valid i: 0 <= k-i < n => k-n+1 <= i <= k
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            g[idx] += a[k - idx]
        # a[i-k] for valid i: 0 <= i-k < n => k <= i <= k+n-1
        i_lo2 = max(0, k)
        i_hi2 = min(n - 1, k + n - 1)
        if i_lo2 <= i_hi2:
            idx2 = np.arange(i_lo2, i_hi2 + 1)
            g[idx2] += a[idx2 - k]
    
    return g, peak_idx


def mirror_descent(a, time_budget=300.0, verbose=True, seed=42):
    """
    Mirror descent with multiplicative (exponential) updates.
    Minimizes 2n*max(conv(a,a)) subject to a>=0, sum(a)=1.
    Uses the exact subgradient of the max convolution peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    lr_base = 8.0 / np.sqrt(n)
    it = 0
    no_improve = 0
    best_no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Adaptive learning rate: start large, decay
        lr = lr_base * (0.2 + 0.8 * (1.0 - progress) ** 0.5)

        conv = compute_conv(a)
        g, peak_idx = peak_subgradient(a, conv)

        g_max = float(np.max(g))
        if g_max > 1e-15:
            # Multiplicative update (mirror descent on simplex)
            a = a * np.exp(-lr * g / g_max)

        s = float(np.sum(a))
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        obj = compute_obj_normalized(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > best_no_improve:
                best_no_improve = no_improve

        # Perturbation when stuck
        if no_improve > 3000 and time.monotonic() - start < time_budget - 5:
            # Targeted perturbation: increase coordinates that contribute to off-peak,
            # decrease coordinates that contribute to peak
            pert = rng.normal(0, 0.15, n)
            # Extra push: reduce peak contributors more
            for k in peak_idx[:3]:
                k = int(k)
                # Coordinates that uniquely contribute to peak k
                i_center = k // 2
                if 0 <= i_center < n:
                    pert[i_center] -= 0.5
            a = a * np.exp(pert)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            no_improve = 0
            best_no_improve = 0

        it += 1
        if verbose and it % 2000 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """
    Exact coordinate descent for minimizing 2n*max(conv(a,a))/sum(a)^2.
    Each coordinate is optimized via golden section search on the exact 1D subproblem.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 5 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks]

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Quick candidate scan
            scale = max(v, S / n, 1e-6)
            candidates = [0.0, v * 0.25, v * 0.5, v, v * 1.5, v * 2.0, v * 3.0]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = 2.0 * n * float(np.max(conv)) / (S * S)
            for x in candidates:
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) ** 2
                if denom < 1e-30:
                    continue
                f_val = 2.0 * n * max_val / denom
                if f_val < best_f:
                    best_f = f_val
                    best_x = x

            # Golden section refinement
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            def f_1d(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) ** 2
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(30):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget=60.0, verbose=False, seed=42):
    """LP-based polishing: find g minimizing max(a*g), blend with current."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 3:
                pert = rng.normal(0, 0.05, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if time.monotonic() - start > time_budget * 0.85:
                    break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // (6 + rng.integers(0, 4))] * 2
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Raised cosine (flat-spectrum hint)
        x = np.arange(n) / n
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = a / a.sum()
    elif mode == 4:
        # Two-hump
        x = np.arange(n) / n
        a = np.exp(-((x - 0.25) ** 2) / 0.02) + np.exp(-((x - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 5:
        # Alternating (period-2)
        a = np.ones(n)
        a[1::2] = 0.3
        a = a + rng.normal(0, 0.05, n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 7:
        # Sinusoidal modulation
        x = np.arange(n) / n
        a = 1.0 + 0.4 * np.sin(2 * np.pi * rng.uniform(2, 8) * x)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def targeted_perturbation(a, conv, rng, strength=0.2):
    """Perturb coordinates that contribute most to the peak, increasing off-peak."""
    n = len(a)
    max_conv = float(np.max(conv))
    peak_idx = np.where(conv >= max_conv - 1e-8 * max_conv)[0]
    
    a_pert = a.copy()
    for k in peak_idx[:5]:
        k = int(k)
        # Reduce the coordinate at the center of the peak
        center = k // 2
        if 0 <= center < n:
            a_pert[center] *= np.exp(-strength * rng.uniform(0.5, 1.0))
        # Slightly increase coordinates away from peak center
        offset = max(1, n // 10)
        for d in [offset, 2 * offset]:
            pos = center + d
            pos2 = center - d
            if 0 <= pos < n:
                a_pert[pos] *= np.exp(strength * 0.3 * rng.uniform(0.5, 1.0))
            if 0 <= pos2 < n:
                a_pert[pos2] *= np.exp(strength * 0.3 * rng.uniform(0.5, 1.0))
    
    a_pert = np.maximum(a_pert, 0)
    s = float(np.sum(a_pert))
    if s > 1e-10:
        a_pert = a_pert / s
    return a_pert

    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)
    if p3_budget > 20:
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(p3_budget / (3 - rnd), remaining * 0.7)
            if alloc < 10:
                break
            a_lp = lp_polish(best_seq, time_budget=alloc, verbose=False,
                             seed=1000 + rnd)
            obj_lp = compute_obj_normalized(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
    
    # CD after LP
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_cd2 = coordinate_descent(best_seq, time_budget=min(remaining * 0.2, 80.0),
                                   verbose=False, seed=2000)
        obj_cd2 = compute_obj_normalized(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  CD after LP: obj={best_score:.8f} **")

    # === Phase 4: Targeted Perturbation + Re-optimize ===
    print(f"\n=== Phase 4: Targeted Perturbation ===")
    remaining = deadline - time.monotonic()
    for attempt in range(3):
        if remaining < 60:
            break
        conv = compute_conv(best_seq)
        a_pert = targeted_perturbation(best_seq, conv, rng, strength=0.25 + 0.1 * attempt)
        
        # Mix with current best
        blend = 0.3
        a_mixed = (1 - blend) * best_seq + blend * a_pert
        a_mixed = np.maximum(a_mixed, 0)
        a_mixed = a_mixed / a_mixed.sum()
        
        # Optimize the perturbed version
        alloc = min(remaining * 0.2, 80.0)
        if alloc > 15:
            a_opt = mirror_descent(a_mixed, time_budget=alloc * 0.5,
                                   verbose=False, seed=3000 + attempt)
            a_opt = coordinate_descent(a_opt, time_budget=alloc * 0.5,
                                       verbose=False, seed=4000 + attempt)
            obj_pert = compute_obj_normalized(a_opt)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_opt.copy()
                print(f"  Perturb {attempt}: obj={best_score:.8f} **")
            else:
                print(f"  Perturb {attempt}: obj={obj_pert:.8f}")
        
        remaining = deadline - time.monotonic()

    # === Phase 5: Mirror Descent again (second pass) ===
    print(f"\n=== Phase 5: MD Pass 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.2, 150.0)
        a_md2 = mirror_descent(best_seq, time_budget=alloc,
                               verbose=False, seed=seed + 1000)
        obj_md2 = compute_obj_normalized(a_md2)
        if obj_md2 < best_score:
            best_score = obj_md2
            best_seq = a_md2.copy()
            print(f"  MD2: obj={best_score:.8f} **")
        else:
            print(f"  MD2: obj={obj_md2:.8f}")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 100), n_best + 50, n_best + 100, n_best + 200,
        600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and abs(n - n_best) > 30]
    n_options = n_options[:4]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick mirror descent + CD
        alloc = min(remaining * 0.2, 60.0)
        if alloc > 15:
            a_try = mirror_descent(a_try, time_budget=alloc * 0.5,
                                   verbose=False, seed=n_try)
            a_try = coordinate_descent(a_try, time_budget=alloc * 0.5,
                                       verbose=False, seed=n_try + 1)

        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = 3 + r  # raised cosine, two-hump, alternating, power law
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 60.0)
        if alloc > 15:
            a_r = mirror_descent(a_r, time_budget=alloc * 0.5,
                                 verbose=False, seed=5000 + r * 100)
            a_r = coordinate_descent(a_r, time_budget=alloc * 0.5,
                                     verbose=False, seed=6000 + r * 100)
            obj_r = compute_obj_normalized(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 8: Final CD Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 150.0)
        a_final = coordinate_descent(best_seq, time_budget=alloc,
                                     verbose=False, seed=seed + 9999)
        obj_final = compute_obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # Final MD pass if time
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_md_final = mirror_descent(best_seq, time_budget=min(remaining * 0.6, 60.0),
                                    verbose=False, seed=7777)
        obj_md_final = compute_obj_normalized(a_md_final)
        if obj_md_final < best_score:
            best_score = obj_md_final
            best_seq = a_md_final.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
