"""
Optimizes step function coefficients using a multi-phase approach:
1. Exact coordinate descent with 1D subproblem solving
2. LP-based polishing with iterative blending
3. Smoothed gradient descent for escape
4. Multi-n exploration
Key improvements: tighter CD convergence, more aggressive LP iterations,
and adaptive n search focused on the parent's length.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective for normalized sequence (sum=1): 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    return compute_obj_norm(a_norm)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    return None, None


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """
    Exact coordinate descent for minimizing 2n*max(conv(a,a)) with sum(a)=1.
    Uses golden section search for each 1D subproblem.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = compute_obj_norm(a)
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 5 == 1:
            conv = compute_conv(a)
            obj = compute_obj_norm(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v

            if S_other < 1e-12:
                continue

            # Determine which conv entries are affected
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)

            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Max over unaffected entries
            if k_min > 0 or k_max < 2 * n - 2:
                unaffected = np.ones(2 * n - 1, dtype=bool)
                unaffected[k_min:k_max + 1] = False
                M_unchanged = float(np.max(conv[unaffected])) if unaffected.any() else 0.0
            else:
                M_unchanged = 0.0

            def f_1d(x):
                if x < 0:
                    x = 0.0
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val, M_unchanged)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Quick scan
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.5, v, v * 1.5, v * 2.0]
            candidates = [max(0.0, float(c)) for c in candidates]

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refine
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(25):
                if x_hi - x_lo < 1e-12 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-13:
                continue

            # Update
            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

        # End of sweep: check improvement
        if sweep % 5 == 0:
            conv = compute_conv(a)
            obj = compute_obj_norm(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                if verbose and sweep % 25 == 0:
                    elapsed = time.monotonic() - start
                    print(f"    CD sweep {sweep}: obj={best_obj:.9f}, t={elapsed:.1f}s")

    return best_a


def lp_polish_iterative(a, time_budget=200.0, verbose=False, seed=42):
    """
    Iterative LP polishing: repeatedly solve LP, blend, and do CD.
    This is the key improvement over the previous approach.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iteration = 0
    while time.monotonic() - start < time_budget:
        iteration += 1
        remaining = time_budget - (time.monotonic() - start)

        # Solve LP
        lp_time = min(remaining * 0.4, 60.0)
        if lp_time < 5:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Try multiple blending factors
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        best_t = 0.0

        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
                best_t = t

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a

            # Do a few CD sweeps to converge
            cd_time = min(remaining * 0.3, 30.0)
            if cd_time > 5:
                a = coordinate_descent(a, time_budget=cd_time, verbose=False,
                                       seed=seed + iteration * 100)
                a = a / a.sum()
                obj_after = compute_obj_norm(a)
                if obj_after < best_obj:
                    best_obj = obj_after
                    best_a = a.copy()
                a = best_a

            if verbose:
                print(f"    LP iter {iteration}: t={best_t:.2f}, obj={best_obj:.9f}")
        else:
            # No improvement, try perturbation
            if iteration > 2:
                pert = rng.normal(0, 0.01, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
            elif iteration > 5:
                break

    return best_a


def smoothed_gradient_fft(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n], conv


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=20.0, tau_end=500.0,
                                verbose=False):
    """Nesterov accelerated smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()

    beta = 0.85
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    lr_schedule = lr

    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
        else:
            y = y / s_y

        g, _ = smoothed_gradient_fft(y, tau)
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-15:
            lr_schedule = min(max(lr_schedule * 0.998 + 0.002 / (g_norm + 0.1),
                                  0.5 * lr), 2.0 * lr)
        else:
            lr_schedule = lr

        v = beta * v - lr_schedule * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj_norm(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 2000 == 0:
            print(f"    GD it{it}: obj={best_obj:.9f}, tau={tau:.0f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use cubic for smoother interpolation
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(x_old, a, bc_type='natural')
        a_new = cs(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
