```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_exact(a, tol=1e-10):
    """
    Compute subgradient of max_k conv(a,a)[k] at a.
    Uses uniform weighting over active set.
    Returns the subgradient vector (length n).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    active = np.where(conv >= max_conv - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        # d(conv(a,a)[k])/da[m] = 2*a[k-m] if 0 <= k-m < n
        # This is equivalent to: grad[m] += 2 * a[k-m] for valid k-m
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for m in range(lo, hi + 1):
            grad[m] += 2.0 * a[k - m]
    
    grad /= len(active)
```

```python
    result = ifft(fft(flip_a, N) * fft(p, N)).real
    grad = 2.0 * result[n-1:2*n-1]
```

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated Nesterov momentum gradient descent with adaptive log-sum-exp
smoothing, active-set tracking, LP polishing, and multi-length exploration.
Key differences from parent: (1) pure gradient-based core loop (faster per iteration),
(2) exact FFT-based gradient for smoothed max, (3) adaptive tau schedule,
(4) Barzilai-Borwein step sizing, (5) active-set equalization refinement,
(6) structured multi-start with chirp initializations.
"""

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient_fft(a, tau):
    """
    Compute gradient of smoothed max of conv(a,a) via FFT.
    f(a) = (1/tau)*log(sum_k exp(tau*conv(a,a)[k]))
    grad[m] = 2 * sum_j a[j]*p[m+j] where p = softmax(tau*conv)
    Computed as: 2 * ifft(fft(flipped_a, N) * fft(p, N)).real[n-1:2n-1]
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    s = float(np.sum(exp_vals))
    if s < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / s

    flip_a = a[::-1]
    N = 1
    while N < n + (2 * n - 1):
        N <<= 1
    result = ifft(fft(flip_a, N) * fft(p, N)).real
    grad = 2.0 * result[n - 1:2 * n - 1]

    return grad, conv, p


def nesterov_gd(a, time_budget, tau_init=10.0, tau_end=500.0, verbose=False):
    """
    Nesterov accelerated projected gradient descent with adaptive smoothing.
    Minimizes smoothed max of conv(a,a) on the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = 0.85
    base_lr = 5.0 / math.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Cosine annealing for tau
        cos_val = 0.5 * (1.0 + math.cos(math.pi * progress))
        tau = tau_init + (tau_end - tau_init) * (1.0 - cos_val)

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        grad, conv, p = smooth_gradient_fft(y, tau)

        # Barzilai-Borwein step size
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = grad - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                bb_lr = min(max(bb_lr, 1e-10), 10.0 / math.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb_lr

        v = momentum * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            lr = base_lr
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1

        if verbose and it % 5000 == 0:
            print(f"      GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, stall={stall_count}")

        prev_a = a.copy()
        prev_g = grad.copy()
        a = a_new
        it += 1

        # Decay learning rate slowly
        lr = max(1e-10, lr * 0.9999)

        # Stagnation handling
        if stall_count > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            stall_count = 0

    return best_a


def exact_subgradient_step(a, step_size, verbose=False):
    """
    Take a step in the negative exact subgradient direction (uniform active set).
    Returns updated a and the subgradient.
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    tol = max_c * 1e-8 + 1e-12
    active = np.where(conv >= max_c - tol)[0]

    # Subgradient: average of gradients at active lags
    grad = np.zeros(n)
    for k in active:
        # d(conv[k])/da[m] = 2*a[k-m] if 0 <= k-m < n
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for m in range(lo, hi + 1):
            grad[m] += 2.0 * a[k - m]
    grad /= len(active)

    # Projected gradient step
    a_new = a - step_size * grad
    a_new = project_simplex(a_new)

    return a_new, grad, active, max_c


def active_set_refine(a, time_budget=60.0, verbose=False):
    """
    Refine using exact subgradient with adaptive step size.
    Focuses on equalizing active lags.
    """
    n = len(a)
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    step_size = 1.0 / (n * 0.1)
    patience = 0

    while time.monotonic() - start < time_budget:
        a_new, grad, active, max_c = exact_subgradient_step(a, step_size)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            step_size = min(step_size * 1.05, 1.0)
        else:
            a = a_new  # Still move, but don't update best
            patience += 1
            step_size *= 0.9

        if verbose and patience % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"      ASR: obj={best_obj:.8f}, |active|={len(active)}, step={step_size:.6f}, t={elapsed:.0f}s")

        if patience > 500:
            break

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns optimal g or None.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.PDLP, cp.SCS]
        for solver in solvers:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except Exception:
        pass

    return None


def lp_polish(a, time_budget=200.0, verbose=False):
    """LP-based polishing: solve LP, line search for best blend."""
    n = len(a)
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    prev_g = None
    momentum = 0.0
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        g_lp = lp_direction(a)
        if g_lp is None:
            break
        it += 1

        # Momentum extrapolation
        if prev_g is not None and it > 1:
            diff = g_lp - prev_g
            momentum = 0.6 * momentum + 0.4 * float(np.linalg.norm(diff))
            g_pred = g_lp + 0.2 * momentum * diff
            g_pred = np.maximum(g_pred, 0)
            s_pred = float(np.sum(g_pred))
            if s_pred > 1e-10:
                g_pred = g_pred / s_pred
                obj_pred = compute_obj(g_pred)
                if obj_pred < best_obj:
                    g_lp = g_pred
        prev_g = g_lp.copy()

        # Line search: try multiple t values
        t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 1.0]
        if stall > 10:
            t_candidates = [0.3, 0.5, 0.7, 1.0, 1.2, 1.5]

        best_t_obj = best_obj
        best_t_a = best_a.copy()

        for t in t_candidates:
            if t <= 1.0:
                a_try = (1.0 - t) * a + t * g_lp
            else:
                a_try = t * g_lp - (t - 1.0) * a
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = a_new = best_a
            stall = 0
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}")
        else:
            stall += 1
            if stall > 20:
                break

    return best_a


def chirp_init(n, alpha=1.0, beta=1.0):
    """Real-valued chirp-like initialization."""
    j = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * beta * j * j / n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def power_init(n, p=0.5):
    """Power-law initialization: a[j] proportional to (j+1)^p."""
    j = np.arange(n, dtype=np.float64) + 1.0
    a = j ** p
    return a / a.sum()


def symmetric_init(n, mode=0):
    """Symmetric structured initializations."""
    j = np.arange(n, dtype=np.float64)
    x = j / (n - 1)
    if mode == 0:
        a = 1.0 + 0.5 * np.cos(np.pi * x)
    elif mode == 1:
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * x)
    elif mode == 2:
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * 2 * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(5, max(1, n // 20))
    for k in range(1, n_modes + 1):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.0) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

        else:
            print(f"  Phase 4 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 5: Second GD pass with higher tau (~100s) ===
    print(f"\n=== Phase 5: Fine GD Polish ===")
    p5_budget = min((deadline - time.monotonic()) * 0.1, 100.0)
    if p5_budget > 20:
        a_gd2 = nesterov_gd(best_seq, time_budget=p5_budget,
                            tau_init=200.0, tau_end=1000.0, verbose=False)
        obj_gd2 = compute_obj(a_gd2)
        if obj_gd2 < best_score:
            best_score = obj_gd2
            best_seq = a_gd2.copy()
            print(f"  Phase 5 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase 5 GD: obj={obj_gd2:.8f} (no improvement)")

    # === Phase 6: Multi-n exploration (~150s) ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_candidates = []
    for dn in [-100, -50, 50, 100, 150, 200]:
        n_try = n_best + dn
        if 500 <= n_try <= 1500:
            n_candidates.append(n_try)
    for n_abs in [700, 750, 800, 850, 900, 950, 1000]:
        if n_abs not in n_candidates and abs(n_abs - n_best) > 30:
            n_candidates.append(n_abs)
    n_candidates = sorted(set(n_candidates))[:6]
    print(f"  Trying n values: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # Quick GD
        alloc = min(remaining * 0.12, 40.0)
        if alloc > 10:
            a_try = nesterov_gd(a_try, time_budget=alloc, tau_init=50.0, tau_end=300.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diversification restarts (~100s) ===
    print(f"\n=== Phase 7: Diversification ===")
    n_current = len(best_seq)
    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if r == 0:
            a_r = fourier_perturb(best_seq, rng, strength=0.03)
        elif r == 1:
            a_r = 0.8 * best_seq + 0.2 * np.ones(n_current) / n_current
            a_r = a_r / a_r.sum()
        else:
            x = np.linspace(0, 1, n_current)
            a_struct = 1.0 + 0.3 * np.cos(2 * np.pi * (r + 2) * x)
            a_struct = np.maximum(a_struct, 0.05)
            a_r = 0.7 * best_seq + 0.3 * a_struct / a_struct.sum()
            a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.15, 50.0)
        if alloc > 10:
            a_r = nesterov_gd(a_r, time_budget=alloc, tau_init=50.0, tau_end=400.0)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 8: Final LP polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 80.0)
        a_final = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final GD micro-polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 30.0)
        a_final = nesterov_gd(best_seq, time_budget=alloc,
                              tau_init=500.0, tau_end=2000.0, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
