def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize_scalar

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    idx = np.arange(n)
    for j in range(n):
        k_start = 0
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[max(0, -j):min(n, 2 * n - 1 - j)]
    return A


def solve_lp_direction(a, M_target=None):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_value) or (None, None).
    """
    import cvxpy as cp
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value).flatten(), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_tr_sscp(a, r):
    """
    Solve trust region SOCP:
    min M s.t. conv(a+delta) <= M (linearized), ||delta||_2 <= r,
    sum(a+delta)=1, a+delta >= 0.
    Returns (delta, M_value) or (None, None).
    """
    import cvxpy as cp
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)

    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta

    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
        cp.norm(delta, 2) <= r,
    ]

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None


def brent_line_search(a, direction, best_obj):
    """
    Brent's method line search along a + alpha * direction.
    direction should be such that a + alpha*direction stays on simplex for alpha in [0,1].
    Returns (best_alpha, best_obj_found).
    """
    n = len(a)

    def objective(alpha):
        a_try = a + alpha * direction
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            return best_obj
        a_try = a_try / s
        return compute_obj(a_try)

    try:
        res = minimize_scalar(objective, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-4, 'maxiter': 40})
        if res.success and res.fun < best_obj - 1e-15:
            return float(res.x), float(res.fun)
    except Exception:
        pass

    # Fallback: try a few fixed alphas
    best_alpha = 0.0
    best_val = best_obj
    for t in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9, 1.0]:
        v = objective(t)
        if v < best_val - 1e-15:
            best_val = v
            best_alpha = t
    return best_alpha, best_val


def smoothed_gd(a, time_budget=100.0, tau_init=15.0, tau_end=400.0,
                lr_init=None, momentum=0.8, seed=42, verbose=False):
    """Smoothed gradient descent with log-sum-exp annealing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    if lr_init is None:
        lr = 3.0 / np.sqrt(n)
    else:
        lr = lr_init
    v = np.zeros(n)
    it = 0
    stall = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n] * tau

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                lr *= 0.5
                v *= 0.2
                stall = 0
                if lr < 1e-12:
                    a_new = a * np.exp(rng.normal(0, 0.015, n))
                    a_new = np.maximum(a_new, 0)
                    s_new = a_new.sum()
                    if s_new > 1e-10:
                        a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                    a = a_new
                    lr = 3.0 / np.sqrt(n)
                    v = np.zeros(n)
                    stall = 0

        a = a_new
        lr = max(1e-13, lr * 0.9998)
        it += 1

        if verbose and it % 40 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

    return best_a

    it = 0

        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break

        weights = np.exp((peak_vals - max_pv) * 10.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            for i in range(n):
                diff = k - i
                if 0 <= diff < n:
                    g[i] += w * a[diff]
                diff2 = i - k
                if 0 <= diff2 < n:
                    g[i] += w * a[diff2]

        g = g / K
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 150:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-13:
                    a = best_a * np.exp(rng.normal(0, 0.01, n))
                    a = np.maximum(a, 0)
                    s_a = a.sum()
                    if s_a > 1e-10:
                        a = a / s_a
                    lr = 2.0 / np.sqrt(n)
                    v = np.zeros(n)
                    stall = 0
                    if it > 500:
                        break
            else:
                lr *= 0.997

        it += 1

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    AS it{it}: obj={best_obj:.8f}, lr={lr:.2e}, t={elapsed:.0f}s")

    return best_a


def lp_polish(a, time_budget=200.0, verbose=False, seed=42):
    """LP direction finding with Brent line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp_direction(a)
        lp_time = time.monotonic() - t0

        if g_lp is None:
            break

        direction = g_lp - best_a
        best_alpha, best_val = brent_line_search(best_a, direction, best_obj)

        if best_alpha > 0 and best_val < best_obj - 1e-12:
            best_obj = best_val
            new_a = best_a + best_alpha * direction
            new_a = np.maximum(new_a, 0)
            s_new = new_a.sum()
            if s_new > 1e-10:
                new_a = new_a / s_new
            best_a = new_a
            a = new_a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                scale = 0.01 + 0.008 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                no_improve = 0
                if it > 40:
                    break

        it += 1
        if verbose and it % 3 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, lp_t={lp_time:.1f}s, t={elapsed:.0f}s")

    return best_a


def tr_sscp_polish(a, time_budget=300.0, r_init=0.3, seed=42, verbose=False):
    """Trust region SOCP polishing with adaptive radius."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    r = r_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        delta_dir, M_val = solve_tr_sscp(a, r)

        if delta_dir is None:
            g_lp, _ = solve_lp_direction(a)
            if g_lp is None:
                break
            delta_dir = g_lp - a
        else:
            norm_d = np.linalg.norm(delta_dir)
            if norm_d > r:
                delta_dir = delta_dir * (r / norm_d)

        best_alpha, best_val = brent_line_search(best_a, delta_dir, best_obj)

        if best_alpha > 0 and best_val < best_obj - 1e-12:
            best_obj = best_val
            new_a = best_a + best_alpha * delta_dir
            new_a = np.maximum(new_a, 0)
            s_new = new_a.sum()
            if s_new > 1e-10:
                new_a = new_a / s_new
            best_a = new_a
            a = new_a.copy()
            no_improve = 0
            r = min(r * 1.1, 2.0)
        else:
            no_improve += 1
            r *= 0.6
            if no_improve >= 5:
                if r < 0.05:
                    a = best_a * np.exp(rng.normal(0, 0.015, n))
                    a = np.maximum(a, 0)
                    s_p = a.sum()
                    if s_p > 1e-10:
                        a = a / s_p
                    r = r_init
                    no_improve = 0
                    if it > 60:
                        break
                else:
                    r = max(r * 0.5, 0.05)

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    TR it{it}: obj={best_obj:.8f}, r={r:.4f}, t={elapsed:.0f}s")

    return best_a


def spectral_shaping_init(n, seed=42):
    """Initialize using frequency-domain spectral shaping."""
    rng = np.random.default_rng(seed)
    freqs = np.arange(n)
    taper = 0.85 + 0.15 * np.cos(np.pi * freqs / n)
    target_spec = np.ones(n) * taper
    target_spec *= (1.0 + 0.15 * rng.standard_normal(n))
    target_spec = np.maximum(target_spec, 0.1)

    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.sqrt(target_spec) * np.exp(1j * phases)
    a = ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def gaussian_init(n, width=3.0):
    """Gaussian-shaped initialization."""
    x = np.arange(n) / n
    a = np.exp(-width * (x - 0.5) ** 2)
    s = a.sum()
    return a / s


def random_init(n, seed=42, dist='uniform'):
    """Random initialization with various distributions."""
    rng = np.random.default_rng(seed)
    if dist == 'uniform':
        a = rng.uniform(0, 1, n)
    elif dist == 'sqrt':
        a = np.sqrt(rng.uniform(0, 1, n))
    elif dist == 'bimodal':
        half = n // 2
        a = np.concatenate([rng.uniform(0.3, 1.0, half),
                            rng.uniform(0.0, 0.4, n - half)])
    else:
        a = rng.uniform(0, 1, n)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a


def resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def targeted_perturbation(a, conv, scale=0.02, seed=42):
    """Perturb values that contribute most to the peaks."""
    n = len(a)
    rng = np.random.default_rng(seed)
    top_k = np.argsort(conv)[::-1][:4]
    # Positions that contribute to peaks get reduced, others get increased
    contrib = np.zeros(n)
    for k in top_k:
        k = int(k)
        for i in range(n):
            if 0 <= k - i < n:
                contrib[i] += a[k - i]
            if 0 <= i - k < n:
                contrib[i] += a[i - k]
    contrib_norm = contrib / (contrib.max() + 1e-10)
    # Reduce high-contributing positions, increase low ones
    perturbation = rng.normal(0, 1, n) * (1 - contrib_norm) * scale
    perturbation -= (perturbation * a).sum() / (a.sum() + 1e-10)
    a_new = a * np.exp(perturbation)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining smoothed GD, LP direction,
    TR-SOCP, active-set subgradient, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Diverse initialization (20s) ===
    print(f"\n=== Phase 1: Diverse Init (n={len(best_seq)}) ===")
    n_cur = len(best_seq)
    remaining = deadline - time.monotonic()
    init_budget = min(remaining * 0.02, 20.0)

    if init_budget > 10:
        inits = {
            'spectral': spectral_shaping_init(n_cur, seed=100),
            'gaussian': gaussian_init(n_cur, width=2.5),
            'gauss_wide': gaussian_init(n_cur, width=1.2),
            'sqrt': random_init(n_cur, seed=200, dist='sqrt'),
            'bimodal': random_init(n_cur, seed=300, dist='bimodal'),
        }
        for name, init_a in inits.items():
            t_alloc = init_budget / (len(inits) + 1)
            if t_alloc < 2:
                break
            obj_try = compute_obj(init_a)
            if obj_try < best_score:
                a_quick = smoothed_gd(init_a, time_budget=t_alloc, tau_init=10, tau_end=200)
                obj_quick = compute_obj(a_quick)
                if obj_quick < best_score:
                    best_score = obj_quick
                    best_seq = a_quick.copy()
                    print(f"  {name}: obj={best_score:.8f} **")
                else:
                    print(f"  {name}: init={obj_try:.6f}, gd={obj_quick:.8f}")
            elif obj_try < best_score + 0.1:
                a_quick = smoothed_gd(init_a, time_budget=t_alloc, tau_init=10, tau_end=200)
                obj_quick = compute_obj(a_quick)
                if obj_quick < best_score:
                    best_score = obj_quick
                    best_seq = a_quick.copy()
                    print(f"  {name}: obj={best_score:.8f} **")
                else:
                    print(f"  {name}: init={obj_try:.6f}, gd={obj_quick:.8f}")

    # === Phase 2: Smoothed GD warmup (100s) ===
    print(f"\n=== Phase 2: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.07, 100.0)
    if p2_budget > 15:
        a_gd = smoothed_gd(best_seq, time_budget=p2_budget,
                           tau_init=12.0, tau_end=350.0, seed=42)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  GD: obj={best_score:.8f} **")
        else:
            print(f"  GD: obj={obj_gd:.8f}")

    # === Phase 3: LP + Line Search (main, 450s) ===
    print(f"\n=== Phase 3: LP + Line Search ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.30, 450.0)
    if p3_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p3_budget, verbose=True, seed=42)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 4: TR-SOCP polish (150s) ===
    print(f"\n=== Phase 4: TR-SOCP ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.10, 150.0)
    if p4_budget > 25:
        a_tr = tr_sscp_polish(best_seq, time_budget=p4_budget,
                              r_init=0.25, seed=77, verbose=True)
        obj_tr = compute_obj(a_tr)
        if obj_tr < best_score:
            best_score = obj_tr
            best_seq = a_tr.copy()
            print(f"  TR: obj={best_score:.8f} **")
        else:
            print(f"  TR: obj={obj_tr:.8f}")

    # === Phase 5: Active set subgradient (80s) ===
    print(f"\n=== Phase 5: Active Set Subgradient ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.05, 80.0)
    if p5_budget > 15:
        conv_cur = compute_conv(best_seq)
        a_as = active_set_subgradient(best_seq, time_budget=p5_budget,
                                      K=6, seed=123)
        obj_as = compute_obj(a_as)
        if obj_as < best_score:
            best_score = obj_as
            best_seq = a_as.copy()
            print(f"  AS: obj={best_score:.8f} **")
        else:
            print(f"  AS: obj={obj_as:.8f}")

    # === Phase 6: Targeted perturbation + optimize (200s) ===
    print(f"\n=== Phase 6: Targeted Perturbation ===")
    remaining = deadline - time.monotonic()
    n_pert = 4
    for p_idx in range(n_pert):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        conv_cur = compute_conv(best_seq)
        scale = 0.015 + 0.01 * p_idx
        a_pert = targeted_perturbation(best_seq, conv_cur, scale=scale,
                                       seed=500 + p_idx)

        # Quick GD then LP
        alloc1 = min(remaining * 0.08, 25.0)
        if alloc1 > 5:
            a_pert = smoothed_gd(a_pert, time_budget=alloc1,
                                 tau_init=10, tau_end=250, seed=600 + p_idx)
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.12, 50.0)
        if alloc2 > 15:
            a_pert = lp_polish(a_pert, time_budget=alloc2, seed=700 + p_idx)

        obj_pert = compute_obj(a_pert)
        if obj_pert < best_score:
            best_score = obj_pert
            best_seq = a_pert.copy()
            print(f"  Pert{p_idx} (scale={scale:.4f}): obj={best_score:.8f} **")
        else:
            print(f"  Pert{p_idx} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 7: Multi-length exploration (250s) ===
    print(f"\n=== Phase 7: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(300, n_current - 200),
        n_current - 100 if n_current > 500 else 300,
        n_current + 100,
        n_current + 300,
        n_current + 500,
        1000, 1200, 1500, 2000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 3000 and abs(n - n_current) > 50]
    n_options = n_options[:6]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)

        # Quick GD
        alloc1 = min(remaining * 0.06, 20.0)
        if alloc1 > 5:
            a_try = smoothed_gd(a_try, time_budget=alloc1,
                                tau_init=10, tau_end=200)

        remaining = deadline - time.monotonic()
        # LP or TR depending on length
        if n_try <= 1500:
            alloc2 = min(remaining * 0.08, 40.0)
            if alloc2 > 10:
                a_try = lp_polish(a_try, time_budget=alloc2, seed=800 + n_try)
        else:
            alloc2 = min(remaining * 0.08, 30.0)
            if alloc2 > 10:
                a_try = smoothed_gd(a_try, time_budget=alloc2,
                                    tau_init=20, tau_end=500)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final intensive polish (remaining) ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_final = len(best_seq)
        if n_final <= 1500:
            alloc = min(remaining * 0.4, 80.0)
            a_final = lp_polish(best_seq, time_budget=alloc, seed=999, verbose=True)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final LP: obj={best_score:.8f} **")
            else:
                print(f"  Final LP: obj={obj_final:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.3, 50.0)
            a_final2 = smoothed_gd(best_seq, time_budget=alloc,
                                   tau_init=50, tau_end=800, seed=1111)
            obj_final2 = compute_obj(a_final2)
            if obj_final2 < best_score:
                best_score = obj_final2
                best_seq = a_final2.copy()
                print(f"  Final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
