import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smooth_max_grad(a, eps):
    """Compute gradient of smooth max of conv(a,a).
    Returns (gradient, smooth_max_value, p_weights).
    """
    n = len(a)
    conv = compute_conv(a)  # length 2n-1
    # Smooth max: (1/eps) * log(sum exp(eps * c_k))
    max_c = np.max(conv)
    shifted = eps * (conv - max_c)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / np.sum(exp_shifted)
    smooth_val = max_c + np.log(np.sum(exp_shifted)) / eps
    
    # Gradient: 2 * cross_correlation(a, p)
    # S1[i] = sum_m a[m] * p[m+i]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    A = rfft(a, size)
    P = rfft(p, size)
    XC = irfft(np.conj(A) * P, size)
    grad = 2.0 * XC[:n]
    
    return grad, smooth_val, p

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = np.sum(w)
    if s > 0:
        w /= s  # Actually, the projection should give sum = 1 by construction
    return w

def smooth_gradient_descent(a, time_budget, initial_eps=1.0, final_eps=200.0, verbose=False):

        return a
