def line_search(a, b, lo=0.0, hi=1.5, max_iter=25):
    """Find t in [lo,hi] minimizing obj((1-t)*a + t*b)"""
    def f(t):
        x = np.maximum((1-t)*a + t*b, 0)
        s = x.sum()
        if s < 1e-10:
            return np.inf
        return compute_obj(x/s)

    gr = 2-phi

    d = lo + (1-gr)*(hi-lo)

            c = lo + gr*(hi-lo)
            fc = f(c)
        else:
            lo, c, fc = c, d, fd
            d = lo + gr*(hi-lo)
            fd = f(d)
    t = c if fc < fd else d

Uses log-sum-exp smoothed objective with L-BFGS for global-ish search,

import numpy as np

_GLOBAL_ALPHA = 100.0


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    an = a / s
    c = compute_conv(an)
    return 2.0 * n * float(np.max(c))


def softmax_fn(z):
    """Numerically stable softmax."""
    zm = np.max(z)
    e = np.exp(z - zm)
    return e / e.sum()


def smooth_obj_and_grad(z, alpha):
    """
    Log-sum-exp smoothed objective and its gradient w.r.t. z.
    f(z) = 2n/alpha * log(sum_k exp(alpha * conv(softmax(z),softmax(z))_k))
    """
    n = len(z)
    a = softmax_fn(z)
    c = compute_conv(a)
    mc = np.max(c)
    exp_shifted = np.exp(alpha * (c - mc))
    logsum = mc + np.log(np.sum(exp_shifted))
    obj = 2.0 * n * logsum / alpha
    w = exp_shifted / exp_shifted.sum()
    C = float(np.dot(w, c))
    w_a = correlate(w, a, mode='valid')
    grad = (4.0 * n / alpha) * a * (w_a - C)
    return obj, grad

def line_search(a, b, lo=0.0, hi=2.0, max_iter=30):

        s = float(x.sum())

    z = z - np.mean(z)  # center
    best_z = z.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for alpha in alphas:
        if time.monotonic() - start >= sum(time_per_alpha for _ in alphas) - 2:
            break
        _GLOBAL_ALPHA = alpha
        try:
            res = minimize(_lbfgs_fun, z, method='L-BFGS-B', jac=True,
                           options={'maxiter': 300, 'maxfun': 600, 'gtol': 1e-8})
            if res.x is not None:
                z_new = res.x
                a_new = softmax_fn(z_new)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_z = z_new.copy()
                    z = z_new - np.mean(z_new)
                    if verbose:
                        print(f"    alpha={alpha}: obj={best_obj:.8f} **")
                else:
                    if verbose:
                        print(f"    alpha={alpha}: obj={obj_new:.8f} (no improve)")
        except Exception as e:
            if verbose:
                print(f"    alpha={alpha}: error {e}")

    return softmax_fn(best_z), best_obj


def u_init(n, sharp=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharp * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    return a / a.sum()


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.01, 0.99, n)
    a = 1.0 / np.sqrt(1.0 - (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    return a / a.sum()


def diverse_init(n, mode=0):
    """Various structured initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=np.float64) / max(1, n - 1)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - 0.25) ** 2) / 0.015) + np.exp(-((idx - 0.75) ** 2) / 0.015) + 0.01
    elif mode == 2:
        a = np.exp(5.0 * (2 * x - 1) ** 2)
    elif mode == 3:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 4:
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 2)
    else:
        a = np.ones(n) * 0.1 + (1 - idx) * idx * 0.0  # flat
        a = 1.0 / (0.1 + 2.0 * (2 * x - 1) ** 3)
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def try_length(a_best, n_try, time_budget, verbose=False):
    """Try a different sequence length by interpolation + optimization."""
    n_old = len(a_best)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = np.maximum(a_try, 0)
    s = float(a_try.sum())
    if s < 0.01:
        return None, np.inf
    a_try = a_try / s

    # Quick smooth optimization
    if time_budget > 20:
        a_try, _ = smooth_optimize(a_try, [100, 300], time_per_alpha=min(10, time_budget * 0.3))

    # LP polish
    remaining = time_budget - (time.monotonic() - time.monotonic())  # approximation
    if time_budget > 30:
        a_try = lp_polish(a_try, min(time_budget * 0.5, 60), max_iters=30)

    obj_try = compute_obj(a_try)
    return a_try, obj_try


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: hybrid smooth optimization + LP polishing with
    multi-phase strategy, adaptive restarts, and length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    # Load parent or initialize
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(parent.sum())
        if s < 0.01 or not np.isfinite(parent).all():
            raise ValueError("bad parent")
        best_a = parent / s
    except Exception:
        best_a = u_init(800)

    best_obj = compute_obj(best_a)
    n = len(best_a)
    print(f"Initial: n={n}, obj={best_obj:.8f}")

    # Also try alternative initializations
    a_u = u_init(n, sharp=3.0)
    obj_u = compute_obj(a_u)
    print(f"  U-shaped: obj={obj_u:.8f}")
    if obj_u < best_obj:
        best_obj = obj_u
        best_a = a_u.copy()

    a_arc = arcsine_init(n)
    obj_arc = compute_obj(a_arc)
    print(f"  Arcsine: obj={obj_arc:.8f}")
    if obj_arc < best_obj:
        best_obj = obj_arc
        best_a = a_arc.copy()

    # === Phase 1: Smooth L-BFGS with increasing alpha ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 1: Smooth L-BFGS (remaining={remaining:.0f}s) ===")
        alphas_p1 = [30, 80, 150, 300, 600, 1200]
        time_per = min(12.0, remaining * 0.08)
        a_new, obj_new = smooth_optimize(best_a, alphas_p1, time_per_alpha=time_per, verbose=True)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            print(f"  Phase 1 result: obj={best_obj:.8f} **")

    # === Phase 2: Intensive LP polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: LP Polish (remaining={remaining:.0f}s) ===")
        alloc = min(remaining * 0.35, 350.0)
        a_lp = lp_polish(best_a, alloc, max_iters=200, verbose=True, seed=123)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            print(f"  Phase 2: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_lp:.8f}")

    # === Phase 3: Smooth optimization at very high alpha ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 3: High-alpha smooth (remaining={remaining:.0f}s) ===")
        alphas_p3 = [2000, 5000, 10000, 20000]
        time_per = min(10.0, remaining * 0.08)
        a_new, obj_new = smooth_optimize(best_a, alphas_p3, time_per_alpha=time_per, verbose=True)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            print(f"  Phase 3: obj={best_obj:.8f} **")

    # === Phase 4: Second LP round with perturbation ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 4: LP Round 2 (remaining={remaining:.0f}s) ===")
        rng = np.random.default_rng(seed + 1)
        noise = rng.normal(0, 0.003, n)
        a_pert = np.maximum(best_a + noise, 0)
        a_pert = a_pert / a_pert.sum()
        alloc = min(remaining * 0.3, 200.0)
        a_lp2 = lp_polish(a_pert, alloc, max_iters=100, verbose=True, seed=456)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()
            print(f"  Phase 4: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 4: obj={obj_lp2:.8f}")

    # === Phase 5: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 5: Length Exploration (remaining={remaining:.0f}s) ===")
        n_best = len(best_a)
        n_options = [n_best - 100, n_best - 50, n_best + 50, n_best + 100,
                     n_best + 200, 600, 700, 900, 1000, 1200, 1400]
        n_options = sorted(set([x for x in n_options if 300 <= x <= 2000 and abs(x - n_best) >= 30]))
        n_options = n_options[:6]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            alloc = min(remaining * 0.3, 80.0)
            a_try, obj_try = try_length(best_a, n_try, alloc, verbose=True)
            if a_try is not None and obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                print(f"  ** n={n_try}: obj={best_obj:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 6: Diverse Restarts (remaining={remaining:.0f}s) ===")
        n_cur = len(best_a)
        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            mode = r
            a_div = diverse_init(n_cur, mode=mode)
            # Blend with best
            a_mix = 0.3 * a_div + 0.7 * best_a
            a_mix = np.maximum(a_mix, 0)
            a_mix = a_mix / a_mix.sum()
            # Optimize
            a_mix, _ = smooth_optimize(a_mix, [100, 400], time_per_alpha=8.0)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_mix = lp_polish(a_mix, min(remaining * 0.4, 40.0), max_iters=25)
            obj_mix = compute_obj(a_mix)
            if obj_mix < best_obj:
                best_obj = obj_mix
                best_a = a_mix.copy()
                print(f"  Restart {r}: obj={best_obj:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_mix:.8f}")

    # === Phase 7: Final intensive LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: Final Polish (remaining={remaining:.0f}s) ===")
        alloc = min(remaining - 10, 150.0)
        a_final = lp_polish(best_a, alloc, max_iters=80, verbose=True, seed=789)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final: obj={best_obj:.8f} **")
        else:
            print(f"  Final: obj={obj_final:.8f}")

    # === Phase 8: One more smooth pass at very high alpha ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alphas_final = [5000, 15000, 30000]
        a_final2, obj_final2 = smooth_optimize(best_a, alphas_final, time_per_alpha=min(6.0, remaining * 0.1))
        if obj_final2 < best_obj:
            best_obj = obj_final2
            best_a = a_final2.copy()
            print(f"  Final smooth: obj={best_obj:.8f} **")

    # Final LP if time
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_polish(best_a, min(remaining - 5, 40.0), max_iters=30)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last.copy()

    # Validate
    if best_a is None:
        best_a = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [0.0125] * 800
        best_obj = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")
    return result
