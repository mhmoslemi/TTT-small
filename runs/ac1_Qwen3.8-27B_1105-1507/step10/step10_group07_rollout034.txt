import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def coordinate_descent(a, time_budget, K=20, L=200, verbose=False, seed=42):
    n = len(a)
    rng = np.random.default_rng(seed)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    pass_count = 0
    no_improve_passes = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        S = float(a.sum())
        max_conv = float(conv.max())
        
        # Find active set: top K peaks
        if K < len(conv):
            top_idx = np.argpartition(conv, -K)[-K:]
        else:
            top_idx = np.arange(len(conv))
        top_idx = top_idx[np.argsort(conv[top_idx])[::-1]]
        top_vals = conv[top_idx]
        
        obj_start = best_obj
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            
            a_orig = a[j]
            if a_orig < 1e-15:
                # Try small positive values
                deltas = [0.01/n, 0.1/n, 1.0/n, 10.0/n]
            else:
                # Grid of delta values
                deltas = a_orig * np.array([-1.0, -0.8, -0.5, -0.3, -0.1, 0.0, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0])

            best_val = 2.0 * n * max_conv / (S*S)  # current objective
            
            for delta in deltas:

                if new_val < 0:
                    continue
                if abs(delta) < 1e-15:
                    continue
                
                # Compute new conv at active set positions
                # For each k in top_idx:
                #   new_conv[k] = conv[k] + delta * c_k (+ delta^2 if k == 2j)
                #   where c_k = 2*a[k-j] if k in [j, j+n-1] and k != 2j
                #              = 2*a[j] if k == 2j (plus delta separately)

                if new_S < 1e-10:

                new_obj = 2.0 * n * new_max / (new_S * new_S)

        pass_count += 1
        if verbose and pass_count % 5 == 0:
            print(f"  CD pass {pass_count}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
    
    return best_a

"""
Coordinate Descent + LP Hybrid Optimizer for minimum peak autocorrelation step function.
Uses fast coordinate descent with incremental convolution updates as the primary local
optimizer, combined with smoothed gradient warmup, LP-based polishing, multi-length
exploration, and perturbation-based restarts to escape local minima.
Key innovation: O(n*K) per-pass coordinate descent using incremental conv updates
and proxy objective (top-K peaks), enabling thousands of precise local optimizations
within the time budget.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coordinate_descent(a, time_budget, K=50, verbose=False, seed=42):

    Fast coordinate descent with incremental conv updates.
    Uses a proxy objective based on top-K peaks for speed.
    """

    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a

    no_improve = 0

        order = rng.permutation(n)
        any_improved = False

        for j in order:

                break

                candidates = np.array([0.01 / n, 0.1 / n, 1.0 / n, 5.0 / n, 20.0 / n])
            else:
                base = a_orig

                    continue

                    if base_val > new_max:

def smoothed_gradient_descent(a, time_budget, tau_init=20.0, tau_end=400.0, verbose=False):
    """Smoothed gradient descent with log-sum-exp approximation of max."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 3.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr = max(1e-12, lr * 0.999)
        it += 1

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x) approximately."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a):
    """Solve LP: min M s.t. a*g[k] <= M, sum(g)=1, g>=0. Returns (g, M) or None."""
    try:
        import cvxpy as cp
    except ImportError:
        return None, None

    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M))

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val, float(M.value)
        except Exception:
            continue
    return None, None

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

            print(f"  Phase3 LP: obj={obj_lp:.8f}")

    # === Phase 4: Fine coordinate descent ===
    print(f"\n=== Phase 4: Fine Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.20, 200.0)
    if p4_budget > 15:
        a_fine = fine_coordinate_descent(best_seq, time_budget=p4_budget, K=80,
                                         verbose=True, seed=200)
        obj_fine = compute_obj(a_fine)
        if obj_fine < best_score:
            best_score = obj_fine
            best_seq = a_fine.copy()
            print(f"  Phase4 FineCD: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 FineCD: obj={obj_fine:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 100), n_current + 100, n_current + 200,
        1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_current) > 50]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Quick CD on new length
        alloc_cd = min(remaining * 0.20, 60.0)
        if alloc_cd > 10:
            a_try = coordinate_descent(a_try, time_budget=alloc_cd, K=40, seed=n_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_current = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Perturbation + rediscovery ===
    print(f"\n=== Phase 6: Perturbation + Rediscovery ===")
    remaining = deadline - time.monotonic()

    for r_idx in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        scale = 0.01 + 0.012 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            break
        a_pert = a_pert / s_pert

        alloc = min(remaining * 0.30, 80.0)
        if alloc > 10:
            a_pert = coordinate_descent(a_pert, time_budget=alloc, K=50,
                                        seed=500 + r_idx)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Final fine CD
        alloc1 = min(remaining * 0.40, 100.0)
        if alloc1 > 10:
            a_final = fine_coordinate_descent(best_seq, time_budget=alloc1,
                                              K=100, verbose=False, seed=999)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final FineCD: obj={best_score:.8f} **")

        # Final LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc2 = min(remaining * 0.40, 60.0)
            if alloc2 > 10:
                a_final_lp = lp_polish(best_seq, time_budget=alloc2, seed=2000)
                obj_flp = compute_obj(a_final_lp)
                if obj_flp < best_score:
                    best_score = obj_flp
                    best_seq = a_final_lp.copy()
                    print(f"  Final LP: obj={best_score:.8f} **")

        # One more CD pass
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc3 = min(remaining * 0.40, 40.0)
            if alloc3 > 5:
                a_final_cd = coordinate_descent(best_seq, time_budget=alloc3,
                                                K=60, seed=3000)
                obj_fcd = compute_obj(a_final_cd)
                if obj_fcd < best_score:
                    best_score = obj_fcd
                    best_seq = a_final_cd.copy()
                    print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
