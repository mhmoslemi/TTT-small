"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated smoothed gradient descent with adaptive temperature,
simplex projection, periodic LP direction hints, and multi-length exploration.
Different from previous: gradient-descent-first (not LP-first), adaptive tau
based on effective peak support, and a multiplicative descent variant.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time
import cvxpy as cp


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))


def simplex_project(x):
    """Project x onto the simplex {y >= 0, sum(y) = 1}."""
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)


def smooth_grad_and_info(a, tau):
    """Compute gradient of smoothed objective and info for adaptive tau."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    exp_v = np.exp(shifted)
    Z = float(np.sum(exp_v))
    p = exp_v / Z
    M_tau = max_c + np.log(Z) / tau
    eff_support = 1.0 / float(np.sum(p ** 2))
    peak_fraction = float(p[int(np.argmax(p))])
    
    # cross[i] = sum_k p[k] * a[k-i]
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(np.conj(P) * A).real[:n]
    
    grad = 4.0 * n * cross
    return grad, M_tau, eff_support, peak_fraction, conv, p


def nesterov_gd(a, time_budget, tau_init=50.0, tau_max=5000.0, verbose=False):
    """
    Nesterov accelerated gradient descent on the simplex with adaptive tau.
    """
    n = len(a)
    a = simplex_project(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 0.5 / np.sqrt(n)
    v = np.zeros(n)
    tau = tau_init
    prev_obj = best_obj
    prev_a = a.copy()
    prev_grad = None
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        
        # Adaptive tau
        tau = tau_init * (tau_max / tau_init) ** (progress ** 0.7)
        
        # Nesterov extrapolation
        y = a + beta * v
        y = simplex_project(y)
        
        # Compute gradient at y
        g, M_tau, eff_sup, peak_frac, conv, p = smooth_grad_and_info(y, tau)
        g_norm = float(np.linalg.norm(g))
        
        if g_norm < 1e-15:
            break
        
        # Barzilai-Borwein step size
        if prev_grad is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_grad
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                bb_lr = float(np.clip(bb_lr, 0.1 * lr, 5.0 * lr))
                lr = 0.7 * lr + 0.3 * bb_lr
        
        # Gradient step
        v = beta * v - lr * g
        a_new = y + v
        a_new = simplex_project(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Adaptive: try reducing lr or adjusting tau
                lr *= 0.8
                if stall > 500:
                    break
        
        # Backtracking if objective increased too much
        if obj_new > prev_obj * 1.01:
            lr *= 0.7
        
        prev_obj = compute_obj(a)  # This is expensive, do less often
        if it % 10 == 0:
            prev_obj = compute_obj(a)
        prev_a = a.copy()
        prev_grad = g.copy()
        a = a_new
        it += 1
        
        if verbose and it % 500 == 0:
            print(f"      GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, "
                  f"lr={lr:.6f}, eff_sup={eff_sup:.1f}, time={elapsed:.0f}s")
    
    return best_a, best_obj


def solve_lp_direction(a, n_constraints=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns g (direction) or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build convolution matrix
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a_norm[diff[valid]]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_line_search(a, g, n_evals=30):
    """Find best t in [0,1] for a + t*(g-a)."""
    ts = np.linspace(0.01, 1.0, n_evals)
    best_t = 0.0
    best_obj = compute_obj(a)
    for t in ts:
        a_try = (1 - t) * a + t * g
        a_try = simplex_project(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
    if best_t > 0:
        return simplex_project((1 - best_t) * a + best_t * g), best_obj
    return a.copy(), best_obj


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n)
    x = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x, a)
    return simplex_project(a_new)


def fourier_perturb(a, rng, strength=0.03):
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    return simplex_project(a_new)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated smoothed gradient descent with adaptive temperature,
multiplicative (mirror) descent, Fourier fixed-point iteration, LP direction hints,
and multi-length exploration. Unlike the previous LP-first approach, this is
gradient-first with periodic LP refinement, and adds a frequency-domain method.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))


def simplex_project(x):
    """Project x onto the simplex {y >= 0, sum(y) = 1}."""
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho = np.nonzero(u * idx > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0)


def smooth_grad_info(a, tau):
    """
    Compute gradient of smoothed objective 2n*LSE(tau*conv(a,a))/tau.
    Returns (grad, M_tau, eff_support, p, conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    exp_v = np.exp(shifted)
    Z = float(np.sum(exp_v))
    p = exp_v / Z
    M_tau = max_c + np.log(Z) / tau
    eff_support = 1.0 / float(np.sum(p ** 2))

    N = 1
    while N < 3 * n:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    cross = ifft(np.conj(A) * P).real[:n]

    grad = 4.0 * n * cross
    return grad, M_tau, eff_support, p, conv


def nesterov_gd(a, time_budget, tau_init=50.0, tau_max=5000.0, verbose=False):
    """Nesterov accelerated GD on simplex with adaptive tau."""
    n = len(a)
    a = simplex_project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 1.0 / np.sqrt(n)
    v = np.zeros(n)
    tau = tau_init
    it = 0
    stall = 0
    prev_a = a.copy()
    prev_grad = None
    obj_eval_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init * (tau_max / tau_init) ** (progress ** 0.6)

        # Nesterov extrapolation
        y = a + beta * v
        y = simplex_project(y)

        # Gradient at y
        g, M_tau, eff_sup, p, conv = smooth_grad_info(y, tau)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Barzilai-Borwein step size adaptation
        if prev_grad is not None and it % 8 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_grad
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-10:
                bb_lr = ss / abs(sy)
                bb_lr = float(np.clip(bb_lr, 0.05 * lr, 10.0 * lr))
                lr = 0.7 * lr + 0.3 * bb_lr

        # Momentum update
        v = beta * v - lr * g
        a_new = y + v
        a_new = simplex_project(a_new)

        it += 1

        # Periodic objective evaluation (expensive)
        if it % 15 == 0 or it < 5:
            obj_new = compute_obj(a_new)
            obj_eval_count += 1
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
                if stall > 300:
                    lr *= 0.7
                    if stall > 800:
                        break

            # Backtracking
            prev_obj = compute_obj(a)
            if obj_new > prev_obj * 1.005:
                lr *= 0.75

        prev_a = a.copy()
        prev_grad = g.copy()
        a = a_new

        if verbose and it % 1000 == 0:
            obj_trace = compute_obj(a)
            print(f"      NGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, "
                  f"lr={lr:.6f}, eff_sup={eff_sup:.1f}, time={elapsed:.0f}s")

    return best_a, best_obj
