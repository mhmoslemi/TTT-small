from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_conv_lp(a, time_limit=5.0):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a, warm_start=None, time_limit=10.0):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    if warm_start is not None:
        g.value = warm_start
        if M.value is None:
            M.value = float(np.max(A @ warm_start))

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_conv_lp_reduced(a, warm_start=None, n_active=800):
    """
    Solve LP with only the most active constraints (reduced problem).
    First solve reduced, then verify and add violated constraints.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A_full = build_conv_matrix(a_norm)
    
    # Use warm start to identify active constraints
    if warm_start is not None:
        conv_ws = A_full @ warm_start
        max_val = float(np.max(conv_ws))
        # Keep top n_active constraints
        threshold = np.percentile(conv_ws, max(0, 100 * (1 - n_active / len(conv_ws))))
        active_idx = np.where(conv_ws >= threshold)[0]
    else:
        active_idx = np.arange(len(A_full))
    
    A = A_full[active_idx, :]
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    if warm_start is not None:
        g.value = warm_start
        M.value = float(np.max(A @ warm_start))

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Verify against full constraints
                    full_conv = A_full @ g_val
                    M_val = float(np.max(full_conv))
                    if M_val <= float(M.value) * 1.01 + 1e-8:
                        return g_val, M_val
                    # If violated, solve full LP
                    return solve_conv_lp(a, warm_start=g_val)
        except Exception:
            continue
    return None, None


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of (a*a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def nesterov_gd_polish(a, time_budget, tau_init=50.0, tau_max=5000.0):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 1.5 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adaptive_lr = 0.5 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.1 * base_lr), 3.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adaptive_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new
        lr *= 0.9999
        tau = min(tau * 1.03, tau_max)
        it += 1

        if patience > 2000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            tau = tau_init
            patience = 0

    return best_a


def fourier_perturb(a, rng, strength=0.03, n_modes=8):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def chirp_init(n, frequency=1.0, rng=None):
    """Create a chirp-like initial sequence with good autocorrelation properties."""
    if rng is None:
        rng = np.random.default_rng(0)
    t = np.arange(n)
    # Squared magnitude of a chirp signal
    phase = np.pi * frequency * t * t / n
    a = (1.0 + np.cos(phase)) ** 2
    # Add slight randomization
    a *= np.exp(0.1 * rng.standard_normal(n))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_structured_init(n, rng, sparsity=0.0):
    """Create a random structured initial sequence."""
    a = np.abs(rng.standard_normal(n))
    if sparsity > 0:
        mask = rng.random(n) < (1 - sparsity)
        a *= mask
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def lp_alternate(a, time_budget, seed=42, verbose=False, use_reduced_lp=False):
    """
    Main LP-based alternating minimization with adaptive blending.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    t_blend = 0.5
    stall_count = 0
    g_warm = None

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        t0 = time.monotonic()
        if use_reduced_lp and g_warm is not None and iter_count > 5:
            g_lp, M_lp = solve_conv_lp_reduced(a, warm_start=g_warm)
        else:
            g_lp, M_lp = solve_conv_lp(a, warm_start=g_warm)
        if g_warm is not None:
            g_warm = g_warm  # keep for next warm start
        lp_time = time.monotonic() - t0

        if g_lp is None:
            # Fallback to reduced LP
            g_lp, M_lp = solve_conv_lp(a, warm_start=g_warm)
            if g_lp is None:
                break

        g_warm = g_lp
        iter_count += 1

        # Adaptive blending: try multiple t values
        candidates_t = [t_blend, t_blend * 0.5, t_blend * 1.5, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if iter_count > 50:
            candidates_t.extend([0.05, 0.02, 0.01, 0.005])
        candidates_t = [min(max(t, 0.001), 1.0) for t in candidates_t]
        candidates_t = list(set(candidates_t))

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend

        for t in candidates_t:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            t_blend = max(0.005, best_local_t * 0.98)
            stall_count = 0
            if verbose and iter_count % 50 == 0:
                print(f"    LP alt {iter_count}: obj={best_obj:.10f}, t={t_blend:.4f}, "
                      f"M={M_lp:.6f}, time={lp_time:.3f}s")
        else:
            stall_count += 1
            if stall_count >= 5:
                strength = 0.01 + 0.003 * min(stall_count, 15)
                a = fourier_perturb(best_a, rng, strength=strength)
                t_blend = max(0.2, min(t_blend * 1.5, 0.8))
                stall_count = 0
            elif stall_count >= 2:
                # Small nudge
                nudge = 0.003 * rng.standard_normal(n) * best_a
                a = best_a + nudge
                a = np.maximum(a, 0)
                a = a / max(a.sum(), 1e-10)

    if verbose:
        print(f"    LP alternating: {iter_count} iters, best_obj={best_obj:.10f}")
    return best_a


def coordinate_descent_step(a, i, conv):
    """
    For coordinate i, find optimal value given others fixed.
    Uses the fact that changing a[i] by delta changes conv[k] by
    2*delta*a[k-i] + delta^2*[k=2i].
    We search for delta in [-a[i], 0] (only decreasing helps).
    Returns optimal delta and new conv.
    """
    n = len(a)
    v = a[i]
    if v < 1e-15:
        return 0.0, conv

    # Precompute a[k-i] for valid k
    # k-i is in [0, n-1] when k is in [i, i+n-1]
    # But conv has indices 0 to 2n-2
    k_min = max(0, i)
    k_max = min(2 * n - 2, i + n - 1)
    
    # a[k-i] for k in [k_min, k_max]
    idx = np.arange(k_min, k_max + 1) - i
    a_vals = a[idx]  # these are a[k-i] values
    
    conv_relevant = conv[k_min:k_max + 1].copy()
    
    # Find optimal t (decrease a[i] by t, t in [0, v])
    # new_conv_relevant[t] = conv_relevant - 2*t*a_vals + t^2*(idx==2i)
    
    quad_mask = (np.arange(k_min, k_max + 1) == 2 * i)
    
    best_t = 0
    best_max = float(np.max(conv))
    
    # Golden section search on t in [0, v]
    phi = (np.sqrt(5) - 1) / 2
    lo = 0.0
    hi = v
    a_pt = lo + (1 - phi) * (hi - lo)
    b_pt = lo + phi * (hi - lo)
    
    def obj_at_t(t):
        new_conv_rel = conv_relevant - 2 * t * a_vals + t * t * quad_mask.astype(float)
        # Other positions don't change
        m_relevant = float(np.max(new_conv_rel))
        m_other = float(np.max(np.delete(conv, np.arange(k_min, k_max + 1))))
        return max(m_relevant, m_other)
    
    fa = obj_at_t(a_pt)
    fb = obj_at_t(b_pt)
    
    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = lo + (1 - phi) * (hi - lo)
            fa = obj_at_t(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + phi * (hi - lo)
            fb = obj_at_t(b_pt)
    
    best_t = (lo + hi) / 2
    best_max = obj_at_t(best_t)
    
    if best_max < float(np.max(conv)) - 1e-15:
        return -best_t, conv
    return 0.0, conv


def run_coordinate_descent(a, time_budget):
    """Run cyclic coordinate descent with renormalization."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    sweep = 0

    while time.monotonic() - start < time_budget:
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            delta, _ = coordinate_descent_step(a, i, conv)
            if abs(delta) > 1e-15:
                a[i] = max(0, a[i] + delta)
                # Update conv incrementally
                for k in range(max(0, i), min(2 * n - 1, i + n)):
                    conv[k] += 2 * delta * a[min(n - 1, k - i)] if k - i < n else 0
                # Simpler: just recompute conv periodically
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    a = best_a.copy()
                    break
                improved = True
        
        if improved:
            a = a / a.sum()
            conv = compute_conv(a)
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        
        sweep += 1
        if sweep % 100 == 0 and sweep % 1000 < 10:
            pass  # just track
    
    return best_a

        n_parent = a_parent.size

        a_final = lp_alternate(best_a, time_budget=alloc, seed=seed + 99999, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final LP: obj={best_obj:.10f} **")
        else:
            print(f"  Final LP: obj={obj_final:.10f}")

        # Final gradient polish
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_final2 = nesterov_gd_polish(best_a, time_budget=min(remaining * 0.5, 50.0),
                                           tau_init=200.0, tau_max=8000.0)
            obj_final2 = compute_obj(a_final2)
            if obj_final2 < best_obj:
                best_obj = obj_final2
                best_a = a_final2.copy()
                print(f"  Final polish: obj={best_obj:.10f} **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        final_obj = val
    else:
        result = [1.0] * 100
        final_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={final_obj:.10f}, time={elapsed:.1f}s ===")

    return result
