"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses aggressive LP fixed-point polishing with adaptive blending, dual-guided
perturbations, and multi-n exploration.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Returns (gradient of length n, exact_max of conv).
    """
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
    # This is a cross-correlation computation
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max

def lp_solve(a, M_target=None):
    """
    Solve LP: min M s.t. conv(a,b)_k <= M for all k, b >= 0, sum(b) = 1.
    Returns (b_optimal, M_optimal, dual_weights) or (None, None, None).
    """
    n = len(a)
    if n > 3000:
        return None, None, None

    # Build convolution matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0
    # A is (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j  # when k-j = 0, k = j
        # A[k, j] = a[k-j] for k = j, j+1, ..., j+n-1
        A[j:j + n, j] = a

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if b_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(b_var.value, 0)
                sb = float(np.sum(b_val))
                if sb > 1e-10:
                    b_val = b_val / sb
                    M_val = float(M_var.value)
                    # Try to get dual variables
                    dual = None
                    try:
                        # The dual of the constraint A @ b <= M
                        dual = np.zeros(2 * n - 1)
                        # cvxpy doesn't easily give all duals, so skip
                    except:
                        pass
                    return b_val, M_val, dual
        except Exception:
            continue

    return None, None, None

def mirror_descent(a, time_budget, verbose=False, lr_init=0.1, tau=100):
    """
    Mirror descent (multiplicative weights) on the simplex.
    Natural for non-negative optimization.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    momentum = 0.9

    v = np.zeros(n)  # momentum buffer (in log-space)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        g, _ = smoothed_grad(a, tau)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Mirror descent update: a_i <- a_i * exp(-lr * g_i) / sum(...)
        # Equivalent to: log(a_i) <- log(a_i) - lr * g_i
        # But we need to handle the constraint sum(a) = 1

        # Standard multiplicative weights: a_i <- a_i * exp(-g_i / ||g||_inf)
        g_max = np.max(np.abs(g))
        if g_max < 1e-30:
            break
        updates = np.exp(-lr * g / g_max)
        a_new = a * updates
        s = a_new.sum()
        if s < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        it += 1

        # Adaptive learning rate
        if stall > 100:
            lr *= 0.7
            stall = 0
        elif it % 100 == 0 and stall == 0:
            lr = min(lr * 1.1, 1.0)

        if it % 1000 == 0 and lr < 1e-6:
            break

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_init(n, mode, rng=None):
    """Create diverse initial sequences of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:  # Triangle
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
        return a / a.sum()
    elif mode == 2:  # Semicircle
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.0001))
        return a / a.sum()
    elif mode == 3:  # Two bumps
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 4:  # Uniform random
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 5:  # Cosine bump
        x = np.linspace(0, np.pi, n)
        a = np.maximum(0.5 + 0.5 * np.cos(x), 0.001)
        return a / a.sum()
    elif mode == 6:  # Flat top with tapered edges
        x = np.linspace(0, 1, n)
        a = np.where((x > 0.15) & (x < 0.85), 1.0,
                     np.maximum(0.3 * (1 - np.abs(2 * x - 1)), 0.01))
        return a / a.sum()
    elif mode == 7:  # Raised cosine
        x = np.linspace(-1, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * x))
        a = np.maximum(a, 0.001)
        return a / a.sum()
    else:
        return np.ones(n) / n

def lp_solve_with_dual(a):
    """
    Solve LP and also extract dual variables.

    cons1 = A @ b_var <= M_var
    cons2 = cp.sum(b_var) == 1
    prob = cp.Problem(cp.Minimize(M_var), [cons1, cons2])

        try:
            prob.solve(solver=solver, verbose=False)
            if b_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

                if sb > 1e-10:

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Primary method: aggressive LP fixed-point polishing with dual-guided perturbations,
interleaved with high-tau smoothed gradient descent. Multi-n exploration with
parametric initializations. Focuses compute budget on LP convergence since it's
the most effective single-step improvement.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def staircase_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9, lr_scale=2.0):
    """Nesterov accelerated gradient descent with staircase tau schedule."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 15, 40, 100, 250, 600, 1500, 3000, 6000]

    it = 0
    best_it = 0
    stall_patience = 8000

    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0
        max_iters_per_level = 8000

        while time.monotonic() - start < time_budget:
            if tau_iters >= max_iters_per_level:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it

            a = a_new
            it += 1
            tau_iters += 1
            lr = max(1e-7, lr * 0.99997)

            if it - best_it > stall_patience:
                a = best_a.copy()
                v = np.zeros(n)
                lr = lr_scale / np.sqrt(n)
                beta = max(0.5, beta * 0.9)
                best_it = it

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    tau={tau:5d}: obj={best_obj:.10f}, iters={tau_iters}, t={elapsed:.0f}s")

def build_conv_matrix(a):
    """Build the convolution matrix A where (A@b)[k] = conv(a,b)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve_full(a):
    """
    Solve LP: min M s.t. conv(a,b)_k <= M for all k, b >= 0, sum(b) = 1.
    Returns (b, M, lambda_dual) or (None, None, None).
    """
    n = len(a)
    if n > 3500:
        return None, None, None

    A = build_conv_matrix(a)

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    cons1 = A @ b_var <= M_var
    cons2 = cp.sum(b_var) == 1
    prob = cp.Problem(cp.Minimize(M_var), [cons1, cons2])

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if b_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(b_var.value, 0)
                sb = float(np.sum(b_val))
                if sb > 1e-10:
                    b_val = b_val / sb
                    M_val = float(M_var.value)
                    lam = np.zeros(2 * n - 1)
                    try:
                        dv = cons1.dual_value
                        if dv is not None:
                            lam = np.abs(np.asarray(dv))
                            lam = np.maximum(lam, 0)
                            s_lam = lam.sum()
                            if s_lam > 1e-10:
                                lam = lam / s_lam
                    except Exception:
                        pass
                    return b_val, M_val, lam
        except Exception:
            continue
    return None, None, None


def lp_solve_linearized(a):
    """
    Solve linearized LP at point a:
    min M s.t. 2*conv(a,b)_k - conv(a,a)_k <= M for all k, b >= 0, sum(b) = 1.
    Equivalent to: 2*A@b <= M*1 + A@a
    """
    n = len(a)
    if n > 3500:
        return None, None

    A = build_conv_matrix(a)
    conv_aa = compute_conv(a)

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    # 2*A@b - M*1 <= conv_aa
    # i.e., 2*A@b - M*1 - conv_aa <= 0
    # Rewrite: A@b <= (M*1 + conv_aa) / 2
    # Or: 2*A@b - M*1 <= conv_aa
    # In cvxpy: 2*(A@b_var) - M_var <= conv_aa
    # This is: (2*A)@b_var + (-1)*M_var <= conv_aa
    # cvxpy needs: A_ub @ [b; M] <= b_ub
    # Let's just use: A@b_var - M_var/2 <= conv_aa/2
    # Better: 2*(A@b_var) - M_var <= conv_aa
    # cvxpy can handle: cp.multiply(2, A @ b_var) - M_var <= conv_aa

    cons1 = 2 * (A @ b_var) - M_var <= conv_aa
    cons2 = cp.sum(b_var) == 1
    prob = cp.Problem(cp.Minimize(M_var), [cons1, cons2])

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if b_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(b_var.value, 0)
                sb = float(np.sum(b_val))
                if sb > 1e-10:
                    b_val = b_val / sb
                    M_val = float(M_var.value)
                    return b_val, M_val
        except Exception:
            continue
    return None, None

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_init(n, mode, rng=None):
    """Create diverse initial sequences of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:  # Triangle
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
        return a / a.sum()
    elif mode == 2:  # Semicircle
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.0001))
        return a / a.sum()
    elif mode == 3:  # Raised cosine
        x = np.linspace(-1, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * x))
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 4:  # Flat top with tapered edges
        x = np.linspace(0, 1, n)
        edge = 0.2
        a = np.ones(n)
        for i in range(n):
            if x[i] < edge:
                a[i] = 0.5 * (1 + np.cos(np.pi * x[i] / edge))
            elif x[i] > 1 - edge:
                a[i] = 0.5 * (1 + np.cos(np.pi * (x[i] - (1 - edge)) / edge))
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 5:  # Gaussian bump
        x = np.arange(n)
        a = np.exp(-((x - n / 2) ** 2) / (2 * (n / 4) ** 2))
        return a / a.sum()
    elif mode == 6:  # Power cosine (flatter top)
        x = np.linspace(-1, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 7:  # Uniform random
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 8:  # Two-scale: broad + narrow
        x = np.arange(n)
        a = 0.7 * np.exp(-((x - n / 2) ** 2) / (2 * (n / 3) ** 2))
        a += 0.3 * np.exp(-((x - n / 2) ** 2) / (2 * (n / 8) ** 2))
        return a / a.sum()
    else:
        return np.ones(n) / n
