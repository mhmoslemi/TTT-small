```python
import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def mirror_descent(a, time_budget=100.0, verbose=False):
    """Mirror descent with soft-max subgradient on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 0.1  # learning rate
    tau = 0.01  # soft-max temperature
    it = 0
    patience = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        obj = 2.0 * n * M
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        # Compute soft-max weights
        # Use shifted conv to avoid overflow
        shifted = conv - M
        w = np.exp(shifted / tau)
        w_sum = np.sum(w)
        w = w / w_sum
        
        # Compute smoothed gradient
        # g_i = 2 * sum_k w_k * a[k-i] (for valid k-i)
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                j_vals = k - np.arange(lo, hi + 1)
                g[lo:hi+1] += 2.0 * w[k] * a[j_vals]
        
        # Mirror descent step
        a_new = a * np.exp(-eta * g)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        a = a_new
        it += 1
        
        # Adaptive eta
        if patience > 100:
            eta *= 0.9
            tau *= 1.1
            patience = 0
            a = best_a.copy()
        
        if verbose and it % 1000 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.4f}, tau={tau:.4f}, t={time.monotonic()-start:.1f}s")
    
    return best_a
```

So the gradient is bounded in l1 norm by 2. For mirror descent with KL divergence on the simplex, the step size eta should be O(1/L) where L is the Lipschitz constant in the dual norm (which is l1 for KL). So eta ~ O(1) should work.

I think tau should be on the order of the "gap" between the peak and the second-peak of the convolution. For a well-optimized sequence, the peaks are relatively flat, so tau should be small.

The QP for fixed k*:
min sum_i a_i * a_{k*-i} s.t. a >= 0, sum(a) = 1

This is a quadratic program with a special structure. The matrix is a permutation matrix (shifted identity), so it's actually very simple to solve.

Actually, `c_k = sum_i a_i * a_{k-i}`. The Hessian of this is `Q_k` where `(Q_k)_{ij} = 1` if `i+j=k` (i.e., `j=k-i`). So Q_k is a "anti-diagonal" matrix for the sub-diagonal at position k.

For a QP: min a^T Q a / 2 s.t. a >= 0, sum(a) = 1
The KKT conditions are: (Q a)_i + lambda - nu_i = 0 for active i, nu_i >= 0
Where (Q a)_i = 2 * a_{k-i} (as we derived before, the gradient of c_k is 2*a_{k-i}).

So the KKT: 2*a_{k-i} + lambda - nu_i = 0 for active i, nu_i >= 0
And 2*a_{k-i} + lambda >= 0 for inactive i (complementary slackness: nu_i = 0 for inactive)

This means: for active i, a_{k-i} = (nu_i - lambda) / 2 >= 0, so nu_i >= lambda
For inactive i: 2*a_{k-i} + lambda >= 0, i.e., a_{k-i} >= -lambda/2 (which is always true since a >= 0 and lambda >= 0... wait, lambda could be negative)

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def mirror_descent(a, time_budget=100.0, verbose=False):
    """Fast mirror descent with FFT-based gradient."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s
    
    fft_size = 1
    while fft_size < 3 * n - 2:
        fft_size <<= 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 0.5
    tau = 0.01
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        obj = 2.0 * n * M

            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        w = w / w_sum

    return best_a

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Build constraint matrix using convolution structure
    # A @ g = conv(a_norm, g), shape (2n-1, n)
    # A[k, j] = a_norm[k-j] if 0 <= k-j < n, else 0
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # Build A efficiently
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = a_norm[:hi - lo]
    
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val, float(M.value)
        except Exception:
            continue
    return None, None
