"""
FISTA + Log-Sum-Exp + SOCP Hybrid for minimum peak linear autocorrelation.
Primary method: Nesterov-accelerated gradient (FISTA) on the log-sum-exp
smoothed objective with adaptive temperature. Secondary: SOCP trust-region
polish. Multiple diverse initializations and length exploration for
globalization.

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    result = irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_candidates = np.where(cond)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_lse(a, M):
    """
    Compute gradient of log-sum-exp smoothing:
    phi_M(a) = (1/M) * log(sum_k exp(M * c[k]))
    where c = conv(a, a).
    
    Returns gradient vector of length n.
    """
    n = len(a)
    c = compute_conv(a)
    # Compute softmax(M * c) numerically stable
    Mc = M * c
    Mc_shifted = Mc - np.max(Mc)
    exp_vals = np.exp(Mc_shifted)
    p = exp_vals / np.sum(exp_vals)
    
    # Gradient: g[i] = 2 * sum_j a[j] * p[i+j]
    # This is cross-correlation of a with p
    # Use FFT: g = ifft(conj(fft(a)) * fft(p))[:n]
    size = 1
    while size < n + len(p) - 1:
        size <<= 1
    A = fft(a, size)
    P = fft(p, size)
    g = ifft(np.conj(A) * P, size).real[:n]
    g *= 2.0
    
    return g


def fista_lse(a0, time_budget=300.0, M_start=10.0, M_end=500.0,
               verbose=False, seed=42):

    FISTA optimization on the log-sum-exp smoothed objective.
    M increases gradually from M_start to M_end.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s  # Normalize to simplex
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    z = a.copy()  # FISTA's "look-ahead" point
    t = 1.0
    step = 1.0  # Initial step size
    
    it = 0
    total_iters = 0
    
    while time.monotonic() - start < time_budget:
        # Progress-based M schedule
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        # Use power schedule: M increases slowly at first, then quickly
        M = M_start * (M_end / M_start) ** (progress ** 0.6)
        
        # Compute gradient at z
        grad_z = compute_gradient_lse(z, M)
        
        # Ensure gradient is finite
        if not np.isfinite(grad_z).all():
            grad_z = np.where(np.isfinite(grad_z), grad_z, 0.0)
        
        # FISTA update
        x_new = project_simplex(z - step * grad_z)
        
        # Check convergence / objective improvement
        obj_new = compute_obj(x_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = x_new.copy()
        
        # Update t and z
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = x_new + ((t - 1.0) / t_new) * (x_new - a)
        t = t_new
        
        # Adapt step size (simple: decrease slowly)
        step *= 0.9998
        
        a = x_new
        it += 1
        total_iters += 1
        
        if verbose and it % 50 == 0:
            print(f"      FISTA it{it}: obj={best_obj:.8f}, M={M:.1f}, step={step:.4f}, t={time.monotonic()-start:.0f}s")

            break
        
        # Backtracking line search
        best_x = a  # Fallback: stay
        best_val = compute_lse_value(a, M)
        
        for _ in range(10):
            x_try = project_simplex(z - step * grad_z)
            val_try = compute_lse_value(x_try, M)
            if val_try <= best_val - 0.5 * step * np.sum(grad_z * (z - x_try)):
                best_x = x_try
                break
            step *= 0.5
        
        if np.array_equal(best_x, a) and it > 20:
            # No improvement, try smaller M
            step = 50.0 / np.sqrt(n)
        
        obj_new = compute_obj(best_x)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = best_x.copy()
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = best_x + ((t - 1.0) / t_new) * (best_x - a)
        t = t_new
        
        a = best_x
        it += 1
        
        if verbose and it % 100 == 0:
            print(f"      FISTA-BT it{it}: obj={best_obj:.8f}, M={M:.1f}, step={step:.4f}")
    
    return best_a


def compute_lse_value(a, M):
    """Compute the log-sum-exp value (for line search)."""
    c = compute_conv(a)
    Mc = M * c
    Mc_shifted = Mc - np.max(Mc)
    return float(np.log(np.sum(np.exp(Mc_shifted))) / M + np.max(Mc) / M)


def socp_polish(a, time_budget=200.0, r_init=0.2, verbose=False):
    """
    SOCP trust-region polish (simplified version of the previous code's approach).
    Solves: min M s.t. c_old + 2*(A_cross @ delta) <= M, ||delta|| <= r, a+delta >= 0, sum=1
    """
    import cvxpy as cp
    
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    r = r_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget:
        # Build cross-correlation matrix
        c_old = compute_conv(a)
        A_cross = build_conv_matrix(a)
        
        delta = cp.Variable(n)
        M = cp.Variable()
        a_new = a + delta
        
        constraints = [
            cp.sum(a_new) == 1,
            a_new >= 0,
            rcp_negative_one * (M * cp.ones(2*n-1)) + c_old + 2 * (A_cross @ delta) <= 0,
            cp.norm(delta, 2) <= r,
        ]
        
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        delta_val = None
        M_val = None
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    delta_val = np.asarray(delta.value).flatten()
                    M_val = float(M.value)
                    break
            except Exception:
                continue
        
        if delta_val is None:
            # Try LP direction
            g = cp.Variable(n, nonneg=True)
            M2 = cp.Variable()
            constraints2 = [A_cross @ g <= M2, cp.sum(g) == 1]
            prob2 = cp.Problem(cp.Minimize(M2), constraints2)
            for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
                try:
                    prob2.solve(solver=solver, verbose=False)
                    if g.value is not None and prob2.status in ('optimal', 'optimal_inaccurate'):
                        g_val = np.maximum(g.value, 0)
                        gs = g_val.sum()
                        if gs > 1e-10:
                            delta_val = g_val / gs - a
                        break
                except Exception:
                    continue
        
        if delta_val is None:
            break
        
        # Line search
        best_alpha = 0
        best_alpha_obj = best_obj
        for t_alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = a + t_alpha * delta_val
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj - 1e-14:
                best_alpha_obj = obj_try
                best_alpha = t_alpha
        
        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-14:
            a = a + best_alpha * delta_val
            a = np.maximum(a, 0)
            s_n = a.sum()
            if s_n > 1e-10:
                a = a / s_n
            best_obj = best_alpha_obj
            best_a = a.copy()
            no_improve = 0
            r = min(r * 1.15, 2.0)
        else:
            no_improve += 1
            r *= 0.6
            if no_improve >= 4:
                if r < 0.05:
                    # Perturb and restart
                    a = best_a * np.exp(rng.normal(0, 0.01, n))
                    a = np.maximum(a, 0)
                    s_p = a.sum()
                    if s_p > 1e-10:
                        a = a / s_p
                    r = r_init
                    no_improve = 0
                    if it > 50:
                        break
                else:
                    r = max(r * 0.5, 0.05)
        
        it += 1
        if verbose and it % 10 == 0:
            print(f"      SOCP it{it}: obj={best_obj:.8f}, r={r:.4f}, t={time.monotonic()-start:.0f}s")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix B (2n-1 x n) such that B @ x = conv(a, x) (cross-correlation)."""
    n = len(a)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        # B[k, j] = a[k-j] if 0 <= k-j < n else 0
        # For fixed j, k ranges from j to j+n-1
        for k in range(j, min(j + n, 2 * n - 1)):
            B[k, j] = a[k - j]
    return B


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def make_diverse_inits(base, n, rng):
    """Generate diverse initializations."""
    inits = []
    
    # 1. Base sequence (resampled to n)
    inits.append(resample(base, n).astype(np.float64))
    
    # 2. Linear ramp
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(ramp / ramp.sum())
    
    # 3. Exponential
    exps = np.exp(np.linspace(0, 2, n))
    inits.append(exps / exps.sum())
    
    # 4. Reverse exponential
    exps2 = np.exp(np.linspace(2, 0, n))
    inits.append(exps2 / exps2.sum())
    
    # 5. Random (Dirichlet)
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)
    
    # 6. Base with noise
    noisy = base[:n] if n <= len(base) else resample(base, n)
    noisy = noisy * np.exp(rng.normal(0, 0.3, n))
    noisy = np.maximum(noisy, 0)
    inits.append(noisy / noisy.sum())
    
    # 7. Two-level (peak at center)
    two_level = np.ones(n) / n
    center = n // 2
    spread = n // 4
    for i in range(max(0, center-spread), min(n, center+spread+1)):
        two_level[i] = 2.0 / n
    inits.append(two_level / two_level.sum())
    
    # 8. Triangle (peak at center)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float) + 1
    inits.append(tri / tri.sum())
    
    return inits
