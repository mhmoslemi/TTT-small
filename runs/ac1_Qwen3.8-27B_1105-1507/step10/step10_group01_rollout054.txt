"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses an enhanced multi-phase approach: LP-driven iteration with range-minimization LP,
coordinate descent polish, multi-n exploration with resampling, and FFT-accelerated
smoothed gradient descent for final polishing.
Key improvements: dual LP objectives (minimax + range), adaptive line search,
coordinate descent, broader n exploration, and structured perturbations.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a.astype(np.float64), size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_range(a):
    """Solve LP: min (Mmax - Mmin) s.t. Mmin <= conv(a,g)[k] <= Mmax, sum(g)=1, g>=0.
    This directly optimizes for flatness of the convolution."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    Mmax = cp.Variable()
    Mmin = cp.Variable()
    constraints = [A @ g <= Mmax, A @ g >= Mmin, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(Mmax - Mmin), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def line_search(a, direction, n_evals=40):
    """Adaptive line search: find best t to minimize obj((1-t)*a + t*direction)."""
    n = len(a)
    best_obj = compute_obj(a)
    best_t = 0.0

    # Coarse grid
    ts = np.concatenate([np.linspace(0.005, 0.3, 10),
                         np.linspace(0.3, 0.7, 15),
                         np.linspace(0.7, 0.995, 15)])[:n_evals]
    objs = np.full(len(ts), np.inf)
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try / s)

    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]

    # Parabolic refinement
    if 0 < best_idx < len(ts) - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.002, 0.998))
            a_par = (1 - t_par) * a + t_par * direction
            a_par = np.maximum(a_par, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                f_par = compute_obj(a_par / s_par)
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par

    return best_t, best_obj

def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def gd_polish(a, time_budget=100.0, tau_init=50.0, tau_end=1500.0, verbose=False):
    """Nesterov accelerated gradient descent with adaptive parameters."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 4.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 2.0

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.3 * base_lr), 3.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adapt

        # BB step
        if prev_g is not None and it % 4 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 6.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9995)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it

    if verbose:
        print(f"    GD: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.04):
    """Add Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 7):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def structured_init(n, rng, variant=0):
    """Generate structured initial sequences with known good properties."""
    if variant == 0:
        # Cosine window
        a = 0.5 * (1 + np.cos(np.pi * np.arange(n) / (n - 1)))
    elif variant == 1:
        # Gaussian window
        center = np.arange(n) - n / 2.0
        a = np.exp(-0.5 * (center / (n / 6.0)) ** 2)
    elif variant == 2:
        # Raised cosine with modulation
        base = 0.5 * (1 + np.cos(np.pi * np.arange(n) / (n - 1)))
        mod = 1.0 + 0.3 * np.cos(2 * np.pi * rng.integers(2, 10) * np.arange(n) / n)
        a = base * mod
    elif variant == 3:
        # Triangular
        a = 1.0 - 2.0 * np.abs(np.arange(n) / (n - 1) - 0.5)
    elif variant == 4:
        # Flat top
        a = np.ones(n)
        w = max(3, n // 20)
        a[:w] = np.linspace(0.5, 1, w)
        a[-w:] = np.linspace(1, 0.5, w)
    else:
        # Power cosine
        p = rng.uniform(1.5, 4.0)
        a = (0.5 * (1 + np.cos(np.pi * np.arange(n) / (n - 1)))) ** p

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase optimization combining LP-driven iteration,
    coordinate descent, multi-n exploration, and GD polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Initialize from parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(800) / 800.0
        best_score = compute_obj(parent)
        best_seq = parent.copy()
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Primary LP Optimization ===
    print(f"\n=== Phase 1: LP Optimization (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 400.0)
    if p1_budget > 30:
        a_lp = lp_driven_optimize(best_seq, time_budget=p1_budget, verbose=True, seed=42)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Coordinate Descent ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.08, 80.0)
    if p2_budget > 15:
        a_cd = coordinate_descent(best_seq, time_budget=p2_budget, verbose=True, seed=43)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase2 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Generate candidate n values
    n_candidates = []
    for offset in [-200, -150, -100, -50, 50, 100, 150, 200, 300, 400]:
        n_try = n_best + offset
        if 400 <= n_try <= 2000:
            n_candidates.append(n_try)
    # Add some fixed values
    for n_fixed in [500, 600, 700, 900, 1000, 1100, 1200, 1500]:
        if abs(n_fixed - n_best) > 50 and 400 <= n_fixed <= 2000:
            n_candidates.append(n_fixed)
    # Add prime-like values
    for n_prime in [769, 811, 997, 1009, 1049]:
        if abs(n_prime - n_best) > 30:
            n_candidates.append(n_prime)

    n_candidates = sorted(set(n_candidates))
    # Select top candidates by diversity
    selected = []
    for nc in n_candidates:
        if all(abs(nc - s) > 50 for s in selected):
            selected.append(nc)
        if len(selected) >= 6:
            break

    print(f"  Trying n values: {selected}")

    for n_try in selected:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        print(f"  --- n={n_try} ---")
        # Resample from best
        a_try = resample_sequence(best_seq, n_try)

        # Also try structured init
        a_struct = structured_init(n_try, rng, variant=n_try % 5)
        # Pick the better starting point
        obj_resample = compute_obj(a_try)
        obj_struct = compute_obj(a_struct)
        if obj_struct < obj_resample - 0.01:
            a_try = a_struct
            print(f"    Using structured init: obj={obj_struct:.8f}")
        else:
            print(f"    Using resampled: obj={obj_resample:.8f}")

        # LP optimization
        alloc_lp = min(remaining * 0.35, 50.0)
        if alloc_lp > 12:
            a_try = lp_driven_optimize(a_try, time_budget=alloc_lp, verbose=False, seed=100 + n_try)

        # CD polish
        remaining = deadline - time.monotonic()
        alloc_cd = min(remaining * 0.05, 15.0)
        if alloc_cd > 8:
            a_try = coordinate_descent(a_try, time_budget=alloc_cd, verbose=False, seed=200 + n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Second LP Round on Best ===
    print(f"\n=== Phase 4: Second LP Round (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.25, 250.0)
    if p4_budget > 20:
        # Slight perturbation before second round
        a_pert = fourier_perturb(best_seq, rng, strength=0.02)
        a_lp2 = lp_driven_optimize(a_pert, time_budget=p4_budget, verbose=True, seed=4242)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase4 LP: obj={best_score:.8f} **")
        else:
            # Try direct (no perturbation)
            a_lp2b = lp_driven_optimize(best_seq, time_budget=min(p4_budget / 3, 50.0), verbose=False, seed=999)
            obj_lp2b = compute_obj(a_lp2b)
            if obj_lp2b < best_score:
                best_score = obj_lp2b
                best_seq = a_lp2b.copy()
                print(f"  Phase4 LP (direct): obj={best_score:.8f} **")
            else:
                print(f"  Phase4 LP: obj={obj_lp2:.8f} (no improvement)")

    # === Phase 5: GD Polish ===
    print(f"\n=== Phase 5: GD Polish ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.20, 150.0)
    if p5_budget > 15:
        a_gd = gd_polish(best_seq, time_budget=p5_budget, tau_init=100.0, tau_end=2000.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase5 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 GD: obj={obj_gd:.8f} (no improvement)")

    # === Phase 6: Final Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Quick LP
    if remaining > 30:
        alloc = min(remaining * 0.4, 100.0)
        a_final = lp_driven_optimize(best_seq, time_budget=alloc, verbose=False, seed=777)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Quick CD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 40.0)
        a_final_cd = coordinate_descent(best_seq, time_budget=alloc, verbose=False, seed=888)
        obj_cd_fin = compute_obj(a_final_cd)
        if obj_cd_fin < best_score:
            best_score = obj_cd_fin
            best_seq = a_final_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
