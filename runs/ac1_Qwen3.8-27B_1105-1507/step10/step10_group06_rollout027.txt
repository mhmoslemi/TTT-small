            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_momentum(a, time_budget, beta_init=0.6, t_init=0.4, seed=42, verbose=False):
    """
    LP best-response with heavy-ball momentum and adaptive step size.
    At each step: solve LP for best response g(a), then update with momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    v = np.zeros(n)
    t = t_init
    beta = beta_init
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    iter_count = 0
    stall = 0
    consecutive_no_improve = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_lp(a)
        lp_time = time.monotonic() - t0

        if g_lp is None:
            break
        iter_count += 1

        # Heavy-ball momentum update
        v = beta * v + t * (g_lp - a)
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            v = np.zeros(n)
            a_new = best_a.copy()
            obj_new = best_obj
        else:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            consecutive_no_improve = 0
            # Gradually increase momentum
            beta = min(0.92, beta + 0.002)
            t = min(0.6, t * 1.01)
        else:
            # Backtrack: halve velocity
            v = v * 0.4
            a_new = a + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                    consecutive_no_improve += 1
                else:
                    a = best_a.copy()
                    v = np.zeros(n)
                    stall += 1
                    consecutive_no_improve += 1
            else:
                a = best_a.copy()
                v = np.zeros(n)
                stall += 1
                consecutive_no_improve += 1

            # Adaptive response to stalling
            if stall >= 3:
                # Reduce step, reset momentum
                t = max(0.05, t * 0.5)
                beta = max(0.3, beta * 0.7)
                stall = 0

            if consecutive_no_improve >= 8:
                # Fourier perturbation to escape local minimum
                strength = 0.015 + 0.008 * min(consecutive_no_improve - 8, 5)
                n_modes = 7
                pert = np.zeros(n)
                for k in range(1, n_modes + 1):
                    freq = k * 2.0 * np.pi / n
                    amp = strength * rng.uniform(0.5, 1.5) / k
                    phase = rng.uniform(0, 2.0 * np.pi)
                    pert += amp * np.cos(freq * np.arange(n) + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                a = a / s
                v = np.zeros(n)
                t = t_init * 0.6
                beta = beta_init * 0.8
                consecutive_no_improve = 0

        if verbose and iter_count % 30 == 0:
            el = time.monotonic() - start
            print(f"        LP-mom {iter_count}: obj={best_obj:.10f}, t={t:.3f}, "
                  f"beta={beta:.3f}, lp_t={lp_time:.2f}s, el={el:.1f}s")

    if verbose:
        print(f"        LP-momentum done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def coordinate_descent_fast(a, time_budget, verbose=False):
    """
    Fast coordinate descent: for each coordinate, try a few multiplier values
    and rescale others proportionally to maintain sum=1.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    a = a / s
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    multipliers = np.array([0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 3.0])

    passes = 0
    while time.monotonic() - start < time_budget:
        passes += 1
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget - 0.5:
                break

            a_j = a[j]
            others = a.copy()
            others[j] = 0.0
            others_sum = float(np.sum(others))
            if others_sum < 1e-15 or a_j < 1e-15:
                continue

            current_obj = compute_obj(a)
            best_mult = 1.0

            for mult in multipliers:
                t_new = a_j * mult
                if t_new <= 0 or t_new >= 0.99:
                    continue
                a[j] = t_new
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] = others[mask] * (1.0 - t_new) / others_sum
                obj_try = compute_obj(a)
                if obj_try < current_obj - 1e-13:
                    current_obj = obj_try
                    best_mult = mult

            if best_mult != 1.0:
                a[j] = a_j * best_mult
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] = others[mask] * (1.0 - a_j * best_mult) / others_sum
                improved = True
                if current_obj < best_obj:
                    best_obj = current_obj

        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s

        if not improved and passes > 2:
            if verbose:
                print(f"        CD converged in {passes} passes: obj={best_obj:.10f}")
            break

    if verbose:
        print(f"        CD done: {passes} passes, obj={best_obj:.10f}")
    return a


def smoothed_gradient_step(a, tau, lr):
    """One step of gradient descent on smoothed max of convolution."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    grad = 2.0 * g_full[:n]
    return grad


def gradient_polish(a, time_budget, tau_init=20.0, tau_max=2000.0, verbose=False):
    """Nesterov-style gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    beta = 0.85
    base_lr = 1.5 / np.sqrt(n)
    lr = base_lr
    tau = tau_init

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        grad = smoothed_gradient_step(y, tau, lr)
        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            adaptive_lr = 0.3 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.1 * base_lr), 3.0 * base_lr)
            lr = 0.85 * lr + 0.15 * adaptive_lr

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr *= 0.9997
        tau = min(tau * 1.03, tau_max)

    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 8
    for k in range(1, n_modes + 1):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_to_length(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def init_structured(n, mode, rng):
    """Generate structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        # Cosine window
        x = np.arange(n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * x / n)
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif mode == 2:
        # Triangular
        x = np.arange(n)
        mid = n / 2.0
        a = 1.0 - np.abs(x - mid) / mid
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif mode == 3:
        # Random sparse
        a = np.zeros(n)
        n_nonzero = max(10, n // 4)
        idx = rng.choice(n, size=n_nonzero, replace=False)
        a[idx] = rng.uniform(0.5, 2.0, size=n_nonzero)
        return a / a.sum()
    elif mode == 4:
        # Random dense
        a = rng.uniform(0.1, 1.0, size=n)
        return a / a.sum()
    elif mode == 5:
        # Geometric
        ratio = rng.uniform(0.7, 0.95)
        a = ratio ** np.arange(n)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP momentum, coordinate descent,
    gradient polish, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0)
        if parent.size == 0 or parent.sum() < 0.01:
            raise ValueError("Invalid parent")
        parent = parent / parent.sum()
        n_init = parent.size
        print(f"Parent: n={n_init}, obj={compute_obj(parent):.10f}")
    except Exception:
        n_init = 800
        parent = np.ones(n_init) / n_init
        print(f"Default init: n={n_init}")

    best_a = parent.copy()
    best_obj = compute_obj(parent)

    # === Phase 1: Multi-start initialization (120s) ===
    print("\n=== Phase 1: Multi-start Initialization ===")
    p1_deadline = start_time + 120.0
    n_candidates = [400, 600, 800, 1000]

    for n_try in n_candidates:
        if time.monotonic() > p1_deadline:
            break
        if n_try == n_init:
            continue
        for mode in range(4):
            if time.monotonic() > p1_deadline:
                break
            a_try = init_structured(n_try, mode, rng)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                print(f"  New best: n={n_try}, mode={mode}, obj={best_obj:.10f}")

    # Also try resampling parent to different lengths
    for n_try in [400, 600, 1000]:
        if time.monotonic() > p1_deadline:
            break
        if n_try == n_init:
            continue
        a_res = resample_to_length(parent, n_try)
        obj_res = compute_obj(a_res)
        if obj_res < best_obj:
            best_obj = obj_res
            best_a = a_res.copy()
            print(f"  Resampled best: n={n_try}, obj={best_obj:.10f}")

    print(f"  After Phase 1: n={len(best_a)}, obj={best_obj:.10f}")

    # === Phase 2: LP Momentum (500s) ===
    print("\n=== Phase 2: LP Momentum ===")
    p2_deadline = start_time + 620.0
    p2_budget = max(30, p2_deadline - time.monotonic())
    a_p2 = lp_momentum(best_a, time_budget=p2_budget, beta_init=0.5, t_init=0.35,
                       seed=seed, verbose=True)
    obj_p2 = compute_obj(a_p2)
    if obj_p2 < best_obj:
        best_obj = obj_p2
        best_a = a_p2.copy()
        print(f"  ** LP Momentum: obj={best_obj:.10f} **")
    else:
        print(f"  LP Momentum: obj={obj_p2:.10f} (no improvement)")

    # === Phase 3: Coordinate Descent Polish (120s) ===
    print("\n=== Phase 3: Coordinate Descent ===")
    p3_deadline = start_time + 740.0
    p3_budget = max(20, p3_deadline - time.monotonic())
    if p3_budget > 20:
        a_p3 = coordinate_descent_fast(best_a, time_budget=p3_budget, verbose=True)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3.copy()
            print(f"  ** CD: obj={best_obj:.10f} **")
        else:
            print(f"  CD: obj={obj_p3:.10f}")

    # === Phase 4: Second LP pass with perturbation (250s) ===
    print("\n=== Phase 4: LP Momentum Pass 2 ===")
    p4_deadline = start_time + 990.0
    p4_budget = max(30, p4_deadline - time.monotonic())
    if p4_budget > 30:
        a_pert = fourier_perturb(best_a, rng, strength=0.012)
        a_p4 = lp_momentum(a_pert, time_budget=p4_budget, beta_init=0.4, t_init=0.3,
                           seed=seed + 555, verbose=True)
        obj_p4 = compute_obj(a_p4)
        if obj_p4 < best_obj:
            best_obj = obj_p4
            best_a = a_p4.copy()
            print(f"  ** LP Pass 2: obj={best_obj:.10f} **")
        else:
            print(f"  LP Pass 2: obj={obj_p4:.10f}")

    # === Phase 5: Multi-length exploration (250s) ===
    print("\n=== Phase 5: Multi-length Exploration ===")
    p5_deadline = start_time + 1240.0
    n_best = len(best_a)
    n_options = sorted(set([
        max(200, n_best - 200), max(200, n_best - 100),
        min(1200, n_best + 100), min(1200, n_best + 200)
    ]))
    n_options = [n for n in n_options if n != n_best and 200 <= n <= 1200]

    for n_try in n_options[:3]:
        remaining = p5_deadline - time.monotonic()
        if remaining < 80:
            break
        a_res = resample_to_length(best_a, n_try)
        alloc = min(remaining * 0.4, 100.0)
        if alloc < 30:
            break
        a_t = lp_momentum(a_res, time_budget=alloc, beta_init=0.5, t_init=0.4,
                          seed=n_try, verbose=False)
        remaining = p5_deadline - time.monotonic()
        if remaining > 20:
            a_t = coordinate_descent_fast(a_t, time_budget=min(remaining * 0.4, 30.0))
        obj_t = compute_obj(a_t)
        if obj_t < best_obj:
            best_obj = obj_t
            best_a = a_t.copy()
            print(f"  ** n={n_try}: obj={best_obj:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_t:.10f}")

    # === Phase 6: Final gradient polish (100s) ===
    print("\n=== Phase 6: Final Gradient Polish ===")
    p6_deadline = start_time + 1340.0
    p6_budget = max(10, p6_deadline - time.monotonic())
    if p6_budget > 15:
        a_p6 = gradient_polish(best_a, time_budget=p6_budget,
                               tau_init=30.0, tau_max=1500.0)
        obj_p6 = compute_obj(a_p6)
        if obj_p6 < best_obj:
            best_obj = obj_p6
            best_a = a_p6.copy()
            print(f"  ** Grad polish: obj={best_obj:.10f} **")
        else:
            print(f"  Grad polish: obj={obj_p6:.10f}")

    # === Phase 7: Ultra-final LP (80s) ===
    print("\n=== Phase 7: Ultra-final LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.7, 80.0)
        a_final = lp_momentum(best_a, time_budget=alloc, beta_init=0.3, t_init=0.2,
                              seed=seed + 9999, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  ** Final LP: obj={best_obj:.10f} **")
        else:
            print(f"  Final LP: obj={obj_final:.10f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
