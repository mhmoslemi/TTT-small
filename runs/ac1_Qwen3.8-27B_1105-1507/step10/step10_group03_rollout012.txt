import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Returns (gradient of length n, exact_max of conv).
    """
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - exact_max))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def subgradient_peak(a, top_k=5):
    """
    Compute subgradient of max_k (a*a)[k] using top-k active lags.
    Returns gradient vector of length n.
    """
    n = len(a)
    conv = compute_conv(a)
    # Find top-k lags
    top_indices = np.argsort(conv)[::-1][:top_k]
    # Weights proportional to how close to max
    max_val = conv[top_indices[0]]
    weights = np.zeros(len(top_indices))
    for idx, k in enumerate(top_indices):
        weights[idx] = conv[k] / max_val
    weights = weights / weights.sum()
    
    g = np.zeros(n)
    for idx, k in enumerate(top_indices):
        # d(a*a)[k]/da[i] = a[k-i] if 0<=k-i<n
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] += weights[idx] * a[j]
    return g


def nesterov_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9, lr_scale=2.0,
                 max_tau_iters=3000, use_subgrad_every=500, subgrad_lr_scale=1.0):

    Nesterov accelerated GD with staircase tau schedule and periodic subgradient steps.
    """

    a = np.maximum(a, 0).copy().astype(np.float64)

        return a
    a = a / s

    best_a = a.copy()

    best_it = 0
    stall_patience = 8000

    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0
        local_best = best_obj
        local_iter = 0

        while time.monotonic() - start < time_budget:
            if tau_iters >= max_tau_iters:
                break

            # Nesterov lookahead
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            # Compute gradient at y
            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-16:
                break

            # Update
            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)

                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it
                local_best = obj_new
                local_iter = tau_iters
            a = a_new
            it += 1
            tau_iters += 1

            # Periodic subgradient step to escape smooth approximation bias
            if tau > 100 and tau_iters % use_subgrad_every == 0:
                sg = subgradient_peak(a, top_k=3)
                sg_norm = np.linalg.norm(sg)

                        if obj_sg < best_obj:

                            best_a = a_sg.copy()
                            a = a_sg.copy()

            # Slow learning rate decay

                v = np.zeros(n)

def lp_solve_given_a(a, method='auto'):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    if n > 3000:
        return None, None

    # Build A matrix: A[k,j] = a[k-j] if 0 <= k-j < n
    # A has shape (2n-1, n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j  # k-j = 0 => k = j
        end_k = min(2 * n - 1, j + n)  # k-j = n-1 => k = j+n-1
        if start_k < 2 * n - 1:
            A[start_k:end_k, j] = a[:end_k - start_k]

    # LP: min M
    # Variables: [g_0, ..., g_{n-1}, M]
    # Objective: [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # A_ub @ vars <= b_ub
    # (a*g)[k] - M <= 0  =>  A[k,:] @ g - M <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)

    # A_eq @ vars = b_eq
    # sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds: g[i] >= 0, M unbounded below (but will be >= 0 from constraints)
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method=method)
        if res.success and res.x is not None:
            g_val = np.maximum(res.x[:n], 0)
            sg = float(np.sum(g_val))
            if sg > 1e-10:
                return g_val / sg, float(res.x[-1])
    except Exception:
        pass

    return None, None

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

        alloc = min(remaining * 0.04, 25.0)
        if alloc < 10:
            alloc = 10
        a_screen = nesterov_gd(a_cand, time_budget=alloc,
                                tau_levels=[10, 50, 200], verbose=False)
        obj_screen = compute_obj(a_screen)
        if obj_screen < best_score:
            best_score = obj_screen
            best_seq = a_screen.copy()
            print(f"  Screen {i} (init obj={obj_cand:.6f}): GD -> {obj_screen:.10f} ** NEW BEST")
        else:
            print(f"  Screen {i}: GD -> {obj_screen:.10f}")

    print(f"\n  After Phase 1: best={best_score:.10f}, n={len(best_seq)}")

    # === Phase 2: Main aggressive optimization ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Main Optimization Cycle")
    print(f"{'='*60}")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 500.0)

    if p2_budget > 50:
        # Cycle between GD, LP, and subgradient refinement
        a_work = best_seq.copy()
        obj_work = compute_obj(a_work)
        p2_start = time.monotonic()
        
        cycle_len = 60  # seconds per cycle
        cycle = 0
        while time.monotonic() - p2_start < p2_budget:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            
            alloc_cycle = min(cycle_len, remaining)
            
            # Step 1: GD with increasing tau
            tau_for_cycle = [20, 80, 300, 1000, 3000, 6000, 10000][min(cycle, 6)]
            gd_budget = alloc_cycle * 0.5
            a_work = nesterov_gd(a_work, time_budget=gd_budget,
                                  tau_levels=[tau_for_cycle, tau_for_cycle * 3, tau_for_cycle * 10],
                                  verbose=False)
            obj_work = compute_obj(a_work)
            if obj_work < best_score - 1e-12:
                best_score = obj_work
                best_seq = a_work.copy()
                print(f"  Cycle {cycle} GD: obj={best_score:.10f} ** NEW BEST")
            
            # Step 2: LP fixed-point
            lp_budget = alloc_cycle * 0.4
            if n_current <= 2500:
                a_work = lp_polish_cycle(a_work, time_budget=lp_budget,
                                          verbose=False, max_iters=50,
                                          seed=100 + cycle)
                obj_work = compute_obj(a_work)
                if obj_work < best_score - 1e-12:
                    best_score = obj_work
                    best_seq = a_work.copy()
                    print(f"  Cycle {cycle} LP: obj={best_score:.10f} ** NEW BEST")
            
            # Step 3: Subgradient targeted refinement
            sg_budget = alloc_cycle * 0.1
            if sg_budget > 5:
                sg_start = time.monotonic()
                sg_lr = 0.1 / np.sqrt(n_current)
                for sg_it in range(200):
                    if time.monotonic() - sg_start > sg_budget:
                        break
                    sg = subgradient_peak(a_work, top_k=5)
                    sg_norm = np.linalg.norm(sg)
                    if sg_norm < 1e-12:
                        break
                    step = a_work - sg_lr * a_work.sum() * sg / sg_norm
                    step = np.maximum(step, 0)
                    ss = float(np.sum(step))
                    if ss < 1e-10:
                        break
                    step = step / ss
                    obj_step = compute_obj(step)
                    if obj_step < obj_work:
                        a_work = step
                        obj_work = obj_step
                        sg_lr *= 0.99  # slow decay
                    else:
                        sg_lr *= 0.95  # faster decay on failure
                        if sg_lr < 1e-6:
                            break
                if obj_work < best_score - 1e-12:
                    best_score = obj_work
                    best_seq = a_work.copy()
                    print(f"  Cycle {cycle} SG: obj={best_score:.10f} ** NEW BEST")
            
            cycle += 1
            elapsed = time.monotonic() - p2_start
            print(f"  Cycle {cycle-1} done: obj={compute_obj(best_seq):.10f}, t={elapsed:.0f}s/{p2_budget:.0f}s")

    # === Phase 3: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-length Exploration")
    print(f"{'='*60}")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 250.0)

    if p3_budget > 50:
        n_current = len(best_seq)
        n_options = sorted(set([
            n_current, n_current - 100, n_current - 200,
            n_current + 100, n_current + 200, n_current + 400,
            500, 600, 700, 800, 900, 1000, 1200, 1500, 2000
        ]))
        n_options = [n for n in n_options if 300 <= n <= 2500]
        # Sort by distance from current, prefer current
        n_options.sort(key=lambda x: abs(x - n_current))
        n_options = n_options[:7]
        
        print(f"  n candidates: {n_options}")
        
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            
            a_try = resample(best_seq, n_try)
            
            # GD
            gd_alloc = min(remaining * 0.15, 50.0)
            if gd_alloc > 15:
                a_try = nesterov_gd(a_try, time_budget=gd_alloc,
                                     tau_levels=[10, 50, 200, 800], verbose=False)
            
            # LP if small enough
            remaining = deadline - time.monotonic()
            if n_try <= 1500:
                lp_alloc = min(remaining * 0.1, 35.0)
                if lp_alloc > 15:
                    a_try = lp_polish_cycle(a_try, time_budget=lp_alloc,
                                              verbose=False, max_iters=30, seed=n_try)
            
            obj_try = compute_obj(a_try)
            if obj_try < best_score - 1e-12:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.10f} ** NEW BEST")
            else:
                print(f"  n={n_try}: obj={obj_try:.10f}")
    
    # === Phase 4: Intensive final polish ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Intensive Final Polish")
    print(f"{'='*60}")
    remaining = deadline - time.monotonic()

    if remaining > 50:
        # High-tau GD (very focused on the peak)
        alloc = min(remaining * 0.25, 120.0)
        a_f = nesterov_gd(best_seq, time_budget=alloc,
                           tau_levels=[500, 1500, 4000, 8000, 15000],
                           verbose=True, lr_scale=1.5, beta=0.9)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final GD: obj={best_score:.10f} ** NEW BEST")

        # LP polish
        remaining = deadline - time.monotonic()
        if remaining > 25 and len(best_seq) <= 2500:
            alloc = min(remaining * 0.25, 100.0)
            a_f = lp_polish_cycle(best_seq, time_budget=alloc, verbose=True,
                                    max_iters=80, seed=7777)
            obj_f = compute_obj(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final LP: obj={best_score:.10f} ** NEW BEST")

        # Subgradient intensive
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_sg = best_seq.copy()
            sg_lr = 0.05 / np.sqrt(len(a_sg))
            sg_start = time.monotonic()
            alloc_sg = min(remaining * 0.2, 60.0)
            for sg_it in range(5000):
                if time.monotonic() - sg_start > alloc_sg:
                    break
                sg = subgradient_peak(a_sg, top_k=3)
                sg_norm = np.linalg.norm(sg)
                if sg_norm < 1e-12:
                    break
                step = a_sg - sg_lr * a_sg.sum() * sg / sg_norm
                step = np.maximum(step, 0)
                ss = float(np.sum(step))
                if ss < 1e-10:
                    break
                step = step / ss
                obj_step = compute_obj(step)
                if obj_step < compute_obj(a_sg):
                    a_sg = step
                    sg_lr *= 0.999
                else:
                    sg_lr *= 0.95
                    if sg_lr < 1e-8:
                        break
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  Final SG: obj={best_score:.10f} ** NEW BEST")

        # One more GD pass on the very best
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.3, 60.0)
            a_final = nesterov_gd(best_seq, time_budget=alloc,
                                   tau_levels=[200, 600, 2000, 5000],
                                   verbose=False, lr_scale=1.0, beta=0.85)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Ultra-final GD: obj={best_score:.10f} ** NEW BEST")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
