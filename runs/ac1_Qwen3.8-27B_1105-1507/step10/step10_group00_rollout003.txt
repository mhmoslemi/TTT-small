"""
Hybrid top-k subgradient + LP polishing optimizer for step function auto-convolution peak minimization.
Uses Adam optimizer with adaptive top-k peak subgradients, LP-based direction finding,
and systematic exploration of sequence lengths.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a * a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def subgradient_topk(a, k=5):
    """Compute subgradient of the objective using top-k peaks."""
    n = len(a)
    s = np.sum(a)
    conv = compute_conv(a)
    
    # Find top-k peaks
    top_k = np.argsort(conv)[-k:]
    
    # Compute weights (softmax of top-k peaks)
    peaks = conv[top_k]
    weights = np.exp(peaks - peaks.max())
    weights /= weights.sum()
    
    # Compute subgradient
    grad = np.zeros(n)
    for w, km in zip(weights, top_k):
        for i in range(n):
            j = km - i
            if 0 <= j < n:
                grad[i] += w * 2.0 * a[j]
    
    # Full gradient of f = 2n*M/S^2
    # df/da[i] = (2n/S^2) * dM/da[i] - (2*2n*M/S^3) * S (since dS/da[i]=1)
    # = (2n/S^2) * (dM/da[i] - 2*M/S)
    M = np.max(conv)
    grad = (2.0 * n / (s * s)) * (grad - 2.0 * M / s)
    
    return grad, conv

def adam_subgradient_descent(a, time_budget=60.0, k_peaks=5, verbose=False):
    """Adam optimizer on the simplex using top-k subgradient."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))

"""
Hybrid optimizer combining top-k subgradient descent with LP polishing
for minimizing the auto-convolution peak of non-negative step function coefficients.
Uses Adam optimizer, multi-peak subgradients, aggressive LP line search,

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a * a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    import cvxpy as cp
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except Exception:
            continue
    return None, None


def subgradient_topk(a, k=5):
    """Compute subgradient using top-k peaks with softmax weighting."""
    n = len(a)
    s = float(np.sum(a))
    conv = compute_conv(a)
    M = float(np.max(conv))
    
    # Top-k peaks
    k_actual = min(k, 2 * n - 1)
    top_k_idx = np.argsort(conv)[-k_actual:]
    peaks = conv[top_k_idx]
    
    # Softmax weights (temperature controls concentration)
    temp = 0.1 * M
    weights = np.exp((peaks - peaks.max()) / temp)
    weights = weights / weights.sum()
    
    # Compute dM/da[i] as weighted sum over peaks
    dM_da = np.zeros(n)
    for w, km in zip(weights, top_k_idx):
        km = int(km)
        # dM/da[i] += w * 2*a[km-i] if valid
        i_min = max(0, km - n + 1)
        i_max = min(n - 1, km)
        if i_max >= i_min:
            j_range = km - np.arange(i_min, i_max + 1)
            dM_da[i_min:i_max + 1] += w * 2.0 * a[j_range]
    
    # Full gradient: df/da[i] = (2n/S^3) * (dM_da[i] * S - 2*M)
    grad = (2.0 * n / (s ** 3)) * (dM_da * s - 2.0 * M)
    
    return grad, conv, M

    beta2 = 0.999

    t = 0
    start = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(42)

        grad, conv, M = subgradient_topk(a, k_peaks)
        t += 1

        # Project gradient onto simplex tangent space
        grad_mean = np.mean(grad)
        grad_proj = grad - grad_mean

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_proj
        v = beta2 * v + (1 - beta2) * grad_proj * grad_proj
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)

def line_search_blend(a, g_dir, time_budget=4.0):
    """Golden section line search for optimal blend factor."""
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    lo, hi = 0.0, 1.5

    def obj_at(t):
        a_try = (1 - t) * a + t * g_dir
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try)

    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    start = time.monotonic()
    for _ in range(30):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    obj_best = min(fc, fd)
