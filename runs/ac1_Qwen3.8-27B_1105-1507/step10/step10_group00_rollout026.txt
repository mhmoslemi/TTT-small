    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v

    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_norm(a):
    """Objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    c = conv_full(a)
    return 2.0 * n * float(np.max(c))


def obj_raw(a):
    """Objective: 2*n*max_b/(sum_a)^2. Handles unnormalized input."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    an = a / s
    return obj_norm(an)


def simplex_proj(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    support = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(support):
        return np.ones(n) / n
    rho = int(np.max(np.nonzero(support)[0]))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgrad_single(a, k):
    """Subgradient of conv_k(a) at point a. Returns vector of length n."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if hi >= lo:
        j_lo = k - hi
        j_hi = k - lo
        grad[lo:hi + 1] = 2.0 * a[j_lo:j_hi + 1][::-1]
    return grad


def subgrad_smoothed(a, rel_eps=0.005):
    """Smoothed subgradient: average over near-max conv indices."""
    n = len(a)
    c = conv_full(a)
    mx = float(np.max(c))
    if mx < 1e-30:
        return np.zeros(n), c, mx
    thresh = mx * (1.0 - rel_eps)
    active = np.where(c >= thresh)[0]

    grad = np.zeros(n)
    for k in active:
        grad += subgrad_single(a, int(k))
    grad /= float(len(active))
    return grad, c, mx


def nesterov_optimize(a0, time_budget=100.0, verbose=False, seed=42):
    """Nesterov-accelerated subgradient descent on the simplex."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 0.01:
        return a0
    a = a / s

    best_a = a.copy()
    best_obj = obj_norm(a)
    start = time.monotonic()

    x = a.copy()
    t = 1.0
    lr = 0.3
    patience = 0
    max_patience = 3000
    it = 0
    eps = 0.02
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new

        y = a + beta * (a - x)
        y = np.maximum(y, 0.0)
        sy = float(np.sum(y))
        if sy < 0.01:
            y = a.copy()
        else:
            y = y / sy

        grad, c, mx = subgrad_smoothed(y, eps=eps)

        step = lr / t_new
        a_new = y - step * grad
        a_new = simplex_proj(a_new)

        x = a.copy()
        a = a_new

        ot = obj_norm(a)
        if ot < best_obj:
            best_obj = ot
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            # Annealing restart
            scale = 1.0 + 0.01 * rng.standard_normal()
            if rng.random() < 0.3:
                # Fourier perturbation
                freq = rng.integers(1, max(2, n // 100))
                phase = rng.uniform(0, 2 * np.pi)
                pert = 0.02 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
                a = best_a * np.exp(pert)
            else:
                noise = rng.normal(0, 0.01, n)
                a = best_a + noise
            a = np.maximum(a, 0)
            sa = float(np.sum(a))
            if sa < 0.01:
                a = best_a.copy()
            else:
                a = a / sa
            x = a.copy()
            t = 1.0
            lr = max(0.05, lr * 0.6)
            eps = max(0.001, eps * 0.5)
            patience = 0

        it += 1
        t = t_new

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    NSD it{it}: obj={best_obj:.8f}, lr={lr:.4f}, eps={eps:.4f}, t={elapsed:.1f}s")

    return best_a


def mirror_descent(a0, time_budget=60.0, verbose=False, seed=42):
    """Mirror descent (multiplicative weights) on the simplex."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 0.01:
        return a0
    a = a / s

    best_a = a.copy()
    best_obj = obj_norm(a)
    start = time.monotonic()
    lr = 0.5
    patience = 0
    max_patience = 2000
    it = 0
    eps = 0.01
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        grad, c, mx = subgrad_smoothed(a, eps=eps)

        log_a = np.log(np.maximum(a, 1e-30)) - lr * grad
        log_a -= np.max(log_a)
        weights = np.exp(log_a)
        z = float(np.sum(weights))
        if z < 1e-30:
            break
        a = weights / z

        ot = obj_norm(a)
        if ot < best_obj:
            best_obj = ot
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            a = best_a.copy()
            lr *= 0.5
            eps = max(0.001, eps * 0.5)
            patience = 0
            if lr < 0.001:
                break
            # Small perturbation
            noise = rng.normal(0, 0.005, n)
            a = np.maximum(a + noise, 0)
            a = a / a.sum()

        it += 1
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it{it}: obj={best_obj:.8f}, lr={lr:.4f}, t={elapsed:.1f}s")

    return best_a


def peak_equalize(a, time_budget=30.0, verbose=False):
    """
    Try to equalize the top convolution values by adjusting a.
    Uses a Newton-like step on the active set.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 0.01:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_norm(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start >= time_budget:
            break

        c = conv_full(a)
        mx = float(np.max(c))
        # Get top 10 values
        top_ks = np.argsort(c)[::-1][:10]
        top_vals = c[top_ks]
        mean_top = float(np.mean(top_vals))

        # If already well-equalized, stop
        if np.max(top_vals) - np.min(top_vals) < 1e-8 * mx:
            break

        # Compute a direction that reduces the spread of top values
        # For each top k, we want to reduce conv_k if it's above mean
        # The gradient tells us how to do this
        grad = np.zeros(n)
        for k in top_ks:
            weight = 1.0 if c[k] > mean_top else -1.0
            grad += weight * subgrad_single(a, int(k))

        # Project gradient to the tangent space of the simplex (sum of adjustments = 0)
        grad -= np.mean(grad)

        # Step
        lr = 0.01
        a_new = a - lr * grad
        a_new = simplex_proj(a_new)

        ot = obj_norm(a_new)
        if ot < best_obj:
            best_obj = ot
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.5
            if lr < 1e-6:
                break

    return best_a


def u_shape_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 0.01:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.01 + 5.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        a = 1.0 + 3.0 * np.cos(2 * np.pi * (2 * x - 1)) ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 0.01:
        return np.ones(n) / n
    return a / s


def interpolate_to_length(a_old, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a_old)
    if n_old == n_new:
        return a_old.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a_old)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 0.01:
        return np.ones(n_new) / n_new
    return a_new / s


def combined_optimize(a0, time_budget=100.0, verbose=False, seed=42):
    """Run multiple optimization methods and return the best result."""
    n = len(a0)
    best = a0.copy().astype(np.float64)
    s = float(np.sum(best))
    if s < 0.01:
        return a0
    best = best / s
    best_obj = obj_norm(best)

    if time_budget < 10:
        return best

    # Phase A: Nesterov subgradient
    t_a = min(time_budget * 0.5, time_budget - 10)
    if t_a > 5:
        result = nesterov_optimize(best, time_budget=t_a, verbose=verbose, seed=seed)
        obj_r = obj_norm(result)
        if obj_r < best_obj:
            best_obj = obj_r
            best = result.copy()
            if verbose:
                print(f"      After NSD: {best_obj:.8f}")

    remaining = time_budget - (time.monotonic() - (time.monotonic() - t_a))  # approximate
    remaining = min(time_budget * 0.25, 40.0)

    # Phase B: Mirror descent from current best
    if remaining > 5:
        result = mirror_descent(best, time_budget=remaining, verbose=verbose, seed=seed + 1)
        obj_r = obj_norm(result)
        if obj_r < best_obj:
            best_obj = obj_r
            best = result.copy()
            if verbose:
                print(f"      After MD: {best_obj:.8f}")

    remaining = min(time_budget * 0.2, 30.0)

    # Phase C: Peak equalization
    if remaining > 3:
        result = peak_equalize(best, time_budget=remaining, verbose=verbose)
        obj_r = obj_norm(result)
        if obj_r < best_obj:
            best_obj = obj_r
            best = result.copy()
            if verbose:
                print(f"      After PE: {best_obj:.8f}")

    return best


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines Nesterov subgradient descent, mirror descent,
    peak equalization, and multi-n exploration to minimize the step function objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Load parent and initialize ---
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj_p = obj_norm(parent)
                best_score = obj_p
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj_p:.8f}")
    except Exception:
        print("No parent found, using default init.")

    # Try several initializations
    inits = []
    if best_seq is not None:
        inits.append(("parent", best_seq.copy()))

    for mode, name in [(0, "u4"), (4, "exp"), (5, "u4b"), (6, "cos2")]:
        a_init = diverse_init(n_current, rng, mode=mode)
        obj_i = obj_norm(a_init)
        inits.append((name, a_init))
        print(f"  Init '{name}' (n={n_current}): obj={obj_i:.8f}")
        if obj_i < best_score:
            best_score = obj_i
            best_seq = a_init.copy()

    u_init = u_shape_init(n_current, sharpness=5.0)
    obj_u = obj_norm(u_init)
    inits.append(("u5", u_init))
    print(f"  Init 'u5' (n={n_current}): obj={obj_u:.8f}")
    if obj_u < best_score:
        best_score = obj_u
        best_seq = u_init.copy()

    # === Phase 1: Main optimization on best init (n = n_current) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 1: Main optimization (n={n_current}), budget={min(400, remaining*0.4):.0f}s ===")
        alloc = min(400.0, remaining * 0.4)
        result = combined_optimize(best_seq, time_budget=alloc, verbose=True, seed=seed)
        obj_r = obj_norm(result)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = result.copy()
            print(f"  Phase 1 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: no improvement (obj={obj_r:.8f})")

    # === Phase 2: Nesterov with different params ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: Extended Nesterov, budget={min(200, remaining*0.3):.0f}s ===")
        alloc = min(200.0, remaining * 0.3)
        result = nesterov_optimize(best_seq, time_budget=alloc, verbose=True, seed=seed + 100)
        obj_r = obj_norm(result)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = result.copy()
            print(f"  Phase 2 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: no improvement (obj={obj_r:.8f})")

    # === Phase 3: Peak equalization + final NSD ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 3: Peak equalize + polish ===")
        alloc_pe = min(30.0, remaining * 0.15)
        alloc_ns = min(100.0, remaining * 0.4)
        if alloc_pe > 5:
            result = peak_equalize(best_seq, time_budget=alloc_pe, verbose=True)
            obj_r = obj_norm(result)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = result.copy()
                print(f"  PE: obj={best_score:.8f} **")
        if alloc_ns > 10:
            result = nesterov_optimize(best_seq, time_budget=alloc_ns, verbose=True, seed=seed + 200)
            obj_r = obj_norm(result)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = result.copy()
                print(f"  Final NSD: obj={best_score:.8f} **")

    # === Phase 4: Explore different n values ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 4: n Exploration ===")
        n_best = len(best_seq)
        n_options = []
        for dn in [-100, -50, 50, 100, 200, -200, 300]:
            nt = n_best + dn
            if 400 <= nt <= 2000 and abs(nt - n_best) >= 30:
                n_options.append(nt)
        # Also try some round numbers
        for nt in [600, 700, 800, 900, 1000, 1200, 1500]:
            if nt not in n_options and abs(nt - n_best) >= 50:
                n_options.append(nt)
        n_options = sorted(set(n_options))[:6]
        print(f"  Trying n values: {n_options}")

        per_n_budget = min(60.0, remaining * 0.12)
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < per_n_budget + 10:
                break

            a_try = interpolate_to_length(best_seq, n_try)
            obj_try = obj_norm(a_try)
            print(f"  n={n_try}: interp obj={obj_try:.8f}", end="")

            a_try = combined_optimize(a_try, time_budget=per_n_budget, verbose=False, seed=seed + n_try)
            obj_try = obj_norm(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f" -> {obj_try:.8f} **")
            else:
                print(f" -> {obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 5: Diverse Restarts ===")
        n_cur = len(best_seq)
        per_restart = min(50.0, remaining * 0.15)

        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < per_restart + 5:
                break

            mode = r % 7
            a_r = diverse_init(n_cur, rng, mode=mode)
            # Blend with current best
            blend = 0.3 + 0.1 * r
            a_r = (1.0 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            a_r = combined_optimize(a_r, time_budget=per_restart, verbose=False, seed=seed + 1000 + r)
            obj_r = obj_norm(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 6: Final Polish (budget={min(remaining-10, 120):.0f}s) ===")
        alloc = min(remaining - 10, 120.0)
        result = nesterov_optimize(best_seq, time_budget=alloc, verbose=True, seed=seed + 999)
        obj_r = obj_norm(result)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = result.copy()
            print(f"  Final: obj={obj_r:.8f} **")
        else:
            # Try peak equalize as last resort
            result2 = peak_equalize(best_seq, time_budget=min(30, remaining - 15), verbose=False)
            obj_r2 = obj_norm(result2)
            if obj_r2 < best_score:
                best_score = obj_r2
                best_seq = result2.copy()
                print(f"  Final PE: obj={obj_r2:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0 / len(result)] * len(result)
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
