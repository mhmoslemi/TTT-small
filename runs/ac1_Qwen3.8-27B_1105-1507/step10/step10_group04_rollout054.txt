    stall = 0
    last_t = 0.3
    momentum = np.zeros(n)
    warm_g = None

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp(a, warm_g)
        if g_lp is None:
            break
        warm_g = g_lp.copy()

        # Adaptive mixing: try multiple t values
        t_candidates = np.unique(np.clip(np.concatenate([
            np.linspace(0.05, 1.0, 15),
            [max(0.02, last_t * 0.5), last_t * 0.7, last_t, last_t * 1.3, min(1.0, last_t * 1.7)]
        ]), 0.01, 1.0))

        best_t_obj = best_obj
        best_t = last_t
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-12:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            # Apply momentum
            momentum = 0.5 * momentum + 0.5 * (best_a_local - a)
            a_mom = best_a_local + 0.1 * momentum
            a_mom = np.maximum(a_mom, 0)
            s_mom = float(np.sum(a_mom))
            if s_mom > 1e-10:
                a_mom = a_mom / s_mom
                obj_mom = compute_obj(a_mom)
                if obj_mom < best_t_obj:
                    best_a_local = a_mom
                    best_t_obj = obj_mom

            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t
            stall = 0
            if verbose and stall == 0 and (time.monotonic() - start) % 15 < 2:
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.4f}, elapsed={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 5:
                break

    return best_a


def compute_subgrad(a, p):
    """Compute subgradient of max(conv) using probability p on active set.
    grad_j = 2 * sum_k p_k * (a_{k-j} + a_{j+k}) with boundary handling.
    Uses FFT for efficiency."""
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A_f = fft(a, N)
    g_full = ifft(P * np.conj(A_f)).real
    return 2.0 * g_full[:n]


def subgradient_backtrack(a, time_budget, tau_start=100, tau_max=20000, verbose=True):
    """Subgradient method with backtracking line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    eta = 0.1

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_max - tau_start) * progress ** 0.7

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        grad = compute_subgrad(a, p)

        # Project gradient: ensure sum(grad) adjustment for normalization
        grad_centered = grad - np.mean(grad)

        g_norm = float(np.linalg.norm(grad_centered))
        if g_norm < 1e-15:
            break

        # Backtracking line search
        eta_try = eta
        found = False
        for _ in range(15):
            a_try = a - eta_try * grad_centered
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                eta_try *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                eta = min(eta * 1.05, 1.0 / (g_norm + 1e-10))
                found = True
                break
            eta_try *= 0.5

        if not found:
            eta *= 0.7
            # Try a small perturbation to escape
            if it % 50 == 0:
                a_pert = a + 0.01 * np.random.randn(n)
                a_pert = np.maximum(a_pert, 0)
                s_pert = float(np.sum(a_pert))
                if s_pert > 1e-10:
                    a_pert = a_pert / s_pert
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                        a = a_pert

        a = best_a if a is None else a
        it += 1
        if it - best_it > 5000:
            a = best_a.copy()
            best_it = it
            eta = 0.5 / (float(np.linalg.norm(compute_subgrad(a, p))) + 1e-10)

        if verbose and it % 2000 == 0:
            print(f"    SubGrad it{it}: obj={best_obj:.8f}, tau={tau:.0f}, eta={eta:.6f}, t={elapsed:.0f}s")

    return best_a


def high_tau_gd(a, time_budget, tau_start=200, tau_end=15000, verbose=True):
    """High-precision smoothed gradient descent with adaptive step and momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.6

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(y, N)
        g_full = ifft(P * np.conj(A_f)).real
        g = 2.0 * g_full[:n]
        g = g - np.mean(g)  # Center for normalization invariance

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Adaptive step size
        lr = min(1.0 / (g_norm * np.sqrt(tau) + 1e-10), 5.0 / np.sqrt(n))
        lr = max(lr, 1e-10)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        beta = min(0.97, beta + 0.00005)

        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it
            if it > 30000:
                break

        if verbose and it % 3000 == 0:
            print(f"    HTGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def log_param_optimize(a, time_budget, tau_min=200, tau_max=10000, verbose=True):
    """Optimize using log-parameterization with L-BFGS on smooth approximation."""
    full_len = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Identify support
    threshold = 1e-6
    support = np.where(a > threshold)[0]
    if len(support) < 3:
        threshold = a.max() * 1e-4
        support = np.where(a > threshold)[0]
    if len(support) < 2:
        return a

    b0 = np.log(np.maximum(a[support], 1e-30))
    b0 = b0 - b0.mean()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    n_stages = 4
    tau_stages = []
    for i in range(n_stages):
        tau_stages.append(tau_min + (tau_max - tau_min) * (i / max(1, n_stages - 1)) ** 2)

    current_b = b0.copy()
    for stage_idx, tau in enumerate(tau_stages):
        if time.monotonic() - start > time_budget:
            break

        stage_alloc = 30
        def obj_grad(b_vec):
            b_shifted = b_vec - b_vec.max()
            a_supp = np.exp(b_shifted)
            a_supp = a_supp / a_supp.sum()
            a_full = np.zeros(full_len)
            a_full[support] = a_supp
            conv = compute_conv(a_full)
            max_conv = float(np.max(conv))
            exp_vals = np.exp(tau * (conv - max_conv))
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp
            obj_val = max_conv + np.log(sum_exp) / tau
            N = 1
            while N < 3 * full_len:
                N <<= 1
            P = fft(p, N)
            A_full_f = fft(a_full, N)
            g_full = ifft(P * np.conj(A_full_f)).real
            grad_full = 2.0 * g_full[:full_len]
            grad_b = a_supp * (grad_full[support] - np.sum(a_supp * grad_full[support]))
            return obj_val, grad_b

        try:
            res = scipy_minimize(obj_grad, current_b, method='L-BFGS-B',
                                 jac=True, options={'maxiter': 100, 'gtol': 1e-10,
                                                    'ftol': 1e-14, 'maxfun': 300})
            if res.x is not None:
                b_shifted = res.x - res.x.max()
                a_supp = np.exp(b_shifted)
                a_supp = a_supp / a_supp.sum()
                a_new = np.zeros(full_len)
                a_new[support] = a_supp
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    current_b = res.x.copy()
                    if verbose:
                        print(f"    LogOpt stage{stage_idx}: tau={tau:.0f}, obj={best_obj:.8f}")
        except Exception:
            pass

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.5
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
        a = a / a.sum()
    elif mode == 5:
        a = np.exp(-np.arange(n) / (n * 0.2)) + 0.3 * np.exp(-np.arange(n) / (n * 0.8))
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x ** 3 * (1 - x)
        a = a / a.sum()
    elif mode == 7:
        # Cosine-based
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = a / a.sum()
    elif mode == 8:
        # Gaussian
        x = np.linspace(-2, 2, n)
        a = np.exp(-x ** 2)
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return np.maximum(a, 0)


def resample_sequence(a, new_n):
    """Resample sequence to new length via spline interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, kind='cubic')
    return np.maximum(a_new, 0)


def support_shrink(a, time_budget, verbose=True):
    """Try zeroing out small entries and re-optimizing."""
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    sorted_vals = np.sort(a)[::-1]
    for frac in [0.25, 0.4, 0.5, 0.6, 0.75, 0.85]:
        if time.monotonic() - start > time_budget:
            break
        k = max(3, int(len(a) * frac))
        threshold = sorted_vals[k - 1] if k <= len(sorted_vals) else 0
        support = np.where(a >= threshold)[0]
        if len(support) < 3:
            continue
        a_shrunk = np.zeros(len(a))
        a_shrunk[support] = a[support]
        a_shrunk = a_shrunk / a_shrunk.sum()

        # Quick GD on reduced support
        alloc = min(15, time_budget - 5 - (time.monotonic() - start))
        if alloc > 3:
            a_shrunk = high_tau_gd(a_shrunk, time_budget=alloc,
                                   tau_start=100, tau_max=3000, verbose=False)
        obj_shrunk = compute_obj(a_shrunk)
        if obj_shrunk < best_obj - 1e-12:
            best_obj = obj_shrunk
            best_a = a_shrunk.copy()
            if verbose:
                print(f"    Shrink frac={frac}: obj={best_obj:.8f}, support={len(support)}")

    return best_a

        alloc = min(remaining * 0.3, 150.0)
        a_final = high_tau_gd(best_seq, time_budget=alloc,
                              tau_start=1000, tau_end=30000, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Round 3: Subgradient
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 80.0)
        a_final = subgradient_backtrack(best_seq, time_budget=alloc,
                                        tau_start=500, tau_max=30000, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final SubGrad: obj={best_score:.8f} **")

    # Round 4: Log-param
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.2, 60.0)
        a_final = log_param_optimize(best_seq, time_budget=alloc,
                                     tau_min=500, tau_max=15000, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Log: obj={best_score:.8f} **")

    # Round 5: Support shrink + refine
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.15, 40.0)
        a_final = support_shrink(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Shrink: obj={best_score:.8f} **")

    # Round 6: Ultra-final LP + GD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_lp = min(remaining * 0.4, 40.0)
        if alloc_lp > 10:
            a_final = lp_polish(best_seq, time_budget=alloc_lp, verbose=False)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Ultra LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc_gd = min(remaining * 0.5, 25.0)
        if alloc_gd > 5:
            a_final = high_tau_gd(best_seq, time_budget=alloc_gd,
                                  tau_start=2000, tau_end=30000, verbose=False)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Ultra GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
