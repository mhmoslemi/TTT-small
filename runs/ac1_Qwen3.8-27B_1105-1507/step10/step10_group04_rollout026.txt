"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP fixed-point iteration, subgradient refinement with Polyak step,
active-set QP, symmetry exploitation, and aggressive multi-n exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_lp(a):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value) if M.value is not None else compute_obj(g_val) / (2 * n)
                    return g_val, M_val
        except Exception:
            continue
    return None, None

def lp_fixed_point(a, time_budget, verbose=True):
    """LP fixed-point iteration with adaptive mixing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    t = 0.5
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        
        # Armijo line search
        obj_cur = compute_obj(a)
        # Approximate gradient direction
        direction = g_lp - a
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-12:
            break
        
        # Try several step sizes
        best_t = t
        best_t_obj = best_obj
        
        t_candidates = np.clip(np.array([t * 0.5, t * 0.7, t * 0.9, t, t * 1.2, t * 1.5, t * 2.0]), 0.01, 1.0)
        t_candidates = np.unique(t_candidates)
        
        for t_try in t_candidates:
            a_try = (1.0 - t_try) * a + t_try * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t_try
        
        if best_t_obj < best_obj - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            best_obj = best_t_obj
            best_a = a_new.copy()
            a = a_new
            t = best_t
        else:
            # Try smaller step
            t = max(0.01, t * 0.5)
            a_new = (1.0 - t) * a + t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                # Stuck, try LP solution directly
                obj_direct = compute_obj(g_lp)
                if obj_direct < best_obj - 1e-14:
                    best_obj = obj_direct
                    best_a = g_lp.copy()
                    a = g_lp.copy()
                    t = 1.0
                else:
                    break
    
    return best_a

def subgradient_polyak(a, time_budget, lower_bound=1.0, verbose=True):
    """Subgradient method with Polyak step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Work on simplex: a is in simplex (sum=1, a>=0)
    # Subgradient of f(a) = max_k (a*a)_k at a is: 2 * a * p where p is the "maximizer"
    # i.e., p_k = 1 if (a*a)_k = max, else 0 (or any convex combination of maximizers)
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        obj_cur = 2.0 * n * max_conv
        
        # Active set (with smoothing)
        tau = 50.0 + 100.0 * (time.monotonic() - start) / max(time_budget, 1)
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp
        
        # Subgradient: grad_j = 2 * sum_k p_k * a_{k-j}
        # This is 2 * (p * a) (cross-correlation)
        from scipy.fft import fft, ifft
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(a, N)
        g_full = ifft(P * np.conj(A_f)).real
        grad = 2.0 * g_full[:n]
        
        # Polyak step size
        gap = obj_cur - 2.0 * n * lower_bound / (2 * n - 1)  # rough lower bound
        # Actually, use a tighter lower bound
        avg_conv = float(np.sum(conv))  # = 1 since sum(a)=1
        lb = 2.0 * n * avg_conv / (2 * n - 1)
        gap = obj_cur - lb
        if gap < 1e-12:
            break
        
        g_norm_sq = float(np.dot(grad, grad))
        if g_norm_sq < 1e-20:
            break
        
        step = 2.0 * gap / g_norm_sq
        step = min(step, 5.0 / np.linalg.norm(grad))  # cap
        
        a_new = a - step * grad
        # Project onto simplex
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def smoothed_gd(a, time_budget, tau_start=50, tau_end=5000, verbose=True):
    """Smoothed gradient descent with Nesterov acceleration on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    v = np.zeros(n)
    
    from scipy.fft import fft, ifft
    
    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.3
        
        # Nesterov
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp
        
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(y, N)
        g_full = ifft(P * np.conj(A_f)).real
        g = 2.0 * g_full[:n]
        
        # Step size
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        lr = min(1.0 / (g_norm + 1e-10), 3.0 / np.sqrt(n))
        lr = max(lr, 1e-8)
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        it += 1
        beta = min(0.98, 0.9 + 0.02 * progress)
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.5
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
    elif mode == 5:
        a = np.exp(-np.arange(n) / (n * 0.2)) + 0.3 * np.exp(-np.arange(n) / (n * 0.8))
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x ** 3 * (1 - x)
    elif mode == 7:
        # Symmetric bell
        x = np.linspace(-1, 1, n)
        a = np.exp(-x**2)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP fixed-point iteration with adaptive mixing, Nesterov-accelerated smoothed
gradient descent, subgradient with Polyak step, symmetry exploitation,
KKT-based refinement, and aggressive multi-n exploration with diverse inits.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value) if M.value is not None else compute_obj(g_val) / (2.0 * n)
                    return g_val, M_val
        except Exception:
            continue
    return None, None


def smoothed_grad(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using softmax smoothing."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A_f = fft(a, N)
    g_full = ifft(P * np.conj(A_f)).real
    grad = 2.0 * g_full[:n]
    return grad, max_conv, p


def lp_fixed_point(a, time_budget, verbose=True):
    """LP fixed-point iteration with adaptive line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    t = 0.5
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Line search over mixing parameter
        t_candidates = np.unique(np.clip(
            np.array([t * 0.5, t * 0.7, t * 0.85, t, t * 1.15, t * 1.4, t * 1.8, 1.0]),
            0.01, 1.0))

        best_t = t
        best_t_obj = best_obj
        best_a_try = None

        for t_try in t_candidates:
            a_try = (1.0 - t_try) * a + t_try * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-14:
                best_t_obj = obj_try
                best_t = t_try
                best_a_try = a_try.copy()

        if best_a_try is not None:
            best_obj = best_t_obj
            best_a = best_a_try
            a = best_a
            t = best_t
            stall = 0
            if verbose and (time.monotonic() - start) % 15 < 2:
                print(f"    LP iter: obj={best_obj:.8f}, t={best_t:.3f}, t={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            t = max(0.01, t * 0.5)
            if stall > 4:
                # Try direct LP solution
                obj_direct = compute_obj(g_lp)
                if obj_direct < best_obj - 1e-14:
                    best_obj = obj_direct
                    best_a = g_lp.copy()
                    a = g_lp.copy()
                    t = 1.0
                    stall = 0
                else:
                    break

    return best_a


def nesterov_smoothed_gd(a, time_budget, tau_start=50, tau_end=10000, verbose=True):
    """Nesterov-accelerated smoothed gradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(float(time_budget), 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.4

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
        else:
            y = y / s_y

        # Smoothed gradient
        grad, max_conv, p = smoothed_grad(y, tau)

        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break

        # Adaptive step size
        lr = min(1.0 / (g_norm + 1e-10), 4.0 / np.sqrt(n))
        lr = max(lr, 1e-9)

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        beta = min(0.98, 0.9 + 0.08 * progress)

        # Reset if stuck
        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            it = best_it
            if it > 60000:
                break

        if verbose and it % 5000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.0f}s")

    return best_a


def subgradient_method(a, time_budget, verbose=True):
    """Subgradient method with Polyak step size on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        obj_cur = 2.0 * n * max_conv

        # Lower bound from average convolution value
        avg_conv = 1.0 / (2 * n - 1)  # since sum(conv) = 1 when sum(a) = 1
        lb = 2.0 * n * avg_conv
        gap = obj_cur - lb
        if gap < 1e-12:
            break

        # Use smoothed distribution for subgradient
        tau = 100.0 + 200.0 * (time.monotonic() - start) / max(time_budget, 1.0)
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = exp_vals / sum_exp

        # Subgradient via FFT
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(a, N)
        g_full = ifft(P * np.conj(A_f)).real
        grad = 2.0 * g_full[:n]

        g_norm_sq = float(np.dot(grad, grad))
        if g_norm_sq < 1e-20:
            break

        # Polyak step
        step = 2.0 * gap / g_norm_sq
        step = min(step, 3.0 / (float(np.linalg.norm(grad)) + 1e-10))

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    return best_a


def kkt_refine(a, time_budget, verbose=True):
    """KKT-based refinement: adjust support based on KKT conditions."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Compute subgradient at current point
        tau = 200.0
        grad, max_conv, p = smoothed_grad(a, tau)

        # KKT: for a_j > 0, grad_j should be approximately constant (= lambda)
        # For a_j = 0, grad_j should be >= lambda
        positive = a > 1e-8
        if np.sum(positive) < 2:
            break

        lambda_val = float(np.mean(grad[positive]))
        variance = float(np.var(grad[positive]))

        # If variance is small, we're near KKT
        if variance < 1e-16:
            break

        # Identify entries that should be zero (grad >> lambda)
        should_zero = positive & (grad > lambda_val + 5.0 * float(np.std(grad[positive]) + 1e-15))
        # Identify zero entries that should be positive (grad << lambda)
        should_add = (~positive) & (grad < lambda_val - 5.0 * float(np.std(grad[positive]) + 1e-15))

        if np.sum(should_zero) == 0 and np.sum(should_add) == 0:
            break

        # Apply modification
        a_new = a.copy()
        if np.sum(should_zero) > 0:
            a_new[should_zero] = 0.0
        if np.sum(should_add) > 0:
            a_new[should_add] = 1e-4

        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Quick LP polish after support change
        lp_alloc = min(10, remaining - 2)
        if lp_alloc > 3:
            a_new = lp_fixed_point(a_new, time_budget=lp_alloc, verbose=False)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.5
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
    elif mode == 5:
        a = np.exp(-np.arange(n) / (n * 0.2)) + 0.3 * np.exp(-np.arange(n) / (n * 0.8))
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = np.maximum(x ** 3 * (1 - x), 1e-6)
    elif mode == 7:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2)
    elif mode == 8:
        x = np.linspace(0, 1, n)
        a = (1.0 - 4.0 * (x - 0.25) * (x - 0.75)) ** 2
    elif mode == 9:
        # Cosine taper
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
    elif mode == 10:
        # Two-peaked symmetric
        x = np.linspace(-1, 1, n)
        a = np.exp(-((x - 0.4) ** 2) / 0.1) + np.exp(-((x + 0.4) ** 2) / 0.1)
    else:
        a = rng.uniform(0, 1, n)
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s > 0:
        a = a / s
    return a


def symmetric_project(a):
    """Project onto symmetric sequences."""
    return 0.5 * (a + a[::-1])


def optimize_candidate(a, time_budget, verbose=False):
    """Run a mini optimization pipeline on a candidate."""
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    # LP (if n small enough)
    lp_alloc = time_budget * 0.4
    if lp_alloc > 8 and n <= 2000:
        a = lp_fixed_point(a, time_budget=lp_alloc, verbose=verbose)

    # GD
    remaining = time_budget - lp_alloc
    gd_alloc = max(5, remaining * 0.5)
    if gd_alloc > 5:
        a = nesterov_smoothed_gd(a, time_budget=gd_alloc,
                                  tau_start=50, tau_end=5000, verbose=verbose)

    # Subgradient
    remaining2 = time_budget - lp_alloc - gd_alloc
    if remaining2 > 5:
        a = subgradient_method(a, time_budget=remaining2, verbose=verbose)

    return a

            s = float(np.sum(parent_arr))

        pass

    if best_seq is None:
        n_init = 800
        best_seq = diverse_init(n_init, rng, mode=7)

        print(f"Default init (n={n_init}): obj={best_score:.8f}")

        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.8f}")
    else:
        print("  Skipped (n too large or insufficient time)")

    # === Phase 2: Nesterov GD (200s) ===
    print(f"\n=== Phase 2: Nesterov GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.18, 200.0)
    if p2_budget > 20:
        a_gd = nesterov_smoothed_gd(best_seq, time_budget=p2_budget,
                                     tau_start=50, tau_end=12000, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase2 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 GD: obj={obj_gd:.8f}")

    # === Phase 3: Subgradient (80s) ===
    print(f"\n=== Phase 3: Subgradient ===")

    p3_budget = min(remaining * 0.07, 80.0)

        a_sub = subgradient_method(best_seq, time_budget=p3_budget, verbose=True)
        obj_sub = compute_obj(a_sub)

            print(f"  Phase3 Sub: obj={best_score:.8f} **")

    # === Phase 4: KKT Refinement (60s) ===
    print(f"\n=== Phase 4: KKT Refine ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.05, 60.0)
    if p4_budget > 15 and len(best_seq) <= 2000:

        obj_kkt = compute_obj(a_kkt)
        if obj_kkt < best_score:
            best_score = obj_kkt
            best_seq = a_kkt.copy()
            print(f"  Phase4 KKT: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 KKT: obj={obj_kkt:.8f}")

            a_r = a_r / s_r

                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  R{round_num} LP: obj={best_score:.8f} **")

        # GD with high tau
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.25, 100.0)
            a_final = nesterov_smoothed_gd(best_seq, time_budget=alloc,
                                           tau_start=500, tau_end=20000, verbose=False)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  R{round_num} GD: obj={best_score:.8f} **")

        # Subgradient
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.15, 40.0)
            a_final = subgradient_method(best_seq, time_budget=alloc, verbose=False)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  R{round_num} Sub: obj={best_score:.8f} **")

        # KKT
        remaining = deadline - time.monotonic()
        if remaining > 15 and len(best_seq) <= 2000:
            alloc = min(remaining * 0.15, 30.0)
            a_final = kkt_refine(best_seq, time_budget=alloc, verbose=False)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  R{round_num} KKT: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
