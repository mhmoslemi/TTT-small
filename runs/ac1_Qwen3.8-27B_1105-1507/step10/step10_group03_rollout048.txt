import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lp_solve(a, solver_list=None):
    """
    Solve the LP: minimize M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    if n > 3000:
        return None, None

    # Build convolution matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if solver_list is None:
        solver_list = [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue

    return None, None

            t_candidates = sorted(set(t_candidates + [
                max(0.02, cur_t * 0.7), max(0.02, cur_t * 0.85),
                cur_t, min(1.0, cur_t * 1.15), min(1.0, cur_t * 1.3)
            ]))
            
            best_t_obj = best_obj
            best_t = cur_t
            best_a_local = best_a.copy()
            
            for t in t_candidates:
                if use_geometric:
                    # Geometric mean: a_new = normalize(a^α * g^β) where α=1-t, β=t
                    # Handle zeros: where a==0, use g directly; where g==0, use a directly
                    a_pos = np.maximum(a, 1e-300)
                    g_pos = np.maximum(g_lp, 1e-300)
                    a_new = a_pos ** (1 - t) * g_pos ** t
                else:
                    a_new = (1.0 - t) * a + t * g_lp
                a_new = np.maximum(a_new, 0)
                s_new = float(np.sum(a_new))
                if s_new < 1e-10:
                    continue
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_t_obj:
                    best_t_obj = obj_new
                    best_t = t
                    best_a_local = a_new.copy()
            
            if best_t_obj < best_obj - 1e-14:
                best_obj = best_t_obj
                best_a = best_a_local.copy()
                a = best_a
                cur_t = best_t
                stall_count = 0
            else:
                a = (1.0 - min(cur_t, 0.3)) * a + min(cur_t, 0.3) * g_lp
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count += 1
                cur_t = max(0.05, cur_t * 0.7)
        else:
            # Fixed t
            a_new = (1.0 - cur_t) * a + cur_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                stall_count = 0
            a = a_new

        # Periodic perturbation
        if perturb_every > 0 and it % perturb_every == perturb_every - 1 and stall_count > 5:
            # Perturb based on autocorrelation structure
            conv = compute_conv(a)
            peak_lag = int(np.argmax(conv))
            # Reduce weight at positions that contribute to the peak
            # (a*a)[peak_lag] = sum_i a[i]*a[peak_lag-i]
            # So positions i where a[peak_lag-i] is large are "responsible"
            responsibility = np.zeros(n)
            for i in range(n):
                j = peak_lag - i
                if 0 <= j < n:
                    responsibility[i] = a[j]
            # Add noise weighted by responsibility
            noise = rng.uniform(0, 0.01, n) * responsibility / max(responsibility.max(), 1e-10)
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall_count = 0

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP it {it}: obj={best_obj:.10f}, t={cur_t:.3f}, stall={stall_count}, t_elapsed={elapsed:.0f}s")

        # Early termination if converged
        if stall_count > 40:
            break

    return best_a

def fast_smoothed_grad(a, tau):
    """Fast computation of smoothed gradient using FFT."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted -= shifted.max()
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    # grad[j] = 2 * sum_k p[k] * a[k-j] for j in [0, n), a[x]=0 outside [0,n)
    # Use FFT: zero-pad both to length >= 2n-1 + n - 1 = 3n-2
    N = 1
    while N < 3 * n:
        N <<= 1
    
    P = fft(p.astype(np.float64), N)
    A = fft(a.astype(np.float64), N)
    # cross-correlation: ifft(conj(P) * A) gives sum_k p[k] a[k-m] at position m
    # Wait: ifft(conj(P) * A)[m] = (1/N) sum_k conj(P[k]) A[k] e^{2πimk/N}
    # = (1/N) sum_k P[conj at k] ... this is the circular cross-correlation
    # For zero-padded signals, circular = linear for the relevant range.
    
    # Let me verify: ifft(conj(fft(p)) * fft(a))[m] = sum_k p[k] a[(m+k) mod N]
    # Hmm, that doesn't seem right either. Let me think again.
    
    # The circular cross-correlation of p and a is:
    # (p ⊛ a)[m] = sum_k p[k] a[(m+k) mod N]
    # And its FFT is conj(FFT(p)) * FFT(a).
    
    # But we want sum_k p[k] a[k-m] (note the sign difference).
    # sum_k p[k] a[k-m] = sum_k p[k+m] a[k] = (p_flipped ⊛ a)[m]
    # where p_flipped[k] = p[-k mod N] = p[N-k] (for k != 0)
    # FFT(p_flipped) = conj(FFT(p))
    # So ifft(FFT(p_flipped) * FFT(a))[m] = ifft(conj(FFT(p)) * FFT(a))[m] = sum_k p_flipped[k] a[(m+k) mod N]
    # = sum_k p[-k mod N] a[(m+k) mod N]
    # Let j = -k mod N: = sum_j p[j] a[(m-j) mod N]
    # 
    # So ifft(conj(FFT(p)) * FFT(a))[m] = sum_j p[j] a[(m-j) mod N]
    # For zero-padded p (length 2n-1) and a (length n), with N >= 3n:
    # For m in [0, n-1]: sum_j p[j] a[(m-j) mod N]
    # = sum_j p[j] a[m-j] (for m-j in [0, n-1]) + 0 (elsewhere)
    # = sum_{j=m-n+1}^{m} p[j] a[m-j] ... wait, this gives a[m-j] not a[j-m].
    
    # Hmm, let me reconsider. We want:
    # grad[m] = 2 * sum_k p[k] * a[k-m]
    # 
    # From above: ifft(conj(FFT(p)) * FFT(a))[m] = sum_j p[j] a[(m-j) mod N]
    # We want sum_k p[k] a[k-m]. Let's substitute: this is sum_k p[k] a[k-m].
    # From the formula with m replaced by -m: ifft(conj(FFT(p)) * FFT(a))[-m mod N] = sum_j p[j] a[(-m-j) mod N]
    # = sum_j p[j] a[N-m-j] (for N-m-j in [0, N-1])
    # This is getting confusing. Let me just use a direct but efficient approach.
    
    # Direct approach using the structure:
    # grad[m] = 2 * sum_k p[k] a[k-m] where a[x]=0 for x<0 or x>=n
    # = 2 * sum_{k=m}^{m+2n-2} p[k] a[k-m]   (since a[k-m] is nonzero only for 0<=k-m<n)
    #     (also need k in [0, 2n-2])
    # = 2 * sum_{i=0}^{n-1} p[i+m] a[i]   (substituting i=k-m, with 0<=i<n and 0<=i+m<=2n-2)
    
    # So grad[m] = 2 * sum_{i=0}^{n-1} p[i+m] * a[i]
    # This is like a "toeplitz" matrix multiplication where the (m,i) entry is p[i+m].
    # We can compute this as a series of dot products, or use FFT.
    
    # Using FFT: define q[i+m] = p[i+m]. For fixed m, this is a shift of p.
    # grad[m] = 2 * <shift_m(p), a> = 2 * <p, shift_{-m}(a)>

    # is the cross-correlation of p and a at lags 0,...,n-1.
    
    # Let's use: result = ifft(FFT(p, N) * conj(FFT(a, N)))
    # result[m] = sum_k p[k] a[(k-m) mod N] (circular cross-correlation, conjugate version)
    # Wait, I keep going back and forth. Let me just test numerically.
    
    # For now, let me use a simple O(n * len(p)) approach which is O(n * 2n) = O(2n^2)
    # For n=2000, that's 8 million operations - fast enough.
    
    # Actually, for the GD loop we call this many times, so let me get the FFT right.
    
    # Let me define it clearly:
    # I want C[m] = sum_i p[i+m] * a[i] for m = 0, 1, ..., n-1
    # (where p is defined on [0, 2n-2] and a on [0, n-1])
    # 
    # C = P @ a where P[m, i] = p[i+m]
    # P is a (n x n) Toeplitz-like matrix.
    #
    # Note that C[m] = sum_i p[m+i] a[i] = (reverse of the cross-correlation)
    # 
    # Actually, C[m] = sum_i p[m+i] a[i]
    # Let u[m] = p[m] (length 2n-1), v[i] = a[i] (length n)
    # C[m] = sum_i u[m+i] v[i] = cross_correlation(u, v) at lag m
    #       = (u * flip(v))[m + (n-1)] where * is linear convolution and flip(v)[i] = v[n-1-i]
    
    # So: compute flip_a = a[::-1]
    # conv = linear_convolution(p, flip_a)  (length (2n-1) + n - 1 = 3n-2)
    # C[m] = conv[m + n - 1] for m = 0, ..., n-1
    
    # Using FFT:
    N2 = 1
    while N2 < (2 * n - 1) + n - 1:
        N2 <<= 1
    P_fft = fft(p.astype(np.float64), N2)
    FA_fft = fft(a[::-1].astype(np.float64), N2)
    conv_result = ifft(P_fft * FA_fft).real
    C = conv_result[n - 1 : 2 * n - 1]  # C[m] = conv_result[m + n - 1]
    
    g = 2.0 * C
    return g, exact_max


def staircase_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9, lr_scale=2.0):
    """Nesterov accelerated gradient descent with staircase tau schedule."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 20, 80, 300, 1000, 3000]

    it = 0
    best_it = 0
    stall_patience = 8000

    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0
        max_iters_per_level = 5000
        local_best = best_obj

        while time.monotonic() - start < time_budget:
            if tau_iters >= max_iters_per_level:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = fast_smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it
            if obj_new < local_best - 1e-15:
                local_best = obj_new

            a = a_new
            it += 1
            tau_iters += 1
            lr = max(1e-8, lr * 0.99999)

            if it - best_it > stall_patience:
                a = best_a.copy()
                v = np.zeros(n)
                lr = lr_scale / np.sqrt(n)
                best_it = it

        if verbose:
            elapsed = time.monotonic() - start
            print(f"      tau={tau:5d}: obj={best_obj:.10f}, iters={tau_iters}, t={elapsed:.0f}s")

    return best_a


def make_init(n, mode, rng=None):
    """Create diverse initial sequences of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x**2, 0.0001))
        return a / a.sum()
    elif mode == 3:
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c1)**2) / (2 * w**2))
        a += np.exp(-((np.arange(n) - c2)**2) / (2 * w**2))
        return a / a.sum()
    elif mode == 4:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, np.pi, n)
        a = np.maximum(0.5 + 0.5 * np.cos(x), 0.001)
        return a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [2, 3, 5, 7]:
            a += 0.15 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 4))
        return a / a.sum()
    elif mode == 8:
        x = np.linspace(0, 1, n)
        a = np.where((x > 0.15) & (x < 0.85), 1.0,
                     np.maximum(0.4 - 2 * np.abs(x - 0.5), 0.01))
        return a / a.sum()
    elif mode == 9:
        # Multi-peak with varying widths
        a = np.zeros(n)
        num_peaks = rng.integers(2, 6)
        for _ in range(num_peaks):
            center = rng.integers(0, n)
            width = rng.integers(max(1, n // 20), max(2, n // 5))
            height = rng.uniform(0.5, 1.0)
            a += height * np.exp(-((np.arange(n) - center)**2) / (2 * width**2))
        a = np.maximum(a, 0.001)
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

    if remaining > 20 and len(best_seq) <= 2500:

    return result
