```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

# Global FFT sizes for different n values (precompute)
def get_fft_size(n):
    """Get the FFT size needed for convolution of two length-n sequences."""
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size

def get_fft_size_grad(n):
    """Get the FFT size needed for gradient computation."""
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    return size

def compute_conv(a, fft_size=None):
    n = len(a)
    if fft_size is None:
        fft_size = get_fft_size(n)
    fa = rfft(a, fft_size)
    return irfft(fa * fa, fft_size)[:2*n-1]

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def grad_on_simplex(a, tau, N):
    """Compute gradient of 2n*LSE(tau*conv(a,a)) for a on simplex. N is precomputed FFT size."""
    n = len(a)
    conv = compute_conv(a)
    
    # Compute p = softmax(tau * conv)
    max_c = np.max(conv)
    shifted = tau * (conv - max_c)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    
    # Compute cross-correlation
    A = np.zeros(N)
    A[:n] = a
    P = np.zeros(N)
    P[:2*n-1] = p
    X = ifft(fft(P) * np.conj(fft(A))).real[:n]
    
    return 2.0 * n * 2.0 * X  # = 4n * X

def fast_subgrad(a, time_budget, tau_start, tau_end, lr, momentum=0.9, 
                 N=None, verbose=False):
    """Fast projected subgradient with momentum on the simplex."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    if N is None:
        N = get_fft_size_grad(n)
    
    best_a = a.copy()
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    
    v = np.zeros(n)
    start = time.monotonic()
    it = 0
    best_it = 0
    prev_grad_norm = None
    prev_a = a.copy()
    prev_grad = None
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        tau = tau_start + (tau_end - tau_start) * progress**2
        
        grad = grad_on_simplex(a, tau, N)
        g_norm = np.linalg.norm(grad)
        
        if g_norm < 1e-15:
            break
        
        # Barzilai-Borwein step size
        if prev_grad is not None and it % 2 == 0:
            s = a - prev_a
            y = grad - prev_grad
            ss = np.dot(s, s)
            sy = np.dot(s, y)
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                bb_lr = np.clip(bb_lr, 0.1 * lr, 10.0 * lr)
                lr = 0.8 * lr + 0.2 * bb_lr
        
        # Momentum step
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        
        # Evaluate
        a_new_conv = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(a_new_conv))
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        prev_a = a.copy()
        prev_grad = grad.copy()
        
        # Periodic restart
        if it - best_it > 3000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = max(1e-10, lr * 0.5)
            best_it = it
            momentum = 0.95  # increase momentum after restart
        
        # Slow decay of lr
        lr = max(1e-12, lr * 0.99995)
        
        if verbose and it % 10000 == 0:
            print(f"  Subgrad it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.2e}")
    
    return best_a
```
