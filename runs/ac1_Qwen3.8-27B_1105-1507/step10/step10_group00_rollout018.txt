"""
Optimized step function coefficient search using:
1. FFT-based fast convolution
2. Reduced-constraint LP polishing with adaptive constraint selection
3. Multi-point line search for optimal blending
4. Systematic n-length exploration with interpolation
5. Diverse restarts with smooth parametric initializations
6. Nesterov-accelerated subgradient descent for warm-up

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_full(a):
    """Solve full LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def subgradient_descent(a, time_budget=30.0, verbose=False):
    """Nesterov-accelerated subgradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.05
    it = 0
    momentum = np.zeros(n)
    beta = 0.7
    best_it = 0
    y = a.copy()  # Nesterov lookahead
    
    while time.monotonic() - start < time_budget:
        # Compute convolution and gradient at the peak
        conv = compute_conv(y)
        k_max = int(np.argmax(conv))
        
        # Subgradient of max(conv[y,y]) w.r.t. y
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += y[j1]
            if 0 <= j2 < n:
                grad[i] += y[j2]
        
        # Nesterov update
        momentum = beta * momentum - lr * grad
        a_new = y + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
            y = a_new.copy()
        else:
            a_new = a_new / s_new
            y = a_new + beta * (a_new - y) if it > 0 else a_new.copy()
            # Actually simpler: y = a_new (standard Nesterov)
            y = a_new.copy()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        lr *= 0.999
        it += 1
        
        if it - best_it > 3000:
            a = best_a.copy()
            y = a.copy()
            momentum = np.zeros(n)
            lr = 0.02
            best_it = it
    
    return best_a
