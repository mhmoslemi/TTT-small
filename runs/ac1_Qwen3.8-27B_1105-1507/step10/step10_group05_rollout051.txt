"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses symmetric LP iteration with Nesterov momentum, subgradient polishing,
and multi-length exploration. Exploits convolution symmetry to halve the LP size.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT. Returns array of length len(a)+len(b)-1."""
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:na + nb - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a, n):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(j, j + n):
            if k < 2 * n - 1:
                A[k, j] = a[k - j]
    return A


def build_symmetric_conv_matrix(a, n):
    """Build matrix A (n x m) for symmetric LP.
    A @ x gives conv(a, g_sym)[:n] where g_sym is determined by x.
    m = (n+1)//2 free variables.
    """
    m = (n + 1) // 2
    A = np.zeros((n, m))
    for k in range(n):
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                idx = min(kj, n - 1 - kj)
                A[k, idx] += a[j]
    return A


def symmetrize(a):
    """Make a sequence symmetric."""
    n = len(a)
    return (a + a[::-1]) / 2.0


def is_symmetric(a, tol=1e-8):
    return np.allclose(a, a[::-1], atol=tol)


def subgradient_step(a, lr):
    """One subgradient step for minimizing max(conv(a,a)) on the simplex."""
    n = len(a)
    a_norm = a / a.sum()
    conv = compute_conv(a_norm)
    k_star = int(np.argmax(conv))
    
    # Subgradient: g_i = a_{k*-i} + a_{i-k*} (with zero padding)
    g = np.zeros(n)
    for i in range(n):
        val = 0.0
        if 0 <= k_star - i < n:
            val += a_norm[k_star - i]
        if 0 <= i - k_star < n:
            val += a_norm[i - k_star]
        g[i] = 2.0 * val  # factor of 2 from the quadratic
    
    # Projected gradient step
    a_new = project_simplex(a_norm - lr * g)
    return a_new, k_star


def nesterov_subgradient_descent(a, time_budget=50.0, lr_init=0.01, verbose=False):
    """Nesterov accelerated projected subgradient descent."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    y = a.copy()
    v = np.zeros(n)
    t = 1.0
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        lr = (t - 1) / t_new  # Nesterov step size
        
        # Nesterov extrapolation
        y_new = y + (t - 1) / t_new * (y - a)
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-10:
            y_new = a.copy()
        else:
            y_new = y_new / s
        
        # Subgradient
        conv = compute_conv(y_new)
        k_star = int(np.argmax(conv))
        g = np.zeros(n)
        for i in range(n):
            val = 0.0
            if 0 <= k_star - i < n:
                val += y_new[k_star - i]
            if 0 <= i - k_star < n:
                val += y_new[i - k_star]
            g[i] = 2.0 * val
        
        # Update
        a_new = project_simplex(y_new - lr * g / max(n, 1))
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            
        a = a_new
        y = y_new
        t = t_new
        it += 1
        
        if it - best_it > 5000:
            a = best_a.copy()
            y = a.copy()
            v = np.zeros(n)
            t = 1.0
            best_it = it
    
    return best_a


def solve_lp_full(a):
    """Solve LP: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0. Full (non-symmetric)."""
    n = len(a)
    a_norm = a / a.sum()
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # Build constraints using the convolution structure
    # conv(a, g)_k = sum_j a_j * g_{k-j} for valid j, k-j
    # We build this as A @ g where A is (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for kk in range(n):
            k = j + kk  # k = j + (k-j), where k-j = kk
            if k < 2 * n - 1:
                A[k, j] = a_norm[kk]
    
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:
            continue
    
    return None, None


def solve_lp_symmetric(a):
    """Solve LP with symmetry constraint on g.
    min M s.t. conv(a,g_sym)_k <= M, sum(g_sym)=1, g_sym>=0, g_sym is symmetric.
    Assumes a is symmetric.
    """
    n = len(a)
    a_norm = a / a.sum()
    m = (n + 1) // 2
    
    # Free variables: x[0], ..., x[m-1]
    # g_sym[i] = x[min(i, n-1-i)]
    x = cp.Variable(m, nonneg=True)
    M = cp.Variable()
    
    # Build constraint matrix: A (n x m) such that A @ x = conv(a, g_sym)[:n]
    A = np.zeros((n, m))
    for k in range(n):
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                idx = min(kj, n - 1 - kj)
                A[k, idx] += a_norm[j]
    
    # Constraint: sum(g_sym) = 1
    # sum(g_sym) = sum of x[i] weighted by how many times each appears
    # x[i] appears in g_sym at positions i and n-1-i (same position if i = n-1-i)
    # For i < n-1-i: x[i] appears twice (at positions i and n-1-i)
    # For i = n-1-i (only when n is odd, i = (n-1)/2): x[i] appears once
    sum_weights = np.zeros(m)
    for i in range(m):
        if i == n - 1 - i:
            sum_weights[i] = 1.0
        else:
            sum_weights[i] = 2.0
    
    constraints = [A @ x <= M, sum_weights @ x == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.maximum(x.value, 0)
                # Reconstruct g_sym
                g_sym = np.zeros(n)
                for i in range(m):
                    g_sym[i] = x_val[i]
                    if n - 1 - i != i:
                        g_sym[n - 1 - i] = x_val[i]
                s = float(np.sum(g_sym))
                if s > 1e-10:
                    g_sym = g_sym / s
                    return g_sym, float(M.value)
        except Exception:
            continue
    
    return None, None


def lp_iteration_symmetric(a, time_budget=300.0, verbose=False, seed=42):
    """LP iteration with symmetry, momentum, and adaptive step size."""
    n = len(a)
    a = symmetrize(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    momentum_vec = np.zeros(n)
    prev_g = None
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        # Solve LP
        g_lp, M_lp = solve_lp_symmetric(a)
        if g_lp is None:
            # Fallback to full LP
            g_lp, M_lp = solve_lp_full(a)
            if g_lp is None:
                break
            g_lp = symmetrize(g_lp)
            g_lp = g_lp / g_lp.sum()
        
        it += 1
        
        # Momentum
        if prev_g is not None:
            momentum_vec = 0.8 * momentum_vec + 0.2 * (g_lp - prev_g)
            g_mom = g_lp + 0.3 * momentum_vec
            g_mom = np.maximum(g_mom, 0)
            s_mom = g_mom.sum()
            if s_mom > 1e-10:
                g_mom = g_mom / s_mom
                # Use momentum version if it's not too different
                if compute_obj(g_mom) < best_obj * 1.1:
                    g_lp = g_mom
        prev_g = g_lp.copy()
        
        # Adaptive step size via backtracking
        t = 1.0
        best_t = 0.0
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        
        # Try a range of step sizes
        t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if it > 10:
            t_candidates = [0.3, 0.5, 0.7, 0.9, 1.0, 1.2, 1.5]
        
        for tc in t_candidates:
            if tc <= 1.0:
                a_try = (1.0 - tc) * a + tc * g_lp
            else:
                a_try = tc * g_lp - (tc - 1.0) * a
            a_try = np.maximum(a_try, 0)
            a_try = symmetrize(a_try)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = tc
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            best_it = it
            if verbose and it % 5 == 0:
                print(f"  LP it{it}: obj={best_obj:.8f}, t_best={best_t:.2f}, M={M_lp:.6f}")
        else:
            # No improvement - try perturbation
            if it - best_it > 10:
                pert = rng.normal(0, 0.01 * (it - best_it), n)
                a_pert = best_a * np.exp(pert)
                a_pert = symmetrize(a_pert)
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                a = a_pert
                prev_g = None
                momentum_vec = np.zeros(n)
                if verbose:
                    print(f"  LP it{it}: perturbed (stuck for {it-best_it} iters)")
            
            if it - best_it > 30:
                # Reset
                a = best_a.copy()
                prev_g = None
                momentum_vec = np.zeros(n)
                best_it = it
    
    return best_a

            g_mom = np.maximum(g_mom, 0)

            if s > 1e-10:
                g_mom = g_mom / s
                if compute_obj(g_mom) < best_obj * 1.05:
                    g_lp = g_mom
        prev_g = g_lp.copy()
        
        # Step size search
        t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]

            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:

            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = tc
                best_a_local = a_try.copy()

            best_a = best_a_local.copy()

                a_pert = a_pert / a_pert.sum()
                a = a_pert
                prev_g = None
                best_it = it
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    return a_new / a_new.sum()
