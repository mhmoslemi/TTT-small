```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

`= sum_n sum_m p[n] a[m] * δ_{(m-n+j) mod N, 0}`

`f(a) = 2n * max_k conv(a,a)_k`

The smoothed version:
`f_smooth(a) = 2n * (1/tau) * log(sum_k exp(tau * conv(a,a)_k))`

The gradient of `f_smooth` with respect to `a_j`:
`d f_smooth / da_j = 2n * (1/tau) * (1/sum_k exp(tau*conv_k)) * sum_k exp(tau*conv_k) * d(conv_k)/da_j`
`= 2n * sum_k p_k * 2 * a_{k-j}` (where `p_k = exp(tau*conv_k) / sum_m exp(tau*conv_m)`)
`= 4n * sum_k p_k * a_{k-j}`

Wait, I had a factor of 2 for the derivative of the auto-convolution. Let me recheck.

`conv(a,a)_k = sum_j a_j * a_{k-j}`
`d(conv(a,a)_k) / da_{j'} = a_{k-j'} + a_{k-j'} = 2 * a_{k-j'}` (as derived earlier)

So `d f_smooth / da_{j'} = 2n * sum_k p_k * 2 * a_{k-j'} = 4n * sum_k p_k * a_{k-j'}`

And `sum_k p_k * a_{k-j'}` is the cross-correlation we computed above.

So the gradient on the simplex is:
`grad[j'] = 4n * (p ⋆ a)[j']`

And we project this onto the tangent space of the simplex (subtract the mean) before taking a step.

Actually, for projected gradient on the simplex, we don't need to project the gradient onto the tangent space. We just take a step in the negative gradient direction and project back onto the simplex.

FISTA algorithm:

For the Lipschitz constant: `||grad(f_smooth(a)) - grad(f_smooth(b))|| <= L * ||a - b||`

This is hard to bound exactly, but we can estimate it empirically or use a backtracking line search.

Let me use a backtracking line search instead of a fixed L:

```

    compute grad at y
    try step size alpha
    if f(y - alpha*grad) <= f(y) - 0.5*alpha*||grad||^2: accept
```

        # Cosine annealing for tau
        cos_val = 0.5 * (1.0 + np.cos(np.pi * progress))
        tau = tau_init + (tau_end - tau_init) * (1.0 - cos_val)
        
        # Compute gradient at y
        _, grad, conv, p = smooth_obj_grad(y, tau)
        
        # FISTA extrapolation is already done (y was set at end of previous iter)
        # Take gradient step and project
        a_new = project_simplex(y - alpha * grad)
        
        # Backtracking line search (check on the actual objective, not smoothed)
        obj_y = compute_obj(y)
        obj_new = compute_obj(a_new)
        backtracks = 0
        while obj_new > obj_y - 1e-14 and backtracks < 15:
            alpha *= 0.5
            a_new = project_simplex(y - alpha * grad)
            obj_new = compute_obj(a_new)
            backtracks += 1
        
        alpha = min(alpha * 1.2, 5.0 / (4.0 * n * tau))
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
        y_new = a_new + ((t_fista - 1.0) / t_new) * (a_new - prev_a)
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = a_new.copy()
        else:
            y_new = y_new / s_new
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a_new.copy()
        t_fista = t_new
        y = y_new
        it += 1
        
        if verbose and it % 200 == 0:
            print(f"    FISTA it{it}: obj={best_obj:.10f}, tau={tau:.1f}, alpha={alpha:.8f}, bt={backtracks}")
        
        # Stagnation detection
        if it - best_it > 5000:
            # Reset momentum
            y = best_a.copy()
            prev_a = best_a.copy()
            t_fista = 1.0
            alpha = 0.5 / (4.0 * n * tau)
            best_it = it
            if verbose:
                print(f"    FISTA it{it}: stagnation reset")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, warm_start=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns optimal g vector or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None
