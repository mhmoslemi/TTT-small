import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_smoothed_grad(a, T):
    """
    Compute gradient of smoothed objective (1/T)*log(sum exp(T*c_k))
    where c = conv(a,a), assuming sum(a)=1.
    Returns gradient vector of length n.
    """
    n = len(a)
    conv = compute_conv(a)
    # Compute softmax weights with numerical stability
    Tc = T * conv
    Tc_max = np.max(Tc)
    exp_Tc = np.exp(Tc - Tc_max)
    p = exp_Tc / np.sum(exp_Tc)
    
    # Compute gradient: grad[m] = 2 * sum_j p[m+j] * a[j]
    # Using convolution: grad[m] = 2 * convolve(p[::-1], a, 'full')[2n-2-m]
    p_rev = p[::-1]
    conv_g = np.convolve(p_rev, a, mode='full')
    # grad[m] = 2 * conv_g[2*n-2-m] for m = 0, ..., n-1
    grad = 2.0 * conv_g[2*n-2 : 2*n-2+n-1 : -1]
    
    return grad, conv, float(np.max(conv))

"""
Hybrid mirror descent + LP polishing for step function coefficient optimization.
Uses log-sum-exp smoothed objective with mirror descent (exponential weights) for fast
local optimization, temperature annealing for smooth-to-sharp transition, and LP polishing
for final refinement. Explores multiple sequence lengths.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_smoothed_grad(a, alpha):
    """
    Compute gradient of smoothed objective log(sum_k exp(alpha*(c_k - max_c)))
    where c = conv(a,a), assuming sum(a)=1 (so c values are ~1/n scale).
    Returns (grad, max_c, p) where grad has length n.
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    # Softmax with numerical stability
    scaled = alpha * (conv - max_c)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / np.sum(exp_scaled)

    # Compute gradient: grad[m] = 2 * sum_j p[m+j] * a[j] for m=0,...,n-1
    # Using: convolve(p[::-1], a, 'full')[n-1 : 2n-1][::-1] * 2
    p_rev = p[::-1]
    conv_g = np.convolve(p_rev, a, mode='full')
    # indices n-1 to 2n-2 (inclusive), reversed
    grad = 2.0 * conv_g[n - 1:2 * n - 1][::-1]

    return grad, max_c, p


def mirror_descent_step(a, grad, lr):
    """Mirror descent update: a * exp(-lr*grad), normalized."""
    # Clip gradient for stability
    grad_clipped = np.clip(grad, -10.0, 10.0)
    logits = -lr * grad_clipped
    # Subtract max for numerical stability
    logits -= np.max(logits)
    weights = np.exp(logits)
    a_new = a * weights
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-15:
        return a.copy()
    return a_new / s


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization: high at edges, low in middle."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(2.0 * (2 * x - 1) - np.log(2))
    elif mode == 7:
        # Power-law edge concentration
        d = np.minimum(idx, n - 1 - idx).astype(float) + 1
        a = d ** (-0.5)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.SCIP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g_target, time_budget=3.0):
    """Golden section search for optimal blend factor."""
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    lo, hi = 0.0, 1.0

    def obj_at(t):
        a_try = (1 - t) * a + t * g_target
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    start = time.monotonic()
    for _ in range(30):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)
    return t_best, obj_best


def mirror_descent_optimize(a, time_budget=100.0, alpha_start=1.0, alpha_end=80.0,
                            lr_init=0.5, max_iters=5000, verbose=False, seed=42):
    """
    Mirror descent with temperature annealing on the smoothed objective.
    a should be normalized (sum=1, non-negative).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    lr = lr_init
    it = 0
    stall = 0
    best_it = 0

    n_anneal = max(1, int(time_budget * 0.6))
    # We'll use iteration-based annealing: alpha increases over first 60% of budget

    while time.monotonic() - start < time_budget and it < max_iters:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(1, n_anneal))
        # Exponential annealing of alpha
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        grad, max_c, p = compute_smoothed_grad(a, alpha)

        # Mirror descent step
        a_new = mirror_descent_step(a, grad, lr)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it
            a = a_new
            stall = 0
            # Slightly increase lr when making progress
            lr = min(lr * 1.01, 5.0)
        else:
            # Backtracking: reduce lr
            lr *= 0.7
            if lr < 0.01:
                # Stuck, try perturbation
                stall += 1
                if stall > 20:
                    # Restart with perturbation
                    noise = rng.normal(0, 0.01 * stall, n)
                    a = best_a * np.exp(noise)
                    a = np.maximum(a, 0)
                    s2 = float(np.sum(a))
                    if s2 > 0.01:
                        a = a / s2
                    else:
                        a = best_a.copy()
                    lr = lr_init * 0.5
                    stall = 0
                a = mirror_descent_step(a, grad, lr * 0.5)
                s2 = float(np.sum(a))
                if s2 > 0.01:
                    a = a / s2

        it += 1
        if verbose and it % 200 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, alpha={alpha:.1f}, lr={lr:.4f}, t={elapsed:.1f}s")

    # Final evaluation
    final_obj = compute_obj(best_a)
    if final_obj < best_obj:
        best_obj = final_obj
    return best_a


def lp_polish(a, time_budget=60.0, verbose=False, max_iters=50):
    """LP-based polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        T0 = time.monotonic()
        g_lp, M_lp = solve_lp_minpeak(a)
        lp_time = time.monotonic() - T0

        if g_lp is None:
            break

        t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=min(2.0, remaining * 0.1))

        if t_opt > 0 and obj_opt < best_obj - 1e-14:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, lp_time={lp_time:.2f}s")

        if stall >= 5:
            break

    a_try = np.interp(x_new, x_old, a_best)

    s = float(np.sum(a_try))

        print(f"\n=== Phase 4: n Exploration (budget={remaining:.0f}s) ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 100, n_best + 100, n_best + 200, n_best + 400,
            600, 700, 900, 1100, 1300, 1500
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) >= 50]
        n_options = n_options[:4]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            t_alloc = min(remaining * 0.6, 60.0)
            a_try, obj_try = try_length(best_seq, n_try, t_alloc, rng)
            if a_try is not None and obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 5: Diverse Restarts ===")
        n_cur = len(best_seq)
        n_restarts = min(4, int(remaining // 30))
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            mode = (r * 3) % 8
            a_r = diverse_init(n_cur, rng, mode=mode)
            blend = 0.4
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 0.01:
                a_r = a_r / s_r

            t_alloc = min(remaining * 0.3, 40.0)
            a_r = mirror_descent_optimize(a_r, time_budget=t_alloc,
                                           alpha_start=5.0, alpha_end=80.0,
                                           lr_init=0.2, max_iters=3000,
                                           verbose=False, seed=seed + 100 + r)
            # Quick LP
            t_lp = min(remaining * 0.1, 15.0)
            if t_lp > 5:
                a_r = lp_polish(a_r, time_budget=t_lp, max_iters=10)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Final Polish ===")
        # MD with high alpha
        t_md = min(remaining * 0.5, 60.0)
        if t_md > 10:
            a_final = mirror_descent_optimize(best_seq, time_budget=t_md,
                                               alpha_start=50.0, alpha_end=200.0,
                                               lr_init=0.05, max_iters=5000,
                                               verbose=False, seed=seed + 999)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final MD: obj={best_score:.8f} **")

        # LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t_lp = min(remaining * 0.6, 40.0)
            if t_lp > 5:
                a_lp_final = lp_polish(best_seq, time_budget=t_lp, verbose=True, max_iters=20)
                obj_lp_final = compute_obj(a_lp_final)
                if obj_lp_final < best_score:
                    best_score = obj_lp_final
                    best_seq = a_lp_final.copy()
                    print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
