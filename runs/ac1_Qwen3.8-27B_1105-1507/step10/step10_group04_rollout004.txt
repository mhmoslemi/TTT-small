def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_lp_fixed_point(a, max_iters=50, tol=1e-10, verbose=False):
    """
    LP fixed-point iteration: a <- g*(a) where g* solves the LP.
    Full step (no mixing). Returns (a_converged, obj, n_iters, converged).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0, False
    a = a / s

    best_obj = compute_obj(a)
    best_a = a.copy()

    # Build matrix once (it depends on a, so rebuild each iteration)
    for it in range(max_iters):
        a_norm = a / np.sum(a)
        A = build_conv_matrix(a_norm)

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        g_val = None
        M_val = None
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:

                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                        M_val = float(M.value)
                        break
            except Exception:
                continue

        if g_val is None:
            break

        # Full step: a = g_val
        old_a = a.copy()
        a = g_val
        obj = compute_obj(a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

            if verbose:
                print(f"      LP converged at iter {it}, obj={obj:.10f}")
            return best_a, best_obj, it + 1, True

    return best_a, best_obj, max_iters, False

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.7

        # Nesterov momentum
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Smoothed gradient via log-sum-exp
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        shifted = tau * (conv - max_conv)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        P = fft(p, N)
        A_f = fft(y, N)
        g_full = ifft(P * np.conj(A_f)).real[:n]
        g = 2.0 * g_full

        # Gradient of the smoothed objective
        obj_smooth = max_conv + np.log(sum_exp) / tau
        # The gradient w.r.t. y is: 2 * (p convolved with y) ... actually
        # d/dy_j [log-sum-exp(tau*(conv(y,y)-max))] = 2 * sum_k p_k * (y conv a)_k's j component
        # = 2 * sum_k p_k * y_{k-j} (with appropriate bounds)
        # Which is what g_full gives us (up to the factor of 2 which we already have)

        # But we need to subtract the mean to stay on the simplex
        g = g - np.sum(g * y)  # tangential projection
        # Actually, for simplex constraint, we project: g <- g - (g @ y) * y... no
        # The constraint is sum(y)=1, so gradient in tangent space: g_tan = g - mean(g)
        # But since we're on the probability simplex, we use: g_tan = g - sum(g*y)
        # Wait, let's just use: g_tan = g - np.mean(g) doesn't preserve the simplex
        # Better: just clip and renormalize

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Adaptive step size
        lr = min(1.0 / (g_norm + 1e-10), 2.0 / np.sqrt(max(n, 1)))
        lr = max(lr, 1e-9)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1

        a = a_new
        it += 1
        beta = min(0.95, beta + 0.0002)

        if stall > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            stall = 0
            if it - best_it > 15000:
                break

        if verbose and it % 3000 == 0:
            print(f"      GD it{it}: obj={best_obj:.10f}, tau={tau:.0f}, t={elapsed:.0f}s")

    return best_a


def kkt_perturbation(a, strength=0.1, rng=None):
    """
    Perturb a based on KKT analysis.
    Positions where the dual constraint is "slack" (g_j > 0 but not at max conv)
    are good candidates for perturbation.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    # Identify "active" positions (where conv is close to max)
    active = conv > max_conv - 1e-6
    active_center = np.where(active)[0]

    # For positions j where a_j > 0, check if they contribute to active positions
    # If a position j only contributes to non-active positions, it's a good candidate
    # for perturbation (we can shift mass away from it)
    contribution = np.zeros(n)
    for k in range(len(conv)):
        if active[k]:
            for j in range(max(0, k - n + 1), min(k + 1, n)):
                contribution[j] += a[k - j]

    # Positions with low contribution to active set are "free"
    free_score = 1.0 - contribution / (contribution.max() + 1e-10)

    # Perturb: move mass from "free" positions to "active-contributing" positions
    a_pert = a.copy()
    n_pert = max(1, int(n * strength))
    free_indices = np.where(free_score > 0.5)[0]
    active_indices = np.where(contribution > contribution.max() * 0.5)[0]

    if len(free_indices) > 0 and len(active_indices) > 0:
        # Sample positions to perturb
        n_f = min(n_pert, len(free_indices))
        n_a = min(n_pert, len(active_indices))
        f_idx = rng.choice(free_indices, size=n_f, replace=False)
        a_idx = rng.choice(active_indices, size=n_a, replace=False)

        # Move small amount of mass
        eps = strength * 0.01 * np.min(a[f_idx])
        for fi in f_idx:
            for ai in a_idx:
                if ai != fi:
                    a_pert[fi] -= eps / len(a_idx)
                    a_pert[ai] += eps / len(a_idx)

    # Also add small random noise
    noise = rng.normal(0, 0.001, n)
    a_pert = a_pert + noise

    a_pert = np.maximum(a_pert, 0)
    s_pert = float(np.sum(a_pert))
    if s_pert > 1e-10:
        a_pert = a_pert / s_pert
    else:
        a_pert = a.copy()

    return a_pert


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 2
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-((np.arange(n) - n / 2) ** 2) / (2 * (n / 4) ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Alternating pattern with decay
        a = np.ones(n) * (0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Two peaks
        centers = [n * 0.25, n * 0.75]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * (n / 10) ** 2))
        a = a / a.sum()
    else:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    return a


def optimize_for_n(a_init, budget, verbose=False):
    """
    Full optimization pipeline for a given length:
    LP fixed-point -> GD refinement -> perturbation loop.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(hash(n) % 10000)

    # Phase A: LP fixed-point (full step)
    lp_budget = min(budget * 0.4, 120.0)
    remaining = budget - (time.monotonic() - start)
    if remaining > 20:
        a_lp, obj_lp, n_iters, converged = solve_lp_fixed_point(
            best_a, max_iters=60, tol=1e-10, verbose=verbose)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            if verbose:
                print(f"      LP: obj={best_obj:.10f}, iters={n_iters}, converged={converged}")

    # Phase B: GD refinement
    remaining = budget - (time.monotonic() - start)
    if remaining > 20:
        gd_budget = min(remaining * 0.3, 60.0)
        a_gd = smoothed_gd_refine(best_a, time_budget=gd_budget,
                                  tau_start=200, tau_end=6000, verbose=verbose)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_obj:
            best_obj = obj_gd
            best_a = a_gd.copy()
            if verbose:
                print(f"      GD: obj={best_obj:.10f}")

    # Phase C: LP again after GD
    remaining = budget - (time.monotonic() - start)
    if remaining > 20:
        a_lp2, obj_lp2, _, _ = solve_lp_fixed_point(
            best_a, max_iters=30, tol=1e-11, verbose=verbose)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()
            if verbose:
                print(f"      LP2: obj={best_obj:.10f}")

    # Phase D: Perturbation loop
    remaining = budget - (time.monotonic() - start)
    if remaining > 30:
        n_perts = max(1, int(remaining / 30))
        for p in range(n_perts):
            remaining = budget - (time.monotonic() - start)
            if remaining < 30:
                break
            a_pert = kkt_perturbation(best_a, strength=0.05 + 0.02 * p, rng=rng)
            # Quick LP + GD on perturbed
            alloc = min(25, remaining - 10)
            if alloc > 10:
                a_try, obj_try, _, _ = solve_lp_fixed_point(a_pert, max_iters=15, tol=1e-9)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    if verbose:
                        print(f"      Pert {p}: obj={best_obj:.10f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP fixed-point iteration, multi-n exploration,
    KKT-guided perturbation, and high-precision GD refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0  # safety margin

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent_arr = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = diverse_init(n_init, rng, mode=2)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.10f}")

    n_best = len(best_seq)

    # === Phase 1: Intensive LP polishing on parent n ===
    print(f"\n=== Phase 1: LP on n={n_best} ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a_p1 = optimize_for_n(best_seq, budget=p1_budget, verbose=True)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase1 complete: obj={best_score:.10f} **")
        else:
            print(f"  Phase1: obj={obj_p1:.10f} (no improvement)")

    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.45, 500.0)

    n_best = len(best_seq)
    n_candidates = sorted(set([
        400, 500, 600, 700, 750, 800, 850, 900, 950, 1000,
        1100, 1200, 1400, 1600, 2000, 2500,
        n_best - 100, n_best + 100, n_best + 200, n_best + 400,
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 3000 and n != n_best]

    # Sort by proximity to n_best (try nearby first)
    n_candidates.sort(key=lambda x: abs(x - n_best))
    n_candidates = n_candidates[:10]  # Limit to 10 values
    print(f"  n candidates: {n_candidates}")

    per_n_budget = p2_budget / max(1, len(n_candidates))
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(per_n_budget * 1.5, remaining - 30)
        if alloc < 25:
            break

        # Resample from best
        a_try = resample_sequence(best_seq, n_try)

        if verbose_alloc := (alloc > 30):
            print(f"  Trying n={n_try} (alloc={alloc:.0f}s) ...", end=" ")

        a_opt = optimize_for_n(a_try, budget=alloc, verbose=False)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f" ** obj={best_score:.10f} **")
        else:
            print(f"obj={obj_try:.10f}")

    # === Phase 3: Diverse restarts at best n ===
    print(f"\n=== Phase 3: Diverse Restarts (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    n_restarts = max(2, int(p3_budget / 35))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break
        alloc = min(35, remaining - 10)
        if alloc < 20:
            break

        mode = r % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with best
        blend = 0.4 + 0.1 * (r % 3)
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        a_r = optimize_for_n(a_r, budget=alloc, verbose=False)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.10f}")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s")

    # 4a: LP with tighter tolerance
    if remaining > 40:
        alloc = min(remaining * 0.3, 120.0)
        a_final, obj_final, n_it, conv_flag = solve_lp_fixed_point(
            best_seq, max_iters=80, tol=1e-12, verbose=True)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    # 4b: GD with very high tau
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 100.0)
        a_final = smoothed_gd_refine(best_seq, time_budget=alloc,
                                     tau_start=500, tau_end=15000, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.10f} **")

    # 4c: Another LP round
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.25, 60.0)
        a_final, obj_final, _, _ = solve_lp_fixed_point(
            best_seq, max_iters=40, tol=1e-12)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra LP: obj={best_score:.10f} **")

    # 4d: Final GD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 40.0)
        a_final = smoothed_gd_refine(best_seq, time_budget=alloc,
                                     tau_start=1000, tau_end=20000)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra GD: obj={best_score:.10f} **")

    # 4e: One more LP for absolute final
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_final, obj_final, _, _ = solve_lp_fixed_point(
            best_seq, max_iters=20, tol=1e-13)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Final LP: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
