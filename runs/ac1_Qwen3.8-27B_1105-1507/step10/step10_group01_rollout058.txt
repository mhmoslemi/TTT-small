def smoothed_gradient_simple(a, tau):
    """
    Simpler gradient computation using direct formula for small n,
    FFT for large n.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    tau_conv = tau * (conv - max_conv)
    tau_conv = np.clip(tau_conv, -500, 500)
    exp_vals = np.exp(tau_conv)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # Use FFT-based gradient
    N = 1
    while N < 2 * n + 1:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    grad = ifft(P * A).real[:n] + ifft(P * np.conj(A)).real[:n]
    
    return grad


def gd_optimize(a, time_budget=100.0, tau_start=10.0, tau_end=5000.0, verbose=True):
    """
    Nesterov accelerated gradient descent on smoothed max autocorrelation.
    Uses adaptive tau, adaptive learning rate, and periodic restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9  # momentum
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr

        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        
        # Adaptive tau: exponential ramp
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.7)
        tau = min(tau, 10000.0)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:

        # Use backtracking: try the step, if objective worsens, reduce lr

        s_try = float(np.sum(a_try))

            obj_try = compute_obj(a_try)
