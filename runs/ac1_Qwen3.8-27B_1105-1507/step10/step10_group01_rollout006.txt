    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s

    conv_curr = compute_conv(a_norm)
    top_indices = np.argsort(conv_curr)[::-1][:top_k]

    A = build_active_matrix(a_norm, top_indices)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.MOSEK]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    return g_val / s_g, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_coarse(a, g, n_evals=20):
    """Coarse grid line search for step size t."""
    n = len(a)
    ts = np.linspace(0.005, 0.99, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try)
    best_idx = int(np.argmin(objs))
    return ts[best_idx], objs[best_idx]


def lp_optimize(a, time_budget=300.0, verbose=False):
    """
    Active-set LP optimization loop with adaptive constraint count.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall = 0
    consec_no_improve = 0
    iter_count = 0
    cur_top_k = 80
    max_top_k = min(n // 2, 400)
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break

        g, M = active_set_lp(a, top_k=cur_top_k)
        if g is None:
            break

        best_t, obj_t = line_search_coarse(a, g, n_evals=18)
        iter_count += 1

        if obj_t < best_obj - 1e-13:
            a_new = (1 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                best_obj = obj_t
                best_a = a_new.copy()
                a = a_new
                stall = 0
                consec_no_improve = 0
                # Gradually increase constraints
                if stall < 2:
                    cur_top_k = min(max_top_k, cur_top_k + 10)
                if verbose and (iter_count % 15 == 1):
                    print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, K={cur_top_k}, t={time.monotonic()-start:.0f}s")
        else:
            consec_no_improve += 1
            stall += 1

            # Try smaller steps
            if consec_no_improve <= 3:
                for t_s in [0.05, 0.02, 0.01]:
                    a_s = (1 - t_s) * a + t_s * g
                    a_s = np.maximum(a_s, 0)
                    s_s = float(np.sum(a_s))
                    if s_s > 1e-10:
                        a_s = a_s / s_s
                        obj_s = compute_obj(a_s)
                        if obj_s < best_obj - 1e-13:
                            best_obj = obj_s
                            best_a = a_s.copy()
                            a = a_s
                            stall = 0
                            consec_no_improve = 0
                            break

            if stall >= 4:
                # Perturb: use smoothed subgradient
                tau = 300.0
                conv_p = compute_conv(a)
                max_p = float(np.max(conv_p))
                exp_p = np.exp(tau * (conv_p - max_p))
                p = exp_p / exp_p.sum()
                N_fft = 1
                while N_fft < 3 * n:
                    N_fft <<= 1
                P = fft(p, N_fft)
                Af = fft(a, N_fft)
                grad = 2.0 * ifft(P * np.conj(Af)).real[:n]
                gnorm = float(np.linalg.norm(grad))
                if gnorm > 1e-15:
                    lr_p = 0.015 / gnorm
                    a = a - lr_p * grad
                a = a + 0.003 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                else:
                    a = best_a.copy()
                stall = 0
                consec_no_improve = 0
                cur_top_k = max(50, cur_top_k - 20)

                if iter_count > 200:
                    if verbose:
                        print(f"    LP: giving up at it{iter_count}, obj={best_obj:.8f}")
                    break
            elif stall >= 2:
                cur_top_k = max(50, cur_top_k - 10)

    return best_a


def smooth_grad(a, tau):
    """Gradient of softmax-smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_v = np.exp(tau * (conv - max_c))
    sum_e = float(np.sum(exp_v))
    if sum_e < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_v / sum_e
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def gd_polish(a, time_budget=100.0, tau_start=100.0, tau_end=3000.0, verbose=False):
    """Nesterov-accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.5 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 1.3

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smooth_grad(y, tau)
        gnorm = float(np.linalg.norm(g))

        if gnorm > 1e-15:
            adapt = 1.5 / (gnorm + 1e-10)
            adapt = min(max(adapt, 0.3 * base_lr), 3.0 * base_lr)
            lr = 0.75 * lr + 0.25 * adapt

        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 4.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-12, lr * 0.9997)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

    if verbose:
        print(f"    GD: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_seq(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, kind, rng):
    """Generate structured initial sequences."""
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "bimodal":
        a = np.abs(x - mid) + 1.0
    elif kind == "unimodal":
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
    elif kind == "ramp":
        a = x + 1.0
    elif kind == "w_shaped":
        peaks = [0.25, 0.75]
        a = np.ones(n) * 0.3
        for p in peaks:
            sigma = n / 12.0
            a += np.exp(-0.5 * ((x - p * (n - 1)) / sigma) ** 2)
    elif kind == "comb":
        a = np.ones(n)
        a[::3] = 2.0
    else:
        a = rng.uniform(0.5, 1.5, n)
    a = np.maximum(a, 0.01)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def threshold_restart(a, rng, frac=0.7):
    """Combinatorial restart: keep top fraction, optimize."""
    n = len(a)
    k = int(n * frac)
    if k < n // 4:
        k = n // 4
    thresh = np.sort(a)[::-1][k - 1] * rng.uniform(0.5, 1.5)
    mask = a >= thresh
    a_new = a.copy()
    a_new[~mask] = 0.0
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def fuzzy_threshold(a, rng, frac=0.7, width=0.1):
    """Soft threshold: sigmoid-based exclusion."""
    n = len(a)
    k = int(n * frac)
    if k < 2 or k >= n:
        return a.copy()
    sorted_a = np.sort(a)[::-1]
    thresh = sorted_a[k - 1]
    spread = width * (sorted_a[0] - thresh + 1e-12)
    sigmoid = 1.0 / (1.0 + np.exp(-(a - thresh) / spread))
    a_new = a * sigmoid
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: active-set LP + GD polish + combinatorial restarts + multi-n.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # === Phase 0: Initialization ===
    print("=== Phase 0: Initialization ===")
    candidates = []

    # Try parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s_p = float(np.sum(parent_arr))
            if s_p > 0.01:
                parent = parent_arr / s_p
                obj_p = compute_obj(parent)
                candidates.append((obj_p, parent.copy(), "parent"))
                print(f"  Parent (n={parent.size}): obj={obj_p:.8f}")
    except Exception:
        pass

    # Try structured initializations at n=800
    n_try = 800
    if not candidates:
        n_try = 800

    for kind in ["uniform", "bimodal", "unimodal", "ramp", "w_shaped"]:
        a_init = structured_init(n_try, kind, rng)
        obj_init = compute_obj(a_init)
        candidates.append((obj_init, a_init.copy(), f"{kind}"))
        print(f"  Structured {kind}: obj={obj_init:.8f}")

    # Also try resampling parent to different lengths
    if candidates and candidates[0][2] == "parent":
        parent = candidates[0][1]
        for n_res in [900, 1000, 1100]:
            if len(parent) != n_res:
                a_res = resample_seq(parent, n_res)
                a_res = a_res / a_res.sum()
                obj_res = compute_obj(a_res)
                candidates.append((obj_res, a_res.copy(), f"resampled_{n_res}"))
                print(f"  Resampled n={n_res}: obj={obj_res:.8f}")

    candidates.sort(key=lambda x: x[0])
    best_obj_init = candidates[0][0]
    best_seq = candidates[0][1].copy()
    best_score = best_obj_init
    n_current = len(best_seq)
    print(f"  Best init: {candidates[0][2]}, obj={best_score:.8f}")

    # === Phase 1: LP Optimization (main) ===
    print(f"\n=== Phase 1: Active-Set LP (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 450.0)
    if p1_budget > 30:
        a_lp = lp_optimize(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_lp:.8f}")

    # === Phase 2: GD Polish ===
    print(f"\n=== Phase 2: GD Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 150.0)
    if p2_budget > 15:
        a_gd = gd_polish(best_seq, time_budget=p2_budget, tau_start=80.0, tau_end=4000.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj_gd:.8f}")

    # === Phase 3: Combinatorial Restarts ===
    print(f"\n=== Phase 3: Combinatorial Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    restarts = [
        ("fuzzy_70", lambda: fuzzy_threshold(best_seq, rng, frac=0.7, width=0.15)),
        ("fuzzy_50", lambda: fuzzy_threshold(best_seq, rng, frac=0.5, width=0.1)),
        ("hard_60", lambda: threshold_restart(best_seq, rng, frac=0.6)),
        ("mirror_mix", None),
        ("foURIer_perturb", None),
    ]

    for name, func in restarts:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        if name == "mirror_mix":
            a_r = 0.6 * best_seq + 0.4 * best_seq[::-1]
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        elif name == "foURIer_perturb":
            n_r = len(best_seq)
            pert = np.zeros(n_r)
            for k_idx in range(1, 7):
                freq = k_idx * 2.0 * np.pi / n_r
                amp = 0.04 * rng.uniform(0.5, 1.5) / k_idx
                phase = rng.uniform(0, 2.0 * np.pi)
                pert += amp * np.cos(freq * np.arange(n_r) + phase)
            a_r = best_seq * np.exp(pert)
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        else:
            a_r = func()

        # LP optimize
        alloc_lp = min(remaining * 0.25, 50.0)
        if alloc_lp > 12:
            a_r = lp_optimize(a_r, time_budget=alloc_lp, verbose=False)

        # GD polish
        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.08, 25.0)
        if alloc_gd > 8:
            a_r = gd_polish(a_r, time_budget=alloc_gd, tau_start=200.0, tau_end=2000.0)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 200), max(400, n_best - 100),
        n_best + 100, n_best + 200, n_best + 400,
        600, 750, 900, 1000, 1100, 1200, 1400, 1500, 1600, 1800
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) > 50]
    n_options = n_options[:5]
    print(f"  Trying n: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_seq(best_seq, n_try)
        a_try = a_try / a_try.sum()

        alloc_lp = min(remaining * 0.3, 55.0)
        if alloc_lp > 12:
            a_try = lp_optimize(a_try, time_budget=alloc_lp, verbose=False)

        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.1, 25.0)
        if alloc_gd > 8:
            a_try = gd_polish(a_try, time_budget=alloc_gd, tau_start=100.0, tau_end=2000.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Final Intensive Polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP round
    if remaining > 30:
        alloc = min(remaining * 0.4, 250.0)
        a_final = lp_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # High-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 120.0)
        a_final2 = gd_polish(best_seq, time_budget=alloc, tau_start=500.0, tau_end=10000.0, verbose=True)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # One more short LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.6, 100.0)
        a_final3 = lp_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_final3 = compute_obj(a_final3)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
