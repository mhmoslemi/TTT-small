    """FISTA optimization on the simplex with log-sum-exp smoothing.
    Anneals temperature from tau_initial to tau_final."""
    n = len(a)

    best_obj = compute_obj(a)

    y = a.copy()

            break

        progress = k / n_steps
        tau = tau_initial * (tau_final / tau_initial) ** progress
        
        # Compute gradient at y

Hybrid FISTA + LP optimization for minimizing step function autocorrelation peak.
Uses log-sum-exp smoothing for fast gradient-based global optimization,
LP polishing for local refinement, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    rho = cssv[int(i) - 1] / i
    return np.maximum(v - rho, 0.0)

    # C[i] = R[-i mod size] = R[size-i] for i>0, R[0] for i=0

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):

        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, M_bound=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g) = 1, g >= 0.
    Optionally with an upper bound on M.
    Returns (g, M) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:

                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

"""
Hybrid FISTA + LP optimization for minimizing step function autocorrelation peak.
Combines log-sum-exp smoothed gradient descent (FISTA/Nesterov accelerated),
LP-based direction finding with inexact solves, criss-cross line search,
spectral perturbations for escape, and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = int(np.max(ind[cond]))
    rho = cssv[i - 1] / i
    return np.maximum(v - rho, 0.0)


def compute_smoothed_gradient(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Returns (gradient, conv_array, smoothed_value).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    exp_vals = np.exp((conv - max_conv) / tau)
    p = exp_vals / np.sum(exp_vals)
    logsumexp = max_conv + tau * np.log(np.sum(exp_vals))

    # grad[i] = 2 * sum_j a[j] * p[j+i]
    # Using FFT cross-correlation: C[i] = R[size-i] where R = ifft(FFT(a)*conj(FFT(p)))
    size = 1
    while size < 3 * n:
        size <<= 1
    fa = rfft(a, size)
    fp = rfft(p, size)
    R = irfft(fa * np.conj(fp), size)
    g = np.zeros(n)
    g[0] = R[0]
    if n > 1:
        g[1:] = R[size - 1 : size - n : -1]
    return 2.0 * g, conv, logsumexp


def compute_subgradient(a):
    """Compute subgradient of max(conv(a,a)) at a. Returns (gradient, k_max, conv)."""
    n = len(a)
    conv = compute_conv(a)
    k_max = int(np.argmax(conv))
    grad = np.zeros(n)
    for i in range(n):
        j1 = k_max - i
        j2 = i - k_max
        if 0 <= j1 < n:
            grad[i] += a[j1]
        if 0 <= j2 < n:
            grad[i] += a[j2]
    return grad, k_max, conv


def fista_optimize(a_init, tau_list, time_budget, verbose=False):
    """
    FISTA with log-sum-exp smoothing and adaptive temperature.
    tau_list: list of tau values to anneal through.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_list:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # FISTA inner loop for this tau
        a_curr = a.copy()
        a_prev = a_curr.copy()
        y = a_curr.copy()
        t_mom = 1.0
        eta = 1.0 / max(1.0, 4.0 / tau)
        tau_time = min(remaining * 0.9, time_budget / (len(tau_list) + 1))

        it = 0
        best_obj_tau = compute_obj(a_curr)
        while time.monotonic() - (start + (time_budget - remaining * 0.9)) < tau_time + 1:
            if time.monotonic() - start >= time_budget - 2:
                break

            grad, conv, _ = compute_smoothed_gradient(y, tau)

            # Gradient step
            z = y - eta * grad
            a_new = project_simplex(z)

            # Nesterov momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            y_new = a_new + ((t_mom - 1.0) / t_new) * (a_new - a_curr)
            y_new = project_simplex(y_new)

            a_prev = a_curr
            a_curr = a_new
            y = y_new
            t_mom = t_new
            it += 1

            # Periodic exact objective check
            if it % 5 == 0:
                obj = compute_obj(a_curr)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_curr.copy()
                if obj < best_obj_tau - 1e-10:
                    best_obj_tau = obj
                    # Reset momentum if we made big progress
                    if it % 50 == 0:
                        y = a_curr.copy()
                        a_prev = a_curr.copy()
                        t_mom = 1.0

            # Adaptive step: shrink if worsening
            if it % 20 == 0 and it > 0:
                obj_check = compute_obj(a_curr)
                if obj_check > best_obj * 1.02:
                    eta *= 0.6
                    y = a_curr.copy()
                    a_prev = a_curr.copy()
                    t_mom = 1.0
                elif obj_check < best_obj * 0.99 and eta < 4.0 / tau:
                    eta *= 1.3

        a = a_curr
        if verbose:
            print(f"    FISTA tau={tau:.6f}: best_obj={best_obj:.8f}, iters={it}")

    return best_a


def build_conv_matrix_sparse(a):
    """Build the convolution matrix but return it in a form suitable for LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, inexact=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_sparse(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            if inexact:
                prob.solve(solver=solver, relative_gap=1e-4, absolute_gap=1e-8)
            else:
                prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def crisscross_line_search(a, direction, obj_curr, max_tries=15):
    """
    Criss-cross method to find optimal step size t in (1-t)*a + t*direction.
    Returns (best_t, best_obj).
    """
    def obj_at(t):
        if t <= 0:
            return obj_curr
        a_try = (1 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    # Establish bracket
    t_lo, t_hi = 0.0, 1.0
    f_lo, f_hi = obj_curr, obj_at(1.0)

    if f_hi >= f_lo:
        # Function increased at t=1, bisect to find where it starts decreasing
        t_mid = 0.5
        f_mid = obj_at(t_mid)
        if f_mid < f_lo:
            t_change, f_change = t_mid, f_mid
            t_hi = 1.0
            f_hi = f_change
            t_lo = 0.0
            f_lo = f_change
        else:
            # Try smaller steps
            for t_try in [0.1, 0.05, 0.01]:
                f_try = obj_at(t_try)
                if f_try < f_lo:
                    t_lo = 0.0
                    f_lo = obj_curr
                    t_hi = t_try
                    f_hi = f_try
                    t_change, f_change = t_try, f_try
                    break
            else:
                return 0.0, obj_curr

    # Criss-cross iterations
    best_t, best_f = t_hi, f_hi
    for _ in range(max_tries):
        f_mid_est = f_lo + (f_hi - f_lo) * t_lo / (t_hi - t_lo)
        f_mid_est2 = f_lo + (f_hi - f_lo) * t_hi / (t_hi - t_lo)
        t_mid = (t_lo + t_hi) / 2.0
        f_mid = obj_at(t_mid)
        if f_mid < best_f:
            best_f = f_mid
            best_t = t_mid
        if f_mid < f_mid_est:
            t_hi = t_mid
            f_hi = f_mid
        else:
            t_lo = t_mid
            f_lo = f_mid
        if t_hi - t_lo < 1e-5:
            break
    return best_t, best_f

        obj_lp_dir = compute_obj(g_lp)
        if obj_lp_dir >= best_obj:
            stall += 1

            if obj_opt < best_obj - 1e-12:

                a_new = np.maximum(a_new, 0)

                # Major restart: blend with diverse init
                x = np.linspace(0, 1, n)
                a_div = 1.0 / (0.05 + 3.0 * (2 * x - 1) ** 2)
                a = 0.5 * a_div + 0.5 * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()
            stall = 0

    return best_a


def subgradient_warmup(a, time_budget=20.0):
    """Quick subgradient descent with momentum for fast initial improvement."""
    n = len(a)
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.05
    momentum = np.zeros(n)
    beta = 0.7
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        grad, k_max, conv = compute_subgradient(a)
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s > 1e-10:
            a_new = a_new / s
        else:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        a = a_new
        lr *= 0.998

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            best_it = it

        it += 1
        if it - best_it > 3000:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = 0.02
            best_it = it

    return best_a


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    s = float(np.sum(a))
    return a / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = 1.0 / (0.05 + 3.0 * (2 * x - 1) ** 2)
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 8) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 8) ** 2)) + 0.01
    elif mode == 2:
        a = 4.0 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5.0 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / np.sqrt(0.1 + (2 * x - 1) ** 4)
    elif mode == 6:
        a = 1.0 / (0.1 + (2 * x - 1) ** 1.5)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def try_length(a_best, n_try, time_budget, rng):
    """Try a different sequence length by interpolating and optimizing."""
    n_old = len(a_best)
    if n_try == n_old:
        return a_best.copy(), compute_obj(a_best)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        a_try = u_shaped_init(n_try)
    else:
        a_try = a_try / s

    # Quick warmup
    if time_budget > 20:
        a_try = subgradient_warmup(a_try, time_budget=min(10.0, time_budget * 0.2))

    # LP polish
    remaining = time_budget - 10.0
    if remaining > 10:
        a_try = lp_polish(a_try, time_budget=remaining, inexact=True)

    obj_try = compute_obj(a_try)
    return a_try, obj_try


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: FISTA + LP hybrid optimization with spectral perturbations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Load parent or create default ---
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default U-init (n={n_current}): obj={best_score:.8f}")

    # Try a few alternative inits at same n
    for mode in [4, 5, 6]:
        a_alt = diverse_init(n_current, rng, mode)
        obj_alt = compute_obj(a_alt)
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"Alt init mode {mode} (n={n_current}): obj={best_score:.8f} **")

    # --- Phase 1: Subgradient warmup ---
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 1: Subgradient warmup ===")
        a_sg = subgradient_warmup(best_seq, time_budget=min(30.0, remaining * 0.08))
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_sg:.8f} (no improvement)")

    # --- Phase 2: FISTA with annealed temperature ---
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 2: FISTA optimization ===")
        # Multi-scale tau annealing
        # Peak of conv is around 1/n ~ 0.001 for n=800
        # tau should be comparable to or smaller than the peak for good resolution
        tau_schedules = [
            [0.01, 0.005, 0.003, 0.002, 0.001],
            [0.002, 0.001, 0.0005, 0.0003],
        ]
        alloc = min(remaining * 0.3, 300.0)
        chunk = alloc / len(tau_schedules)
        for sched in tau_schedules:
            remaining_sched = deadline - time.monotonic()
            if remaining_sched < 10:
                break
            a_fista = fista_optimize(best_seq, sched, time_budget=min(chunk, remaining_sched * 0.8), verbose=False)
            obj_f = compute_obj(a_fista)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_fista.copy()
                print(f"  FISTA: obj={best_score:.8f} **")

    # --- Phase 3: Intensive LP polishing ---
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 3: LP Polishing (intensive) ===")
        alloc = min(remaining * 0.3, 350.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True, inexact=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase3 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 LP: obj={obj_lp:.8f} (no improvement)")

    # --- Phase 4: FISTA again on improved point ---
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 4: FISTA refinement on LP result ===")
        tau_refine = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        alloc = min(remaining * 0.15, 120.0)
        a_f2 = fista_optimize(best_seq, tau_refine, time_budget=alloc, verbose=False)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Phase4: obj={best_score:.8f} **")

        # More LP
        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc2 = min(remaining * 0.2, 200.0)
            a_lp2 = lp_polish(best_seq, time_budget=alloc2, verbose=True, inexact=True)
            obj_lp2 = compute_obj(a_lp2)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  Phase4 LP: obj={best_score:.8f} **")

    # --- Phase 5: n exploration ---
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 5: n Exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 100, n_best - 50, n_best + 50, n_best + 100, n_best + 200,
            n_best + 400, n_best - 200,
            500, 600, 700, 900, 1000, 1200, 1400, 1600, 1800
        ]))
        n_options = [n for n in n_options if 300 <= n <= 2500 and abs(n - n_best) >= 40]
        n_options = n_options[:6]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            a_try, obj_try = try_length(best_seq, n_try, remaining * 0.6, rng)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # --- Phase 6: Diverse restarts ---
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print("\n=== Phase 6: Diverse Restarts ===")
        n_cur = len(best_seq)
        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            mode = r % 7
            a_r = diverse_init(n_cur, rng, mode)
            a_r = 0.4 * a_r + 0.6 * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            if remaining > 30:
                a_r = subgradient_warmup(a_r, time_budget=min(10.0, remaining * 0.15))
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_r = lp_polish(a_r, time_budget=min(remaining * 0.5, 60.0), inexact=True)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # --- Phase 7: Final intensive polish ---
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 7: Final Polish ===")
        # FISTA with very fine tau
        if remaining > 40:
            tau_final = [0.0003, 0.0001, 0.00005, 0.00002]
            alloc = min(remaining * 0.3, 80.0)
            a_fin = fista_optimize(best_seq, tau_final, time_budget=alloc)
            obj_fin = compute_obj(a_fin)
            if obj_fin < best_score:
                best_score = obj_fin
                best_seq = a_fin.copy()
                print(f"  Final FISTA: obj={best_score:.8f} **")

        # Final LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining - 5, 150.0)
            a_fin2 = lp_polish(best_seq, time_budget=alloc, verbose=True, inexact=False)
            obj_fin2 = compute_obj(a_fin2)
            if obj_fin2 < best_score:
                best_score = obj_fin2
                best_seq = a_fin2.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
