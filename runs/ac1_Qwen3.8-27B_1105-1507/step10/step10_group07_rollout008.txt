def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

def lp_direction(a, M_target=None, solver=None):
    """Solve LP: min M s.t. A@a_new <= M, sum(a_new)=1, a_new>=0"""
    n = len(a)

        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    solvers = [cp.CLP, cp.GLPK, cp.GUROBI, cp.MOSEK, cp.CBC, cp.SCS]
    if solver is not None:

    # Remove duplicates while preserving order

            unique_solvers.append(s)

                sv = float(np.sum(x_val))

def tr_sscp(a, r, solver_list=None):

    a = a / a.sum()

    A_cross = build_conv_matrix(a)

        a_new >= 0,

        cp.norm(delta, 2) <= r,
    ]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    if solver_list is None:

import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    return np.maximum(np.interp(x_new, x_old, a), 0)

        A_f = fft(a, N)
        g_full = ifft(P * np.conj(A_f)).real
        g = 2.0 * g_full[:n]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr = max(1e-12, lr * 0.9992)

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimizer combining smoothed gradient warmup, LP/TR-SOCP polishing,
    broad multi-length exploration, perturbation restarts, and final tight polish.
    Designed to push below 1.5027 by exploiting better solver accuracy,
    broader search, and adaptive line search.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 12.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Short smoothed GD warmup ===
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.05, 50.0)
    if p1_budget > 8:
        print(f"\n=== Phase 1: Smoothed GD (budget={p1_budget:.0f}s) ===")
        a_sgd = smoothed_gd(best_seq, time_budget=p1_budget, tau_init=10.0, tau_end=200.0)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_sgd:.8f}")

    # === Phase 2: Main LP/TR polishing ===
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.32, 320.0)
    if p2_budget > 25:
        print(f"\n=== Phase 2: LP/TR Polish (budget={p2_budget:.0f}s) ===")
        a_tr = polish(best_seq, time_budget=p2_budget, seed=42, verbose=True)
        obj_tr = compute_obj(a_tr)
        if obj_tr < best_score:
            best_score = obj_tr
            best_seq = a_tr.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj_tr:.8f}")

    # === Phase 3: Broad multi-length exploration ===
    print(f"\n=== Phase 3: Multi-Length Exploration ===")
    remaining = deadline - time.monotonic()

    # Generate a broad set of candidate lengths
    n_candidates = set()
    for delta in range(-500, 600, 50):
        n_try = n_current + delta
        if 400 <= n_try <= 2000:
            n_candidates.add(n_try)
    # Add natural lengths
    for n_nat in [450, 500, 550, 600, 650, 700, 750, 800, 850, 900,
                  950, 1000, 1100, 1200, 1300, 1400, 1500, 1600, 1800, 2000]:
        n_candidates.add(n_nat)

    # Sort by distance from current (closest first) but exclude current
    n_sorted = sorted([n for n in n_candidates if n != n_current],
                      key=lambda x: abs(x - n_current))

    for n_try in n_sorted[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)

        alloc1 = min(remaining * 0.08, 15.0)
        if alloc1 > 5:
            a_try = smoothed_gd(a_try, time_budget=alloc1, tau_init=8.0, tau_end=150.0)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.1, 50.0)
        if alloc2 > 8:
            a_try = polish(a_try, time_budget=alloc2, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Perturbation restarts ===
    print(f"\n=== Phase 4: Perturbation Restarts ===")
    remaining = deadline - time.monotonic()

    for r_idx in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        scale = 0.005 + 0.005 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = float(a_pert.sum())
        if s_pert < 1e-10:
            break
        a_pert = a_pert / s_pert

        alloc = min(remaining * 0.18, 55.0)
        if alloc > 8:
            a_pert = polish(a_pert, time_budget=alloc, seed=600 + r_idx * 17)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 5: Final tight polish ===
    print(f"\n=== Phase 5: Final Tight Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.55, 120.0)
        a_final = polish(best_seq, time_budget=alloc, seed=9999, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final: obj={best_score:.8f} **")
        else:
            print(f"  Final: obj={obj_final:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
