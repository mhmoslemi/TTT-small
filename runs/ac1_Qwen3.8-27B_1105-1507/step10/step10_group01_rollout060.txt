import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_fast(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_fast(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g, n_evals=30):
    """
    Line search: find best t in [0,1] to minimize obj((1-t)*a + t*g).
    """
    n = len(a)
    ts = np.linspace(0.005, 0.995, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try)

    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]

    if 0 < best_idx < n_evals - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.005, 0.995))
            a_par = (1 - t_par) * a + t_par * g
            a_par = np.maximum(a_par, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                f_par = compute_obj(a_par)
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par

    return best_t, best_obj

def spectral_perturb(a, rng, strength=0.03):
    """Add spectral-domain perturbation to escape local minima."""
    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1
    A = fft(a, N)
    # Perturb in log-amplitude domain (preserves non-negativity better)
    log_amp = np.log(np.abs(A) + 1e-10)
    # Low-frequency perturbation
    for k in range(1, 4):
        amp = strength * rng.uniform(0.5, 1.5) / k
        log_amp[k] += amp * rng.standard_normal()
        log_amp[N - k] += amp * rng.standard_normal()
    new_amp = np.exp(log_amp)
    # Preserve phase
    phase = np.angle(A)
    A_new = new_amp * np.exp(1j * phase)
    a_new = irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def peak_shave(a, rng, n_shaves=5):
    """Directly reduce peak contributions by modifying sequence at peak-inducing positions."""
    n = len(a)
    a = a.copy()
    for _ in range(n_shaves):
        conv = compute_conv(a)
        max_idx = int(np.argmax(conv))
        max_val = float(conv[max_idx])
        
        # The peak at lag k is sum_i a[i]*a[i+k]
        # To reduce it, we can decrease a[i] where the product a[i]*a[i+k] is large
        if 0 <= max_idx < n:
            products = a[:n-max_idx] * a[max_idx:n]
        else:
            k = max_idx - (n - 1)
            products = a[k:] * a[:n-k]
        
        # Find the top contributors
        top_k = max(1, n // 20)
        top_idx = np.argsort(products)[-top_k:]
        
        # Reduce them slightly
        reduction = 0.01 * max(0.0, max_val - float(np.mean(conv)))
        for idx in top_idx:
            a[idx] *= (1.0 - reduction * 0.1)
        
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s
    
    return a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-driven optimization as the main workhorse, combined with Fourier-domain
perturbation for escaping local minima, adaptive multi-n exploration, and
Nesterov-accelerated smoothed gradient descent for final polishing.
Key improvements: targeted spectral perturbation, soaker-style LP/GD alternation,
and systematic n scanning with resampling.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_fast(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_fast(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g, n_evals=35):
    """
    Line search: find best t in [0,1] to minimize obj((1-t)*a + t*g).
    Uses coarse grid + parabolic refinement.
    """
    n = len(a)
    ts = np.linspace(0.005, 0.995, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try)

    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]

    if 0 < best_idx < n_evals - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.005, 0.995))
            a_par = (1 - t_par) * a + t_par * g
            a_par = np.maximum(a_par, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:

    Focuses perturbation on modes that contribute most to the peak.
    """
    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1
    A = fft(a, N)
    log_amp = np.log(np.abs(A) + 1e-12)
    phase = np.angle(A)

    # Identify which frequency modes contribute to the peak
    # The peak in time domain corresponds to coherent phase alignment
    # Perturb low-frequency modes more (they control overall shape)
    for k in range(1, 6):
        amp = strength * rng.uniform(0.5, 1.5) / (k * 0.5)
        log_amp[k] += amp * rng.standard_normal()
        if k < N:
            log_amp[N - k] += amp * rng.standard_normal()

    new_amp = np.exp(log_amp)
    A_new = new_amp * np.exp(1j * phase)
    a_new = irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def fourier_perturb_random(a, rng, strength=0.03):
    """Add random low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 5):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def peak_shave(a, rng, n_shaves=8):
    """Directly reduce peak contributions by modifying sequence at peak-inducing positions."""
    n = len(a)
    a = a.copy()
    for _ in range(n_shaves):
        conv = compute_conv(a)
        max_idx = int(np.argmax(conv))
        max_val = float(conv[max_idx])
        mean_val = float(np.mean(conv))

        if 0 <= max_idx < n:
            k = max_idx
            products = a[:n - k] * a[k:n]
        else:
            k = max_idx - (n - 1)
            products = a[k:] * a[:n - k]

        top_k = max(3, n // 15)
        top_idx = np.argsort(products)[-top_k:]

        reduction = 0.015 * max(0.0, max_val - mean_val) / (max_val + 1e-10)
        for idx in top_idx:
            a[idx] *= (1.0 - reduction * 0.5)

        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s

    return a


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def gd_polish(a, time_budget=100.0, tau_init=100.0, tau_end=1200.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.5 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.15 * base_lr), 3.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adapt

        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 5.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        if it % 8 == 0:
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it

    obj_final = compute_obj(a)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a.copy()

    if verbose:
        print(f"    GD: obj={best_obj:.8f}, iters={it}")
    return best_a


def lp_driven_optimize(a, time_budget=300.0, verbose=False, max_iters=500):
    """
    Main LP-driven optimization loop with peak-focusing perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall_count = 0
    iter_count = 0
    consecutive_no_improve = 0
    rng = np.random.default_rng(98765)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break
        if iter_count >= max_iters:
            break

        g, M = solve_lp_minpeak(a)
        if g is None:
            break

        best_t, obj_at_t = line_search(a, g, n_evals=30)
        iter_count += 1

        if obj_at_t < best_obj - 1e-12:
            a_try = (1 - best_t) * a + best_t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                best_obj = obj_at_t
                best_a = a_try.copy()
                a = a_try
                stall_count = 0
                consecutive_no_improve = 0
                if verbose and (iter_count % 15 == 0 or iter_count == 1):
                    elapsed = time.monotonic() - start
                    print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, "
                          f"LP_M={M:.6f}, time={elapsed:.0f}s")
        else:
            consecutive_no_improve += 1
            stall_count += 1

            if consecutive_no_improve <= 3:
                for t_small in [0.05, 0.02, 0.01, 0.005]:
                    a_small = (1 - t_small) * a + t_small * g
                    a_small = np.maximum(a_small, 0)
                    s_small = float(np.sum(a_small))
                    if s_small > 1e-10:
                        a_small = a_small / s_small
                        obj_small = compute_obj(a_small)
                        if obj_small < best_obj - 1e-12:
                            best_obj = obj_small
                            best_a = a_small.copy()
                            a = a_small
                            stall_count = 0
                            consecutive_no_improve = 0
                            break

            if stall_count >= 4:
                # Combined perturbation: spectral + subgradient
                conv_p = compute_conv(a)
                max_p = float(np.max(conv_p))
                tau = 300.0
                exp_p = np.exp(tau * (conv_p - max_p))
                p = exp_p / exp_p.sum()
                N_fft = 1
                while N_fft < 3 * n:
                    N_fft <<= 1
                P = fft(p, N_fft)
                Af = fft(a, N_fft)
                grad = 2.0 * ifft(P * np.conj(Af)).real[:n]

                lr_pert = 0.025 / (float(np.linalg.norm(grad)) + 1e-10)
                a = a - lr_pert * grad
                a = np.maximum(a, 0)
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                else:
                    a = best_a.copy()

                a = spectral_perturb(a, rng, strength=0.03)
                stall_count = 0
                consecutive_no_improve = 0

                if iter_count > 250:
                    if verbose:
                        print(f"    LP: stopping at it{iter_count}, obj={best_obj:.8f}")
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def soaker_lp_gd(a, time_budget=200.0, lp_frac=0.6, verbose=False):
    """
    Soaker-style alternation: alternate between LP iterations and GD steps.
    This helps escape flat regions where LP gets stuck.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(54321)

    block_lp = max(10, int(time_budget * lp_frac / 30))
    block_gd = max(5, int(time_budget * (1 - lp_frac) / 30))

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # LP block
        lp_time = min(block_lp, remaining * 0.6)
        if lp_time > 5:
            a = lp_driven_optimize(a, time_budget=lp_time, verbose=False, max_iters=50)
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        # GD block
        remaining = time_budget - (time.monotonic() - start)
        gd_time = min(block_gd, remaining * 0.4)
        if gd_time > 5:
            a = gd_polish(a, time_budget=gd_time, tau_init=200.0, tau_end=800.0)
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        # Perturb to escape
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20:
            a = spectral_perturb(best_a, rng, strength=0.02)

    # Final check
    if compute_obj(best_a) < best_obj:
        best_obj = compute_obj(best_a)

    if verbose:
        print(f"    Soaker: obj={best_obj:.8f}")
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP-driven optimization + spectral perturbation + multi-n exploration
    + soaker alternation + final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"Starting: n={n_current}, score={best_score:.8f}")
    print(f"Budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # === Phase 1: Aggressive LP-Driven Optimization ===
    print(f"\n=== Phase 1: LP-Driven (main) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 400.0)
    if p1_budget > 30:
        a_lp = lp_driven_optimize(best_seq, time_budget=p1_budget, verbose=True, max_iters=400)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Soaker LP/GD Alternation ===
    print(f"\n=== Phase 2: Soaker LP/GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 150.0)
    if p2_budget > 20:
        a_sk = soaker_lp_gd(best_seq, time_budget=p2_budget, lp_frac=0.5, verbose=True)
        obj_sk = compute_obj(a_sk)
        if obj_sk < best_score:
            best_score = obj_sk
            best_seq = a_sk.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_sk:.8f} (no improvement)")

    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 100, n_best - 50,
        n_best + 50, n_best + 100, n_best + 200, n_best + 400,
        600, 700, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) > 40]
    n_options = n_options[:5]
    print(f"  n options: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        a_try = resample_sequence(best_seq, n_try)

        alloc_lp = min(remaining * 0.22, 40.0)
        if alloc_lp > 10:
            a_try = lp_driven_optimize(a_try, time_budget=alloc_lp, verbose=False, max_iters=60)

        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.08, 15.0)
        if alloc_gd > 5:
            a_try = gd_polish(a_try, time_budget=alloc_gd, tau_init=150.0, tau_end=600.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Diverse Restarts ===
    print(f"\n=== Phase 4: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    # Restart 1: Spectral perturbation
    if remaining > 40:
        a_r = spectral_perturb(best_seq, rng, strength=0.06)

        remaining = deadline - time.monotonic()

        if alloc_gd > 5:
            a_r = gd_polish(a_r, time_budget=alloc_gd, tau_init=200.0, tau_end=800.0)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart (spectral): obj={best_score:.8f} **")
        else:
            print(f"  Restart (spectral): obj={obj_r:.8f}")

    # Restart 2: Peak shaving
    if remaining > 35:

        alloc = min(remaining * 0.12, 40.0)
        if alloc > 10:
            a_r = lp_driven_optimize(a_r, time_budget=alloc, verbose=False, max_iters=50)
        remaining = deadline - time.monotonic()

    # Restart 3: Fourier random perturbation

        alloc = min(remaining * 0.12, 35.0)
        if alloc > 10:
            a_r = lp_driven_optimize(a_r, time_budget=alloc, verbose=False, max_iters=45)
        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.05, 10.0)
        if alloc_gd > 5:
            a_r = gd_polish(a_r, time_budget=alloc_gd, tau_init=200.0, tau_end=800.0)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart (fourier): obj={best_score:.8f} **")
        else:
            print(f"  Restart (fourier): obj={obj_r:.8f}")

    # Restart 4: Smoothed blend
    if remaining > 25:
        ks = max(5, n_cur // 10)
        if ks % 2 == 0:
            ks += 1
        a_smooth = uniform_filter1d(best_seq, size=ks)
        a_r = 0.75 * best_seq + 0.25 * a_smooth
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        alloc = min(remaining * 0.10, 30.0)
        if alloc > 8:
            a_r = lp_driven_optimize(a_r, time_budget=alloc, verbose=False, max_iters=40)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:

            print(f"  Restart (smooth): obj={best_score:.8f} **")

            print(f"  Restart (smooth): obj={obj_r:.8f}")

    # === Phase 5: Final Intensive Polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final LP round
    if remaining > 30:
        alloc = min(remaining * 0.35, 120.0)
        a_final = lp_driven_optimize(best_seq, time_budget=alloc, verbose=True, max_iters=300)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:

            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final high-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.40, 50.0)
        a_final2 = gd_polish(best_seq, time_budget=alloc, tau_init=300.0, tau_end=2000.0, verbose=True)
        obj_final2 = compute_obj(a_final2)

        obj_final3 = compute_obj(a_final3)

    return result
