        return np.inf
    conv_seq = np.convolve(sequence, sequence, mode='full')
    max_conv = np.max(conv_seq)
    return float(2 * n_val * max_conv / (sum_val**2))

def generate_candidate(n_len: int, n_best: float, n_old: float) -> list[float]:
    """Generates a candidate sequence by reflection and projecion."""
    if n_len < 3 or np.abs(1 - n_best) > 1e-3 or np.abs(n_len - n_old) > 5:
        shift = float(np.amax(n_best) + 1.0) * float(n_old) - n_old
    else:
        shift = float(np.amax(n_old)) / 2.5
    scale = float(shift) / 2.5 if shift != 0 else 1.0
    a = np.linspace(shift * np.pi * float(n_len) / 10, scale * np.exp(-shift * np.pi / 8 * float(n_len)), n_len)
    a = [max(0, x) for x in a]
    base = float(np.sum(a))
    if base < 0.001:
        base *= 1.0001
    a = [float(x) * 10.0 / base for x in a]
    return a

def generate_exterior_candidate(previous_len: int, n_beta: float) -> list[float]:
    """Generates a distant candidate for exploratory search."""
    while True:
        b = np.random.beta(0.5, 0.5, size=previous_len)
        b = [max(0, x) for x in b]
        if sum(b) > 0.1:
            c = np.linspace(sum(b) * 0.5, sum(b) * 0.5 * n_beta, n_beta)
            c = c * n_beta if n_beta > 1 else c * 10.0 ** (1 / n_beta)
        else:
            c = [0.01] * previous_len
            c = [x * (previous_len / 1.5) if x > 0 else 0 for x in c]
        return c

def initial_reflections(n_parent: int, n_budget: int) -> tuple[

def build_linear_convolution_matrix(seq_length):
    """Builds a Toeplitz matrix for linear convolution."""
    n = seq_length
    matrix_size = 2*n-1
    A = np.zeros((n, matrix_size))
    for i in range(n):
        # Source row
        A[i, :min(n, matrix_size-i)] = [1.0] * min(n, matrix_size-i)
    return A

def apply_interior_polish(seq_len, current_point):
    """Performs interior point line search using scaled simplex."""
    try:
        A = build_linear_convolution_matrix(seq_len)
        delta = np.random.normal(0, 0.1, size=seq_len)
        h = current_point + delta * np.var(delta)
        obj_val = evaluate_sequence(h)
        return h, float(obj_val)
    except:
        return current_point, evaluate_sequence(current_point)

def ternary_search_segment(seq, start, end):
    """Performs ternary search to minimize peak."""
    n_samples = min(3, end-start)
    segment_vals = np.random.uniform(start, end, n_samples)
    conv_vals = [np.max(np.convolve(x, x, mode='full')) for x in segment_vals]
    pivot = np.median(segment_vals)
    left = segment_vals[np.argmin(conv_vals)][:5]
    right = segment_vals[np.argmin(conv_vals)][-5:]
    return seq[left.astype(int):right.astype(int)+1].copy()

def main_optimization_loop(sequence_length: int, initial_budget: int) -> list[float]:
    """Main optimization cycle with reflection and return."""
    deadline = time.time() + initial_budget  # Round binding clock
    
    # Initial reflection space candidates

    tentative_seq = [max(0, x) for x in tentative_seq]
    s0 = np.sum(tentative_seq)
    tentative_seq = [x / s0 for x in tentative_seq]
    
    best_sequence = tentative_seq.copy()
    best_score = evaluate_sequence(best_sequence)
    
    improvement_flags = [1, 1, 1, 1]  # Traversal mechanism and consistent epsilon
    
    # Extensive BPROP with probabilistic exploration
    for _ in range(10):  # Explore multiple lengths with BPROP
        try:
            if time.time() > deadline:
                break
            
            # Interior point search: gradient-free method
            new_sequence, new_score = apply_interior_polish(sequence_length, best_sequence)
            if new_score < best_score:
                best_sequence = new_sequence.copy()
                best_score = new_score
            
            # Extensive reflection mechanism
            delta, delta_score = initial_reflections(sequence_length, 300)
            best_sequence = delta.copy()
            best_score = delta_score
            
            # Exterior persistent search: larger scale
            exterior_multiplier = float(np.sqrt(sequence_length)) * 0.8
            big_candidate = generate_exterior_candidate(sequence_length, exterior_multiplier)
            big_candidate_score = float(np.min([evaluate_sequence(x) for x in [best_sequence, big_candidate]]))
            best_sequence = big_candidate.copy()
            best_score = big_candidate_score
            
        except:
            # Handling computational infeasibility: reduce dimension
            best_sequence = best_sequence[::2]  # Subsampling
            best_score = evaluate_sequence(best_sequence)
    
    # Final precision refinement: ensure no floating point errors
    formatted_seq = [np.around(x, decimals=5) for x in best_sequence]
    formatted_score = evaluate_sequence(formatted_seq)
    
    if formatted_score < best_score:
        best_score = formatted_score
    
    return best_sequence
