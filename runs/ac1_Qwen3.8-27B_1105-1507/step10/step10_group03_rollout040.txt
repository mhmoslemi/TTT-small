"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses p-norm smoothed gradient descent, LP fixed-point iteration, pair-swap
local search, coordinate descent, and multi-n exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = full_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def pnorm_gradient(a, p):
    """
    Compute gradient of ||conv(a,a)||_p for non-negative a.
    Returns (gradient of length n, the p-norm value).
    """
    n = len(a)
    conv = full_conv(a)
    norm_p = float(np.sum(conv ** p)) ** (1.0 / p)
    if norm_p < 1e-300:
        return np.zeros(n), 0.0
    c = conv ** (p - 1)
    N = 1
    while N < 4 * n:
        N <<= 1
    C = fft(c, N)
    A = fft(a, N)
    T1 = ifft(C * np.conj(A)).real[:n]
    T2 = ifft(C * A).real[:n]
    g = (T1 + T2) / norm_p
    return g, norm_p


def lse_gradient(a, tau):
    """Log-sum-exp smoothed gradient of max(conv(a,a))."""
    n = len(a)
    conv = full_conv(a)
    cmax = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - cmax))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 4 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    T1 = ifft(P * np.conj(A)).real[:n]
    T2 = ifft(P * A).real[:n]
    g = (T1 + T2)  # No division needed since p is normalized
    return g, cmax


def normalize(a):
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s

                best_it = iters

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    tau={tau:5d}: obj={best_obj:.10f}, iters={iters}, t={elapsed:.0f}s")

    return best_a


def lp_solve(a):
    """
    Solve LP: min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    if n > 3000:
        return None, None

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget, verbose=False, seed=42, max_iters=100):
    """
    Iterative LP polishing with adaptive blending (additive and multiplicative).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    last_t = 0.3

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        g_lp, M_lp = lp_solve(a)
        if g_lp is None:
            break

        # Try multiple strategies
        candidates = []

        # Additive blending
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0,
                  max(0.03, last_t * 0.7), min(1.0, last_t * 1.3)]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            candidates.append(a_try)

        # Multiplicative (exponential weighting) blending
        for t in [0.1, 0.3, 0.5, 1.0, 2.0]:
            a_try = a * (g_lp / (a + 1e-30)) ** t
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            candidates.append(a_try)

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_strategy = None

        for a_try in candidates:
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a
            stall = 0
            if verbose:
                print(f"    LP it{it}: obj={best_obj:.10f}")
        else:
            stall += 1
            # Perturb to escape
            pert_mag = 0.01 + 0.01 * stall
            a = a + pert_mag * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            if stall > 8:
                break

    return best_a


def pair_swap_search(a, time_budget, verbose=False, seed=42):
    """
    Local search: randomly swap/transfer mass between pairs of coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    conv = full_conv(a)
    current_max = float(np.max(conv))

    it = 0
    no_improve = 0
    max_no_improve = 500

    while time.monotonic() - start < time_budget and no_improve < max_no_improve:
        # Pick random pair
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Transfer small amount of mass
        d = rng.uniform(0.01, 0.15) * a[i]
        if d > a[i] * 0.5:
            d = a[i] * 0.5

        a_trip = a.copy()
        a_trip[i] -= d
        a_trip[j] += d
        a_trip = np.maximum(a_trip, 0)
        s_trip = float(np.sum(a_trip))
        if s_trip < 1e-10:
            continue
        a_trip = a_trip / s_trip

        obj_trip = compute_obj(a_trip)

        if obj_trip < best_obj - 1e-12:
            a = a_trip
            best_obj = obj_trip
            best_a = a.copy()
            no_improve = 0
            it += 1
            if verbose and it % 500 == 0:
                print(f"    PS it{it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            it += 1

    return best_a
