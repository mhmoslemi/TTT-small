import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_lp_active(a, top_k=None, warm_start=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for k in active set, sum(g)=1, g>=0.
    Uses scipy.linprog for speed.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Compute conv(a, a) to find active lags
    conv_aa = compute_conv(a_norm)
    if top_k is None:
        top_k = min(2 * n - 1, max(50, int(0.1 * (2 * n - 1))))
    
    # Get top lags
    active_idx = np.argsort(conv_aa)[::-1][:top_k]
    active_idx = np.sort(active_idx)
    
    # Build constraint matrix for active lags only
    # conv(a,g)[k] = sum_j a[j] * g[k-j] for valid j
    # This is a linear function of g.
    # A[i, j] = a[j] if k=i maps to valid j, else 0
    # Specifically: conv(a,g)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * g[k-j]
    # Let m = k-j, so j = k-m. Then conv(a,g)[k] = sum_{m: 0<=m<n, 0<=k-m<n} a[k-m] * g[m]
    # So A[k, m] = a[k-m] if 0 <= k-m < n, else 0
    
    m_constraints = len(active_idx)
    A = np.zeros((m_constraints, n))
    for i, k in enumerate(active_idx):
        # A[i, m] = a[k-m] if 0 <= k-m < n and 0 <= m < n
        m_min = max(0, k - (n - 1))
        m_max = min(n - 1, k)
        if m_min <= m_max:
            A[i, m_min:m_max+1] = a[k - np.arange(m_min, m_max + 1)]
    
    # Objective: minimize M (variable index n)
    c = np.zeros(n + 1)
    c[n] = 1.0  # M is the last variable
    
    # Inequality constraints: A @ g <= M => A @ g - M <= 0
    A_ub = np.hstack([A, -np.ones((m_constraints, 1))])
    b_ub = np.zeros(m_constraints)

    A_eq = np.zeros((1, n + 1))

    b_eq = np.ones(1)
    
    # Bounds: g >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]
    
    # Try solving
    from scipy.optimize import linprog
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

"""

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ab(a, b):
    """Full linear convolution a*b via FFT. Returns array of length 2n-1."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp_active(a, n_active=None, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for k in top lags, sum(g)=1, g>=0.
    Returns (g, M, active_indices) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    if n_active is None:
        n_active = min(2 * n - 1, max(30, int(0.05 * (2 * n - 1))))
    
    # Find active lags (where conv(a,a) is largest)
    conv_aa = compute_conv(a_norm)
    active_idx = np.argsort(conv_aa)[::-1][:n_active]
    active_idx = np.sort(active_idx)
    
    # Build constraint matrix
    # conv(a,g)[k] = sum_m a[k-m] * g[m] for valid m
    m_cons = len(active_idx)
    A = np.zeros((m_cons, n))
    for i, k in enumerate(active_idx):
        m_min = max(0, k - (n - 1))
        m_max = min(n - 1, k)
        if m_min <= m_max:
            A[i, m_min:m_max+1] = a_norm[k - np.arange(m_min, m_max + 1)]
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.hstack([A, -np.ones((m_cons, 1))])
    b_ub = np.zeros(m_cons)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 5.0})
        if result.success and result.x is not None:
            g = np.maximum(result.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
            M = float(result.x[n])
            return g, M, active_idx
    except Exception:
        pass
    
    return None, None, None


def backtracking_line_search(a, g, best_obj, max_iter=30):
    """
    Find best t in [0, 1] such that obj((1-t)*a + t*g) is minimized.
    Uses backtracking from t=1.0.
    """
    best_t = 0.0
    best_local_obj = best_obj
    best_a = a.copy()
    
    t = 1.0
    for _ in range(max_iter):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            t *= 0.5
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        
        if obj_try < best_local_obj:
            best_local_obj = obj_try
            best_t = t
            best_a = a_try.copy()
        
        # Armijo condition
        if obj_try < best_obj - 1e-14:
            # Acceptable step, but try to go further
            if t >= 0.99:
                break
            t = min(1.0, t * 1.3)  # Try superlinear step
        else:
            t *= 0.5
            if t < 0.01:
                break
    
    return best_a, best_local_obj, best_t


def gradient_smoothed(a, tau=100.0):

    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))

    # The derivative is: for each term a[j]*a[k-j], d/d(a[m]) gives:
    #   a[k-m] if j=m (i.e., the second factor is a[k-m])
    #   + a[m]... no, if k-j=m, i.e., j=k-m, then d/d(a[m]) a[k-j] = 1, so contribution is a[k-m]
    # Wait, let me redo this.
    # conv(a,a)[k] = sum_j a[j] * a[k-j]
    # d/d(a[m]) = sum_j [delta(j,m) * a[k-j] + a[j] * delta(k-j,m)]
    # = a[k-m] (if 0<=k-m<n) + a[k-m] (if 0<=k-m<n)
    # Wait: delta(j,m) * a[k-j] gives a[k-m] when j=m, provided 0<=k-m<n.
    # delta(k-j,m) gives j=k-m, so a[j] = a[k-m], provided 0<=k-m<n.
    # So d/d(a[m]) conv(a,a)[k] = 2*a[k-m] if 0<=k-m<n, else 0.
    # But wait, this counts the term twice when j = k-j, i.e., k = 2j.
    # Actually no, the two terms are from different sources (derivative of first factor vs second factor),
    # so they're both valid even when j = k-j.

    # This is a cross-correlation of p and a!

    A = fft(a, N)

    # This is (a * p)[m] in the cross-correlation sense... 
    # Actually, let's think of it as: grad[m] = 2 * (a cross-correlated with p)[m]
    # where (a ⊛ p)[m] = sum_l a[l] * p[l+m]
    # This is the same as: (a ⊛ p)[m] = sum_l a[l] * p_flip[n-1-l] flipped... 
    # Let me just use direct computation for clarity, or use the FFT trick.
    # 
    # sum_l a[l] * p[l+m] = (a * p)[m + (2n-2)] in the FFT sense? 
    # Let's use: ifft(fft(a) * fft(p)) gives circular convolution.
    # We need cross-correlation: sum_l a[l] * p[l+m] = sum_l a[l] * p_flip[(2n-2)-(l+m)] ... 
    # This is getting complicated. Let me just use the direct formula.

    # because we need linear, not circular.
    
    # Let me use the identity: sum_l a[l] p[l+m] for m in [0, 2n-2]

    # In frequency: conj(Fa) * Fp gives ifft as sum_l a[l] * p[m-l] (circular)

    # Or: let q = p[::-1] (reversed), then sum_l a[l] * p[l+m] = sum_l a[l] * q[(2n-2)-(l+m)]
    # This is (a * q) at index 2n-2-m... this is getting too complicated.
    
    # Let me just do it directly with numpy for the gradient (n is at most ~2000, so 2n-1 is ~4000)

def accelerated_gd(a, time_budget=100.0, tau_init=50.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    base_lr = 5.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        cos_val = 0.5 * (1 + np.cos(np.pi * progress))
        tau = tau_init + (tau_end - tau_init) * (1 - cos_val)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g, _ = gradient_smoothed(y, tau)
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)
        
        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it
    
    return best_a


def spectral_init(n, mode='chirp', rng=None):
    """Initialize a non-negative sequence with good autocorrelation properties."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    if mode == 'chirp':
        # Chirp signal: a[j] = exp(i * pi * j^2 / n), take magnitude or real part
        # For non-negative: use |chirp| or cos^2
        x = np.arange(n)
        phase = np.pi * x**2 / n
        a = 0.5 + 0.5 * np.cos(phase + rng.uniform(0, 2*np.pi))
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'gaussian':
        x = np.linspace(-3, 3, n)
        a = np.exp(-x**2 / 2) + 0.01
    elif mode == 'uniform_noise':
        a = 1.0 + 0.1 * rng.random(n)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        
        # Adaptive active set size
        n_active = int(n_active_start + (n_active_end - n_active_start) * progress)

        g_lp, M_lp, active = solve_lp_active(a, n_active=n_active)
        if g_lp is None:
            break

        g_mom = a + momentum_vec
        g_mom = np.maximum(g_mom, 0)
        s_mom = float(np.sum(g_mom))
        if s_mom > 1e-10:
            g_mom = g_mom / s_mom
            # Use momentum version if it looks reasonable
            obj_mom = compute_obj(g_mom)
            if obj_mom < best_obj:
                g_lp = g_mom
        
        it += 1
        
        # Backtracking line search
        a_new, obj_new, t_used = backtracking_line_search(a, g_lp, best_obj)
        
        if obj_new < best_obj - 1e-14:

            best_a = a_new.copy()
            a = a_new
            best_it = it
            stall_count = 0
            if verbose and it % 20 == 0:

        else:

                a = best_a * (1 + strength * rng.normal(0, 1, n))

                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                momentum_vec = np.zeros(n)
                stall_count = 0
                if verbose:
                    print(f"    LPA it{it}: perturbed, strength={strength:.3f}")
