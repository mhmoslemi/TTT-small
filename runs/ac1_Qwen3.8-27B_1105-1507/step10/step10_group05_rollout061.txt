def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_A(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, time_limit=5.0):
    """

    Returns (g, t) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_A(a_norm)

    # Minimize t: c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))

    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

                      bounds=bounds, method='highs')

            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
            t = float(res.x[n])
            return g, t
    except Exception:
        pass
    
    return None, None

"""

Uses fast LP-based iteration (scipy HiGHS), smoothed gradient polishing,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_A(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_fast(a):
    """
    Solve LP: min t s.t. conv(a,g)[k] <= t for all k, sum(g)=1, g>=0.
    Uses scipy HiGHS. Returns (g, t_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_A(a_norm)

    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality: A @ g - t <= 0  =>  [A | -1] @ x <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds
    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10.0})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
            t_opt = float(res.x[n])
            return g, t_opt
        elif res.status in (1, 3, 5):
            # Iteration limit or time limit - use solution if available
            if res.x is not None:
                g = np.maximum(res.x[:n], 0.0)
                s = float(np.sum(g))
                if s > 1e-10:
                    g = g / s
                    t_opt = float(res.x[n])
                    return g, t_opt
    except Exception:
        pass

    return None, None

def smoothed_gradient(a, tau):
    """Compute gradient of smoothed objective using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    grad_full = ifft(P * np.conj(A)).real
    grad = 2.0 * grad_full[:n]
    return grad, conv, max_conv


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gd_polish(a, time_budget=60.0, tau_init=20.0, tau_max=2000.0, verbose=False):
    """
    Smoothed gradient descent with Nesterov acceleration and adaptive tau.
    Continuation method: tau increases over time for sharper max approximation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0

    # Nesterov parameters
    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Cosine annealing for tau
        cos_val = 0.5 * (1.0 + np.cos(np.pi * progress))
        tau = tau_init + (tau_max - tau_init) * (1.0 - cos_val)

        # Nesterov lookahead
        y = a + beta * v
        y = project_simplex(y)

        grad, conv, max_conv = smoothed_gradient(y, tau)

        # Subtract mean for simplex constraint
        grad_centered = grad - np.mean(grad)

        # Adaptive learning rate (Barzilai-Borwein style)
        lr = base_lr / (1.0 + 0.1 * progress)

        v = beta * v - lr * grad_centered
        a_new = y + v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        # Reset if stuck
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it

        if verbose and it % 2000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def fourier_perturbation(a, rng, strength=0.05):
    """Add low-frequency Fourier modes as structured perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(8, n // 10)
    for k in range(1, n_modes + 1):
        freq = 2.0 * np.pi * k / n
        amp = strength * rng.uniform(0.3, 1.0) / math.sqrt(k)
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturbation(a, rng, strength=0.05):
    """Add random multiplicative perturbation."""
    n = len(a)
    pert = 1.0 + rng.normal(0, strength, n)
    a_new = a * np.maximum(pert, 0)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def init_parent(n):
    """Initialize from parent sequence."""
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                return resample_sequence(parent_arr / s, n)
    except Exception:
        pass
    return np.ones(n) / n


def init_ramp(n, direction=1):
    """Initialize with a smooth ramp."""
    x = np.linspace(0, 1, n)
    if direction == 1:
        a = 0.5 + 0.5 * x
    else:
        a = 0.5 + 0.5 * (1 - x)
    a = a ** 2  # Make it more peaked
    s = float(np.sum(a))
    return a / s


def init_gaussian(n, center=0.3, width=0.2):
    """Initialize with a Gaussian bump."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 0.01)
    s = float(np.sum(a))
    return a / s


def init_semicircle(n):
    """Initialize with a semicircle (Wigner semicircle) profile."""
    x = np.linspace(-1, 1, n)
    a = np.sqrt(np.maximum(1 - x**2, 0))
    a = np.maximum(a, 0.01)
    s = float(np.sum(a))
    return a / s


def init_fourier_flat(n, rng):
    """Initialize with approximately flat power spectrum."""
    # Random phases in frequency domain
    N = n
    phases = rng.uniform(0, 2 * np.pi, N)
    mags = np.ones(N)
    # Enforce Hermitian symmetry for real signal
    spectrum = mags * np.exp(1j * phases)
    spectrum[0] = np.abs(spectrum[0])
    if N % 2 == 0:
        spectrum[N // 2] = np.abs(spectrum[N // 2])
    for k in range(1, N // 2 + 1):
        if N - k < N:
            spectrum[N - k] = np.conj(spectrum[k])
    a = ifft(spectrum).real
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def full_optimize(a, time_budget=200.0, verbose=False, seed=42):
    """
    Full optimization pipeline: LP iteration + gradient polish.
    """
    start = time.monotonic()
    n = len(a)

    # Phase 1: LP iteration (main workhorse)
    lp_budget = min(time_budget * 0.6, time_budget - 20)
    if lp_budget > 10:
        a = lp_optimize(a, time_budget=lp_budget, verbose=verbose, seed=seed)

    # Phase 2: Gradient polish
    remaining = time_budget - (time.monotonic() - start)
    gd_budget = min(remaining * 0.5, 60.0)
    if gd_budget > 10:
        a_gd = gd_polish(a, time_budget=gd_budget, tau_init=50.0, tau_max=1500.0)
        obj_gd = compute_obj(a_gd)
        obj_lp = compute_obj(a)
        if obj_gd < obj_lp:
            a = a_gd

    # Phase 3: Quick LP polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        a = lp_optimize(a, time_budget=min(remaining * 0.7, 50.0), verbose=False, seed=seed + 1)

    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-initialization, multi-length exploration with
    LP iteration and gradient polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try multiple (n, initialization) combinations
    # Priority: parent n first, then nearby, then diverse
    n_values = [800, 700, 900, 600, 1000, 1100, 1200]
    init_methods = ['parent', 'ramp1', 'ramp2', 'gauss', 'semicircle', 'fourier']

    # Score-based allocation: give more time to promising configs
    results = []

    for n_try in n_values:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Determine how many initializations to try for this n
        if n_try == 800:
            # Parent length: try all initializations
            inits_for_n = init_methods
            alloc_each = min(remaining / (len(inits_for_n) * (len(n_values) - 1 + 1)), 120.0)
        else:
            # Other lengths: try fewer initializations
            inits_for_n = init_methods[:3]
            alloc_each = min(remaining / (len(inits_for_n) * 3), 80.0)

        for init_name in inits_for_n:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            # Choose initialization
            if init_name == 'parent':
                a_init = init_parent(n_try)
            elif init_name == 'ramp1':
                a_init = init_ramp(n_try, direction=1)
            elif init_name == 'ramp2':
                a_init = init_ramp(n_try, direction=-1)
            elif init_name == 'gauss':
                a_init = init_gaussian(n_try, center=rng.uniform(0.2, 0.8),
                                       width=rng.uniform(0.15, 0.35))
            elif init_name == 'semicircle':
                a_init = init_semicircle(n_try)
            elif init_name == 'fourier':
                a_init = init_fourier_flat(n_try, rng)
            else:
                a_init = np.ones(n_try) / n_try

            # Optimize
            a_opt = full_optimize(a_init, time_budget=alloc_each, verbose=False,
                                  seed=seed + n_try + hash(init_name) % 100)

            obj = compute_obj(a_opt)
            results.append((obj, n_try, init_name, a_opt))

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  n={n_try}, init={init_name}: obj={obj:.8f} **")
            else:
                print(f"  n={n_try}, init={init_name}: obj={obj:.8f}")

    # Sort results and do intensive polish on the top ones
    results.sort(key=lambda x: x[0])
    print(f"\n  Top results: {[f'{r[0]:.6f}(n={r[1]},{r[2]})' for r in results[:5]]}")

    # Intensive polish on best
    remaining = deadline - time.monotonic()
    if remaining > 60 and best_seq is not None:
        print(f"\n=== Intensive Polish (remaining {remaining:.0f}s) ===")
        n_best = len(best_seq)
        alloc = min(remaining * 0.5, 300.0)
        a_pol = full_optimize(best_seq, time_budget=alloc, verbose=True, seed=99999)
        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Intensive polish: obj={best_score:.8f} **")

        # Gradient polish with longer budget
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc_gd = min(remaining * 0.4, 100.0)
            a_gd = gd_polish(best_seq, time_budget=alloc_gd,
                             tau_init=100.0, tau_max=3000.0, verbose=True)
            obj_gd = compute_obj(a_gd)
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
                print(f"  Final GD: obj={best_score:.8f} **")

        # One more LP pass
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_lp = lp_optimize(best_seq, time_budget=min(remaining * 0.6, 80.0),
                               verbose=False, seed=77777)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

    # Diversification: try a few perturbed versions of the best
    remaining = deadline - time.monotonic()
    if remaining > 40 and best_seq is not None:
        n_best = len(best_seq)
        for r in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            if r == 0:
                a_pert = fourier_perturbation(best_seq, rng, strength=0.03)
            elif r == 1:
                a_pert = random_perturbation(best_seq, rng, strength=0.05)
            else:
                # Blend with uniform
                a_pert = 0.8 * best_seq + 0.2 * np.ones(n_best) / n_best
                a_pert = a_pert / a_pert.sum()

            alloc = min(remaining * 0.3, 60.0)
            if alloc > 10:
                a_pert = full_optimize(a_pert, time_budget=alloc, verbose=False, seed=50000 + r)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  Diversify {r}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
