"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based minimax with golden-section line search, periodic perturbation,
support reduction, and multi-length exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:
            continue
    return None, None

import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_prob=None):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M, prob) or (None, None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value), prob
        except Exception:
            continue
    return None, None, None


def golden_section_obj(a, g, n, lo=0.0, hi=1.0, tol=1e-12, max_iter=60):

    Uses the fact that max of quadratics is convex."""

    C_g = compute_conv(g)
    C_x = compute_conv_cross(a, g)

    def f_of_t(t):
        vals = (1.0 - t) ** 2 * C_a + 2.0 * t * (1.0 - t) * C_x + t ** 2 * C_g
        return float(np.max(vals))

    inv_phi2 = (3.0 - np.sqrt(5.0)) / 2.0
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0

    c = hi - inv_phi * (hi - lo)
    d = lo + inv_phi * (hi - lo)

    fd = f_of_t(d)

        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - inv_phi * (hi - lo)

            c = d
            fc = fd
            d = lo + inv_phi * (hi - lo)

def flat_spectrum_init(n, rng):
    """Initialize with flat power spectrum, rectified to be non-negative."""
    # Random phase, flat magnitude
    freqs = np.zeros(n)
    freqs[1:] = 1.0
    phases = rng.uniform(0, 2 * np.pi, n)
    spectrum = freqs * np.exp(1j * phases)
    # Make it real by conjugate symmetry
    if n % 2 == 0:
        spectrum[n // 2] = float(spectrum[n // 2])
    spectrum[:n // 2 + 1] = freqs[:n // 2 + 1] * np.exp(1j * phases[:n // 2 + 1])
    # Build full spectrum with conjugate symmetry
    full_spec = np.zeros(n, dtype=complex)
    half = n // 2
    full_spec[:half + 1] = freqs[:half + 1] * np.exp(1j * phases[:half + 1])
    for i in range(1, n - half):
        full_spec[n - i] = np.conj(full_spec[i])
    a = ifft(full_spec).real
    a = np.abs(a)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def smooth_gradient_step(a, tau, lr_scale=1.0):
    """One step of smoothed gradient descent on the normalized objective."""
    n = len(a)
    a = a / np.sum(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = conv - max_conv
    # Clip for numerical stability
    shifted = np.clip(shifted, -50.0 / tau, 0.0)
    exp_vals = np.exp(tau * shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    # Gradient via FFT: grad[j] = 2 * sum_k p[k] * a[k-j] (valid range)
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A_f = fft(a, N)
    g_full = ifft(P * np.conj(A_f)).real
    grad = 2.0 * g_full[:n]

    a_new = a - lr_scale * grad
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a
    return a_new / s


def high_tau_polish(a, time_budget, tau_start=100, tau_end=5000, verbose=False):

    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))

        return a

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress

        lr = 0.5 / (tau + 1.0) * lr_decay
        a = smooth_gradient_step(a, tau, lr)
        
        obj = compute_obj(a)

            best_a = a.copy()

            best_it = it

def support_shrink(a, frac, refine_budget=20, verbose=False):
    """Zero out entries below the frac-th percentile and re-optimize."""
    n = len(a)
    a = a / np.sum(a)
    sorted_vals = np.sort(a)[::-1]
    k = max(2, int(n * frac))
    threshold = sorted_vals[k - 1]
    mask = a >= threshold
    if mask.sum() < 2:
        return a
    a_shrunk = np.zeros(n)
    a_shrunk[mask] = a[mask]
    a_shrunk = a_shrunk / a_shrunk.sum()

    if refine_budget > 5:
        a_shrunk = lp_polish(a_shrunk, time_budget=refine_budget, verbose=False)

    return a_shrunk

    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.10f}")
    except Exception:
        pass

    n_base = 800
    if best_seq is None:
        best_seq = diverse_init(n_base, rng, mode=5)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_base}): obj={best_score:.10f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.10f}")

    # === Phase 1: Intensive LP Polish on parent (250s) ===
    print(f"\n=== Phase 1: LP Polish (parent) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 250.0)
    if p1_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p1_budget, verbose=True, rng=rng)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1: obj={best_score:.10f} **")
        else:
            print(f"  Phase1: obj={obj_lp:.10f} (no improvement)")

    # === Phase 2: High-tau refinement (100s) ===
    print(f"\n=== Phase 2: High-Tau GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.1, 100.0)
    if p2_budget > 15:
        a_gd = high_tau_polish(best_seq, time_budget=p2_budget,
                               tau_start=200, tau_end=8000, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            n_best = len(best_seq)
            print(f"  Phase2: obj={best_score:.10f} **")
        else:
            print(f"  Phase2: obj={obj_gd:.10f}")

    # === Phase 3: Multi-n exploration (500s) ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.4, 500.0)

    # Generate candidate n values
    n_candidates = sorted(set([
        n_best - 200, n_best - 100, n_best - 50,
        n_best + 50, n_best + 100, n_best + 200,
        n_best + 400, n_best + 800, n_best + 1500,
        400, 500, 600, 1000, 1200, 1600, 2000, 2500, 3000
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 3500 and n != n_best]
    n_candidates = n_candidates[:10]
    print(f"  Trying n values: {n_candidates}")

    per_n_budget = p3_budget / max(1, len(n_candidates))
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(per_n_budget, remaining - 30)
        if alloc < 40:
            break

        # Resample from best
        a_try = resample_sequence(best_seq, n_try)

        # LP polish
        lp_alloc = min(alloc * 0.45, remaining - 30)
        if lp_alloc > 15:
            a_try = lp_polish(a_try, time_budget=lp_alloc, verbose=False, rng=rng)

        # High-tau
        remaining = deadline - time.monotonic()
        gd_alloc = min(alloc * 0.3, remaining - 20)
        if gd_alloc > 10:
            a_try = high_tau_polish(a_try, time_budget=gd_alloc,
                                    tau_start=100, tau_end=4000, verbose=False)

        # Final LP
        remaining = deadline - time.monotonic()
        if remaining > 25:
            a_try = lp_polish(a_try, time_budget=min(20, remaining - 10),
                              verbose=False, rng=rng)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 4: Support shrink exploration (150s) ===
    print(f"\n=== Phase 4: Support Shrink ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.12, 150.0)
    for frac in [0.5, 0.6, 0.7, 0.8]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(30, remaining - 10)
        if alloc < 10:
            break
        a_shrunk = support_shrink(best_seq, frac, refine_budget=alloc, verbose=False)
        obj_shrunk = compute_obj(a_shrunk)
        if obj_shrunk < best_score:
            best_score = obj_shrunk
            best_seq = a_shrunk.copy()
            print(f"  ** Shrink frac={frac}: obj={best_score:.10f} **")
        else:
            print(f"  Shrink frac={frac}: obj={obj_shrunk:.10f}")

    # === Phase 5: Diverse restarts (200s) ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.15, 200.0)
    n_restarts = max(1, int(p5_budget / 50))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(50, remaining - 10)
        if alloc < 20:
            break

        mode = r % 7
        n_r = n_best
        a_r = diverse_init(n_r, rng, mode=mode)
        # Blend with best
        blend = 0.4
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        # Optimize
        half = alloc * 0.5
        a_r = lp_polish(a_r, time_budget=half, verbose=False, rng=rng)
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_r = lp_polish(a_r, time_budget=min(alloc * 0.3, remaining - 10),
                           verbose=False, rng=rng)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  ** Restart {r} (mode={mode}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.10f}")

    # === Phase 6: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s")

    # 6a: LP polish
    if remaining > 40:
        alloc = min(remaining * 0.25, 120.0)
        a_f = lp_polish(best_seq, time_budget=alloc, verbose=True, rng=rng)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    # 6b: High-tau at very high tau
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 100.0)
        a_f = high_tau_polish(best_seq, time_budget=alloc,
                              tau_start=1000, tau_end=20000, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final HTau: obj={best_score:.10f} **")

    # 6c: Another LP round
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 80.0)
        a_f = lp_polish(best_seq, time_budget=alloc, verbose=True, rng=rng)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final2 LP: obj={best_score:.10f} **")

    # 6d: Support shrink on final
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for frac in [0.7, 0.8, 0.9]:
            a_f = support_shrink(best_seq, frac, refine_budget=min(10, remaining - 5), verbose=False)
            obj_f = compute_obj(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final Shrink({frac}): obj={best_score:.10f} **")
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break

    # 6e: Last LP push
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_f = lp_polish(best_seq, time_budget=min(remaining - 5, 30), verbose=False, rng=rng)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Ultra-final LP: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val != np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
