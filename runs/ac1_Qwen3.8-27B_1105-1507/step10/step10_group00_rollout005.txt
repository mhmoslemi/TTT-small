Multi-strategy optimization combining softmax gradient descent,
projected gradient with momentum, coordinate descent, and LP polishing

import numpy as np
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_gradient(a, tau):
    """Compute smoothed gradient of 2n * max_k conv_k using softmax approximation."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    if max_conv < 1e-15:
        return np.zeros(n), conv
    
    # Softmax weights
    scaled = conv / tau
    scaled -= np.max(scaled)
    w = np.exp(scaled)
    W = np.sum(w)
    
    # Gradient: g_i = 4n * sum_k (w_k/W) * a[k-i] / ... 
    # c[i] = sum_k w[k] * a[k-i] for i = 0,...,n-1
    # Using FFT cross-correlation: ifft(fft(w) * conj(fft(a)))
    size = 1
    while size < (2*n-1) + n - 1:
        size <<= 1
    Wf = rfft(w, size)
    Af = rfft(a, size)
    c = irfft(Wf * np.conj(Af), size)
    c = c[:n]  # We want c[0], ..., c[n-1]
    
    g = 4.0 * n * c / W
    return g, conv

        g, conv = softmax_gradient(a, tau)

def coordinate_descent(a, time_budget=60.0, rng=None, verbose=False):

        rng = np.random.default_rng(42)

    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))

        return a

    best_obj = compute_obj(a)

    it = 0

        # a_i + t, a_j - t for t in some range

            continue

            a_try = a.copy()

            a_try[j] -= t
            if a_try[i] < -1e-15 or a_try[j] < -1e-15:

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_init=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    if M_init is not None:
        constraints.append(M <= M_init)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

    elif mode == 4:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / (0.05 + 5.0 * (2 * x - 1) ** 4)
    elif mode == 7:
        a = (2 * x - 1) ** 6 + 0.02
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
