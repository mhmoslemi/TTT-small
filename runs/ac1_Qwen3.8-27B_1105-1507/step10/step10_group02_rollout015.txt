def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj_normalized(a):
    """Assume sum(a) = 1, return 2*n*max(conv)."""
    n = len(a)
    return 2.0 * n * float(np.max(compute_conv(a)))

def symmetrize(a):
    """Symmetrize: a = (a + flip(a)) / 2."""
    return (a + a[::-1].copy()) / 2.0

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # (a*g)[k] = sum_j a[j] * g[k-j]
        # For fixed j, g index is k-j, valid when 0 <= k-j < n, i.e., j <= k < j+n
        k_start = j
        k_end = min(2*n-1, j+n)
        A[k_start:k_end, j] = a[:k_end-k_start]
    return A

def solve_lp(a, time_limit=1.0):
    """Solve LP: min M s.t. a*g <= M (componentwise), sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:

                s = float(np.sum(g_val))

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    return 2.0 * n * float(np.max(compute_conv(a_norm)))


def symmetrize(a):
    """Symmetrize sequence: (a + flip(a))/2, then renormalize."""
    a_sym = (a + a[::-1].copy()) / 2.0
    s = float(np.sum(a_sym))
    if s > 1e-10:
        a_sym = a_sym / s
    return a_sym


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

def compute_pair_obj(v1, v2, i, j, a, conv, n, S_other):
    """Compute objective for a pair update (approximate, using current conv as base)."""
    # Quick approximation: evaluate the max convolution after updating i and j
    # For speed, we only check the most affected positions
    S = S_other + v1 + v2
    if S < 1e-10:
        return 1e30

    # The change in conv at position k due to changing a[i] from old_v1 to v1
    # and a[j] from old_v2 to v2:
    # conv[k] changes by: 2*(v1-old_v1)*a[k-i] + 2*(v2-old_v2)*a[k-j] + (v1-old_v1)^2 (if k==2i) + ...
    # For speed, just check a few key positions
    max_val = 0.0
    # Check positions near 2i, 2j, i+j
    check_ks = set()
    for k in [2*i, 2*j, i+j]:
        if 0 <= k <= 2*n-2:
            check_ks.add(k)
    # Also check current max position
    k_max_cur = int(np.argmax(conv))
    check_ks.add(k_max_cur)
    check_ks.add(k_max_cur + 1 if k_max_cur + 1 <= 2*n-2 else k_max_cur - 1)

    for k in check_ks:
        if k < 0 or k > 2*n-2:
            continue
        # Approximate new conv[k]
        new_val = conv[k]
        # Contribution from i
        ki = k - i
        if 0 <= ki < n:
            new_val += 2.0 * (v1 - a[i]) * a[ki]
        if k == 2*i:
            new_val += v1*v1 - a[i]*a[i]
        # Contribution from j
        kj = k - j
        if 0 <= kj < n:
            new_val += 2.0 * (v2 - a[j]) * a[kj]
        if k == 2*j:
            new_val += v2*v2 - a[j]*a[j]
        # Cross term (i contribution to j's position and vice versa)
        if i != j:
            if 0 <= k - i < n and 0 <= k - j < n:
                pass  # Already counted above
        max_val = max(max_val, new_val)

    return 2.0 * n * max_val / (S * S)

    best_obj = compute_obj_norm(a_current)
    best_params = params.copy()

    def obj_func(p):
        return parametric_obj(p, n)

    try:
        result = minimize(obj_func, params, method='L-BFGS-B',
                          options={'maxiter': 200, 'maxfun': 500, 'ftol': 1e-10})
        if result.fun < best_obj:
            best_obj = result.fun
            best_params = result.x
    except Exception:
        pass

    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        # Try different M values
        for M2 in [3, 5, 10]:
            if M2 >= n // 3:
                continue
            if time.monotonic() - start >= time_budget:
                break
            params2 = np.zeros(2 * M2)
            for m in range(M2):
                k = m + 1
                c = 2.0 * n * np.sum(a_dev * np.cos(2.0 * np.pi * k * j / n))
                s_val = 2.0 * n * np.sum(a_dev * np.sin(2.0 * np.pi * k * j / n))
                params2[2 * m] = c
                params2[2 * m + 1] = np.arctan2(-s_val, c) if (c != 0 or s_val != 0) else 0.0
            try:
                result2 = minimize(obj_func, params2, method='L-BFGS-B',
                                   options={'maxiter': 100, 'maxfun': 300, 'ftol': 1e-10})
                if result2.fun < best_obj:
                    best_obj = result2.fun
                    best_params = result2.x
            except Exception:
                pass

    a_opt = parametric_seq(n, best_params)
    obj_opt = compute_obj_norm(a_opt)

    # Blend with current
    best_blend_obj = best_obj
    best_result = a_opt
    for t in [1.0, 0.8, 0.6, 0.4, 0.2]:
        a_blended = (1 - t) * a_current + t * a_opt
        a_blended = np.maximum(a_blended, 0)
        s_b = float(np.sum(a_blended))
        if s_b < 1e-10:
            continue
        a_blended = a_blended / s_b
        obj_b = compute_obj_norm(a_blended)
        if obj_b < best_blend_obj:
            best_blend_obj = obj_b
            best_result = a_blended

    if verbose:
        print(f"    Parametric opt: obj={best_blend_obj:.8f} (vs {compute_obj_norm(a_current):.8f} current)")

    return best_result


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        # Cosine modulation
        freqs = rng.integers(1, 12, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.arange(n, dtype=np.float64)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x / n + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-5, 5)
        a = a / a.sum()
    elif mode == 6:
        # Alternating
        a = np.ones(n)
        a[::2] = 1.0 + rng.uniform(0, 0.5)
        a[1::2] = 1.0 - rng.uniform(0, 0.4)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def proposal_candidate(seed=42, budget_s=1500):
    """
    Main search combining enhanced CD, pair CD, LP polishing, parametric optimization,
    and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj_norm(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj_norm(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Symmetrize + Enhanced CD ===
    print(f"\n=== Phase 1: Symmetrize + Enhanced CD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 500.0)
    if p1_budget > 30:
        # Symmetrize first
        a_sym = symmetrize(best_seq)
        obj_sym = compute_obj_norm(a_sym)
        if obj_sym <= best_score:
            best_seq = a_sym
            best_score = obj_sym
            print(f"  After symmetrize: obj={best_score:.8f}")

        a_cd = coordinate_descent(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_cd = compute_obj_norm(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: Pair Coordinate Descent ===
    print(f"\n=== Phase 2: Pair CD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 150.0)
    if p2_budget > 20:
        a_pcd = pair_coordinate_descent(best_seq, time_budget=p2_budget, verbose=True, seed=seed+7)
        obj_pcd = compute_obj_norm(a_pcd)
        if obj_pcd < best_score:
            best_score = obj_pcd
            best_seq = a_pcd.copy()
            print(f"  PCD: obj={best_score:.8f} **")
        else:
            print(f"  PCD: obj={obj_pcd:.8f}")

    # === Phase 3: LP Polishing ===
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.18, 200.0)
    if p3_budget > 20:
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(p3_budget / (3 - rnd), remaining * 0.6)
            if alloc < 10:
                break
            a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True, seed=2000 + rnd)
            obj_lp = compute_obj_norm(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")

            # CD after LP
            remaining = deadline - time.monotonic()
            if remaining > 20 and obj_lp < best_score + 1e-6:
                a_cd_after = coordinate_descent(a_lp, time_budget=min(20, remaining * 0.15),
                                                 verbose=False, seed=3000 + rnd)
                obj_cd_after = compute_obj_norm(a_cd_after)
                if obj_cd_after < best_score:
                    best_score = obj_cd_after
                    best_seq = a_cd_after.copy()
                    print(f"  CD after LP {rnd}: obj={best_score:.8f} **")

    # === Phase 4: Parametric Optimization ===
    print(f"\n=== Phase 4: Parametric Optimization ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.12, 120.0)
    if p4_budget > 15:
        a_param = parametric_optimize(best_seq, time_budget=p4_budget, verbose=True)
        obj_param = compute_obj_norm(a_param)
        if obj_param < best_score:
            best_score = obj_param
            best_seq = a_param.copy()
            print(f"  Parametric: obj={best_score:.8f} **")
        else:
            print(f"  Parametric: obj={obj_param:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 100), min(1200, n_best + 100),
        max(400, n_best - 200), min(1500, n_best + 200),
        600, 700, 900, 1000, 1100
    ]))
    n_options = [n for n in n_options if abs(n - n_best) > 30]
    n_options = n_options[:5]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = symmetrize(a_try)

        alloc = min(remaining * 0.2, 50.0)
        if alloc > 15:
            a_try = coordinate_descent(a_try, time_budget=alloc, verbose=False, seed=n_try)

        obj_try = compute_obj_norm(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        mode = 1 + (r % 6)
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        a_r = symmetrize(a_r)

        alloc = min(remaining * 0.2, 50.0)
        if alloc > 15:
            a_r = coordinate_descent(a_r, time_budget=alloc, verbose=False, seed=5000 + r * 100)
            obj_r = compute_obj_norm(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Final CD
        alloc = min(remaining * 0.5, 150.0)
        a_final = coordinate_descent(best_seq, time_budget=alloc, verbose=False, seed=seed + 9999)
        obj_final = compute_obj_norm(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

        # Final LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_lp_final = lp_polish(best_seq, time_budget=min(remaining * 0.5, 40.0),
                                   verbose=False, seed=7777)
            obj_lp_final = compute_obj_norm(a_lp_final)
            if obj_lp_final < best_score:
                best_score = obj_lp_final
                best_seq = a_lp_final.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # Final symmetrization check
        a_sym_final = symmetrize(best_seq)
        obj_sym_final = compute_obj_norm(a_sym_final)
        if obj_sym_final < best_score:
            best_score = obj_sym_final
            best_seq = a_sym_final.copy()
            print(f"  Final symmetrize: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_norm(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
