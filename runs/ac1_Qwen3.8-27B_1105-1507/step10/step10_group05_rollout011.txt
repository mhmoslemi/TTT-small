Let me now design my algorithm:

**GCA Algorithm for flat convolution:**
1. Initialize a (e.g., uniform or from parent)
2. Repeat:
   a. Zero-pad a to length N >= 2n-1 (use N = next power of 2)
   b. Compute A_freq = FFT(a_padded)
   c. Modify A_freq to flatten: A_freq_new = A_freq / |A_freq| * target_spectrum
      - target_spectrum could be 1 (unit magnitude) or some other target
   d. a_new = IFFT(A_freq_new)
   e. a_new = a_new[:n] (take first n elements)
   f. a_new = max(a_new, 0) (enforce non-negativity)
   g. a_new = a_new / sum(a_new) (normalize)
   h. a = a_new
3. Return best a found (by tracking objective)

The key parameter is the target spectrum. Options:
- target = 1 (unit magnitude everywhere) - this is the standard GCA
- target = |A_freq|^alpha for some 0 < alpha < 1 (partial flattening)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Primary novel approach: Gerchberg-Saxton-Alt (GCA) spectral flattening,
which exploits the identity FFT(conv(a,a)) = |FFT(a_padded)|^2 to directly
flattens the autocorrelation via iterative time/frequency domain projections.
Combined with LP polishing, Adam-smoothed gradient descent, and multi-n search.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_nonneg_normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build convolution matrix A (2n-1 x n) where A @ g = conv(a_norm, g)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val
        except Exception:
            continue
    return None


def gca_iteration(a, target_power, clip_type='hard', normalize_freq=True):
    """
    Single GCA iteration for spectral flattening.
    
    a: input sequence (length n)
    target_power: target power spectrum (length N) or scalar
    clip_type: 'hard' (clip negatives to 0) or 'soft' (sqruare then clip)
    normalize_freq: whether to normalize magnitude in freq domain
    Returns: new sequence of length n
    """
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    # Zero-pad and go to frequency domain
    a_padded = np.zeros(N)
    a_padded[:n] = a
    A_freq = fft(a_padded)
    
    # Modify magnitude to approach target
    mag = np.abs(A_freq)
    phase = np.angle(A_freq)
    
    if np.isscalar(target_power) or (isinstance(target_power, (int, float)) and not isinstance(target_power, np.ndarray)):
        target_mag = np.full(N, float(target_power))
    else:
        target_mag = target_power
    
    # Blend: new_mag = (1-blend)*mag + blend*target_mag would be one option
    # Standard GCA: set magnitude to target
    new_mag = mag.copy()
    # Use target where mag > target (reduce peaks), keep where mag < target
    # This is the "peel-off" approach
    new_mag = np.where(mag > target_mag, target_mag, mag)
    
    if normalize_freq:
        # Alternating: also boost where below target (full GCA)
        target_mean = float(np.mean(target_mag)) if isinstance(target_power, np.ndarray) else float(target_power)
        new_mag = target_mean * np.ones(N)
        # Actually for standard GCA, just set to target
        new_mag = target_mag.copy() if isinstance(target_power, np.ndarray) else np.full(N, float(target_power))
    
    A_new = new_mag * np.exp(1j * phase)
    
    # Back to time domain
    a_new_full = ifft(A_new).real
    
    # Take first n elements
    a_new = a_new_full[:n]
    
    # Project onto non-negative with sum=1
    if clip_type == 'hard':
        a_new = np.maximum(a_new, 0)
    elif clip_type == 'soft':
        a_new = a_new ** 2  # Square to make positive
    else:
        a_new = np.maximum(a_new, 0)
    
    s = float(np.sum(a_new))
    if s < 1e-300:
        return np.ones(n) / n
    a_new = a_new / s
    
    return a_new


def gca_optimize(a, time_budget=100.0, verbose=False, seed=42):
    """
    GCA optimization with adaptive target spectrum.
    Tries multiple strategies and keeps the best result.
    """
    n = len(a)
    a = project_nonneg_normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    it = 0
    no_improve = 0
    best_it = 0
    
    # Strategy 1: Standard GCA with unit target
    while time.monotonic() - start < time_budget * 0.4:
        A_freq = fft(np.concatenate([a, np.zeros(N - n)]))
        mag = np.abs(A_freq)
        phase = np.angle(A_freq)
        
        # Target: flat spectrum
        target_mag = np.full(N, 1.0)
        
        A_new = target_mag * np.exp(1j * phase)
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-300:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            no_improve = 0
        else:
            no_improve += 1
            # Try partial update
            blend = max(0.1, 1.0 - no_improve * 0.05)
            A_blend = (1 - blend) * A_freq + blend * A_new
            a_blend = ifft(A_blend).real[:n]
            a_blend = np.maximum(a_blend, 0)
            s_blend = float(np.sum(a_blend))
            if s_blend > 1e-300:
                a_blend = a_blend / s_blend
                obj_blend = compute_obj(a_blend)
                if obj_blend < best_obj:
                    best_obj = obj_blend
                    best_a = a_blend.copy()
        
        a = a_new
        it += 1
        
        if no_improve > 200:
            break
    
    if verbose and time.monotonic() - start < time_budget:
        print(f"    GCA strategy1: obj={best_obj:.8f} after {it} iters")
    
    # Strategy 2: Adaptive target (geometric mean of spectrum)
    it = 0
    no_improve = 0
    a = best_a.copy()
    while time.monotonic() - start < time_budget * 0.7:
        A_freq = fft(np.concatenate([a, np.zeros(N - n)]))
        mag = np.abs(A_freq)
        phase = np.angle(A_freq)
        
        # Adaptive target: geometric mean
        log_mag = np.log(mag + 1e-30)
        target_mag = np.exp(np.mean(log_mag))
        
        # Smooth transition
        alpha = max(0.3, 1.0 - it * 0.01)
        new_mag = alpha * target_mag + (1 - alpha) * mag
        
        A_new = new_mag * np.exp(1j * phase)
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-300:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it += 1
        
        if no_improve > 300:
            break
    
    if verbose and time.monotonic() - start < time_budget:
        print(f"    GCA strategy2: obj={best_obj:.8f} after {it} more iters")
    
    # Strategy 3: Iterative with water-filling target
    it = 0
    no_improve = 0
    a = best_a.copy()
    while time.monotonic() - start < time_budget:
        A_freq = fft(np.concatenate([a, np.zeros(N - n)]))
        mag = np.abs(A_freq)
        phase = np.angle(A_freq)
        
        # Water-filling: reduce high, boost low
        mean_mag = float(np.mean(mag))
        target_mag = np.where(mag > mean_mag, mean_mag, mag * 1.01)
        # Ensure DC component is reasonable
        target_mag[0] = mag[0] * 0.9 + mean_mag * 0.1
        
        A_new = target_mag * np.exp(1j * phase)
        a_new = ifft(A_new).real[:n]
        
        # Soft clipping: use ReLU with small negative allowance
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-300:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it += 1
        
        if no_improve > 500:
            break
    
    if verbose:
        print(f"    GCA final: obj={best_obj:.8f}")
    
    return best_a


def gca_diverse(a, time_budget=80.0, verbose=False, seed=0):

    x = np.linspace(0, 1, N)
    targets.append(1.0 + 0.5 * x)
    targets.append(1.0 + 0.5 * (1 - x))
    # Cosine bump
    targets.append(1.0 + 0.3 * np.cos(2 * np.pi * x))
    # Random smooth
    for _ in range(3):
        freqs = rng.choice(np.arange(1, 10), size=3, replace=False)

        t = np.ones(N)
        for f, amp, ph in zip(freqs, amps, phases):
            t += amp * np.cos(2 * np.pi * f * x + ph)
        t = np.maximum(t, 0.5)
        targets.append(t)
    
    for t_idx, target in enumerate(targets):
        if time.monotonic() - start > time_budget:

        a_local = best_a.copy()

            # Normalize target to have same mean as current mag
            target_norm = target * (float(np.mean(mag)) / float(np.mean(target)))
            
            A_new = target_norm * np.exp(1j * phase)
            a_new = ifft(A_new).real[:n]
            a_new = np.maximum(a_new, 0)
            s = float(np.sum(a_new))

            it += 1

    return best_a


def lp_polish(a, time_budget=100.0, verbose=False, seed=42):
    """LP-based polishing (simplified from parent)."""
    n = len(a)
    a = project_nonneg_normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    momentum = 0.0
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Momentum
        if prev_g is not None:
            momentum = 0.7 * momentum + 0.3 * np.linalg.norm(g_lp - prev_g)
            if momentum > 1e-8:
                g_pred = g_lp + 0.2 * momentum * (g_lp - prev_g)
                g_pred = np.maximum(g_pred, 0)
                s_pred = float(np.sum(g_pred))
                if s_pred > 1e-10:
                    g_pred = g_pred / s_pred
                    obj_pred = compute_obj(g_pred)
                    if obj_pred < best_obj * 1.01:
                        g_lp = g_pred
        
        prev_g = g_lp.copy()
        it += 1
        
        # Line search
        best_t_obj = best_obj
        best_t_val = 0.0
        best_a_local = best_a.copy()
        
        t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            best_it = it
            if verbose and it % 20 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}")
        else:
            if it - best_it > 20:
                # Perturb and restart
                strength = min(0.1, 0.02 * (it - best_it))
                pert = 1.0 + rng.normal(0, strength, n)
                a = best_a * np.maximum(pert, 0)
                a = project_nonneg_normalize(a)
                prev_g = None
                momentum = 0.0
                if it - best_it > 50:
                    break
    
    return best_a


def smoothed_gradient_descent(a, time_budget=80.0, verbose=False, seed=0):
    """Adam optimizer with smooth-max approximation."""
    n = len(a)
    a = project_nonneg_normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    # Adam parameters
    lr = 5.0 / np.sqrt(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    
    it = 0
    best_it = 0
    tau = 50.0
    
    while time.monotonic() - start < time_budget:
        # Anneal tau
        progress = (time.monotonic() - start) / time_budget
        tau = 50.0 + 500.0 * progress
        
        # Compute smooth-max gradient
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Soft-max weights
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp
        
        # Gradient via FFT
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
        else:
            a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        lr *= 0.9999
        
        if it - best_it > 5000:
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n)
            best_it = it
    
    if verbose:
        print(f"    SGD final: obj={best_obj:.8f}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.05):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    x = np.arange(n)
    pert = np.zeros(n)
    n_modes = min(5, n // 20)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * x + phase)
    a_new = a * np.exp(pert)
    return project_nonneg_normalize(a_new)

    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 250.0)
    if p1_budget > 30:
        a_gca = gca_optimize(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_gca = compute_obj(a_gca)
        if obj_gca < best_score:
            best_score = obj_gca
            best_seq = a_gca.copy()
            print(f"  GCA optimize: obj={best_score:.8f} **")
        else:
            print(f"  GCA optimize: obj={obj_gca:.8f} (no improvement)")
    
    # === Phase 2: GCA diverse targets ===
    print(f"\n=== Phase 2: GCA Diverse Targets ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 150.0)
    if p2_budget > 20:
        a_div = gca_diverse(best_seq, time_budget=p2_budget, verbose=True, seed=seed+10)
        obj_div = compute_obj(a_div)
        if obj_div < best_score:
            best_score = obj_div
            best_seq = a_div.copy()
            print(f"  GCA diverse: obj={best_score:.8f} **")
        else:
            print(f"  GCA diverse: obj={obj_div:.8f}")
    
    # === Phase 3: LP polishing ===
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 250.0)
    if p3_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p3_budget, verbose=True, seed=seed+100)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP polish: obj={best_score:.8f} **")
        else:
            print(f"  LP polish: obj={obj_lp:.8f}")
    
    # === Phase 4: Smoothed gradient descent (Adam) ===
    print(f"\n=== Phase 4: Adam Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.1, 100.0)
    if p4_budget > 20:
        a_gd = smoothed_gradient_descent(best_seq, time_budget=p4_budget, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Adam GD: obj={best_score:.8f} **")
        else:
            print(f"  Adam GD: obj={obj_gd:.8f}")
    
    # === Phase 5: GCA + LP alternation ===
    print(f"\n=== Phase 5: GCA-LP Alternation ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.15, 150.0)
    if p5_budget > 40:
        a_alt = best_seq.copy()
        sub_budget = p5_budget / 3
        for i in range(3):
            if time.monotonic() > deadline - 30:
                break
            a_alt = gca_optimize(a_alt, time_budget=sub_budget * 0.5, seed=seed + 200 + i)
            a_alt = lp_polish(a_alt, time_budget=sub_budget * 0.5, seed=seed + 300 + i)
            obj_alt = compute_obj(a_alt)
            if obj_alt < best_score:
                best_score = obj_alt
                best_seq = a_alt.copy()
                print(f"  Alt round {i}: obj={best_score:.8f} **")
        print(f"  GCA-LP alternation final: obj={compute_obj(a_alt):.8f}")
    
    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    n_candidates = []
    for dn in [-200, -100, -50, 50, 100, 200, 300]:
        n_try = n_best + dn
        if 400 <= n_try <= 2000:
            n_candidates.append(n_try)
    for n_abs in [600, 700, 900, 1000, 1200, 1400]:
        if n_abs not in n_candidates and abs(n_abs - n_best) > 30:
            n_candidates.append(n_abs)
    
    n_candidates = sorted(set(n_candidates))[:6]
    print(f"  Trying n: {n_candidates}")
    
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        a_try = resample_sequence(best_seq, n_try)
        a_try = project_nonneg_normalize(a_try)
        
        # GCA on new n
        alloc = min(remaining * 0.12, 50.0)
        if alloc > 15:
            a_try = gca_optimize(a_try, time_budget=alloc * 0.6, seed=n_try)
            a_try = lp_polish(a_try, time_budget=alloc * 0.4, seed=n_try + 1)
        
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 7: Diversified restarts ===
    print(f"\n=== Phase 7: Diversified Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    
    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        
        if r == 0:
            a_r = fourier_perturb(best_seq, rng, strength=0.03)
        elif r == 1:
            a_r = fourier_perturb(best_seq, rng, strength=0.06)
        elif r == 2:
            a_r = 0.8 * best_seq + 0.2 * np.ones(n_cur) / n_cur
            a_r = project_nonneg_normalize(a_r)
        elif r == 3:
            # Sinusoidal initialization blend
            x = np.linspace(0, 2 * np.pi, n_cur)
            a_struct = 1.0 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(7 * x)
            a_struct = project_nonneg_normalize(a_struct)
            a_r = 0.6 * best_seq + 0.4 * a_struct
            a_r = project_nonneg_normalize(a_r)
        else:
            # Random Fourier
            a_r = best_seq.copy()
            N_r = 1
            while N_r < 2 * n_cur - 1:
                N_r <<= 1
            Af = fft(np.concatenate([a_r, np.zeros(N_r - n_cur)]))
            Af[np.random.default_rng(r*7).choice(N_r, size=n_cur//10, replace=False)] *= 0.5
            a_r = ifft(Af).real[:n_cur]
            a_r = project_nonneg_normalize(a_r)
        
        alloc = min(remaining * 0.12, 60.0)
        if alloc > 20:
            a_r = gca_optimize(a_r, time_budget=alloc * 0.5, seed=500 + r)
            a_r = lp_polish(a_r, time_budget=alloc * 0.5, seed=600 + r)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")
    
    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    if remaining > 40:
        alloc = min(remaining * 0.4, 100.0)
        a_f = gca_optimize(best_seq, time_budget=alloc * 0.5, verbose=True, seed=9999)
        a_f = lp_polish(a_f, time_budget=alloc * 0.5, verbose=True, seed=8888)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final: obj={best_score:.8f} **")
    
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 80.0)
        a_f = smoothed_gradient_descent(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
