"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FISTA with log-sum-exp smoothing and adaptive annealing as the primary method,
combined with LP best-response refinement, multi-length exploration, and structured restarts.
Key innovations: direct smoothed gradient optimization on the simplex (avoiding LP overhead
per step), geometric tau annealing, periodic LP "jumps" for basin switching, and
Fourier-structured initializations for diversity.
"""

import numpy as np
import math
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def smoothed_obj_and_grad(a, tau):
    """
    Compute log-sum-exp smoothed objective and gradient.
    f_tau(a) = max + (1/tau)*log(sum exp(tau*(conv - max)))
    grad = 2 * cross_correlation(p, a)[:n]
    where p = softmax(tau * (conv - max))
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    obj = max_conv + math.log(sum_exp) / tau

    # Gradient: g[m] = sum_k p[k] * a[k-m] for m=0..n-1
    # Using FFT: g = ifft(fft(p_pad) * conj(fft(a_pad)))[:n]
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    grad = 2.0 * g_full[:n]

    return obj, grad

                          n_phases=8, verbose=True, seed=42):

def solve_lp_best_response(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns optimal g or None.
    """
    n = len(a)
    if n > 1500:
        # Too large for LP, use subgradient approximation
        return None
    a_norm = a / float(np.sum(a))

    # Build convolution matrix A (2n-1) x n where A[k,j] = a_norm[j] if j<=k<j+n
    # Actually A @ g should give conv(a_norm, g)
    # conv(a, g)[k] = sum_j a[j] * g[k-j] for valid j
    # = sum_j a[k-j] * g[j] (reindexing)
    # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0

    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = a_norm[k - j]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    if verbose:
                        print(f"      LP solved with {solver}, M={float(M.value):.8f}")
                    return g_val
        except Exception as e:
            continue
    return None

def fourier_init(n, mode, rng):
    """Generate structured initial sequences."""
    x = np.arange(n)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        # Gaussian centered
        sigma = n / 4
        a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
        return a / a.sum()
    elif mode == 2:
        # Cosine bump
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
        return np.maximum(a, 0.001) / np.maximum(a, 0.001).sum()
    elif mode == 3:
        # Two bumps
        a = np.exp(-0.5 * ((x - n * 0.25) / (n * 0.1)) ** 2) + np.exp(-0.5 * ((x - n * 0.75) / (n * 0.1)) ** 2)
        return a / a.sum()
    elif mode == 4:
        # Power law
        a = 1.0 / (x + 1) ** 0.5
        return a / a.sum()
    elif mode == 5:
        # Random smooth
        a = np.exp(0.3 * rng.standard_normal(n).cumsum() / math.sqrt(n))
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation on [0,1]."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
