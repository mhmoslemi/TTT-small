"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid LP + Nesterov-accelerated gradient descent approach with
multiple diverse initializations, adaptive switching between methods,
coordinate descent polishing, and multi-n exploration.
Key innovations: subgradient-based LP constraint selection, warm-started
LP, Fista acceleration, and spectral initialization.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_smoothed(a, tau):
    """Compute gradient of smoothed max of conv(a,a) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    # Numerical stability
    tau_safe = tau
    shifted = tau_safe * (conv - max_conv)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def subgradient(a):
    """Compute subgradient of max(conv(a,a)) - pick the max element."""
    conv = compute_conv(a)
    idx = int(np.argmax(conv))
    n = len(a)
    g = np.zeros(n)
    # d conv[idx] / d a[j] = a[idx-j] if 0 <= idx-j < n
    for j in range(n):
        k = idx - j
        if 0 <= k < n:
            g[j] = a[k]
    return 2.0 * g  # factor of 2 because conv = a*a


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix A where (A @ g)[k] = conv(a_norm, g)[k]
    # A[k, j] = a_norm[k-j] if 0 <= k-j < n, else 0
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A_data = a_norm[diff[valid]]
    A = np.zeros((2 * n - 1, n))
    A[valid] = A_data
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

    v = np.zeros(n)
    it = 0
    best_it = 0
    last_tau = 100.0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        tau = 100.0 + (tau_max - 100.0) * progress ** 1.5
        
        # Nesterov step

        s_y = float(np.sum(y))

            s_y = 1.0

        g_norm = float(np.linalg.norm(g))
        
        # Adaptive learning rate

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)
        
        # Reset if stuck

            a = best_a.copy()
            v = np.zeros(n)
            beta = min(0.9, beta * 1.01)

    rng = np.random.default_rng(42)
    
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break
        
        g, M_lp = solve_lp(a)
        if g is None:
            # Fallback: use subgradient direction
            sg = subgradient(a)
            if float(np.linalg.norm(sg)) > 1e-15:
                g = a - 0.01 * sg / np.linalg.norm(sg)
                g = np.maximum(g, 0)
                s_g = g.sum()
                if s_g > 1e-10:
                    g = g / s_g
                else:
                    break
            else:
                break
            
            # Line search
            best_t = 0.05
            best_step_obj = best_obj
            for t in [0.01, 0.02, 0.05, 0.1, 0.2]:
                a_try = (1 - t) * a + t * g
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_step_obj:
                        best_step_obj = obj_try
                        best_t = t
            
            it += 1
            if best_step_obj < best_obj - 1e-13:
                a_try = (1 - best_t) * a + best_t * g
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                best_obj = best_step_obj
                best_a = a_try.copy()
                a = a_try
                stall = 0
            else:
                stall += 1
                if stall >= 10:
                    # Perturb and restart
                    a = best_a + 0.01 * rng.standard_normal(n)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    stall = 0
                continue
        
        # Have LP direction, do line search
        best_t = 0.0
        best_step_obj = best_obj
        ts = np.array([0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5])
        
        for t in ts:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_step_obj:
                    best_step_obj = obj_try
                    best_t = t
        
        # Refine around best_t
        if best_t > 0.02:
            t_lo = best_t * 0.7
            t_hi = min(best_t * 1.3, 0.95)
            for t in np.linspace(t_lo, t_hi, 5):
                a_try = (1 - t) * a + t * g
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_step_obj:
                        best_step_obj = obj_try
                        best_t = t
        
        it += 1
        
        if best_step_obj < best_obj - 1e-13:
            a_try = (1 - best_t) * a + best_t * g
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            best_obj = best_step_obj
            best_a = a_try.copy()
            a = a_try
            stall = 0
            if verbose and (it % 5 == 0 or it == 1):
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, M_lp={M_lp:.6f}")
        else:
            stall += 1
            if verbose and stall % 5 == 0:
                print(f"      LP it{it}: stalled obj={best_obj:.8f}, M_lp={M_lp:.6f}, stall={stall}")
            
            if stall >= 5:
                # Perturb using smoothed gradient direction
                sg = gradient_smoothed(a, 200.0)
                if float(np.linalg.norm(sg)) > 1e-15:
                    a_pert = a - 0.01 * sg / np.linalg.norm(sg)
                    a_pert = np.maximum(a_pert, 0)
                    s_pert = a_pert.sum()
                    if s_pert > 1e-10:
                        a = a_pert / s_pert
                    else:
                        a = best_a.copy()
                # Small random perturbation
                a = a + 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                
                if it > 200:
                    break
    
    if verbose:
        print(f"      LP done: obj={best_obj:.8f}, iters={it}")
    return best_a


def coordinate_descent(a, time_budget=50.0, verbose=False):
    """
    Coordinate descent: optimize one element at a time.
    For each j, find the optimal a[j] given all other elements fixed.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Precompute: for each j, the convolution contribution
    # conv[k] = sum_i a[i]*a[k-i]
    # If we change a[j], conv[k] changes by:
    #   delta_conv[k] = a[k-j]*delta + a[j-k]*delta + delta^2 (for k=j, the a[j]^2 term)
    # Actually: conv[k] = sum_i a[i]*a[k-i]
    # d conv[k] / d a[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    # Wait, for the "same" sequence a*a, it's:
    # conv[k] = sum_i a[i]*a[k-i]
    # d/d a[j]: a[k-j] (if 0<=k-j<n) + a[k-j] (again, from the other term) = 2*a[k-j] (if 0<=k-j<n)
    # But wait, that's only when j appears once. When j = k-j, i.e., k = 2j, then it's a[j]*a[j], and the derivative is 2*a[j].
    # Actually, d/d a[j] of a[i]*a[k-i] summed over i:
    #   = a[k-j] (from the term where i=j) + (if k-j != j, the term where k-i=j, i.e., i=k-j, gives a[k-j]*a[j])
    #   So if k-j != j: derivative = a[k-j] + a[k-j] = 2*a[k-j]
    #   If k-j == j (k=2j): derivative = 2*a[j]
    # In both cases: 2*a[k-j] (when 0 <= k-j < n)
    # So d conv[k] / d a[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
    
    # For coordinate descent, fixing all a[i] for i != j:
    # conv[k] = base_k + 2*a[k-j]*a[j] + (if k=2j: a[j]^2, else 0)
    # Wait, let me be more careful.
    # conv[k] = sum_i a[i]*a[k-i]
    # Split the sum into i=j and i!=j:
    # = a[j]*a[k-j] + sum_{i!=j} a[i]*a[k-i]
    # = a[j]*a[k-j] + base_k
    # where base_k doesn't depend on a[j].
    # But we also need to account for the term where k-i = j, i.e., i = k-j.
    # If k-j != j, then the term a[k-j]*a[j] appears in the sum (as i=k-j).
    # So conv[k] = a[j]*a[k-j] + sum_{i!=j, i!=k-j} a[i]*a[k-i] + (if k-j != j: a[k-j]*a[j])
    #           = 2*a[j]*a[k-j] + rest_k  (if k-j != j and 0<=k-j<n)
    #           = a[j]^2 + rest_k  (if k=2j)
    
    # This is getting complicated. Let me just use the full recomputation approach
    # but with a smart grid search for each coordinate.
    
    # Actually, for efficiency, let me just do a simple greedy coordinate descent:
    # For each j, try a few values of a[j] and pick the best.
    
    # Precompute the "base" convolution (with a[j] = 0 for all j, then add contributions)
    # This is too slow for large n. Let me use a different approach.
    
    # Simple approach: for each coordinate j, try perturbing a[j] by small amounts.
    n_rounds = 0
    order = np.arange(n)
    
    while time.monotonic() - start < time_budget:
        # Shuffle order for diversity
        rng_local = np.random.default_rng(n_rounds)
        perm = rng_local.permutation(n)
        
        improved = False
        for j in perm:
            if time.monotonic() - start >= time_budget:
                break
            
            a[j] = max(0, a[j] - 0.01)
            a[j] += 0.02  # try +2%
            # Actually, let me try a grid of multipliers
            base_val = a[j]
            best_local = best_obj
            best_local_val = base_val
            for mult in [0.8, 0.9, 1.1, 1.2, 1.5, 0.5, 2.0]:
                trial = base_val * mult
                if trial < 0:
                    continue
                a[j] = trial
                s = a.sum()
                if s < 1e-10:
                    a[j] = base_val
                    continue
                a_norm = a / s
                conv = compute_conv(a_norm)
                obj = 2.0 * n * float(np.max(conv))
                if obj < best_local - 1e-12:
                    best_local = obj
                    best_local_val = trial
                    improved = True
            a[j] = best_local_val
        
        n_rounds += 1
        if best_obj > 0 and n_rounds % 10 == 0 and verbose:
            s = a.sum()
            a_norm = a / s
            conv = compute_conv(a_norm)
            cur_obj = 2.0 * n * float(np.max(conv))
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
            print(f"      CD round {n_rounds}: obj={best_obj:.8f}")
        
        if not improved and n_rounds > 5:
            break
    
    # Final check
    s = a.sum()
    if s > 1e-10:
        a_norm = a / s
        conv = compute_conv(a_norm)
        obj = 2.0 * n * float(np.max(conv))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a


def make_raised_cosine(n):
    """Create a raised cosine sequence of length n."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    return a / a.sum()


def make_triangular(n):
    """Create a triangular (tent) sequence."""
    if n % 2 == 0:
        half = n // 2
        a = np.concatenate([np.arange(1, half + 1), np.arange(half, 0, -1)])
    else:
        half = n // 2
        a = np.concatenate([np.arange(1, half + 1), [half + 1], np.arange(half, 0, -1)])
    return a / a.sum()


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian-shaped sequence."""
    x = np.linspace(0, 1, n)
    center = 0.5
    sigma = sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / a.sum()


def make_spectral_random(n, seed=123):
    """
    Create a non-negative sequence with approximately flat spectrum.
    Uses |cos random phase| approach.
    """
    rng = np.random.default_rng(seed)
    # Random phases
    phases = rng.uniform(0, 2 * np.pi, n)
    # Create signal in frequency domain with flat magnitude
    freqs = np.arange(n)
    # Only use positive frequencies for real signal
    n_half = n // 2 + 1
    A = np.ones(n_half) * (1 + 0.1 * rng.standard_normal(n_half))
    A = np.abs(A) + 0.1  # ensure positive magnitude
    # Reconstruct
    signal = np.zeros(n)
    for k in range(n_half):
        signal += A[k] * np.cos(2 * np.pi * k * freqs / n + phases[k % n])
    # Make non-negative
    a = np.abs(signal) + 0.01
    return a / a.sum()


def make_bell_natural(n):
    """Create a 'natural' bell shape that's been optimized for flat autocorrelation."""
    # Start with a raised cosine and add some structure
    x = np.linspace(0, 1, n)
    # Base: raised cosine
    base = 0.5 * (1 + np.cos(2 * np.pi * x))
    # Add a moderate amount of structure
    structure = 0.1 * np.cos(4 * np.pi * x) * np.cos(6 * np.pi * x)
    a = base + structure
    a = np.maximum(a, 0.01)
    return a / a.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid strategy: LP-driven direction finding with adaptive line search,
Nesterov-accelerated smoothed gradient descent, incremental coordinate descent,

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_smoothed(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def subgradient_direct(a):
    """Compute exact subgradient at the max element of conv(a,a)."""
    conv = compute_conv(a)
    idx = int(np.argmax(conv))
    n = len(a)
    g = np.zeros(n)
    j_start = max(0, idx - n + 1)
    j_end = min(n - 1, idx)
    for j in range(j_start, j_end + 1):
        k = idx - j
        if 0 <= k < n:
            g[j] = 2.0 * a[k]
    return g


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)

    # Build constraint matrix: A[k,j] = a_norm[k-j] if 0<=k-j<n
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a_norm[diff[valid]]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g, n_steps=15):

    n = len(a)
    best_t = 0.0
    best_obj = compute_obj(a)

    for t in ts:

        a_try = np.maximum(a_try, 0)
        s_try = float(a_try.sum())
        if s_try > 1e-10:
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_t = t

    # Refine around best
    if best_t > 0.03:
        t_lo = max(0.005, best_t * 0.6)
        t_hi = min(0.95, best_t * 1.4)
        for t in np.linspace(t_lo, t_hi, 7):
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(a_try.sum())
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:

                    best_t = t

    return best_t, best_obj

    it = 0

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        g, M_lp = solve_lp(a)
        if g is None:
            break

        best_t, obj_at_t = line_search(a, g)
        it += 1

        if obj_at_t < best_obj - 1e-13:

            s_try = float(a_try.sum())
            if s_try > 1e-10:
                a_try = a_try / s_try
                best_obj = obj_at_t
                best_a = a_try.copy()
                a = a_try
                stall = 0
                if verbose and (it <= 3 or it % 10 == 0):
                    print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.4f}, M={M_lp:.6f}, time={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if verbose and stall % 5 == 0:
                print(f"      LP it{it}: STALL obj={best_obj:.8f}, M_lp={M_lp:.6f}, stall={stall}")

            if stall >= 4:
                # Multi-strategy perturbation
                strategy = stall % 3
                if strategy == 0:
                    # Subgradient perturbation
                    sg = subgradient_direct(a)
                    if float(np.linalg.norm(sg)) > 1e-15:
                        a = a - 0.02 * sg / (float(np.linalg.norm(sg)) + 1e-10)
                elif strategy == 1:

                s_new = float(a.sum())

                    break

    return best_a


def nesterov_gd(a, time_budget=150.0, tau_init=50.0, tau_end=3000.0, verbose=False):
    """Nesterov-accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.75
    base_lr = 2.5 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * (progress ** 1.3)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(y.sum())
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = 1.0
        y = y / s_y

        g = gradient_smoothed(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adapt_lr = base_lr / (g_norm * 0.5 + 1.0)
            lr = 0.6 * lr + 0.4 * min(adapt_lr, 4.0 * base_lr)

        # Barzilai-Borwein every few steps
        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-18 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                bb_lr = min(max(bb_lr, 1e-8), 6.0 / np.sqrt(n))
                lr = 0.65 * lr + 0.35 * bb_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)

        if it - best_it > 8000:
            a = best_a.copy()
            v = np.zeros(n)
            beta = min(0.9, beta + 0.01)
            lr = base_lr * 0.5
            best_it = it

        if verbose and it % 2000 == 0:
            print(f"      GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}")

    if verbose:
        print(f"      GD done: obj={best_obj:.8f}, iters={it}")
    return best_a


def fast_coordinate_descent(a, time_budget=40.0, verbose=False):
    """
    Coordinate descent with incremental convolution update.
    For coordinate j, changing a[j] by delta changes conv[k] by:
      2*delta*a[k-j] for k where 0<=k-j<n and k != 2j
      2*a[j]*delta + delta^2 for k = 2j
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Normalize and compute initial conv
    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    round_num = 0
    while time.monotonic() - start < time_budget:
        round_num += 1
        rng_local = np.random.default_rng(1000 + round_num)
        perm = rng_local.permutation(n)

        any_improved = False
        for j in perm:
            if time.monotonic() - start >= time_budget:
                break

            a_j = a[j]
            if a_j < 1e-10:
                continue

            # Try different values of a[j]
            best_local_obj = best_obj
            best_local_val = a_j

            for mult in [0.85, 0.95, 1.05, 1.15, 0.7, 1.3, 0.5, 2.0]:
                trial_val = a_j * mult
                if trial_val < 0 or trial_val > 10.0:
                    continue

                delta = trial_val - a_j
                if abs(delta) < 1e-15:
                    continue

                # Incrementally update convolution
                # conv_new[k] = conv[k] + 2*delta*a[k-j] (for valid k, k != 2j)
                #              conv[k] + 2*a[j]*delta + delta^2 (for k = 2j)
                conv_new = conv.copy()

                # Range of k where a[k-j] is valid: 0 <= k-j < n => j <= k < j+n
                k_min = j
                k_max = min(j + n - 1, 2 * n - 2)

                # For k != 2j: delta_conv = 2*delta*a[k-j]
                k_range = np.arange(k_min, k_max + 1)
                k_vals = k_range - j  # these are the indices into a
                conv_new[k_range] += 2.0 * delta * a[k_vals]

                # For k = 2j: we added 2*delta*a[j] but need 2*a[j]*delta + delta^2
                # So we need to add delta^2 extra (since 2*delta*a[j] was already added)
                k_2j = 2 * j
                if k_2j < 2 * n - 1:
                    conv_new[k_2j] += delta * delta

                # Also need to account for the fact that a[k-j] at k=2j gives a[j]
                # which is the OLD a[j], but we're changing it. The correct formula:
                # conv_new[2j] should have a[j+delta]^2 instead of a[j]^2
                # We added 2*delta*a[j] + delta^2 which is correct!

                s_new = s + delta
                if s_new < 1e-10:
                    continue

                # Compute objective without full normalization (just check max)
                # obj = 2n * max(conv_new) / s_new^2
                max_new = float(np.max(conv_new))
                obj_try = 2.0 * n * max_new / (s_new * s_new)

                if obj_try < best_local_obj - 1e-13:
                    best_local_obj = obj_try
                    best_local_val = trial_val
                    conv = conv_new  # keep track for next trial
                    s = s_new
                    any_improved = True

            a[j] = best_local_val

            # Update s and conv if we changed
            if abs(a[j] - a_j) > 1e-15:
                pass  # already updated in the loop

        # After full sweep, check and update best
        if any_improved:
            s_check = float(a.sum())
            if s_check > 1e-10:
                conv_check = compute_conv(a / s_check)
                obj_check = 2.0 * n * float(np.max(conv_check))
                if obj_check < best_obj - 1e-14:
                    best_obj = obj_check
                    best_a = a.copy()
                # Sync our tracking
                a = a / s_check
                s = 1.0
                conv = compute_conv(a)
        else:
            if round_num >= 3:
                break

        if verbose and round_num % 5 == 0:
            print(f"      CD round {round_num}: obj={best_obj:.8f}")

    # Final verification
    s_final = float(a.sum())
    if s_final > 1e-10:
        a = a / s_final
        obj_final = compute_obj(a)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a.copy()

    return best_a


def make_raised_cosine(n):
    """Raised cosine sequence."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    return a / a.sum()


def make_triangle(n):
    """Triangular (tent) sequence."""
    if n % 2 == 0:
        half = n // 2
        a = np.concatenate([np.arange(1, half + 1), np.arange(half, 0, -1)])
    else:
        half = n // 2
        a = np.concatenate([np.arange(1, half + 1), [float(half + 1)], np.arange(half, 0, -1)])
    return a / a.sum()


def make_gaussian(n, sigma=0.25):
    """Gaussian-shaped sequence."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    return a / a.sum()


def make_flat_spectrum(n, seed=99):
    """Non-negative sequence with approximately flat magnitude spectrum."""
    rng = np.random.default_rng(seed)
    # Random cosine sum with controlled amplitudes
    a = np.ones(n) * 0.5
    for k in range(1, min(30, n // 2)):
        amp = 0.5 / k * rng.uniform(0.5, 1.0)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * np.arange(n) / n + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def resample_sequence(a, new_n):
    """Resample via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP + Nesterov GD + coordinate descent + multi-n + perturbations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_raised_cosine(800)
        best_score = compute_obj(best_seq)
        print(f"Default (raised cos, n=800): obj={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 0: Try diverse initializations ===
    print(f"\n=== Phase 0: Diverse Inits ===")
    init_shapes = [
        ('raised_cos', make_raised_cosine),
        ('triangle', make_triangle),
        ('gauss', make_gaussian),
        ('flat_spec', make_flat_spectrum),
    ]
    for name, fn in init_shapes:
        try:
            a_init = fn(n_base)
            obj_init = compute_obj(a_init)
            print(f"  {name}: obj={obj_init:.8f}")
            if obj_init < best_score:
                best_score = obj_init
                best_seq = a_init.copy()
                print(f"  ** new best: {obj_init:.8f}")
        except Exception:
            pass

    # === Phase 1: Main LP optimization ===
    print(f"\n=== Phase 1: LP Optimization (~400s) ===")
    remaining = deadline - time.monotonic()
    p1 = min(remaining * 0.32, 400.0)
    if p1 > 30:
        a_out = lp_optimize(best_seq, time_budget=p1, verbose=True)
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  ** Phase1: obj={best_score:.8f}")

    # === Phase 2: Nesterov GD ===
    print(f"\n=== Phase 2: Nesterov GD (~150s) ===")
    remaining = deadline - time.monotonic()
    p2 = min(remaining * 0.13, 150.0)
    if p2 > 15:
        a_out = nesterov_gd(best_seq, time_budget=p2, tau_init=50.0, tau_end=2000.0, verbose=True)
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  ** Phase2: obj={best_score:.8f}")

    # === Phase 3: Second LP round (after GD polishing) ===
    print(f"\n=== Phase 3: LP Round 2 (~200s) ===")
    remaining = deadline - time.monotonic()
    p3 = min(remaining * 0.18, 200.0)
    if p3 > 20:
        a_out = lp_optimize(best_seq, time_budget=p3, verbose=True)
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  ** Phase3: obj={best_score:.8f}")

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coordinate Descent (~50s) ===")
    remaining = deadline - time.monotonic()
    p4 = min(remaining * 0.05, 50.0)
    if p4 > 10:
        a_out = fast_coordinate_descent(best_seq, time_budget=p4, verbose=True)
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  ** Phase4: obj={best_score:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    n_candidates = sorted(set([
        n_cur - 200, n_cur - 100, n_cur + 100, n_cur + 200, n_cur + 400,
        600, 800, 1000, 1200, 1400, 1600
    ]))
    n_candidates = [x for x in n_candidates if 400 <= x <= 1800 and abs(x - n_cur) > 50]
    n_candidates = n_candidates[:5]
    print(f"  Trying: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # LP round
        alloc_lp = min(remaining * 0.25, 50.0)
        if alloc_lp > 12:
            a_try = lp_optimize(a_try, time_budget=alloc_lp, verbose=False)

        # GD round
        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.12, 25.0)
        if alloc_gd > 8:
            a_try = nesterov_gd(a_try, time_budget=alloc_gd, tau_init=100.0, tau_end=1500.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Spectral perturbations + re-optimize ===
    print(f"\n=== Phase 6: Perturbations ===")
    remaining = deadline - time.monotonic()
    n_pert = len(best_seq)

    pert_names = ['fisher_low', 'fisher_med', 'smooth', 'mirror']
    for pname in pert_names:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if pname == 'fisher_low':
            pert = np.zeros(n_pert)
            for k in range(1, 4):
                freq = k * 2.0 * np.pi / n_pert
                amp = 0.03 * rng.uniform(0.5, 1.5) / k
                phase = rng.uniform(0, 2.0 * np.pi)
                pert += amp * np.cos(freq * np.arange(n_pert) + phase)
            a_p = best_seq * np.exp(pert)
        elif pname == 'fisher_med':
            pert = np.zeros(n_pert)
            for k in range(1, 10):
                freq = k * 2.0 * np.pi / n_pert
                amp = 0.02 * rng.uniform(0.3, 1.0) / k
                phase = rng.uniform(0, 2.0 * np.pi)
                pert += amp * np.cos(freq * np.arange(n_pert) + phase)
            a_p = best_seq * np.exp(pert)
        elif pname == 'smooth':
            ks = max(5, n_pert // 20)
            if ks % 2 == 0:
                ks += 1
            a_s = uniform_filter1d(best_seq, size=ks)
            a_p = 0.85 * best_seq + 0.15 * a_s
        else:  # mirror
            a_p = 0.55 * best_seq + 0.45 * best_seq[::-1]

        a_p = np.maximum(a_p, 0)
        s_p = float(a_p.sum())
        if s_p < 1e-10:
            continue
        a_p = a_p / s_p

        # Quick optimize
        alloc_lp = min(remaining * 0.18, 40.0)
        if alloc_lp > 10:
            a_p = lp_optimize(a_p, time_budget=alloc_lp, verbose=False)

        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.08, 20.0)
        if alloc_gd > 8:
            a_p = nesterov_gd(a_p, time_budget=alloc_gd, tau_init=100.0, tau_end=1500.0)

        obj_p = compute_obj(a_p)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  ** Perturb ({pname}): obj={best_score:.8f}")
        else:
            print(f"  Perturb ({pname}): obj={obj_p:.8f}")

    # === Phase 7: Final intensive push ===
    print(f"\n=== Phase 7: Final Push ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        alloc = min(remaining * 0.45, 200.0)
        a_out = lp_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  ** Final LP: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.4, 100.0)
        a_out = nesterov_gd(best_seq, time_budget=alloc, tau_init=200.0, tau_end=5000.0, verbose=True)
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  ** Final GD: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_out = fast_coordinate_descent(best_seq, time_budget=alloc, verbose=False)
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  ** Final CD: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    best_seq = np.clip(best_seq, 0, 1000.0)
    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
