import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_minpeak(a, time_limit=30):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    return g_val / s_g
        except Exception:
            continue
    return None

        current_obj = 2.0 * n * float(np.max(conv)) / (S * S) if S > 1e-10 else np.inf
        if current_obj < best_obj:
            best_obj = current_obj
            best_a = a.copy()

        # Randomly select blocks
        n_blocks = max(1, n // (block_size * 2))
        for _ in range(n_blocks):
            if time.monotonic() - start >= time_budget:
                break
            
            # Pick a random block start
            start_idx = rng.integers(0, n - block_size)
            block = a[start_idx:start_idx + block_size].copy()
            
            # Try small random perturbations
            best_block = block.copy()
            best_block_obj = np.inf
            
            for trial in range(20):
                pert = rng.normal(0, 0.1, block_size)
                trial_block = np.maximum(block * np.exp(pert), 0)
                a_try = a.copy()
                a_try[start_idx:start_idx + block_size] = trial_block
                S_try = float(np.sum(a_try))
                if S_try < 1e-10:
                    continue
                conv_try = compute_conv(a_try)
                obj_try = 2.0 * n * float(np.max(conv_try)) / (S_try * S_try)
                if obj_try < best_block_obj:
                    best_block_obj = obj_try
                    best_block = trial_block.copy()
            
            if best_block_obj < current_obj:
                a[start_idx:start_idx + block_size] = best_block
                conv = compute_conv(a)
                S = float(np.sum(a))
                current_obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if current_obj < best_obj:
                    best_obj = current_obj
                    best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    Block CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def spectral_refinement(a, time_budget=50.0, verbose=False):
    """
    Refine in frequency domain. The autocorrelation peak is related to
    the inverse DFT of |A(f)|^2. We want to flatten |A(f)|^2 as much as possible.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    it = 0
    while time.monotonic() - start < time_budget:
        # Compute spectrum
        A = rfft(a, 2 * n)
        power = np.abs(A) ** 2
        
        # Target: flat spectrum
        # Update in frequency domain: A_new = sign(A) * target_power
        # Then inverse FFT and project
        target = np.sqrt(np.mean(power)) * np.ones_like(power)
        A_new = A / np.abs(A + 1e-10) * np.sqrt(power * 0.5 + target * 0.5)
        
        a_new = irfft(A_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
        
        # Also try random spectral perturbations
        if it % 10 == 5:
            A = rfft(a, 2 * n)
            phase_pert = rng.normal(0, 0.1, len(A))
            A_pert = A * np.exp(1j * phase_pert)
            a_pert = irfft(A_pert, 2 * n)[:n]
            a_pert = np.maximum(a_pert, 0)
            s_pert = float(np.sum(a_pert))
            if s_pert > 1e-10:
                a_pert = a_pert / s_pert
                obj_pert = 2.0 * n * float(np.max(compute_conv(a_pert)))
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    a = a_pert
        
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x**2)
        a = a / a.sum()
    elif mode == 4:
        # Fourier-based
        freqs = rng.integers(1, 10, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.uniform(0.1, 0.5, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Cosine taper
        x = np.arange(n) / (n - 1)
        a = 0.5 * (1 + np.cos(np.pi * (2 * x - 1)))
        a = a / a.sum()
    elif mode == 7:
        # Random walk projected to positive
        w = rng.normal(0, 1, n).cumsum()
        a = w - w.min() + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def lp_polish_iterative(a, time_budget=60.0, verbose=False, seed=42):
    """
    Iterative LP polishing with multiple blending strategies and perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # Solve LP
        g_lp = solve_lp_minpeak(a, time_limit=min(remaining * 0.3, 20))
        if g_lp is None:
            stall += 1
            if stall > 5:
                break
            # Perturb and retry
            pert = rng.normal(0, 0.02, n)
            a = a * np.exp(pert)
            a = np.maximum(a, 0)
            a = a / a.sum()
            continue

        # Try multiple blending factors
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            pert = rng.normal(0, 0.015, n)
            a = a * np.exp(pert)
            a = np.maximum(a, 0)
            a = a / a.sum()
            if stall > 8:
                break

    return best_a

import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_minpeak(a, time_limit=25):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    return g_val / s_g
        except Exception:
            continue
    return None


def eval_1d_subproblem(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i):
    """Evaluate 1D subproblem objective at point x."""
    if len(a_ki_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom


def fast_coordinate_descent(a, time_budget=300.0, verbose=True, seed=42, anneal_temp=0.0):
    """
    Fast coordinate descent with golden section on exact 1D subproblem.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        sweep += 1

        conv = compute_conv(a)
        S = float(np.sum(a))
        current_obj = 2.0 * n * float(np.max(conv)) / (S * S) if S > 1e-10 else np.inf

        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10 and anneal_temp > 0:
                if rng.random() < math.exp(-no_improve * 0.15 * anneal_temp):
                    no_improve = 0
                elif no_improve > 25:
                    pert = rng.normal(0, 0.01, n)
                    a = a * np.exp(pert)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    no_improve = 0

        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-15:
                continue

            k_min = i
            k_max = min(2 * n - 2, i + n - 1)

            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            quad_idx = np.where(ks == 2 * i)
            if len(quad_idx) > 0:
                quad_pos = quad_idx[0]
                c_2i = c_k[quad_pos]
                lin_mask = np.ones(len(ks), dtype=bool)
                lin_mask[quad_pos] = False
            else:
                c_2i = 0.0
                lin_mask = np.ones(len(ks), dtype=bool)

            ks_lin = ks[lin_mask]
            a_ki_lin = a_ki[lin_mask]
            c_k_lin = c_k[lin_mask]

            scale = max(v, S / n, 1e-6)
            candidates = [0.0, v * 0.25, v * 0.5, v * 0.75, v, v * 1.25,
                         v * 1.5, v * 2.0, S_other * 0.5, S_other, S_other + v]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = np.inf
            for x in candidates:
                f_val = eval_1d_subproblem(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                if f_val < best_f:
                    best_f = f_val
                    best_x = x

            x_lo = max(0.0, best_x - scale * 1.5)
            x_hi = best_x + scale * 2.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = eval_1d_subproblem(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            fd = eval_1d_subproblem(d_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)

            for _ in range(45):
                if x_hi - x_lo < 1e-13 * max(1.0, abs(x_hi)):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = eval_1d_subproblem(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = eval_1d_subproblem(d_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= 2 * i <= 2 * n - 2:
                conv[2 * i] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def parametric_sequence(n, params, family):
    """Generate a sequence from a parametric family."""
    x = np.arange(n) / max(n - 1, 1)
    if family == 0:
        # Polynomial: 1 + p1*(2x-1) + p2*(2x-1)^2 + p3*(2x-1)^3 + p4*(2x-1)^4
        t = 2 * x - 1
        a = 1.0 + params[0] * t + params[1] * t**2 + params[2] * t**3 + params[3] * t**4
    elif family == 1:
        # Cosine: 1 + sum pi*cos(2*pi*fi*x + phi)
        a = np.ones(n)
        for k in range(min(4, len(params) // 2)):
            freq = params[2 * k]
            amp = params[2 * k + 1]
            a = a + amp * np.cos(2 * np.pi * freq * x)
    elif family == 2:
        # Gaussian mixture
        a = np.zeros(n)
        for k in range(min(3, len(params) // 2)):
            center = params[2 * k]
            width = max(params[2 * k + 1], 0.01)
            a = a + np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif family == 3:
        # Power law with offset
        a = (x + params[0]) ** params[1] + (1 - x + params[0]) ** params[2] + 0.01
    elif family == 4:
        # Sinc-like
        a = np.sinc(params[0] * (x - 0.5)) ** 2 + 0.01
    elif family == 5:
        # B-spline approximation
        a = np.ones(n)
        for k in range(min(3, len(params) // 2)):
            c = params[2 * k]
            w = max(params[2 * k + 1], 0.02)
            a = a + np.exp(-((x - c) ** 2) / (2 * w ** 2))
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def optimize_parametric(n, family, time_budget=30.0, rng=None):
    """Optimize parameters of a parametric family."""
    if rng is None:
        rng = np.random.default_rng()

    n_params = 6
    best_obj = np.inf
    best_a = np.ones(n) / n

    for trial in range(15):
        if time.monotonic() - time_budget > 0:
            break
        if family == 0:
            params = rng.uniform(-0.8, 0.8, n_params)[:4]
        elif family == 1:
            freqs = rng.uniform(0.5, 5.0, 3)
            amps = rng.uniform(-0.5, 0.5, 3)
            params = np.concatenate([freqs, amps])
        elif family == 2:
            centers = rng.uniform(0.1, 0.9, 3)
            widths = rng.uniform(0.05, 0.3, 3)
            params = np.concatenate([centers, widths])
        elif family == 3:
            params = np.array([rng.uniform(0.01, 0.5), rng.uniform(-3, 3), rng.uniform(-3, 3)])
        elif family == 4:
            params = np.array([rng.uniform(2, 20)])
        else:
            centers = rng.uniform(0, 1, 3)
            widths = rng.uniform(0.05, 0.4, 3)
            params = np.concatenate([centers, widths])

        a = parametric_sequence(n, params, family)
        obj = compute_obj_normalized(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def lp_polish_iterative(a, time_budget=60.0, verbose=False, seed=42):
    """
    Iterative LP polishing with multiple blending strategies and perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        g_lp = solve_lp_minpeak(a, time_limit=min(remaining * 0.4, 20))
        if g_lp is None:
            stall += 1
            if stall > 6:
                break
            pert = rng.normal(0, 0.02, n)
            a = a * np.exp(pert)
            a = np.maximum(a, 0)
            a = a / a.sum()
            continue

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            pert = rng.normal(0, 0.015, n)
            a = a * np.exp(pert)
            a = np.maximum(a, 0)
            a = a / a.sum()
            if stall > 10:
                break

    return best_a


def subgradient_step(a, tau):
    """One subgradient step using smoothed max."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))

    shifted = conv - max_conv
    exp_vals = np.exp(tau * shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp > 1e-300:
        p = exp_vals / sum_exp
    else:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad_full = ifft(P * np.conj(A)).real
    grad = 2.0 * grad_full[:n]

    S = float(np.sum(a))
    if S < 1e-10:
        return a
    obj_grad = (2.0 * n / (S * S)) * (grad - 2.0 * max_conv * a / S)
    return obj_grad


def subgradient_descent(a, time_budget=100.0, verbose=False, seed=42):
    """Subgradient descent with momentum and adaptive smoothing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.85
    base_lr = 3.0 / np.sqrt(n)
    lr = base_lr

    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = 20.0 + 600.0 * progress * progress

        grad = subgradient_step(a, tau)
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break

        lr = base_lr / (g_norm / base_lr + 1.0)
        lr = min(lr, 0.3)

        momentum = beta * momentum + (1 - beta) * grad
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new

        it += 1
        if verbose and it % 500 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.arange(n) / max(n - 1, 1)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        t = 2 * x - 1
        a = 1.0 + 0.3 * t**2 - 0.2 * t**4
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        t = 2 * x - 1
        a = 1.0 + 0.5 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-((x - 0.25) ** 2) / 0.02) + np.exp(-((x - 0.75) ** 2) / 0.02) + 0.1
        a = a / a.sum()
    elif mode == 5:
        a = (x + 0.1) ** 1.5 + (1 - x + 0.1)
