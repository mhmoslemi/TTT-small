"""
Optimizes step function coefficients using L-BFGS-B with log-sum-exp smoothing
(continuation method), combined with exact coordinate descent and multi-n exploration.
Key innovation: FFT-based gradient computation for the smoothed objective enables
very fast L-BFGS-B optimization, with a tau continuation schedule for global convergence.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute true objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_objective_and_grad(a, tau):
    """
    Compute the log-sum-exp smoothed objective and its gradient.
    f_tau(a) = 2n * (1/tau) * log(sum_k exp(tau * R[k] / S^2))
    where R = conv(a,a), S = sum(a).
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return 1e30, np.zeros(n)
    
    R = compute_conv(a)  # length 2n-1
    x = R / (S * S)  # x_k = R[k] / S^2
    max_x = np.max(x)
    
    # Numerically stable log-sum-exp
    shifted = tau * (x - max_x)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    
    if sum_exp < 1e-300:
        # Fallback: uniform p
        p = np.ones(len(x)) / len(x)
        log_sum_exp = np.log(len(x))
    else:
        p = exp_vals / sum_exp
        log_sum_exp = max_x * tau + np.log(sum_exp)  # tau * max_x + log(sum_exp)
        # Actually: log(sum exp(tau*x)) = tau*max_x + log(sum exp(tau*(x-max_x)))
        log_sum_exp = tau * max_x + np.log(sum_exp)
    
    f = 2.0 * n * log_sum_exp / tau
    
    # Gradient: (4n/S^3) * (S * C - PR * 1)
    # where C[i] = sum_j p[j+i] * a[j] (cross-correlation of a and p)
    # and PR = sum_k p[k] * R[k]
    
    # Compute C via FFT
    N = 1
    while N < 3 * n:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    C_full = ifft(A * np.conj(P)).real
    C = C_full[:n]
    
    PR = float(np.dot(p, R))
    
    grad = (4.0 * n / (S**3)) * (S * C - PR * np.ones(n))
    
    return f, grad


def lbfgsb_smoothed(a, tau_schedule, time_budget, verbose=True, seed=42):
    """
    Run L-BFGS-B with continuation on tau.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in tau_schedule:
        if time.monotonic() - start >= time_budget:
            break
        
        def func(x):
            val, _ = smoothed_objective_and_grad(x, tau)
            return val
        
        def jac(x):
            _, g = smoothed_objective_and_grad(x, tau)
            return g
        
        try:
            result = minimize(
                func, a, jac=jac, method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options={'maxiter': 500, 'gtol': 1e-8, 'maxls': 50,
                         'maxfun': 2000}
            )
            a_new = np.maximum(result.x, 0)
            S_new = float(np.sum(a_new))
            if S_new > 1e-10:
                a_new = a_new / S_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                a = a_new
        except Exception as e:
            if verbose:
                print(f"    L-BFGS-B failed at tau={tau}: {e}")
            break
        
        if verbose:
            elapsed = time.monotonic() - start
            print(f"    tau={tau}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """
    Exact coordinate descent with golden section search on 1D subproblems.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 3 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks]

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Quick scan
            scale = max(v, S / n, 1e-6)
            candidates = [0.0, v * 0.3, v * 0.5, v, v * 1.5, v * 2.0, v * 3.0]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            # Evaluate at current and candidates
            def f_1d(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(30):
                if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-15:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        # Cosine window
        a = np.cos(np.pi * np.arange(n) / (2 * n)) ** 2
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Ramp
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Sinc-like
        x = np.arange(n) - n / 2
        a = np.sinc(x / (n / 4))
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / a.sum()
    elif mode == 7:
        # Multi-frequency
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [2, 3, 5]:
            a += 0.3 * np.cos(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def subgradient_descent(a, time_budget=100.0, verbose=False, seed=42):
    """
    Subgradient/ smoothed gradient descent with Polyak-like step size
    and Nesterov momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Adaptive tau schedule
    tau = 20.0
    tau_growth = 1.02
    beta = 0.7  # momentum
    v = np.zeros(n)
    lr = 5.0 / np.sqrt(n)

    it = 0
    stall_count = 0
    stall_best = best_obj

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = 1.0
        else:
            y = y / s_y

        # Compute smoothed gradient
        _, grad = smoothed_objective_and_grad(y, tau)
        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            grad = grad / g_norm

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall_count = 0
            stall_best = best_obj
        else:
            stall_count += 1
            if stall_count % 50 == 0:
                # Increase tau (make gradient sharper)
                tau = min(tau * 1.5, 2000.0)
                # Reduce learning rate
                lr *= 0.9
                # Reset momentum if stuck too long
                if stall_count > 200:
                    v = np.zeros(n)
                    # Small perturbation
                    pert = rng.normal(0, 0.01, n)
                    a_new = a_new * np.exp(pert)
                    a_new = np.maximum(a_new, 0)
                    a_new = a_new / a_new.sum()
                    stall_count = 0

        a = a_new
        it += 1

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.1f}s")

    return best_a


def lp_polish(a, time_budget=60.0, verbose=False, seed=42):
    """LP-based polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    def solve_lp_minpeak(a_cur):
        n = len(a_cur)
        a_norm = a_cur / np.sum(a_cur)
        # Build convolution matrix: A @ g = conv(a, g)
        # A is (2n-1) x n, A[k, j] = a[k-j] if 0<=k-j<n
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            A[j:j + r, j] = a_norm[:r]

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        return g_val / s_g
            except Exception:
                continue
        return None

    stall = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 2:
                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if time.monotonic() - start > time_budget * 0.85:
                    break

    return best_a
