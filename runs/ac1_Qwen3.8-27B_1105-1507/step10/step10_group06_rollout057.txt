def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v.copy()
    if n == 1:
        return np.array([max(0.0, v[0]) if v[0] != 0 else 1.0])

    m = 2 * n - 1

    # Variables: [g_0, ..., g_{n-1}, M]
    # Minimize: [0, ..., 0, 1] (minimize M)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Constraints: A@g - M <= 0 for each k
    # => A @ g + (-1) * M <= 0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)

    # sum(g) = 1 => -sum(g) <= 0 and sum(g) <= 1
    # Actually use bounds and equality
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g_val = np.maximum(res.x[:n], 0)
            M_val = res.x[n]
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(M_val)
    except Exception:
        pass
    return None, None

def lp_restart(a_init, time_budget, max_iters=200, verbose=False, seed=42):
    """
    LP-based alternating with aggressive blending.
    At each step, solve LP for best response, then blend with large t.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(max_iters):
        if time.monotonic() - start >= time_budget:
            break

        g_lp, M_lp = solve_lp_fast(a)
        if g_lp is None:
            break

        # Try multiple blending factors
        candidate_ts = [0.3, 0.5, 0.7, 1.0, 0.1, 0.2]
        rng.shuffle(candidate_ts)

        best_local_obj = best_obj
        best_local_a = best_a.copy()

        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            if verbose and it % 20 == 0:
                print(f"    LP restart {it}: obj={best_obj:.10f}")
        else:
            # Perturb to escape
            a = best_a + 0.01 * rng.standard_normal(n) * best_a
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s

    return best_a

def generate_initializations(n, seed=42):
    """Generate diverse initializations for sequence of length n."""
    rng = np.random.default_rng(seed)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Arc-sine distribution
    i = np.arange(n).astype(np.float64)
    x = i / max(n - 1, 1)
    x = np.clip(x, 1e-6, 1 - 1e-6)
    arc = 1.0 / np.sqrt(x * (1 - x))
    inits.append(arc / arc.sum())

    # Sinusoidal (various frequencies)
    for freq in [1, 2, 3, 5, 7, 10, 15]:
        amp = 0.3
        sig = 1.0 + amp * np.cos(2 * np.pi * freq * i / n)
        sig = np.maximum(sig, 0.01)
        inits.append(sig / sig.sum())

    # Gaussian centered
    center = (n - 1) / 2.0
    for sigma_frac in [0.2, 0.3, 0.4, 0.6]:
        gauss = np.exp(-((i - center) ** 2) / (2 * (sigma_frac * n / 2) ** 2))
        inits.append(gauss / gauss.sum())

    # U-shaped (inverse Gaussian)
    for alpha in [0.5, 1.0, 2.0]:
        u_shape = (1.0 + alpha * x ** 2) / (1.0 + alpha * (1 - x) ** 2)
        u_shape = np.maximum(u_shape, 0.01)
        inits.append(u_shape / u_shape.sum())

    # Triangular
    tri = 1.0 - np.abs(2 * x - 1)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    # Random smooth
    for _ in range(3):
        coarse = rng.standard_normal(max(5, n // 50))
        coarse = np.maximum(coarse, 0) + 0.5
        smooth = np.interp(i, np.linspace(0, n - 1, len(coarse)), coarse)
        smooth = np.maximum(smooth, 0.01)
        inits.append(smooth / smooth.sum())

    return inits

def coordinate_refinement(a, time_budget):
    """
    Coordinate descent: for each coordinate j, find optimal value by
    1D optimization (the objective is max of linear functions in a_j
    when other coordinates are fixed, after normalization).
    Uses golden section search.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    def obj_with_j(a_base, j, val):
        trial = a_base.copy()
        trial[j] = val
        s = float(np.sum(trial))
        if s < 1e-10:
            return np.inf
        trial = trial / s
        return compute_obj(trial)

    order = list(range(n))
    rounds = 0
    while time.monotonic() - start < time_budget:
        rounds += 1
        for j in order:
            if time.monotonic() - start >= time_budget:
                break
            # Golden section search for optimal a[j]
            base = a[j]
            s_rest = float(np.sum(a) - base)
            if s_rest < 1e-10:
                continue

            # Search range: [0, 5*base + s_rest*0.1]
            lo = 0.0
            hi = max(5.0 * base, 0.1 * s_rest / n) + 0.01

            # Golden section
            gr = (np.sqrt(5) - 1) / 2
            a1 = hi - gr * (hi - lo)
            a2 = lo + gr * (hi - lo)
            f1 = obj_with_j(a, j, a1)
            f2 = obj_with_j(a, j, a2)

            for _ in range(20):
                if time.monotonic() - start >= time_budget:
                    break
                if f1 < f2:
                    hi = a2
                    a2 = a1
                    f2 = f1
                    a1 = hi - gr * (hi - lo)
                    f1 = obj_with_j(a, j, a1)
                else:
                    lo = a1
                    a1 = a2
                    f1 = f2
                    a2 = lo + gr * (hi - lo)
                    f2 = obj_with_j(a, j, a2)

            best_j_val = (lo + hi) / 2
            new_obj = obj_with_j(a, j, best_j_val)
            if new_obj < best_obj:
                a[j] = best_j_val
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_obj = new_obj

        # Shuffle order each round
        rng.shuffle(order)

    return a

def resample_to_length(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, kind='cubic')
    return np.maximum(a_new, 0)

def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-start FISTA with adaptive smoothing, subgradient escape,
    LP restarts, coordinate refinement, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        a_parent = np.asarray(height_sequence_1, dtype=np.float64)
        a_parent = np.maximum(a_parent, 0)
        if a_parent.size == 0 or a_parent.sum() < 0.01:
            raise ValueError("Invalid parent")
        a_parent = a_parent / a_parent.sum()
        n_parent = a_parent.size
        obj_parent = compute_obj(a_parent)
        print(f"Parent: n={n_parent}, obj={obj_parent:.10f}")
    except Exception:
        a_parent = None
        n_parent = 800
        obj_parent = np.inf
        print("No valid parent, using initialization from scratch.")

    best_a = None
    best_obj = np.inf
    global_start = start_time

    # Phase 1: Multi-start with diverse initializations at parent length
    print("\n=== Phase 1: Multi-start FISTA ===")
    p1_deadline = start_time + budget_s * 0.45
    n = n_parent

    inits = generate_initializations(n, seed=seed)
    # Add parent as first initialization
    if a_parent is not None and len(a_parent) == n:
        inits.insert(0, a_parent.copy())

    # Allocate time per initialization
    per_init_time = max(10, (p1_deadline - time.monotonic()) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() >= p1_deadline:
            break
        remaining_for_init = min(per_init_time, p1_deadline - time.monotonic())
        if remaining_for_init < 5:
            break

        a_try = fista_adaptive(init.copy(), time_budget=remaining_for_init,
                               tau_init=5.0, tau_growth=1.008, tau_max=8000.0,
                               subgrad_every=150, seed=seed + idx)
        obj_try = compute_obj(a_try)
        tag = "parent" if idx == 0 and a_parent is not None else f"init{idx}"
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  {tag}: obj={obj_try:.10f} **")
        else:
            print(f"  {tag}: obj={obj_try:.10f}")

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    print(f"  Phase 1 best: obj={best_obj:.10f}")

    # Phase 2: Subgradient escape + FISTA polish
    print("\n=== Phase 2: Subgradient Escape ===")
    p2_deadline = start_time + budget_s * 0.60
    remaining = p2_deadline - time.monotonic()
    if remaining > 20:
        # Use subgradient to potentially escape
        a_sg = subgradient_method(best_a.copy(), time_budget=min(remaining * 0.4, 120.0),
                                  step_scale=0.05, seed=seed + 100)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            print(f"  Subgradient: obj={best_obj:.10f} **")
        else:
            print(f"  Subgradient: obj={obj_sg:.10f}")

        # FISTA polish on the result
        remaining = p2_deadline - time.monotonic()
        if remaining > 10:
            a_polish = fista_adaptive(best_a.copy(), time_budget=remaining,
                                      tau_init=50.0, tau_growth=1.003, tau_max=5000.0,
                                      subgrad_every=0, seed=seed + 200)
            obj_polish = compute_obj(a_polish)
            if obj_polish < best_obj:
                best_obj = obj_polish
                best_a = a_polish.copy()
                print(f"  FISTA polish: obj={best_obj:.10f} **")
            else:
                print(f"  FISTA polish: obj={obj_polish:.10f}")

    # Phase 3: LP-based refinement (different from before - more aggressive)
    print("\n=== Phase 3: LP Refinement ===")
    p3_deadline = start_time + budget_s * 0.75
    remaining = p3_deadline - time.monotonic()
    if remaining > 30:
        a_lp = lp_restart(best_a.copy(), time_budget=min(remaining * 0.6, 200.0),
                          max_iters=100, verbose=True, seed=seed + 300)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            print(f"  LP: obj={best_obj:.10f} **")
        else:
            print(f"  LP: obj={obj_lp:.10f}")

        # FISTA after LP
        remaining = p3_deadline - time.monotonic()
        if remaining > 15:
            a_after = fista_adaptive(best_a.copy(), time_budget=remaining,
                                     tau_init=100.0, tau_growth=1.002, tau_max=3000.0,
                                     subgrad_every=0, seed=seed + 400)
            obj_after = compute_obj(a_after)
            if obj_after < best_obj:
                best_obj = obj_after
                best_a = a_after.copy()
                print(f"  FISTA after LP: obj={best_obj:.10f} **")

    # Phase 4: Coordinate refinement
    print("\n=== Phase 4: Coordinate Refinement ===")
    p4_deadline = start_time + budget_s * 0.88
    remaining = p4_deadline - time.monotonic()
    if remaining > 15:
        n_best = len(best_a)
        # Only do coordinate descent if n is manageable
        if n_best <= 1500:
            a_coord = coordinate_refinement(best_a.copy(), time_budget=remaining)
            obj_coord = compute_obj(a_coord)
            if obj_coord < best_obj:
                best_obj = obj_coord
                best_a = a_coord.copy()
                print(f"  Coordinate: obj={best_obj:.10f} **")
            else:
                print(f"  Coordinate: obj={obj_coord:.10f}")

    # Phase 5: Multi-length exploration
    print("\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_best = len(best_a)
        n_options = sorted(set([
            n_best - 50, n_best + 50,
            n_best - 100, n_best + 100,
            n_best - 200, n_best + 200,
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2000]

        for n_try in n_options[:4]:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            a_try = resample_to_length(best_a, n_try)
            alloc = min(remaining * 0.35, 150.0)
            if alloc < 15:
                break

            a_try = fista_adaptive(a_try, time_budget=alloc * 0.7,
                                   tau_init=10.0, tau_growth=1.006, tau_max=5000.0,
                                   subgrad_every=200, seed=n_try)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_try = fista_adaptive(a_try, time_budget=min(remaining * 0.4, 40.0),
                                       tau_init=50.0, tau_growth=1.003, tau_max=3000.0,
                                       subgrad_every=0, seed=n_try + 7)

            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                print(f"  ** n={n_try}: obj={best_obj:.10f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.10f}")

    # Phase 6: Final aggressive polish
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Very high tau for final sharpening
        a_final = fista_adaptive(best_a.copy(), time_budget=min(remaining * 0.6, 60.0),
                                 tau_init=200.0, tau_growth=1.001, tau_max=2000.0,
                                 subgrad_every=0, seed=seed + 9999)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final: obj={best_obj:.10f} **")
        else:
            print(f"  Final: obj={obj_final:.10f}")

        # One more LP shot if time permits
        remaining = deadline - time.monotonic()
        if remaining > 20 and len(best_a) <= 1200:
            a_lp2 = lp_restart(best_a.copy(), time_budget=min(remaining * 0.5, 40.0),
                               max_iters=50, verbose=False, seed=seed + 55555)
            obj_lp2 = compute_obj(a_lp2)
            if obj_lp2 < best_obj:
                best_obj = obj_lp2
                best_a = a_lp2.copy()
                print(f"  Final LP: obj={best_obj:.10f} **")

    # Final validation
    if best_a is None:
        best_a = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
