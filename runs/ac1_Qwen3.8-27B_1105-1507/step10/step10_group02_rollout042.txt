import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def objective_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix_sparse(a):
    """Build sparse convolution matrix A such that A @ g = conv(a, g)."""
    import scipy.sparse as sp
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for i in range(max(0, k - n + 1), min(k + 1, n)):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = sp.coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A.tocsr()

def solve_lp(a):
    """Solve the LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix_sparse(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.MOSEK]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

    max_conv = np.max(conv)
    p = np.exp(tau * (conv - max_conv))
    p_sum = p.sum()
    if p_sum < 1e-300:
        # Fallback: hard max
        p = np.zeros(2*n-1)
        p[np.argmax(conv)] = 1.0
    else:
        p = p / p_sum
    
    # h[i] = sum_j a[j] * p[i+j]
    # = conv[a_rev, p][n-1 + i]
    a_rev = a[::-1]
    N = 1
    while N < n + (2*n-1) - 1 + 1:
        N <<= 1
    FA = fft(a_rev, N)
    FP = fft(p, N)
    h_full = ifft(FA * FP).real
    h = h_full[n-1 : 2*n-1]
    
    return 2.0 * h, conv

    #      sum(g) = 1
    #      g >= 0
    #      M free (unconstrained)
