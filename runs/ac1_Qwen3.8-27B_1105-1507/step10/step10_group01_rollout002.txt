"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of LP-driven optimization, mirror descent (multiplicative updates),
and smoothed gradient descent with aggressive multi-n exploration.
Key improvements: mirror descent for better local minima escape, more aggressive

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a/g_norm, g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_s = float(np.sum(a))
    if a_s < 1e-10:
        return None, None
    a_norm = a / a_s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g, n_evals=30):
    """Line search for best t in (0,1) minimizing obj((1-t)*a + t*g)."""
    n = len(a)
    ts = np.linspace(0.01, 0.95, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        objs[i] = compute_obj(a_try)

    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]

    # Parabolic refinement
    if 0 < best_idx < n_evals - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.01, 0.95))
            a_par = np.maximum((1 - t_par) * a + t_par * g, 0)
            f_par = compute_obj(a_par)
            if f_par < best_obj:
                best_t = t_par
                best_obj = f_par

    return best_t, best_obj

def mirror_descent(a, time_budget=150.0, tau_init=50.0, tau_end=1500.0, verbose=False):
    """
    Mirror descent (exponential weights) on smoothed max autocorrelation.
    Multiplicative update: a_new[i] = a[i] * exp(-eta * grad[i])
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget - 1.0:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.2

        # Compute smoothed gradient
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = exp_vals / sum_exp

        N_fft = 1
        while N_fft < 3 * n:
            N_fft <<= 1
        P = fft(p, N_fft)
        A = fft(a, N_fft)
        grad = 2.0 * ifft(P * np.conj(A)).real[:n]

        # Adaptive step size for mirror descent
        grad_norm = float(np.linalg.norm(grad))
        base_eta = 1.5 / np.sqrt(n)
        eta = base_eta / (grad_norm + 1e-10)
        eta = min(eta, 3.0 / np.sqrt(n))
        eta = max(eta, 1e-8)

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        # Restart if stalled
        if it - best_it > 8000:
            a = best_a.copy()
            eta = base_eta * 0.3
            best_it = it

        if verbose and it % 500 == 0:
            print(f"    MIR it{it}: obj={best_obj:.8f}, tau={tau:.0f}")

    if verbose:
        print(f"    MIR done: obj={best_obj:.8f}, iters={it}")
    return best_a

    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget - 1.0:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = exp_vals / sum_exp

        N_fft = 1
        while N_fft < 3 * n:
            N_fft <<= 1
        P = fft(p, N_fft)
        Y = fft(y, N_fft)
        g = 2.0 * ifft(P * np.conj(Y)).real[:n]

        # Adaptive LR
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.3 * base_lr), 3.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adapt

        # BB step
        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 4.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)

            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        if it - best_it > 12000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.4
            best_it = it

    if verbose:
        print(f"    GD: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 6):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()


def symmetric_init(n):
    """Create a symmetric initial sequence."""
    half = n // 2
    x = np.linspace(0, 1, half)
    # Raised cosine
    a_half = 0.5 * (1.0 + np.cos(np.pi * x))
    if n % 2 == 0:
        a = np.concatenate([a_half, a_half[::-1]])
    else:
        a = np.concatenate([a_half, a_half[-2::-1]])
    return a / a.sum()


def make_symmetric(a):
    """Force sequence to be symmetric."""
    n = len(a)
    return (a + a[::-1]) / 2.0


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP-driven + mirror descent + smoothed GD with multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = symmetric_init(1000)
        best_score = compute_obj(best_seq)
        print(f"Default init (n=1000): obj={best_score:.8f}")

    # === Phase 1: LP-Driven on parent ===
    print(f"\n=== Phase 1: LP-Driven on n={len(best_seq)} ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 350.0)
    if p1_budget > 30:
        a_p1 = lp_optimize(best_seq, time_budget=p1_budget, verbose=True)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP: obj={obj_p1:.8f} (no improvement)")

    # === Phase 2: Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 150.0)
    if p2_budget > 20:
        a_p2 = mirror_descent(best_seq, time_budget=p2_budget, tau_init=50.0, tau_end=1200.0, verbose=True)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_score:
            best_score = obj_p2
            best_seq = a_p2.copy()
            print(f"  Phase2 MIR: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 MIR: obj={obj_p2:.8f} (no improvement)")

    # === Phase 3: Second LP round ===
    print(f"\n=== Phase 3: Second LP ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.18, 250.0)
    if p3_budget > 20:
        a_p3 = lp_optimize(best_seq, time_budget=p3_budget, verbose=True)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_score:
            best_score = obj_p3
            best_seq = a_p3.copy()
            print(f"  Phase3 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 LP: obj={obj_p3:.8f} (no improvement)")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Try a range of n values, prioritizing larger n
    n_candidates = []
    for n_try in [800, 900, 1000, 1100, 1200, 1300, 1400, 1500, 1600]:
        if abs(n_try - n_best) > 30:
            n_candidates.append(n_try)

    # Sort by distance from best (try nearby first)
    n_candidates.sort(key=lambda x: abs(x - n_best))
    n_candidates = n_candidates[:7]
    print(f"  Trying n: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_seq(best_seq, n_try)

        # LP optimization
        alloc_lp = min(remaining * 0.25, 70.0)
        if alloc_lp > 15:
            a_try = lp_optimize(a_try, time_budget=alloc_lp, verbose=False)

        # Mirror descent
        remaining = deadline - time.monotonic()
        alloc_mir = min(remaining * 0.08, 30.0)
        if alloc_mir > 8:
            a_try = mirror_descent(a_try, time_budget=alloc_mir, tau_init=80.0, tau_end=800.0)

        # Quick GD
        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.06, 20.0)
        if alloc_gd > 5:
            a_try = smoothed_gd(a_try, time_budget=alloc_gd, tau_init=100.0, tau_end=600.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts on best ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    # Restart 1: Fourier perturbation
    if remaining > 80:
        a_r = fourier_perturb(best_seq, rng, strength=0.04)
        alloc = min(remaining * 0.2, 100.0)
        if alloc > 20:
            a_r = lp_optimize(a_r, time_budget=alloc * 0.5, verbose=False)
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.08, 25.0)
        if alloc2 > 8:
            a_r = mirror_descent(a_r, time_budget=alloc2, tau_init=100.0, tau_end=600.0)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart fourier: obj={best_score:.8f} **")
        else:
            print(f"  Restart fourier: obj={obj_r:.8f}")

    # Restart 2: Symmetrized
    if remaining > 60:
        a_r = make_symmetric(best_seq)
        a_r = a_r / a_r.sum()
        alloc = min(remaining * 0.2, 80.0)
        if alloc > 15:
            a_r = lp_optimize(a_r, time_budget=alloc, verbose=False)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart sym: obj={best_score:.8f} **")
        else:
            print(f"  Restart sym: obj={obj_r:.8f}")

    # Restart 3: Blended with raised cosine
    if remaining > 50:
        rc = symmetric_init(n_cur)
        a_r = 0.7 * best_seq + 0.3 * rc
        a_r = np.maximum(a_r, 1e-15)
        a_r = a_r / a_r.sum()
        alloc = min(remaining * 0.15, 60.0)
        if alloc > 10:
            a_r = lp_optimize(a_r, time_budget=alloc, verbose=False)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart blend: obj={best_score:.8f} **")
        else:
            print(f"  Restart blend: obj={obj_r:.8f}")

    # Restart 4: Random perturbation
    if remaining > 50:
        a_r = best_seq + 0.02 * rng.standard_normal(n_cur)
        a_r = np.maximum(a_r, 1e-15)
        a_r = a_r / a_r.sum()
        alloc = min(remaining * 0.15, 60.0)
        if alloc > 10:
            a_r = lp_optimize(a_r, time_budget=alloc, verbose=False)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart random: obj={best_score:.8f} **")
        else:
            print(f"  Restart random: obj={obj_r:.8f}")

    # === Phase 6: Final Intensive Optimization ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final LP
    if remaining > 30:
        alloc = min(remaining * 0.35, 150.0)
        a_final = lp_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final mirror descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 60.0)
        a_final2 = mirror_descent(best_seq, time_budget=alloc, tau_init=200.0, tau_end=2000.0, verbose=True)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final MIR: obj={best_score:.8f} **")

    # Final GD with high tau
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 50.0)
        a_final3 = smoothed_gd(best_seq, time_budget=alloc, tau_init=200.0, tau_end=2500.0, verbose=True)
        obj_final3 = compute_obj(a_final3)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    # Scale to reasonable values
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
