Uses exact coordinate-wise optimization (no linearization) as the primary method,

Key innovation: exact 1D minimax coordinate descent that directly optimizes the

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_descent(a, time_budget=60.0, tau_init=20.0, tau_end=300.0, seed=42):
    """Smoothed gradient descent with increasing temperature and adaptive LR."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 3.0 / np.sqrt(n)
    momentum = 0.85
    v = np.zeros(n)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        if no_improve > 50:
            lr *= 0.7
            no_improve = 0
        lr = max(1e-10, lr * 0.999)
        it += 1

    return best_a

"""
Exact Coordinate Descent + Smoothed Gradient + LP Hybrid Optimizer.
Primary innovation: exact 1D minimax coordinate descent that solves the true
quadratic subproblem at each coordinate (no linearization), enabling convergence
to tighter local minima than trust-region or LP-based methods.
Combined with smoothed gradient warmup, LP polishing, and multi-length search.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_descent(a, time_budget=60.0, tau_init=15.0, tau_end=400.0, seed=42):
    """Smoothed gradient descent with increasing temperature and adaptive LR."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 4.0 / np.sqrt(n)
    momentum = 0.82
    v = np.zeros(n)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        if no_improve > 40:
            lr *= 0.75
            no_improve = 0
        lr = max(1e-12, lr * 0.998)
        it += 1

    return best_a


def coordinate_descent(a, time_budget=120.0, max_sweeps=200, seed=42, verbose=False,
                       n_evals=25):
    """
    Exact coordinate descent for min max_k conv(b,b)[k] s.t. b>=0.
    Each coordinate step solves the exact 1D quadratic minimax subproblem
    via golden section search. Uses incremental convolution updates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    conv = compute_conv(a)
    best_conv_max = float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    no_improve_sweeps = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate

    while time.monotonic() - start < time_budget and sweep < max_sweeps:
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            x_old = a[i]

            # Build the 1D problem: conv_new[k] = C[k] + A[k]*x + B[k]*x^2
            # A[k] = 2*a[k-i] if 0<=k-i<n and k != 2*i, else 0
            # B[k] = 1 if k == 2*i, else 0
            # C[k] = conv[k] - A[k]*x_old - B[k]*x_old^2

            A_vec = np.zeros(2 * n - 1)
            B_val = 0.0
            B_pos = -1

            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)

            if k_lo <= k_hi:
                idx = np.arange(k_lo, k_hi + 1)
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    mask = idx != 2 * i
                    A_vec[idx[mask]] = 2.0 * a[idx[mask] - i]
                    B_val = 1.0
                    B_pos = 2 * i
                else:
                    A_vec[idx] = 2.0 * a[idx - i]

            C_vec = conv - A_vec * x_old - B_val * x_old * x_old * (np.arange(2 * n - 1) == B_pos).astype(np.float64) if B_val > 0 else conv - A_vec * x_old

            # Golden section search for min_x max_k (C + A*x + B*x^2) s.t. x >= 0
            x_upper = max(x_old * 4.0 + 1.0 / n, 5.0 / n)

            def f1d(x):
                if B_val > 0:
                    vals = C_vec + A_vec * x
                    vals[B_pos] += x * x
                    return float(np.max(vals))
                else:
                    return float(np.max(C_vec + A_vec * x))

            # Check if x=0 is better
            f_0 = f1d(0.0)
            f_old = f1d(x_old)

            if f_0 < f_old - 1e-15:
                # Try searching in [0, x_old]
                lo, hi = 0.0, x_old
            else:
                # Search around x_old
                lo, hi = 0.0, x_upper

            # Golden section
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f_c1 = f1d(c1)
            f_c2 = f1d(c2)

            for _ in range(n_evals):
                if f_c1 < f_c2:
                    hi = c2
                    c2 = c1
                    f_c2 = f_c1
                    c1 = hi - gr * (hi - lo)
                    f_c1 = f1d(c1)
                else:
                    lo = c1
                    c1 = c2
                    f_c1 = f_c2
                    c2 = lo + gr * (hi - lo)
                    f_c2 = f1d(c2)

            x_new = (lo + hi) / 2.0
            x_new = max(0.0, x_new)
            f_new = f1d(x_new)

            # Also check x=0 explicitly
            if f_0 < f_new:
                x_new = 0.0
                f_new = f_0

            if f_new < best_conv_max - 1e-15:
                # Update a[i] and conv incrementally
                delta = x_new - x_old
                if abs(delta) > 1e-15:
                    a[i] = x_new
                    # Update conv: conv[k] += A[k]*delta + B*(x_new^2 - x_old^2)
                    if B_val > 0:
                        conv[k_lo:k_hi + 1] += A_vec[k_lo:k_hi + 1] * delta
                        conv[B_pos] += x_new * x_new - x_old * x_old
                    else:
                        conv[k_lo:k_hi + 1] += A_vec[k_lo:k_hi + 1] * delta

                    new_max = float(np.max(conv))
                    if new_max < best_conv_max:
                        best_conv_max = new_max
                        best_a = a.copy()
                        improved = True
                else:
                    pass
            else:
                # Still update conv if x changed (for consistency)
                delta = x_new - x_old
                if abs(delta) > 1e-15:
                    a[i] = x_new
                    if B_val > 0:
                        conv[k_lo:k_hi + 1] += A_vec[k_lo:k_hi + 1] * delta
                        conv[B_pos] += x_new * x_new - x_old * x_old
                    else:
                        conv[k_lo:k_hi + 1] += A_vec[k_lo:k_hi + 1] * delta

        sweep += 1
        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 4:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweep % 20 == 0:
            obj_cur = 2.0 * n * best_conv_max / (float(np.sum(a)) ** 2)
            print(f"    CD sweep {sweep}: obj={obj_cur:.8f}, t={time.monotonic()-start:.1f}s")

    # Final normalization
    s_final = best_a.sum()
    if s_final > 1e-10:
        best_a = best_a / s_final
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, M_target=None):
    """Solve LP: min M s.t. A_cross @ g <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_tr_sscp(a, r):
    """Trust region SOCP: min M s.t. linearized conv <= M, ||delta||<=r."""
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)

    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta

    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
        cp.norm(delta, 2) <= r,
    ]

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None

    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        best_t = 0
        best_t_obj = best_obj
        for t_val in [0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]:
            a_try = (1 - t_val) * best_a + t_val * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-15:
                best_t_obj = obj_try
                best_t = t_val

        if best_t > 0 and best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = (1 - best_t) * best_a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            s_ba = float(np.sum(best_a))
            if s_ba > 1e-10:
                best_a = best_a / s_ba
            a = best_a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                scale = 0.006 + 0.004 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                no_improve = 0
                if it > 80:
                    break

        it += 1

    return best_a


def tr_polish(a, time_budget=100.0, r_init=0.3, seed=42):
    """Trust region SOCP polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    r = r_init
    no_improve = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        delta_dir, M_val = solve_tr_sscp(a, r)

        if delta_dir is None:
            g_lp, _ = solve_lp_direction(a)
            if g_lp is None:
                break
            delta_dir = g_lp - a
        else:
            norm_d = np.linalg.norm(delta_dir)
            if norm_d > r:
                delta_dir = delta_dir * (r / norm_d)

        best_alpha = 0
        best_alpha_obj = best_obj
        for t_alpha in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0, 1.5]:
            a_try = a + t_alpha * delta_dir
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj - 1e-15:
                best_alpha_obj = obj_try
                best_alpha = t_alpha

        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-15:
            a = a + best_alpha * delta_dir
            a = np.maximum(a, 0)
            s_n = float(np.sum(a))
            if s_n > 1e-10:
                a = a / s_n
            best_obj = best_alpha_obj
            best_a = a.copy()
            no_improve = 0
            r = min(r * 1.1, 2.0)
        else:
            no_improve += 1
            r *= 0.6
            if no_improve >= 6:
                if r < 0.05:
                    a = best_a * np.exp(rng.normal(0, 0.01, n))
                    a = np.maximum(a, 0)
                    s_p = float(np.sum(a))
                    if s_p > 1e-10:
                        a = a / s_p
                    r = r_init
                    no_improve = 0

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0.0)


def multi_start_cd(a, time_budget=60.0, n_restarts=3, seed=100):
    """Run coordinate descent from multiple perturbed starting points."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    per_restart = time_budget / n_restarts

    for r in range(n_restarts):
        if time.monotonic() - start >= time_budget - 10:
            break
        scale = 0.005 + 0.005 * r
        a_pert = best_a * np.exp(rng.normal(0, scale, n))
        a_pert = np.maximum(a_pert, 0)
        s = float(np.sum(a_pert))
        if s < 1e-10:
            continue
        a_pert = a_pert / s
        a_res = coordinate_descent(a_pert, time_budget=per_restart,
                                   max_sweeps=50, seed=seed + r)
        obj_res = compute_obj(a_res)
        if obj_res < best_obj:
            best_obj = obj_res
            best_a = a_res.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Coordinate Descent + Smoothed GD + LP + TR hybrid.
    Uses exact 1D minimax coordinate descent as the primary optimizer.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Smoothed GD Warmup ===
    print(f"\n=== Phase 1: Smoothed GD Warmup ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.06, 60.0)
    if p1_budget > 10:
        a_sgd = smoothed_gradient_descent(best_seq, time_budget=p1_budget,
                                          tau_init=15.0, tau_end=350.0)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 GD: obj={obj_sgd:.8f}")

    # === Phase 2: Coordinate Descent (primary) ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.30, 300.0)
    if p2_budget > 20:
        a_cd = coordinate_descent(best_seq, time_budget=p2_budget,
                                  max_sweeps=150, verbose=True, seed=42, n_evals=28)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase2 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 CD: obj={obj_cd:.8f}")

    # === Phase 3: LP Polish ===
    print(f"\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 100.0)
    if p3_budget > 15:
        a_lp = lp_polish(best_seq, time_budget=p3_budget, seed=77)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase3 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 LP: obj={obj_lp:.8f}")

    # === Phase 4: More Coordinate Descent after LP ===
    print(f"\n=== Phase 4: CD after LP ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 120.0)
    if p4_budget > 15:
        a_cd2 = coordinate_descent(best_seq, time_budget=p4_budget,
                                   max_sweeps=100, verbose=True, seed=200, n_evals=30)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Phase4 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 CD: obj={obj_cd2:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current - 50, n_current + 50,
        n_current - 100, n_current + 100,
        n_current + 200, n_current + 400,
        1000, 1200, 1500, 600
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500 and abs(n - n_current) > 30]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        alloc2 = min(remaining * 0.15, 80.0)
        if alloc2 > 12:
            a_try = coordinate_descent(a_try, time_budget=alloc2,
                                       max_sweeps=40, seed=300 + n_try, n_evals=25)

        a_tr = tr_polish(best_seq, time_budget=alloc, r_init=0.2, seed=999)
        obj_tr = compute_obj(a_tr)
        if obj_tr < best_score:
            best_score = obj_tr
            best_seq = a_tr.copy()

                                     max_sweeps=80, verbose=True, seed=777, n_evals=35)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:

        a_final_lp = lp_polish(best_seq, time_budget=alloc, seed=2000)

        if obj_flp < best_score:
            best_score = obj_flp

    # Final validation
    if best_seq is None:

    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
