    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A_fft = fft(a, N)
    corr = ifft(P * np.conj(A_fft)).real[:n]
    return 2.0 * corr


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.maximum(v, 0)
    s = float(np.sum(v))
    if s < 1e-300:
        return np.ones(n) / n
    return v / s


def nesterov_gd(a, time_budget, tau_init=50.0, tau_max=2000.0):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = project_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    lr = 1.0 / np.sqrt(n)
    beta = 0.5
    y = a.copy()
    tau_cur = tau_init
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        g = smoothed_grad(y, tau_cur)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Adaptive step size
        lr_step = min(2.0 / (np.sqrt(tau_cur) * g_norm + 1.0), 5.0 / np.sqrt(n))
        y = a + beta * (a - y)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = a.copy()
        else:
            y = y / s_y

        a_new = y - lr_step * g
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Perturb to escape
                a = best_a * (1 + 0.02 * rng.standard_normal(n))
                a = project_simplex(a)
                y = a.copy()
                lr_step = 0.5 * lr_step
                stall = 0

        a = a_new
        tau_cur = min(tau_cur * 1.01, tau_max)
        it += 1

    return best_a


def lp_alternate(a, time_budget, seed=42, verbose=True):
    """LP alternating minimization with line-search blending."""
    n = len(a)
    a = project_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    stall_count = 0
    M_current = best_obj / (2.0 * n)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.5:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_lp(a, M_init=M_current)
        lp_time = time.monotonic() - t0

        if g_lp is None:
            break

        iter_count += 1

        # Try several blending factors via line search (golden section)
        def blend_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return np.inf
            return compute_obj(a_try)

        # Coarse search over t
        t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        best_t = 0.0
        best_t_obj = best_obj
        for t in t_candidates:
            obj_t = blend_obj(t)
            if obj_t < best_t_obj:
                best_t_obj = obj_t
                best_t = t

        # Fine search around best_t
        if best_t > 0:
            t_lo = max(0.0, best_t - 0.1)
            t_hi = min(1.0, best_t + 0.1)
            gr = (1.0 + np.sqrt(5.0)) / 2.0
            c = t_hi - (t_hi - t_lo) / gr
            d = t_lo + (t_hi - t_lo) / gr
            fc, fd = blend_obj(c), blend_obj(d)
            for _ in range(8):
                if fc < fd:
                    t_hi, d, fd = d, c, fc
                    c = t_hi - (t_hi - t_lo) / gr
                    fc = blend_obj(c)
                else:
                    t_lo, c, fc = c, d, fd
                    d = t_lo + (t_hi - t_lo) / gr
                    fd = blend_obj(d)
            t_mid = (t_lo + t_hi) / 2.0
            obj_mid = blend_obj(t_mid)
            if obj_mid < best_t_obj:
                best_t_obj = obj_mid
                best_t = t_mid

        if best_t_obj < best_obj - 1e-13:
            a_new = (1.0 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            a = a_new / s_new
            best_obj = best_t_obj
            best_a = a.copy()
            M_current = M_lp
            stall_count = 0
            if verbose and iter_count % 20 == 0:
                print(f"    LP {iter_count}: obj={best_obj:.10f}, t={best_t:.3f}, lp={lp_time:.2f}s")
        else:
            stall_count += 1
            if stall_count >= 3:
                # Perturb and try again
                strength = 0.005 + 0.003 * min(stall_count, 10)
                a = best_a * (1 + strength * rng.standard_normal(n))
                a = project_simplex(a)
                stall_count = 0

    if verbose:
        print(f"    LP done: {iter_count} iters, best_obj={best_obj:.10f}")
    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_initialization(n, seed=42, mode='bump'):
    """Create diverse initializations."""
    rng = np.random.default_rng(seed)
    if mode == 'uniform':
        a = np.ones(n)
    elif mode == 'bump':
        # Cosine bump concentrated at edges
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2 + 0.1
        a = np.abs(a)
    elif mode == 'edge':
        # Concentrated at edges
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.5) ** 2) + 0.05
    elif mode == 'sparse':
        a = np.zeros(n)
        n_nonzero = max(3, n // 4)
        indices = rng.choice(n, size=n_nonzero, replace=False)
        a[indices] = rng.uniform(0.5, 2.0, size=n_nonzero)
    elif mode == 'random':
        a = rng.uniform(0.1, 1.0, size=n)
    elif mode == 'sine':
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.5 * np.sin(2 * x + rng.uniform(0, 2 * np.pi))
        a = np.abs(a) + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating minimization,
    gradient polishing, and multi-length/multi-start exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    # Load parent or default
    best_a = None
    best_obj = np.inf

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0)
        if parent.size > 0 and parent.sum() > 0.01:
            parent = parent / parent.sum()
            n_init = parent.size
            obj_init = compute_obj(parent)
            best_a = parent.copy()
            best_obj = obj_init
            print(f"Parent: n={n_init}, obj={obj_init:.10f}")
        else:
            raise ValueError("Invalid parent")
    except Exception:
        n_init = 800
        best_a = np.ones(n_init) / n_init
        best_obj = compute_obj(best_a)
        print(f"Default init: n={n_init}, obj={best_obj:.10f}")

    n_best = len(best_a)

    # === Strategy: Multiple phases with diverse approaches ===

    # Phase 1: LP alternating from parent (most reliable)
    print("\n=== Phase 1: LP Alternating (parent) ===")
    p1_deadline = start_time + budget_s * 0.45
    p1_budget = max(30, p1_deadline - time.monotonic())
    a1 = lp_alternate(best_a, time_budget=p1_budget, seed=seed, verbose=True)
    obj1 = compute_obj(a1)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()
        print(f"  Phase 1: obj={best_obj:.10f} **")
    else:
        print(f"  Phase 1: obj={obj1:.10f}")

    # Phase 2: Gradient polishing
    print("\n=== Phase 2: Gradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = min(remaining * 0.15, 120.0)
        a2 = nesterov_gd(best_a, time_budget=p2_budget, tau_init=30.0, tau_max=1500.0)
        obj2 = compute_obj(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            print(f"  Phase 2: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 2: obj={obj2:.10f}")

    # Phase 3: LP again from polished (different seed)
    print("\n=== Phase 3: LP (polished start) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = min(remaining * 0.25, 250.0)
        a3 = lp_alternate(best_a, time_budget=p3_budget, seed=seed + 1234, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            print(f"  Phase 3: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 3: obj={obj3:.10f}")

    # Phase 4: Diverse initialization at same n
    print("\n=== Phase 4: Diverse Starts ===")
    remaining = deadline - time.monotonic()
    modes = ['edge', 'bump', 'sine', 'sparse']
    for mode in modes:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        a_div = make_initialization(n_best, seed=seed + hash(mode) % 10000, mode=mode)

        # Quick LP warm-up
        alloc = min(remaining * 0.2, 120.0)
        if alloc < 30:
            break
        a_div = lp_alternate(a_div, time_budget=alloc * 0.7, seed=seed + hash(mode) % 10000, verbose=False)
        obj_div = compute_obj(a_div)

        remaining = deadline - time.monotonic()
        if remaining > 30 and obj_div < best_obj + 0.05:
            obj_div2 = nesterov_gd(a_div, time_budget=min(remaining * 0.3, 40.0),
                                    tau_init=50.0, tau_max=1000.0)
            obj_div = compute_obj(obj_div2)
            a_div = obj_div2

        if compute_obj(a_div) < best_obj:
            best_obj = compute_obj(a_div)
            best_a = a_div.copy()
            print(f"  Mode '{mode}': obj={best_obj:.10f} **")
        else:
            print(f"  Mode '{mode}': obj={compute_obj(a_div):.10f}")

    # Phase 5: Different n values
    print("\n=== Phase 5: n Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        n_best - 50, n_best - 100, n_best + 50, n_best + 100,
        n_best - 200, n_best + 200, 600, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and n != n_best]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break

        a_try = resample_to_length(best_a, n_try)
        alloc = min(remaining * 0.25, 150.0)
        if alloc < 40:
            break

        a_try = lp_alternate(a_try, time_budget=alloc * 0.65, seed=n_try, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_try = nesterov_gd(a_try, time_budget=min(remaining * 0.3, 50.0),
                                 tau_init=50.0, tau_max=1000.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            n_best = n_try
            print(f"  n={n_try}: obj={best_obj:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # Phase 6: Final LP refinement
    print("\n=== Phase 6: Final LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p6_budget = min(remaining * 0.5, 200.0)
        a6 = lp_alternate(best_a, time_budget=p6_budget, seed=seed + 99999, verbose=True)
        obj6 = compute_obj(a6)
        if obj6 < best_obj:
            best_obj = obj6
            best_a = a6.copy()
            print(f"  Phase 6: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 6: obj={obj6:.10f}")

    # Phase 7: Ultra-fine gradient polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 7: Final Gradient Polish ===")
        p7_budget = min(remaining * 0.5, 60.0)
        a7 = nesterov_gd(best_a, time_budget=p7_budget, tau_init=100.0, tau_max=3000.0)
        obj7 = compute_obj(a7)
        if obj7 < best_obj:
            best_obj = obj7
            best_a = a7.copy()
            print(f"  Phase 7: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 7: obj={obj7:.10f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_obj:
        best_obj = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_obj = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
