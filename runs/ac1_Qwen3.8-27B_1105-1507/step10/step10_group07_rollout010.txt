"""
LSE-LBFGS + MGA Peak Peeling + LP Hybrid Optimizer for minimum peak autocorrelation.
Uses log-sum-exp smoothed objective with L-BFGS-B quasi-Newton optimization,
MGA-inspired peak peeling, and LP polishing. Employs adaptive temperature schedule
and multi-restart strategy to escape local minima.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lse_obj_grad(z, T):
    """
    Compute log-sum-exp smoothed objective and gradient w.r.t. z.
    a = softmax(z), obj = 2n/T * log(sum_k exp(T*conv_k))
    Returns (objective, gradient).
    """
    n = len(z)
    # Compute a = softmax(z) with numerical stability
    z_shift = z - np.max(z)
    exp_z = np.exp(z_shift)
    a = exp_z / np.sum(exp_z)

    # Compute convolution
    conv = compute_conv(a)

    # Compute log-sum-exp with stability
    Tconv = T * conv
    Tconv_max = np.max(Tconv)
    lse_val = Tconv_max / T + np.log(np.sum(np.exp(Tconv - Tconv_max))) / T
    obj = 2.0 * n * lse_val

    # Compute softmax weights over conv values
    w = np.exp(Tconv - Tconv_max)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = w / w_sum

    # Compute grad_a[j] = sum_k w[k] * 2*a[k-j] (a[m]=0 if m notin [0,n-1])
    # This is 2 * cross_correlation(w, a)[j]
    # (w ⊛ a)[j] = sum_m w[m+j] * a[m]
    N = 1
    while N < (2 * n - 1) + n:
        N <<= 1

    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(np.conj(W) * A, N)[:n]
    grad_a = 2.0 * corr

    # Convert grad_a to grad_z via softmax Jacobian
    # df/dz[j] = a[j] * (grad_a[j] - sum_i a[i]*grad_a[i])
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)

    return obj, grad_z


def lse_lbfgs_optimize(a_init, time_budget, T_start=5.0, T_end=800.0, n_temps=12, max_iter_per_T=50):
    """
    Optimize using LSE-smoothed objective with L-BFGS-B and increasing temperature.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    # Initialize z from a: z = log(a) (with small epsilon for zeros)
    eps = 1e-15
    z = np.log(a + eps)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Temperature schedule
    if n_temps > 1:
        T_schedule = np.geomspace(T_start, T_end, n_temps)
    else:
        T_schedule = [T_start]

    for Ti, T in enumerate(T_schedule):
        if time.monotonic() - start > time_budget:
            break

        def fun_grad(z_vec):
            obj, grad = lse_obj_grad(z_vec, T)
            if not np.isfinite(obj):
                return 1e10, np.zeros_like(z_vec)
            if not np.all(np.isfinite(grad)):
                grad = np.where(np.isfinite(grad), grad, 0.0)
            return obj, grad

        result = minimize(fun_grad, z, method='L-BFGS-B',
                          jac=True, options={'maxiter': max_iter_per_T,
                                             'gtol': 1e-8, 'maxfun': max_iter_per_T * 3})

        if result.x is not None:

            a_new = a_new / np.sum(a_new)

                best_a = a_new.copy()
                z = z_new

        if time.monotonic() - start > time_budget:

    return best_a


def mga_peak_peel(a, num_iters=30, step_size=0.3):
    """
    MGA (Monz Gradient Algorithm) style peak peeling.
    Iteratively find the peak lag and reduce it.
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(num_iters):
        conv = compute_conv(a)
        k_peak = int(np.argmax(conv))
        peak_val = conv[k_peak]

        if peak_val < 1e-15:
            break

        # Compute interference pattern for this peak
        # conv[k_peak] = sum_j a[j] * a[k_peak - j]
        # For each j where both a[j] and a[k_peak-j] are valid, the product contributes
        # We reduce the larger element in each pair

        # Build the "shadow" vector: shadow[j] = a[k_peak - j] if valid
        shadow = np.zeros(n)
        for j in range(n):
            k_idx = k_peak - j
            if 0 <= k_idx < n:
                shadow[j] = a[k_idx]

        # The gradient of conv[k_peak] w.r.t. a is 2*shadow (approximately)
        # We want to reduce conv[k_peak], so move a in the direction -shadow
        # But we need to maintain non-negativity and normalization

        # MGA update: a <- a - step * shadow (for positions where shadow > a, cap it)
        alpha = step_size * peak_val / (np.sum(shadow * a) + 1e-15)
        alpha = min(alpha, 0.5)

        a_new = a - alpha * shadow
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            for frac in [0.5, 0.25, 0.1]:
                a_try = a - alpha * frac * shadow
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                # No improvement, try reducing step size globally
                step_size *= 0.7
                if step_size < 0.01:
                    break

    return best_a


def lp_polish_simple(a, time_budget=60.0):
    """
    Simple LP-based polishing: find direction g that minimizes max(conv(a,g))
    subject to sum(g)=1, g>=0, then line search.
    Uses scipy linprog for speed.
    """
    from scipy.optimize import linprog

    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    # Build constraint matrix: conv(a, g)[k] = sum_j A[k,j] * g[j] <= M
    # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(j, j + n):
            if k < 2 * n - 1:
                A_mat[k, j] = a[k - j]

    while time.monotonic() - start < time_budget:
        # Solve LP: min M s.t. A*g <= M*1, sum(g) = 1, g >= 0
        # Variables: [g_0, ..., g_{n-1}, M] (total n+1 variables)
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize M

        # Inequality constraints: A*g - M <= 0
        # [A_mat, -ones] @ [g; M] <= 0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A_mat
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)

        # Equality: sum(g) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(None, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        if res.status != 0 or res.x is None:
            break

        g = np.maximum(res.x[:n], 0)
        s_g = g.sum()
        if s_g < 1e-10:
            break
        g = g / s_g

        # Line search
        best_t = 0
        best_t_obj = best_obj
        for t in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * best_a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-15:
                best_t_obj = obj_try
                best_t = t

        if best_t > 0:
            best_obj = best_t_obj
            best_a = (1 - best_t) * best_a + best_t * g
            best_a = np.maximum(best_a, 0)
            s_ba = best_a.sum()
            if s_ba > 1e-10:
                best_a = best_a / s_ba
        else:
            # Perturb and try again
            rng = np.random.default_rng(it + 42)
            a_pert = best_a * np.exp(rng.normal(0, 0.01, n))
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-10:
                a_pert = a_pert / s_pert
            a = a_pert
            best_a = a_pert.copy()
            best_obj = compute_obj(a)

        it += 1

    return best_a


def multi_peak_lse_grad(a, T, top_k=5):
    """
    Compute gradient using only top-K peaks (faster approximation).
    """
    n = len(a)
    conv = compute_conv(a)

    # Get top K peak indices
    top_idx = np.argsort(conv)[::-1][:top_k]
    peak_vals = conv[top_idx]
    max_pv = peak_vals[0]

    w_top = w_top / np.sum(w_top)

    for w_k, k in zip(w_top, top_idx):
        k = int(k)
        for j in range(n):
            k_idx = k - j
            if 0 <= k_idx < n:
                grad_a[j] += 2.0 * w_k * a[k_idx]

    # Convert to z-gradient (assuming a is already normalized)
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)

    return grad_a, grad_z


def projected_gradient_descent_lse(a, time_budget, T=100.0, n_iters=2000):
    """
    Projected gradient descent on the LSE-smoothed objective.
    Works directly with a (projected onto simplex) rather than z.
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / (T * np.max(compute_conv(a)) + 1e-10)
    momentum = np.zeros(n)
    beta = 0.7
    it = 0

    while time.monotonic() - start < time_budget and it < n_iters:
        conv = compute_conv(a)
        max_conv = np.max(conv)

        # Adaptive temperature
        T_cur = T * (1.0 + 0.5 * it / max(1, n_iters // 4))

        # Softmax weights
        Tconv = T_cur * conv
        Tconv_max = np.max(Tconv)
        w = np.exp(Tconv - Tconv_max)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum

        # Compute gradient via cross-correlation
        N = 1
        while N < (2 * n - 1) + n:
            N <<= 1
        W = rfft(w, N)
        A = rfft(a, N)
        corr = irfft(np.conj(W) * A, N)[:n]
        grad_a = 2.0 * corr

        # Nesterov-style momentum
        momentum = beta * momentum + grad_a
        a_new = a - lr * momentum

        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr = lr * 0.999
        it += 1

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LSE-LBFGS + MGA + LP hybrid with multi-restart.
    Uses adaptive temperature L-BFGS-B as primary optimizer,
    MGA peak peeling for targeted improvement, and LP for final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_cur = len(best_seq)

    # === Phase 1: Quick PGD warmup ===
    print(f"\n=== Phase 1: PGD Warmup ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.05, 40.0)
    if p1_budget > 10:
        a_pgd = projected_gradient_descent_lse(best_seq, time_budget=p1_budget, T=50.0)
        obj_pgd = compute_obj(a_pgd)
        if obj_pgd < best_score:
            best_score = obj_pgd
            best_seq = a_pgd.copy()
            print(f"  PGD: obj={best_score:.8f} **")
        else:
            print(f"  PGD: obj={obj_pgd:.8f}")

    # === Phase 2: LSE-LBFGS (primary optimizer) ===
    print(f"\n=== Phase 2: LSE-LBFGS ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.30, 300.0)
    if p2_budget > 30:
        a_lse = lse_lbfgs_optimize(best_seq, time_budget=p2_budget,
                                   T_start=3.0, T_end=600.0, n_temps=10, max_iter_per_T=40)
        obj_lse = compute_obj(a_lse)
        if obj_lse < best_score:
            best_score = obj_lse
            best_seq = a_lse.copy()
            print(f"  LSE-LBFGS: obj={best_score:.8f} **")
        else:
            print(f"  LSE-LBFGS: obj={obj_lse:.8f}")

    # === Phase 3: MGA Peak Peeling ===
    print(f"\n=== Phase 3: MGA Peak Peeling ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.10, 80.0)
    if p3_budget > 10:
        a_mga = mga_peak_peel(best_seq, num_iters=50, step_size=0.3)
        obj_mga = compute_obj(a_mga)
        if obj_mga < best_score:
            best_score = obj_mga
            best_seq = a_mga.copy()
            n_cur = len(best_seq)
            print(f"  MGA: obj={best_score:.8f} **")
        else:
            print(f"  MGA: obj={obj_mga:.8f}")

    # === Phase 4: Second LSE-LBFGS pass (higher T) ===
    print(f"\n=== Phase 4: LSE-LBFGS (high T) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.20, 200.0)
    if p4_budget > 20:
        a_lse2 = lse_lbfgs_optimize(best_seq, time_budget=p4_budget,
                                    T_start=50.0, T_end=2000.0, n_temps=8, max_iter_per_T=30)
        obj_lse2 = compute_obj(a_lse2)
        if obj_lse2 < best_score:
            best_score = obj_lse2
            best_seq = a_lse2.copy()
            print(f"  LSE-LBFGS2: obj={best_score:.8f} **")
        else:
            print(f"  LSE-LBFGS2: obj={obj_lse2:.8f}")

    # === Phase 5: LP Polish ===
    print(f"\n=== Phase 5: LP Polish ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.12, 100.0)
    if p5_budget > 15:
        a_lp = lp_polish_simple(best_seq, time_budget=p5_budget)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 6: Multi-restart with perturbations ===
    print(f"\n=== Phase 6: Multi-restart ===")
    remaining = deadline - time.monotonic()

    for r_idx in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        scale = 0.005 + 0.004 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        # Quick optimization
        alloc1 = min(remaining * 0.15, 60.0)
        if alloc1 > 10:
            a_pert = lse_lbfgs_optimize(a_pert, time_budget=alloc1,
                                        T_start=20.0, T_end=800.0, n_temps=6, max_iter_per_T=25)

        obj_pert = compute_obj(a_pert)
        if obj_pert < best_score:
            best_score = obj_pert
            best_seq = a_pert.copy()
            print(f"  Restart{r_idx} (scale={scale:.4f}): obj={best_score:.8f} **")
        else:
            print(f"  Restart{r_idx} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 7: Try different lengths ===
    print(f"\n=== Phase 7: Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = []
    for delta in [-200, -100, 100, 200, 400]:
        n_try = n_current + delta
        if 400 <= n_try <= 2000 and abs(n_try - n_current) > 50:
            n_options.append(n_try)
    n_options.append(1000)
    n_options.append(1200)
    n_options = sorted(set(n_options))

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break
        if abs(n_try - n_current) < 30:
            continue

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)

        alloc = min(remaining * 0.15, 70.0)
        if alloc > 10:
            a_try = lse_lbfgs_optimize(a_try, time_budget=alloc,
                                       T_start=10.0, T_end=600.0, n_temps=7, max_iter_per_T=25)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 100.0)
        a_final = lse_lbfgs_optimize(best_seq, time_budget=alloc,
                                     T_start=100.0, T_end=3000.0, n_temps=6, max_iter_per_T=30)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LSE: obj={best_score:.8f} **")
        else:
            print(f"  Final LSE: obj={obj_final:.8f}")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 40.0)
        a_flp = lp_polish_simple(best_seq, time_budget=alloc)
        obj_flp = compute_obj(a_flp)
        if obj_flp < best_score:
            best_score = obj_flp
            best_seq = a_flp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final MGA
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_fmga = mga_peak_peel(best_seq, num_iters=20, step_size=0.15)
        obj_fmga = compute_obj(a_fmga)
        if obj_fmga < best_score:
            best_score = obj_fmga
            best_seq = a_fmga.copy()
            print(f"  Final MGA: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
