import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_obj_grad(x, tau):
    """
    Compute smoothed objective and gradient using FFT.
    Objective: 2n * logsumexp(tau * conv(a)) / (sum(a))^2
    where conv(a) = a * a (full linear convolution).
    """

    a = np.maximum(x, 0.0)
    S = float(np.sum(a))
    if S < 1e-10:
        return 1e10, np.zeros(n)

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)[:2 * n - 1]

    shifted = tau * (conv - max_c)
    exp_shifted = np.exp(shifted)

    if total < 1e-300:

    p = exp_shifted / total  # softmax weights

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lbfgs_anneal(a, time_budget=60.0, tau_schedule=None, verbose=False):
    """
    L-BFGS-B with temperature annealing on the smoothed objective.
    Starts at high tau (smooth) and decreases to low tau (sharp).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_schedule is None:
        tau_schedule = [50.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.2, 1.0, 0.8, 0.6, 0.5]

    for tau in tau_schedule:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        time_per_tau = remaining / max(1, len(tau_schedule) - tau_schedule.index(tau))
        maxiter = max(50, int(time_per_tau * 200))

        def fun_and_grad(x):
            return smooth_obj_grad(x, tau)

        try:
            res = minimize(fun_and_grad, a, jac=True, method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                                    'maxfun': maxiter * 3})
            a_new = np.maximum(res.x, 0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    if verbose:
                        print(f"      tau={tau:.2f}: obj={best_obj:.8f}")
        except Exception as e:
            if verbose:
                print(f"      tau={tau:.2f}: error {e}")

    # Final polish with very low temperature using many iterations
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        for tau_final in [0.4, 0.3]:
            remaining = time_budget - (time.monotonic() - start)
            if remaining < 1:
                break
            try:
                def fun_and_grad2(x):
                    return smooth_obj_grad(x, tau_final)
                res = minimize(fun_and_grad2, a, jac=True, method='L-BFGS-B',
                               bounds=[(0, None)] * n,
                               options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-12})
                a_new = np.maximum(res.x, 0)
                s_new = a_new.sum()
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
            except Exception:
                pass

    return best_a


def lp_polish(a, time_budget=120.0, verbose=False):
    """LP-based polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        best_t = 0
        best_t_obj = best_obj
        for t_val in [0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t_val) * best_a + t_val * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-15:
                best_t_obj = obj_try
                best_t = t_val

        if best_t > 0 and best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = (1 - best_t) * best_a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            s_ba = best_a.sum()
            if s_ba > 1e-10:
                best_a = best_a / s_ba
            a = best_a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                scale = 0.005 + 0.003 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                no_improve = 0
                if it > 50:
                    break

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def support_reduction(a, time_budget=30.0, threshold=0.001):
    """
    Identify small entries, set them to zero, and re-optimize.
    This can help escape flat regions of the landscape.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Sort by value, try zeroing the smallest entries
    for frac in [0.05, 0.1, 0.15, 0.2]:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        k_zero = int(n * frac)
        if k_zero < 1:
            continue
        sort_idx = np.argsort(a)
        a_try = a.copy()
        a_try[sort_idx[:k_zero]] = 0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Quick L-BFGS polish
        a_try = lbfgs_anneal(a_try, time_budget=min(remaining * 0.4, 20.0),
                              tau_schedule=[10.0, 5.0, 2.0, 1.0, 0.5])
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: L-BFGS-B annealing + LP hybrid with multi-start and diverse
    initializations. Uses FFT-based gradient for fast optimization.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        parent = None
        n_target = 800

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: L-BFGS-B Annealing on parent ===
    print(f"\n=== Phase 1: L-BFGS-B Annealing (parent) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.20, 200.0)
    if p1_budget > 20:
        a1 = lbfgs_anneal(best_seq, time_budget=p1_budget, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1 L-BFGS: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 L-BFGS: obj={obj1:.8f}")

    # === Phase 2: LP Polish ===
    print(f"\n=== Phase 2: LP Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 150.0)
    if p2_budget > 15:
        a2 = lp_polish(best_seq, time_budget=p2_budget, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj2:.8f}")

    # === Phase 3: Multi-start with diverse initializations ===
    print(f"\n=== Phase 3: Multi-Start ===")
    remaining = deadline - time.monotonic()

    initializations = []

    # Perturbation 1: small
    a_p1 = best_seq * np.exp(rng.normal(0, 0.01, n_current))
    a_p1 = np.maximum(a_p1, 0)
    a_p1 = a_p1 / a_p1.sum()
    initializations.append(("small_pert", a_p1))

    # Perturbation 2: medium
    a_p2 = best_seq * np.exp(rng.normal(0, 0.05, n_current))
    a_p2 = np.maximum(a_p2, 0)
    a_p2 = a_p2 / a_p2.sum()
    initializations.append(("med_pert", a_p2))

    # Perturbation 3: large
    a_p3 = best_seq * np.exp(rng.normal(0, 0.15, n_current))
    a_p3 = np.maximum(a_p3, 0)
    a_p3 = a_p3 / a_p3.sum()
    initializations.append(("large_pert", a_p3))

    # Random exponential
    a_p4 = rng.exponential(1.0, n_current)
    a_p4 = a_p4 / a_p4.sum()
    initializations.append(("rand_exp", a_p4))

    # One-sided ramp
    a_p5 = np.linspace(1.0, 0.1, n_current)
    a_p5 = a_p5 / a_p5.sum()
    initializations.append(("ramp", a_p5))

    # Reverse ramp
    a_p6 = np.linspace(0.1, 1.0, n_current)
    a_p6 = a_p6 / a_p6.sum()
    initializations.append(("ramp_rev", a_p6))

    # Gaussian bump
    x = np.linspace(0, 1, n_current)
    a_p7 = np.exp(-20 * (x - 0.3) ** 2) + 0.1
    a_p7 = a_p7 / a_p7.sum()
    initializations.append(("gauss_left", a_p7))

    # Two bumps
    a_p8 = np.exp(-30 * (x - 0.25) ** 2) + np.exp(-30 * (x - 0.75) ** 2) + 0.05
    a_p8 = a_p8 / a_p8.sum()
    initializations.append(("two_bump", a_p8))

    # LP direction from parent
    g_lp_init, _ = solve_lp_direction(best_seq)
    if g_lp_init is not None:
        initializations.append(("lp_dir", g_lp_init))

    for name, a_init in initializations:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        alloc = min(remaining * 0.12, 60.0)
        if alloc < 10:
            continue

        print(f"  Start '{name}' (remaining={remaining:.0f}s, alloc={alloc:.0f}s)")
        a_try = lbfgs_anneal(a_init, time_budget=alloc * 0.7,
                              tau_schedule=[30.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5])
        # Quick LP polish
        remaining_inner = deadline - time.monotonic()
        if remaining_inner > 10:
            a_try = lp_polish(a_try, time_budget=min(remaining_inner * 0.15, 20.0))

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"    ** '{name}': obj={best_score:.8f} **")
        else:
            print(f"    '{name}': obj={obj_try:.8f}")

    # === Phase 4: Support reduction ===
    print(f"\n=== Phase 4: Support Reduction ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_sr = support_reduction(best_seq, time_budget=min(remaining * 0.1, 40.0))
        obj_sr = compute_obj(a_sr)
        if obj_sr < best_score:
            best_score = obj_sr
            best_seq = a_sr.copy()
            print(f"  Support reduction: obj={best_score:.8f} **")
        else:
            print(f"  Support reduction: obj={obj_sr:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current - 100 if n_current - 100 >= 400 else 400,
        n_current + 100,
        n_current + 200,
        n_current + 400,
        600, 800, 1000, 1200, 1400, 1600,
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_current) > 50]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)

        alloc1 = min(remaining * 0.10, 40.0)
        if alloc1 > 8:
            a_try = lbfgs_anneal(a_try, time_budget=alloc1,
                                  tau_schedule=[20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.6])

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.08, 30.0)
        if alloc2 > 8:
            a_try = lp_polish(a_try, time_budget=alloc2)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Intensive final polish ===
    print(f"\n=== Phase 6: Final Intensive Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # L-BFGS with very fine annealing
        alloc = min(remaining * 0.4, 120.0)
        a_final = lbfgs_anneal(best_seq, time_budget=alloc,
                                tau_schedule=[5.0, 3.0, 2.0, 1.5, 1.2, 1.0, 0.8, 0.6, 0.5, 0.4, 0.3])
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final L-BFGS: obj={best_score:.8f} **")
        else:
            print(f"  Final L-BFGS: obj={obj_final:.8f}")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 60.0)
        a_flp = lp_polish(best_seq, time_budget=alloc)
        obj_flp = compute_obj(a_flp)
        if obj_flp < best_score:
            best_score = obj_flp
            best_seq = a_flp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more short L-BFGS
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.2, 30.0)
        a_f2 = lbfgs_anneal(best_seq, time_budget=alloc,
                             tau_schedule=[1.0, 0.7, 0.5, 0.4, 0.3])
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final L-BFGS2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
