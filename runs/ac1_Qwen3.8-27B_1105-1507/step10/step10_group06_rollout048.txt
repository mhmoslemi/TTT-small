    n = len(a)
    N = 1
    while N < 2*n:
        N <<= 1

    return a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternating minimization, Gerchberg-Saxton spectral flattening,
subgradient polishing, and simulated annealing with multi-length exploration.
Key improvements: spectral domain diversification, adaptive temperature schedule,
richer initialization pool, and tighter LP tolerances.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a, tight_tol=False):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    if tight_tol:
        solvers = [cp.CLARABEL, cp.MOSEK] + solvers

    for solver in solvers:
        try:
            solve_kwargs = {}
            if tight_tol:
                solve_kwargs['tol_gap_abs'] = 1e-10
                solve_kwargs['tol_gap_rel'] = 1e-10
                solve_kwargs['tol_infeas'] = 1e-10
            prob.solve(solver=solver, **solve_kwargs)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    return g_val / s_g, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of (a*a)_k using FFT.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def subgradient_polish(a, time_budget, iters_per_sec=2000):
    """
    Subgradient method with adaptive step size on the max-conv objective.
    Works on normalized sequences (sum=1, a>=0).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    step_size = 0.5 / np.sqrt(n)
    momentum = 0.9
    v = np.zeros(n)
    no_improve = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient of conv[k_star] w.r.t. a
        # conv[k] = sum_j a[j]*a[k-j], d/d a[i] = a[k-i] + a[i-k] (with bounds)
        sg = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            j2 = i - k_star
            if 0 <= j1 < n:
                sg[i] += a[j1]
            if 0 <= j2 < n:
                sg[i] += a[j2]
        sg *= 2.0  # because max over k, and we differentiate through

        # Momentum update
        v = momentum * v - step_size * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            v = np.zeros(n)
            step_size *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                step_size *= 0.8
                if step_size < 1e-6 / n:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                no_improve = 0

        a = a_new

    return best_a


def gerchberg_saxton(a, iters=200, n_target=None):
    """
    Gerchberg-Saxton: find non-negative signal with flat power spectrum.
    This produces sequences with approximately flat auto-correlation.
    """
    n = len(a)
    N = 1
    while N < 4 * n:
        N <<= 1
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Target flat magnitude
    target_mag = np.ones(N) / np.sqrt(N)

    for it in range(iters):
        # FFT domain: flatten magnitude, keep phase
        A = fft(a, N)
        mag = np.abs(A) + 1e-30
        phase = A / mag
        A_new = target_mag * phase
        a_new = ifft(A_new).real
        # Time domain: keep only first n elements, enforce non-negativity
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a = a_new / s_new

    return a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = max(3, n // 100)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, rng, strength=0.1):
    """Random multiplicative perturbation."""
    n = len(a)
    mult = 1.0 + strength * rng.standard_normal(n)
    a_new = a * mult
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_alternate_aaa(a, time_budget, seed=42, verbose=False, use_annealing=True):
    """
    LP-based alternating minimization with simulated annealing on blend factor.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    t_blend = 0.6
    temperature = 0.3
    stall_count = 0
    best_ever_obj = best_obj
    last_improve_iter = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_conv_lp(a)
        lp_time = time.monotonic() - t0

        if g_lp is None:
            break

        iter_count += 1

        # Generate candidate blend factors
        candidates_t = [t_blend, 0.1, 0.3, 0.5, 0.7, 1.0]
        if iter_count > 50:
            candidates_t.extend([0.05, 0.02, 0.01])
        if use_annealing and temperature > 0.01:
            candidates_t.append(rng.uniform(0.01, 1.0))

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend

        for t in candidates_t:
            t = min(max(t, 0.001), 1.0)
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)

            # Simulated annealing: occasionally accept worse solutions
            if obj_try < best_local_obj or (
                use_annealing and temperature > 0.05 and
                rng.random() < np.exp(-(obj_try - best_local_obj) / temperature)
            ):
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_local_a = a_try.copy()
                    best_local_t = t

        # Accept best local
        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            t_blend = max(0.01, best_local_t * 0.98)
            stall_count = 0
            last_improve_iter = iter_count
            if verbose and iter_count % 50 == 0:
                print(f"    LP {iter_count}: obj={best_obj:.10f}, t={t_blend:.4f}, "
                      f"T={temperature:.4f}, lp={lp_time:.3f}s")
        else:
            stall_count += 1
            # Temperature decay
            temperature *= 0.995
            if use_annealing:
                temperature = max(temperature, 0.01)

            # Perturbation to escape local minima
            if stall_count >= 5:
                strength = 0.01 + 0.005 * min(stall_count, 15)
                if rng.random() < 0.5:
                    a = fourier_perturb(best_a, rng, strength=strength)
                else:
                    a = random_perturb(best_a, rng, strength=strength)
                t_blend = max(0.1, min(t_blend * 1.3, 0.8))
                temperature = max(temperature, 0.1)
                stall_count = 0
            elif stall_count >= 2:
                a = best_a + 0.003 * rng.standard_normal(n) * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()

        # Periodic GS diversification
        if iter_count % 300 == 0 and remaining > 30:
            a_gs = gerchberg_saxton(a, iters=50)
            obj_gs = compute_obj(a_gs)
            if obj_gs < best_obj:
                best_obj = obj_gs
                best_a = a_gs.copy()
                a = a_gs.copy()
                if verbose:
                    print(f"    LP {iter_count}: GS improvement! obj={best_obj:.10f}")

    if verbose:
        print(f"    LP alternating: {iter_count} iters, best={best_obj:.10f}")

    return best_a


def nesterov_gd_polish(a, time_budget, tau_init=30.0, tau_max=2000.0):
    """
    Nesterov accelerated gradient descent on the smoothed objective.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adaptive_lr = 0.5 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.1 * base_lr), 3.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adaptive_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr *= 0.9998
        tau = min(tau * 1.02, tau_max)
        it += 1

        if it % 5000 == 0 and compute_obj(a) > best_obj + 1e-6:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5

    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def create_structured_init(n, rng):
    """Create diverse structured initializations."""
    method = rng.integers(0, 5)
    if method == 0:
        # Near-uniform with small perturbation
        a = np.ones(n) * (1 + 0.1 * rng.standard_normal(n))
    elif method == 1:
        # Two-scale pattern
        a = np.ones(n)
        for i in range(0, n, 3):
            a[i] = rng.uniform(0.5, 1.5)
        for i in range(1, n, 3):
            a[i] = rng.uniform(0.3, 1.0)
        for i in range(2, n, 3):
            a[i] = rng.uniform(0.3, 1.0)
    elif method == 2:
        # Smooth random (low-pass filtered noise)
        a = rng.standard_normal(n) + 0.5 * rng.standard_normal(n)
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=n/20)
        a = a - a.min() + 0.1
    elif method == 3:
        # Cosine-based
        x = np.linspace(0, 2 * np.pi * rng.uniform(0.5, 3), n)
        a = 1.0 + 0.3 * np.cos(x) + 0.2 * np.cos(2*x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
    else:
        # Piecewise constant
        n_segs = rng.integers(5, 20)
        a = np.repeat(rng.uniform(0.5, 1.5, n_segs), n // n_segs + 1)[:n]

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

    p1_end = start_time + budget_s * 0.55

    alloc_per_start = (p1_end - time.monotonic()) / n_starts
