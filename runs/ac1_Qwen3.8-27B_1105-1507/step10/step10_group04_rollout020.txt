```python
import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import cvxpy as cp
import time

def compute_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def solve_lp(a, max_time=30):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val /= s
                    return g_val, float(M.value)
        except:
            continue
    return None, None
```

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft

import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, time_limit=10.0):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]
    for solver in solvers:
        if time.monotonic() - start > time_limit:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None

def high_tau_gd(a, time_budget, tau_start=200, tau_end=10000, verbose=True):
    """Nesterov accelerated gradient descent with adaptive tau."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0

    N = 1
    while N < 3 * n:
        N <<= 1

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 0.7

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient of smoothed objective
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -50, 0)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient
        P = fft(p, N)
        A_f = fft(y, N)
        cross = ifft(P * np.conj(A_f)).real
        grad = np.zeros(n)
        grad[0] = 2.0 * cross[0]
        for j in range(1, n):
            grad[j] = 2.0 * cross[N - j]

        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break

        # Adaptive learning rate
        lr = min(2.0 / (g_norm + 1e-10), 5.0 / np.sqrt(n))
        lr = max(lr, 1e-10)

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        if verbose and it % 3000 == 0:

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it
            if it > 40000:
                break

    return best_a


def log_param_optimize(a, time_budget, verbose=True):

        else:

        obj_val = max_conv + np.log(sum_exp) / tau_local

        P = fft(p, N)
        A_f = fft(a_full, N)
        cross = ifft(P * np.conj(A_f)).real
        grad_full = np.zeros(full_len)
        grad_full[0] = 2.0 * cross[0]
        for j in range(1, full_len):

        # Chain rule for log-parameterization

        return obj_val, grad_b

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        a = np.arange(1, n + 1, dtype=np.float64)
    elif mode == 3:
        x = np.linspace(0.01, 1, n)
        a = x**2 * (1 - x)**0.5
    elif mode == 4:
        a = np.exp(-np.arange(n) / (n * 0.3))
    elif mode == 5:
        a = np.exp(-((np.arange(n) - n/2)**2) / (2 * (n/4)**2))
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x * (1 - x) ** 2
    elif mode == 7:
        a = np.ones(n) + 0.3 * np.sin(2 * np.pi * np.arange(n) / n * 3)
    else:
        a = rng.uniform(0.5, 1.5, n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    return a

def kkt_polish(a, time_budget, verbose=True):
    """Polish using KKT conditions: at optimum, gradient of active constraints is in span of {1, active_mask}."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 3 * n:
        N <<= 1

    # Use high-tau gradient to identify active constraints and do targeted updates
    for it in range(20):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        # Identify "active" convolution positions (within epsilon of max)
        active_thresh = max_conv * 0.95
        active_k = np.where(conv >= active_thresh)[0]

        # Compute which a_j are most important for active constraints
        # grad_j = 2 * sum_{k in active} p_k * a[k-j]
        # For simplicity, use uniform p over active set
        p = np.zeros(len(conv))
        p[active_k] = 1.0 / len(active_k)

        P = fft(p, N)
        A_f = fft(a, N)
        cross = ifft(P * np.conj(A_f)).real
        grad = np.zeros(n)
        grad[0] = 2.0 * cross[0]
        for j in range(1, n):
            grad[j] = 2.0 * cross[N - j]

        # Project gradient to zero-sum (since we're on the simplex)
        grad = grad - np.mean(grad)

        # Small step in negative gradient direction
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break
        lr = min(0.5 / g_norm, 1.0 / n)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP fixed-point iteration, high-precision GD,
    KKT polishing, n-sweep, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.9f}")
    except NameError:
        pass
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = diverse_init(n_init, rng, mode=0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.9f}")

    n_best = len(best_seq)

    # === Phase 1: LP Fixed-Point Iteration (main workhorse, ~400s) ===
    print(f"\n=== Phase 1: LP Fixed-Point (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.28, 420.0)
    if p1_budget > 30:
        a_lp = lp_fixed_point(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP: obj={best_score:.9f} **")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.9f}")

    # === Phase 2: KKT Polish (~50s) ===
    print(f"\n=== Phase 2: KKT Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.04, 60.0)
    if p2_budget > 10:
        a_kkt = kkt_polish(best_seq, time_budget=p2_budget, verbose=True)
        obj_kkt = compute_obj(a_kkt)
        if obj_kkt < best_score:
            best_score = obj_kkt
            best_seq = a_kkt.copy()
            print(f"  Phase2 KKT: obj={best_score:.9f} **")

    # === Phase 3: High-Tau GD (~150s) ===
    print(f"\n=== Phase 3: High-Tau GD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 150.0)
    if p3_budget > 20:
        a_gd = high_tau_gd(best_seq, time_budget=p3_budget,
                           tau_start=200, tau_end=8000, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase3 GD: obj={best_score:.9f} **")

    # === Phase 4: Log-Param Optimization (~80s) ===
    print(f"\n=== Phase 4: Log-Param Opt ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.06, 80.0)
    if p4_budget > 10:
        a_log = log_param_optimize(best_seq, time_budget=p4_budget, verbose=True)
        obj_log = compute_obj(a_log)
        if obj_log < best_score:
            best_score = obj_log
            best_seq = a_log.copy()
            print(f"  Phase4 Log: obj={best_score:.9f} **")

    # === Phase 5: n-Sweep (~350s) ===
    print(f"\n=== Phase 5: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.25, 350.0)
    n_best = len(best_seq)

    n_options = sorted(set([
        max(200, n_best - 100), max(200, n_best - 50),
        n_best + 50, n_best + 100, n_best + 200,
        n_best + 400, n_best + 600, n_best + 1000,
        500, 700, 900, 1100, 1400, 1800, 2000, 2500
    ]))
    n_options = [n for n in n_options if 300 <= n <= 3000 and abs(n - n_best) > 20]
    n_options = n_options[:7]
    print(f"  n options: {n_options}")

    per_n_budget = p5_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        alloc = min(per_n_budget, remaining - 25)
        if alloc < 30:
            break

        a_try = resample_sequence(best_seq, n_try)

        # LP polish
        lp_alloc = alloc * 0.45
        if lp_alloc > 8:
            a_try = lp_fixed_point(a_try, time_budget=lp_alloc, verbose=False)

        # GD polish
        remaining = deadline - time.monotonic()
        gd_alloc = min(alloc * 0.35, remaining - 15)
        if gd_alloc > 8:
            a_try = high_tau_gd(a_try, time_budget=gd_alloc,
                                tau_start=100, tau_end=5000, verbose=False)

        # Quick log-param
        remaining = deadline - time.monotonic()
        log_alloc = min(alloc * 0.15, remaining - 10)
        if log_alloc > 5:
            a_try = log_param_optimize(a_try, time_budget=log_alloc, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.9f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.9f}")

    # === Phase 6: Diverse Restarts (~150s) ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p6_budget = min(remaining * 0.1, 150.0)
    n_current = len(best_seq)
    n_restarts = max(1, int(p6_budget / 45))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(50, remaining - 15)
        if alloc < 20:
            break

        mode = (r + 1) % 8
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.3 + 0.08 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        # Optimize
        half = alloc * 0.5
        a_r = lp_fixed_point(a_r, time_budget=half, verbose=False)
        remaining = deadline - time.monotonic()
        gd_alloc = min(alloc * 0.4, remaining - 10)
        if gd_alloc > 5:
            a_r = high_tau_gd(a_r, time_budget=gd_alloc,
                              tau_start=100, tau_end=3000, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.9f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.9f}")

    # === Phase 7: Final Intensive Polish (remaining time) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Time remaining: {remaining:.0f}s")

    # Round 1: LP
    if remaining > 30:
        alloc = min(remaining * 0.2, 100.0)
        a_f = lp_fixed_point(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.9f} **")

    # Round 2: KKT
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.1, 40.0)
        a_f = kkt_polish(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final KKT: obj={best_score:.9f} **")

    # Round 3: High-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.25, 120.0)
        a_f = high_tau_gd(best_seq, time_budget=alloc,
                          tau_start=500, tau_end=15000, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final GD: obj={best_score:.9f} **")

    # Round 4: Log-param
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 60.0)
        a_f = log_param_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final Log: obj={best_score:.9f} **")

    # Round 5: Support shrink
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.15, 30.0)
        a_f = support_refine(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final Shrink: obj={best_score:.9f} **")

    # Round 6: One more LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 50.0)
        a_f = lp_fixed_point(best_seq, time_budget=alloc, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Ultra LP: obj={best_score:.9f} **")

    # Final GD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 30.0)
        a_f = high_tau_gd(best_seq, time_budget=alloc,
                          tau_start=1000, tau_end=20000, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f
            print(f"  Ultra GD: obj={best_score:.9f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.9f}, time={elapsed:.1f}s ===")

    return result
