Optimizes step function coefficients using alternating LP + exact coordinate descent
with analytic 1D subproblems, multi-length exploration, and multiplicative weight updates.
Key innovations over previous:
- Alternating LP (two LPs per iteration) for stronger directional updates
- Better coordinate descent with adaptive bracketing

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)

def solve_lp_one_side(a_fixed, n):

    Returns (g, M)."""

    M = cp.Variable()

    # Actually total non-zeros: for each j, there are min(n, 2n-1-j) non-zeros in column j
    # Sum over j: sum_{j=0}^{n-1} min(n, 2n-1-j) = sum_{j=0}^{n-1} (n if j <= n-1) 
    # = n * n = n^2 (since 2n-1-j >= n for j <= n-1)
    # So the matrix is dense with n^2 non-zeros!

    """Alternating LP: iterate between minimizing max(a*g) and max(g*a)."""

        return a
    a = a / s

        # Check if a_new improves

        grad_full = ifft_full(A * np.conj(P)).real * 2.0

        a_new = np.maximum(a_new, 1e-15)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 3:
        freqs = rng.integers(1, 12, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Arcsine distribution (optimal for some related problems)
        x = np.linspace(0.01, 0.99, n)
        a = 1.0 / (np.pi * np.sqrt(x * (1 - x)))
        a = a / a.sum()
    elif mode == 5:
        # Combination of arcsine and uniform
        x = np.linspace(0.01, 0.99, n)
        a = 1.0 / (np.pi * np.sqrt(x * (1 - x))) + 0.5
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

Optimizes step function coefficients using alternating LP (two LPs per iteration for
stronger directional updates), exact coordinate descent with golden section search,
multiplicative weight mirror descent, and multi-length exploration.
Key improvements over previous: alternating LP provides exact best-response updates,
better time allocation, and more aggressive local polishing.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp

import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp_minpeak(a_fixed, n, initial_g=None):
    """Solve LP: min M s.t. conv(a_fixed, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_value)."""
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    # Build constraint matrix: A[k, j] = a_fixed[k-j] if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = a_fixed[:end - j]

    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0.0)
                s = float(np.sum(g_val))
                if s > 1e-10:

            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Define 1D objective without closure
            def f_1d(x, v=v, S_other=S_other, n=n, c_k_lin=c_k_lin,
                     a_ki_lin=a_ki_lin, c_2i=c_2i):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Quick scan for bracket
            scale = max(v, S / n, 1e-8)
            candidates_x = [0.0, v * 0.3, v * 0.5, v * 0.7, v,
                            v * 1.3, v * 1.5, v * 2.0, v * 3.0]
            candidates_x = [max(0.0, min(10.0 * S, float(c))) for c in candidates_x]

            best_x = v
            best_f = f_1d(v)
            for xc in candidates_x:
                fx = f_1d(xc)
                if fx < best_f:
                    best_f = fx
                    best_x = xc

            # Golden section refinement
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 1.5

            if x_hi - x_lo < 1e-14:
                x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            else:
                c_pt = x_hi - gr * (x_hi - x_lo)
                d_pt = x_lo + gr * (x_hi - x_lo)
                fc = f_1d(c_pt)
                fd = f_1d(d_pt)

                for _ in range(45):
                    if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                        break
                    if fc < fd:
                        x_hi = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = x_hi - gr * (x_hi - x_lo)
                        fc = f_1d(c_pt)
                    else:
                        x_lo = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = x_lo + gr * (x_hi - x_lo)
                        fd = f_1d(d_pt)

                x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-15:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, t={elapsed:.1f}s")

    return best_a


def multiplicative_descent(a, time_budget=60.0, eta_init=0.08, eta_decay=0.997, verbose=False):
    """Multiplicative weight (mirror descent) with subgradient from top convolution peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    eta = eta_init
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))

        # Use top-k argmax positions for subgradient
        k_top = min(7, len(conv))
        top_idx = np.argsort(conv)[-k_top:]
        p = np.zeros(len(conv))
        p[top_idx] = 1.0 / k_top

        # Gradient via cross-correlation
        N = 1
        while N < 3 * n:
            N <<= 1
        from scipy.fft import fft as fft_full, ifft as ifft_full
        A_f = fft_full(a, N)
        P_f = fft_full(p, N)
        grad_full = ifft_full(A_f * np.conj(P_f)).real * 2.0
        grad = grad_full[:n]

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        eta *= eta_decay
        it += 1

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    MW iter {it}: obj={best_obj:.10f}, eta={eta:.6f}, t={elapsed:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1.0 - x * x + 1e-6)
        a = a / a.sum()
    elif mode == 3:
        freqs = rng.integers(1, 12, size=6)
        phases = rng.uniform(0, 2 * np.pi, size=6)
        x = np.linspace(0, 1, n)
        a = np.ones(n) * 0.5
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0.01, 0.99, n)
        a = 1.0 / (np.pi * np.sqrt(x * (1.0 - x)))
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 0.99, n)
        a = 1.0 / (np.pi * np.sqrt(x * (1.0 - x))) + 0.3
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: alternating LP + coordinate descent + multiplicative updates +
    multi-n exploration + diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = 2.0 * n_default * float(np.max(compute_conv(best_seq)))
        print(f"Default init (n={n_default}): obj={best_score:.10f}")

    n_best = len(best_seq)

    # === Phase 1: Alternating LP (main improvement) ===
    print(f"\n=== Phase 1: Alternating LP ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 250.0)
    if p1_budget > 30:
        a_lp = alternating_lp(best_seq, max_iters=25, time_budget=p1_budget, verbose=True, blend=1.0)
        obj_lp = 2.0 * a_lp.size * float(np.max(compute_conv(a_lp / a_lp.sum())))
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  AltLP: obj={best_score:.10f} **")
        else:
            print(f"  AltLP: obj={obj_lp:.10f}")
            # Try blends
            for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
                a_blend = (1.0 - t) * best_seq + t * a_lp
                a_blend = np.maximum(a_blend, 0.0)
                a_blend = a_blend / a_blend.sum()
                obj_blend = 2.0 * a_blend.size * float(np.max(compute_conv(a_blend)))
                if obj_blend < best_score:
                    best_score = obj_blend
                    best_seq = a_blend.copy()
                    print(f"  AltLP blend t={t}: obj={best_score:.10f} **")

    # === Phase 2: Coordinate Descent (main polish) ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 500.0)
    if p2_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p2_budget, verbose=True, seed=seed)
        obj_cd = 2.0 * a_cd.size * float(np.max(compute_conv(a_cd / a_cd.sum())))
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.10f} **")
        else:
            print(f"  CD: obj={obj_cd:.10f} (no improvement)")

    # === Phase 3: AltLP Round 2 (after CD polish) ===
    print(f"\n=== Phase 3: AltLP Round 2 ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 130.0)
    if p3_budget > 20:
        a_lp2 = alternating_lp(best_seq, max_iters=20, time_budget=p3_budget, verbose=True, blend=1.0)
        obj_lp2 = 2.0 * a_lp2.size * float(np.max(compute_conv(a_lp2 / a_lp2.sum())))
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  AltLP2: obj={best_score:.10f} **")
        else:
            print(f"  AltLP2: obj={obj_lp2:.10f}")

        # Quick CD after AltLP
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_cd2 = coordinate_descent(a_lp2, time_budget=min(remaining * 0.15, 60.0),
                                       verbose=False, seed=seed + 111)
            obj_cd2 = 2.0 * a_cd2.size * float(np.max(compute_conv(a_cd2 / a_cd2.sum())))
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  CD after AltLP2: obj={best_score:.10f} **")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current - 50, n_current + 50,
        n_current - 100, n_current + 100,
        n_current + 200,
        700, 850, 900, 1000, 1100
    ]))
    n_options = [n for n in n_options if 500 <= n <= 2000 and abs(n - n_current) > 20]
    n_options = n_options[:6]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 70:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Coordinate descent
        alloc = min(remaining * 0.15, 60.0)
        if alloc > 15:
            a_try = coordinate_descent(a_try, time_budget=alloc, verbose=False, seed=n_try)

        # Quick AltLP
        remaining2 = deadline - time.monotonic()
        alloc2 = min(remaining2 * 0.08, 25.0)
        if alloc2 > 10:
            a_try = alternating_lp(a_try, max_iters=5, time_budget=alloc2, verbose=False)

        obj_try = 2.0 * a_try.size * float(np.max(compute_conv(a_try / a_try.sum())))
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 5: Multiplicative Weight Updates ===
    print(f"\n=== Phase 5: Multiplicative Updates ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.12, 70.0)
        a_mw = multiplicative_descent(best_seq, time_budget=alloc, eta_init=0.05,
                                      eta_decay=0.998, verbose=False)
        obj_mw = 2.0 * a_mw.size * float(np.max(compute_conv(a_mw / a_mw.sum())))
        if obj_mw < best_score:
            best_score = obj_mw
            best_seq = a_mw.copy()
            print(f"  MW: obj={best_score:.10f} **")
        else:
            print(f"  MW: obj={obj_mw:.10f}")
            # Try blend
            for t in [0.1, 0.2, 0.3]:
                a_bl = (1 - t) * best_seq + t * a_mw
                a_bl = np.maximum(a_bl, 0)
                a_bl = a_bl / a_bl.sum()
                obj_bl = 2.0 * a_bl.size * float(np.max(compute_conv(a_bl)))
                if obj_bl < best_score:
                    best_score = obj_bl
                    best_seq = a_bl.copy()
                    print(f"  MW blend t={t}: obj={best_score:.10f} **")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = 2 + r
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend_frac = 0.5 + 0.1 * r
        a_r = (1.0 - blend_frac) * a_r + blend_frac * best_seq
        a_r = np.maximum(a_r, 0.0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 60.0)
        if alloc > 20:
            a_r = coordinate_descent(a_r, time_budget=alloc, verbose=False, seed=5000 + r * 100)
            obj_r = 2.0 * a_r.size * float(np.max(compute_conv(a_r / a_r.sum())))
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.10f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.10f}")

    # === Phase 7: Final CD Polish ===
    print(f"\n=== Phase 7: Final CD Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 150.0)
        a_final = coordinate_descent(best_seq, time_budget=alloc, verbose=False, seed=seed + 9999)
        obj_final = 2.0 * a_final.size * float(np.max(compute_conv(a_final / a_final.sum())))
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.10f} **")
        else:
            print(f"  Final CD: obj={obj_final:.10f}")

    # === Phase 8: Final AltLP ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_final = min(remaining * 0.5, 50.0)
        a_lp_final = alternating_lp(best_seq, max_iters=8, time_budget=alloc_final, verbose=False)
        obj_final = 2.0 * a_lp_final.size * float(np.max(compute_conv(a_lp_final / a_lp_final.sum())))
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_lp_final.copy()
            print(f"  Final AltLP: obj={best_score:.10f} **")
        else:
            print(f"  Final AltLP: obj={obj_final:.10f}")

        # Tiny final CD
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_tiny = coordinate_descent(best_seq, time_budget=min(remaining * 0.5, 20.0),
                                        verbose=False, seed=seed + 12345)
            obj_tiny = 2.0 * a_tiny.size * float(np.max(compute_conv(a_tiny / a_tiny.sum())))
            if obj_tiny < best_score:
                best_score = obj_tiny
                best_seq = a_tiny.copy()
                print(f"  Tiny CD: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
