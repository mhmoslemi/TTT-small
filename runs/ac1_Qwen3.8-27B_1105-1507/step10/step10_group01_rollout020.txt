import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix_fast(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A

def solve_lp_minpeak(a, max_time=30.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_fast(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def projection_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    # Numerically stable log-sum-exp
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if new_n == 1:
        return np.array([float(np.sum(a))])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use cubic spline for smoother resampling
    from scipy.interpolate import CubicSpline
    cs = CubicSpline(x_old, a, bc_type='natural')
    a_new = cs(x_new)
    # Ensure non-negativity
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated adaptive smoothing gradient descent (Nesterov), LP-driven
iterations for smaller n, peak equalization, multi-n exploration with diverse
parametric initializations, and aggressive restarts to escape local minima.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
from scipy.interpolate import CubicSpline
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_fast(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_minpeak(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_fast(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def projection_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, s / n)
    rho_val = rho[-1]
    theta = (cssv[rho_val] - s) / (rho_val + 1.0)
    return np.maximum(v - theta, 0)


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -60, 60)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def gd_optimize(a, time_budget=100.0, tau_init=30.0, tau_end=500.0, verbose=False, beta=0.9):
    """
    Nesterov accelerated gradient descent with adaptive smoothing and BB steps.
    Operates on the simplex (a >= 0, sum(a) = 1).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    base_lr = 8.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Exponential tau schedule with slight noise
        tau = tau_init * (tau_end / tau_init) ** progress

        y = a + beta * v
        y = projection_simplex(y, 1.0)

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            target_lr = 1.0 / (g_norm + 1e-10)
            target_lr = min(max(target_lr, 0.2 * base_lr), 8.0 * base_lr)
            lr = 0.85 * lr + 0.15 * target_lr

        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                bb_lr = min(max(bb_lr, 1e-8), 15.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = projection_simplex(a_new, 1.0)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 3000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.4
            stall_count = 0

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-12, lr * 0.99995)

    if verbose:
        print(f"    GD(n={n}): obj={best_obj:.8f}, iters={it}, tau_end={tau_end:.0f}")
    return best_a


def lp_driven_optimize(a, time_budget=100.0, verbose=False, max_iters=150):
    """
    LP-driven optimization with line search and perturbation on stall.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall_count = 0
    iter_count = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget and iter_count < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g, M = solve_lp_minpeak(a)
        if g is None:
            break

        iter_count += 1

        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_try = (1 - t) * a + t * g
            a_try = projection_simplex(a_try, 1.0)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-12:
            a = (1 - best_t) * a + best_t * g
            a = projection_simplex(a, 1.0)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall_count = 0
            if verbose and (iter_count % 5 == 0 or iter_count == 1):
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.2f}, M={M:.6f}")
        else:
            stall_count += 1
            if stall_count >= 6:
                tau = 400.0
                conv_p = compute_conv(a)
                max_p = float(np.max(conv_p))
                shifted = tau * (conv_p - max_p)
                shifted = np.clip(shifted, -50, 50)
                exp_p = np.exp(shifted)
                p = exp_p / exp_p.sum()
                N_fft = 1
                while N_fft < 3 * n:
                    N_fft <<= 1
                P = fft(p, N_fft)
                Af = fft(a, N_fft)
                grad = 2.0 * ifft(P * np.conj(Af)).real[:n]
                lr_pert = 0.1 / (float(np.linalg.norm(grad)) + 1e-10)
                a = a - lr_pert * grad
                a = a + 0.02 * rng.standard_normal(n)
                a = projection_simplex(a, 1.0)
                stall_count = 0
                if iter_count > 40:
                    if verbose:
                        print(f"    LP: stopping after {iter_count} iterations, obj={best_obj:.8f}")
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic spline interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if new_n == 1:
        return np.array([float(np.sum(a))])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    try:
        cs = CubicSpline(x_old, a, bc_type='natural')
        a_new = cs(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def parametric_init(n, rng):
    """Generate diverse parametric initializations."""
    strategies = []

    for alpha in [-2, -1.5, -1, -0.5, 0, 0.5, 1, 1.5, 2, 2.5]:
        j = np.arange(n, dtype=np.float64)
        a = (j + 1) ** alpha
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            strategies.append(a / s)

    for sigma in [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
        j = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        a = np.exp(-0.5 * ((j - center) / (sigma * n)) ** 2)
        strategies.append(a / a.sum())

    for freq_mult in [1, 2, 3, 5, 7]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            j = np.arange(n, dtype=np.float64)
            a = 1.0 + amp * np.cos(2 * np.pi * freq_mult * j / n)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 0:
                strategies.append(a / s)

    for _ in range(5):
        a = rng.exponential(1.0, n)
        s = a.sum()
        if s > 0:
            strategies.append(a / s)

    for ratio in [0.15, 0.2, 0.3, 0.35, 0.4]:
        j = np.arange(n, dtype=np.float64)
        c1 = ratio * (n - 1)
        c2 = (1 - ratio) * (n - 1)
        a = np.exp(-((j - c1) / (0.08 * n)) ** 2) + np.exp(-((j - c2) / (0.08 * n)) ** 2)
        strategies.append(a / a.sum())

    # Single spike in the middle
    a_spike = np.zeros(n)
    a_spike[n // 2] = 1.0
    strategies.append(a_spike)

    # Edge concentrated
    a_edge = np.zeros(n)
    a_edge[0] = 1.0
    strategies.append(a_edge)

    return strategies


def fourier_perturb(a, rng, strength=0.05):
    """Add low-frequency Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def peak_equalization(a, n_iters=50, verbose=False):
    """
    Iteratively reduce the top peaks of the autocorrelation.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(n_iters):
        conv = compute_conv(a)
        max_val = float(np.max(conv))

        sort_idx = np.argsort(conv)[::-1]
        top_k = min(20, len(conv))
        top_peaks = sort_idx[:top_k]

        weights = conv[top_peaks] ** 3
        w_sum = weights.sum()
        if w_sum < 1e-30:
            break
        weights = weights / w_sum

        grad = np.zeros(n)
        for w, k in zip(weights, top_peaks):
            k_int = int(k)
            j_max = min(n - 1, k_int)
            j_min = max(0, k_int - n + 1)
            for j in range(j_min, j_max + 1):
                grad[j] += 2.0 * w * a[k_int - j]

        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break

        for lr_mult in [0.5, 0.3, 0.1, 0.05, 0.01]:
            lr = lr_mult / g_norm
            a_new = a - lr * grad
            a_new = projection_simplex(a_new, 1.0)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                break

    if verbose:
        print(f"    PeakEq(n={n}): obj={best_obj:.8f}")
    return best_a


def optimize_at_n(a_init, n, time_budget, use_lp=True, verbose=False):
    """
    Full optimization pipeline for a given length n:
    GD -> (LP if use_lp) -> peak_eq -> GD polish
    """
    remaining = time_budget
    a = a_init.copy()

    # Phase A: Main GD
    gd_alloc = remaining * 0.45
    a = gd_optimize(a, time_budget=gd_alloc, tau_init=20, tau_end=400, verbose=verbose)
    remaining = time_budget - (time.monotonic() - (time.monotonic() - time_budget))

    # Phase B: LP (if feasible)
    if use_lp and n <= 1500:
        lp_alloc = time_budget * 0.25
        elapsed_so_far = time.monotonic() - _optimize_start
        if time_budget - elapsed_so_far > 15:
            a = lp_driven_optimize(a, time_budget=lp_alloc, verbose=verbose)

    # Phase C: Peak equalization
    a = peak_equalization(a, n_iters=30, verbose=verbose)

    # Phase D: Final GD polish with high tau
    gd2_alloc = time_budget * 0.2
    a = gd_optimize(a, time_budget=gd2_alloc, tau_init=100, tau_end=1500, verbose=verbose)

    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-n exploration with FFT-GD, LP iterations,
    peak equalization, diverse parametric inits, and aggressive restarts.
    """
    global _optimize_start
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if parent is None:
        n_def = 800
        parent = np.ones(n_def) / n_def
        best_score = compute_obj(parent)
        best_seq = parent.copy()
        print(f"Default init (n={n_def}): obj={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Quick optimization of parent ===
    print(f"\n=== Phase 1: Optimize parent (n={n_parent}) ===")
    _optimize_start = time.monotonic()
    alloc_p1 = min((deadline - time.monotonic()) * 0.12, 120.0)

    a1 = lp_driven_optimize(parent, time_budget=alloc_p1 * 0.5, verbose=True)
    obj1 = compute_obj(a1)
    if obj1 < best_score:
        best_score = obj1
        best_seq = a1.copy()
        print(f"  LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    alloc_gd = min(remaining * 0.1, 80.0)
    a1g = gd_optimize(a1, time_budget=alloc_gd, tau_init=40, tau_end=600, verbose=True)
    obj1g = compute_obj(a1g)
    if obj1g < best_score:
        best_score = obj1g
        best_seq = a1g.copy()
        print(f"  GD: obj={best_score:.8f} **")

    a1p = peak_equalization(best_seq, n_iters=30, verbose=False)
    obj1p = compute_obj(a1p)
    if obj1p < best_score:
        best_score = obj1p
        best_seq = a1p.copy()
        print(f"  PeakEq: obj={best_score:.8f} **")

    n_best = len(best_seq)
    print(f"  After Phase 1: n={n_best}, obj={best_score:.8f}")

    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()

    # Build list of n values to try
    n_candidates = set()
    n_candidates.add(n_best)
    n_candidates.add(n_best + 200)
    n_candidates.add(n_best + 400)
    n_candidates.add(n_best + 600)
    n_candidates.add(n_best + 1000)
    n_candidates.add(1200)
    n_candidates.add(1500)
    n_candidates.add(2000)
    n_candidates.add(2500)
    n_candidates.add(3000)
    n_candidates.add(400)
    n_candidates.add(600)

    n_list = sorted([n for n in n_candidates if 300 <= n <= 3500])
    # Prioritize: near-best first, then larger, then smaller
    n_list.sort(key=lambda x: (abs(x - n_best), -x))
    n_list = n_list[:10]
    print(f"  n options: {n_list}")

    for idx, n_try in enumerate(n_list):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        print(f"\n  --- n={n_try} (remaining={remaining:.0f}s) ---")

        # Generate initializations
        inits = []
        inits.append(resample_sequence(best_seq, n_try))
        inits.append(fourier_perturb(resample_sequence(best_seq, n_try), rng, strength=0.06))

        param_inits = parametric_init(n_try, rng)
        chosen = rng.choice(len(param_inits), size=min(4, len(param_inits)), replace=False)
        for c in chosen:
            inits.append(param_inits[c])

        time_per_init = max(15, (remaining * 0.8) / len(inits))

        for init_idx, a_init in enumerate(inits):
            remaining = deadline - time.monotonic()
            if remaining < 35:
                break

            _optimize_start = time.monotonic()
            alloc = min(time_per_init, remaining * 0.35)

            a_cur = gd_optimize(a_init, time_budget=alloc * 0.5,
                               tau_init=20, tau_end=300, verbose=False)

            remaining = deadline - time.monotonic()
            if n_try <= 1400 and remaining > 20:
                a_cur = lp_driven_optimize(a_cur,
                                          time_budget=min(alloc * 0.3, remaining * 0.3),
                                          verbose=False)

            a_cur = peak_equalization(a_cur, n_iters=20, verbose=False)

            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_cur = gd_optimize(a_cur,
                                   time_budget=min(alloc * 0.3, remaining * 0.3),
                                   tau_init=80, tau_end=800, verbose=False)

            obj_cur = compute_obj(a_cur)
            if obj_cur < best_score:
                best_score = obj_cur
                best_seq = a_cur.copy()
                n_best = n_try
                print(f"    ** init{init_idx}, n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"    init{init_idx}, n={n_try}: obj={obj_cur:.8f}")

    # === Phase 3: Intensive refinement ===
    print(f"\n=== Phase 3: Intensive Refinement ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    print(f"  Best: n={n_final}, obj={best_score:.8f}, remaining={remaining:.0f}s")

    if n_final <= 1500 and remaining > 30:
        alloc = min(remaining * 0.2, 120.0)
        a_r = lp_driven_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.25, 150.0)
        a_r = gd_optimize(best_seq, time_budget=alloc,
                         tau_init=50, tau_end=2000, verbose=True)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  GD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_r = peak_equalization(best_seq, n_iters=80, verbose=False)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  PeakEq: obj={best_score:.8f} **")

    # === Phase 4: Diverse restarts ===
    print(f"\n=== Phase 4: Restarts ===")
    remaining = deadline - time.monotonic()

    methods = ["fourier", "smooth", "mirror", "random", "spike_add"]

    for method in methods:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        n_cur = len(best_seq)

        if method == "fourier":
            a_r = fourier_perturb(best_seq, rng, strength=0.08)
        elif method == "smooth":
            ks = max(5, n_cur // 15)
            if ks % 2 == 0:
                ks += 1
            a_smooth = uniform_filter1d(best_seq, size=ks)
            a_r = 0.6 * best_seq + 0.4 * a_smooth
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        elif method == "mirror":
            a_r = (best_seq + best_seq[::-1]) / 2.0
            a_r = a_r / a_r.sum()
        elif method == "random":
            a_r = 0.7 * best_seq + 0.3 * (rng.uniform(0, 1, n_cur) / n_cur)
            a_r = a_r / a_r.sum()
        else:  # spike_add
            a_r = best_seq.copy()
            pos = rng.integers(0, n_cur)
            a_r[pos] += 0.1
            a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 60.0)
        a_r = gd_optimize(a_r, time_budget=alloc * 0.6, tau_init=40, tau_end=500, verbose=False)

        remaining = deadline - time.monotonic()
        if len(a_r) <= 1400 and remaining > 15:
            a_r = lp_driven_optimize(a_r,
                                    time_budget=min(alloc * 0.3, remaining * 0.3),
                                    verbose=False)

        a_r = peak_equalization(a_r, n_iters=20, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart({method}): obj={best_score:.8f} **")
        else:
            print(f"  Restart({method}): obj={obj_r:.8f}")

    # === Phase 5: Final high-tau GD ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_final = gd_optimize(best_seq,
                             time_budget=min(remaining * 0.4, 80.0),
                             tau_init=300, tau_end=5000, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final high-tau GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
