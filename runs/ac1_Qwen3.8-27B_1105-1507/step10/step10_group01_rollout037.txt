def compute_conv(a):

    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):

def peak_shave(a, tau=100.0, top_k=20, strength=0.05):
    """Targeted peak shaving: reduce mass where peaks are high."""
    n = len(a)
    conv = compute_conv(a)
    
    # Get top peaks
    sorted_idx = np.argsort(conv)[-top_k:]
    max_val = conv[sorted_idx[0]]
    
    # Compute "peak pressure" on each coordinate
    pressure = np.zeros(n)
    for k in sorted_idx:
        w = np.exp(tau * (conv[k] - max_val))
        # Contribution to pressure from peak k
        # d(conv[k])/d(a[j]) = 2*a[k-j] (with bounds)
        for j in range(n):
            k_minus_j = k - j
            if 0 <= k_minus_j < n:
                pressure[j] += w * a[k_minus_j]
    
    # Normalize pressure
    p_sum = float(np.sum(pressure))
    if p_sum > 1e-10:
        pressure = pressure / p_sum
    
    # Reduce a where pressure is high

    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:

def gaussian_init(n, width_ratio=0.3):
    """Initialize with a Gaussian-shaped sequence."""
    x = np.arange(n)
    center = n / 2.0
    width = width_ratio * n
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s


def triangular_init(n, peak_pos=0.5):
    """Initialize with a triangular sequence."""
    x = np.arange(n) / (n - 1)
    a = np.maximum(0, 1 - np.abs(x - peak_pos) / max(peak_pos, 1-peak_pos))
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def arcosh_init(n, asymmetry=0.3):
    """Anti-symmetric initialization to reduce center peak."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    # Base: uniform
    a = np.ones(n)
    # Add anti-symmetric component
    t = (x - center) / ((n - 1) / 2.0 + 1e-10)
    a += asymmetry * t  # This makes a[i] + a[n-1-i] roughly constant
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def multi_scale_optimize(a_init, n_target, time_budget, verbose=False):
    """Optimize at multiple scales: coarse to fine."""
    n = n_target
    
    # Start at coarse scale
    scales = [max(64, n // 8), max(128, n // 4), max(256, n // 2), n]
    scales = sorted(set([s for s in scales if s <= n]))
    
    a_current = resample_sequence(a_init, scales[0])
    
    for i, scale_n in enumerate(scales):
        remaining = time_budget - (time.monotonic() - (multi_scale_optimize._start))
        if remaining < 5:
            break
        alloc = remaining * (0.2 + 0.15 * i + 0.05 * i * i)
        alloc = min(alloc, remaining * 0.4)
        
        if scale_n < n:
            # Upsample from previous scale
            if i > 0:
                a_current = resample_sequence(a_current, scale_n)
        
        a_current = gd_optimize(a_current, time_budget=alloc, 
                                 tau_init=50.0, tau_end=2000.0,
                                 top_k="adaptive", verbose=verbose)
    
    return a_current


def local_search(a, time_budget, rng, verbose=False):
    """Local random search: try small perturbations and keep improvements."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    step_size = 0.02
    
    while time.monotonic() - start < time_budget:
        # Random perturbation
        perturb = rng.normal(0, step_size / np.sqrt(n), n)
        a_try = best_a + perturb
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            step_size = min(step_size * 1.2, 0.1)
        else:
            step_size = max(step_size * 0.95, 0.001)
    
    if verbose:
        print(f"      LocalSearch: obj={best_obj:.8f}")
    return best_a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive subgradient with peak-targeting, high-tau smoothing, momentum,
multi-scale optimization, diverse initializations, and local search.
Key differences from parent: no LP dependency, pure subgradient with much higher tau,
peak-targeting strategy, multi-n search, and coordinate-level local search.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_subgradient(a, tau, top_k=None):
    """
    Compute subgradient of smoothed max of conv(a,a)/sum(a)^2.
    Since objective is scale-invariant, we work with sum(a)=1 and
    minimize 2n * max(conv(a,a)).
    
    Returns (gradient, conv) where gradient is the subgradient of 
    sum_k p[k]*conv_k with p from softmax(tau*conv).
    """
    n = len(a)
    conv = compute_conv(a)
    conv_len = len(conv)
    
    if top_k is not None and top_k < conv_len:
        sorted_idx = np.argsort(conv)[-top_k:]
        weights = np.zeros(conv_len)
        max_val = conv[sorted_idx[0]]
        for idx in sorted_idx:
            weights[idx] = np.exp(tau * (conv[idx] - max_val))
    else:
        max_val = float(np.max(conv))
        weights = np.exp(tau * (conv - max_val))
    
    total = float(np.sum(weights))
    if total < 1e-300:
        p = np.zeros(conv_len)
        p[int(np.argmax(conv))] = 1.0
    else:
        p = weights / total
    
    # Compute gradient: grad[j] = 2 * sum_k p[k] * a[k-j]
    # Using FFT: grad/2 = ifft(fft(p, N) * conj(fft(a_padded, N)))
    a_padded = np.zeros(2 * n - 1)
    a_padded[:n] = a
    
    N = 1
    while N < 4 * n:
        N <<= 1
    
    P = fft(p, N)
    Ap = fft(a_padded, N)
    grad_half = ifft(P * np.conj(Ap)).real[:n]
    grad = 2.0 * grad_half
    
    return grad, conv


def gd_optimize(a, time_budget, tau_init=100.0, tau_end=5000.0, top_k_mode="adaptive",
                beta_init=0.9, verbose=False, seed=42):
    """
    Gradient descent with Nesterov momentum, adaptive step size, and peak targeting.
    top_k_mode: None (all peaks), int (fixed k), "adaptive" (decreasing k)
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    it = 0
    best_it = 0
    lr = 2.0 / np.sqrt(n)
    beta = beta_init
    
    prev_a = a.copy()
    prev_g = None
    bb_counter = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        
        # Adaptive tau
        tau = tau_init + (tau_end - tau_init) * progress ** 0.8
        
        # Adaptive top_k
        if top_k_mode == "adaptive":
            if progress < 0.2:
                k_use = None
            elif progress < 0.5:
                k_use = max(20, n // 4)
            elif progress < 0.8:
                k_use = max(10, n // 10)
            else:
                k_use = max(5, n // 30)
        elif isinstance(top_k_mode, int):
            k_use = top_k_mode
        else:
            k_use = None
        
        # Nesterov momentum
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = float(np.sum(y))
        else:
            y = y / s_y
        
        # Compute subgradient
        g, conv_y = compute_subgradient(y, tau, k_use)
        
        # Project to tangent space of simplex
        g_proj = g - np.mean(g)
        g_norm = float(np.linalg.norm(g_proj))
        
        if g_norm < 1e-15:
            break
        
        # Barzilai-Borwein step size
        if prev_g is not None and it % 4 == 0:
            s_vec = y - prev_a
            y_vec = g_proj - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-18 and abs(sy) > 1e-14:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-10), 10.0 / np.sqrt(n))
                lr = 0.5 * lr + 0.5 * bb
        
        # Effective step size (normalize by gradient magnitude)
        lr_eff = lr / (g_norm + 1e-10)
        lr_eff = min(lr_eff, 0.05)
        
        # Momentum update
        v = beta * v - lr_eff * g_proj
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        # Evaluate
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        # Save for BB
        if it % 4 == 0:
            prev_a = y.copy()
            prev_g = g_proj.copy()
        
        a = a_new
        it += 1
        
        # Decay
        lr = max(1e-12, lr * 0.9995)
        beta = max(0.5, beta * 0.9999)
        
        # Reset if stuck
        if it - best_it > 3000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n)
            beta = 0.85
            best_it = it
            if verbose:
                print(f"      GD reset at it={it}, obj={best_obj:.10f}")
        
        if verbose and it % 500 == 0:
            print(f"      GD it={it}, obj={best_obj:.10f}, tau={tau:.0f}, k={k_use}, lr={lr:.5f}")
    
    if verbose:
        print(f"      GD done: obj={best_obj:.10f}, iters={it}")
    return best_a


def coordinate_descent(a, time_budget, verbose=False, seed=42):
    """
    Coordinate descent: try adjusting each coordinate to reduce max peak.
    More targeted than full gradient descent.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    conv = compute_conv(a)
    
    while time.monotonic() - start < time_budget:
        # Identify the peak
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]
        
        # Find coordinates that contribute most to this peak
        # conv[peak_idx] = sum_i a[i]*a[peak_idx-i]
        contributions = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]
        
        # Try reducing the largest contributors
        top_contrib = np.argsort(contributions)[-5:][::-1]
        
        improved = False
        for idx in top_contrib:
            # Try reducing a[idx] slightly
            for frac in [0.7, 0.8, 0.9]:
                a_try = a.copy()
                a_try[idx] *= frac
                # Redistribute to maintain sum=1
                deficit = a[idx] - a_try[idx]
                # Give deficit to coordinates with smallest value (spread out)
                other_idx = [i for i in range(n) if i != idx]
                other_idx.sort(key=lambda i: a[i])
                # Spread among the smallest
                spread_to = other_idx[:max(1, n // 10)]
                for si in spread_to:
                    a_try[si] += deficit / len(spread_to)
                
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                
                conv_try = compute_conv(a_try)
                max_try = float(np.max(conv_try))
                obj_try = 2.0 * n * max_try
                
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    conv = conv_try
                    improved = True
                    break
        
        if not improved:
            # Try increasing small coordinates (spread mass)
            small_idx = np.argsort(a)[:5]
            for idx in small_idx:
                for frac in [1.1, 1.3, 1.5]:
                    a_try = a.copy()
                    old_val = a_try[idx]
                    a_try[idx] *= frac
                    # Take from largest
                    large_idx = int(np.argmax(a_try))
                    if large_idx == idx:
                        continue
                    a_try[large_idx] -= (a_try[idx] - old_val)
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    
                    conv_try = compute_conv(a_try)
                    obj_try = 2.0 * n * float(np.max(conv_try))
                    
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        conv = conv_try
                        improved = True
                        break
        
        if not improved:
            # Random perturbation to escape
            pert = rng.normal(0, 0.01, n)
            a_try = a + pert
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                conv_try = compute_conv(a_try)
                if float(np.max(conv_try)) < float(np.max(conv)) * 1.01:
                    a = a_try
                    conv = conv_try
    
    if verbose:
        print(f"      CoordDescent: obj={best_obj:.10f}")
    return best_a


def peak_shave(a, tau=200.0, top_k=20, strength=0.03):
    """Reduce mass at positions contributing to top peaks."""
    n = len(a)
    conv = compute_conv(a)
    
    sorted_idx = np.argsort(conv)[-top_k:]
    max_val = conv[sorted_idx[0]]
    
    pressure = np.zeros(n)
    for k in sorted_idx:
        w = np.exp(tau * (conv[k] - max_val))
        for j in range(n):
            k_minus_j = k - j
            if 0 <= k_minus_j < n:
                pressure[j] += w * a[k_minus_j]
    
    max_pressure = float(np.max(pressure))
    if max_pressure > 1e-15:
        a_new = a * (1.0 - strength * pressure / max_pressure)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
    else:
        a_new = a.copy()
    
    return a_new


def gaussian_init(n, width_ratio=0.3):
    x = np.arange(n)
    center = n / 2.0
    width = width_ratio * n
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def antisymmetric_init(n, strength=0.2):
    """Initialize with anti-symmetric structure to reduce center peak."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    t = (x - center) / (center + 1e-10)
    a = 1.0 + strength * t
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def local_search(a, time_budget, rng, verbose=False):
    """Adaptive random perturbation search."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    step_size = 0.03
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        # Adaptive perturbation
        if no_improve > 50:
            step_size *= 0.8
            no_improve = 0
            if step_size < 0.001:
                break
        
        # Gaussian perturbation with structure
        freq = rng.integers(1, max(2, n // 50))
        phase = rng.uniform(0, 2 * np.pi)
        structured = np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        structured = structured - structured.mean()
        
        noise = rng.normal(0, 1, n)
        noise = noise - noise.mean()
        
        perturb = step_size * (0.5 * structured / (np.std(structured) + 1e-10) 
                               + 0.5 * noise / (np.std(noise) + 1e-10))
        
        a_try = best_a + perturb
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            step_size = min(step_size * 1.05, 0.1)
            no_improve = 0
        else:
            no_improve += 1
    
    if verbose:
        print(f"      LocalSearch: obj={best_obj:.10f}")
    return best_a

    deadline = start_time + float(budget_s) - 25.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")
    
    n_parent = len(best_seq)
    
    # === Phase 1: Main GD (long, high tau) ===
    print(f"\n=== Phase 1: Main GD (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 350.0)
    if p1_budget > 30:
        a_gd = gd_optimize(best_seq, time_budget=p1_budget,
                          tau_init=50.0, tau_end=8000.0,
                          top_k_mode="adaptive", verbose=True, seed=1)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase1 GD: obj={best_score:.10f} **")
        else:
            print(f"  Phase1 GD: obj={obj_gd:.10f} (no improvement)")
    
    # === Phase 2: Coordinate Descent ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 120.0)
    if p2_budget > 15:
        a_cd = coordinate_descent(best_seq, time_budget=p2_budget, verbose=True, seed=7)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase2 CoordDesc: obj={best_score:.10f} **")
        else:
            print(f"  Phase2 CoordDesc: obj={obj_cd:.10f} (no improvement)")
    
    # === Phase 3: Peak shaving + GD ===
    print(f"\n=== Phase 3: Peak shaving + GD ===")
    remaining = deadline - time.monotonic()
    for shave_iter in range(4):
        if remaining < 15:
            break
        a_shave = peak_shave(best_seq, tau=300.0, top_k=30, strength=0.02)
        alloc = min(remaining * 0.15, 50.0)
        a_shave = gd_optimize(a_shave, time_budget=alloc,
                             tau_init=500.0, tau_end=5000.0,
                             top_k_mode=10, verbose=False, seed=100 + shave_iter)
        obj_shave = compute_obj(a_shave)
        if obj_shave < best_score:
            best_score = obj_shave
            best_seq = a_shave.copy()
            print(f"  Phase3 shave{shave_iter}: obj={best_score:.10f} **")
        else:
            print(f"  Phase3 shave{shave_iter}: obj={obj_shave:.10f}")
        remaining = deadline - time.monotonic()
    
    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    n_candidates = sorted(set([
        n_best - 200, n_best - 100, n_best + 100, n_best + 200,
        n_best + 400, n_best + 600, n_best + 1000,
        500, 600, 700, 900, 1000, 1200, 1400, 1600, 2000
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 2500 and n != n_best]
    # Sort by distance from best, take mix
    n_candidates.sort(key=lambda n: abs(n - n_best))
    n_try_list = n_candidates[:8]
    
    n_results = []
    for n_try in n_try_list:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break
        
        a_try = resample_sequence(best_seq, n_try)
        alloc_gd = min(remaining * 0.15, 45.0)
        if alloc_gd > 10:
            a_try = gd_optimize(a_try, time_budget=alloc_gd,
                               tau_init=100.0, tau_end=5000.0,
                               top_k_mode="adaptive", verbose=False, seed=n_try % 10000)
        
        obj_try = compute_obj(a_try)
        n_results.append((n_try, obj_try, a_try))
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")
    
    # === Phase 5: Diverse initializations ===
    print(f"\n=== Phase 5: Diverse initializations ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    
    inits = [
        ("gauss_03", gaussian_init(n_cur, 0.3)),
        ("gauss_05", gaussian_init(n_cur, 0.5)),
        ("gauss_02", gaussian_init(n_cur, 0.15)),
        ("antisym_02", antisymmetric_init(n_cur, 0.2)),
        ("antisym_04", antisymmetric_init(n_cur, 0.4)),
        ("blend_anti", 0.7 * (best_seq / best_seq.sum()) + 0.3 * (antisymmetric_init(n_cur, 0.3) / antisymmetric_init(n_cur, 0.3).sum())),
    ]
    
    for name, a_init in inits:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        
        a_init = np.maximum(a_init, 0)
        s_init = float(np.sum(a_init))
        if s_init < 1e-10:
            continue
        a_init = a_init / s_init
        
        alloc_gd = min(remaining * 0.12, 40.0)
        if alloc_gd < 8:
            break
        a_opt = gd_optimize(a_init, time_budget=alloc_gd,
                           tau_init=100.0, tau_end=4000.0,
                           top_k_mode="adaptive", verbose=False, seed=hash(name) % 10000)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Init ({name}): obj={best_score:.10f} **")
        else:
            print(f"  Init ({name}): obj={obj_opt:.10f}")
    
    # === Phase 6: Second main GD (higher tau) ===
    print(f"\n=== Phase 6: Second main GD (high tau) ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.35, 250.0)
        a_gd2 = gd_optimize(best_seq, time_budget=alloc,
                           tau_init=500.0, tau_end=10000.0,
                           top_k_mode="adaptive", verbose=True, seed=999)
        obj_gd2 = compute_obj(a_gd2)
        if obj_gd2 < best_score:
            best_score = obj_gd2
            best_seq = a_gd2.copy()
            print(f"  Phase6 GD: obj={best_score:.10f} **")
        else:
            print(f"  Phase6 GD: obj={obj_gd2:.10f} (no improvement)")
    
    # === Phase 7: Coordinate descent again ===
    print(f"\n=== Phase 7: Second coordinate descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 80.0)
        a_cd2 = coordinate_descent(best_seq, time_budget=alloc, verbose=True, seed=42)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Phase7 CoordDesc: obj={best_score:.10f} **")
    
    # === Phase 8: Local search polish ===
    print(f"\n=== Phase 8: Local search polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 80.0)
        a_ls = local_search(best_seq, time_budget=alloc, rng=rng, verbose=True)
        obj_ls = compute_obj(a_ls)
        if obj_ls < best_score:
            best_score = obj_ls
            best_seq = a_ls.copy()
            print(f"  LocalSearch: obj={best_score:.10f} **")
    
    # === Phase 9: Final ultra-high tau GD ===
    print(f"\n=== Phase 9: Final ultra-high tau GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 60.0)
        a_final = gd_optimize(best_seq, time_budget=alloc,
                             tau_init=2000.0, tau_end=15000.0,
                             top_k_mode=3, verbose=True, seed=777)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final: obj={best_score:.10f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    # Clip to valid range
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    
    return result
