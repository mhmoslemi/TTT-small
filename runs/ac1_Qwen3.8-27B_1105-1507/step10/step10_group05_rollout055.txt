def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_a_grad(a, k):
    """Gradient of conv(a,a)[k] with respect to a. Efficient O(n) computation."""
    n = len(a)
    grad = np.zeros(n)
    i_vals = np.arange(n)
    j_vals = k - i_vals
    valid = (j_vals >= 0) & (j_vals < n)
    grad[i_vals[valid]] = 2.0 * a[j_vals[valid]]
    return grad

def solve_lp_alternating(a, use_warm_start=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, optimal_M, dual_weights) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix A such that A @ g = conv(a, g)
    # A has shape (2n-1, n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm[:n]  # a[j+i] at position j+i for i=0..n-1
        # Actually conv(a,g)[k] = sum_j a_norm[j] * g[k-j]
        # So A[k,j] = a_norm[j] if 0 <= k-j < n, i.e., j <= k <= j+n-1
        # So for column j, rows k=j..j+n-1 have value a_norm[j]
    
    # Rebuild properly
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm[j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    best_g = None
    best_M = None
    best_dual = None
    
    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    if best_g is None or M_val < best_M - 1e-12:
                        best_g = g_val
                        best_M = M_val
                        # Get dual values
                        try:
                            dual_vals = [(c.dual_value for c in c_.constraints) 
                                        for c_ in [prob.constraints[0]]]
                            if dual_vals and dual_vals[0]:
                                dv = np.array(dual_vals[0])
                                if dv is not None and len(dv) > 0:
                                    best_dual = dv
                        except:
                            best_dual = None
                    break
        except Exception:
            continue
    
    if best_g is None:
        return None, None, None
    
    return best_g, best_M, best_dual

                best_t_val = t
                best_a_local = a_try.copy()

        # Also try the momentum-extrapolated step
        a_mom = a + 0.3 * v
        a_mom = np.maximum(a_mom, 0)
        s_mom = float(np.sum(a_mom))
        if s_mom > 1e-10:
            a_mom = a_mom / s_mom
            obj_mom = compute_obj(a_mom)
            if obj_mom < best_t_obj:
                best_t_obj = obj_mom
                best_a_local = a_mom.copy()
                best_t_val = -1.0  # flag for momentum step

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a.copy()
            best_it = it
            no_improve = 0
            if verbose and it % 20 == 0:
                print(f"    AALP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, M={M_lp:.6f}, solve={solve_time:.2f}s")
        else:
            no_improve += 1
            a = best_a.copy()
            if no_improve > 20:
                # Perturb and restart momentum
                pert = a + rng.normal(0, 0.01, n)
                pert = np.maximum(pert, 0)
                s_p = float(np.sum(pert))
                if s_p > 1e-10:
                    a = pert / s_p
                v = np.zeros(n)
                prev_g = None
                no_improve = 10

    return best_a


def exact_subgradient_descent(a, time_budget=100.0, verbose=False):
    """
    Gradient descent using exact subgradients (from argmax of convolution).
    With adaptive step size and projection onto simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 5.0 / n
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_conv = float(conv[k_star])

        grad = conv_a_grad(a, k_star)
        # Full gradient of f = 2n * max_conv / S^2 (S=1 since normalized)
        # df/da_i = 2n * (grad_i - 2*max_conv)  [since S=1, dS=1]
        # Actually: f = 2n * max_conv (when S=1)
        # df/da_i = 2n * grad_i (but we need to account for normalization)
        # If we work on the simplex (sum=1), the projected gradient is:
        # grad_proj = grad - mean(grad) (for L2 projection onto simplex)
        
        obj_grad = 2.0 * n * (grad - 2.0 * max_conv)
        
        # Backtracking line search
        a_new = a - lr * obj_grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            lr *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            best_it = it
            lr = min(lr * 1.05, 20.0 / n)
        else:
            lr *= 0.7
            # Keep a, don't update

        it += 1

        if it - best_it > 500:
            a = best_a.copy()
            lr = 5.0 / n
            best_it = it
            if verbose:
                print(f"    SGD it{it}: reset, obj={best_obj:.8f}")

        if verbose and it % 200 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def smoothed_gradient_descent(a, time_budget=100.0, tau_init=50.0, tau_end=500.0, verbose=False):
    """
    Smoothed gradient descent with adaptive tau and Nesterov momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.8
    v = np.zeros(n)
    lr = 3.0 / np.sqrt(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = a.copy()
        else:
            y = y / s_y

        grad, smooth_val = smoothed_logsumexp_grad(y, tau)
        
        # Scale gradient: f = 2n * smooth_max / S^2, with S=1: f = 2n * smooth_max
        # grad of f = 2n * grad_smooth
        obj_grad = 2.0 * n * grad

        v = beta * v - lr * obj_grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            lr *= 0.5
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            best_it = it
            lr = min(lr * 1.02, 15.0 / np.sqrt(n))
        else:
            lr *= 0.85

        it += 1

        if it - best_it > 3000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n)
            best_it = it
            if verbose:
                print(f"    SmGD it{it}: reset, obj={best_obj:.8f}, tau={tau:.0f}")

        if verbose and it % 500 == 0:
            print(f"    SmGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}")

    return best_a


def coordinate_descent(a, time_budget=60.0, passes=2, verbose=False):
    """
    Cyclic coordinate descent: for each coordinate, do 1D golden-section search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for p in range(passes):
        for i in range(n):
            if time.monotonic() - start > time_budget:
                return best_a

            # 1D search for a[i]
            # Objective as function of a[i]: 2n * max(conv(a,a)) / (sum(a))^2
            # We search a[i] in [0, 3*a[i]] or [0, 2/n + a[i]]
            a_i_center = a[i]
            a_i_low = max(0, a_i_center * 0.3)
            a_i_high = max(a_i_center * 2.0, 3.0 / n)

            # Golden section search
            gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
            
            best_xi = a_i_center
            best_fi = best_obj

            for _ in range(12):
                x1 = a_i_low + gr * (a_i_high - a_i_low)
                x2 = a_i_low + (1 - gr) * (a_i_high - a_i_low)

                a[i] = x1
                s1 = float(np.sum(a))
                if s1 < 1e-10:
                    f1 = np.inf
                else:
                    a_norm = a / s1
                    f1 = compute_obj(a_norm)

                a[i] = x2
                s2 = float(np.sum(a))
                if s2 < 1e-10:
                    f2 = np.inf
                else:
                    a_norm = a / s2
                    f2 = compute_obj(a_norm)

                if f1 < f2:
                    a_i_high = x2
                    if f1 < best_fi:
                        best_fi = f1
                        best_xi = x1
                else:
                    a_i_low = x1
                    if f2 < best_fi:
                        best_fi = f2
                        best_xi = x2

            a[i] = best_xi

            # Renormalize
            s = float(np.sum(a))
            if s < 1e-10:
                a = best_a.copy()
            else:
                a = a / s

            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        if verbose:
            print(f"    CD pass {p+1}/{passes}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation on [0,1]."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, style, rng):
    """Generate structured initial sequences of length n."""
    x = np.linspace(0, 1, n)
    if style == 0:  # Uniform
        return np.ones(n) / n
    elif style == 1:  # Cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        return a / a.sum()
    elif style == 2:  # Raised cosine (bell)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        return np.maximum(a, 1e-10) / np.maximum(a, 1e-10).sum()
    elif style == 3:  # Two peaks
        a = np.exp(-20 * (x - 0.2) ** 2) + np.exp(-20 * (x - 0.8) ** 2)
        return a / a.sum()
    elif style == 4:  # Linear ramp
        a = np.linspace(0.1, 1.0, n)
        return a / a.sum()
    elif style == 5:  # Random smooth
        a = np.ones(n) + rng.normal(0, 0.5, n)
        # Apply simple smoothing
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif style == 6:  # Fourth power cosine
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500):

            s = float(np.sum(parent_arr))

                    a_lp = a_grad

                best_local_obj = obj_lp

                best_seq = best_local.copy()
                best_n = n_trial
