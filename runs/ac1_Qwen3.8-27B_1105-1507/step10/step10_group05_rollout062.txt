        # g_full[j] = g_half[j] for j < n/2
        # g_full[j] = g_half[n-1-j] for j >= n/2
        # So A @ g_full = A[:, :n/2] @ g_half + A[:, n/2:] @ g_half_mirrored
        # g_half_mirrored[j] = g_half[n/2-1-j]
        # So A @ g_full = (A[:, :n/2] + A[:, n/2:][:, ::-1]) @ g_half
        A_half = A[:, :n_half] + A[:, n_half:][:, ::-1]
        g_half = cp.Variable(n_half, nonneg=True)
        M = cp.Variable()
        constraints = [A_half @ g_half <= M, cp.sum(g_half) == 0.5]
        prob = cp.Problem(cp.Minimize(M), constraints)
    else:
        g_var = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g_var <= M, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        g_half = None  # not used
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                if symmetric and n % 2 == 0:
                    val = np.maximum(g_half.value, 0)
                    s = float(np.sum(val))
                    if s < 1e-10:
                        continue
                    g = np.zeros(n)
                    g[:n_half] = val
                    g[n_half:] = val[::-1]
                    g = g / np.sum(g)
                else:
                    val = np.maximum(g_var.value, 0)
                    s = float(np.sum(val))
                    if s < 1e-10:
                        continue
                    g = val / s
                M_val = float(M.value)
                return g, M_val
        except Exception:
            continue
    
    return None, None


def lp_descent(a, time_budget=100.0, verbose=False, symmetric=False, momentum_init=0.7):
    """LP-based descent with adaptive step size and momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    if symmetric and n % 2 == 0:
        a = (a + a[::-1]) / 2.0
        a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    prev_g = None
    momentum = 0.0
    step_size = 0.5
    it = 0
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        g_lp, M_lp = solve_lp(a, symmetric=symmetric)
        if g_lp is None:
            break
        
        # Momentum extrapolation
        if prev_g is not None:
            diff = g_lp - prev_g
            momentum = 0.8 * momentum + 0.2 * np.linalg.norm(diff)
            g_lp = g_lp + 0.2 * momentum * diff
            g_lp = np.maximum(g_lp, 0)
            if symmetric and n % 2 == 0:
                g_lp = (g_lp + g_lp[::-1]) / 2.0
            g_lp = g_lp / g_lp.sum()
        
        prev_g = g_lp.copy()
        it += 1
        
        # Adaptive step size with backtracking
        t = step_size
        a_new = a.copy()
        obj_new = best_obj
        improved = False
        
        for _ in range(20):
            a_candidate = (1 - t) * a + t * g_lp
            a_candidate = np.maximum(a_candidate, 0)
            if symmetric and n % 2 == 0:
                a_candidate = (a_candidate + a_candidate[::-1]) / 2.0
            sc = float(np.sum(a_candidate))
            if sc < 1e-10:
                t *= 0.5
                continue
            a_candidate = a_candidate / sc
            obj_c = compute_obj(a_candidate)
            if obj_c < best_obj - 1e-15 * best_obj:
                a_new = a_candidate
                obj_new = obj_c
                improved = True
                break
            t *= 0.7
        
        if improved:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step_size = min(1.0, step_size * 1.1)
            stall_count = 0
            momentum = 0.0  # Reset momentum on improvement
        else:
            stall_count += 1
            step_size *= 0.8
            if stall_count > 20:
                # Restart from best with perturbation
                rng = np.random.default_rng(it * 7919)
                a = best_a * (1 + 0.05 * rng.normal(0, 1, n))
                a = np.maximum(a, 0)
                if symmetric and n % 2 == 0:
                    a = (a + a[::-1]) / 2.0
                a = a / a.sum()
                prev_g = None
                momentum = 0.0
                step_size = 0.3
                stall_count = 0
        
        if verbose and it % 20 == 0:
            print(f"  LP it{it}: obj={best_obj:.8f}, step={step_size:.3f}, stall={stall_count}")
    
    return best_a


def lp_gradient(a, p=20):
    """
    Compute gradient of sum_k (a*a)_k^p using FFT.
    Returns gradient of the same shape as a.
    """
    n = len(a)
    conv = compute_conv(a)
    # p-like weighting
    max_c = float(np.max(conv))
    if max_c < 1e-30:
        return np.zeros(n)
    # Use a smooth max approximation: w[k] = p * conv[k]^(p-1) / max_c^(p-1)
    # This normalizes so that the max weight is p
    w = p * np.power(np.maximum(conv, 0), p - 1)
    max_w = float(np.max(w))
    if max_w < 1e-300:
        return np.zeros(n)
    w = w / max_w
    
    # grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) (with bounds)
    # Using FFT:
    N = 1
    while N < 3 * n:
        N <<= 1
    W = fft(w, N)
    A = fft(a, N)
    # Cross-correlation: ifft(conj(W) * A) gives sum_k w[k]*a[i-k]
    # Convolution: ifft(W * A) gives sum_k w[k]*a[i-k]... 
    # Actually: ifft(W * A) = circ_conv(w, a)
    # ifft(conj(W) * A) = circ_xcorr(w, a)
    conv_part = ifft(W * A).real  # sum_k w[k] * a[i-k+n] (circular)
    xcorr_part = ifft(np.conj(W) * A).real  # sum_k w[k] * a[i+k] (circular)
    # For N >= 3n, no wraparound for the linear case
    grad = 2.0 * conv_part[:n]  # Approximation: when w is symmetric, conv ≈ xcorr
    # More precisely:
    grad = (conv_part[:n] + xcorr_part[:n])
    return grad


def adam_lp_descent(a, time_budget=100.0, p_start=10, p_end=200, verbose=False):
    """
    Adam gradient descent on sum_k (a*a)_k^p with p increasing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 1.0 / np.sqrt(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Log-linear p schedule
        log_p = math.log(p_start) + (math.log(p_end) - math.log(p_start)) * progress
        p = math.exp(log_p)
        
        g = lp_gradient(a, p=int(p))
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        
        # Learning rate decay
        lr = lr * 0.9999 + (0.01 / np.sqrt(n)) * (1 - 0.9999)
        
        if verbose and it % 10000 == 0:
            print(f"    Adam it{it}: obj={best_obj:.8f}, p={p:.0f}, lr={lr:.6f}")
        
        if it - best_it > 50000:
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            best_it = it
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturbation(a, rng, strength=0.05):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(10, n // 10)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.2, 1.0) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def build_sym_conv_matrix(a):
    """Build matrix for symmetric LP: (2n-1 x n/2) such that A_sym @ g_half = conv(a, g_full)."""
    n = len(a)
    n_half = n // 2
    m = 2 * n - 1
    A = build_conv_matrix(a)
    # g_full[j] = g_half[j] for j < n/2, g_full[j] = g_half[n-1-j] for j >= n/2
    # A @ g_full = A[:, :n/2] @ g_half + A[:, n/2:] @ g_half_reversed
    # = (A[:, :n/2] + A[:, n/2:][:, ::-1]) @ g_half
    A_sym = A[:, :n_half] + A[:, n_half:][:, ::-1]
    return A_sym


def solve_lp(a, symmetric=False):
    """
    Solve LP: min M s.t. A_a @ g <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)

    if symmetric and n % 2 == 0:
        n_half = n // 2
        A_sym = build_sym_conv_matrix(a_norm)
        g_half_var = cp.Variable(n_half, nonneg=True)
        M = cp.Variable()
        constraints = [A_sym @ g_half_var <= M, cp.sum(g_half_var) == 0.5]
        prob = cp.Problem(cp.Minimize(M), constraints)
    else:
        A = build_conv_matrix(a_norm)
        g_var = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g_var <= M, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        g_half_var = None

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                if symmetric and n % 2 == 0:
                    val = np.maximum(g_half_var.value, 0)
                    s = float(np.sum(val))
                    if s < 1e-10:
                        continue
                    g = np.zeros(n)
                    g[:n // 2] = val
                    g[n // 2:] = val[::-1]
                    g = g / np.sum(g)
                else:
                    val = np.maximum(g_var.value, 0)
                    s = float(np.sum(val))
                    if s < 1e-10:
                        continue
                    g = val / s
                M_val = float(M.value)
                return g, M_val
        except Exception:
            continue

    return None, None


def lp_descent(a, time_budget=100.0, verbose=False, symmetric=False, seed=42):
    """LP-based descent with adaptive step size and momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    if symmetric and n % 2 == 0:
        a = (a + a[::-1]) / 2.0
        a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    prev_g = None
    momentum = 0.0
    step_size = 0.5
    it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = solve_lp(a, symmetric=symmetric)
        if g_lp is None:
            break

        # Momentum extrapolation
        if prev_g is not None:
            diff = g_lp - prev_g
            momentum = 0.7 * momentum + 0.3 * np.linalg.norm(diff)
            if momentum > 1e-8:
                g_lp = g_lp + 0.25 * momentum * diff
                g_lp = np.maximum(g_lp, 0)
                if symmetric and n % 2 == 0:
                    g_lp = (g_lp + g_lp[::-1]) / 2.0
                gs = float(np.sum(g_lp))
                if gs > 1e-10:
                    g_lp = g_lp / gs

        prev_g = g_lp.copy()
        it += 1

        # Adaptive step size with backtracking (Armijo-like)
        t = step_size
        improved = False
        for _ in range(25):
            a_candidate = (1 - t) * a + t * g_lp
            a_candidate = np.maximum(a_candidate, 0)
            if symmetric and n % 2 == 0:
                a_candidate = (a_candidate + a_candidate[::-1]) / 2.0
            sc = float(np.sum(a_candidate))
            if sc < 1e-10:
                t *= 0.5
                continue
            a_candidate = a_candidate / sc
            obj_c = compute_obj(a_candidate)
            if obj_c < best_obj / (1 + 1e-10):
                a = a_candidate
                best_obj = obj_c
                best_a = a_candidate.copy()
                improved = True
                break
            t *= 0.7

        if improved:
            step_size = min(1.0, step_size * 1.15)
            stall_count = 0
        else:
            stall_count += 1
            step_size *= 0.7
            if stall_count > 25:
                # Perturb and restart

                a = best_a * (1 + strength * rng.normal(0, 1, n))
                a = np.maximum(a, 0)
                if symmetric and n % 2 == 0:
                    a = (a + a[::-1]) / 2.0
                a = a / a.sum()
                prev_g = None
                momentum = 0.0
                step_size = 0.3
                stall_count = 0

        if verbose and it % 15 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, step={step_size:.3f}, stall={stall_count}, t={elapsed:.0f}s")

def lp_gradient_fft(a, p):
    """
    Compute gradient of sum_k (a*a)_k^p using FFT.
    Returns gradient array of same shape as a.
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    if max_c < 1e-30:
        return np.zeros(n)

    # Weight: w[k] = p * conv[k]^(p-1), normalized
    w = p * np.power(np.maximum(conv, 1e-30), p - 1)
    max_w = float(np.max(w))
    if max_w < 1e-300:
        return np.zeros(n)
    w = w / max_w

    # grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) with bounds
    N = 1
    while N < 3 * n:
        N <<= 1
    W = fft(w, N)
    A = fft(a, N)
    # ifft(W * A) gives circular conv: (w*a)[i] = sum_k w[k]*a[i-k]
    # ifft(conj(W) * A) gives cross-corr: sum_k w[k]*a[i+k]
    conv_part = ifft(W * A).real[:n]
    xcorr_part = ifft(np.conj(W) * A).real[:n]
    grad = conv_part + xcorr_part
    return grad


def adam_lp_descent(a, time_budget=100.0, p_start=10, p_end=500, verbose=False, seed=42):
    """
    Adam gradient descent on sum_k (a*a)_k^p with p annealing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Log-linear p schedule with slight oscillation
        log_p = math.log(p_start) + (math.log(p_end) - math.log(p_start)) * progress
        p = int(math.exp(log_p))
        p = max(p, 2)

        g = lp_gradient_fft(a, p)

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        # Learning rate schedule: decay with progress
        lr = base_lr * (0.3 + 0.7 * (1 - progress))

        if verbose and it % 5000 == 0:
            print(f"    Adam it{it}: obj={best_obj:.8f}, p={p}, lr={lr:.6f}")

        # Restart if stuck
        if it - best_it > 30000:
            a = best_a * (1 + 0.03 * rng.normal(0, 1, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            best_it = it

    return best_a


def smoothed_gd_descent(a, time_budget=100.0, tau_start=30, tau_end=2000, verbose=False, seed=42):

        return a

    best_a = a.copy()
    best_obj = compute_obj(a)

    beta = 0.85
    base_lr = 3.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:

            break

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Smoothed gradient via log-sum-exp
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        # Barzilai-Borwein
        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                bb_lr = min(max(bb_lr, 1e-8), 5.0 / np.sqrt(n))
                lr = 0.6 * lr + 0.4 * bb_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        lr = max(1e-8, lr * 0.9998)

        if verbose and it % 5000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

        if it - best_it > 20000:
            a = best_a * (1 + 0.03 * rng.normal(0, 1, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            best_it = it

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturbation(a, rng, strength=0.05):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(10, max(2, n // 10))
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.2, 1.0) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def structured_init(n, kind, rng):
    """Generate structured initializations."""
    x = np.linspace(0, 1, n)
    if kind == 0:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif kind == 1:
        # Gaussian
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif kind == 2:
        # Bump (compact support)
        a = np.maximum(1 - np.abs(2 * x - 1) * 2, 0) ** 2
    elif kind == 3:
        # Multi-bump
        a = 0.5 + 0.5 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
    else:
        # Near-uniform with noise
        a = 1.0 + 0.3 * rng.normal(0, 1, n)
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines LP descent, smoothed gradient descent,
    Lp descent, symmetric optimization, multi-length search, and diversification.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={best_seq.size}): obj={best_score:.8f}")
    except Exception:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: LP Descent (asymmetric) - main workhorse ===
    print(f"\n=== Phase 1: LP Descent (asymmetric) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.22, 280.0)
    if p1_budget > 30:
        a1 = lp_descent(best_seq, time_budget=p1_budget, verbose=True, symmetric=False, seed=42)
        obj1 = compute_obj(a1)
        print(f"  Phase 1 LP: obj={obj1:.8f}")
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase 1: ** improved to {best_score:.8f} **")

    # === Phase 2: Smoothed GD (Nesterov) ===
    print(f"\n=== Phase 2: Smoothed Gradient Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 120.0)
    if p2_budget > 20:
        a2 = smoothed_gd_descent(best_seq, time_budget=p2_budget,
                                 tau_start=30, tau_end=1500, verbose=True, seed=100)
        obj2 = compute_obj(a2)
        print(f"  Phase 2 GD: obj={obj2:.8f}")
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase 2: ** improved to {best_score:.8f} **")

    # === Phase 3: Symmetric LP ===
    print(f"\n=== Phase 3: LP Descent (symmetric) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.13, 150.0)
    if p3_budget > 30:
        a_sym = (best_seq + best_seq[::-1]) / 2.0
        a_sym = a_sym / a_sym.sum()
        a3 = lp_descent(a_sym, time_budget=p3_budget, verbose=True, symmetric=True, seed=200)
        obj3 = compute_obj(a3)
        print(f"  Phase 3 SymLP: obj={obj3:.8f}")
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase 3: ** improved to {best_score:.8f} **")

    # === Phase 4: Lp Adam Descent ===
    print(f"\n=== Phase 4: Lp Adam Descent ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.12, 120.0)
    if p4_budget > 20:
        a4 = adam_lp_descent(best_seq, time_budget=p4_budget,
                             p_start=10, p_end=500, verbose=True, seed=300)
        obj4 = compute_obj(a4)
        print(f"  Phase 4 Lp: obj={obj4:.8f}")
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase 4: ** improved to {best_score:.8f} **")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    n_best = len(best_seq)
    n_candidates = sorted(set([
        max(400, n_best - 200), max(400, n_best - 100),
        n_best + 100, n_best + 200, n_best + 400,
        600, 1000, 1200
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 2000 and n != n_best]
    n_candidates = n_candidates[:5]
    print(f"  Trying n values: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # Quick LP
        alloc1 = min(remaining * 0.1, 50.0)
        if alloc1 > 15:
            a_try = lp_descent(a_try, time_budget=alloc1, symmetric=False, seed=n_try)

        # Quick GD
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.05, 25.0)
        if alloc2 > 10:
            a_try = smoothed_gd_descent(a_try, time_budget=alloc2,
                                        tau_start=50, tau_end=500, seed=n_try + 1)

        obj_try = compute_obj(a_try)
        print(f"  n={n_try}: obj={obj_try:.8f}")
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  n={n_try}: ** improved to {best_score:.8f} **")

    # === Phase 6: Diversification with structured inits ===
    print(f"\n=== Phase 6: Diversification ===")
    n_current = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if r == 0:
            a_r = fourier_perturbation(best_seq, rng, strength=0.04)
        elif r == 1:
            a_r = best_seq * (1 + 0.08 * rng.normal(0, 1, n_current))
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        elif r == 2:
            # Blend with structured init
            a_struct = structured_init(n_current, r - 2, rng)
            a_r = 0.6 * best_seq + 0.4 * a_struct
            a_r = a_r / a_r.sum()
        elif r == 3:
            # Symmetric perturbation
            a_r = best_seq.copy()
            half = n_current // 2
            a_r[:half] = best_seq[:half] * (1 + 0.05 * rng.normal(0, 1, half))
            a_r[half:] = a_r[:half][::-1] if n_current % 2 == 0 else np.pad(a_r[:half][::-1], (0, 1))[:n_current - half]
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        elif r == 4:
            # Low-frequency dominant
            a_r = structured_init(n_current, 0, rng)
            a_r = 0.5 * best_seq + 0.5 * a_r
            a_r = a_r / a_r.sum()
        else:
            # High contrast (sharper peaks)
            a_r = best_seq ** 2
            a_r = np.maximum(a_r, 1e-6)
            a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.1, 70.0)
        if alloc > 15:
            a_r = lp_descent(a_r, time_budget=alloc, symmetric=(r % 2 == 1), seed=500 + r * 13)
            obj_r = compute_obj(a_r)
            print(f"  Div {r}: obj={obj_r:.8f}")
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Div {r}: ** improved to {best_score:.8f} **")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 50:
        alloc = min(remaining * 0.35, 180.0)
        a_final = lp_descent(best_seq, time_budget=alloc, verbose=True, symmetric=False, seed=9999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: ** {best_score:.8f} **")
        else:
            print(f"  Final LP: {obj_final:.8f} (no improvement)")

    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.25, 100.0)
        a_final = smoothed_gd_descent(best_seq, time_budget=alloc,
                                      tau_start=100, tau_end=3000, verbose=True, seed=8888)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: ** {best_score:.8f} **")
        else:
            print(f"  Final GD: {obj_final:.8f} (no improvement)")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 80.0)
        a_final = adam_lp_descent(best_seq, time_budget=alloc,
                                  p_start=50, p_end=1000, seed=7777)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Lp: ** {best_score:.8f} **")
        else:
            print(f"  Final Lp: {obj_final:.8f} (no improvement)")

    # One more LP if time permits
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 60.0)
        a_final = lp_descent(best_seq, time_budget=alloc, symmetric=False, seed=6666)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra LP: ** {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
