Soft-max gradient descent with Adam + basin hopping + multi-length exploration

Key ideas:

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_idx = np.arange(1, n + 1)
    cond = u * rho_idx > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """
    Compute gradient of f_tau(a) = 2n * tau * log(sum_k exp(conv_k/tau))
    assuming sum(a) = 1.
    
    Returns (gradient, conv_values).
    """

    conv = compute_conv(a)
    # Numerically stable softmax
    conv_shifted = conv - np.max(conv)
    p = np.exp(conv_shifted / tau)

    A = fft(a, N)

def peak_targeted_refine(a, time_budget=60.0, K=5, verbose=False):
    """
    Refine by explicitly targeting the top-K peaks.
    Uses a subgradient-like approach focusing on active constraints.
    """

    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 5.0 / np.sqrt(n)
    momentum = 0.75
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        max_val = conv[top_idx[0]]
        if max_val < 1e-15:
            break

        # Weight peaks by how close they are to the max
        weights = np.exp(8.0 * (conv[top_idx] - max_val) / max_val)
        weights /= weights.sum()

        # Compute subgradient from active peaks
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            # d(conv_k)/d(a[i]) = 2*a[k-i] (approx)
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            if j_start <= j_end:
                # g[i] += w * 2 * a[k-i] for valid i
                # i ranges from j_start to j_end, and a[k-i] is the corresponding element
                idx = np.arange(j_start, j_end + 1)
                g[idx] += 2.0 * w * a[k - idx]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.998

        it += 1

    return best_a


def lp_direction(a):

    where A_k @ g = conv(a, g)[k].

        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:

                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except Exception:
        pass
    return None


def tr_soCP_step(a, radius=0.3):
    """
    One trust-region SOCP step.
    Linearize the convolution around a:
    conv(a+delta, a+delta)[k] ≈ conv(a,a)[k] + 2*(conv(a, delta))[k]
    Minimize M s.t. conv(a,a)[k] + 2*conv(a, delta)[k] <= M, ||delta|| <= r
    """
    try:
        import cvxpy as cp
        n = len(a)
        a = a / np.sum(a)
        C = compute_conv(a)

        # Build matrix for conv(a, delta)[k] = sum_i a[i] * delta[k-i]
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            lo = j
            hi = min(2 * n - 2, j + n - 1)
            if lo <= hi:
                length = hi - lo + 1
                A[lo:hi + 1, j] = a[:length]

        delta = cp.Variable(n)
        M = cp.Variable()
        a_new = a + delta

        constraints = [
            cp.sum(a_new) == 1,
            a_new >= 0,
            C + 2 * (A @ delta) <= M,
            cp.norm(delta, 2) <= radius,
        ]

        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    return np.asarray(delta.value).flatten()
            except Exception:
                continue
    except Exception:
        pass
    return None


def tr_polish(a, time_budget=120.0, r_init=0.25, verbose=False):
    """Trust-region SOCP polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    r = r_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - start < time_budget:
        delta = tr_soCP_step(a, r)
        if delta is None:
            # Fallback to LP direction
            g = lp_direction(a)
            if g is None:
                break
            delta = g - a

        norm_d = np.linalg.norm(delta)
        if norm_d > r:
            delta = delta * (r / norm_d)

        delta = np.where(np.isfinite(delta), delta, 0.0)

        # Line search
        best_alpha = 0
        best_alpha_obj = best_obj
        for t_alpha in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0, 1.3, 1.7]:
            a_try = a + t_alpha * delta
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj - 1e-15:
                best_alpha_obj = obj_try
                best_alpha = t_alpha

        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-15:
            a = a + best_alpha * delta
            a = np.maximum(a, 0)
            s_n = a.sum()
            if s_n > 1e-10:
                a = a / s_n
            best_obj = best_alpha_obj
            best_a = a.copy()
            no_improve = 0
            r = min(r * 1.1, 1.5)
        else:
            no_improve += 1
            r *= 0.6
            if no_improve >= 4:
                if r < 0.02:
                    a = best_a * np.exp(rng.normal(0, 0.01, n))
                    a = np.maximum(a, 0)
                    s_p = a.sum()
                    if s_p > 1e-10:
                        a = a / s_p
                    r = r_init
                    no_improve = 0
                    if it > 60:
                        break
                else:
                    r = max(r * 0.5, 0.02)

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    TR it{it}: obj={best_obj:.8f}, r={r:.4f}, t={elapsed:.0f}s")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def generate_diverse_init(n, seed=42):
    """Generate multiple diverse initializations of length n."""
    rng = np.random.default_rng(seed)
    inits = []

    # 1. Uniform
    inits.append(np.ones(n) / n)

    # 2. Exponential decay
    w = np.exp(-np.linspace(0, 3, n))
    inits.append(w / w.sum())

    # 3. Triangular
    w = np.arange(1, n + 1)
    w = np.concatenate([w, w[::-1]])[:n] if n > 2 else w
    inits.append(w / w.sum())

    # 4. Gaussian (centered)
    x = np.linspace(-2, 2, n)
    w = np.exp(-x ** 2 / 2)
    inits.append(w / w.sum())

    # 5. Random Dirichlet (concentrated)
    w = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(w)

    # 6. Random Dirichlet (uniform-like)
    w = rng.dirichlet(np.ones(n) * 2.0)
    inits.append(w)

    # 7. Sinusoidal modulation
    base = np.ones(n) / n
    for freq in [1, 2, 3]:
        mod = 1.0 + 0.3 * np.cos(2 * np.pi * freq * np.arange(n) / n)
        w = base * mod
        w = np.maximum(w, 0)
        inits.append(w / w.sum())

    # 8. Two-level
    w = np.ones(n) / n
    mask = np.arange(n) % 2 == 0
    w[mask] *= 1.5
    w[~mask] *= 0.5
    inits.append(w / w.sum())

    # 9. Random smooth
    w = rng.random(n)
    # Smooth with a simple kernel
    kernel = np.array([0.2, 0.6, 0.2])
    w_smooth = np.convolve(w, kernel, mode='same')
    w_smooth = np.maximum(w_smooth, 0)
    inits.append(w_smooth / w_smooth.sum())

    # 10. Cosine window
    w = 0.5 * (1 + np.cos(np.pi * np.arange(n) / (n - 1))) if n > 1 else np.ones(n)
    inits.append(w / w.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Soft-max GD + TR-SOCP + basin hopping + multi-length exploration.
    Combines fast gradient-based local search with structured global exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj_parent = compute_obj(parent)
                best_score = obj_parent
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={n_parent}): obj={obj_parent:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_parent}): obj={best_score:.8f}")

    # === Phase 1: Diverse initialization + quick evaluation ===
    print(f"\n=== Phase 1: Divers
e Initialization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.12, 120.0)

    if p1_budget > 30:
        inits = generate_diverse_init(n_parent, seed=seed)
        # Also add parent as an initialization
        inits.append(best_seq.copy())

        for i, init in enumerate(inits):
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            alloc = min(remaining * 0.15, 30.0)
            if alloc < 5:
                break
            a_try = softmax_gd(init, time_budget=alloc,
                               tau_init=30.0, tau_end=0.1, lr=0.008)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Init {i}: obj={best_score:.8f} **")
            else:
                print(f"  Init {i}: obj={obj_try:.8f}")

    # === Phase 2: Intensive soft-max GD ===
    print(f"\n=== Phase 2: Intensive Soft-Max GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 30:
        a_sgd = softmax_gd(best_seq, time_budget=p2_budget,
                           tau_init=80.0, tau_end=0.02, lr=0.015,
                           verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase2 SMAX: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 SMAX: obj={obj_sgd:.8f}")

    # === Phase 3: TR-SOCP Polish ===
    print(f"\n=== Phase 3: TR-SOCP Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 200.0)
    if p3_budget > 20:
        a_tr = tr_polish(best_seq, time_budget=p3_budget, r_init=0.25, verbose=True)
        obj_tr = compute_obj(a_tr)
        if obj_tr < best_score:
            best_score = obj_tr
            best_seq = a_tr.copy()
            print(f"  Phase3 TR: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 TR: obj={obj_tr:.8f}")

    # === Phase 4: Peak-targeted refinement ===
    print(f"\n=== Phase 4: Peak-Targeted Refinement ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 60.0)
    if p4_budget > 10:
        # Try different K values
        for K in [3, 5, 10]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.25, 30.0)
            if alloc < 5:
                break
            a_pt = peak_targeted_refine(best_seq, time_budget=alloc, K=K)
            obj_pt = compute_obj(a_pt)
            if obj_pt < best_score:
                best_score = obj_pt
                best_seq = a_pt.copy()
                print(f"  PT K={K}: obj={best_score:.8f} **")
            else:
                print(f"  PT K={K}: obj={obj_pt:.8f}")

    # === Phase 5: Basin Hopping ===
    print(f"\n=== Phase 5: Basin Hopping ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.25, 250.0)

    if p5_budget > 20:
        n_hops = int(p5_budget // 30) + 3
        n = len(best_seq)
        hop_start = time.monotonic()

        for hop in range(n_hops):
            elapsed_hop = time.monotonic() - hop_start
            if elapsed_hop >= p5_budget:
                break

            # Vary perturbation scale
            scale = 0.008 + 0.012 * rng.random() + 0.003 * hop
            a_pert = best_seq * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert < 1e-10:
                continue
            a_pert = a_pert / s_pert

            # Allocate time for optimization
            alloc1 = min((p5_budget - elapsed_hop) * 0.3, 25.0)
            if alloc1 < 5:
                break
            a_new = softmax_gd(a_pert, time_budget=alloc1,
                               tau_init=40.0, tau_end=0.05, lr=0.01)

            # Quick TR polish
            remaining_hop = p5_budget - (time.monotonic() - hop_start)
            if remaining_hop > 15:
                alloc2 = min(remaining_hop * 0.3, 20.0)
                a_new = tr_polish(a_new, time_budget=alloc2, r_init=0.15)

            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Hop {hop}: obj={best_score:.8f} **")
            else:
                print(f"  Hop {hop}: obj={obj_new:.8f}")

    # === Phase 6: Multi-length Exploration ===
    print(f"\n=== Phase 6: Multi-Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Generate candidate lengths
    n_options = sorted(set([
        max(300, n_current - 200),
        max(300, n_current - 100),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_current) > 50]

    for n_try in n_options[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)

        # Quick soft-max GD
        alloc1 = min(remaining * 0.12, 25.0)
        if alloc1 > 5:
            a_try = softmax_gd(a_try, time_budget=alloc1,
                               tau_init=30.0, tau_end=0.1, lr=0.01)

        # TR polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.12, 40.0)
        if alloc2 > 10:
            a_try = tr_polish(a_try, time_budget=alloc2, r_init=0.2)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        # Soft-max GD (fine tuned)
        alloc = min(remaining * 0.3, 80.0)
        if alloc > 10:
            a_final = softmax_gd(best_seq, time_budget=alloc,
                                 tau_init=20.0, tau_end=0.01, lr=0.005,
                                 verbose=True)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final SMAX: obj={best_score:.8f} **")
            else:
                print(f"  Final SMAX: obj={obj_final:.8f}")

    if deadline - time.monotonic() > 20:
        # TR polish
        alloc = min((deadline - time.monotonic()) * 0.4, 60.0)
        if alloc > 10:
            a_final = tr_polish(best_seq, time_budget=alloc, r_init=0.1, verbose=True)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final TR: obj={best_score:.8f} **")
            else:
                print(f"  Final TR: obj={obj_final:.8f}")

    if deadline - time.monotonic() > 10:
        # Peak refinement
        alloc = min((deadline - time.monotonic()) * 0.3, 30.0)
        if alloc > 5:
            a_final = peak_targeted_refine(best_seq, time_budget=alloc, K=5)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final PT: obj={best_score:.8f} **")
            else:
                print(f"  Final PT: obj={obj_final:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
