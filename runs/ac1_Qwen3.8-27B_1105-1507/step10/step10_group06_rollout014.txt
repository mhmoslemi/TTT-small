"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Multiplicative Weights (MW) as the primary fast optimizer, combined with
Nesterov accelerated gradient descent on the log-sum-exp smoothed objective,
and LP-based refinement for final polishing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def mw_gradient(a, beta):
    """
    Compute the smoothed gradient for the MW algorithm.
    Uses log-sum-exp smoothing: p[k] = exp(beta*c[k]) / sum exp(beta*c[k])
    Gradient: g[j] = 2 * (p_rev * a)_j (truncated convolution)
    """
    n = len(a)
    c = compute_conv(a)  # length 2n-1
    max_c = float(np.max(c))
    exp_vals = np.exp(beta * (c - max_c))
    p = exp_vals / np.sum(exp_vals)
    
    # g[j] = sum_k p[k] * 2 * a[k-j] (where 0 <= k-j < n)
    # = 2 * (p_rev * a)[j] where p_rev[k] = p[2n-2-k]
    # Actually, let me just compute it via FFT
    # (p * a)[k] = sum_j p[j] a[k-j], we want sum_k p[k] a[k-j] for each j
    # = (p_rev * a)[j] where p_rev[i] = p[len(p)-1-i]
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    P = fft(p, N)
    A = fft(a, N)
    # conv(p, a) = ifft(P * A)
    conv_pa = ifft(P * A).real[:2*n-1]
    # We want: g[j] = 2 * sum_k p[k] * a[k-j] for valid k-j
    # This is conv(p_rev, a)[j] where p_rev[i] = p[2n-2-i]
    p_rev = p[::-1]
    P_rev = fft(p_rev, N)
    ga = ifft(P_rev * A).real[:n]
    g = 2.0 * ga
    
    return g, c


def mw_iteration(a, beta, eta):
    """One MW iteration: compute smoothed gradient, update via multiplicative weights."""
    n = len(a)
    g, c = mw_gradient(a, beta)
    a_new = a * np.exp(-eta * g)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a, np.max(c)
    a_new = a_new / s
    return a_new, float(np.max(c))


def mw_algorithm(a, time_budget, beta=100.0, eta_init=0.5, eta_decay=0.9999, verbose=False):
    """
    Multiplicative Weights algorithm for minimizing max_k (a*a)[k].
    Fast: O(n log n) per iteration.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init
    it = 0
    
    while time.monotonic() - start < time_budget:
        a_new, max_c = mw_iteration(a, beta, eta)
        a = a_new
        
        obj = 2.0 * n * max_c  # Since sum(a) = 1
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        eta *= eta_decay
        it += 1
        
        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MW iter {it}: obj={best_obj:.10f}, eta={eta:.6f}, elapsed={elapsed:.1f}s")
    
    if verbose:
        print(f"    MW done: {it} iterations, best_obj={best_obj:.10f}")
    
    return best_a


def nesterov_smoothed_gd(a, time_budget, tau_init=20.0, tau_max=500.0, verbose=False):
    """
    Nesterov accelerated gradient descent on the smoothed objective.
    f_smooth(a) = (1/tau) log(sum_k exp(tau * (a*a)[k]))
    Projected onto {a >= 0, sum(a) = 1}.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    tau = tau_init
    lr = 2.0 / (tau * np.sqrt(n))
    v = np.zeros(n)
    momentum = 0.9
    it = 0
    
    def project_simplex(v):
        """Euclidean projection onto {x >= 0, sum(x) = 1}."""
        n = len(v)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
        theta = (cssv[rho] - 1) / (rho + 1.0)
        return np.maximum(v - theta, 0)
    
    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = a + momentum * v
        y = project_simplex(y)
        
        # Compute gradient of smoothed objective
        c = compute_conv(y)
        max_c = float(np.max(c))
        exp_vals = np.exp(tau * (c - max_c))
        p = exp_vals / np.sum(exp_vals)
        
        # Gradient: g[j] = 2 * sum_k p[k] * a_y[k-j] (truncated)
        N_fft = 1
        while N_fft < 2 * n - 1:
            N_fft <<= 1
        P = fft(p, N_fft)
        Y = fft(y, N_fft)
        p_rev = p[::-1]
        P_rev = fft(p_rev, N_fft)
        g = 2.0 * ifft(P_rev * Y).real[:n]
        
        # Update
        v = momentum * v - lr * g
        a_new = project_simplex(y + v)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        tau = min(tau * 1.01, tau_max)
        lr = 2.0 / (tau * np.sqrt(n))
        it += 1
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    Nesterov iter {it}: obj={best_obj:.10f}, tau={tau:.1f}, elapsed={elapsed:.1f}s")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_best_response(a):
    """Solve LP: min M s.t. (a_norm@g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a, time_budget, verbose=False):
    """LP-based alternating minimization."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    t_blend = 0.3
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        if time_budget - (time.monotonic() - start) < 5:
            break
        
        g_lp, M_lp = lp_best_response(a)
        if g_lp is None:
            break
        it += 1
        
        best_local = best_obj
        best_local_a = best_a.copy()
        best_t = t_blend
        
        for t in [t_blend, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local:
                best_local = obj_try
                best_local_a = a_try.copy()
                best_t = t
        
        if best_local < best_obj - 1e-12:
            best_obj = best_local
            best_a = best_local_a
            a = best_a.copy()
            t_blend = max(0.05, best_t * 0.9)
            stall = 0
        else:
            stall += 1
            if stall >= 5:
                # Perturb
                rng = np.random.default_rng(it)
                a = best_a * (1 + 0.01 * rng.standard_normal(n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
        
        if verbose and it % 10 == 0:
            print(f"    LP iter {it}: obj={best_obj:.10f}")
    
    return best_a


def raised_cosine(n):
    """Raised cosine window, normalized to sum 1."""
    x = np.arange(n)
    w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    return w / w.sum()


def gaussian_window(n, sigma_frac=0.3):
    """Gaussian window, normalized to sum 1."""
    x = np.arange(n) - (n - 1) / 2
    w = np.exp(-x**2 / (2 * (sigma_frac * n)**2))
    return w / w.sum()


def triangular_window(n):
    """Triangular (Bartlett) window, normalized to sum 1."""
    x = np.arange(n)
    w = 1 - 2 * np.abs(x - (n-1)/2) / (n-1)
    w = np.maximum(w, 0)
    return w / w.sum()


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
