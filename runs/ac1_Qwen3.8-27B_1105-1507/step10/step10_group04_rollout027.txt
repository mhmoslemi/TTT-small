"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov accelerated projected gradient with log-sum-exp smoothing,
adaptive tau scheduling, multi-n exploration, diverse initializations,
and optional LP polishing. Key insight: the linear convolution (not autocorrelation)
allows non-symmetric sequences to beat the Cauchy-Schwarz bound of 2.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_grad(a, p):
    """Compute gradient of smoothed objective: grad_j = 2 * sum_k p[k]*a[k-j].
    Uses FFT-based cross-correlation."""
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    p_fft = rfft(p, N)
    a_fft = rfft(a, N)
    cross_corr = irfft(p_fft * np.conj(a_fft), N)[:n]
    return 2.0 * cross_corr


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(a):
    """Project a onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-300:
        return np.full(len(a), 1.0 / len(a))
    return a / s


def nesterov_optimize(a, time_budget, tau_start=30, tau_end=5000, verbose=False):
    """Nesterov accelerated projected gradient with log-sum-exp smoothing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov parameters
    y = a.copy()
    v = np.zeros(n)
    beta = 1.0
    lr = 1.0 / np.sqrt(n)
    it = 0
    stall = 0
    best_iter = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Adaptive tau: increase over time
        tau = tau_start + (tau_end - tau_start) * (progress ** 0.7)

        # Nesterov extrapolation
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y_new = y + ((beta - 1.0) / beta_new) * (y - v)
        y_new = project_simplex(y_new)

        # Compute gradient at y_new
        conv = compute_conv(y_new)
        max_conv = float(np.max(conv))
        shifted = tau * (conv - max_conv)
        # Numerical safety
        max_shifted = np.max(shifted)
        if max_shifted > 500:
            shifted = shifted - max_shifted + max_shifted - 500
        exp_vals = np.exp(np.clip(shifted, -500, 500))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        grad = compute_grad(y_new, p)
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break

        # Adaptive step size
        lr = min(1.0 / (g_norm + 1e-8), 2.0 / np.sqrt(n))
        lr = max(lr, 1e-10)

        # Gradient step
        new_v = y_new - lr * grad
        new_v = project_simplex(new_v)

        # Check improvement
        obj_new = compute_obj(new_v)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = new_v.copy()
            best_iter = it
            stall = 0
        else:
            stall += 1
            # If stalled, dampen the step
            new_v = project_simplex(0.9 * y + 0.1 * new_v)
            obj_new = compute_obj(new_v)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = new_v.copy()
                best_iter = it
                stall = 0

        v = y
        y = new_v
        beta = beta_new
        it += 1

        # Reset if stuck too long
        if stall > 500:
            y = best_a.copy()
            v = best_a.copy()
            beta = 1.0
            stall = 0
            if it > 50000:
                break

        if verbose and it % 2000 == 0:
            print(f"      NGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def subgradient_optimize(a, time_budget, verbose=False):
    """Pure subgradient method (no smoothing) with adaptive step sizes."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Use multiple "active" lags with softmax-like weights
    gamma = 0.1  # temperature for lag selection
    it = 0

    # Nesterov
    y = a.copy()
    v = np.zeros(n)
    beta = 1.0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Increase gamma (sharper peak selection)
        gamma = 0.01 + 0.5 * progress

        conv = compute_conv(y)
        max_conv = float(np.max(conv))

        # Select lags with softmax
        shifted = gamma * (conv - max_conv)
        exp_vals = np.exp(np.clip(shifted, -500, 0))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        grad = compute_grad(y, p)
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break

        # Step size: O(1/sqrt(t))
        lr = 1.0 / (np.sqrt(it + 1) + 1)
        lr = min(lr, 1.0 / (g_norm + 1e-8))

        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y_new = y + ((beta - 1.0) / beta_new) * (y - v)
        y_new = project_simplex(y_new)

        # Recompute at y_new
        conv2 = compute_conv(y_new)
        max_conv2 = float(np.max(conv2))
        shifted2 = gamma * (conv2 - max_conv2)
        exp_vals2 = np.exp(np.clip(shifted2, -500, 0))
        sum_exp2 = float(np.sum(exp_vals2))
        if sum_exp2 < 1e-300:
            p2 = np.zeros(len(conv2))
            p2[np.argmax(conv2)] = 1.0
        else:
            p2 = exp_vals2 / sum_exp2
        grad2 = compute_grad(y_new, p2)

        new_v = y_new - lr * grad2
        new_v = project_simplex(new_v)

        obj_new = compute_obj(new_v)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = new_v.copy()

        v = y
        y = new_v
        beta = beta_new
        it += 1

        if verbose and it % 5000 == 0:
            print(f"      SGD it{it}: obj={best_obj:.8f}, gamma={gamma:.3f}, t={elapsed:.0f}s")

    return best_a

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Front-loaded (geometric decay)
        a = 0.7 ** np.arange(n)
        a = a / a.sum()
    elif mode == 3:
        # Back-loaded
        a = 0.7 ** np.arange(n - 1, -1, -1)
        a = a / a.sum()
    elif mode == 4:
        # U-shaped (high at edges)
        x = np.linspace(0, 1, n)
        a = 1.0 + 3.0 * (2 * x - 1) ** 2
        a = a / a.sum()
    elif mode == 5:
        # Two peaks
        center = n // 2
        a = np.exp(-((np.arange(n) - center) ** 2) / (2 * (n * 0.15) ** 2)) + \
            0.5 * np.exp(-((np.arange(n) - 0) ** 2) / (2 * (n * 0.1) ** 2)) + \
            0.5 * np.exp(-((np.arange(n) - (n - 1)) ** 2) / (2 * (n * 0.1) ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Linear ramp decreasing
        a = np.linspace(1, 0.1, n)
        a = a / a.sum()
    elif mode == 7:
        # Power law
        a = (np.arange(n) + 1) ** (-0.5)
        a = a / a.sum()
    elif mode == 8:
        # Concentrated center
        center = n // 2
        a = np.exp(-np.abs(np.arange(n) - center) / (n * 0.2))
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n))
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.full(new_n, 1.0 / new_n)
    return a_new
