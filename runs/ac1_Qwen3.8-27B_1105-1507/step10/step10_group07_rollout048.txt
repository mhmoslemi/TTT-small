import numpy as np
from scipy.fft import fft, ifft
import time

def next_power_of_2(n):
    return 1 << (n - 1).bit_length()

def compute_conv(a):
    n = len(a)
    size = next_power_of_2(2 * n - 1)
    fa = fft(a, size)
    return ifft(fa * fa, size)[:2 * n - 1].real

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_and_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Returns (smooth_max_value, gradient_vector).
    """
    n = len(a)
    size = next_power_of_2(2 * n - 1)
    fa = fft(a, size)
    conv = ifft(fa * fa, size)[:2 * n - 1].real
    
    # Softmax weights
    scaled = tau * conv
    max_scaled = scaled.max()
    shifted = scaled - max_scaled
    p = np.exp(shifted)
    p_sum = p.sum()
    p = p / p_sum
    
    # Smooth max value = (log(sum(exp(tau*conv))) ) / tau
    smooth_max = (np.log(p_sum) + max_scaled) / tau
    
    # Gradient: grad_i = sum_k p_k * d(conv_k)/d(a_i)
    # d(conv_k)/d(a_i) ≈ 2 * a[k-i] (for valid k-i, ignoring k=2i correction)
    # grad = 2 * correlate(a, p) [cross-correlation]
    # Using FFT: grad = 2 * ifft(fft(a, N) * conj(fft(p, N))).real[:n]
    # where N >= n + len(p) - 1 = n + 2n-2 - 1 = 3n-3
    
    size2 = next_power_of_2(3 * n - 2)
    A = fft(a, size2)
    P = fft(p, size2)
    grad = 2.0 * ifft(A * np.conj(P), size2).real[:n]
    
    return smooth_max, grad

    Smooth max gradient descent with adaptive tau and learning rate.
    """

    s = a.sum()
    if s < 1e-10:
        return a

    no_improve = 0

        progress = min(1.0, elapsed / time_budget)

        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.5)

        v = momentum * v + grad

        a_new = project_simplex(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0

                v *= 0.5
                no_improve = 0

def resample(a, new_n):
    n = len(a)
    if n == new_n:

        return np.full(new_n, a[0])

    x_new = np.linspace(0, 1, new_n)
    result = np.interp(x_new, x_old, a)
    return np.maximum(result, 0)
