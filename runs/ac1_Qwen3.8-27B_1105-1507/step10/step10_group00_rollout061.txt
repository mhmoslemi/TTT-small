```python
    # Using formula: grad[0] = 2*a[3-0] = 2*a3 ✓, grad[1] = 2*a[3-1] = 2*a2 ✓
    
    # Perfect. So the subgradient is simply:
    # grad[j] = 2*a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
    
    # Hmm, but this means the "subgradient" has a very specific structure - it's just a (reflected) copy of a.
    # That's a neat observation.
    
    for i in range(n):
        j = k_max - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    # Wait, I need to be more careful. The valid range for i (which is j in my notation above) is:
    # max(0, k-n+1) <= i <= min(n-1, k)
    # And for each such i, grad[i] = 2*a[k-i]
    
    # Actually, let me just re-derive more carefully.
    # S = sum_{i=lo}^{hi} a[i]*a[k-i] where lo=max(0,k-n+1), hi=min(n-1,k)
    # dS/da[m] = sum_{i=lo}^{hi} [delta(i,m)*a[k-i] + a[i]*delta(k-i,m)]
    #          = a[k-m] (if lo<=m<=hi) + a[k-m] (if lo<=k-m<=hi)
    # Note: k-m in [lo,hi] means lo <= k-m <= hi
    #   => k-hi <= m <= k-lo
    #   => k-min(n-1,k) <= m <= k-max(0,k-n+1)
    #   => max(0, k-n+1) <= m <= min(n-1, k)  (same as before!)
    
    # So: dS/da[m] = 2*a[k-m] if max(0,k-n+1) <= m <= min(n-1,k), else 0
    
    # Implementation:
    lo = max(0, k_max - n + 1)
    hi = min(n - 1, k_max)
    for m in range(lo, hi + 1):
        grad[m] = 2.0 * a[k_max - m]
    
    return grad
```

OK so now I have a clean subgradient. Let me think about the full algorithm:

**Algorithm: Nesterov-Accelerated Projected Subgradient Descent + LP Polishing**

1. Normalize a to be on the simplex (sum=1).
2. At each iteration:
   a. Compute conv(a) via FFT, find k_max.
   b. Compute subgradient g = 2*a[k_max - i] for valid i.
   c. Nesterov step: y = a + beta*(a - a_prev); a_new = proj_simplex(y - eta*g)
   d. Adapt eta based on recent progress.
3. Every K iterations, do one LP polish step.
4. Periodically try different n values.

The projection onto the simplex (with lower bound 0) is: proj_simplex(v) = max(v - tau, 0) where tau is chosen so that sum = 1. This can be done by sorting.

Actually, for the simplex {x : x >= 0, sum(x) = 1}, the Euclidean projection is:
proj(v)_i = max(v_i - tau, 0) where tau is the value such that sum_i max(v_i - tau, 0) = 1.

This is a 1D root-finding problem that can be solved by sorting.

Let me also think about the mirror descent / multiplicative update approach:
- Update: a_new_i = a_i * exp(-eta * grad_i) / Z
- This is the multiplicative weights algorithm, equivalent to mirror descent with entropy regularization.
- It naturally keeps values positive.
- For our gradient structure (grad_i = 2*a[k-i]), the update becomes:
  a_new_i = a_i * exp(-eta * 2*a[k-i]) / Z

This is interesting because it's a "self-consistent" update.

Let me code up a complete solution:

```python
"""
Optimization of step function coefficients to minimize 2*n*max(autocorrelation)/(sum)^2.
Uses Nesterov-accelerated subgradient descent with multiplicative updates,
periodic LP polishing, parametric initialization, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_of_peak(a, k_max):
    """Compute subgradient of max_k (a*a)[k] at the peak k_max."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k_max - n + 1)
    hi = min(n - 1, k_max)
    for m in range(lo, hi + 1):
        grad[m] = 2.0 * a[k_max - m]
    return grad

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def nesterov_subgradient(a, time_budget=60.0, verbose=False):
    """Nesterov-accelerated projected subgradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov parameters
    a_prev = a.copy()
    beta = 0.0  # momentum (will increase)
    eta = 1.0 / np.sqrt(n)  # initial step size
    decrease_rate = 0.999
    it = 0
    best_it = 0
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        grad = subgradient_of_peak(a, k_max)
        
        # Nesterov momentum
        beta = 2.0 / (it + 3.0)  # Nesterov sequence
        y = a + beta * (a - a_prev)
        a_prev = a.copy()
        
        # Projected gradient step
        a_new = project_simplex(y - eta * grad)
        
        # Adaptive step size (using standard S德甲 rule)
        # If objective increased, decrease step
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        eta *= decrease_rate
        it += 1
        
        if stall_count > 200:
            # Restart from best with smaller step
            a = best_a.copy()
            a_prev = a.copy()
            eta = 0.5 / np.sqrt(n)
            stall_count = 0
            best_it = it
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    NES it{it}: obj={best_obj:.8f}, eta={eta:.6f}, t={elapsed:.1f}s")
    
    return best_a

def multiplicative_weights(a, time_budget=60.0, verbose=False):
    """Mirror descent with entropy (multiplicative weights) on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 1.0 / n  # very small initial step
    it = 0
    stall_count = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        grad = subgradient_of_peak(a, k_max)
        
        # Multiplicative update: a_new = a * exp(-eta * grad) / Z
        log_a = np.log(a + 1e-300)
        log_new = log_a - eta * grad
        # Shift for numerical stability
        log_new -= np.max(log_new)
        a_new = np.exp(log_new)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        it += 1
        if stall_count % 100 == 0 and stall_count > 0:
            eta *= 0.95
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    MW it{it}: obj={best_obj:.8f}, eta={eta:.8f}, t={elapsed:.1f}s")
    
    return best_a

def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def parametric_init(n, rng, mode=0):
    """Generate parametric initializations with random parameters."""
    x = np.linspace(0, 1, n)
    t = 2 * x - 1  # in [-1, 1]
    
    if mode == 0:  # Power U-shape
        p = rng.uniform(0.5, 8.0)
        c = rng.uniform(0.01, 1.0)
        a = 1.0 / (c + (1 - c) * np.abs(t) ** p)
    elif mode == 1:  # Exponential U
        beta = rng.uniform(1.0, 15.0)
        a = np.exp(beta * t ** 2)
    elif mode == 2:  # Rational U
        p = rng.uniform(0.5, 4.0)
        q = rng.uniform(0.5, 4.0)
        a = (1 + np.abs(t) ** p) / (1 + np.abs(t) ** q) if q > p else (1 + np.abs(t) ** q) / (1 + np.abs(t) ** p)
    elif mode == 3:  # Arcsine-like
        eps = rng.uniform(0.01, 0.2)
        a = 1.0 / (eps + np.sqrt(1 - t**2 + eps**2))
    elif mode == 4:  # Bimodal
        center = rng.uniform(0.1, 0.4)
        width = rng.uniform(0.05, 0.3)
        a = np.exp(-((t + center) ** 2) / (2 * width ** 2)) + np.exp(-((t - center) ** 2) / (2 * width ** 2)) + 0.01
    elif mode == 5:  # Cosine power
        p = rng.uniform(1.0, 6.0)
        a = np.cos(np.pi * t / 2) ** p + 0.01
    else:
        a = 1.0 / (0.05 + 3.0 * t ** 2)
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def lp_polish(a, time_budget=60.0, max_iters=30, verbose=False):
    """LP polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(42 + it)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search: try several blend factors
        best_t = 0
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 0.001:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_t = t
                best_a = a_try.copy()
        
        if best_t > 0:
            a = best_a.copy()
            stall = 0
        else:
            stall += 1
            # Perturb
            noise = rng.normal(0, 0.01 * stall, n)
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}, t_elapsed={elapsed:.1f}s")
    
    return best_a
```

Wait, I realize there's an issue with my Nesterov implementation. The standard Nesterov accelerated gradient method uses:
- y_k = x_k + (t_{k-1} - 1)/t_k * (x_k - x_{k-1})
- x_{k+1} = proj(y_k - eta * grad(y_k))
- t_{k+1} = (1 + sqrt(1 + 4*t_k^2)) / 2

But I'm using a simpler momentum approach. Let me fix that. Actually, for non-smooth problems, the "Nesterov" acceleration is a bit different. Let me use a simpler approach that's known to work well:

```python
"""
Optimization of step function coefficients to minimize 2*n*max(autocorrelation)/(sum)^2.
Key improvement over parent: uses precomputed convolution decomposition for O(n) line search
(fast quadratic interpolation along blend direction), Nesterov subgradient warm-up with
multi-peak awareness, intensive LP polishing with extrapolation, and systematic n exploration.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_xy(a, b):
    """Full linear convolution a*b via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_of_peak(a, k_max):
    """Compute subgradient of max_k (a*a)[k] at the peak k_max."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k_max - n + 1)
    hi = min(n - 1, k_max)
    for m in range(lo, hi + 1):
        grad[m] = 2.0 * a[k_max - m]
    return grad


def multi_peak_subgradient(a, conv, top_k=3, temperature=0.5):
    """Weighted subgradient using top-k peaks with softmax weighting."""
    n = len(a)
    # Get top k peak values
    peak_vals = np.sort(conv)[::-1][:top_k]
    max_val = peak_vals[0]
    # Softmax weights
    weights = np.exp(-(peak_vals - max_val) / (temperature * max_val + 1e-10))
    weights /= weights.sum()
    
    # Find which lags achieve top peaks
    grad = np.zeros(n)
    # Get indices of top peaks
    top_indices = np.argsort(conv)[::-1][:top_k]
    for idx, w in zip(top_indices, weights):
        k = int(idx)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for m in range(lo, hi + 1):
            grad[m] += w * 2.0 * a[k - m]
    return grad


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def fast_line_search(conv_aa, conv_ag, conv_gg, t_max=1.5, n_evals=40):
    """
    Fast line search using precomputed convolution decomposition.
    Along a_t = (1-t)*a + t*g (both normalized to sum 1):
    conv(a_t, a_t) = (1-t)^2 * conv_aa + 2*t*(1-t) * conv_ag + t^2 * conv_gg
    Each element is a quadratic in t, so we minimize max_k of quadratics.
    Uses golden section search.
    """
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    lo = 0.0
    hi = t_max

    def obj_at_t(t):
        t2 = t * t
        one_minus_t = 1.0 - t
        val = one_minus_t * one_minus_t * conv_aa + 2.0 * t * one_minus_t * conv_ag + t2 * conv_gg
        return float(np.max(val))

    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at_t(c)
    fd = obj_at_t(d)

    for _ in range(n_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at_t(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)
    return t_best, obj_best


def nesterov_subgradient(a, time_budget=50.0, verbose=False, multi_peak=True, top_k=3):
    """Nesterov-accelerated projected subgradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    a_prev = a.copy()
    t_nest = 1.0
    eta = 2.0 / (n ** 0.5)  # initial step size
    eta_min = 1e-6
    it = 0
    best_it = 0
    stall_count = 0
    obj_history = []

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        
        if multi_peak and n > 100:
            grad = multi_peak_subgradient(a, conv, top_k=top_k, temperature=0.3)
        else:
            k_max = int(np.argmax(conv))
            grad = subgradient_of_peak(a, k_max)

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_new - 1.0) / t_new
        y = a + beta * (a - a_prev)
        a_prev = a.copy()
        t_nest = t_new

        # Projected gradient step
        a_new = project_simplex(y - eta * grad)

        obj_new = compute_obj(a_new)
        obj_history.append(obj_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        # Adaptive step size

            eta *= 0.9
        elif stall_count == 0:
            eta = min(eta * 1.01, 2.0 / (n ** 0.5))

            eta = eta_min

            a = best_a.copy()
            a_prev = a.copy()
            eta = 1.0 / (n ** 0.5)
            t_nest = 1.0
            stall_count = 0

            print(f"    NES it{it}: obj={best_obj:.8f}, eta={eta:.6f}, stall={stall_count}, t={elapsed:.1f}s")

def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)

def parametric_init(n, rng, mode=0):

    elif mode == 3:

        return np.ones(n) / n
    return a / s


def try_different_length(a_best, n_try, time_budget, rng):
    """Interpolate to different length and optimize."""
    n_old = len(a_best)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        return None, np.inf
    a_try = a_try / s

    # Quick Nesterov warm-up
    if time_budget > 20:
        a_try = nesterov_subgradient(a_try, time_budget=min(15.0, time_budget * 0.3))

    # LP polish
    if time_budget > 30:
        a_try = lp_polish_fast(a_try, time_budget=min(50.0, time_budget * 0.5), max_iters=20)

    obj_try = compute_obj(a_try)
    return a_try, obj_try


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: fast LP polishing with O(n) line search, Nesterov subgradient
    warm-up with multi-peak awareness, parametric initialization, and n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_initial = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_initial = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = u_shaped_init(n_initial, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default U-shaped init (n={n_initial}): obj={best_score:.8f}")

    # Try several parametric initializations quickly
    for mode in range(7):
        a_p = parametric_init(n_initial, rng, mode=mode)
        obj_p = compute_obj(a_p)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parametric init mode={mode}: obj={best_score:.8f} **")

    # === Phase 1: Nesterov subgradient warm-up (fast) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 1: Nesterov Subgradient Warm-up ===")
        alloc = min(40.0, remaining * 0.08)
        a_nes = nesterov_subgradient(best_seq, time_budget=alloc, verbose=True, multi_peak=True, top_k=3)
        obj_nes = compute_obj(a_nes)
        if obj_nes < best_score:
            best_score = obj_nes
            best_seq = a_nes.copy()
            print(f"  Phase1 complete: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: Nesterov obj={obj_nes:.8f} (no improvement from {best_score:.8f})")

    # === Phase 2: Intensive LP polishing with fast line search ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 2: Intensive LP Polishing (Fast Line Search) ===")
        alloc = min(remaining * 0.4, 450.0)
        a_lp = lp_polish_fast(best_seq, time_budget=alloc, max_iters=200, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Nesterov from improved point ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        print("\n=== Phase 3: Nesterov Round 2 ===")
        alloc = min(30.0, remaining * 0.06)
        a_nes2 = nesterov_subgradient(best_seq, time_budget=alloc, multi_peak=True, top_k=5)
        obj_nes2 = compute_obj(a_nes2)
        if obj_nes2 < best_score:
            best_score = obj_nes2
            best_seq = a_nes2.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj_nes2:.8f}")

    # === Phase 4: Second LP round ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print("\n=== Phase 4: LP Round 2 ===")
        alloc = min(remaining * 0.25, 250.0)
        a_lp2 = lp_polish_fast(best_seq, time_budget=alloc, max_iters=100, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase4 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 LP: obj={obj_lp2:.8f}")

    # === Phase 5: n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 5: n Exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 100, n_best - 50, n_best + 50, n_best + 100,
            n_best + 200, n_best - 200, n_best + 300,
            600, 900, 1000, 1100
        ]))
        n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_best) >= 40]
        n_options = n_options[:4]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            a_try, obj_try = try_different_length(best_seq, n_try, remaining * 0.5, rng)
            if a_try is not None and obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts with LP ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print("\n=== Phase 6: Diverse Restarts ===")
        n_cur = len(best_seq)
        for r in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            mode = (r * 3 + 1) % 7
            a_r = parametric_init(n_cur, rng, mode=mode)
            blend = 0.25
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            # Quick Nesterov
            if remaining > 50:
                a_r = nesterov_subgradient(a_r, time_budget=min(10.0, remaining * 0.15))

            # LP
            if remaining > 50:
                a_r = lp_polish_fast(a_r, time_budget=min(remaining * 0.4, 60.0), max_iters=20)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 7: Final LP Polish ===")
        alloc = min(remaining - 10, 100.0)
        if alloc > 15:
            a_final = lp_polish_fast(best_seq, time_budget=alloc, max_iters=50, verbose=True)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final: obj={best_score:.8f} **")
            else:
                print(f"  Final: obj={obj_final:.8f}")

    # Extra final check
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_check = lp_polish_fast(best_seq, time_budget=min(remaining - 5, 50.0), max_iters=20)
        obj_check = compute_obj(a_check)
        if obj_check < best_score:
            best_score = obj_check
            best_seq = a_check.copy()
            print(f"  Extra: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
