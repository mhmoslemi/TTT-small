Actually, here's another idea: **use the KKT conditions**. At the optimal `a`, there exist Lagrange multipliers `mu_k >= 0` for the constraints `(a*a)_k <= M` and `nu` for `sum a = 1` such that:
`2 * sum_k mu_k * (a*a)'_k - nu = 0` (for `a_j > 0`)
`2 * sum_k mu_k * (a*a)'_k - nu <= 0` (for `a_j = 0`)

This gives us a system of equations that we can solve iteratively.

At the optimum, `M = max_k (a*a)_k` and the active set is `{k : (a*a)_k = M}`.

So the KKT condition is: for all `j` with `a_j > 0`:
`2 * sum_{k in active} mu_k * gradient_k(j) = nu`

where `gradient_k(j) = a_{k-j} + a_{j-k}` (the j-th component of the gradient of `(a*a)_k`).

This is a system of `n` equations in `n + |active|` unknowns (the `a_j` and `mu_k`). We can solve it iteratively:
1. Start with some `a`
2. Find the active set: `active = {k : (a*a)_k >= M - epsilon}`
3. Solve the KKT system for `a` and `mu`
4. Update `a` and check convergence

Step 3 is itself an optimization problem (solve for `a` given the active set). But if the active set is small, it's tractable.

Actually, this is equivalent to the LP if we fix `a` and solve for `g`! The LP is:
```
min M
s.t. (a*g)_k <= M for k in active (or all k)
     sum g = 1
     g >= 0
```

The dual of this LP gives us the `mu_k` values. And at the fixed point, `g = a`.

So the LP iteration IS the KKT-based approach. The question is just how to make it converge to a better fixed point.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: LP fixed-point iteration (t=1) with perturbation for escape, combined with
smoothed gradient descent in log-space, multi-n exploration (800-2000), and
window-function-based diverse initializations. Key improvements over previous:
- Pure LP fixed-point (no blending) for faster convergence to fixed points
- Log-space parameterization for GD to handle non-negativity naturally
- Stronger emphasis on larger n where theoretical bound is tighter
- Perturbation via random Fourier modes to escape local minima
- Adaptive time allocation based on observed LP solve times
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Uses FFT-based cross-correlation for O(n log n) gradient.
    """
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -100, 0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    # Gradient: grad_m = sum_j a_j * (p[j+m] + p[m-j])
    # = c[m] + fc[m] where c = ifft(fft(a)*conj(fft(p))), fc = ifft(fft(a)*fft(p))
    N = 1
    while N < 3 * n + 2:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    c = ifft(A * np.conj(P)).real[:n]
    fc = ifft(A * P).real[:n]
    g = c + fc
    return g, exact_max


def logspace_smoothed_grad(log_a, tau):
    """
    Compute gradient in log-space. a_j = exp(log_a_j) / Z.
    Returns gradient w.r.t. log_a (length n) and the exact max.
    """
    n = len(log_a)
    # Softmax
    log_a_shifted = log_a - np.max(log_a)
    exp_a = np.exp(log_a_shifted)
    Z = float(np.sum(exp_a))
    a = exp_a / Z

    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -100, 0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    # Gradient of f w.r.t. a (in a-space)
    N = 1
    while N < 3 * n + 2:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    c = ifft(A * np.conj(P)).real[:n]
    fc = ifft(A * P).real[:n]
    grad_a = c + fc  # gradient w.r.t. a

    # Chain rule: d f / d log_a_j = sum_m (df/da_m)(da_m/d log_a_j)
    # da_m / d log_a_j = a_m * (delta_mj - a_j)
    # So df/d log_a_j = sum_m grad_a_m * a_m * (delta_mj - a_j)
    # = grad_a_j * a_j - a_j * sum_m grad_a_m * a_m
    grad_log = grad_a * a - a * float(np.dot(grad_a, a))
    return grad_log, exact_max, a


def lp_solve(a, solver_list=None):
    """
    Solve LP: min M s.t. (a*g)_k <= M for all k, g>=0, sum(g)=1.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    if n == 0:
        return None, None

    # Build Toeplitz matrix A[k,j] = a[k-j] if 0<=k-j<n else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if solver_list is None:
        solver_list = [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.SCS]
    else:
        solver_list = solver_list

    for solver in solver_list:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue

    return None, None


def lp_fixed_point(a, time_budget, verbose=False, seed=42, max_iters=300, perturb_every=15):
    """
    LP fixed-point iteration with t=1 (pure replacement).
    Adds perturbation every 'perturb_every' iterations to escape local minima.
    """
    rng = np.random.default_rng(seed)
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        g_lp, M_lp = lp_solve(a)
        if g_lp is None:
            if verbose:
                print(f"    LP failed at iter {it}")
            break

        # Pure fixed-point: a <- g
        a_new = g_lp.copy()
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose and it % 5 == 0:
                print(f"    LP-FP it{it}: obj={best_obj:.10f}, M_lp={M_lp:.6f}")

        # Update
        a = a_new

        # Perturbation every few iterations
        if it > 0 and it % perturb_every == 0:
            # Check if we're stuck
            if obj_new > best_obj - 1e-10:
                # Add Fourier perturbation
                num_modes = rng.integers(1, 4)
                for _ in range(num_modes):
                    f = rng.integers(1, max(2, n // 10))
                    phase = rng.uniform(0, 2 * np.pi)
                    amplitude = rng.uniform(0.01, 0.05) * best_obj / (2.0 * n * np.max(compute_conv(a)))
                    x = np.arange(n)
                    a += amplitude * np.sin(2 * np.pi * f * x / n + phase)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s

    return best_a


def gd_logspace(a, time_budget, tau_levels=None, verbose=False, beta=0.9):
    """
    Nesterov-accelerated gradient descent in log-space.
    a_j = exp(log_a_j) / Z ensures non-negativity and normalization.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    s = float(np.sum(a))
    a = a / s
    log_a = np.log(a)

    if tau_levels is None:
        tau_levels = [5, 20, 80, 300, 1000, 3000, 8000]

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    log_v = np.zeros(n)  # momentum in log-space
    lr = 1.0 / np.sqrt(n)

    for tau in tau_levels:
        if tau > 10000:
            lr = lr * 0.5
        lr = max(1e-6, min(5.0 / np.sqrt(n), lr))
        it_level = 0
        max_iters = 3000

        while time.monotonic() - start < time_budget:
            if it_level >= max_iters:
                break

            # Nesterov lookahead in log-space
            y_log = log_a + beta * log_v
            g_log, exact_max, a_y = logspace_smoothed_grad(y_log, tau)
            g_norm = float(np.linalg.norm(g_log))
            if g_norm < 1e-18:
                break

            # Adaptive step size
            step = lr / max(1.0, g_norm)
            log_v = beta * log_v - step * g_log
            log_a_new = log_a + log_v

            # Convert back to a-space
            log_a_new_shifted = log_a_new - np.max(log_a_new)
            a_new = np.exp(log_a_new_shifted)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10 or not np.isfinite(s_new):
                log_a = best_a_log = np.log(np.maximum(best_a, 1e-15))
                log_v = np.zeros(n)
                continue
            a_new = a_new / s_new
            log_a = np.log(np.maximum(a_new, 1e-30))

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()

            log_a = np.log(np.maximum(a_new, 1e-30))
            it_level += 1
            lr = max(1e-8, lr * 0.9998)

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    GD-log tau={tau:5d}: obj={best_obj:.10f}, iters={it_level}, t={elapsed:.0f}s")

    return best_a


def gd_direct(a, time_budget, tau_levels=None, verbose=False, beta=0.85):
    """
    Nesterov-accelerated GD in a-space with simplex projection.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    if tau_levels is None:
        tau_levels = [5, 20, 80, 300, 1000, 3000]

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    lr = 3.0 / np.sqrt(n)

    for tau in tau_levels:
        it_level = 0
        max_iters = 3000

        while time.monotonic() - start < time_budget:
            if it_level >= max_iters:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                continue
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-18:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            it_level += 1
            lr = max(1e-8, lr * 0.9999)

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    GD tau={tau:5d}: obj={best_obj:.10f}, iters={it_level}, t={elapsed:.0f}s")

    return best_a


def make_window(n, mode, rng=None):
    """Generate window function initializations."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 1, n)

    if mode == 0:
        # Uniform
        a = np.ones(n)
    elif mode == 1:
        # Hann window
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif mode == 2:
        # Blackman window
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    elif mode == 3:
        # Gaussian bump (centered)
        sigma = n / 4
        xm = (np.arange(n) - n / 2) / sigma
        a = np.exp(-0.5 * xm ** 2)
    elif mode == 4:
        # Exponential (centered)
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 3))
    elif mode == 5:
        # Triangular
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0) + 0.01
    elif mode == 6:
        # Semicircle
        a = np.sqrt(np.maximum(1 - (2 * x - 1) ** 2, 0.001))
    elif mode == 7:
        # Raised cosine with random frequency
        f = rng.uniform(1.5, 3.5)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
    elif mode == 8:
        # Power law taper
        a = (1 - (2 * x - 1) ** 2) ** 0.5
        a = np.maximum(a, 0.01)
    elif mode == 9:
        # Two-lobe
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 10)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
    elif mode == 10:
        # Flat with sharp edges
        edge = max(2, n // 20)
        a = np.ones(n)
        for i in range(edge):
            a[i] = np.sqrt(1 - ((edge - i) / edge) ** 2)
            a[n - 1 - i] = a[i]
    else:
        a = np.ones(n)

    a = np.maximum(a, 1e-6)
    return a / a.sum()


def resample(a, new_n):
    """Resample sequence to new length via spline interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if a[0] > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use cubic for smooth resampling
    from scipy.interpolate import PchipInterpolator
    try:
        interp = PchipInterpolator(x_old, a)
        a_new = interp(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def multi_start_lp(a, time_budget, n_starts=5, verbose=False, seed=42):
    """
    Run LP fixed-point from multiple perturbed starts, keep best.
    """
    rng = np.random.default_rng(seed)
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    per_start = time_budget / n_starts

    for s in range(n_starts):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < per_start * 0.5:
            break
        alloc = min(per_start, remaining)

        # Perturb
        a_pert = a.copy()
        if s > 0:
            # Random Fourier perturbation
            n_modes = rng.integers(2, 6)
            for _ in range(n_modes):
                f = rng.integers(1, max(2, n // 5))
                phase = rng.uniform(0, 2 * np.pi)
                amp = rng.uniform(0.02, 0.1)
                x = np.arange(n)
                a_pert += amp * np.sin(2 * np.pi * f * x / n + phase)
            a_pert = np.maximum(a_pert, 0)
            sp = float(np.sum(a_pert))
            if sp < 1e-10:
                a_pert = a.copy()
            else:
                a_pert = a_pert / sp

        a_result = lp_fixed_point(a_pert, time_budget=alloc, verbose=False,
                                  seed=seed + s, max_iters=200, perturb_every=20)
        obj_result = compute_obj(a_result)
        if obj_result < best_obj:
            best_obj = obj_result
            best_a = a_result.copy()
            if verbose:
                print(f"    Multi-start LP {s}: obj={best_obj:.10f} **")

    return best_a, best_obj

                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        parent = None

    if best_seq is None:
        n0 = 1000
        best_seq = make_window(n0, 3, rng)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.10f}")

    print(f"\nStarting search. Initial best: {best_score:.10f}")

    # === Main loop: try different n values ===
    n_values = [800, 1000, 1200, 1500, 2000]
    # Put current n first
    cur_n = len(best_seq)
    n_values.sort(key=lambda x: 0 if x == cur_n else abs(x - cur_n))
    n_values = [x for x in n_values if 500 <= x <= 2500]

    # Deduplicate
    n_values = list(dict.fromkeys(n_values))

    print(f"n values to try: {n_values}")

    for n_try in n_values:
        remaining = deadline - time.monotonic()
        if remaining < 120:
            print(f"  Skipping n={n_try}: not enough time ({remaining:.0f}s)")
            break

        print(f"\n{'='*50}")
        print(f"Processing n={n_try}, remaining={remaining:.0f}s")
        print(f"{'='*50}")

        # Initialize for this n
        a_init = resample(best_seq, n_try)

        # Try a few different initializations and pick the best after quick GD
        inits = [a_init]
        for mode in [1, 3, 6, 7, 8]:
            inits.append(make_window(n_try, mode, rng))

        # Quick screening with short GD
        best_init = a_init
        best_init_obj = compute_obj(a_init)
        screen_budget = min(remaining * 0.1, 30.0)
        per_screen = screen_budget / len(inits)

        for a_s in inits:
            if time.monotonic() + per_screen > deadline - 100:
                break
            if per_screen < 3:
                break
            a_screen = gd_direct(a_s, time_budget=per_screen,
                                  tau_levels=[10, 100, 500], verbose=False)
            obj_s = compute_obj(a_screen)
            if obj_s < best_init_obj:
                best_init_obj = obj_s
                best_init = a_screen
                print(f"    Better init: obj={best_init_obj:.10f}")

        a_work = best_init.copy()
        n_b
        # === LP Fixed-Point (main computation) ===
        lp_budget = min(remaining * 0.4, 300.0)
        if lp_budget > 30 and n_try <= 2000:
            print(f"  LP fixed-point (budget={lp_budget:.0f}s)")
            a_lp = lp_fixed_point(a_work, time_budget=lp_budget, verbose=True,
                                  seed=seed, max_iters=200, perturb_every=20)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  ** LP: obj={best_score:.10f} **")
            a_work = a_lp

        # === Log-space GD ===
        remaining = deadline - time.monotonic()
        gd_budget = min(remaining * 0.2, 100.0)
        if gd_budget > 20:
            print(f"  Log-space GD (budget={gd_budget:.0f}s)")
            a_gd = gd_logspace(a_work, time_budget=gd_budget,
                               tau_levels=[20, 100, 500, 2000], verbose=False)
            obj_gd = compute_obj(a_gd)
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
                print(f"  ** Log-GD: obj={best_score:.10f} **")
            a_work = a_gd

        # === Direct GD ===
        remaining = deadline - time.monotonic()
        gd2_budget = min(remaining * 0.15, 80.0)
        if gd2_budget > 15:
            print(f"  Direct GD (budget={gd2_budget:.0f}s)")
            a_gd2 = gd_direct(a_work, time_budget=gd2_budget,
                              tau_levels=[50, 300, 1000, 3000], verbose=False)
            obj_gd2 = compute_obj(a_gd2)
            if obj_gd2 < best_score:
                best_score = obj_gd2
                best_seq = a_gd2.copy()
                print(f"  ** Direct-GD: obj={best_score:.10f} **")
            a_work = a_gd2

        # === Multi-start LP at this n ===
        remaining = deadline - time.monotonic()
        ms_budget = min(remaining * 0.2, 150.0)
        if ms_budget > 60 and n_try <= 1500:
            print(f"  Multi-start LP (budget={ms_budget:.0f}s)")
            a_ms, obj_ms = multi_start_lp(a_work, time_budget=ms_budget,
                                           n_starts=4, verbose=True, seed=seed + n_try)
            if obj_ms < best_score:
                best_score = obj_ms
                best_seq = a_ms.copy()
                print(f"  ** Multi-start: obj={best_score:.10f} **")
            a_work = a_ms

        print(f"  After n={n_try}: best={best_score:.10f}")

    # === Final intensive polish on the best sequence found ===
    print(f"\n{'='*50}")
    print(f"FINAL POLISH PHASE")
    print(f"{'='*50}")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    print(f"  Final n={n_final}, obj={best_score:.10f}, remaining={remaining:.0f}s")

    # LP polish
    if remaining > 60 and n_final <= 2000:
        alloc = min(remaining * 0.3, 120.0)
        print(f"  Final LP (budget={alloc:.0f}s)")
        a_fin1 = lp_fixed_point(best_seq, time_budget=alloc, verbose=True,
                                 seed=seed + 999, max_iters=200, perturb_every=25)
        obj_fin1 = compute_obj(a_fin1)
        if obj_fin1 < best_score:
            best_score = obj_fin1
            best_seq = a_fin1.copy()
            print(f"  ** Final LP: obj={best_score:.10f} **")

    # High-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.25, 80.0)
        print(f"  High-tau GD (budget={alloc:.0f}s)")
        a_fin2 = gd_direct(best_seq, time_budget=alloc,
                            tau_levels=[200, 800, 2000, 5000], verbose=True)
        obj_fin2 = compute_obj(a_fin2)
        if obj_fin2 < best_score:
            best_score = obj_fin2
            best_seq = a_fin2.copy()
            print(f"  ** High-tau GD: obj={best_score:.10f} **")

    # Log-space GD
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 60.0)
        print(f"  Final Log-GD (budget={alloc:.0f}s)")
        a_fin3 = gd_logspace(best_seq, time_budget=alloc,
                              tau_levels=[100, 500, 2000, 5000], verbose=False)
        obj_fin3 = compute_obj(a_fin3)
        if obj_fin3 < best_score:
            best_score = obj_fin3
            best_seq = a_fin3.copy()
            print(f"  ** Final Log-GD: obj={best_score:.10f} **")

    # One more LP if time allows
    remaining = deadline - time.monotonic()
    if remaining > 30 and n_final <= 1800:
        alloc = min(remaining * 0.3, 50.0)
        print(f"  Ultra-final LP (budget={alloc:.0f}s)")
        a_fin4 = lp_fixed_point(best_seq, time_budget=alloc, verbose=True,
                                 seed=seed + 1234, max_iters=100, perturb_every=30)
        obj_fin4 = compute_obj(a_fin4)
        if obj_fin4 < best_score:
            best_score = obj_fin4
            best_seq = a_fin4.copy()
            print(f"  ** Ultra-final LP: obj={best_score:.10f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    return result
