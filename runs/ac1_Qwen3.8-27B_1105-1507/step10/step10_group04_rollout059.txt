import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_lp(a, time_budget=5.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / max(float(np.sum(a)), 1e-30)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:
            continue
    return None, None

def resample_sequence(a, new_n):
    """Resample via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes non-negative step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses: LP game iteration (Nash equilibrium / best-response method), high-precision smoothed
gradient descent with Nesterov acceleration, log-parameterization with L-BFGS, multi-length
exploration, and structured initializations. The LP game converges to the KKT optimum for
fixed n; smoothed GD provides sub-solver-tolerance precision for final polish.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_normalized, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / max(float(np.sum(a)), 1e-30)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value).flatten(), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    return g_val, M_val
        except Exception:
            continue
    return None, None

        a_shrunk[support] = a[support]
        a_shrunk = a_shrunk / a_shrunk.sum()
        obj_shrunk = compute_obj(a_shrunk)
        if obj_shrunk < best_obj - 1e-12:
            best_obj = obj_shrunk
            best_a = a_shrunk.copy()
            # Quick refine on shrunk support
            remaining = time_budget - (time.monotonic() - start)
            if remaining > 10:
                alloc = min(20, remaining * 0.5)
                best_a = smoothed_gd(best_a, time_budget=alloc,
                                     tau_start=500, tau_end=5000, verbose=False)
                obj_shrunk = compute_obj(best_a)
                if obj_shrunk < best_obj:
                    best_obj = obj_shrunk
            if verbose:
                print(f"    Shrink frac={frac}: obj={best_obj:.10f}, support={len(support)}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
        a = a / a.sum()
    elif mode == 3:
        a = 1.0 - np.abs(np.linspace(-1, 1, n))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        a = 0.5 + 0.35 * np.cos(2 * np.pi * 3 * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * 7 * np.arange(n) / n) + 0.15 * np.cos(2 * np.pi * 13 * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Symmetric with gap in middle
        half = n // 2
        a = np.ones(n)
        a[half - n // 10:half + n // 10] = 0.3
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a


def optimize_single(a, time_budget, verbose=True):
    """Run full optimization pipeline on a single sequence."""
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    # Phase A: LP game
    remaining = time_budget - (time.monotonic() - start)
    lp_alloc = min(remaining * 0.5, time_budget * 0.5)
    if lp_alloc > 5:
        a_lp = lp_game_iterate(best_a, time_budget=lp_alloc, verbose=verbose)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp

    # Phase B: Smoothed GD
    remaining = time_budget - (time.monotonic() - start)
    gd_alloc = min(remaining * 0.3, 40.0)
    if gd_alloc > 5:
        a_gd = smoothed_gd(best_a, time_budget=gd_alloc,
                           tau_start=200, tau_end=10000, verbose=verbose)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_obj:
            best_obj = obj_gd
            best_a = a_gd

    # Phase C: Log-param refine
    remaining = time_budget - (time.monotonic() - start)
    log_alloc = min(remaining * 0.2, 20.0)
    if log_alloc > 5:
        a_log = log_param_optimize(best_a, time_budget=log_alloc,
                                   tau_min=500, tau_max=10000, verbose=False)
        obj_log = compute_obj(a_log)
        if obj_log < best_obj:
            best_obj = obj_log
            best_a = a_log

    # Phase D: Support shrink
    remaining = time_budget - (time.monotonic() - start)
    sh_alloc = min(remaining * 0.1, 10.0)
    if sh_alloc > 3:
        a_sh = support_shrink_refine(best_a, time_budget=sh_alloc, verbose=False)
        obj_sh = compute_obj(a_sh)
        if obj_sh < best_obj:
            best_obj = obj_sh
            best_a = a_sh

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP game iteration, smoothed GD, multi-n exploration,
    diverse restarts, and final intensive polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0  # safety margin

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = diverse_init(n_init, rng, mode=0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.10f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.10f}")

    # === Phase 1: Intensive LP game on parent (180s) ===
    print(f"\n=== Phase 1: LP Game (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.18, 180.0)
    if p1_budget > 10:
        a_lp = lp_game_iterate(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp
            print(f"  Phase1 LP: obj={best_score:.10f} **")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.10f} (no improvement)")

    # === Phase 2: High-precision smoothed GD (80s) ===
    print(f"\n=== Phase 2: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.08, 80.0)
    if p2_budget > 10:
        a_gd = smoothed_gd(best_seq, time_budget=p2_budget,
                           tau_start=500, tau_end=50000, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd
            print(f"  Phase2 GD: obj={best_score:.10f} **")
        else:
            print(f"  Phase2 GD: obj={obj_gd:.10f}")

    # === Phase 3: Multi-n exploration (400s) ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.35, 400.0)

    # Generate candidate n values around the current best
    n_candidates = set()
    for delta in [-300, -200, -100, -50, 0, 50, 100, 200, 300, 500, 700, 1000, 1500]:
        n_try = n_best + delta
        if 200 <= n_try <= 5000:
            n_candidates.add(n_try)
    # Also add some absolute values
    for n_abs in [300, 400, 500, 600, 900, 1100, 1500, 2000, 3000]:
        if 200 <= n_abs <= 5000:
            n_candidates.add(n_abs)

    # Sort by proximity to current best (try nearby first)
    n_candidates = sorted([n for n in n_candidates if n != n_best],
                          key=lambda x: abs(x - n_best))
    n_candidates = n_candidates[:10]  # Limit to 10 values
    print(f"  Trying n values: {n_candidates}")

    per_n = p3_budget / max(1, len(n_candidates))
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_n, remaining - 20)
        if alloc < 15:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Run optimization pipeline
        a_opt, obj_opt = optimize_single(a_try, time_budget=alloc, verbose=False)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.10f}")

    # === Phase 4: Diverse restarts (150s) ===
    print(f"\n=== Phase 4: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 150.0)
    n_rest = max(2, int(p4_budget / 50))
    n_cur = len(best_seq)

    for r in range(n_rest):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(50, remaining - 15)
        if alloc < 15:
            break

        mode = r % 7
        a_r = diverse_init(n_cur, rng, mode=mode)
        # Blend with best
        blend = 0.2 + 0.15 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        a_opt, obj_opt = optimize_single(a_r, time_budget=alloc, verbose=False)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt
            print(f"  Restart {r} (mode={mode}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_opt:.10f}")

    # === Phase 5: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s")
    n_final = len(best_seq)

    # 5a: Another LP game round
    if remaining > 30:
        alloc = min(remaining * 0.3, 120.0)
        a_final = lp_game_iterate(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final
            print(f"  Final LP: obj={best_score:.10f} **")
        else:
            print(f"  Final LP: obj={obj_final:.10f}")

    # 5b: Very high tau GD
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.25, 80.0)
        a_final = smoothed_gd(best_seq, time_budget=alloc,
                              tau_start=2000, tau_end=100000, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final
            print(f"  Final high-tau GD: obj={best_score:.10f} **")
        else:
            print(f"  Final high-tau GD: obj={obj_final:.10f}")

    # 5c: Log-param with very high tau
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 50.0)
        a_final = log_param_optimize(best_seq, time_budget=alloc,
                                     tau_min=2000, tau_max=50000, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final
            print(f"  Final log-param: obj={best_score:.10f} **")
        else:
            print(f"  Final log-param: obj={obj_final:.10f}")

    # 5d: Support shrink + refine
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.15, 30.0)
        a_final = support_shrink_refine(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final
            print(f"  Final shrink: obj={best_score:.10f} **")

    # 5e: One more LP
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.3, 30.0)
        a_final = lp_game_iterate(best_seq, time_budget=alloc, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final
            print(f"  Ultra-final LP: obj={best_score:.10f} **")

    # 5f: Final GD
    remaining = deadline - time.monotonic()
    if remaining > 8:
        alloc = min(remaining * 0.4, 20.0)
        a_final = smoothed_gd(best_seq, time_budget=alloc,
                              tau_start=5000, tau_end=200000, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final
            print(f"  Ultra-final GD: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
