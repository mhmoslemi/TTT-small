"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: FFT-accelerated LP-driven optimization with adaptive smoothed Nesterov GD,
emphasis on larger sequence lengths, KKT-based convergence detection, and 
dual-guided perturbation for escaping local minima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = sum_j a[j]*g[k-j]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. (A@g)[k] <= M for all k, sum(g)=1, g>=0.
    Where A is the convolution matrix with a_norm.
    Returns (g, M, dual_vars) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Try to get dual variables
                    dual = None
                    try:
                        dual = constraints[0].dual_value
                    except:
                        pass
                    return g_val, float(M.value), dual
        except Exception:
            continue
    return None, None, None


def line_search(a, g, n_evals=40):
    """Line search with parabolic refinement."""
    n = len(a)
    ts = np.linspace(0.01, 0.95, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try / s)

    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]

    # Parabolic refinement
    if 0 < best_idx < n_evals - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.01, 0.95))
            a_par = (1 - t_par) * a + t_par * g
            a_par = np.maximum(a_par, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                f_par = compute_obj(a_par / s_par)
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par

    return best_t, best_obj


def smoothed_gradient(a, tau):
    """Compute gradient of smoothed max of conv(a,a) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    # Use log-sum-exp for numerical stability
    shifted = tau * (conv - max_conv)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def kkt_violation(a, tau=50.0):
    """Compute KKT violation for the non-negative constrained problem."""
    n = len(a)
    g = smoothed_gradient(a / np.sum(a), tau)
    a_norm = a / np.sum(a)
    # For interior points (a_i > 0), gradient should be ~0
    # For boundary points (a_i = 0), gradient should be >= 0
    interior = a_norm > 1e-10
    boundary = ~interior
    violation = 0.0
    if interior.any():
        violation += float(np.sum(g[interior] ** 2))
    if boundary.any():
        neg_grad = np.minimum(g[boundary], 0)
        violation += float(np.sum(neg_grad ** 2))
    return violation

def nesterov_gd(a, time_budget=100.0, tau_init=100.0, tau_end=3000.0, verbose=False):
    """Nesterov accelerated GD on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 5.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget - 1.0:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.2

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.3 * base_lr), 3.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adapt

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-12, lr * 0.99995)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.05):
    """Add Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / np.sqrt(k)
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 1e-10)
    return a_new / a_new.sum()


def block_coordinate_descend(a, time_budget=50.0, block_size=20):
    """Cyclic block coordinate descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget - 2.0:
        improved = False
        for start_idx in range(0, n, block_size):
            end_idx = min(start_idx + block_size, n)
            a_block = a[start_idx:end_idx].copy()
            
            # Minimize objective by varying a[start_idx:end_idx]
            # Use simple gradient steps on the block
            for inner_it in range(20):
                g = smoothed_gradient(a, tau=200.0)
                g_block = g[start_idx:end_idx]
                # Projected gradient step on block
                a_block -= 0.1 * g_block
                a_block = np.maximum(a_block, 0)
                a[start_idx:end_idx] = a_block
            
            # Renormalize
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
                obj = compute_obj(a)
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a


def dual_guided_perturb(a, dual_vars, rng, strength=0.03):
    """Perturb using dual variable information."""
    if dual_vars is None:
        return fourier_perturb(a, rng, strength)
    
    n = len(a)
    # dual_vars has length 2n-1, corresponding to convolution lags
    # High dual values mean those constraints are tight
    # Perturb more where the convolution is "flat" (low dual)
    d = np.asarray(dual_vars, dtype=np.float64)
    if len(d) != 2 * n - 1:
        return fourier_perturb(a, rng, strength)
    
    # Create perturbation proportional to (max_dual - dual)
    max_d = float(np.max(d))
    if max_d > 0:
        pert_weights = 1.0 - d / (max_d + 1e-10)
    else:
        pert_weights = np.ones(2 * n - 1)
    
    # Map dual weights to position weights via inverse convolution
    N = 1
    while N < 3 * n:
        N <<= 1
    D = fft(pert_weights, N)
    A = fft(a, N)
    pos_weights = ifft(D * A).real[:n]
    pos_weights = np.maximum(pos_weights, 0)
    if pos_weights.sum() > 1e-10:
        pos_weights = pos_weights / pos_weights.sum()
    
    # Perturb
    a_new = a * (1.0 + strength * (2.0 * pos_weights - 1.0) * rng.standard_normal(n))
    a_new = np.maximum(a_new, 1e-10)
    return a_new / a_new.sum()

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: FFT-accelerated LP-driven optimization with adaptive Nesterov GD,
emphasis on larger sequence lengths, dual-guided perturbation, and aggressive
multi-scale exploration. Key improvements: faster LP loop, better time allocation
for large n, structured random initializations for diversity, and high-tau
smoothing for final convergence.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = sum_j a[j]*g[k-j]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. (A@g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M, dual_vars) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    dual = None
                    try:
                        dual = np.asarray(constraints[0].dual_value, dtype=np.float64)
                    except Exception:
                        pass
                    return g_val, float(M.value), dual
        except Exception:
            continue
    return None, None, None


def line_search(a, g, n_evals=35):
    """Line search with parabolic refinement."""
    ts = np.linspace(0.01, 0.90, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try / s)

    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]

    if 0 < best_idx < n_evals - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.01, 0.90))
            a_par = (1 - t_par) * a + t_par * g
            a_par = np.maximum(a_par, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                f_par = compute_obj(a_par / s_par)
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par

    return best_t, best_obj


def smoothed_gradient(a, tau):
    """Compute gradient of smoothed max of conv(a,a) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]

def nesterov_gd(a, time_budget=100.0, tau_init=100.0, tau_end=3000.0, verbose=False):
    """Nesterov accelerated GD on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 5.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget - 1.5:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.3

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.2 * base_lr), 4.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adapt

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:

            v = np.zeros(n)

            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-12, lr * 0.99995)

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it

        print(f"    GD: obj={best_obj:.8f}, iters={it}")

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.05):
    """Add multi-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / np.sqrt(k)
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 1e-10)
    return a_new / a_new.sum()

            ks += 1
        a = uniform_filter1d(a, size=ks)
    elif mode == 4:
        # Cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n)) + 0.01
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def dual_guided_perturb(a, dual_vars, rng, strength=0.04):
    """Perturb using dual variable information to target weak spots."""
    if dual_vars is None:
        return fourier_perturb(a, rng, strength)

    n = len(a)
    d = np.asarray(dual_vars, dtype=np.float64)
    if len(d) != 2 * n - 1:
        return fourier_perturb(a, rng, strength)

    max_d = float(np.max(d))
    if max_d > 1e-10:
        pert_weights = 1.0 - d / (max_d + 1e-10)
    else:
        pert_weights = np.ones(2 * n - 1)

    N = 1
    while N < 3 * n:
        N <<= 1
    D = fft(pert_weights, N)
    A = fft(a, N)
    pos_weights = ifft(D * A).real[:n]
    pos_weights = np.maximum(pos_weights, 0)
    if pos_weights.sum() > 1e-10:
        pos_weights = pos_weights / pos_weights.sum()

    a_new = a * (1.0 + strength * (2.0 * pos_weights * n - 1.0) * rng.standard_normal(n))
    a_new = np.maximum(a_new, 1e-10)
    return a_new / a_new.sum()


def multi_method_optimize(a, time_budget, n, verbose=False):
    """Combine LP and GD with adaptive time allocation based on n."""
    start = time.monotonic()

    # For large n, LP is slower, so allocate less time to it
    if n <= 1000:
        lp_frac = 0.55
    elif n <= 1500:
        lp_frac = 0.40
    else:
        lp_frac = 0.25

    lp_budget = time_budget * lp_frac
    gd_budget = time_budget * (1.0 - lp_frac) * 0.6
    gd2_budget = time_budget * (1.0 - lp_frac) * 0.4

    # LP phase
    if lp_budget > 15:
        a = lp_optimize(a, time_budget=lp_budget, verbose=verbose)

    # First GD phase (medium tau)
    remaining = time_budget - (time.monotonic() - start)
    if gd_budget > 8 and remaining > 10:
        a = nesterov_gd(a, time_budget=min(gd_budget, remaining - 5),
                        tau_init=max(50.0, n / 10.0), tau_end=max(500.0, n))

    # Second GD phase (high tau)
    remaining = time_budget - (time.monotonic() - start)
    if gd2_budget > 5 and remaining > 8:
        a = nesterov_gd(a, time_budget=min(gd2_budget, remaining - 3),
                        tau_init=max(200.0, n / 5.0), tau_end=max(2000.0, n * 3))

    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization with LP + Nesterov GD,
    multi-n exploration emphasizing larger sequences, structured
    random initializations, and dual-guided restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: LP optimization on parent length ===
    print(f"\n=== Phase 1: Optimization on n={n_parent} ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.22, 320.0)
    if p1_budget > 30:
        a_opt = multi_method_optimize(best_seq, p1_budget, n_parent, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_opt:.8f} (no improvement)")

    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Generate candidate n values with emphasis on larger
    n_candidates = set()
    # Around parent
    n_candidates.update([n_best - 200, n_best - 100, n_best + 100, n_best + 300])
    # Larger values
    n_candidates.update([1000, 1200, 1500, 1800, 2000, 2500, 3000])
    # Smaller
    n_candidates.update([600, 700])

    n_options = sorted([n for n in n_candidates if 500 <= n <= 3500 and abs(n - n_best) > 80])
    # Prioritize: larger n first (more degrees of freedom), then closest to best
    n_options.sort(key=lambda n: (-min(n, 2000), abs(n - n_best)))
    n_options = n_options[:7]
    print(f"  Trying n values: {n_options}")

    for idx, n_try in enumerate(n_options):
        remaining = deadline - time.monotonic()
        if remaining < 70:
            break

        per_n_budget = min((remaining - 30) / (len(n_options) - idx), 180.0)
        if per_n_budget < 30:
            break

        # Choose initialization: resample from best, or structured for some
        if idx < len(n_options) // 2:
            a_try = resample(best_seq, n_try)
        else:
            mode = idx % 5
            a_try = structured_init(n_try, rng, mode=mode)
            # Blend with resampled best for continuity
            a_res = resample(best_seq, n_try)
            a_try = 0.5 * a_try + 0.5 * a_res
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()

        a_opt = multi_method_optimize(a_try, per_n_budget, n_try, verbose=False)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try} (init={'resample' if idx < len(n_options)//2 else 'structured'}): "
                  f"obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 3: Second round on best n ===
    print(f"\n=== Phase 3: Second Round on n={n_best} ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.18, 250.0)
    if p3_budget > 30:
        a_opt = multi_method_optimize(best_seq, p3_budget, n_best, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj_opt:.8f} (no improvement)")

    # === Phase 4: Restarts ===
    print(f"\n=== Phase 4: Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    rng2 = np.random.default_rng(seed + 777)

    restart_modes = [
        ('fourier_strong', 0.08),
        ('fourier_weak', 0.03),
        ('random_blend', None),
        ('mirror_blend', None),
        ('smooth_blend', None),
    ]

    for r, (name, strength) in enumerate(restart_modes):
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        per_restart = min((remaining - 20) / (len(restart_modes) - r), 100.0)
        if per_restart < 20:
            break

        if name == 'fourier_strong' or name == 'fourier_weak':
            a_r = fourier_perturb(best_seq, rng2, strength=strength)
        elif name == 'random_blend':
            noise = rng2.uniform(0.3, 1.0, n_cur)
            a_r = 0.8 * best_seq + 0.2 * noise / noise.sum() * best_seq.sum()
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        elif name == 'mirror_blend':
            a_r = 0.6 * best_seq + 0.4 * best_seq[::-1]
            a_r = a_r / a_r.sum()
        else:
            ks = max(7, n_cur // 15)
            if ks % 2 == 0:
                ks += 1
            a_smooth = uniform_filter1d(best_seq, size=ks)
            a_r = 0.75 * best_seq + 0.25 * a_smooth
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

        a_r = multi_method_optimize(a_r, per_restart, n_cur, verbose=False)
        obj_r = compute_obj(a_r)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 5: Dual-guided restart ===
    print(f"\n=== Phase 5: Dual-Guided Perturbation ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        g_tmp, M_tmp, dual_tmp = solve_lp_direction(best_seq)
        for attempt in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            strength = 0.03 + 0.02 * attempt
            a_r = dual_guided_perturb(best_seq, dual_tmp, rng2, strength=strength)
            per_attempt = min((remaining - 10) / (3 - attempt), 80.0)
            a_r = multi_method_optimize(a_r, per_attempt, len(best_seq), verbose=False)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Dual restart {attempt}: obj={best_score:.8f} **")
            else:
                print(f"  Dual restart {attempt}: obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    # Final LP
    if remaining > 35 and n_final <= 1500:
        alloc = min(remaining * 0.35, 200.0)
        a_final = lp_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Final high-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.55, 120.0)
        a_final2 = nesterov_gd(best_seq, time_budget=alloc,
                               tau_init=max(300.0, n_final / 3.0),
                               tau_end=max(3000.0, n_final * 4), verbose=True)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
        else:
            print(f"  Final GD: obj={obj_final2:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
