    i = np.max(ind[cond])
    theta = cssv[i - 1] / i

def build_conv_matrix_sparse(a_n):
    """Build conv matrix as needed for LP."""
    n = len(a_n)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):

def solve_lp_minpeak(a):
    n = len(a)
    a_n = a / float(np.sum(a))
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j+r, j] = a_n[:r]
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None

def lp_alternate(a0, time_budget=100.0, verbose=False):

def coordinate_descent(a0, time_budget=60.0, verbose=False):
    """Coordinate descent: optimize one element at a time."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    sweep = 0
    while time.monotonic() - start < time_budget:
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                # Try small values
                for new_val in [orig, 0.001/n, 0.01/n]:
                    a[i] = new_val
                    a = np.maximum(a, 0)
                    st = float(np.sum(a))
                    if st < 1e-10:
                        a[i] = orig
                        continue
                    a = a / st
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                    else:
                        a[i] = orig
                        a = a / float(np.sum(a))
                continue
            
            # Try a range of values around current
            # The optimal a[i] for fixed others: 
            # We want to minimize max_k conv[k] / sum^2
            # Try golden section or simple grid
            candidates = [orig * 0.5, orig * 0.7, orig, orig * 1.3, orig * 1.5, 0.0]
            # Also add some based on the gradient direction
            conv = compute_conv(a)
            peak = float(np.max(conv))
            # Try values that reduce the peak
            for c in candidates:
                a[i] = c
                a = np.maximum(a, 0)
                st = float(np.sum(a))
                if st < 1e-10:
                    a[i] = orig
                    continue
                a = a / st
                obj_try = compute_obj(a)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break
            else:
                a[i] = orig
                a = a / float(np.sum(a))
        
        sweep += 1
        if not improved and sweep > 3:
            break
        a = best_a.copy()
        if verbose and sweep % 5 == 0:
            print(f"      CD sweep={sweep}: obj={best_obj:.8f}")
    
    return best_a

def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

def nesterov_subgrad(a0, time_budget=60.0, top_k=5, verbose=False):
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.06
    patience = 0
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                m_start = k - i_end
                m_end = k - i_start
                grad[m_start:m_end+1] += 2.0 * y[i_start:i_end+1][::-1]
        grad /= len(top_idx)
        grad = grad - np.mean(grad)
        y_new = y - step_size * grad
        a_new = project_simplex(y_new)
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1.0 - t_k) * (a_new - a)
        y = project_simplex(y)
        a = a_new
        if it % 20 == 0 or it < 20:

                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 300:

                patience = 0
        if it % 1000 == 999:
            step_size = max(step_size * 0.95, 0.002)
        if verbose and it % 5000 == 0:

    return best_a

"""
Multi-phase optimizer for non-negative step function minimizing peak autocorrelation.
Uses: (1) log-sum-exp smoothed objective with multiplicative weights,
(2) Nesterov subgradient with top-k peaks, (3) LP alternating with line search,
(4) coordinate descent polish, (5) n-sweep, (6) random restarts.
Key innovation: smooth objective approximates max via softmax weighting,
giving gradient signal from ALL peaks (not just top-k), enabling escape from
local minima that plague exact-subgradient methods.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_obj_grad(a, beta):
    """
    Log-sum-exp smoothed objective and gradient for a (sum=1, a>=0).
    Returns (smoothed_obj, gradient) where gradient is centered for simplex.
    """
    n = len(a)
    conv = compute_conv(a)
    bc = beta * conv
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    sum_exp = np.sum(exp_bc)
    obj = (1.0 / beta) * (max_bc + np.log(sum_exp))
    w = exp_bc / sum_exp  # softmax weights over convolution positions
    # grad[m] = 2 * sum_j a[j] * w[m+j] (valid indices only)
    # Use convolution trick: grad/2 = conv(a, w_rev)[n-1:2n-1][::-1]
    w_rev = w[::-1]
    conv_aw = np.convolve(a, w_rev, mode='full')
    grad = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
    grad = grad - np.mean(grad)
    return obj, grad


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def multiplicative_weights_opt(a0, time_budget=120.0, beta_start=5.0, beta_end=100.0, verbose=False):
    """

    Works in log-space for natural non-negativity handling.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    # Initialize in log-space
    a_safe = np.maximum(a, 1e-15)
    x = np.log(a_safe / np.sum(a_safe))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break
        # Adaptive beta: start low (smooth), increase (sharp)
        progress = min(1.0, it / 5000.0)
        beta = beta_start + (beta_end - beta_start) * progress

        # Current a from log-space
        x_shifted = x - np.max(x)
        a = np.exp(x_shifted)
        a = a / np.sum(a)

        _, grad = smoothed_obj_grad(a, beta)

        # Multiplicative weights step with decreasing step size
        eta = 3.0 / (np.sqrt(it + 1) * np.sqrt(n))
        x = x - eta * grad

        # Periodic evaluation
        if it % 20 == 0:
            obj_true = compute_obj(a)
            if obj_true < best_obj - 1e-14:
                best_obj = obj_true
                best_a = a.copy()
                stall_count = 0
                if verbose:
                    print(f"        MW it={it}: obj={best_obj:.8f}, beta={beta:.1f}")
            else:
                stall_count += 1
            if stall_count > 500:
                # Reset with perturbation
                x = np.log(np.maximum(a * np.exp(np.random.default_rng(it).normal(0, 0.01, n)), 1e-15))
                x = x - np.sum(x) / n
                stall_count = 0

    # Final: also try direct subgradient from the smoothed solution
    return best_a


def nesterov_subgrad(a0, time_budget=60.0, top_k=5, verbose=False):
    """Nesterov-accelerated subgradient on the simplex."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.06
    patience = 0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                m_start = k - i_end
                m_end = k - i_start
                grad[m_start:m_end + 1] += 2.0 * y[i_start:i_end + 1][::-1]
        grad /= len(top_idx)
        grad = grad - np.mean(grad)
        y_new = y - step_size * grad
        a_new = project_simplex(y_new)
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1.0 - t_k) * (a_new - a)
        y = project_simplex(y)
        a = a_new
        if it % 20 == 0 or it < 30:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 300:
                step_size = max(step_size * 0.6, 0.001)
                patience = 0
        if it % 2000 == 1999:
            step_size = max(step_size * 0.95, 0.002)
        if verbose and it % 5000 == 0:
            print(f"        Nest it={it}: best={best_obj:.8f}, step={step_size:.4f}")

    return best_a


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k]<=M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_n = a / float(np.sum(a))
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a_n[:r]
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=100.0, verbose=False):
    """LP alternating with 1D line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)

        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 40, 'xatol': 0.003})
            best_t = res.x if res.success else 0.5
        except Exception:
            best_t = 0.5

        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            stall_count = 0
            if verbose:
                print(f"        LP: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            stall_count += 1
            if stall_count > 20:
                break

    return best_a


def coordinate_descent(a0, time_budget=80.0, verbose=False):
    """
    Coordinate descent: for each position, try a set of candidate values
    and keep the best. Uses adaptive candidate generation.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    sweep = 0

    while time.monotonic() - start < time_budget:
        improved = False
        # Randomize order for diversity
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            # Generate candidates
            if orig < 1e-14:
                candidates = [0.0, 0.5 / n, 1.0 / n, 2.0 / n]
            else:
                candidates = [orig, orig * 0.3, orig * 0.6, orig * 1.4, orig * 2.0, 0.0]
                # Also try values based on local structure
                candidates.append(orig + 1.0 / n)
                candidates.append(max(0.0, orig - 1.0 / n))

            best_local_val = orig
            best_local_obj = best_obj
            for c in candidates:
                a[i] = c
                a = np.maximum(a, 0)
                st = float(np.sum(a))
                if st < 1e-10:
                    a[i] = orig
                    a = a / max(float(np.sum(a)), 1e-10)
                    continue
                a = a / st
                obj_try = compute_obj(a)
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_local_val = c

            if best_local_val != orig:
                a[i] = best_local_val
                a = np.maximum(a, 0)
                st = float(np.sum(a))
                if st > 1e-10:
                    a = a / st
                    if best_local_obj < best_obj - 1e-14:
                        best_obj = best_local_obj
                        best_a = a.copy()
                        improved = True

        sweep += 1
        if not improved:
            break
        a = best_a.copy()
        if verbose and sweep % 3 == 0:
            print(f"        CD sweep={sweep}: obj={best_obj:.8f}")

    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def peak_deflation(a0, time_budget=60.0, n_peaks=8, verbose=False):
    """
    Targeted peak deflation: identify top peaks, find contributing a values,
    and redistribute mass to reduce the peak while maintaining sum.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for outer in range(100):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        sorted_peaks = np.argsort(conv)[::-1][:n_peaks]
        improved = False

        for k in sorted_peaks:
            k = int(k)
            # Find contributors to this peak
            # conv[k] = sum_j a[j]*a[k-j], valid j: max(0,k-n+1) <= j <= min(n-1,k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # Contribution of pair (j, k-j) to conv[k] is a[j]*a[k-j]
            # We want to reduce this. Strategy: reduce the larger of a[j], a[k-j]
            # and add to a position that doesn't contribute to this peak much.

            # Find the pair with maximum product
            best_pair = None
            best_product = -1
            for j in range(i_start, i_end + 1):
                m = k - j
                if m < n:
                    prod = a[j] * a[m]
                    if prod > best_product:
                        best_product = prod
                        best_pair = (j, m)

            if best_pair is None or best_product < 1e-12:
                continue
            j, m = best_pair
            if j == m:
                continue  # diagonal, harder to fix

            # Try reducing a[j] and adding to a random position far from peak k
            # Or try reducing both slightly
            for frac in [0.05, 0.1, 0.2, 0.3]:
                delta = frac * a[j]
                a[j] -= delta
                a[m] += delta  # keep sum approximately same
                a = np.maximum(a, 0)
                st = float(np.sum(a))
                if st < 1e-10:
                    a[j] += delta
                    a[m] -= delta
                    continue
                a = a / st
                obj_try = compute_obj(a)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[j] += delta
                    a[m] -= delta
                    a = a / st

        if not improved and outer > 20:
            break
        a = best_a.copy()
        if verbose and outer % 10 == 0:
            print(f"        PD iter={outer}: obj={best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase approach:
    1. Warm start from parent + quick Nesterov
    2. Multiplicative weights with log-sum-exp smoothing (novel)
    3. LP alternating (main workhorse)
    4. Peak deflation
    5. Coordinate descent
    6. n-sweep
    7. Final polish + random restarts
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        # Try a few initializations
        for n_init in [400, 600, 800]:
            a_try = np.ones(n_init) / n_init
            obj = compute_obj(a_try)
            if obj < best_score:
                best_score = obj
                best_seq = a_try.copy()
        if best_seq is None:
            best_seq = np.ones(400) / 400
            best_score = compute_obj(best_seq)
        print(f"Init (n={len(best_seq)}): obj={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: Quick Nesterov warm-up (60s) ===
    print(f"\n=== Phase 1: Nesterov Warm-up ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(60.0, remaining * 0.05)
        a_ns = nesterov_subgrad(best_seq, time_budget=alloc, top_k=5)
        obj = compute_obj(a_ns)
        if obj < best_score:
            best_score = obj
            best_seq = a_ns.copy()
            print(f"  Nest warm: obj={best_score:.8f} **")
        else:
            print(f"  Nest warm: obj={obj:.8f}")

    # === Phase 2: Multiplicative weights with smoothed objective (200s) ===
    print(f"\n=== Phase 2: Multiplicative Weights (log-sum-exp) ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        alloc = min(200.0, remaining * 0.15)
        a_mw = multiplicative_weights_opt(best_seq, time_budget=alloc,
                                          beta_start=5.0, beta_end=100.0, verbose=True)
        obj = compute_obj(a_mw)
        if obj < best_score:
            best_score = obj
            best_seq = a_mw.copy()
            print(f"  MW: obj={best_score:.8f} **")
        else:
            print(f"  MW: obj={obj:.8f}")
        # Follow up with Nesterov from MW result
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_ns2 = nesterov_subgrad(a_mw, time_budget=min(30.0, remaining * 0.05), top_k=5)
            obj2 = compute_obj(a_ns2)
            if obj2 < best_score:
                best_score = obj2
                best_seq = a_ns2.copy()
                print(f"  MW+Nest: obj={best_score:.8f} **")

    # === Phase 3: LP Alternating - main workhorse (400s) ===
    print(f"\n=== Phase 3: LP Alternating ===")
    a = best_seq.copy()
    total_lp_time = 0.0
    for round_idx in range(15):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        # Less time per round as we go
        alloc = min(50.0, remaining * 0.2)
        a_new = lp_alternate(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new
            total_lp_time += alloc
            print(f"  LP R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP R{round_idx}: obj={obj:.8f}")
            # Perturb to escape
            scale = 0.01 + 0.005 * round_idx
            a = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
            a = np.maximum(a, 0)
            a /= max(np.sum(a), 1e-10)

    # === Phase 4: Peak Deflation (100s) ===
    print(f"\n=== Phase 4: Peak Deflation ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(100.0, remaining * 0.1)
        a_pd = peak_deflation(best_seq, time_budget=alloc, n_peaks=8, verbose=True)
        obj = compute_obj(a_pd)
        if obj < best_score:
            best_score = obj
            best_seq = a_pd.copy()
            print(f"  PD: obj={best_score:.8f} **")
        else:
            print(f"  PD: obj={obj:.8f}")
        # LP follow-up
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_lp = lp_alternate(a_pd, time_budget=min(40.0, remaining * 0.05), verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  PD+LP: obj={best_score:.8f} **")

    # === Phase 5: Coordinate Descent (150s) ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(150.0, remaining * 0.12)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_cd)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj:.8f}")

    # === Phase 6: n-Sweep (150s) ===
    print(f"\n=== Phase 6: n-Sweep ===")
    n_options = sorted(set([200, 300, 400, 500, 600, 700, 800, 900, 1000,
                            1200, 1500, 2000,
                            n_base, max(100, n_base - 100), min(2500, n_base + 100)]))
    n_options = [mn for mn in n_options if 100 <= mn <= 2500]
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)
            a_try = a_try / max(np.sum(a_try), 1e-10)
        alloc = min(15.0, remaining * 0.08)
        a_opt = nesterov_subgrad(a_try, time_budget=alloc, top_k=5)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final Polish (LP + Nesterov) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(50.0, remaining * 0.25)
        a_f = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(30.0, remaining * 0.25)
        a_ns3 = nesterov_subgrad(best_seq, time_budget=alloc, top_k=3)
        obj_ns3 = compute_obj(a_ns3)
        if obj_ns3 < best_score:
            best_score = obj_ns3
            best_seq = a_ns3.copy()
            print(f"  Final Nest: obj={best_score:.8f} **")

    # === Phase 8: Random Restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 8: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(30):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            scale = 0.01 + 0.003 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr < 1e-10:
                continue
            a_r = a_r / sr
            alloc = min(remaining * 0.15, 15.0)
            # Quick Nesterov + LP
            a_opt = nesterov_subgrad(a_r, time_budget=alloc * 0.5, top_k=5)
            a_opt = lp_alternate(a_opt, time_budget=alloc * 0.5)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  R{r}: obj={best_score:.8f} **")
            else:
                print(f"  R{r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
