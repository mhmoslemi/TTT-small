"""
Optimized step function search using fast FFT-based gradient descent with
log-sum-exp smoothing, Adam optimization, LP polish, and dimension hopping.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_gradient_fft(a, beta):
    """
    Fast FFT-based gradient of log-sum-exp approximation of max(conv(a,a)).
    Returns gradient vector of length n.
    """
    n = len(a)
    conv = compute_conv(a)
    m = conv.max()
    if m < 1e-30:
        return np.zeros(n)
    shifted = beta * (conv - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= w_sum
    
    # c[i] = sum_j a[j] * weights[i+j] (valid indices)
    # = conv(a_rev, weights)[n-1+i]
    a_rev = a[::-1]
    # Need FFT size >= len(a_rev) + len(weights) - 1 = n + (2n-1) - 1 = 3n-2
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    fa = rfft(a_rev, size)
    fw = rfft(weights, size)
    c_full = irfft(fa * fw, size)
    g = 2.0 * c_full[n-1:2*n-1]
    return g

                         lr_start=0.5, verbose=False):

    Adam-style mirror descent on the simplex for minimizing max(conv(a,a)).
    Uses log-sum-exp smoothing with adaptive beta.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adam state
    m_t = np.zeros(n)  # first moment
    v_t = np.zeros(n)  # second moment
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    stall = 0
    patience = 50
    
    while time.monotonic() - start < time_budget:

        frac = min(1.0, t / max(1, int(time_budget * 5)))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Compute gradient
        g = smooth_gradient_fft(a, beta)

        t += 1
        m_t = beta1 * m_t + (1 - beta1) * g
        v_t = beta2 * v_t + (1 - beta2) * g**2
        m_hat = m_t / (1 - beta1**t)
        v_hat = v_t / (1 - beta2**t)
        
        lr = lr_start / np.sqrt(t)
        update = m_hat / (np.sqrt(v_hat) + eps)

        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()

def lp_refine(a, time_budget=60.0, verbose=False):
    """LP-based refinement: alternate between current a and LP solution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    iteration = 0
    while time.monotonic() - start < time_budget:
        # Build constraint matrix: A[k,j] = a_norm[k-j] if valid
        a_norm = a / a.sum()
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            A[j:j + r, j] = a_norm[:r]
        
        # Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0
        g = None
        try:
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            
            solved = False
            for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
                try:
                    prob.solve(solver=solver)
                    if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                        g = np.maximum(g_var.value, 0)
                        solved = True
                        break
                except Exception:
                    continue
            
            if not solved:
                from scipy.optimize import linprog
                c = np.zeros(n + 1)
                c[-1] = 1.0
                A_ub = np.zeros((2 * n - 1, n + 1))
                A_ub[:, :n] = A
                A_ub[:, n] = -1.0
                b_ub = np.zeros(2 * n - 1)
                A_eq = np.zeros((1, n + 1))
                A_eq[0, :n] = 1.0
                b_eq = np.array([1.0])
                bounds = [(0, None)] * n + [(None, None)]
                res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                              bounds=bounds, method='highs')
                if res.status == 0:
                    g = np.maximum(res.x[:n], 0)
        except Exception:
            return best_a
        
        if g is None:
            break
        
        s_g = float(np.sum(g))
        if s_g < 1e-10:
            break
        g = g / s_g
        
        # Line search
        best_t_obj = np.inf
        best_a_local = best_a.copy()
        for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 1.0]:
            a_try = (1 - t_val) * a + t_val * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        iteration += 1
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local.copy()
        else:
            if iteration > 3:
                break
        
        if verbose and iteration % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"        LP it{iteration}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent(a, time_budget=30.0):
    """
    Fast coordinate descent: for each coordinate, find optimal value
    given all others fixed (broadly, reduce the peak convolution).
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for rnd in range(20):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Try reducing the largest contributors to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)
        
        for contrib, i, j in contributors[:15]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break
            
            # Try reducing a[i] (or a[j]) and redistributing
            for src_idx in [i, j] if i != j else [i]:
                for target_idx in range(0, n, max(1, n // 50)):
                    if target_idx == src_idx or (target_idx == j and src_idx == i) or (target_idx == i and src_idx == j):
                        continue
                    for frac in [0.02, 0.05, 0.1]:
                        delta = frac * a[src_idx]
                        if delta < 1e-12:
                            continue
                        a[src_idx] -= delta
                        a[target_idx] += delta
                        obj_new = compute_obj(a)
                        if obj_new < best_obj - 1e-13:
                            best_obj = obj_new
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src_idx] += delta
                            a[target_idx] -= delta
                            break
                if improved:
                    conv = compute_conv(a)
                    k_star = int(np.argmax(conv))
        
        if not improved:
            break
    
    return best_a

"""
Optimized step function search using fast FFT-based gradient descent with
log-sum-exp smoothing, Adam mirror descent, LP polish, dimension hopping,
and coordinate descent refinement.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_gradient_fft(a, beta, conv=None):
    """
    Fast FFT-based gradient of log-sum-exp approximation of max(conv(a,a)).
    Returns gradient vector of length n.
    """
    n = len(a)
    if conv is None:
        conv = compute_conv(a)
    m = conv.max()
    if m < 1e-30:
        return np.zeros(n)
    shifted = beta * (conv - m)
    shifted = np.clip(shifted, -400, 400)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= w_sum

    # c[i] = sum_j a[j] * weights[i+j]
    # = conv(a_rev, weights)[n-1+i]
    a_rev = a[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    fa = rfft(a_rev, size)
    fw = rfft(weights, size)
    c_full = irfft(fa * fw, size)
    g = 2.0 * c_full[n - 1:2 * n - 1]
    return g


def adam_mirror_descent(a, time_budget, beta_start=100.0, beta_end=5000.0,
                        lr_start=0.5, verbose=False):
    """
    Adam-style mirror descent on the simplex for minimizing max(conv(a,a)).
    Uses log-sum-exp smoothing with adaptive beta.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adam state
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    stall = 0
    patience = 80
    best_stall_obj = best_obj

    while time.monotonic() - start < time_budget:
        # Adaptive beta: increase over time
        frac = min(1.0, t / max(1, int(time_budget * 3)))
        beta = beta_start * (beta_end / beta_start) ** frac

        # Compute gradient
        conv = compute_conv(a)
        g = smooth_gradient_fft(a, beta, conv=conv)

        # Adam update
        t += 1
        m_t = beta1 * m_t + (1 - beta1) * g
        v_t = beta2 * v_t + (1 - beta2) * g ** 2
        m_hat = m_t / (1 - beta1 ** t)
        v_hat = v_t / (1 - beta2 ** t)

        lr = lr_start / (1.0 + 0.01 * t)
        update = m_hat / (np.sqrt(v_hat) + eps)

        # Mirror descent step in log-space
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * update
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            best_stall_obj = best_obj
        else:
            stall += 1
            if stall > patience:
                # Restart with structured perturbation
                rng = np.random.default_rng(t * 2654435761 + n)
                noise_scale = 0.03 + 0.03 * min(stall // patience, 4)
                # Multiplicative noise
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                # Also try a structured perturbation (low-frequency modulation)
                if rng.random() < 0.3:
                    x = np.linspace(0, 1, n)
                    freq = rng.uniform(1, 5)
                    phase = rng.uniform(0, 2 * np.pi)
                    mod = 1.0 + noise_scale * 2 * np.sin(2 * np.pi * freq * x + phase)
                    a = a * np.maximum(mod, 0.1)
                    a = a / a.sum()

                m_t = np.zeros(n)
                v_t = np.zeros(n)
                stall = 0
                lr_start = max(0.05, lr_start * 0.6)
                best_obj = compute_obj(a)
                best_a = a.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"        Adam it{t}: obj={best_obj:.8f}, beta={beta:.0f}, lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a


def momentum_subgradient(a, time_budget, lr_start=0.3, verbose=False):
    """
    Simple subgradient method with momentum for the exact max.
    Faster per iteration than log-sum-exp (no softmax computation).
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    velocity = np.zeros(n)
    mu = 0.7
    t = 0
    stall = 0
    patience = 100

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_val = conv[k_star]

        # Subgradient: 2 * a[k_star - i] (valid indices)
        g = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        if i_start <= i_end:
            j_start = k_star - i_end
            j_end = k_star - i_start
            g[i_start:i_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        t += 1
        lr = lr_start / (1.0 + 0.005 * t)
        velocity = mu * velocity + (1 - mu) * g

        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * velocity
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(t * 31337 + n)
                noise = 0.03 + 0.02 * min(stall // patience, 5)
                a = best_a * np.exp(rng.normal(0, noise, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                velocity = np.zeros(n)
                stall = 0
                lr_start = max(0.05, lr_start * 0.5)
                best_obj = compute_obj(a)
                best_a = a.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"        SubGrad it{t}: obj={best_obj:.8f}, lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a


def lp_refine(a, time_budget=60.0, verbose=False):
    """LP-based refinement: alternate between current a and LP solution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    iteration = 0
    no_improve_count = 0

    while time.monotonic() - start < time_budget:
        a_norm = a / a.sum()
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            A[j:j + r, j] = a_norm[:r]

        g = None
        try:
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), constraints)

            solved = False
            for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
                try:
                    prob.solve(solver=solver)
                    if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                        g = np.maximum(g_var.value, 0)
                        solved = True
                        break
                except Exception:
                    continue

            if not solved:
                from scipy.optimize import linprog
                c = np.zeros(n + 1)
                c[-1] = 1.0
                A_ub = np.zeros((2 * n - 1, n + 1))
                A_ub[:, :n] = A
                A_ub[:, n] = -1.0
                b_ub = np.zeros(2 * n - 1)
                A_eq = np.zeros((1, n + 1))
                A_eq[0, :n] = 1.0
                b_eq = np.array([1.0])
                bounds = [(0, None)] * n + [(None, None)]
                res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                              bounds=bounds, method='highs')
                if res.status == 0:
                    g = np.maximum(res.x[:n], 0)
        except Exception:
            break

        if g is None:
            break

        s_g = float(np.sum(g))
        if s_g < 1e-10:
            break
        g = g / s_g

        # Line search
        best_t_obj = np.inf
        best_a_local = best_a.copy()
        for t_val in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 1.0]:
            a_try = (1 - t_val) * a + t_val * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        iteration += 1
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 5:
                break

        if verbose and iteration % 3 == 0:
            elapsed = time.monotonic() - start
            print(f"        LP it{iteration}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent(a, time_budget=30.0):
    """
    Coordinate descent: try moving mass from peak contributors to reduce max convolution.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for rnd in range(30):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        # Target indices (spread out)
        target_indices = list(range(0, n, max(1, n // 40)))
        if len(target_indices) < 10:
            target_indices = list(range(0, n, max(1, n // 80)))

        for contrib, i, j in contributors[:20]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-12:
                break

            src_list = [i, j] if i != j else [i]
            for src_idx in src_list:
                for target_idx in target_indices:
                    if target_idx == src_idx or target_idx == (j if src_idx == i else i):
                        continue
                    for frac in [0.01, 0.03, 0.05, 0.1]:
                        delta = frac * a[src_idx]
                        if delta < 1e-14:
                            continue
                        a[src_idx] -= delta
                        a[target_idx] += delta
                        obj_new = compute_obj(a)
                        if obj_new < best_obj - 1e-13:
                            best_obj = obj_new
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src_idx] += delta
                            a[target_idx] -= delta
                            break
                if improved:
                    conv = compute_conv(a)
                    k_star = int(np.argmax(conv))

        if not improved:
            break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Gaussian
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.5) ** 2)
        a = a / a.sum()
    elif mode == 3:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1.0 - 2.0 * np.abs(x - 0.5)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Semicircle
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.01))
        a = a / a.sum()
    elif mode == 5:
        # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-30 * (x - 0.3) ** 2) + np.exp(-30 * (x - 0.7) ** 2)
        a = a / a.sum()
    elif mode == 6:
        # Flat with edges
        a = np.ones(n)
        a[:n // 5] *= 0.3
        a[-n // 5:] *= 0.3
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Fast FFT-based gradient descent (Adam + subgradient) with
    LP polish, dimension hopping, and coordinate descent.
    Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"Initial: n={n_current}, obj={best_score:.8f}")

    # === Phase 1: Adam mirror descent at multiple dimensions ===
    print("\n=== Phase 1: Adam Mirror Descent (multiple n) ===")

    n_list = sorted(set([
        n_current, n_current + 100, n_current + 200,
        n_current - 100, n_current + 50,
        900, 1000, 1200, 1400
    ]))
    n_list = [n for n in n_list if 400 <= n <= 2000]
    n_list.sort(key=lambda x: abs(x - n_current))

    for n_try in n_list[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        if n_try == n_current:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.18, 200.0)
        print(f"  Adam for n={n_try}, budget={alloc:.0f}s")

        a_opt = adam_mirror_descent(a_try, time_budget=alloc,
                                     beta_start=80.0, beta_end=4000.0,
                                     lr_start=0.4, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 2: LP Polish ===
    print(f"\n=== Phase 2: LP Polish ===")
    best_n = len(best_seq)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 120.0)
        print(f"  LP for n={best_n}, budget={alloc:.0f}s")
        a_lp = lp_refine(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 3: Subgradient method (exact, no smoothing) ===
    print(f"\n=== Phase 3: Subgradient Method ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        n_best = len(best_seq)
        # Try at current and nearby dimensions
        n_list3 = [n_best]
        if remaining > 100:
            n_list3.append(n_best + 100)
            n_list3.append(n_best - 100)
        n_list3 = [n for n in n_list3 if 400 <= n <= 2000]

        for n_try in n_list3[:2]:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            if n_try == n_best:
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)

            alloc = min(remaining * 0.15, 120.0)
            print(f"  SubGrad for n={n_try}, budget={alloc:.0f}s")

            a_opt = momentum_subgradient(a_try, time_budget=alloc,
                                          lr_start=0.3, verbose=True)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 4: Second Adam round with higher beta ===
    print(f"\n=== Phase 4: Adam Round 2 (high beta) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n_best = len(best_seq)
        # Perturb best and re-optimize
        a_perturbed = best_seq * np.exp(rng.normal(0, 0.01, n_best))
        a_perturbed = np.maximum(a_perturbed, 1e-15)
        a_perturbed = a_perturbed / a_perturbed.sum()

        alloc = min(remaining * 0.15, 120.0)
        print(f"  Adam (high beta) for n={n_best}, budget={alloc:.0f}s")

        a_opt = adam_mirror_descent(a_perturbed, time_budget=alloc,
                                     beta_start=500.0, beta_end=8000.0,
                                     lr_start=0.15, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_best}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_best}: obj={obj_opt:.8f}")

    # === Phase 5: More dimension exploration ===
    print(f"\n=== Phase 5: Dimension Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_explore = sorted(set([n_best - 150, n_best + 150, n_best - 300, n_best + 300]))
    n_explore = [n for n in n_explore if 400 <= n <= 2000 and n != n_best]

    for n_try in n_explore[:3]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining *
