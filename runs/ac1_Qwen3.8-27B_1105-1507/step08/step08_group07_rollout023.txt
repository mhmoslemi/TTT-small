import numpy as np
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. A@a_conv(z) <= M, sum(z)=1, z>=0.
    Returns (z_normalized, M_opt, dual_y) or (None, None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])

    z_val = None
    M_val = None
    dual_y = None

    for solver in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    M_val = float(M.value)
                    try:
                        dual_y = np.maximum(np.array(c1.dual_value), 0)
                    except Exception:
                        dual_y = None
                    break
        except Exception:
            continue

    if z_val is None:
        try:
            from scipy.optimize import linprog
            c = np.zeros(n + 1)
            c[-1] = 1.0
            A_ub = np.zeros((2 * n - 1, n + 1))
            A_ub[:, :n] = A
            A_ub[:, n] = -1.0
            b_ub = np.zeros(2 * n - 1)
            A_eq = np.zeros((1, n + 1))

            b_eq = np.array([1.0])

            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

def anderson_accelerate_fp(a, time_budget, fp_func, memory=5, verbose=False):
    """
    Anderson acceleration for fixed point iteration.
    fp_func(a) returns the fixed point map applied to a.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    X = [a.copy()]
    F = [fp_func(a).copy()]
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        m = min(len(X), memory)

        if m >= 2:
            # Build the Anderson TABLE
            X_recent = np.array(X[-m:])
            F_recent = np.array(F[-m:])
            D = F_recent - X_recent  # residuals
            dX = X_recent[-1] - X_recent[:-1]
            dD = D[-1] - D[:-1]

            # Solve least squares: min ||dD*gamma|| s.t. sum(gamma)=1
            try:
                A_and = dD.T @ dD + 1e-10 * np.eye(m - 1)
                b_and = dD.T @ D[-1]
                gamma, _, _, _ = np.linalg.lstsq(A_and, b_and, rcond=None)
                gamma = gamma / np.sum(gamma) if np.sum(gamma) != 0 else np.ones(m - 1) / (m - 1)
            except Exception:
                gamma = np.ones(m - 1) / (m - 1)

            # Extrapolated point
            a_new = X_recent[-1] + np.sum(gamma * dX, axis=0)
        else:
            a_new = F[-1].copy()

        # Project to valid region
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / sn

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose and it % 50 == 0:
                print(f"    Anderson: it={it}, obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

        # Plain step if Anderson diverges
        if not np.all(np.isfinite(a_new)) or np.max(a_new) > 10:
            a_new = F[-1].copy()
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)

        a = a_new
        X.append(a.copy())
        F.append(fp_func(a).copy())
        if len(X) > memory + 1:

    """Fixed point map for LP alternating: a -> LP solution given a."""
    result = solve_lp(a)
    if result[0] is None:
        return a.copy()
    return result[0]


def gradient_refine(a, time_budget, verbose=False):
    """
    Gradient-based refinement using soft-max smoothing of the max-convolution objective.
    Uses Adam optimizer with the softmax parameterization a = softmax(x) to maintain a>0, sum(a)=1.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Initialize x in log-space (with floor to avoid log(0))
    x = np.log(np.maximum(a, 1e-10))

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adam optimizer state
    m_x = np.zeros(n)
    v_x = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps_adam = 1e-8
    lr_base = 0.05
    t_adam = 0

    # Temperature schedule for soft-max
    t_init = 0.1
    t_min = 0.01
    temp_decay = 0.9995

    temp = t_init
    stall_count = 0

    while time.monotonic() - start < time_budget:
        t_adam += 1
        temp = max(t_min, t_init * (temp_decay ** t_adam))

        a_s = np.exp(x - np.max(x))
        a_s = a_s / np.sum(a_s)

        # Compute convolution
        conv = compute_conv(a_s)
        conv_max = float(np.max(conv))

        # Soft-max weights
        conv_shifted = (conv - conv_max) / temp
        conv_exp = np.exp(conv_shifted)
        w = conv_exp / np.sum(conv_exp)

        # Gradient: grad[i] = 4n * (w * a_reversed)[i] where * is correlation
        # (w correlated with a_s): grad[i] = sum_k w[k] * a_s[k-i] for valid k-i
        # This is the cross-correlation of w and a_s
        # corr[i] = sum_k w[k] * a_s[k-i] (with bounds)
        # Equivalently: corr = ifft(conj(fft(w)) * fft(a_s)) but we need linear correlation
        # Simpler: corr[i] = sum_{k: 0<=k-i<n} w[k] * a_s[k-i]
        # This is the convolution of w with reversed a_s, taking the appropriate slice
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fw = rfft(w, size)
        fa_s = rfft(a_s, size)
        # cross-correlation: ifft(conj(fw) * fa_s) gives correlation
        corr = irfft(np.conj(fw) * fa_s, size)[:2 * n - 1]
        # The gradient of soft-max objective w.r.t. a is 2n * temp * sum_k w_k * d(conv_k)/da
        # = 2n * sum_k w_k * 2 * a[k-i] (for valid k-i)
        # = 4n * corr[i] where corr[i] = sum_k w_k * a[k-i] (the cross-correlation)
        # But we need to be careful: corr from FFT gives us the circular correlation
        # For linear: the valid range is i=0,...,n-1
        grad_a = 4.0 * n * corr[:n]

        # Convert to x-gradient (softmax parameterization)
        # dF/dx_i = a_i * (dF/da_i - sum_j dF/da_j * a_j)
        mean_grad = float(np.dot(grad_a, a_s))
        grad_x = a_s * (grad_a - mean_grad)

        # Adam update
        lr_t = lr_base * np.sqrt(1 - beta2 ** t_adam) / (1 - beta1 ** t_adam)
        m_x = beta1 * m_x + (1 - beta1) * grad_x
        v_x = beta2 * v_x + (1 - beta2) * (grad_x ** 2)
        m_hat = m_x / (1 - beta1 ** t_adam)
        v_hat = v_x / (1 - beta2 ** t_adam)
        x_new = x - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_try = np.exp(x_new - np.max(x_new))
        a_try = a_try / np.sum(a_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            x = x_new
            a = a_try
            stall_count = 0
            if verbose and t_adam % 200 == 0:
                print(f"    GradRefine: it={t_adam}, obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
        else:
            stall_count += 1
            if stall_count > 200:
                # Add noise to escape
                rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
                x = x + rng.normal(0, 0.02, n)
                stall_count = 0

    return best_a


def lp_alternating_improved(a, time_budget, verbose=False):
    """
    Improved LP alternating with:
    - Anderson acceleration
    - Adaptive mixing
    - Dual-guided perturbation
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Use Anderson acceleration on the LP fixed point map
    a_and = anderson_accelerate_fp(a, time_budget * 0.5, lp_fixed_point_map,
                                    memory=4, verbose=verbose)
    obj_and = compute_obj(a_and)
    if obj_and < best_obj:
        best_obj = obj_and
        best_a = a_and.copy()
        a = a_and.copy()

    # Continue with standard LP alternating with adaptive step
    remaining = time_budget - (time.monotonic() - start)
    if remaining < 10:
        return best_a

    g_momentum = a.copy()
    stall = 0
    prev_a = a.copy()
    prev_prev_a = None
    mix_ratio = 0.5

    while time.monotonic() - start < time_budget:
        result = solve_lp(a)
        if result[0] is None:
            break
        g_lp, M_lp, dual_y = result

        # Update momentum
        g_momentum = 0.5 * g_momentum + 0.5 * g_lp
        g_momentum = np.maximum(g_momentum, 0)
        gs = float(np.sum(g_momentum))
        if gs > 1e-10:
            g_momentum = g_momentum / gs

        # Try multiple targets and mixing ratios
        best_cand_obj = np.inf
        best_cand = a.copy()
        targets = [g_lp, g_momentum]
        mixes = [mix_ratio * 0.7, mix_ratio, mix_ratio * 1.3, 0.3, 0.7, 1.0]

        for target in targets:
            for mix in mixes:
                if mix <= 0 or mix > 1.5:
                    continue
                a_try = (1.0 - mix) * a + mix * target
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_cand_obj:
                    best_cand_obj = obj_try
                    best_cand = a_try.copy()

        if best_cand_obj < best_obj - 1e-13:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            a = best_a
            stall = 0
            prev_prev_a = prev_a
            prev_a = a.copy()
            if verbose and int((time.monotonic() - start)) % 30 == 0:
                print(f"    LP-Improved: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 10:
                # Dual-guided perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
                if dual_y is not None and np.max(dual_y) > 0:
                    A = build_conv_matrix(a)
                    active_idx = np.where(dual_y > 0.05 * np.max(dual_y))[0]
                    if len(active_idx) > 0:
                        col_imp = np.sum(A[active_idx, :], axis=0)
                        pert = -np.sign(col_imp - np.mean(col_imp)) * 0.03
                        pert += rng.normal(0, 0.02, n)
                    else:
                        pert = rng.normal(0, 0.03, n)
                else:
                    pert = rng.normal(0, 0.04, n)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                prev_prev_a = None
                g_momentum = a.copy()
                stall = 0
                mix_ratio = 0.5

    return best_a


def coordinate_polish(a, time_budget, verbose=False):
    """
    Fine-grained coordinate descent polish.
    For each coordinate, try to find a better value by scanning.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    coords = np.arange(n)

    while time.monotonic() - start < time_budget:
        # Shuffle coordinates for diversity
        rng.shuffle(coords)
        improved = False
        it_count = 0

        for i in coords:
            if time.monotonic() - start > time_budget:
                break
            it_count += 1

            # Try multiple perturbations of coordinate i
            base = best_a.copy()
            current_val = best_a[i]

            # Candidate fractions of total mass
            candidates = []
            candidates.append(current_val * 0.7)
            candidates.append(current_val * 0.85)
            candidates.append(current_val * 1.15)
            candidates.append(current_val * 1.3)
            candidates.append(current_val * 0.5)
            candidates.append(current_val * 1.5)
            candidates.append(0.0)
            candidates.append(current_val + (1.0 / n - current_val) * 0.3)

            for frac in candidates:
                if frac < 0 or frac > 1:
                    continue
                a_try = base.copy()
                # Redistribute: set a[i] = frac, scale others proportionally
                others_sum = 1.0 - current_val
                if others_sum < 1e-10:
                    continue
                target_others = 1.0 - frac
                if target_others < 0:
                    continue
                scale = target_others / others_sum
                a_try[i] = frac
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_try[mask] = base[mask] * scale
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st

                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True

        if not improved and it_count == n:
            # Try pairwise moves
            for _ in range(50):
                if time.monotonic() - start > time_budget:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1, 0.02]:
                    delta = frac * min(best_a[i], best_a[j])
                    if delta <= 0:
                        continue
                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    a_try = a_try / np.sum(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break

            if not improved:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_initialization(n, kind=0, rng=None):
    """Create various initializations."""
    if rng is None:
        rng = np.random.default_rng(0)
    t = np.linspace(0, 1, n)

    if kind == 0:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    elif kind == 1:
        # Sine squared
        a = np.sin(np.pi * t) ** 2
    elif kind == 2:
        # Triangle
        a = 1.0 - 2.0 * np.abs(t - 0.5)
    elif kind == 3:
        # Modulated raised cosine
        a = (0.5 + 0.5 * np.cos(2 * np.pi * t)) * (1.0 + 0.2 * np.cos(4 * np.pi * t))
    elif kind == 4:
        # Modulated sine
        a = (np.sin(np.pi * t) ** 2) * (1.0 + 0.3 * np.cos(6 * np.pi * t))
    elif kind == 5:
        # Gaussian window
        a = np.exp(-4.0 * (t - 0.5) ** 2)
    elif kind == 6:
        # Flat top with taper
        taper = 0.2
        a = np.ones(n)
        k = int(taper * n)
        for i in range(k):
            frac = i / k
            a[i] = 0.5 * (1 - np.cos(np.pi * frac))
            a[n-1-i] = 0.5 * (1 - np.cos(np.pi * frac))
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.01)
    a = a + rng.normal(0, 0.01, n)
    a = np.maximum(a, 0)
    a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function:
    1. Load parent, run improved LP alternating
    2. Gradient-based refinement
    3. Multi-n exploration with diverse initializations
    4. Coordinate polish
    5. Diverse restarts with structured perturbations
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        a_init = np.ones(800) / 800
        best_seq = a_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: LP Alternating with Anderson Acceleration (500s) ===
    print(f"\n=== Phase 1: LP+Anderson (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    alloc1 = min(500.0, remaining * 0.3)
    if alloc1 > 30:
        a_new = lp_alternating_improved(best_seq, time_budget=alloc1, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 1 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 done: obj={obj:.8f}")

    # === Phase 2: Gradient Refinement (300s) ===
    print(f"\n=== Phase 2: Gradient Refine ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc2 = min(300.0, remaining * 0.2)
        a_new = gradient_refine(best_seq, time_budget=alloc2, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 2 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 done: obj={obj:.8f}")

    # === Phase 3: Multi-n Exploration (350s) ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Generate candidate n values
    n_candidates = set()
    n_candidates.add(n_best)
    for delta in [-200, -150, -100, -50, 50, 100, 150, 200, 300, 400, 500]:
        n_try = n_best + delta
        if 200 <= n_try <= 3000:
            n_candidates.add(n_try)
    # Add some round numbers
    for n_round in [400, 500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600, 2000]:
        if 200 <= n_round <= 3000:
            n_candidates.add(n_round)

    n_candidates = sorted([n for n in n_candidates if n != n_best],
                          key=lambda x: abs(x - n_best))

    n_tried = 0
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_tried >= 8:
            break

        # Try different initializations for this n
        best_for_n = None
        best_obj_n = np.inf

        for init_kind in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            if init_kind == 0:
                a_init = resample_sequence(best_seq, n_try)
            else:
                a_init = make_initialization(n_try, kind=init_kind, rng=rng)

            obj_init = compute_obj(a_init)
            if obj_init < best_obj_n:
                best_obj_n = obj_init
                best_for_n = a_init.copy()

        if best_for_n is None:
            continue

        # Quick LP optimization
        alloc_n = min(40.0, remaining * 0.1)
        a_opt = lp_alternating_improved(best_for_n, time_budget=alloc_n)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_obj_n:
            best_obj_n = obj_opt
            best_for_n = a_opt.copy()

        if best_obj_n < best_score - 1e-10:
            best_score = best_obj_n
            best_seq = best_for_n.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: best={best_obj_n:.8f}")

        n_tried += 1

    # === Phase 4: Final LP + Gradient on best (200s) ===
    print(f"\n=== Phase 4: Final Optimization (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc4a = min(100.0, remaining * 0.3)
        a_new = lp_alternating_improved(best_seq, time_budget=alloc4a, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  LP final: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc4b = min(80.0, remaining * 0.25)
            a_new = gradient_refine(best_seq, time_budget=alloc4b)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Grad final: obj={best_score:.8f} **")

    # === Phase 5: Coordinate Polish (120s) ===
    print(f"\n=== Phase 5: Coordinate Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc5 = min(120.0, remaining * 0.25)
        a_new = coordinate_polish(best_seq, time_budget=alloc5, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Polish: obj={best_score:.8f} **")
        else:
            print(f"  Polish: obj={obj:.8f}")

    # === Phase 6: Diverse Restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(8, int(remaining // 35))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            if r % 4 == 0:
                # Gaussian perturbation
                sigma = 0.03 + 0.01 * r
                a_r = best_seq * np.exp(rng.normal(0, sigma, n_best))
            elif r % 4 == 1:
                # Block perturbation
                a_r = best_seq.copy()
                n_blocks = max(5, n_best // 100)
                block_size = n_best // n_blocks
                blocks = rng.choice(n_blocks, min(4, n_blocks), replace=False)
                for b in blocks:
                    si = int(b) * block_size
                    ei = si + block_size
                    a_r[si:ei] *= rng.uniform(0.4, 1.6)
            elif r % 4 == 2:
                # Low-frequency modulation
                t = np.linspace(0, 1, n_best)
                freq = rng.uniform(1, 5)
                amp = rng.uniform(0.1, 0.3)
                a_r = best_seq * (1.0 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2*np.pi)))
            else:
                # Re-initialize with different structure
                a_r = make_initialization(n_best, kind=rng.integers(0, 7), rng=rng)
                # Blend with best
                a_r = 0.5 * best_seq + 0.5 * a_r
                a_r = a_r / np.sum(a_r)

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr

            # Quick optimization
            alloc_r = min(remaining * 0.3, 30.0)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()

            a_opt = lp_alternating_improved(a_r, time_budget=alloc_r)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc7 = min(remaining * 0.6, 20.0)
        a_final = coordinate_polish(best_seq, time_budget=alloc7)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final polish: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)

    # Ensure all values are valid for the evaluator
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
