    while size < 2 * n - 1:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    n = len(a)

    if s < 1e-10:

    a_norm = a / s

    return 2.0 * n * float(np.max(conv))

import numpy as np

import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_obj_grad(a, T):
    """
    Compute smooth objective and gradient for L-BFGS.
    f(a) = 2*n*(1/T)*log(sum_k exp(T*(a/s * a/s)_k)) where s=sum(a)
    Returns (objective, gradient).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf, np.zeros(n)
    u = a / s  # normalized

    conv = compute_conv(u)
    Tconv = T * conv
    max_Tconv = np.max(Tconv)
    exp_Tconv = np.exp(Tconv - max_Tconv)
    sum_exp = np.sum(exp_Tconv)
    p = exp_Tconv / sum_exp  # soft-max weights, length 2n-1
    S = (max_Tconv + np.log(sum_exp)) / T  # smooth max value

    # q[j] = sum_{m=0}^{n-1} p[j+m] * u[m]  (cross-correlation)
    N_fft = 1
    while N_fft < (2 * n - 1) + n:
        N_fft <<= 1
    p_fft = rfft(p, N_fft)
    u_fft = rfft(u, N_fft)
    q = irfft(np.conj(u_fft) * p_fft, N_fft)[:n]

    # r = sum_k q[k] * u[k]
    r = float(np.dot(q, u))

    # Objective
    obj = 2.0 * n * S

    # Gradient: df/da_j = (4*n/s) * (q[j] - r)
    grad = (4.0 * n / s) * (q - r)

    return obj, grad


def lgbfs_continue(a, time_budget=100.0, T_start=5.0, T_end=80.0, n_steps=8):
    """
    L-BFGS with continuation on soft-max temperature T.
    Start with smooth T, gradually increase to sharp T.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_true_obj = compute_obj(a)
    start = time.monotonic()

    # Log-space continuation from T_start to T_end
    T_values = np.geomspace(T_start, T_end, n_steps)

    for T in T_values:
        if time.monotonic() - start > time_budget:
            break

        def fun_grad(x):
            elapsed = time.monotonic() - start
            if elapsed > time_budget:
                return 0.0, np.zeros(len(x))
            obj, grad = softmax_obj_grad(x, T)
            if not np.isfinite(obj):
                return 0.0, np.zeros(len(x))
            return obj, grad

        try:
            res = minimize(fun_grad, a, method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': 200, 'maxfun': 500,
                                    'ftol': 1e-14, 'gtol': 1e-10,
                                    'maxls': 20})
            if res.x is not None:
                a_try = np.maximum(res.x, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    true_obj = compute_obj(a_try)
                    if true_obj < best_true_obj:
                        best_true_obj = true_obj
                        best_a = a_try.copy()
                        a = a_try.copy()
        except Exception:
            pass

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt, dual_y) or (None, None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])

    dual_y = None
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    try:
                        dual_y = np.maximum(np.array(c1.dual_value), 0)
                    except Exception:
                        dual_y = None
                    return z_val, float(M.value), dual_y
        except Exception:
            continue

    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n]), None
    except Exception:
        pass

    return None, None, None

def build_hankel(weights):
    """Build Hankel matrix Q where Q[i,j] = weights[i+j]."""
    n = len(weights) // 2 + (1 if len(weights) % 2 == 1 else 0)
    # weights has length 2n-1
    n = (len(weights) + 1) // 2
    Q = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            Q[i, j] = weights[i + j]
    return Q


def solve_irls_qp(a, T, max_iters=50):
    """
    IRLS-QP: iteratively reweight and solve QP to minimize weighted sum of conv peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(max_iters):
        conv = compute_conv(a)
        # Compute weights
        Tconv = T * conv
        max_Tconv = np.max(Tconv)
        exp_Tconv = np.exp(Tconv - max_Tconv)
        weights = exp_Tconv / np.sum(exp_Tconv)  # length 2n-1

        # Build Hankel matrix
        Q = np.zeros((n, n))
        for i in range(n):
            for j in range(n):
                Q[i, j] = weights[i + j]

        # Ensure Q is PSD (add small regularization)
        Q = (Q + Q.T) / 2.0
        Q += 1e-10 * np.eye(n)

        # Solve QP: min a^T Q a s.t. sum(a) = 1, a >= 0
        x = cp.Variable(n, nonneg=True)
        obj = cp.quad_form(x, cp.psd_wrap(Q))
        prob = cp.Problem(cp.Minimize(obj), [cp.sum(x) == 1])

        solved = False
        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    x_val = np.maximum(x.value, 0)
                    xs = float(np.sum(x_val))
                    if xs > 1e-10:

                                    a = a_try.copy()

            except Exception:

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    best_seq = None

    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        print("Could not load parent, using default initialization.")

    if best_seq is None:
        # Structured initializations
        best_init = None
        best_init_obj = np.inf
        for n_init in [400, 600, 800, 1000, 1200]:
            t = np.linspace(0, 1, n_init)
            windows = [
                0.5 + 0.5 * np.cos(2 * np.pi * t),
                np.sin(np.pi * t) ** 2,
                1.0 - 0.3 * np.cos(4 * np.pi * t),
                np.exp(-2 * (t - 0.5) ** 2 / 0.05),
            ]
            for w in windows:
                a_init = np.maximum(w, 0.01)
                a_init = a_init / np.sum(a_init)
                obj_init = compute_obj(a_init)
                if obj_init < best_init_obj:
                    best_init_obj = obj_init
                    best_init = a_init.copy()
        best_seq = best_init if best_init is not None else np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Initial (n={len(best_seq)}): obj={best_score:.8f}")

    a = best_seq.copy()
    n_parent = len(a)
    print(f"Starting optimization from n={n_parent}, obj={best_score:.8f}")

    # === Phase 1: L-BFGS continuation (300s) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 1: L-BFGS continuation (n={len(a)}) ===")
        alloc = min(300.0, remaining * 0.25)
        a_new = lgbfs_continue(a, time_budget=alloc, T_start=5.0, T_end=100.0, n_steps=8)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 1 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 done: obj={obj:.8f} (no improvement)")

    # === Phase 2: LP alternating (200s) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: LP alternating (n={len(a)}) ===")
        alloc = min(200.0, remaining * 0.18)
        a_new = lp_alternating(a, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 2 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 done: obj={obj:.8f}")

    # === Phase 3: IRLS-QP (150s) ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and len(a) <= 1200:
        print(f"\n=== Phase 3: IRLS-QP (n={len(a)}) ===")
        alloc = min(150.0, remaining * 0.15)
        a_new = irls_qp_loop(a, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 3 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 done: obj={obj:.8f}")

    # === Phase 4: L-BFGS again on current best (150s) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: L-BFGS round 2 (n={len(a)}) ===")
        alloc = min(150.0, remaining * 0.15)
        a_new = lgbfs_continue(best_seq, time_budget=alloc, T_start=20.0, T_end=120.0, n_steps=6)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 4 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4 done: obj={obj:.8f}")

    # === Phase 5: Multi-n exploration (300s) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 5: Multi-n exploration ===")
        n_best = len(best_seq)
        n_candidates = sorted(set([
            n_best - 100, n_best - 50, n_best + 50, n_best + 100,
            n_best + 200, n_best + 400,
            500, 600, 700, 800, 900, 1000, 1100, 1200, 1400,
        ]))
        n_candidates = [n for n in n_candidates if 300 <= n <= 2000]
        n_candidates = [n for n in n_candidates if n != n_best]
        n_candidates.sort(key=lambda x: abs(x - n_best))

        for n_try in n_candidates[:8]:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            a_try = resample_sequence(best_seq, n_try)
            # L-BFGS on resampled
            alloc1 = min(40.0, remaining * 0.15)
            a_opt = lgbfs_continue(a_try, time_budget=alloc1, T_start=10.0, T_end=80.0, n_steps=5)
            obj1 = compute_obj(a_opt)

            # LP on result
            remaining = deadline - time.monotonic()
            if remaining > 20 and obj1 < best_score + 0.01:
                alloc2 = min(30.0, remaining * 0.1)
                a_opt = lp_alternating(a_opt, time_budget=alloc2)
                obj1 = compute_obj(a_opt)

            if obj1 < best_score:
                best_score = obj1
                best_seq = a_opt.copy()
                a = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj1:.8f}")

    # === Phase 6: LP alternating on absolute best (150s) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: LP final (n={len(best_seq)}) ===")
        alloc = min(150.0, remaining * 0.15)
        a_new = lp_alternating(best_seq, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 6 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 6 done: obj={obj:.8f}")

    # === Phase 7: L-BFGS final polish (100s) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: L-BFGS polish ===")
        alloc = min(100.0, remaining * 0.1)
        a_new = lgbfs_continue(best_seq, time_budget=alloc, T_start=40.0, T_end=150.0, n_steps=5)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 7 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 7 done: obj={obj:.8f}")

    # === Phase 8: Peak shaving (50s) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 8: Peak shaving ===")
        alloc = min(50.0, remaining * 0.05)
        a_new = peak_shave(best_seq, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 8 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 8 done: obj={obj:.8f}")

    # === Phase 9: Random restarts with local optimization ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 9: Random restarts ===")
        n_best = len(best_seq)
        max_restarts = min(5, int(remaining // 40))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            if r % 3 == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.03 + 0.01 * r, n_best))
            elif r % 3 == 1:
                a_r = best_seq.copy()
                n_blocks = max(5, n_best // 100)
                block_size = n_best // n_blocks
                blocks = rng.choice(n_blocks, min(3, n_blocks), replace=False)
                for b in blocks:
                    si = int(b) * block_size
                    ei = si + block_size
                    a_r[si:ei] *= rng.uniform(0.5, 1.5)
            else:
                a_r = best_seq.copy()
                n_sparse = max(5, n_best // 20)
                idx = rng.choice(n_best, n_sparse, replace=False)
                a_r[idx] *= rng.uniform(0.4, 2.0, n_sparse)

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr

            # Quick L-BFGS
            alloc = min(remaining * 0.3, 30.0)
            a_opt = lgbfs_continue(a_r, time_budget=alloc, T_start=15.0, T_end=80.0, n_steps=4)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Final: L-BFGS with very high T for precision ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 30.0)
        a_final = lgbfs_continue(best_seq, time_budget=alloc, T_start=50.0, T_end=200.0, n_steps=3)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final L-BFGS: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluator
    try:
        val = evaluate_sequence(result)
        if val < np.inf and val < best_score:
            best_score = val
        elif val == np.inf:
            result = [1.0] * 100
            best_score = 2.0
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
