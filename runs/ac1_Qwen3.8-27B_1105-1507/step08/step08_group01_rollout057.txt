```python
"""
LP-alternating optimization with adaptive active set and coordinate descent
for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses scipy HiGHS LP solver, FFT-based convolution, and quadratic line search.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end+1] = a[k - j_end : k - j_start + 1][::-1]
    return A


def solve_lp(a, active_indices=None):
    """
    Solve LP: min M s.t. A_active @ g <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    m = 2 * n - 1
    
    if active_indices is None:
        active_indices = np.arange(m)
    
    A = build_conv_matrix(a)
    A_sub = A[active_indices]
    
    # Variables: [g_0, ..., g_{n-1}, M]
    # Minimize M -> c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: A_sub @ g - M <= 0
    A_ub = np.zeros((len(active_indices), n + 1))
    A_ub[:, :n] = A_sub
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(active_indices))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def line_search(a, g, conv_aa, n_evals=12):
    """
    Find optimal t in [0,1] for a_new = (1-t)*a + t*g.
    Uses the precomputed conv_aa and computes the quadratic form.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)
    
    conv_gg = compute_conv(g_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    
    def obj_at_t(t):
        val = (1-t)**2 * conv_aa + 2*t*(1-t) * conv_ag + t**2 * conv_gg
        return 2.0 * n * float(np.max(val))
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
    lo, hi = 0.0, 1.0
    c_ = hi - gr * (hi - lo)
    d_ = lo + gr * (hi - lo)
    fc = obj_at_t(c_)
    fd = obj_at_t(d_)
    
    for _ in range(n_evals):
        if fc < fd:
            hi = d_
            d_ = c_
            fd = fc
            c_ = hi - gr * (hi - lo)
            fc = obj_at_t(c_)
        else:
            lo = c_
            c_ = d_
            fc = fd
            d_ = lo + gr * (hi - lo)
            fd = obj_at_t(d_)
    
    t_opt = (lo + hi) / 2.0
    return t_opt


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
```

```python
"""
LP-alternating optimization with adaptive active set, golden-section line search,
coordinate descent polish, and multi-n exploration for minimizing
2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key improvements over previous: tighter active set management, more LP iterations,
smarter perturbation, and focused time allocation.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp(a, active_indices):
    """
    Solve LP: min M s.t. A[active] @ g <= M, sum(g)=1, g>=0, M>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    m = len(active_indices)
    if m == 0:
        return None, None

    A = build_conv_matrix(a)
    A_sub = A[active_indices]

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A_sub
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def golden_section_line_search(a, g, conv_aa, conv_ag, conv_gg, n_evals=14):
    """
    Golden section search for optimal t in [0,1] minimizing
    2n * max((1-t)^2 * conv_aa + 2t(1-t) * conv_ag + t^2 * conv_gg).
    """
    n = len(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    def obj_at_t(t):
        val = (1.0 - t) ** 2 * conv_aa + 2.0 * t * (1.0 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    lo = 0.0
    hi = 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = obj_at_t(c1)
    f2 = obj_at_t(c2)

    for _ in range(n_evals):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = obj_at_t(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = obj_at_t(c2)

    return (lo + hi) / 2.0


def lp_alternating(a, deadline, verbose=True):

def coordinate_descent(a, deadline, verbose=True):
    """

        pass_num += 1
        improved = False
        order = rng.permutation(n)

        for i in order:

                break

            # Try increasing
            for d in [delta, delta * 0.1]:
                a_try = a.copy()
                a_try[i] += d
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))

                    best_a = a_try.copy()

                    break

                continue

            # Try decreasing
            for d in [delta, delta * 0.1]:

                    continue
                a_try = a.copy()
                a_try[i] -= d
                a_try = np.maximum(a_try, 0)

                a_try = a_try / s_try

                    best_a = a_try.copy()

                    improved = True
                    break

        if verbose and pass_num % 10 == 0:

        if not improved:
            no_improve_passes += 1
            delta *= 0.6
            if no_improve_passes > 8 or delta < 1e-8:
                break
        else:
            no_improve_passes = 0

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def smooth_max_gradient_descent(a, deadline, T_init=0.0002, T_min=0.00001):
    """
    Gradient descent using smooth max (log-sum-exp) with momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    T = T_init
    lr = 0.002
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        if total < 1e-300:
            break
        weights = exp_vals / total

        # Gradient of smooth max w.r.t. a[i] = 2 * sum_k weights[k] * a[k-i]
        # = 2 * correlate(weights, a, 'valid')
        grad = 2.0 * np.correlate(weights, a, mode='valid')
        # Center the gradient (we're on the simplex)
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = a + v

        # Project onto simplex
        a_new = _simplex_project(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                lr *= 0.7
                if lr < 1e-7:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
                T = max(T_min, T * 0.8)

        if it > 50:
            T = max(T_min, T_init * (0.997 ** it))

    return best_a


def _simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
