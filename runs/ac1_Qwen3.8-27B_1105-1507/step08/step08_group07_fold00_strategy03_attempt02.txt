def evaluate_sequence(seq: list[float]) -> float:
    """Evaluates a sequence of coefficients with defensive checks.

    if not isinstance(seq, list):
        return np.inf
    if not seq:
        return np.inf
    
    # Validate all elements
    n = len(seq)
    seq_float = []
    for x in seq:
        if isinstance(x, bool) or not isinstance(x, (int, float)) or np.isnan(x) or np.isinf(x):
            return np.inf
        seq_float.append(float(x))
        if x < 0 or x > 1000:
            return np.inf
    
    seq = seq_float
    
    # Additional defense: ensure sum is at least 0.01
    s = np.sum(seq)
    if s < 0.01:
        # Pad to increase sum
        shortfall = 0.01 - s
        seq = [max(x, 0) for x in seq]  # Ensure negatives are turned to 0
        # Distribute minimal padding to least significant elements
        seq[:int(n * 0.2)] = [x + shortfall * 0.01 if x > 0 else 0.01 + shortfall * 0.01 for x in seq[:int(n * 0.2)]]
        s = 0.01
        seq = [x / s for x in seq]  # Normalize the entire sequence to sum 0.01
    
    # Corrected convolution function
    n_len = len(seq)
