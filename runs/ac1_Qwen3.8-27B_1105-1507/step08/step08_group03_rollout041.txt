The alternating method is essentially a form of block coordinate descent on a non-convex problem. The convergence rate near a local minimum is typically linear, with the rate depending on the curvature of the objective.

To speed up convergence, I can:
1. Use Nesterov acceleration (but this is tricky for non-smooth objectives)
2. Use a line search to find the optimal step in the direction of g
3. Use the dual variables to get additional descent information

Actually, I just realized something important. The LP solution g minimizes max_k (a*g)[k]. But what we really want is to minimize max_k (g*g)[k]. These are different objectives!

The reason the alternating scheme works is:
- If a is "close" to the optimum, then (a*g) ≈ (g*g) in some sense
- The LP solution g is "good" in the sense that it doesn't create peaks when convolved with a
- By moving towards g, we reduce the peaks in the autocorrelation

But for a better approach, we could solve a different LP:
min M s.t. (g*g)[k] <= M, sum(g)=1, g>=0

But this is NOT an LP because (g*g)[k] is quadratic in g!

Alternative: We could use a "linearized" version:
min M s.t. (a*g)[k] <= M - alpha*(g-a)^2, sum(g)=1, g>=0

But this is getting complicated. Let me stick with the original LP and focus on implementation efficiency.

Here's another idea: **use the LP solution as a warm start for the next LP**. Since we already have g (the LP solution), we can use it as the initial iterate for the next LP solve. Many LP solvers support warm starting.

With cvxpy, we can use `prob.solve(warm_start=True)` after changing the problem slightly. But since the constraint matrix A changes each iteration (because a changes), warm starting might not directly help.

However, we can use a different approach: instead of solving the LP from scratch each time, we can use the **dual solution** from the previous LP to guide the current solve. The dual variables tell us which constraints are "tight" and which are "slack."

"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast LP alternating with adaptive step size, multi-n exploration,
diverse initializations, and smooth polishing.
Key differences from previous approach:
1. More aggressive use of modern LP solvers (CLARABEL, MOSEK)
2. Adaptive step size with golden section search
3. Systematic multi-n exploration
4. Better restart strategy based on dual information
5. Smooth log-sum-exp polishing at the end
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize as sp_minimize

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except Exception:
    HAVE_CVXPY = False


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_with_cvxpy(a, M_init=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    # Try solvers in order of expected speed
    for solver_name in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver_name, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.asarray(g.value).flatten()
                g_val = np.maximum(g_val, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None


def solve_lp_with_scipy(a):
    """Fallback LP solver using scipy."""
    n = len(a)
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def solve_lp(a):
    """Solve LP with best available solver."""
    if HAVE_CVXPY:
        result = solve_lp_with_cvxpy(a)
        if result[0] is not None:
            return result
    return solve_lp_with_scipy(a)


def lp_alternating(a, time_budget=300.0, verbose=False):
    """LP alternating with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345)
    it = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break
        
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        
        # Adaptive step size: try several values
        best_t_obj = best_obj
        best_a_new = best_a.copy()
        
        # Primary candidates
        ts = [1.0, 0.9, 0.7, 0.5, 0.3]
        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-15:
                best_t_obj = obj_try
                best_a_new = a_try.copy()
        
        it += 1
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a
            stall = 0
            if verbose and it % 50 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, M_lp={M_lp:.6f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 20:
                # Escape: perturb with noise
                noise_scale = min(0.3, 0.05 + 0.02 * (stall // 20))
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
    
    return best_a


def smooth_optimize(a, time_budget=60.0, T=50.0, verbose=False):
    """Smooth optimization using log-sum-exp with L-BFGS-B."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    def obj_grad(x):
        x = np.maximum(x, 1e-15)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x = x / s
        conv = compute_conv(x)
        max_c = conv.max()
        z = T * (conv - max_c)
        e = np.exp(z)
        sum_e = e.sum()
        obj_val = 2.0 * n * (max_c + np.log(sum_e) / T)
        # Gradient
        w = e / sum_e
        g = np.zeros(n)
        for k in range(len(conv)):
            if w[k] > 1e-8:
                # grad of conv[k] w.r.t. x
                g += w[k] * conv_grad_at_k(x, k)
        g *= 2.0 * n
        return obj_val, g
    
    def conv_grad_at_k(x, k):
        """Gradient of (x*x)[k] w.r.t. x."""
        n = len(x)
        g = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            if j_start == j_end:
                # Special case: i == j for some terms
                for i in range(i_start, i_end + 1):
                    j = k - i
                    if j == i:
                        g[i] += 2.0 * x[i] * x[j]  # Wait, this isn't right
                    else:
                        g[i] += x[j]
            else:
                g[i_start:i_end + 1] = x[j_start:j_end + 1][::-1]
        return g
    
    # Actually, let me reconsider the gradient
    # (x*x)[k] = sum_i x[i] * x[k-i]
    # d/dx[m] (x*x)[k] = x[k-m] + x[m-k] (if both in range), or 2*x[k-m] if m = k-m
    # More carefully:
    # d/dx[m] sum_i x[i]*x[k-i] = sum_i (delta_{i,m}*x[k-i] + x[i]*delta_{k-i,m})
    # = x[k-m] (if 0<=k-m<n) + x[m-k] (if 0<=m-k<n)
    # If m = k-m, i.e., 2m = k, then we get 2*x[k-m] = 2*x[m]
    
    def conv_grad_k(x, k):
        n = len(x)
        g = np.zeros(n)
        for m in range(n):
            km = k - m
            if 0 <= km < n:
                g[m] += x[km]
            mk = m - k
            if 0 <= mk < n:
                g[m] += x[mk]
        return g
    
    # This is O(n^2) per evaluation. For n=800, that's 640000 ops per gradient.
    # With FFT, we can do better, but for L-BFGS this should be OK.
    
    max_iter = 300
    try:
        res = sp_minimize(obj_grad, a.copy(), method='L-BFGS-B',
                          bounds=[(1e-12, None)] * n,
                          options={'maxiter': max_iter, 'maxfun': max_iter * 3, 'disp': False})
        a_opt = np.maximum(res.x, 0)
        s_opt = float(np.sum(a_opt))
        if s_opt > 1e-10:
            a_opt = a_opt / s_opt
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                if verbose:
                    print(f"    Smooth: obj={best_obj:.8f}")
    except Exception:
        pass
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Triangular
        x = np.linspace(-1, 1, n)
        a = np.maximum(0, 1.0 - np.abs(x))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Semicircle
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(0, 1 - x**2)) + 0.01
        a = a / a.sum()
    elif mode == 5:
        # Periodic
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Exponential decay from center
        center = n // 2
        a = np.exp(-np.abs(np.arange(n) - center) / (n // 4))
        a = a / a.sum()
    elif mode == 7:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.01
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

"""

Uses fast LP alternating (CLARABEL/MOSEK via cvxpy) with adaptive step size,
active-set gradient polishing, multi-n exploration, diverse initializations,

2. Fine-grained adaptive step size (golden-section-like)
3. Symmetrization to exploit reversal invariance
4. Active-set gradient descent for final polish
5. Systematic multi-n exploration with per-n optimization
6. Smarter restart strategy using dual information

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize as sp_minimize

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except Exception:
    HAVE_CVXPY = False


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None)."""
    n = len(a)
    A = build_conv_matrix(a)

    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        # Try solvers in order of expected speed for this problem size
        for solver_name in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS, cp.GLOP]:
            try:
                prob.solve(solver=solver_name, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.asarray(g.value).flatten()
                    g_val = np.maximum(g_val, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue

    # Fallback to scipy HiGHS
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(-np.inf, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def conv_grad_k(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    for m in range(n):
        km = k - m
        if 0 <= km < n:
            g[m] += a[km]
        mk = m - k
        if 0 <= mk < n:
            g[m] += a[mk]
    return g


def enforce_symmetry(a):

    sym = 0.5 * (a + a[::-1])
    s = sym.sum()
    if s > 1e-10:
        sym /= s
    return sym

            if s_try < 1e-10:
                break
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-15:
                a = a_try
                best_obj = obj_try
                best_a = a.copy()
                lr = lr_try * 1.05
                improved = True
                break
            lr_try *= 0.5

        it += 1
        if not improved:
            lr = lr_try
            if lr < 1e-10:
                # Escape
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                lr = 1.0 / n
            if verbose and it % 50 == 0:
                print(f"    AS-Grad it{it}: obj={best_obj:.8f}, |A|={len(active)}, el={time.monotonic()-start:.1f}s")

    return best_a


def smooth_optimize(a, time_budget=60.0, T=50.0, verbose=False):
    """Smooth optimization using log-sum-exp approximation with L-BFGS-B."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)

    def make_obj_grad(T_val):
        def obj_grad(x):
            x = np.maximum(x, 1e-15)
            s = x.sum()
            if s < 1e-10:
                return 1e10, np.zeros(len(x))
            x = x / s
            conv = compute_conv(x)
            max_c = conv.max()
            z = T_val * (conv - max_c)
            e = np.exp(z)
            sum_e = e.sum()
            obj_val = 2.0 * n * (max_c + np.log(sum_e) / T_val)
            w = e / sum_e
            g = np.zeros(n)
            for k in range(len(conv)):
                if w[k] > 1e-6:
                    g += w[k] * conv_grad_k(x, int(k))
            g *= 2.0 * n
            return obj_val, g
        return obj_grad

    max_iter = 200
    try:
        res = sp_minimize(make_obj_grad(T), a.copy(), method='L-BFGS-B',
                          bounds=[(1e-12, None)] * n,
                          options={'maxiter': max_iter, 'maxfun': max_iter * 2, 'disp': False})
        a_opt = np.maximum(res.x, 0)
        s_opt = float(np.sum(a_opt))
        if s_opt > 1e-10:
            a_opt = a_opt / s_opt
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                if verbose:
                    print(f"    Smooth T={T}: obj={best_obj:.8f}")
    except Exception:
        pass

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = np.maximum(0.01, 1.0 - np.abs(x))
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(0, 1 - x ** 2)) + 0.01
        a = a / a.sum()
    elif mode == 5:
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        center = n // 2
        a = np.exp(-np.abs(np.arange(n) - center) / (n // 4))
        a = a / a.sum()
    elif mode == 7:
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.01
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase optimization:
    Phase 1: LP alternating on parent sequence (main workhorse)
    Phase 2: Active-set gradient polish
    Phase 3: Multi-n exploration
    Phase 4: Diverse restarts with LP
    Phase 5: Final intensive polish (LP + gradient + smooth)
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Loaded parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: LP Alternating (main workhorse, ~550s) ===
    print(f"\n=== Phase 1: LP Alternating (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.38, 550.0)
    if p1_budget > 30:
        a_lp = lp_alternating(best_seq, time_budget=p1_budget, verbose=True, use_symmetry=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 1 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Active-set gradient polish (~100s) ===
    print(f"\n=== Phase 2: Active-Set Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = min(remaining * 0.08, 100.0)
        a_grad = active_set_gradient(best_seq, time_budget=p2_budget, max_active=8, verbose=True)
        obj_grad = compute_obj(a_grad)
        if obj_grad < best_score:
            best_score = obj_grad
            best_seq = a_grad.copy()
            print(f"  Phase 2 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_grad:.8f}")

        # LP polish after gradient
        remaining = deadline - time.monotonic()
        if remaining > 30:
            p2b_budget = min(remaining * 0.08, 80.0)
            a_lp2 = lp_alternating(best_seq, time_budget=p2b_budget, verbose=False, use_symmetry=True)
            obj_lp2 = compute_obj(a_lp2)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  Phase 2b LP: obj={best_score:.8f} **")

    # === Phase 3: Multi-n exploration (~350s) ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = [400, 500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600]
    n_options = [n for n in n_options if 300 <= n <= 2000 and n != n_current]
    # Sort by closeness to current n
    n_options.sort(key=lambda x: abs(x - n_current))
    n_options = n_options[:6]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        # Run LP + gradient on this n
        alloc_lp = min(remaining * 0.12, 100.0)
        if alloc_lp > 20:
            a_try = lp_alternating(a_try, time_budget=alloc_lp, verbose=False, use_symmetry=True)
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc_grad = min(remaining * 0.04, 40.0)
            if alloc_grad > 10:
                a_try = active_set_gradient(a_try, time_budget=alloc_grad, max_active=6)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Diverse restarts (~200s) ===
    print(f"\n=== Phase 4: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    restart_configs = [
        (0, 0.6),   # uniform, blend 0.6
        (4, 0.7),   # semicircle, blend 0.7
        (5, 0.5),   # periodic, blend 0.5
        (7, 0.6),   # cosine, blend 0.6
        (1, 0.4),   # random, blend 0.4
    ]

    for i, (mode, blend) in enumerate(restart_configs):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_r = diverse_init(n_best, rng, mode=mode)
        a_r = (1 - blend) * best_seq + blend * a_r
        a_r = np.maximum(a_r, 0)
        s_r = a_r.sum()
        if s_r > 1e-10:
            a_r = a_r / s_r

        restart_budget = min(remaining * 0.12, 60.0)
        a_opt = lp_alternating(a_r, time_budget=restart_budget, verbose=False, use_symmetry=True)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {i} (mode={mode}, blend={blend}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {i} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 5: Final intensive polish (~remaining) ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 30:
        lp_alloc = min(remaining * 0.35, 100.0)
        a_lp = lp_alternating(best_seq, time_budget=lp_alloc, verbose=True, use_symmetry=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Gradient polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        grad_alloc = min(remaining * 0.2, 50.0)
        a_grad = active_set_gradient(best_seq, time_budget=grad_alloc, max_active=12, verbose=True)
        obj_grad = compute_obj(a_grad)
        if obj_grad < best_score:
            best_score = obj_grad
            best_seq = a_grad.copy()
            print(f"  Final Grad: obj={best_score:.8f} **")

    # Smooth optimization
    remaining = deadline - time.monotonic()
    if remaining > 15:
        sm_alloc = min(remaining * 0.2, 30.0)
        a_sm = smooth_optimize(best_seq, time_budget=sm_alloc, T=40.0, verbose=True)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Final Smooth: obj={best_score:.8f} **")

    # One more LP round if time allows
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.5, 25.0), verbose=False, use_symmetry=True)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
