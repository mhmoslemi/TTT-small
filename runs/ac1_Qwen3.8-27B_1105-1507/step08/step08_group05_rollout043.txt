def compute_conv(a):

def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

def smoothed_lp_step(a, g_init, tau, max_iter=5000, verbose=False):

    g = g_init.copy()
    y = g.copy()

    lr = tau * 0.5  # conservative step size (Lipschitz constant ~ 1/tau)
    best_f = np.inf
    start = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - start > 30:
            break
        # Compute gradient at y
        grad = compute_grad_smoothed(a, y, tau)
        # Center the gradient (remove mean for better simplex behavior)
        grad = grad - np.mean(grad)
        # Gradient step
        g_new = project_simplex(y - lr * grad)
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = g_new + (t - 1.0) / t_new * (g_new - g)
        y_new = project_simplex(y_new)
        g = g_new
        y = y_new
        t = t_new
        
        # Evaluate objective periodically
        if it % 100 == 0:
            f_val = compute_obj(g)
            if f_val < best_f:
                best_f = f_val
    
    return g


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n - j, m - j)  # number of valid entries
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g_var = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g_var <= M, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g_var.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=100.0, verbose=False):

    n = len(a0)

        return a0.copy()
    a = a / s

            break
        # Try full update and blends
        candidates = [g_lp]
        for t in [0.3, 0.5, 0.7]:
            candidates.append((1.0 - t) * a + t * g_lp)
        
        best_candidate = None
        best_candidate_obj = best_obj
        for cand in candidates:

            sc = float(np.sum(cand))

            cand = cand / sc
            obj_c = compute_obj(cand)

                best_candidate = cand
        
        if best_candidate is not None:
            best_obj = best_candidate_obj
            best_a = best_candidate
            a = best_a
            stall = 0
            if verbose:

        else:
            stall += 1
            if stall > 10:
                break
            a = best_a * np.exp(np.random.default_rng().normal(0, 0.05, n))

def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def gaussian_window(n, sigma_frac=0.3):
    """Gaussian window centered at n/2."""
    x = np.arange(n)
    w = np.exp(-0.5 * ((x - n / 2.0) / (n * sigma_frac)) ** 2)
    return w / np.sum(w)


def raised_cosine_window(n, roll_off=0.5):
    """Raised cosine (Generalized Hamming) window."""
    x = np.arange(n)
    center = n / 2.0
    max_d = n * roll_off / 2.0
    w = np.ones(n)
    for i in range(n):
        d = abs(x[i] - center)
        if d > max_d - n * roll_off / 4:
            # In the transition region
            t = (d - (max_d - n * roll_off / 4)) / (n * roll_off / 4)
            t = min(max(t, 0), 1)
            w[i] = 0.5 * (1 + np.cos(np.pi * t))
    w = np.maximum(w, 1e-10)
    return w / np.sum(w)

"""
Smoothed LP alternating optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothing of the LP subproblem solved via FISTA (accelerated
projected gradient descent) with FFT-based gradient computation. Combines
diversified initialization, smoothed LP, exact LP, and Nesterov subgradient.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_diff(a, b):
    """Full linear convolution a*b via FFT."""
    na = len(a)
    nb = len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:na + nb - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def compute_grad_smoothed(a, g, tau, a_rev_fft=None, fft_size=None):
    """
    Compute gradient of f(g) = tau * log(sum_k exp(conv(a,g)[k]/tau)) w.r.t. g.
    grad[j] = sum_i a[i] * p[j+i] where p = softmax(conv(a,g)/tau)
    Uses FFT: grad = conv(a[::-1], p)[n-1 : 2n-1]
    """
    n = len(a)
    c = compute_conv_diff(a, g)
    c_scaled = c / tau
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    p = exp_c / np.sum(exp_c)

    a_rev = a[::-1]
    if fft_size is None:
        fft_size = 1
        while fft_size < 3 * n - 2:
            fft_size <<= 1
    if a_rev_fft is None:
        a_rev_fft = rfft(a_rev, fft_size)
    fp = rfft(p, fft_size)
    conv_result = irfft(a_rev_fft * fp, fft_size)
    grad = conv_result[n - 1: 2 * n - 1]
    return grad


def smoothed_lp_fista(a, g_init, tau, time_budget=60.0, verbose=False):
    """
    Minimize f(g) = tau * log(sum exp(conv(a,g)/tau)) over simplex via FISTA.
    """

    g = g_init.copy()

        g_new = project_simplex(y - lr * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = g_new + (t - 1.0) / t_new * (g_new - g)

        y = y_new

    return best_g


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n - j, m - j)
        if r > 0:
            A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g_var = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g_var <= M, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g_var.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_min_sum_top(a, m=5):
    """LP: min sum of top-m peaks of conv(a,g), approximated as min max + alpha*mean."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g_var = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    alpha = 0.3
    constraints = [A @ g_var <= M, cp.sum(g_var) == 1]
    obj = M + alpha * cp.sum(A @ g_var) / (2 * n - 1)
    prob = cp.Problem(cp.Minimize(obj), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g_var.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs
        except Exception:
            continue
    return None

def nesterov_subgrad(a0, time_budget=60.0, top_k=5, verbose=False):
    """Nesterov-accelerated subgradient on the simplex."""
    n = len(a0)
    a = a0.astype(np.float64).copy()

        return a0.copy()

            break
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                m_start = k - i_end
                m_end = k - i_start
                grad[m_start:m_end + 1] += 2.0 * y[i_start:i_end + 1][::-1]
        grad /= len(top_idx)
        grad = grad - np.mean(grad)

        y_new = y - step_size * grad
        a_new = project_simplex(y_new)
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1.0 - t_k) * (a_new - a)

                patience = 0

    return best_a


def coordinate_descent_peak(a0, time_budget=60.0, verbose=False):

def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def gaussian_window(n, sigma_frac=0.3):
    x = np.arange(n)
    w = np.exp(-0.5 * ((x - n / 2.0) / (n * sigma_frac)) ** 2)
    return w / np.sum(w)


def raised_cosine_window(n, roll_off=0.5):
    x = np.arange(n)
    center = n / 2.0
    half_width = n * roll_off / 2.0
    transition = n * (1 - roll_off) / 2.0
    w = np.ones(n)
    for i in range(n):
        d = abs(x[i] - center)
        if d > transition:
            t = (d - transition) / half_width
            t = min(max(t, 0.0), 1.0)
            w[i] = 0.5 * (1 + np.cos(np.pi * t))
    w = np.maximum(w, 1e-10)
    return w / np.sum(w)


def exponential_window(n, rate=3.0):
    x = np.arange(n)
    center = n / 2.0
    w = np.exp(-rate * abs(x - center) / (n / 2.0))
    return w / np.sum(w)

        n_best = len(best_seq)
        for r in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break
            scale = 0.01 + 0.008 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr < 1e-10:
                continue
            a_r = a_r / sr
            # Quick optimize: smoothed LP + Nesterov
            alloc = min(remaining * 0.25, 15.0)
            a_opt = smoothed_lp_fista(a_r, a_r, 0.01, time_budget=alloc * 0.5)
            a_opt = nesterov_subgrad(a_opt, time_budget=alloc * 0.5, top_k=5)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  R{r}: obj={best_score:.8f} **")
            else:
                print(f"  R{r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
