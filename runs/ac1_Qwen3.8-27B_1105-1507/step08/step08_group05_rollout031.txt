R = ifft(fft(a, 2*len) * conj(fft(w, 2*len)))... 

Actually, the standard formula: if we define the cross-correlation as

= sum_n sum_m x[n] y[m] exp(2πi(n-m)k/N)

= (1/N) sum_n sum_m x[n] y[m] sum_k exp(2πi(n-m+p)k/N)
= sum_n sum_m x[n] y[m] * delta_{n-m+p, 0 (mod N)}
= sum_n x[n] y[n+p (mod N)]

So ifft(conj(FFT(x)) * FFT(y))[p] = sum_n x[n] * y[(n+p) mod N]

With zero-padding to N >= len(x) + len(y) - 1, the circular wrap-around doesn't happen for valid indices. So:
R[p] = sum_n x[n] * y[n+p] for p = 0, ..., N-1 (with the understanding that y[n+p] = 0 if n+p >= len(y))

This is exactly what I want! g[m] = R[m] = ifft(conj(FFT(a)) * FFT(w_full))[m] for m = 0, ..., n-1.

"""
Hybrid optimizer: Log-param L-BFGS (smoothed max with FFT gradient) + reduced LP alternating + diverse inits.
Key improvements: FFT-accelerated gradient, tau annealing, reduced LP with violation checking, aggressive iteration counts.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def softmax_params_to_a(x):
    """Convert log-parameters to simplex point."""
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)


def a_to_log_params(a):
    """Convert simplex point to log-parameters."""
    a = np.maximum(a, 1e-15)
    s = np.sum(a)
    return np.log(a / s)


def smooth_obj_grad(x, tau, top_k):
    """
    Smoothed max objective and gradient w.r.t. log-parameters x.
    Uses FFT for O(n log n) gradient computation.
    """
    n = len(x)
    a = softmax_params_to_a(x)
    conv = compute_conv(a)

    if top_k is not None and top_k < len(conv):
        top_idx = np.argsort(conv)[::-1][:top_k]
    else:
        top_idx = np.arange(len(conv))

    conv_s = conv[top_idx]
    max_c = np.max(conv_s)
    shifted = tau * (conv_s - max_c)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)

    f_val = 2.0 * n * (max_c + np.log(np.sum(exp_s)) / tau)

    w_full = np.zeros(2 * n - 1)
    w_full[top_idx] = w

    # FFT cross-correlation: g[m] = sum_j a[j] * w_full[j+m]
    size = 1
    while size < 3 * n:
        size <<= 1
    A_c = np.fft.fft(a, size)
    W_c = np.fft.fft(w_full, size)
    g = np.fft.ifft(np.conj(A_c) * W_c)[:n].real

    g *= 4.0 * n
    sum_ag = np.dot(a, g)
    grad_a = g - a * sum_ag
    grad_x = a * grad_a

    return f_val, grad_x

def build_conv_matrix_rows(a, indices):
    """Build rows of convolution matrix for specific peak indices."""
    n = len(a)
    rows = []
    for k in indices:
        row = np.zeros(n)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for j in range(i_start, i_end + 1):
            row[j] = a[k - j]
        rows.append(row)
    return np.array(rows) if rows else np.zeros((0, n))


def solve_lp_reduced(a, top_k=50, time_limit=5.0):
    """
    Reduced LP: min M s.t. A_topk @ g <= M, sum(g)=1, g>=0.
    Uses only top_k peak constraints plus violation checking.
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    conv = compute_conv(a_n)
    peak_vals = np.max(conv)

    active_idx = np.argsort(conv)[::-1][:top_k]

    for iteration in range(5):
        A_rows = build_conv_matrix_rows(a_n, active_idx)
        if len(A_rows) == 0:
            return None, None

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A_rows @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        sol_found = False
        for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, **{'time_limit': time_limit} if solver != cp.SCS else {'max_iters': 5000})
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    gv = np.maximum(g.value, 0)
                    gs = float(np.sum(gv))
                    if gs > 1e-10:
                        gv = gv / gs
                        sol_found = True
                        break
            except Exception:
                continue

        if not sol_found:
            return None, None

        # Check for violated constraints
        conv_g = compute_conv(a_n * gv)
        # Actually conv(a, g) not conv(g, g). Let me compute A_full @ g
        # For violated check, compute conv(a_n, gv) at all indices
        # conv(a_n, gv)[k] = sum_j a_n[j] * gv[k-j]
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fa = rfft(a_n, size)
        fg = rfft(gv, size)
        cross_conv = irfft(fa * fg, size)[:2 * n - 1]

        violated = np.where(cross_conv > M.value * 1.001 + 1e-9)[0]

def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw

            a = a / np.sum(a)

    # === Phase 3: L-BFGS polish after LP (200s) ===
    print(f"\n=== Phase 3: L-BFGS Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(80.0, remaining * 0.4)
        a_pol = lbfgs_optimize(best_seq, time_budget=alloc,
                              tau_list=[10, 30, 100, 300, 1000],
                              top_k_list=[20, 50, 100], max_iter=50)
        obj = compute_obj(a_pol)
        if obj < best_score:
            best_score = obj
            best_seq = a_pol.copy()
            print(f"  LBFGS polish: obj={best_score:.8f} **")
        else:
            print(f"  LBFGS polish: obj={obj:.8f}")

    # === Phase 4: N-sweeping (250s) ===
    print(f"\n=== Phase 4: N-Sweep ===")
    n_best = len(best_seq)
    n_options = sorted(set([400, 500, 600, 700, 800, 900, 1000, 1200, 1500, 2000,
                            n_best, max(200, n_best - 100), n_best + 100, n_best + 200]))
    n_options = [n for n in n_options if 200 <= n <= 2500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)
        alloc = min(20.0, remaining * 0.15)
        a_opt = lbfgs_optimize(a_try, time_budget=alloc * 0.6,
                              tau_list=[10, 50, 200], top_k_list=[50], max_iter=25)
        # Quick LP
        a_opt = lp_alternate(a_opt, time_budget=alloc * 0.4, top_k=40)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj:.8f}")

    # === Phase 5: Final combined polish (150s) ===
    print(f"\n=== Phase 5: Final Polish ===")
    for phase in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        # L-BFGS
        alloc1 = min(30.0, remaining * 0.3)
        a1 = lbfgs_optimize(best_seq, time_budget=alloc1,
                           tau_list=[50, 200, 500, 1000], top_k_list=[20, 50], max_iter=40)
        if compute_obj(a1) < best_score:
            best_score = compute_obj(a1)
            best_seq = a1.copy()
        # LP
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        alloc2 = min(30.0, remaining * 0.3)
        a2 = lp_alternate(best_seq, time_budget=alloc2, top_k=50)
        if compute_obj(a2) < best_score:
            best_score = compute_obj(a2)
            best_seq = a2.copy()
        print(f"  Final phase {phase}: obj={best_score:.8f}")

    # === Phase 6: Random restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(15):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            scale = 0.01 + 0.003 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr < 1e-10:
                continue
            a_r = a_r / sr
            alloc = min(remaining * 0.25, 15.0)
            a_opt = lbfgs_optimize(a_r, time_budget=alloc * 0.5,
                                  tau_list=[20, 100, 400], top_k_list=[40], max_iter=20)
            a_opt = lp_alternate(a_opt, time_budget=alloc * 0.5, top_k=40)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score - 1e-12:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  R{r}: obj={best_score:.8f} **")
            else:
                print(f"  R{r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
