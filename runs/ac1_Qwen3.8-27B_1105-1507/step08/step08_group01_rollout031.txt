        n_active = min(n, m // 2)
    active_k = np.argsort(y)[::-1][:n_active]

    # Use scipy linprog
    from scipy.optimize import linprog

"""

- Symmetric initialization and optimization

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def solve_lp_fast(a, active_indices=None, max_iter=5000):
    """
    Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Uses scipy linprog with active set reduction.
    """
    n = len(a)
    a = a / np.sum(a)
    
    if active_indices is not None and len(active_indices) < 2 * n - 1:
        A = build_conv_matrix(a)
        A_red = A[active_indices, :]
        n_cons = len(active_indices)
    else:
        A = build_conv_matrix(a)
        A_red = A
        n_cons = 2 * n - 1
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((n_cons, n + 1))
    A_ub[:, :n] = A_red
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_cons)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_val = g_val / s
            # Verify against ALL constraints
            conv_check = build_conv_matrix(a) @ g_val
            M_true = float(np.max(conv_check))
            return g_val, M_true
    return None, None


def solve_by_pw(a, T=30000, eta_init=3.0):
    """
    Solve inner problem using multiplicative weights on the dual.
    Dual: max_y min_i (y * P)[:,i] where P[k,i] = a[k-i].
    Returns (g, M_estimate).
    """
    n = len(a)
    m = 2 * n - 1
    
    y = np.ones(m) / m
    eta = eta_init
    y_avg = y.copy()
    
    arange_m = np.arange(m)
    
    for t in range(T):
        # D[i] = sum_k y[k] * a[k-i] = cross-correlation
        D = np.correlate(y, a, mode='valid')
        i_star = np.argmin(D)
        
        # MW update: y_k <- y_k * exp(eta * a[k - i_star])
        offset = arange_m - i_star
        a_shifted = np.zeros(m)
        valid = (offset >= 0) & (offset < n)
        a_shifted[valid] = a[offset[valid]]
        
        y = y * np.exp(eta * a_shifted)
        y_sum = np.sum(y)
        if y_sum < 1e-300:
            y = np.exp(eta * a_shifted) / np.sum(np.exp(eta * a_shifted))
        else:
            y = y / y_sum
        
        if t % 10 == 0:
            y_avg = (t * y_avg + y) / (t + 1)
        
        eta = eta_init / (1.0 + 0.02 * t)
    
    # Estimate M* from dual
    D_avg = np.correlate(y_avg, a, mode='valid')
    M_est = float(np.min(D_avg))
    
    # Identify active constraints
    y_sorted = np.sort(y)[::-1]
    # Active set: where y is significantly above uniform
    threshold = max(y_sorted[m // 4] if m // 4 < len(y_sorted) else 1e-10, 
                    5.0 / m)
    active_k = np.where(y > threshold)[0]
    
    if len(active_k) < 5:
        active_k = np.argsort(y)[::-1][:n]
    
    # Solve reduced LP to get primal solution
    g, M_lp = solve_lp_fast(a, active_indices=active_k)
    if g is None:
        g = np.ones(n) / n
        M_lp = float(np.max(np.convolve(a, g)))
    
    return g, M_lp


def line_search_optimal(a, g, n_evals=12):
    """
    Find optimal t in [0,1] for a_new = (1-t)*a + t*g using golden section + parabolic.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)
    
    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)
    
    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val)), val
    
    # Golden section search
    golden = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - golden * (hi - lo)
    d = lo + golden * (hi - lo)
    fc, _ = obj_at_t(c)
    fd, _ = obj_at_t(d)
    
    for _ in range(30):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - golden * (hi - lo)
            fc, _ = obj_at_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + golden * (hi - lo)
            fd, _ = obj_at_t(d)
        
        if hi - lo < 1e-10:
            break
    
    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    a_new = a_new / s
    return t_opt, a_new

def smooth_bump_init(n, alpha=1.0):
    """Create a smooth bump initial sequence."""
    x = np.linspace(0, 1, n)
    # Raised cosine: (1 + cos(2*pi*(x-0.5)))/2 peaked at center
    # Or: sin^2(pi*x) which is zero at endpoints
    # Or: a general bump that's flat in the middle
    # Use Tukey (tapered cosine) window
    r = 0.5  # proportion of tapered ends
    a = np.ones(n)
    n_taper = int(r * n / 2)
    if n_taper > 0:
        t = np.linspace(0, np.pi / 2, n_taper)
        taper = 0.5 * (1 + np.cos(t * 2))
        # Wait, that's wrong. Tukey window:
        # For the tapered part: 0.5 * (1 + cos(pi * (2*x/r - 1 + 1)))
        # Simpler: use sin^2
        a[:n_taper] = 0.5 * (1 - np.cos(np.pi * np.linspace(0, 1, n_taper)))
        a[-n_taper:] = a[:n_taper][::-1]
    
    # Alternative: use a power bump
    x_centered = np.linspace(-1, 1, n)
    a = np.maximum(0, 1 - np.abs(x_centered) ** (2 * alpha))
    
    s = np.sum(a)
    return a / s


def smoothed_max_grad_desc(a, time_budget=60.0, T_init=0.001, T_min=0.00005):
    """Gradient descent with smooth max and adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    T = T_init
    lr = 0.01
    v = np.zeros(n)
    momentum = 0.9
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        if it > 10000:
            break
            
        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -400, 400)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        
        # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i]
        # This is 2 * cross-correlation of w and a
        grad = 2.0 * np.correlate(w, a, mode='valid')
        
        # Center gradient to preserve sum
        grad_centered = grad - np.mean(grad)
        
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if stall == 0 and it % 50 == 0:
                lr = min(lr * 1.05, 0.1)
        else:
            stall += 1
            if stall > 20:
                lr *= 0.8
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
                if lr < 1e-8:
                    break
        
        T = max(T_min, T_init * (0.9995 ** it))
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_symmetric(a):
    """Make a sequence symmetric: a[i] = a[n-1-i]."""
    n = len(a)
    half = (n + 1) // 2
    a_sym = (a[:half] + a[n - 1:n - half - 1:-1]) / 2.0
    result = np.zeros(n)
    result[:half] = a_sym
    result[half:] = a_sym[-1:-(half - (n - half) + 1):-1] if n % 2 == 0 else a_sym[-2::-1]
    # Simpler approach
    result = np.zeros(n)
    for i in range(n):
        result[i] = (a[i] + a[n - 1 - i]) / 2.0
    return result

        obj_sym = compute_obj(a_sym_opt)
        if obj_sym < best_score:
            best_score = obj_sym
            best_seq = a_sym_opt.copy()
            a = a_sym_opt.copy()
            print(f"  Phase 5 (sym) done: obj={best_score:.8f}")
        else:
            print(f"  Phase 5 (sym) done: obj={obj_sym:.8f} (no improvement)")
    
    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n ===")
    n_best = len(best_seq)
    n_options = []
    for dn in [-100, -50, -25, -10, 10, 25, 50, 100, 150, 200]:
        n_try = n_best + dn
        if 500 <= n_try <= 1500:
            n_options.append(n_try)
    # Also try some round numbers
    for n_round in [600, 700, 750, 800, 850, 900, 1000, 1100, 1200]:
        if 500 <= n_round <= 1500 and n_round not in n_options:
            n_options.append(n_round)
    n_options = sorted(set(n_options))
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break
        
        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        
        # Quick optimization
        alloc = min(remaining * 0.1, 30.0)
        a_opt = alternating_minimize(a_try, time_budget=alloc, use_pw=True)
        obj_try = compute_obj(a_opt)
        
        # CD polish if promising
        remaining = deadline - time.monotonic()
        if remaining > 20 and obj_try < best_score + 0.003:
            a_opt = coord_descent_2d(a_opt, time_budget=min(15.0, remaining * 0.3))
            obj_try = compute_obj(a_opt)
        
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = best_seq.copy()
        
        # More aggressive CD
        if remaining > 30:
            alloc = min(80.0, remaining * 0.3)
            a_cd = coord_descent_2d(a_final, time_budget=alloc, verbose=False)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                a_final = a_cd.copy()
                print(f"  Final CD: obj={best_score:.10f}")
        
        # Quick smooth max
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(40.0, remaining * 0.3)
            a_sm = smoothed_max_grad_desc(a_final, time_budget=alloc, T_init=0.0003, T_min=0.00001)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                a_final = a_sm.copy()
                print(f"  Final SmoothMax: obj={best_score:.10f}")
        
        # Tiny CD
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(30.0, remaining * 0.4)
            a_cd = coord_descent_2d(a_final, time_budget=alloc, verbose=False)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final tiny CD: obj={best_score:.10f}")
    
    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if np.isfinite(val) and val < best_score:
            best_score = val
    except Exception:
        pass
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    
    return result
