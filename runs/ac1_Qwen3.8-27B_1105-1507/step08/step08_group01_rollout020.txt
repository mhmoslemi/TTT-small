"""
Optimized search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements:
- Fast pairwise coordinate descent using O(n) sparse convolution updates (no FFT per step)
- Improved LP alternating with warm-started active sets
- Spectral-domain resampling for multi-n exploration
- Adaptive step sizes with backtracking
- Multiple diverse initializations to escape local minima

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def solve_lp_for_direction(a, active_indices=None, max_time=5.0):
    """Solve LP: min M s.t. conv(a,g) <= M for k in active (or all), sum(g)=1, g>=0."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A_full = build_conv_matrix(a)
    
    if active_indices is not None and len(active_indices) < 2 * n - 1:
        A = A_full[active_indices, :]
    else:
        A = A_full
    
    n_constraints = A.shape[0]
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((n_constraints, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_constraints)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': max_time})
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_norm = g_val / s
            # Verify against ALL constraints
            conv_check = A_full @ g_norm
            true_M = float(np.max(conv_check))
            return g_norm, true_M
    return None, None


def convex_blend(a, g, n_evals=12):
    """Find optimal blend t in [0,1] for a_new = (1-t)*a + t*g."""
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)
    
    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)
    
    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))
    
    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    a_new = a_new / s
    return t_opt, a_new

    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    
    max_iters = 500
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute current active set
        conv_cur = compute_conv(a)
        max_conv = float(np.max(conv_cur))
        threshold = max_conv * 0.97
        active_set = np.where(conv_cur >= threshold)[0]
        
        # Solve LP with active set
        g_lp, M_lp = solve_lp_for_direction(a, active_set=active_set, max_time=5.0)
        if g_lp is None:
            g_lp, M_lp = solve_lp_for_direction(a, active_set=None, max_time=5.0)
        if g_lp is None:

        # Convex blend
        t_opt, a_new = convex_blend(a, g_lp, n_evals=10)

    """
    Fast pairwise coordinate descent using O(n) sparse convolution updates.
    Transfer mass between pairs and track the max convolution value incrementally.
    """
    n = len(a)
    a = (np.maximum(a, 0) / np.sum(a)).copy().astype(np.float64)
    c = compute_conv(a)
    max_c = float(np.max(c))
    best_obj = 2.0 * n * max_c
    best_a = a.copy()
    
    start = time.monotonic()
    delta = 0.002
    rng = np.random.default_rng(777)
    stall = 0
    pass_count = 0
    
    # Precompute the max of left and right parts for quick global max
    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        
        # Random permutation for diversity
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            # Try a few random donors
            for _trial in range(5):
                j = int(rng.integers(n))
                if j == i:
                    continue
                if a[j] < delta * 0.3:
                    continue
                
                d = delta
                if a[j] < d:
                    d = a[j] * 0.5
                    if d < 1e-8:
                        continue
                
                # Compute affected range
                lo = min(i, j)
                hi = max(i + n - 1, j + n - 1)
                hi = min(hi, 2 * n - 2)
                
                # Extract affected portion
                affected = c[lo:hi + 1].copy()
                
                # Apply: c[k] += 2*d*a[k-i] - 2*d*a[k-j]
                ki_s = max(lo, i)
                ki_e = min(hi, i + n - 1)
                if ki_s <= ki_e:
                    affected[ki_s - lo:ki_e - lo + 1] += 2.0 * d * a[ki_s - i:ki_e - i + 1]
                
                kj_s = max(lo, j)
                kj_e = min(hi, j + n - 1)
                if kj_s <= kj_e:
                    affected[kj_s - lo:kj_e - lo + 1] -= 2.0 * d * a[kj_s - j:kj_e - j + 1]
                
                new_max_aff = float(np.max(affected))
                
                # Check unaffected regions
                max_unaff = 0.0
                if lo > 0:
                    max_unaff = max(max_unaff, float(np.max(c[:lo])))
                if hi < 2 * n - 2:
                    max_unaff = max(max_unaff, float(np.max(c[hi + 1:])))
                
                new_global_max = max(new_max_aff, max_unaff)
                
                if new_global_max < max_c - 1e-10:
                    # Accept the move
                    a[i] += d
                    a[j] -= d
                    c[lo:hi + 1] = affected
                    max_c = new_global_max
                    best_obj = 2.0 * n * max_c
                    best_a = a.copy()
                    improved = True
                    break
        
        if verbose and pass_count % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"      FastCD pass {pass_count}: obj={best_obj:.8f}, max_c={max_c:.8f}, delta={delta:.6f}, t={elapsed:.1f}s")
        
        if not improved:
            stall += 1
            delta *= 0.6
            if delta < 1e-7:
                break
            if stall > 80:
                break
        else:
            stall = 0
            # Slightly increase delta if improving well
            if stall == 0 and delta < 0.01:
                delta *= 1.05
    
    return best_a


def gradient_cd(a, time_budget=60.0, verbose=False):
    """
    Gradient-based CD: use the subgradient of max(conv) to guide updates.
    """
    n = len(a)
    a = (np.maximum(a, 0) / np.sum(a)).copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.05
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    T = 0.001
    
    max_iters = 3000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        
        # Soft weights on peaks
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -400, 400)
        exp_w = np.exp(shifted)
        w = exp_w / np.sum(exp_w)
        
        # Subgradient: grad[i] = sum_k w[k] * (a[k-i] + a[k-i+n-1]...) 
        # More precisely: d/d a[i] of sum_k w[k]*c[k]
        # c[k] = sum_m a[m]*a[k-m], so d c[k]/d a[i] = a[k-i] (if valid) + a[k-i] (if valid, the other term)
        # Actually for linear conv: d c[k] / d a[i] = a[k-i] (if 0<=k-i<=n-1)
        # So grad[i] = sum_k w[k] * a[k-i] (for valid k-i)
        
        grad = np.zeros(n)
        for i in range(n):
            # k-i valid when k in [i, i+n-1]
            ks = i
            ke = min(i + n - 1, 2 * n - 2)
            grad[i] = np.dot(w[ks:ke + 1], a[:ke - ks + 1])
        
        # Center the gradient (to maintain sum constraint approximately)
        grad_centered = grad - np.mean(grad)
        
        v = momentum * v - lr * grad_centered
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 200 == 0:
                lr *= 1.02
        else:
            stall += 1
            if stall > 30:
                lr *= 0.7
                if lr < 1e-5:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
        
        # Anneal T
        if it > 20:
            T = max(0.0001, T * 0.999)
    
    return best_a


def resample_sequence(a, new_n):
    """Resample via spectral interpolation for better quality."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    
    # Spectral resampling
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = rfft(a, size)
    
    new_size = 1
    while new_size < 2 * new_n:
        new_size <<= 1
    
    # Interpolate spectrum
    freqs_old = np.arange(size // 2 + 1)
    freqs_new = np.arange(new_size // 2 + 1)
    fa_interp = np.interp(freqs_new, freqs_old, fa.real, left=0, right=0)
    fa_interp_imag = np.interp(freqs_new, freqs_old, fa.imag, left=0, right=0)
    fa_new = fa_interp + 1j * fa_interp_imag
    
    a_new = irfft(fa_new, new_size)[:new_n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        # Fallback to linear interpolation
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, new_n)
        a_new = np.maximum(np.interp(x_new, x_old, a), 0)
        s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def resample_linear(a, new_n):
    """Simple linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating + fast pairwise CD + gradient CD + multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 50.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={len(parent)}): obj={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")
    
    n_best = len(best_seq)
    
    # === Phase 1: LP Alternating ===
    print(f"\n=== Phase 1: LP Alternating (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    phase1_budget = min(250.0, remaining * 0.25)
    print(f"  Budget: {phase1_budget:.0f}s")
    
    a_opt = lp_alternating(best_seq.copy(), time_budget=phase1_budget, verbose=True)
    obj_opt = compute_obj(a_opt)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        print(f"  After LP: obj={best_score:.8f}")
    else:
        print(f"  After LP: obj={obj_opt:.8f} (no improvement)")
    
    # === Phase 2: Fast Pairwise CD ===
    print(f"\n=== Phase 2: Fast Pairwise CD (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    phase2_budget = min(200.0, remaining * 0.25)
    print(f"  Budget: {phase2_budget:.0f}s")
    
    a_cd = fast_pairwise_cd(best_seq.copy(), time_budget=phase2_budget, verbose=True)
    obj_cd = compute_obj(a_cd)
    if obj_cd < best_score:
        best_score = obj_cd
        best_seq = a_cd.copy()
        print(f"  After FastCD: obj={best_score:.8f}")
    else:
        print(f"  After FastCD: obj={obj_cd:.8f} (no improvement)")
    
    # === Phase 3: Gradient CD ===
    print(f"\n=== Phase 3: Gradient CD (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    phase3_budget = min(80.0, remaining * 0.1)
    if remaining > 60:
        a_grad = gradient_cd(best_seq.copy(), time_budget=phase3_budget, verbose=False)
        obj_grad = compute_obj(a_grad)
        if obj_grad < best_score:
            best_score = obj_grad
            best_seq = a_grad.copy()
            print(f"  After GradCD: obj={best_score:.8f}")
        else:
            print(f"  After GradCD: obj={obj_grad:.8f} (no improvement)")
    
    # === Phase 4: Another LP round ===
    print(f"\n=== Phase 4: Second LP Round ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        phase4_budget = min(120.0, remaining * 0.2)
        a_opt = lp_alternating(best_seq.copy(), time_budget=phase4_budget, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  After LP2: obj={best_score:.8f}")
        else:
            print(f"  After LP2: obj={obj_opt:.8f} (no improvement)")
    
    # === Phase 5: Fast CD again ===
    print(f"\n=== Phase 5: Second Fast CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        phase5_budget = min(100.0, remaining * 0.15)
        a_cd = fast_pairwise_cd(best_seq.copy(), time_budget=phase5_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  After FastCD2: obj={best_score:.8f}")
        else:
            print(f"  After FastCD2: obj={obj_cd:.8f} (no improvement)")
    
    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_current = len(best_seq)
    
    # Generate candidate n values
    n_candidates = set()
    for dn in [-100, -75, -50, -25, -10, 10, 25, 50, 75, 100, 150, 200]:
        n_candidates.add(n_current + dn)
    n_candidates.update([500, 600, 700, 750, 800, 850, 900, 950, 1000, 1100, 1200])
    n_candidates = sorted([n for n in n_candidates if 400 <= n <= 1500])
    
    # Shuffle to try diverse values
    np.random.seed(seed)
    np.random.shuffle(n_candidates)
    # Put current n first
    if n_current in n_candidates:
        n_candidates.remove(n_current)
        n_candidates.insert(0, n_current)
    
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        if n_try == n_current:
            a_try = best_seq.copy()
        else:
            # Try both spectral and linear resampling
            a_spec = resample_sequence(best_seq, n_try)
            a_lin = resample_linear(best_seq, n_try)
            # Pick the better one
            obj_spec = compute_obj(a_spec)
            obj_lin = compute_obj(a_lin)
            a_try = a_spec if obj_spec < obj_lin else a_lin
        
        # Quick LP
        alloc_lp = min(remaining * 0.2, 25.0)
        a_opt = lp_alternating(a_try, time_budget=alloc_lp, verbose=False)
        obj_try = compute_obj(a_opt)
        
        # Quick fast CD
        remaining = deadline - time.monotonic()
        if remaining > 20 and obj_try < best_score + 0.01:
            alloc_cd = min(remaining * 0.2, 20.0)
            a_opt = fast_pairwise_cd(a_opt, time_budget=alloc_cd, verbose=False)
            obj_try = compute_obj(a_opt)
        
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    
    if remaining > 40:
        # LP
        alloc = min(60.0, remaining * 0.3)
        a_lp = lp_alternating(best_seq.copy(), time_budget=alloc, verbose=False)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f}")
        
        # Fast CD
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(60.0, remaining * 0.3)
            a_cd = fast_pairwise_cd(best_seq.copy(), time_budget=alloc, verbose=False)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final CD: obj={best_score:.8f}")
        
        # Gradient
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(30.0, remaining * 0.3)
            a_g = gradient_cd(best_seq.copy(), time_budget=alloc, verbose=False)
            obj_g = compute_obj(a_g)
            if obj_g < best_score:
                best_score = obj_g
                best_seq = a_g.copy()
                print(f"  Final Grad: obj={best_score:.8f}")
    
    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
