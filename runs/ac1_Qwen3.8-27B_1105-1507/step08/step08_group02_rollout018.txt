import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * np.max(conv)

def solve_lp(a):
    """Solve LP: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0"""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix A: (2n-1) x n
    # A[k, j] = a_norm[k-j] if 0 <= k-j < n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        end = min(n, 2*n-1-j)
        A[j:j+end, j] = a_norm[:end]
    
    # LP: min M
    # s.t. A@g - M <= 0  for each row
    #      sum(g) = 1
    #      g >= 0, M free
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        return g / g.sum()
    return None

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2*n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < 2*n - 1:
        fft_size <<= 1
    fa = rfft(a, fft_size)
    return irfft(fa * fa, fft_size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(a):
    """Solve LP: min M s.t. (a_norm * g)_k <= M for all k, sum(g)=1, g>=0.
    Returns g (normalized) or None."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a_norm = a / s
    
    # Build A matrix: A[k,j] = a_norm[k-j] if 0 <= k-j < n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        end = min(n, 2*n-1-j)
        A[j:j+end, j] = a_norm[:end]
    
    # LP: min M
    # s.t. A@g - M*1 <= 0
    #      sum(g) = 1
    #      g >= 0
    # Variables: [g, M]
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)]*n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g))
        if gs > 1e-10:
            return g / gs
    return None

            g = g / g_norm

        v = momentum_coeff * v - lr * g
        a_new = project_simplex(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            a = a_new.copy()  # Still move, but don't update best
            stall += 1

        lr = max(0.005, lr * 0.998)
        if stall > 200:
            break

        it += 1
        if verbose and it % 100 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}, lr={lr:.4f}, elapsed={time.monotonic()-start:.1f}s")

    return best_a


def coordinate_descent_polish(a, time_budget=60.0, verbose=False):
    """Coordinate descent: try moving mass from peak contributors to reduce max conv."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    max_rounds = 20
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break

        k_star = int(np.argmax(conv))
        improved_in_round = False

        # Find top contributors to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                c = a[i] * a[j]
                contributors.append((c, i, j))
        contributors.sort(reverse=True)

        # Non-contributor targets (positions not involved in the peak)
        contrib_set = set()
        for _, i, j in contributors[:20]:
            contrib_set.add(i)
            if i != j:
                contrib_set.add(j)
        targets = [k for k in range(n) if k not in contrib_set]

        if not targets:
            break

        for c_val, i, j in contributors[:15]:
            if time.monotonic() - start > time_budget:
                break
            if c_val < 1e-10:
                break

            src = i if a[i] >= a[j] else j
            for target in targets[:6]:
                for frac in [0.02, 0.05, 0.1, 0.15, 0.2]:
                    delta = frac * a[src]
                    if delta < 1e-12 or delta >= a[src]:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved_in_round = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break

        if not improved_in_round:
            break

        if verbose and rnd % 5 == 0:
            print(f"    CD round{rnd}: obj={best_obj:.8f}, elapsed={time.monotonic()-start:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * (x - 0.3) ** 2) + np.exp(-2 * (x - 0.7) ** 2)
        return a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = (1 + x) * (1 + x) * (1 - x) + 0.01
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.4 * np.sin(2 * np.pi * 2 * x + rng.uniform(0, 6.28))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 5:
        a = rng.dirichlet(np.ones(n) * 0.5)
        return a
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x * (1 - x) + 0.05
        return a / a.sum()
    else:
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP-alternating with momentum, dimension sweep,
    subgradient polishing, coordinate descent, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial best: n={n_current}, obj={best_score:.8f}")

    # ============================================================
    # Phase 1: Quick dimension and initialization exploration
    # ============================================================
    print(f"\n=== Phase 1: Quick Exploration ===")
    remaining = deadline - time.monotonic()

    # Try different n values with a few initializations each
    n_options = sorted(set([
        n_current - 100, n_current, n_current + 100,
        n_current + 200, 500, 700, 900, 1100, 1300
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1600]

    candidates = []
    exp_budget = min(remaining * 0.15, 200.0)
    per_candidate = max(15.0, exp_budget / (len(n_options) * 2))

    for n_try in n_options[:7]:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Try 2 initializations per dimension
        for mode in [0, 4]:
            alloc = min(per_candidate, remaining * 0.1)
            if alloc < 10:
                break

            if n_try == n_current and best_seq is not None:
                a_init = best_seq.copy()
            else:
                if best_seq is not None and abs(n_try - n_current) <= 200:
                    a_init = resample_sequence(best_seq, n_try)
                    a_init = 0.5 * a_init + 0.5 * diverse_init(n_try, rng, mode)
                    a_init = a_init / a_init.sum()
                else:
                    a_init = diverse_init(n_try, rng, mode)

            a_opt = lp_alternate(a_init, time_budget=alloc)
            obj_opt = compute_obj(a_opt)
            candidates.append((obj_opt, a_opt.copy(), n_try))
            print(f"  n={n_try}, mode={mode}: obj={obj_opt:.8f}")

    candidates.sort(key=lambda x: x[0])
    if candidates:
        best_obj_c, best_a_c, best_n_c = candidates[0]
        if best_obj_c < best_score:
            best_score = best_obj_c
            best_seq = best_a_c.copy()
            n_current = best_n_c
            print(f"  ** Phase 1 best: n={n_current}, obj={best_score:.8f}")

    # ============================================================
    # Phase 2: Intensive LP alternating on the best
    # ============================================================
    print(f"\n=== Phase 2: Intensive LP Alternating ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.4, 500.0)

    if p2_budget > 30:
        a_opt = lp_alternate(best_seq, time_budget=p2_budget, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** Phase 2: obj={best_score:.8f} (improved)")
        else:
            print(f"  Phase 2: obj={obj_opt:.8f} (no improvement)")

    # ============================================================
    # Phase 3: Subgradient polishing + coordinate descent
    # ============================================================
    print(f"\n=== Phase 3: Polishing ===")
    remaining = deadline - time.monotonic()

    if remaining > 60:
        sg_budget = min(remaining * 0.15, 120.0)
        a_sg = subgradient_polish(best_seq, time_budget=sg_budget, topK=4, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  ** Subgradient: obj={best_score:.8f}")
        else:
            print(f"  Subgradient: obj={obj_sg:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 40:
        cd_budget = min(remaining * 0.15, 80.0)
        a_cd = coordinate_descent_polish(best_seq, time_budget=cd_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  ** Coordinate Descent: obj={best_score:.8f}")
        else:
            print(f"  Coordinate Descent: obj={obj_cd:.8f}")

    # ============================================================
    # Phase 4: Perturbation + LP restarts to escape local minima
    # ============================================================
    print(f"\n=== Phase 4: Perturbation Restarts ===")
    remaining = deadline - time.monotonic()

    for restart in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        alloc = min(remaining * 0.1, 60.0)
        if alloc < 20:
            break

        # Perturb the best solution
        noise_scale = 0.02 + 0.01 * restart
        a_pert = best_seq * (1.0 + noise_scale * rng.normal(0, 1, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        # Sometimes blend with a diverse init
        if rng.random() < 0.3:
            blend = 0.1 + 0.1 * restart
            a_div = diverse_init(len(best_seq), rng, mode=restart % 7)
            a_pert = (1 - blend) * a_pert + blend * a_div
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()

        a_opt = lp_alternate(a_pert, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** Restart {restart}: obj={best_score:.8f}")
        else:
            print(f"  Restart {restart}: obj={obj_opt:.8f}")

    # ============================================================
    # Phase 5: Try resampling to nearby dimensions
    # ============================================================
    print(f"\n=== Phase 5: Dimension Adjustment ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    nearby_ns = [n_best - 50, n_best + 50, n_best - 100, n_best + 100]
    nearby_ns = [n for n in nearby_ns if 300 <= n <= 2000]

    for n_adj in nearby_ns:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        alloc = min(remaining * 0.08, 50.0)
        if alloc < 15:
            continue

        a_resamp = resample_sequence(best_seq, n_adj)
        a_opt = lp_alternate(a_resamp, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** Resample n={n_adj}: obj={best_score:.8f}")
        else:
            print(f"  Resample n={n_adj}: obj={obj_opt:.8f}")

    # ============================================================
    # Phase 6: Final intensive LP + polish
    # ============================================================
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 50:
        final_budget = min(remaining * 0.4, 150.0)
        a_final = lp_alternate(best_seq, time_budget=final_budget, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final LP: obj={best_score:.8f}")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_sg2 = subgradient_polish(best_seq, time_budget=min(remaining * 0.2, 50.0), topK=3)
        obj_sg2 = compute_obj(a_sg2)
        if obj_sg2 < best_score:
            best_score = obj_sg2
            best_seq = a_sg2.copy()
            print(f"  ** Final SG: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_cd2 = coordinate_descent_polish(best_seq, time_budget=min(remaining * 0.3, 40.0))
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  ** Final CD: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternate(best_seq, time_budget=min(remaining * 0.5, 30.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  ** Last LP: obj={best_score:.8f}")

    # ============================================================
    # Final validation
    # ============================================================
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
