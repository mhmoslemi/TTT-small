def compute_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] w.r.t. a."""
    n = len(a)

    i_end = min(n-1, k)

        j_end = k - i_start

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=5.0):

        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            try:

                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    sv = float(np.sum(g_val))
                    if sv > 1e-10:
                        return g_val / sv, float(M.value)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))

    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))

        w = np.exp(z)
        w_sum = w.sum()

    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        contrib = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib.append((a[i] * a[j], i, j))
        contrib.sort(key=lambda x: -x[0])

        non_contrib = [i for i in range(n) if not (0 <= k_star - i < n)]
        if not non_contrib:
            break

        improved = False
        for val, i, j in contrib[:10]:
            if val < 1e-12:
                break
            for target in non_contrib[:5]:
                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * a[min(i, j)]
                    if delta < 1e-15:
                        continue
                    src = min(i, j) if i == j else (i if a[i] > a[j] else j)
                    a[src] -= delta
                    a[target] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
            if improved:
                break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x) * 0.5
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.05))
        a = a / a.sum()
    elif mode == 5:
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Geometric decay from center
        x = np.linspace(-1, 1, n)
        a = np.exp(-2.0 * x**2)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Hybrid optimization combining mirror descent with BB step,
    Nesterov subgradient, LP alternating with quadratic line search, adaptive
    log-sum-exp, multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Nesterov subgradient (fast warmup) ===
    print(f"\n=== Phase 1: Nesterov Subgradient (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 80.0)
    if p1_budget > 10:
        a_ns = nesterov_subgradient(best_seq, time_budget=p1_budget, K=7, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_ns:.8f}")

    # === Phase 2: Mirror descent with BB step (main workhorse) ===
    print(f"\n=== Phase 2: Mirror Descent + BB (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 20:
        a_md = mirror_descent_bb(best_seq, time_budget=p2_budget, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Mirror+BB: obj={best_score:.8f} **")
        else:
            print(f"  Mirror+BB: obj={obj_md:.8f}")

    # === Phase 3: LP Alternating with quadratic line search ===
    print(f"\n=== Phase 3: LP Alternating v2 ===")
    remaining = deadline - time.monotonic()
    p3_total = min(remaining * 0.3, 400.0)

    n_rounds = 3
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p3_total / (n_rounds - rnd), remaining * 0.6)
        if alloc < 20:
            break

        if rnd > 0:
            a_pert = best_seq * np.exp(rng.normal(0, 0.02 * (rnd + 1), len(best_seq)))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
        else:
            a_pert = best_seq.copy()

        a_lp = lp_alternating_v2(a_pert, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f}")

    # === Phase 4: Adaptive log-sum-exp ===
    print(f"\n=== Phase 4: Adaptive Log-Sum-Exp ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 150.0)
    if p4_budget > 15:
        a_sm = smooth_optimize_adaptive(best_seq, time_budget=p4_budget, verbose=True)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Smooth: obj={best_score:.8f} **")
        else:
            print(f"  Smooth: obj={obj_sm:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(200, n_current - 100),
        max(200, n_current - 200),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        400, 500, 600, 700, 800, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50 or n_tried >= 8:
            break
        if n_try == n_current:
            continue

        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1

        # Quick optimize: mirror descent + short LP
        alloc_md = min(remaining * 0.08, 40.0)
        if alloc_md > 10:
            a_try = mirror_descent_bb(a_try, time_budget=alloc_md)

        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.08, 40.0)
        if alloc_lp > 10:
            a_try = lp_alternating_v2(a_try, time_budget=alloc_lp)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts with blending ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = r % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.15 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.12, 50.0)
        if restart_budget > 10:
            a_opt = mirror_descent_bb(a_r, time_budget=restart_budget)
            obj_r = compute_obj(a_opt)
        else:
            a_opt = a_r
            obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        lp_alloc = min(remaining * 0.25, 60.0)
        a_lp = lp_alternating_v2(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        md_alloc = min(remaining * 0.2, 40.0)
        a_md = mirror_descent_bb(best_seq, time_budget=md_alloc)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        cd_alloc = min(remaining * 0.15, 20.0)
        a_cd = coordinate_refine(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more LP if time allows
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_alternating_v2(best_seq, time_budget=min(remaining * 0.5, 30.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
