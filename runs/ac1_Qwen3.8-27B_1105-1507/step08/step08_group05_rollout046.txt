def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:

    return 2.0 * n * float(np.max(conv))


def project_simplex(v):

def build_conv_matrix_a(a_norm):

def build_conv_matrix_a_fast(a_norm):

    n = len(a_norm)

    k_idx = np.arange(m)[:, None]  # m x 1
    j_idx = np.arange(n)[None, :]  # 1 x n
    diff = k_idx - j_idx  # m x n

def solve_lp(a, max_iters=10):
    """
    Solve: min M s.t. conv(a, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns normalized g.
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    # Use the fact that conv(a, g)[k] = sum_j a[k-j] * g[j]
    # So the constraint matrix A has A[k,j] = a[k-j] (with a=0 outside [0,n-1])
    
    # Build A efficiently
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # A[k, j] = a[k-j] for k from j to j+n-1 (clipped to 0..m-1)
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a_n[:k_end - k_start]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs
        except Exception:
            continue
    return None


def lp_alternate(a0, time_budget=60.0, verbose=False):
    """LP alternating with line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5

    while time.monotonic() - start < time_budget:
        g = solve_lp(a)
        if g is None:
            break
        
        # Line search: minimize obj((1-t)*a + t*g)
        def line_obj(t):
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)

                                  options={'maxiter': 25, 'xatol': 0.02})
            best_t = max(0.01, min(1.0, res.x))
        except:
            best_t = last_t
        
        a_try = (1.0 - best_t) * a + best_t * g
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))

            break
        a_try = a_try / st

            a = best_a
            last_t = best_t
            stall = 0
        else:

                break
    
    return best_a

            break

        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                m = k - i
                grad[m] += y[i]
                if m != i:
                    grad[i] += y[m]

def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def power_law(n, p):
    """a[i] = (i+1)^p, normalized."""
    i = np.arange(n) + 1.0
    a = i ** p
    return a / np.sum(a)


def exp_ramp(n, p):
    """a[i] = exp(p * i / (n-1)), normalized."""
    if n == 1:
        return np.array([1.0])
    i = np.arange(n)
    a = np.exp(p * i / (n - 1))
    return a / np.sum(a)


def skew_beta(n, alpha, beta):
    """Approximate beta distribution on [0, 1] sampled at n points."""
    x = (np.arange(n) + 0.5) / n
    a = x ** (alpha - 1) * (1 - x) ** (beta - 1)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def sine_shape(n, p):
    """a[i] = sin((i+1)*pi/(2n))^p."""
    i = np.arange(n) + 1.0
    a = np.sin(i * np.pi / (2 * n)) ** p
    return a / np.sum(a)

= (k+1)*n(n-k) + k*k(k+1)/2 - k(k+1)(2k+1)/6
= (k+1)[n(n-k) + k^2/2 - k(2k+1)/6]
= (k+1)[n(n-k) + 3k^2/6 - (2k^2+k)/6]
= (k+1)[n(n-k) + k^2/6 - k/6]
= (k+1)[n(n-k) + k(k-1)/6]

So conv[k] = 4(k+1)[n(n-k) + k(k-1)/6] / (n^2(n+1)^2) for 0 <= k < n.

At k=0: conv[0] = 4*n*n / (n^2(n+1)^2) = 4/(n+1)^2. For large n ≈ 4/n^2.
At k=n-1: conv[n-1] = 4*n*[n*1 + (n-1)(n-2)/6] / (n^2(n+1)^2)
= 4*n*[n + (n^2-3n+2)/6] / (n^2(n+1)^2)
= 4*n*[6n + n^2 - 3n + 2]/6 / (n^2(n+1)^2)
= 4*n*(n^2+3n+2)/6 / (n^2(n+1)^2)
= 4*n*(n+1)(n+2)/6 / (n^2(n+1)^2)
= 4(n+2) / (6n(n+1))
= 2(n+2) / (3n(n+1))
≈ 2/(3n) for large n. ✓
