Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP blinding, mirror descent, and smart perturbations.

from scipy.optimize import linprog
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    
    # Build constraint matrix A (2n-1 x n): A[k,j] = a[k-j] if 0<=k-j<n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.ones((1, n + 1))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g))
        if gs > 1e-10:
            return g / gs, float(res.x[n])
    return None, None


def subgradient_direction(a, n_peaks=10):
    """Compute subgradient direction for max conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    # Get top peaks
    peak_indices = np.argsort(conv)[::-1][:n_peaks]
    
    grad = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        for j in range(n):
            d = k - j
            if 0 <= d < n:
                grad[j] += a[d]
    grad *= 2.0 / n_peaks
    return grad, conv


def mirror_descent(a, time_budget, n_peaks=10, verbose=False):
    """Mirror descent (exponential weights) with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 1.0  # Step size
    patience = 0
    max_patience = 50
    
    while time.monotonic() - start < time_budget:
        grad, conv = subgradient_direction(a, n_peaks)
        
        # Mirror descent update (exponential weights / multiplicative weights)
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= np.sum(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
            # Try smaller step
            a_new2 = a * np.exp(-0.5 * eta * grad)
            a_new2 = np.maximum(a_new2, 1e-30)
            a_new2 /= np.sum(a_new2)
            obj_new2 = compute_obj(a_new2)
            if obj_new2 < best_obj - 1e-12:

                a = a_new2

**Key insight about LP alternating convergence:**

The LP alternating method defines a mapping `T(a) = g*` where `g*` is the LP solution given `a`. Then we iterate `a_{k+1} = (1-t) a_k + t T(a_k)`.

This is a contractive mapping if `t` is small enough (under some conditions). The fixed points of `T` are the stationary points of the original problem.

The issue is that there can be many fixed points (local minima), and the method converges to whichever one the iteration path leads to.

To escape a bad fixed point, we need to perturb `a` in a way that moves it to the basin of attraction of a better fixed point.

"""
Optimizer combining LP blinding, mirror descent, and structured perturbations
to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    return 2.0 * n * float(np.max(compute_conv(a_n)))


def solve_lp(a):
    n = len(a)
    a = a / np.sum(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.ones((1, n + 1))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g))
        if gs > 1e-10:
            return g / gs, float(res.x[n])
    return None, None


def subgrad_dir(a, n_peaks=10):
    n = len(a)
    conv = compute_conv(a)
    peaks = np.argsort(conv)[::-1][:n_peaks]
    grad = np.zeros(n)
    for k in peaks:
        k = int(k)
        for j in range(n):
            d = k - j
            if 0 <= d < n:
                grad[j] += a[d]
    grad *= 2.0 / n_peaks
    return grad
