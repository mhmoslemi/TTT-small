Enhanced optimizer: Hybrid LP/QP alternating + Nesterov subgradient + multiplicative updates.
Key improvements over previous:

2. Top-20% active set for smoother updates

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize_scalar
import time

def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def conv_grad_at_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        m_start = k - i_end
        m_end = k - i_start
        grad[m_start:m_end + 1] = 2.0 * a[i_start:i_end + 1][::-1]
    return grad

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

def build_conv_matrix(a, active_set=None):
    """Build convolution matrix for active set of lag positions."""
    n = len(a)
    if active_set is None:
        active_set = range(2 * n - 1)
    m = len(active_set)
    A = np.zeros((m, n))
    for idx, k in enumerate(active_set):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        A[idx, i_start:i_end+1] = a[i_start:i_end+1][::-1]
    return A

def solve_lp_minpeak(a, active_frac=0.2):
    """LP: min M s.t. (a*g)[k] <= M for active peaks, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    conv_a = compute_conv(a_n)
    # Select active set: top fraction of peaks
    n_active = max(10, int(len(conv_a) * active_frac))
    active_idx = np.argsort(conv_a)[::-1][:n_active]
    A = build_conv_matrix(a_n, active_idx)
    # LP: min M
    # Variables: g[0..n-1], M
    # Constraints: A@g - M <= 0 (for each row)
    #              sum(g) = 1
    #              g >= 0
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    # Inequality constraints: A@g - M <= 0
    A_ub = np.zeros((A.shape[0], n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(A.shape[0])
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g))
        if gs > 1e-10:
            return g / gs
    return None

def solve_qp_minl2(a, active_frac=0.2):
    """QP: min ||a*g||_2^2 s.t. sum(g)=1, g>=0."""
    # This is: min g^T (A^T A) g s.t. sum(g)=1, g>=0
    # We can use scipy minimize with SLSQP for small n, or use the KKT conditions
    # For efficiency, use the fact that at optimum, (A^T A) g = lambda * 1 (on support)
    # This is an eigenvalue-like problem on the support
    # Simpler: use scipy.optimize.minimize
    from scipy.optimize import minimize
    n = len(a)
    a_n = a / float(np.sum(a))
    conv_a = compute_conv(a_n)
    n_active = max(10, int(len(conv_a) * active_frac))
    active_idx = np.argsort(conv_a)[::-1][:n_active]
    A = build_conv_matrix(a_n, active_idx)
    ATA = A.T @ A
    
    def obj(g):
        return g @ ATA @ g
    
    def grad(g):
        return 2 * ATA @ g
    
    x0 = a_n.copy()  # warm start
    constraints = [{'type': 'eq', 'fun': lambda g: np.sum(g) - 1,
                    'jac': lambda g: np.ones(len(g))}]
    bounds = [(0, None)] * n
    res = minimize(obj, x0, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': 200, 'ftol': 1e-10})
    if res.success:
        g = np.maximum(res.x, 0)
        gs = float(np.sum(g))
        if gs > 1e-10:
            return g / gs
    return None

def nesterov_subgrad(a0, time_budget=60.0, top_k=5, verbose=False):
    """Nesterov-accelerated subgradient on the simplex with EMA."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.1
    prev_step = step_size
    t_k = 2.0
    patience = 0
    ema_grad = np.zeros(n)
    beta = 0.7

        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            grad += conv_grad_at_k(y, int(k))
        grad /= len(top_idx)
        # EMA of gradient
        ema_grad = beta * ema_grad + (1 - beta) * grad
        # Center for simplex
        ema_grad = ema_grad - np.mean(ema_grad)
        # Adaptive step size
        t_k = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2

        a_new = project_simplex(y_new)

For a normalized sequence (sum=1), this is `2 * n * max(conv(a,a))`.

For the uniform sequence of length n: max(conv) = 1/n, so objective = 2.

For a sequence that's concentrated: max(conv) is larger, so objective > 2.

So the uniform sequence gives exactly 2, and we want to go below 2. The parent achieves 1.506330, which is well below 2.

How is this possible? Because the uniform sequence's convolution is triangular, and the peak is at the center. But if we make the sequence non-uniform in the right way, we can make the convolution flatter, reducing the peak below 1/n.

For example, if a is "U-shaped" (more mass at the edges), the convolution will have a "dip" at the center and peaks at the edges... wait, that would increase the peak.

If a is "inverted U-shaped" (more mass in the middle), the convolution would have a sharper peak at the center.

Hmm, let me think about this differently. The convolution (a*a)[k] = sum_i a[i]*a[k-i]. For the uniform a[i] = 1/n, this gives (k+1)/n^2 for k < n. The peak is at k=n-1 with value 1/n.

To reduce this peak, we need to reduce the "alignment" of a with its reverse. If a[i] is high where a[n-1-i] is low, then (a*a)[n-1] = sum_i a[i]*a[n-1-i] will be lower.

So the strategy is: make a asymmetric! Put more mass where the "opposite" side has less mass.

For example, if a is increasing (a[0] < a[1] < ... < a[n-1]), then (a*a)[n-1] = sum_i a[i]*a[n-1-i] is a "cross-correlation" of a with its reverse, which is smaller than sum a[i]^2 (the peak for symmetric a).

But wait, we need to check ALL positions k, not just k=n-1. For an increasing sequence, (a*a)[2n-2] = a[n-1]^2 might be large.

So the optimal sequence balances: reduce the central peak by being asymmetric, but don't create large peaks at the edges.

This suggests the optimal sequence might have a "sideways" or "diagonal" structure.
