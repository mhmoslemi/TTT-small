from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def fast_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    a = a / s

def fast_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2n * max(conv(a,a)) for normalized a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = fast_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def fft_subgradient(a, active_indices):
    """
    Compute subgradient of max_{k in S} (a*a)[k] using FFT.
    subgrad[i] = (2/|S|) * sum_{k in S, 0<=k-i<n} a[k-i]
    """

    m = 2 * n - 1
    S = len(active_indices)
    if S == 0:
        return np.zeros(n)

    # Build w vector: w[k] = 1 if k in active_indices
    w = np.zeros(m)
    w[active_indices] = 1.0

    # subgrad[i] = (2/S) * sum_{j=0}^{n-1} a[j] * w[i+j]
    # = (2/S) * conv(flip(a), w)[n-1+i]

def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def nesterov_subgrad(a, time_budget=200.0, verbose=False, top_k=20):
    """
    Nesterov-accelerated subgradient method for minimizing 2n*max(conv(a,a)).
    Uses FFT-based O(n log n) subgradient computation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()       # current point
    x = a.copy()       # Nesterov lookahead
    momentum = np.zeros(n)
    beta = 0.7
    lr = 0.1
    lr_decay = 0.9995
    stall_count = 0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        # Compute convolution at lookahead point
        conv_x = fast_conv(x)
        max_val = float(np.max(conv_x))

        # Find active set: top-k peaks
        if top_k >= 2 * n - 1:
            active = np.arange(2 * n - 1)
        else:
            active = np.argsort(conv_x)[::-1][:top_k]

        # Compute subgradient at x (using a for the "linearization")
        grad = fft_subgradient(a, active)

        # Center gradient (project to sum=0 subspace for simplex)
        grad_centered = grad - np.mean(grad)

        # Nesterov momentum update
        momentum = beta * momentum - lr * grad_centered
        x_new = y + momentum
        y_new = project_simplex(x_new)

        # Evaluate
        obj = compute_obj(y_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = y_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive step size
        if stall_count > 100:
            lr *= 0.9
            stall_count = 0
            # Perturb to escape
            if stall_count % 500 == 0 or (it % 2000 == 0 and lr < 0.01):
                rng = np.random.default_rng(it)
                scale = 0.05
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = project_simplex(a_pert)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                y = a_pert.copy()
                x = a_pert.copy()
                momentum = np.zeros(n)
                lr = 0.1
                stall_count = 0

        # Update Nesterov parameters
        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        x = y_new + (1 - t_k) * (y_new - y)
        x = project_simplex(x)
        y = y_new

        # Decay learning rate
        lr *= lr_decay
        if lr < 0.001:
            lr = 0.001

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    NSub: it={it}, obj={best_obj:.8f}, lr={lr:.5f}, time={elapsed:.0f}s")

    return best_a

def lp_best_response(a):
    """
    Solve LP: minimize M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns optimized g.
    """
    n = len(a)
    m = 2 * n - 1
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None
    a = a / s

    # Build constraint matrix: (a*g)[k] = sum_i g[i] * a[k-i]
    # For each k, the constraint is: sum_i a[k-i] * g[i] <= M (for valid k-i)
    # Using scipy linprog:
    # min M
    # s.t. sum_i a[k-i]*g[i] - M <= 0 for all k
    #      sum_i g[i] = 1
    #      g[i] >= 0

    # A_ub has m + n + 1 columns (n for g, 1 for M)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for k in range(m):
        for i in range(n):
            ki = k - i
            if 0 <= ki < n:
                A_ub[k, i] = a[ki]
        A_ub[k, n] = -1.0  # -M

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        from scipy.optimize import linprog
        res = linprog(np.zeros(n + 1) + np.array([0]*n + [1]),
                      A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g))
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

def pairwise_descent(a, time_budget=60.0):
    """
    Pairwise coordinate descent: move mass between pairs to reduce peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()

    rng = np.random.default_rng(12345)

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        conv = fast_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to the peak
        # (a*a)[k_star] = sum_j a[j]*a[k_star-j]
        contributions = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                contributions[j] = a[j] * a[kj]

        sorted_idx = np.argsort(contributions)[::-1]
        top_contributors = sorted_idx[:min(20, n)]

        improved = False
        for i in top_contributors:
            if time.monotonic() - start > time_budget:
                break
            # Try moving mass from i to a random other position
            for _ in range(3):
                j = rng.integers(0, n)
                if j == i or a[j] >= a[i]:
                    continue
                for frac in [0.01, 0.05, 0.1]:
                    delta = frac * a[i]
                    if delta <= 0 or delta > a[i]:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    a = project_simplex(a)
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        a = project_simplex(a)
            if improved:
                break

        if not improved and it > 200:
            # Random perturbation
            idx = rng.integers(0, n)
            j = rng.integers(0, n)
            if idx != j:
                delta = 0.02 * a[idx]
                a[idx] -= delta
                a[j] += delta
                a = project_simplex(a)
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

def generate_initialization(kind, n, rng):
    """Generate diverse initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        a = np.maximum(a, 0) + 0.01
        return a / np.sum(a)
    elif kind == 'sine':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2) + 0.01
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'bimodal':
        a = np.ones(n) * 0.3
        a[n//4] = 1.0
        a[3*n//4] = 1.0
        # Smooth
        kernel = np.ones(max(3, n//20)) / max(3, n//20)
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0) + 0.01
        return a / np.sum(a)
    elif kind == 'triangular':
        x = np.arange(n)
        center = n / 2.0
        a = np.maximum(0, 1 - np.abs(x - center) / center) + 0.01
        return a / np.sum(a)
    else:
        return np.ones(n) / n

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 0:
        a_new = a_new / s
    return a_new

            best_score = obj1
            best_seq = a1.copy()
            print(f"  ** Phase 1: obj={best_score:.8f}")
        else:
            print(f"  Phase 1: obj={obj1:.8f} (no improvement)")

    # === Phase 2: Fictitious Play (150s) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 2: Fictitious Play ===")
        budget2 = min(150.0, remaining * 0.15)
        a2 = fictitious_play(best_seq, time_budget=budget2, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  ** Phase 2: obj={best_score:.8f}")
        else:
            print(f"  Phase 2: obj={obj2:.8f}")

    # === Phase 3: LP Polish (200s) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: LP Polish ===")
        budget3 = min(200.0, remaining * 0.2)
        a3 = lp_polish(best_seq, time_budget=budget3, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  ** Phase 3: obj={best_score:.8f}")
        else:
            print(f"  Phase 3: obj={obj3:.8f}")

    # === Phase 4: Pairwise Descent (80s) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: Pairwise Descent ===")
        budget4 = min(80.0, remaining * 0.1)
        a4 = pairwise_descent(best_seq, time_budget=budget4)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  ** Phase 4: obj={best_score:.8f}")
        else:
            print(f"  Phase 4: obj={obj4:.8f}")

    # === Phase 5: Length Sweep (200s) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 5: Length Sweep ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            max(100, n_best - 200),
            max(100, n_best - 100),
            max(100, n_best - 50),
            n_best + 50,
            n_best + 100,
            n_best + 200,
            400, 500, 700, 900, 1000, 1200, 1500,
        ]))
        n_options = [n for n in n_options if 100 <= n <= 3000 and n != n_best]
        
        # Sort by distance from current best
        n_options.sort(key=lambda x: abs(x - n_best))
        
        for n_try in n_options[:6]:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            a_try = resample_sequence(best_seq, n_try)
            # Quick subgradient polish
            budget_sg = min(30.0, remaining * 0.3)
            a_opt = nesterov_subgrad(a_try, time_budget=budget_sg, top_k=20)
            obj_try = compute_obj(a_opt)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts (150s) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: Diverse Restarts ===")
        n_best = len(best_seq)
        kinds = ['cosine', 'gaussian', 'sine', 'bimodal', 'triangular', 'random']
        per_restart = max(10.0, (remaining * 0.15) / len(kinds))
        
        for kind in kinds:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_init = generate_initialization(kind, n_best, rng)
            budget_sr = min(per_restart, remaining * 0.2)
            a_opt = nesterov_subgrad(a_init, time_budget=budget_sr, top_k=20)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** Restart '{kind}': obj={best_score:.8f}")
            else:
                print(f"  Restart '{kind}': obj={obj_opt:.8f}")

    # === Phase 7: Final intensive subgradient + LP (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: Final Polish ===")
        n_final = len(best_seq)
        
        # Long subgradient run
        budget7a = min(remaining * 0.5, 200.0)
        a7 = nesterov_subgrad(best_seq, time_budget=budget7a, verbose=True, top_k=50)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  ** Final subgrad: obj={best_score:.8f}")
        
        # LP polish
        remaining = deadline - time.monotonic()
        if remaining > 30:
            budget7b = min(remaining * 0.5, 100.0)
            a8 = lp_polish(best_seq, time_budget=budget7b, verbose=True)
            obj8 = compute_obj(a8)
            if obj8 < best_score:
                best_score = obj8
                best_seq = a8.copy()
                print(f"  ** Final LP: obj={best_score:.8f}")

    # === Phase 8: Final coordinate descent ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a9 = pairwise_descent(best_seq, time_budget=min(remaining * 0.6, 30.0))
        obj9 = compute_obj(a9)
        if obj9 < best_score:
            best_score = obj9
            best_seq = a9.copy()
            print(f"  ** Final CD: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
