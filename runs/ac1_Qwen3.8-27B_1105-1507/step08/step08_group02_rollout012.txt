"""
Optimized step function search using FFT-accelerated gradient methods,
spectral equalization, LP alternating, and multi-dimension exploration.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv_fft(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    target = 2 * n - 1
    while size < target:
        size <<= 1
    fa = rfft(a, size)
    result = irfft(fa * fa, size)
    return result[:target]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(conv))

def smooth_grad_fft(a, beta):
    """
    Compute gradient of log-sum-exp smooth max of conv(a,a).
    Returns (gradient, smooth_value, weights) where smooth_value = (1/beta)*log(sum(exp(beta*conv)))
    Gradient is w.r.t. a (unnormalized).
    """
    n = len(a)
    conv = compute_conv_fft(a)
    m = conv.max()
    shifted = beta * (conv - m)
    weights = np.exp(shifted)
    weights /= weights.sum()
    smooth_val = m + np.log(weights.sum()) / beta  # = m (since weights sum to 1 now)
    # Actually smooth_val = m + (1/beta)*log(sum(exp(beta*(conv-m)))) = m + (1/beta)*log(sum(weights_original))
    # But we normalized weights, so let me just compute it as:
    # smooth_val = m + (1/beta)*log(sum(exp(shifted)))
    # We already have weights = exp(shifted)/sum(exp(shifted)), so sum(exp(shifted)) = 1/weights.min()... no
    # Let me just return the weights and let the caller compute the smooth value if needed.
    
    # Gradient: grad_i = 2 * sum_j a_j * w_{i+j}
    # This is a cross-correlation. Compute via FFT.
    # grad_i = 2 * sum_{j=0}^{n-1} a_j * w_{i+j} (where w_{i+j} = 0 if i+j >= 2n-1)
    # Let w_rev[j] = w[2n-2-j]. Then grad_i/2 = sum_j a_j * w_rev[2n-2-i-j] = (a * w_rev)[2n-2-i]
    w_rev = weights[::-1]  # w_rev[j] = w[2n-2-j]
    # Linear convolution of a (length n) and w_rev (length 2n-1) has length n + 2n-1 - 1 = 3n-2
    conv_size = 1
    while conv_size < n + (2*n-1) - 1:
        conv_size <<= 1
    fa = rfft(a, conv_size)
    fw = rfft(w_rev, conv_size)
    conv_aw = irfft(fa * fw, conv_size)
    # conv_aw[k] = sum_j a_j * w_rev[k-j] for k=0,...,3n-3
    # We want conv_aw[2n-2-i] for i=0,...,n-1
    # 2n-2-i ranges from 2n-2 down to n-1
    grad = 2.0 * conv_aw[n-1:2*n-1][::-1]
    return grad, weights

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_equalize(a, time_budget=50.0, alpha=0.5, verbose=False):
    """
    Spectral equalization: iteratively flatten the power spectrum to reduce
    peak of the convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Compute FFT
        A = rfft(a, fft_size)
        # Compute spectrum of convolution: B = A^2
        B = A * A
        # We want to flatten B (reduce dynamic range)
        # Method: A_new = A / |A|^alpha, which makes |A_new| = |A|^(1-alpha)
        abs_A = np.abs(A)
        abs_A_safe = np.maximum(abs_A, 1e-30)
        # Apply spectral shaping
        shape = abs_A_safe ** (1.0 - alpha)
        A_new = A * (shape / abs_A_safe)  # This gives |A_new| = |A|^(1-alpha), same phase
        A_new = A * (abs_A_safe ** (-alpha))
        
        # Recover time domain
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            # Decrease alpha as we converge
            alpha = max(0.01, alpha * 0.99)
        else:
            alpha = max(0.01, alpha * 0.95)
            if alpha < 0.05:
                break
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"      Spectral it{it}: obj={best_obj:.8f}, alpha={alpha:.4f}, t={time.monotonic()-start:.1f}s")
    
    return best_a

        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Compute gradient
        grad, weights = smooth_grad_fft(a, beta)

        # a_new = a * exp(-lr * grad) / Z

            break

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    # Try cvxpy
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except Exception:
        pass
    
    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2*n-1), A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    
    return None

    conv = compute_conv_fft(a)
    k_star = int(np.argmax(conv))
    
    max_rounds = 10
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv_fft(a)
        k_star = int(np.argmax(conv))
        
        # Find top contributors to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)
        
        # Non-contributing positions
        contrib_set = set()
        for _, i, j in contributors[:50]:
            contrib_set.add(i)
            contrib_set.add(j)
        non_contrib = [i for i in range(n) if i not in contrib_set]
        
        improved = False
        for contrib_val, i, j in contributors[:20]:
            if time.monotonic() - start > time_budget:
                break
            if contrib_val < 1e-10:
                break
            src = i if a[i] >= a[j] else j
            for target in non_contrib[:10]:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = frac * a[src]
                    if delta < 0 or delta >= a[src]:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
        
        if not improved:
            break
    
    return best_a

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = x  # ramp up
    elif mode == 2:
        a = 1.0 - x  # ramp down
    elif mode == 3:
        a = np.abs(x - 0.5) * 2  # V-shape
    elif mode == 4:
        a = np.sin(np.pi * x)  # arc
    elif mode == 5:
        a = np.cos(np.pi * (x - 0.5))  # cosine bump
    elif mode == 6:
        a = x**2  # quadratic up
    elif mode == 7:
        a = (1-x)**2  # quadratic down
    elif mode == 8:
        a = np.abs(np.sin(2 * np.pi * 3 * x)) + 0.1  # oscillating
    elif mode == 9:
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))  # sigmoid
    elif mode == 10:
        a = np.exp(-((x - 0.3)**2) / 0.02) + 0.5*np.exp(-((x - 0.7)**2) / 0.02)  # two bumps
    else:
        a = rng.uniform(0, 1, n)
    a = np.maximum(a, 0.01)
    return a / a.sum()

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
