"""
Smoothed max-convolution optimizer with adaptive temperature, LP peak reduction,
and multi-phase refinement. Uses log-sum-exp smoothing of the peak to create a
differentiable objective, optimized via projected gradient on the simplex with
backtracking line search and temperature annealing.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar
import time
import cvxpy as cp


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

    Objective: 2*n*(max(conv) + t*log(sum(exp((conv-max(conv))/t))))
    This is a smooth approximation to 2*n*max(conv).
    
    Gradient is computed via FFT-based cross-correlation.
    """
    n = len(a)
    conv = compute_conv(a)  # length 2n-1
    max_conv = np.max(conv)
    conv_shifted = conv - max_conv
    exp_vals = np.exp(conv_shifted / t)
    weights = exp_vals / np.sum(exp_vals)
    
    # Smoothed objective
    obj = 2.0 * n * (max_conv + t * np.log(np.sum(exp_vals)))
    
    # Gradient via cross-correlation
    # grad[m] = (4n/t) * sum_j a[j] * weights[m+j]
    # = (4n/t) * (a * weights_rev)[2n-2-m]
    w_rev = weights[::-1]
    # Full convolution of a (length n) and w_rev (length 2n-1)
    N = 1
    while N < n + (2 * n - 1) - 1:
        N <<= 1
    A = rfft(a, N)
    W = rfft(w_rev, N)
    conv_full = irfft(A * W, N)
    # We need indices 2n-2, 2n-3, ..., n-1 of conv_full
    grad = (4.0 * n / t) * conv_full[n - 1:2 * n - 1][::-1]
    
    return obj, grad

The problem is:
minimize `M` subject to `conv(a,a)[k] <= M` for all k, `sum(a) = 1`, `a >= 0`

At the optimum, by complementary slackness:
- `lambda_k > 0` only if `conv(a,a)[k] = M`
- `nu_i > 0` only if `a_i = 0`

The KKT conditions give:
`1 - sum_k lambda_k * 2*a[k-i] = nu_i` for all i (with appropriate index handling)

`sum_k lambda_k * 2*a[k-i] = 1 - nu_i = constant = c` (for all i)

This means `a * lambda_rev = c * ones` (approximately), where `lambda_rev` is the reversed lambda vector.

1. Guess the active set (which k values have `conv(a,a)[k] = M`).

3. Check if the solution satisfies all constraints.

"""
Multi-phase optimizer: smoothed max-convolution via softmax L-BFGS with
multi-scale temperature annealing, LP peak reduction, and coordinate descent
fine-tuning. Key innovation: differentiable log-sum-exp smoothing of the peak
objective enables efficient gradient-based optimization on the simplex manifold.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar, minimize
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def smooth_obj_grad_a(a, t):
    """
    Smoothed objective and gradient w.r.t. a.
    L(a) = 2*n*(max(conv) + t*log(sum(exp((conv-max(conv))/t))))
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = conv - max_c
    exp_shifted = np.exp(shifted / t)
    sum_exp = np.sum(exp_shifted)
    obj = 2.0 * n * (max_c + t * np.log(sum_exp))
    
    # Weights (softmax of conv/t)
    w = exp_shifted / sum_exp  # length 2n-1
    
    # grad[m] = (4n/t) * sum_j a[j] * w[m+j]
    # = (4n/t) * (a * w_rev)[2n-2-m]
    w_rev = w[::-1]
    N = 1
    while N < n + (2 * n - 1):
        N <<= 1
    A = rfft(a, N)
    W = rfft(w_rev, N)
    conv_full = irfft(A * W, N)
    grad = (4.0 * n / t) * conv_full[n - 1:2 * n - 1][::-1]
    
    return obj, grad


def softmax_from_z(z):
    """Compute softmax of z (numerically stable)."""
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    return exp_z / np.sum(exp_z)


def softmax_obj_grad_z(z, t):
    """
    Objective and gradient w.r.t. z where a = softmax(z).
    Returns (obj, grad_z).
    """
    a = softmax_from_z(z)
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = conv - max_c
    exp_shifted = np.exp(shifted / t)
    sum_exp = np.sum(exp_shifted)
    obj = 2.0 * n * (max_c + t * np.log(sum_exp))
    
    w = exp_shifted / sum_exp
    w_rev = w[::-1]
    N = 1
    while N < n + (2 * n - 1):
        N <<= 1
    A = rfft(a, N)
    W = rfft(w_rev, N)
    conv_full = irfft(A * W, N)
    grad_a = (4.0 * n / t) * conv_full[n - 1:2 * n - 1][::-1]
    
    # Convert to z-gradient: grad_z_i = a_i * (grad_a_i - dot(a, grad_a))
    centering = np.dot(a, grad_a)
    grad_z = a * (grad_a - centering)
    
    return obj, grad_z


def multi_scale_smooth_opt(z_init, time_budget=200.0, verbose=False):
    """
    Multi-scale temperature optimization using L-BFGS in the softmax parameterization.
    Starts with large t (smooth) and decreases to small t (sharp).
    """
    n = len(z_init)
    a = softmax_from_z(z_init)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    
    # Temperature schedule
    t_values = [max_conv, max_conv / 4, max_conv / 16, max_conv / 64, max_conv / 256]
    t_values = [max(t, 1e-6) for t in t_values]
    
    z = z_init.copy()
    best_z = z.copy()
    best_obj = compute_obj(softmax_from_z(z))
    start = time.monotonic()
    time_per_scale = time_budget / len(t_values)
    
    for scale_idx, t in enumerate(t_values):
        scale_time = min(time_per_scale, time_budget - (time.monotonic() - start))
        if scale_time < 5:
            break
        
        # Build objective function for this temperature
        class ObjFunc:
            def __init__(self, t_val):
                self.t = t_val
            
            def __call__(self, z):
                obj, grad = softmax_obj_grad_z(z, self.t)
                return obj, grad
        
        func = ObjFunc(t)
        
        try:
            res = minimize(func, z, method='L-BFGS-B', 
                          options={'maxiter': 200, 'gtol': 1e-6, 'maxfun': 500})
            if res.fun < best_obj:
                best_obj = res.fun
                best_z = res.x.copy()
        except Exception:
            pass
        
        z = best_z.copy()
        if verbose:
            a_check = softmax_from_z(z)
            obj_check = compute_obj(a_check)
            print(f"      Scale {scale_idx} (t={t:.6f}): obj={best_obj:.8f}, actual={obj_check:.8f}, elapsed={time.monotonic()-start:.1f}s")
    
    return best_z

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def lp_peak_reduce(a, time_budget=100.0, verbose=False):
    """
    LP alternating: fix a, solve LP for g that minimizes max conv(a,g)
    subject to sum(g)=1, g>=0. Then blend.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5
    
    while time.monotonic() - start < time_budget:
        a_n = a / float(np.sum(a))
        A = build_conv_matrix(a_n)
        
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        g_val = None
        for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    gv = np.maximum(g.value, 0)
                    gs = float(np.sum(gv))
                    if gs > 1e-10:
                        g_val = gv / gs
                        break
            except Exception:
                continue
        
        if g_val is None:
            break
        
        # Line search for blending
        def blend_obj(t_val):
            a_try = (1.0 - t_val) * a + t_val * g_val
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)
        
        try:
            res = minimize_scalar(blend_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 30, 'xatol': 0.01})
            best_t = res.x
        except Exception:
            best_t = last_t
        
        a_try = (1.0 - best_t) * a + best_t * g_val
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            last_t = best_t
            stall = 0
            if verbose:
                print(f"      LP: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 10:
                break
    
    return best_a


def coordinate_descent(a0, time_budget=100.0, verbose=False):
    """
    Coordinate descent: try changing individual (or pairs of) coefficients
    to reduce the objective.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - start < time_budget and no_improve < max_no_improve:
        # Pick a random index to modify
        i = rng.integers(0, n)
        
        # Try moving mass from i to j
        j = rng.integers(0, n)
        if i == j:
            continue
        
        improved = False
        for frac in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
            delta = frac * a[i]
            if delta < 1e-12:
                continue
            a[i] -= delta
            a[j] += delta
            a = np.maximum(a, 0)
            st = float(np.sum(a))
            if st < 1e-10:
                a[i] += delta
                a[j] -= delta
                continue
            a = a / st
            obj_try = compute_obj(a)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a.copy()
                improved = True
                break
            else:
                # Revert
                a[i] += delta
                a[j] -= delta
                a = a / st
        
        if not improved:
            no_improve += 1
            a = best_a.copy()
        else:
            no_improve = 0
            a = best_a.copy()
        
        if verbose and no_improve % 50 == 0:
            print(f"        CD: obj={best_obj:.8f}, no_improve={no_improve}, elapsed={time.monotonic()-start:.1f}s")
    
    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    if n == 1:
        return np.array([1.0])
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def poly_taper(n, p, q):
    """Polynomial taper: (1 - (2x-1)^p)^q normalized."""
    x = np.linspace(0, 1, n)
    t = 1.0 - np.abs(2.0 * x - 1.0) ** p
    t = t ** q
    t = np.maximum(t, 0)
    s = np.sum(t)
    if s < 1e-10:
        return np.ones(n) / n
    return t / s


def gaussian_taper(n, sigma):
    """Gaussian taper normalized."""
    x = np.linspace(-1, 1, n)
    g = np.exp(-0.5 * (x / sigma) ** 2)
    s = np.sum(g)
    if s < 1e-10:
        return np.ones(n) / n
    return g / s
