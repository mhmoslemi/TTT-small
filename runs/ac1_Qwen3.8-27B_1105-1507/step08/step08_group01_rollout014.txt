"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Fast LP alternating with PDLP/SCS and active-set reduction,
(2) FISTA with smooth max for broad exploration,
(3) Pair-coordinate descent for fine-tuning,
(4) Multi-n exploration, (5) Final polish.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end_k = min(2 * n - 1, j + n)
        A[j:end_k, j] = a[:end_k - j]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp(a, solver_list=None, use_active_set=False, active_indices=None):
    """
    Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s

    A_full = build_conv_matrix(a)

    if use_active_set and active_indices is not None:
        A = A_full[active_indices, :]
    else:
        A = A_full

    if solver_list is None:
        solver_list = ['PDLP', 'MOSEK', 'SCS', 'CLARABEL', 'ECOS']

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        for solver in solver_list:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        g_val = g_val / gs
                        # Verify against ALL constraints
                        conv_check = A_full @ g_val
                        true_M = float(np.max(conv_check))
                        return g_val, true_M
            except Exception:
                continue
    except ImportError:
        pass

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((A.shape[0], n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(A.shape[0])
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                conv_check = A_full @ g_val
                true_M = float(np.max(conv_check))
                return g_val, true_M
    except Exception:
        pass

    return None, None


def convex_line_search(a, g, n_evals=12):
    """Find optimal t in [0,1] for a_new = (1-t)*a + t*g."""
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return t_opt, a_norm
    return t_opt, a_new / s


def get_active_indices(conv, fraction=0.15):
    """Get indices where conv is in the top fraction of peaks."""
    max_conv = np.max(conv)
    threshold = max_conv - fraction * max_conv
    return np.where(conv >= threshold)[0]

def pair_coordinate_descent(a, time_budget=150.0, verbose=True):
    """Pair coordinate descent: try mass transfers between pairs."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Precompute which pairs to try each pass
    rng = np.random.default_rng(42)

    pass_num = 0
    delta_base = 0.01
    while time.monotonic() - start < time_budget:
        pass_num += 1
        improved = False

        # Sample O(n) pairs
        n_pairs = min(n, 400)
        indices = np.arange(n)
        i_idx = rng.choice(indices, size=n_pairs, replace=False)
        j_idx = rng.choice(indices, size=n_pairs, Replace=False)

        for i, j in zip(i_idx, j_idx):
            if i == j:
                continue
            if time.monotonic() - start > time_budget:
                break

            # Try transferring mass from j to i (and vice versa)
            for direction in [1, -1]:
                for frac in [0.01, 0.02, 0.05, 0.005]:
                    amount = direction * frac * a[j]
                    if a[j] + (amount if direction == -1 else 0) < 0:
                        continue
                    a_try = a.copy()
                    a_try[i] += amount
                    a_try[j] -= amount
                    a_try = np.maximum(a_try, 0)
                    st = float(np.sum(a_try))
                    if st < 1e-10:
                        continue
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break

        if verbose and pass_num % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    PC pass {pass_num}: obj={best_obj:.8f}, time={elapsed:.1f}s")

        if not improved and pass_num > 5:
            delta_base *= 0.7
            if delta_base < 1e-6:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

    # === PHASE 1: FISTA for broad improvement (200s) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 1: FISTA Smooth Max (budget={min(200, remaining*0.2):.0f}s) ===")
        budget_p1 = min(200.0, remaining * 0.2)
        a_fista = fista_smooth_max(best_seq, time_budget=budget_p1, T_init=0.001, T_final=0.00003, verbose=True)
        obj_fista = compute_obj(a_fista)
        if obj_fista < best_score:
            best_score = obj_fista
            best_seq = a_fista.copy()
            print(f"  Phase 1 result: obj={best_score:.8f}")
        else:
            print(f"  Phase 1 result: obj={obj_fista:.8f} (no improvement)")
            a_fista = best_seq.copy()

    # === PHASE 2: LP Alternating (300s) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: LP Alternating (budget={min(300, remaining*0.25):.0f}s) ===")
        budget_p2 = min(300.0, remaining * 0.25)
        a_lp = lp_alternating(a_fista, time_budget=budget_p2, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 2 result: obj={best_score:.8f}")
        else:
            print(f"  Phase 2 result: obj={obj_lp:.8f} (no improvement, keeping best)")
            a_lp = best_seq.copy()

    # === PHASE 3: FISTA again with lower T (200s) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 3: FISTA Fine (budget={min(200, remaining*0.18):.0f}s) ===")
        budget_p3 = min(200.0, remaining * 0.18)
        a_fista2 = fista_smooth_max(best_seq, time_budget=budget_p3, T_init=0.0003, T_final=0.00001, verbose=True)
        obj_fista2 = compute_obj(a_fista2)
        if obj_fista2 < best_score:
            best_score = obj_fista2
            best_seq = a_fista2.copy()
            print(f"  Phase 3 result: obj={best_score:.8f}")
        else:
            print(f"  Phase 3 result: obj={obj_fista2:.8f} (no improvement)")

    # === PHASE 4: LP Alternating again on best (200s) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: LP Alternating 2 (budget={min(200, remaining*0.2):.0f}s) ===")
        budget_p4 = min(200.0, remaining * 0.2)
        a_lp2 = lp_alternating(best_seq, time_budget=budget_p4, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase 4 result: obj={best_score:.8f}")
        else:
            print(f"  Phase 4 result: obj={obj_lp2:.8f} (no improvement)")

    # === PHASE 5: Pair Coordinate Descent (150s) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 5: Pair Coord Descent (budget={min(150, remaining*0.2):.0f}s) ===")
        budget_p5 = min(150.0, remaining * 0.2)
        a_pcd = pair_coordinate_descent(best_seq, time_budget=budget_p5, verbose=True)
        obj_pcd = compute_obj(a_pcd)
        if obj_pcd < best_score:
            best_score = obj_pcd
            best_seq = a_pcd.copy()
            print(f"  Phase 5 result: obj={best_score:.8f}")
        else:
            print(f"  Phase 5 result: obj={obj_pcd:.8f} (no improvement)")

    # === PHASE 6: Multi-n exploration (200s) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 6: Multi-n Exploration ===")
        n_base = len(best_seq)
        n_options = sorted(set([
            n_base - 20, n_base - 10, n_base + 10, n_base + 20,
            n_base - 40, n_base + 40, n_base + 60,
            750, 800, 850, 900
        ]))
        n_options = [n for n in n_options if 500 <= n <= 1200]

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            if n_try == len(best_seq):
                continue

            a_try = resample_sequence(best_seq, n_try)
            # Quick FISTA
            b_this = min(60.0, remaining * 0.2)
            a_opt = fista_smooth_max(a_try, time_budget=b_this, T_init=0.0005, T_final=0.00002, verbose=False)
            obj_try = compute_obj(a_opt)

            # Quick LP
            remaining = deadline - time.monotonic()
            if remaining > 30 and obj_try < best_score + 0.01:
                b_lp = min(30.0, remaining * 0.3)
                a_opt = lp_alternating(a_opt, time_budget=b_lp, verbose=False)
                obj_try = compute_obj(a_opt)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === PHASE 7: Final intensive LP (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: Final LP Polish (budget={min(remaining*0.6):.0f}s) ===")
        budget_p7 = min(remaining * 0.6, 100.0)
        a_final = lp_alternating(best_seq, time_budget=budget_p7, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f}")

    # === PHASE 8: Final FISTA polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        budget_p8 = min(remaining * 0.7, 50.0)
        a_final2 = fista_smooth_max(best_seq, time_budget=budget_p8, T_init=0.0001, T_final=0.000005, verbose=False)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final FISTA: obj={best_score:.8f}")

    # === Final validation ===
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if np.isfinite(val) and val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
