```python
import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_gradient(a, active_lags):
    """
    Compute subgradient of max_k conv(a,a)_k with respect to a.
    
    active_lags: array of lag indices where conv is "active" (near max).
    
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    conv_len = 2 * n - 1
    
    # active_mask: length 2n-1, 1 where active
    active_mask = np.zeros(conv_len)
    active_mask[active_lags] = 1.0
    
    # grad[i] = sum_k active[k] * (2 - [k==2*i]) * a[k-i]
    #         = 2 * sum_k active[k] * a[k-i] - [2i in active] * a[i]
    # 
    # The first term is a cross-correlation: h[i] = sum_k active[k] * a[k-i]
    # = sum_m active[m+i] * a[m] (substituting m = k-i)
    
    # Compute h = cross-correlation of active_mask and a
    N = 1
    while N < conv_len + n:
        N <<= 1
    
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(active_mask, N)
    # Cross-correlation: ifft(W * conj(A))
    h = np.fft.irfft(W * np.conj(A), N)[:n]
    
    grad = 2.0 * h
    
    # Subtract diagonal correction
    for i in range(min(n, conv_len)):
        if active_mask[2 * i]:
            grad[i] -= a[i]
    
    return grad
```

```python
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def fast_mirror_descent(a, time_budget=300.0, initial_eta=1.0, power=10, verbose=True):
    """
    Fast mirror descent for minimizing max(conv(a,a)) subject to a>=0, sum(a)=1.
    Uses soft active set weighting.
    """
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    a = a / np.sum(a)
    n = len(a)
    
    # Precompute FFT size
    N = 1
    while N < 2 * n:
        N <<= 1
    
    conv_len = 2 * n - 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    eta = initial_eta
    start = time.monotonic()
    stagnant = 0
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Compute convolution
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        
        # Soft active set
        ratio = conv / max_c
        weights = ratio ** power
        w_sum = float(np.sum(weights))
        if w_sum < 1e-15:
            break
        weights = weights / w_sum
        
        # Compute gradient via FFT
        W = rfft(weights, N)
        A = rfft(a, N)
        h = irfft(W * np.conj(A), N)[:n]
        
        grad = 2.0 * h
        # Diagonal correction
        diag = weights[0:conv_len:2]
        grad -= diag * a
        
        # Mirror descent update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Check improvement
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
            # Slightly increase eta
            eta = min(eta * 1.01, 5.0)
        else:
            stagnant += 1
            if stagnant > 20:
                # Reduce eta
                eta *= 0.5
                stagnant = 0
                if eta < 0.01:
                    # Try a perturbation
                    rng = np.random.default_rng(it)
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 1e-15)
                    a = a / np.sum(a)
                    eta = initial_eta
                    stagnant = 0
            # Keep old a (don't accept the step)
        
        it += 1
        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD iter {it}: obj={2.0*n*float(np.max(compute_conv(best_a))):.8f}, eta={eta:.3f}, t={elapsed:.0f}s")
    
    return best_a, best_obj
```

```python
"""
Fast mirror descent with soft active set + periodic LP refinement + multi-n exploration.
Key innovations:
1. Mirror descent (multiplicative weights) for O(n log n) per iteration via FFT
2. Soft active set (power weighting) for smooth gradient computation
3. Adaptive step size with momentum
4. Periodic LP steps via scipy linprog for escape from local minima
5. Multi-initialization strategy with structured windows
6. Basin hopping with adaptive perturbation
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_gradient(a, weights, N):
    n = len(a)
    conv_len = 2 * n - 1
    W = rfft(weights, N)
    A = rfft(a, N)
    h = irfft(W * np.conj(A), N)[:n]
    grad = 2.0 * h
    diag = weights[0:conv_len:2]
    grad -= diag * a
    return grad


def mirror_descent(a, time_budget=300.0, eta=0.5, power=8, verbose=False):
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    a = a / np.sum(a)
    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1
    conv_len = 2 * n - 1

    best_a = a.copy()
    best_obj = compute_obj(a)
    current_eta = eta
    start = time.monotonic()
    stagnant = 0
    momentum_grad = np.zeros(n)
    beta_mom = 0.3
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        obj = 2.0 * n * max_c

        ratio = conv / max_c
        weights = ratio ** power
        w_sum = float(np.sum(weights))
        if w_sum < 1e-15:
            break
        weights = weights / w_sum

        grad = compute_gradient(a, weights, N)
        momentum_grad = beta_mom * momentum_grad + (1.0 - beta_mom) * grad

        a_new = a * np.exp(-current_eta * momentum_grad)
        a_new = np.maximum(a_new, 1e-30)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        conv_new = compute_conv(a_new)
        max_new = float(np.max(conv_new))
        obj_new = 2.0 * n * max_new

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
            current_eta = min(current_eta * 1.005, 3.0)
        else:
            stagnant += 1
            if stagnant % 10 == 0:
                current_eta *= 0.7
                if stagnant > 200:
                    rng = np.random.default_rng(it % (2**31))
                    a = best_a * np.exp(rng.normal(0, 0.03 + 0.01 * (stagnant // 200), n))
                    a = np.maximum(a, 1e-15)
                    a = a / np.sum(a)
                    momentum_grad = np.zeros(n)
                    stagnant = 0
                    current_eta = eta

        it += 1
        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD iter {it}: obj={best_obj:.8f}, eta={current_eta:.3f}, t={elapsed:.0f}s")

    return best_a, best_obj


def solve_lp_fast(a):
    n = len(a)
    conv_len = 2 * n - 1
    A = np.zeros((conv_len, n))
    for j in range(n):
        r = min(n, conv_len - j)
        A[j:j + r, j] = a[:r]

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((conv_len, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(conv_len)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0 and res.x is not None:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val
    except:
        pass
    return None


def lp_alternating(a, time_budget=100.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = a.copy()
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp = solve_lp_fast(a)
        if g_lp is None:
            break

        momentum = 0.5 * momentum + 0.5 * g_lp
        momentum = np.maximum(momentum, 1e-30)
        momentum = momentum / np.sum(momentum)

        best_cand = a.copy()
        best_cand_obj = np.inf
        for mix in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - mix) * a + mix * momentum
            a_try = np.maximum(a_try, 1e-30)
            a_try = a_try / np.sum(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_cand_obj:
                best_cand_obj = obj_try
                best_cand = a_try

        if best_cand_obj < best_obj - 1e-12:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            a = best_a
            stall = 0
            if verbose:
                print(f"    LP-Alt: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 10:
                rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 1e-15)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                momentum = a.copy()
                stall = 0

    return best_a, best_obj


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-15)


def coordinate_polish(a, time_budget=30.0):
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    improved = True
    passes = 0
    while time.monotonic() - start < time_budget and improved:
        improved = False
        passes += 1
        for _ in range(50):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for frac in [0.02, 0.05, 0.1, 0.2]:
                delta = frac * min(a[i], a[j])
                if delta <= 0:
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0 or a_try[j] < 0:
                    continue
                a_try = a_try / np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
    return best_a, best_obj
```
