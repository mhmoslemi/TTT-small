import numpy as np
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Compute linear autocorrelation a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(y):
    """Project y onto {x: sum(x)=1, x>=0} via Euclidean projection."""
    n = len(y)
    if n == 0:
        return y.copy()
    if np.sum(y) <= 0:
        return np.ones(n) / n
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u)
    ar = np.arange(1, n + 1)
    cond = 1 + u * ar - cssv > 0
    rho = np.max(np.where(cond)[0]) if np.any(cond) else 0
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(y - theta, 0.0)


def soft_subgrad(a, kstar):
    """Subgradient of (a*a)_{kstar} w.r.t. a. Returns 2*a[kstar-j] with bounds."""
    n = len(a)
    g = np.zeros(n)
    j_lo = max(0, kstar - n + 1)
    j_hi = min(n - 1, kstar)
    for j in range(j_lo, j_hi + 1):
        g[j] = 2.0 * a[kstar - j]
    return g


def soft_subgrad_fast(a, kstar):
    """Fast subgradient using slicing."""
    n = len(a)
    g = np.zeros(n)
    j_lo = max(0, kstar - n + 1)
    j_hi = min(n - 1, kstar)
    idx = np.arange(j_lo, j_hi + 1)
    src_idx = kstar - idx
    valid = (src_idx >= 0) & (src_idx < n)
    g[idx[valid]] = 2.0 * a[src_idx[valid]]
    return g


def logsumexp_smooth_grad(a, t):
    """Gradient of (1/t)*log(sum_k exp(t*(a*a)_k))."""
    n = len(a)
    conv = compute_conv(a)
    tconv = t * conv
    tconv_shifted = tconv - np.max(tconv)
    weights = np.exp(tconv_shifted)
    weights /= np.sum(weights)
    # Gradient is weighted sum of subgradients
    # grad[j] = 2 * sum_k weights[k] * a[k-j] (with bounds)
    # This is 2 * correlate(weights, a_reversed)[offset]
    ar = a[::-1]
    # conv of weights and ar gives sum_k weights[k] * ar[k-j+n-1]
    # ar[m] = a[n-1-m], so ar[k-j+n-1] = a[n-1-k+j-n+1] = a[j-k]
    # We want sum_k weights[k] * a[k-j]
    # Let's use FFT: cross-correlation
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fw = rfft(weights, size)
    fa_r = rfft(ar, size)
    cross = irfft(fw * fa_r, size)
    # cross[m] = sum_k weights[k] * ar[m-k] = sum_k weights[k] * a[n-1-m+k]
    # We want sum_k weights[k] * a[k-j]
    # a[k-j] = ar[n-1-(k-j)] = ar[j-k+n-1]
    # So sum_k weights[k] * a[k-j] = sum_k weights[k] * ar[j-k+n-1]
    # cross[m] = sum_k weights[k] * ar[m-k]
    # Set m-k = j-k+n-1, so m = j+n-1
    # sum_k weights[k] * ar[j-k+n-1] is not directly cross
    # Let's just do it directly for clarity
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for k in range(max(0, j), min(n, 2 * n - 1 - j + 1)):
            kj = k - j
            if 0 <= kj < n and k < len(weights):
                s += weights[k] * a[kj]
        grad[j] = 2.0 * s
    return grad


def logsumexp_smooth_grad_fft(a, t):
    """Efficient gradient of smooth max using FFT."""
    n = len(a)
    conv = compute_conv(a)
    tconv = t * conv
    tconv_shifted = tconv - np.max(tconv)
    weights = np.exp(tconv_shifted)
    weights /= np.sum(weights)
    # grad[j] = 2 * sum_k weights[k] * a[k-j] for 0<=k-j<n
    # Define w[k] = weights[k], want (w * a_shifted) where a_shifted depends on j
    # sum_k w[k]*a[k-j] = cross-correlation of w and a, evaluated at position j
    # cross[j] = sum_k w[k] * a[k-j] (with a[m]=0 for m<0 or m>=n)
    # This is the 'full' cross-correlation, which can be computed via FFT
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    # cross[j] = sum_k w[k]*conj(a[k-j]) for real = sum_k w[k]*a[(k-j) mod ...]
    # Actually for linear cross-correlation:
    # cross[j] = sum_{k} w[k] * a[k-j] where a[m]=0 outside [0,n-1]
    # = (w cross-correlate a)[j]
    # Using FFT: cross-correlation = IFFT(FFT(w) * conj(FFT(a)))
    fw = rfft(w if False else weights, size)
    fa = rfft(a, size)
    cross = irfft(fw * np.conj(fa), size)
    # cross[m] = sum_k w[k] * a[m+k] ... need to be careful
    # Actually: IFFT(FFT(w) * conj(FFT(a)))[m] = sum_n w[n] * a[n-m] (circular)
    # For linear: cross[m] = sum_k weights[k] * a[k-m]
    # We want grad[j] = 2 * sum_k weights[k] * a[k-j] = 2 * cross[j]
    # But we need to handle the linear (not circular) convolution
    # For our case, since a is zero outside [0,n-1] and weights is zero outside [0,2n-2],
    # the linear cross-correlation is:
    # cross_lin[j] = sum_{k=0}^{2n-2} weights[k] * a[k-j] (a[m]=0 if m<0 or m>=n)
    # This equals sum_{k=j}^{min(2n-2, j+n-1)} weights[k] * a[k-j]
    # The circular version wraps around, so we need to be careful
    # For the non-wrapping part: cross[j] for j in [0, n-1] should give the right answer
    # if the sizes are correct (size >= 2n-1 ensures no wrap for relevant indices)
    grad = 2.0 * cross[:n]
    return grad, conv, weights


def fista_smooth(a, n_iters=5000, t_start=0.5, t_end=50.0, n_epochs=8, verbose=False):
    """
    FISTA on log-sum-exp smoothed max autocorrelation, with geometric temperature schedule.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    temps = np.geomspace(t_start, t_end, n_epochs)
    iters_per_epoch = max(100, n_iters // n_epochs)

    for ei, t in enumerate(temps):
        a_curr = a.copy()
        y = a_curr.copy()
        t_mom = 1.0
        x_prev = a_curr.copy()

        for it in range(iters_per_epoch):
            # Compute gradient
            try:
                grad, conv, weights = logsumexp_smooth_grad_fft(y, t)
            except:
                break

            # Lipschitz constant estimate: L ~ t * 4 * max(weights) * ||a||_2^2 + 2
            # Conservative: use backtracking
            f_y = (1.0 / t) * np.log(np.sum(np.exp(t * conv - np.max(t * conv))))
            L = t * 4.0 * 1.0 + 2.0  # conservative upper bound

            # FISTA update
            x_new = project_simplex(y - grad / L)
            ntry = x_new.copy()
            f_ntry = (1.0 / t) * np.log(np.sum(np.exp(t * compute_conv(ntry) - np.max(t * compute_conv(ntry)))))

            # Backtracking if needed
            bt = 0
            while f_ntry > f_y + np.dot(grad, ntry - y) + 0.5 * L * np.sum((ntry - y) ** 2) and bt < 10:
                L *= 2.0
                x_new = project_simplex(y - grad / L)
                ntry = x_new.copy()
                f_ntry = (1.0 / t) * np.log(np.sum(np.exp(t * compute_conv(ntry) - np.max(t * compute_conv(ntry)))))
                bt += 1

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - x_prev)
            y = np.maximum(y, 0.0)
            sy = float(np.sum(y))
            if sy > 1e-10:
                y = y / sy
            x_prev = x_new.copy()
            t_mom = t_new

        a = x_new.copy()
        a = np.maximum(a, 0.0)
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        if verbose:
            print(f"    FISTA epoch {ei+1}/{n_epochs}, t={t:.2f}, obj={obj:.8f}")

    return best_a


def subgrad_averaging(a, n_iters=20000, verbose=False):
    """
    Subgradient method with Nesterov averaging for the non-smooth max autocorrelation.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    a_curr = a.copy()
    y = a_curr.copy()
    t_mom = 1.0
    x_prev = a_curr.copy()
    a_sum = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(1, n_iters + 1):
        conv = compute_conv(y)
        kstar = int(np.argmax(conv))
        g = soft_subgrad_fast(y, kstar)

        # Lipschitz bound for subgradient: ||g||_2 <= 2*||y||_2 <= 2
        eta = 1.0 / (np.sqrt(2.0 * it) + 1.0)

        x_new = project_simplex(y - eta * g)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - x_prev)
        y = np.maximum(y, 0.0)
        sy = float(np.sum(y))
        if sy > 1e-10:
            y = y / sy
        else:
            y = best_a.copy()
        x_prev = x_new.copy()
        t_mom = t_new

        a_sum += x_new

        # Periodically check averaged iterate
        if it % 500 == 0:
            a_avg = a_sum / it
            a_avg = np.maximum(a_avg, 0.0)
            sa = float(np.sum(a_avg))
            if sa > 1e-10:
                a_avg = a_avg / sa
                obj_avg = compute_obj(a_avg)
                if obj_avg < best_obj:
                    best_obj = obj_avg
                    best_a = a_avg.copy()
                    if verbose:
                        print(f"    Subgrad avg at it={it}: obj={best_obj:.8f}")

    # Final averaged iterate
    a_avg = a_sum / n_iters
    a_avg = np.maximum(a_avg, 0.0)
    sa = float(np.sum(a_avg))
    if sa > 1e-10:
        a_avg = a_avg / sa
        obj_avg = compute_obj(a_avg)
        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = a_avg.copy()

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_cross(a):
    """
    Solve LP: min M s.t. (a*g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ z <= M, cp.sum(z) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    return z_val, float(M.value)
        except:
            continue

    # Fallback
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n])
    except:
        pass

    return None, None


def lp_alternate_improved(a, time_budget=200.0, verbose=False):
    """
    LP alternating with:
    - Backtracking line search for mixing ratio
    - Momentum with adaptive coefficient
    - 2-cycle detection
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = a.copy()
    prev_a = None
    prev_prev_a = None
    stall = 0
    mom_beta = 0.5

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_cross(a)
        if g_lp is None:
            break

        # 2-cycle detection
        if prev_prev_a is not None:
            if np.max(np.abs(a - prev_prev_a)) < 5e-8:
                a_break = 0.5 * (a + prev_a)
                a_break = np.maximum(a_break, 0)
                a_break = a_break / np.sum(a_break)
                obj_break = compute_obj(a_break)
                if obj_break < best_obj:
                    best_obj = obj_break
                    best_a = a_break.copy()
                a = a_break
                prev_a = None
                prev_prev_a = None
                momentum = a.copy()
                stall = 0
                continue

        # Update momentum
        momentum = (1 - mom_beta) * momentum + mom_beta * g_lp
        momentum = np.maximum(momentum, 0)
        ms = float(np.sum(momentum))
        if ms > 1e-10:
            momentum = momentum / ms

        # Backtracking line search: try multiple mixing ratios
        targets = [g_lp, momentum]
        best_cand = a.copy()
        best_cand_obj = np.inf

        for target in targets:
            for mix in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1]:
                a_try = (1.0 - mix) * a + mix * target
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_cand_obj:
                    best_cand_obj = obj_try
                    best_cand = a_try.copy()

        if best_cand_obj < best_obj - 1e-14:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            prev_prev_a = prev_a
            prev_a = a.copy()
            a = best_a
            stall = 0
            # Slightly increase momentum when making progress
            mom_beta = min(mom_beta * 1.01, 0.8)
            if verbose and int((time.monotonic() - start) * 2) % 20 == 0:
                print(f"    LP-alt: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 20:
                # Perturbation to escape
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                # Dual-guided: find active lags and perturb
                conv_a = compute_conv(a)
                k_max = int(np.argmax(conv_a))
                threshold = conv_a[k_max] * 0.95
                active_lags = np.where(conv_a >= threshold)[0]

                pert = np.zeros(n)
                for k in active_lags[:10]:
                    g_k = soft_subgrad_fast(a, int(k))
                    pert += g_k
                pert -= np.mean(pert)
                pert += rng.normal(0, 0.01, n)

                a = best_a * np.exp(0.1 * pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                prev_prev_a = None
                momentum = a.copy()
                stall = 0
                mom_beta = 0.5
                if verbose:
                    print(f"    [Perturbed] obj={best_obj:.8f}")

    return best_a


def direct_slsqp(a, time_budget=60.0, verbose=False):
    """
    Direct optimization using SLSQP for moderate n.
    Minimize max_k (a*a)_k subject to sum(a)=1, a>=0.
    """
    from scipy.optimize import minimize
    n = len(a)
    if n > 400:
        return a  # Too large for SLSQP

    a = np.maximum(a, 0.0).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def objective(x):
        x = np.maximum(x, 0)
        s = np.sum(x)
        if s < 1e-10:
            return 1e10
        xn = x / s
        conv = compute_conv(xn)
        return np.max(conv)

    def gradient(x):
        x = np.maximum(x, 0)
        s = np.sum(x)
        if s < 1e-10:
            return np.zeros(n)
        xn = x / s
        conv = compute_conv(xn)
        kstar = int(np.argmax(conv))
        g = soft_subgrad_fast(xn, kstar)
        return g / s

    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0, 'jac': lambda x: np.ones(n)}
    ]
    bounds = [(0, None)] * n

    try:
        res = minimize(objective, a, method='SLSQP', jac=gradient,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success or (hasattr(res, 'status') and res.status == 0):
            x_opt = np.maximum(res.x, 0)
            s = np.sum(x_opt)
            if s > 1e-10:
                x_opt = x_opt / s
                obj = compute_obj(x_opt)
                if obj < best_obj:
                    best_obj = obj
                    best_a = x_opt.copy()
                    if verbose:
                        print(f"    SLSQP: obj={best_obj:.8f}")
    except Exception as e:
        if verbose:
            print(f"    SLSQP error: {e}")

    elapsed = time.monotonic() - start
    if verbose:
        print(f"    SLSQP done in {elapsed:.1f}s, best={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent(a, time_budget=30.0):
    """Coordinate descent: optimize one element at a time with line search."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    order = rng.permutation(n)
    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        i = order[it % n]
        ai = a[i]
        if ai < 1e-12:
            continue
        # Try several new values for a[i]
        # The constraint is sum(a)=1, so if we change a[i] by delta, we need to adjust others
        # Simpler: try scaling a[i] by various factors, renormalize
        for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]:
            a_try = a.copy()
            a_try[i] = ai * factor
            a_try = np.maximum(a_try, 0)
            s = np.sum(a_try)
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                ai = a[i]
                break

    return best_a


def pairwise_exchange(a, time_budget=40.0):
    """Pairwise exchange with line search."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        ai, aj = a[i], a[j]
        if ai < 1e-10 and aj < 1e-10:
            continue
        for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
            delta = frac * min(ai, aj)
            if delta <= 0:
                continue
            a_try = a.copy()
            a_try[i] = ai - delta
            a_try[j] = aj + delta
            a_try = np.maximum(a_try, 0)
            s = np.sum(a_try)
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break

    return best_a


def try_all_perturbations(a, n_pert=20, verbose=False):
    """Try multiple perturbation strategies and return the best."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(12345)

    for p in range(n_pert):
        if p % 4 == 0:
            # Small Gaussian
            a_p = a * np.exp(rng.normal(0, 0.03, n))
        elif p % 4 == 1:
            # Block perturbation
            a_p = a.copy()
            n_blocks = max(3, n // 200)
            block_sz = n // n_blocks
            b = rng.integers(0, n_blocks)
            si = b * block_sz
            a_p[si:si + block_sz] *= rng.uniform(0.5, 1.5)
        elif p % 4 == 2:
            # Sparse perturbation
            a_p = a.copy()
            n_sp = max(5, n // 20)
            idx = rng.choice(n, n_sp, replace=False)
            a_p[idx] *= rng.uniform(0.3, 2.0, n_sp)
        else:
            # Shift perturbation
            a_p = a.copy()
            shift = rng.integers(1, max(2, n // 50))
            a_p[shift:] = a[:-shift] * 0.95
            a_p[:shift] = a[-shift:] * 0.95

        a_p = np.maximum(a_p, 0)
        s = np.sum(a_p)
        if s > 1e-10:
            a_p = a_p / s
            obj = compute_obj(a_p)
            if obj < best_obj:
                best_obj = obj
                best_a = a_p.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining FISTA (smoothed), LP alternating (with backtracking),
    subgradient averaging, multi-n search, and final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        # Generate initial candidates
        for n_init in [400, 600, 800, 1000]:
            t = np.linspace(0, 1, n_init)
            windows = [
                0.5 + 0.5 * np.cos(2 * np.pi * t),
                np.sin(np.pi * t) ** 2,
                1.0,
            ]
            for w in windows:
                a_init = np.maximum(w, 0.01)
                a_init = a_init / np.sum(a_init)
                obj_init = compute_obj(a_init)
                if obj_init < best_score:
                    best_score = obj_init
                    best_seq = a_init.copy()
        if best_seq is None:
            best_seq = np.ones(600) / 600
            best_score = compute_obj(best_seq)
        print(f"Initial (n={len(best_seq)}): obj={best_score:.8f}")

    # === Phase 1: FISTA on smoothed objective (fast, ~30-60s) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_fista = len(best_seq)
        alloc = min(80.0, remaining * 0.15)
        n_iters = max(2000, int(alloc * 5000))
        print(f"\n=== Phase 1: FISTA smoothed (n={n_fista}) ===")
        a_fista = fista_smooth(best_seq, n_iters=n_iters, verbose=True)
        obj_f = compute_obj(a_fista)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_fista.copy()
            print(f"  FISTA improved: {best_score:.8f} **")
        else:
            print(f"  FISTA: {obj_f:.8f} (no improvement)")

    # === Phase 2: LP Alternating with backtracking (200s) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(250.0, remaining * 0.3)
        print(f"\n=== Phase 2: LP Alternating (n={len(best_seq)}, {alloc:.0f}s) ===")
        a_lp = lp_alternate_improved(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP-alt improved: {best_score:.8f} **")
        else:
            print(f"  LP-alt: {obj_lp:.8f}")

    # === Phase 3: Subgradient averaging (60s) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n_sg = len(best_seq)
        alloc = min(60.0, remaining * 0.1)
        n_iters_sg = max(5000, int(alloc * 10000))
        print(f"\n=== Phase 3: Subgradient averaging (n={n_sg}) ===")
        a_sg = subgrad_averaging(best_seq, n_iters=n_iters_sg, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgrad improved: {best_score:.8f} **")
        else:
            print(f"  Subgrad: {obj_sg:.8f}")

    # === Phase 4: Multi-n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 4: Multi-n exploration ===")
        n_parent = len(best_seq)
        n_candidates = sorted(set([
            max(200, n_parent - 200),
            max(200, n_parent - 100),
            n_parent,
            n_parent + 100,
            n_parent + 200,
            400, 500, 600, 700, 800, 900, 1000, 1200, 1500,
        ]))
        n_candidates = [n for n in n_candidates if 200 <= n <= 2000]
        # Sort by proximity to parent
        n_candidates.sort(key=lambda x: abs(x - n_parent))

        for n_try in n_candidates:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            if n_try == len(best_seq):
                continue

            a_try = resample_sequence(best_seq, n_try)
            # Quick FISTA + LP for this n
            alloc_f = min(30.0, remaining * 0.1)
            alloc_l = min(40.0, remaining * 0.1)
            a_opt = fista_smooth(a_try, n_iters=3000, t_start=1.0, t_end=30.0, n_epochs=5)
            obj_f = compute_obj(a_opt)
            a_opt = lp_alternate_improved(a_opt, time_budget=alloc_l)
            obj_try = compute_obj(a_opt)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: {best_score:.8f} **")
            else:
                print(f"  n={n_try}: {obj_try:.8f}")

    # === Phase 5: Direct SLSQP for moderate n ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and len(best_seq) <= 400:
        print(f"\n=== Phase 5: SLSQP (n={len(best_seq)}) ===")
        alloc = min(40.0, remaining * 0.15)
        a_slsqp = direct_slsqp(best_seq, time_budget=alloc, verbose=True)
        obj_s = compute_obj(a_slsqp)
        if obj_s < best_score:
            best_score = obj_s
            best_seq = a_slsqp.copy()
            print(f"  SLSQP improved: {best_score:.8f} **")
        else:
            print(f"  SLSQP: {obj_s:.8f}")

    # === Phase 6: Final LP on best (150s) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(150.0, remaining * 0.3)
        print(f"\n=== Phase 6: Final LP (n={len(best_seq)}, {alloc:.0f}s) ===")
        a_final = lp_alternate_improved(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: {best_score:.8f} **")
        else:
            print(f"  Final LP: {obj_final:.8f}")

    # === Phase 7: Random restarts with LP ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 7: Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(5, int(remaining // 50))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            if r == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.05, n_best))
            elif r == 1:
                # Anti-peak: reduce at lags where autocorrelation is high
                conv = compute_conv(best_seq)
                k_max = int(np.argmax(conv))
                g = soft_subgrad_fast(best_seq, k_max)
                a_r = best_seq - 0.05 * g / np.max(g)
            elif r == 2:
                # Random block swap
                a_r = best_seq.copy()
                n_blocks = max(3, n_best // 100)
                bs = n_best // n_blocks
                i, j = rng.choice(n_blocks, 2, replace=False)
                tmp = a_r[i*bs:(i+1)*bs].copy()
                a_r[i*bs:(i+1)*bs] = a_r[j*bs:(j+1)*bs]
                a_r[j*bs:(j+1)*bs] = tmp
            else:
                a_r = best_seq * np.exp(rng.normal(0, 0.08, n_best))

            a_r = np.maximum(a_r, 0)
            sr = np.sum(a_r)
            if sr > 1e-10:
                a_r = a_r / sr

            alloc = min(remaining * 0.35, 40.0)
            a_opt = lp_alternate_improved(a_r, time_budget=alloc)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: {best_score:.8f} **")
            else:
                print(f"  Restart {r}: {obj_r:.8f}")

    # === Phase 8: Coordinate descent + pairwise polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 8: Polish ===")
        alloc_cd = min(20.0, remaining * 0.3)
        alloc_pw = min(20.0, remaining * 0.3)
        a_cd = coordinate_descent(best_seq, time_budget=alloc_cd)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coord descent: {best_score:.8f} **")

        a_pw = pairwise_exchange(best_seq, time_budget=alloc_pw)
        obj_pw = compute_obj(a_pw)
        if obj_pw < best_score:
            best_score = obj_pw
            best_seq = a_pw.copy()
            print(f"  Pairwise: {best_score:.8f} **")

    # Final LP polish if time remains
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.6, 30.0)
        a_fin = lp_alternate_improved(best_seq, time_budget=alloc)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final polish LP: {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluator
    try:
        val = evaluate_sequence(result)
        if np.isfinite(val) and val < best_score:
            best_score = val
    except:
        pass

    if not np.isfinite(best_score):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
