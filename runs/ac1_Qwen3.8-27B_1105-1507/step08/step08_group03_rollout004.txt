    def project_simplex(v):
        """Project v onto the probability simplex."""
        n = len(v)

        cssv = np.cumsum(u)
        idx = np.arange(1, n + 1)
        cond = u * idx > (cssv - 1.0)

        rho = rho[-1]

        return np.maximum(v - theta, 0.0)
