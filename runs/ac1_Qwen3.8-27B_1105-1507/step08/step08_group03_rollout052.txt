"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: faster LP with vectorized matrix construction, LP-dual-informed
subgradients, adaptive time allocation, structured multi-n exploration, and
aggressive local search with diverse restarts.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize as sp_minimize

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except Exception:
    HAVE_CVXPY = False


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_fast(a):
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    valid = (diff >= 0) & (diff < n)
    A[valid] = a[diff[valid]]
    return A


def solve_lp_minpeak(a, A=None):
    n = len(a)
    a = a / float(np.sum(a))
    if A is None:
        A = build_conv_matrix_fast(a)

    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        dual_val = None
                        try:
                            dual_val = np.array(prob.constraints[0].dual_value).flatten()
                        except Exception:
                            pass
                        return g_val / s, float(M.value), dual_val
            except Exception:
                continue

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n]), None
    return None, None, None


def grad_conv_at_peak(a, k):
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return 2.0 * g


def lpdual_subgradient(a, dual_weights):
    n = len(a)
    if dual_weights is None:
        return None
    dual_weights = np.maximum(dual_weights, 0)
    s = dual_weights.sum()
    if s < 1e-12:
        return None
    dual_weights = dual_weights / s
    g = np.zeros(n)
    for k in range(len(dual_weights)):
        if dual_weights[k] > 1e-10:
            g += dual_weights[k] * grad_conv_at_peak(a, k)
    return g


def nesterov_subgradient(a, time_budget=60.0, K=5, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 60
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        spread = (peak_vals[0] - peak_vals[-1]) / max_val
        temp = 5.0 / (1.0 + spread * 8.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / max_val)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                g += w * grad_conv_at_peak(y, int(k))

        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                t_k = 1.0
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.4)

        a_prev = a_new.copy()
        y = y_new.copy()
        t_k = t_new
        it += 1

    return best_a

    best_t = 0.3

        ts = sorted(set([best_t, best_t * 0.5, best_t * 1.5, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9, 1.0]))

        it += 1
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            best_t = best_t_val
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, M={M_lp:.6f}, el={time.monotonic()-start:.1f}s")

def mirror_descent(a, time_budget=80.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    min_lr = 1e-5
    stall = 0
    momentum = 0.0
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()
        if max_val < 1e-15:
            break

        temps = [3.0, 8.0, 20.0]
        all_w = np.zeros(len(conv))
        for temp in temps:
            w = np.exp((conv - max_val) / (temp * max_val + 1e-15))
            w /= w.sum()
            all_w += w
        all_w /= len(temps)

        threshold = np.percentile(all_w, 90)
        active = all_w > threshold
        if not active.any():
            active[np.argmax(conv)] = True
        active_w = all_w[active]
        active_w /= active_w.sum()
        active_idx = np.where(active)[0]

        g = np.zeros(n)
        for w, k in zip(active_w, active_idx):
            if w > 1e-10:
                g += w * grad_conv_at_peak(a, int(k))

        v = momentum * v + g
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = max(min_lr, lr * 0.998)
            momentum = min(0.7, momentum + 0.001)
        else:
            stall += 1
            if stall > 80:
                rng = np.random.default_rng(it * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.2, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                momentum = 0.0
                lr = max(min_lr, lr * 0.4)
                stall = 0
            else:
                lr = max(min_lr, lr * 0.97)
        it += 1

    return best_a


def smooth_optimize(a, time_budget=60.0, T=40.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)

    def obj_grad(x):
        x = np.maximum(x, 1e-15)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x = x / s
        conv = compute_conv(x)
        max_c = conv.max()
        z = T * (conv - max_c)
        z -= z.max()
        e = np.exp(z)
        sum_e = e.sum()
        obj_val = 2.0 * n * (max_c + np.log(sum_e) / T)
        w = e / sum_e
        g = np.zeros(n)
        active = w > 1e-6
        for k in np.where(active)[0]:
            g += w[k] * grad_conv_at_peak(x, k)
        g *= 2.0 * n
        return obj_val, g

    try:
        res = sp_minimize(obj_grad, a, method='L-BFGS-B',
                          bounds=[(1e-12, None)] * n,
                          options={'maxiter': 300, 'maxfun': 600, 'disp': False})
        a_opt = np.maximum(res.x, 0)
        s_opt = float(np.sum(a_opt))
        if s_opt > 1e-10:
            a_opt = a_opt / s_opt
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
    except Exception:
        pass

    return best_a


def coordinate_refine(a, time_budget=30.0):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        contrib = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib.append((a[i] * a[j], i, j))
        contrib.sort(reverse=True)

        non_contrib = []
        for i in range(n):
            j = k_star - i
            if j < 0 or j >= n:
                non_contrib.append(i)

        if not non_contrib:
            break

        improved = False
        for ci, (val, i, j) in enumerate(contrib[:20]):
            if val < 1e-12 or ci >= 20:
                break
            for target in non_contrib[:10]:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = frac * a[min(i, j)]
                    if delta < 1e-15:
                        continue
                    src = i if a[i] >= a[j] else j
                    old_val = a[src]
                    a[src] -= delta
                    a[target] += delta
                    new Obj = compute_obj(a)
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] = old_val
                        a[target] -= delta
                        break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.05))
        a = a / a.sum()
    elif mode == 5:
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2 + 0.05
        a = a / a.sum()
    elif mode == 7:
        a = np.zeros(n)
        a[rng.choice(n, size=n//3, replace=False)] = 1.0
        a += 0.01
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # Phase 1: Quick Nesterov
    print(f"\n=== Phase 1: Nesterov (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1 = min(remaining * 0.06, 60.0)
    if p1 > 10:
        a_ns = nesterov_subgradient(best_seq, time_budget=p1, K=7)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov: {best_score:.8f} **")
        else:
            print(f"  Nesterov: {obj_ns:.8f}")

    # Phase 2: LP Alternating (main)
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p2_total = min(remaining * 0.4, 500.0)

    for rnd in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(p2_total / (3 - rnd), remaining * 0.7)
        if alloc < 15:
            break
        if rnd > 0:
            a_pert = best_seq * np.exp(rng.normal(0, 0.02 * (rnd + 1), len(best_seq)))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
        else:
            a_pert = best_seq.copy()

        a_lp = lp_alternating(a_pert, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP r{rnd}: {best_score:.8f} **")
        else:
            print(f"  LP r{rnd}: {obj_lp:.8f}")

    # Phase 3: Mirror descent
    print(f"\n=== Phase 3: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p3 = min(remaining * 0.08, 80.0)
    if p3 > 10:
        a_md = mirror_descent(best_seq, time_budget=p3)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Mirror: {best_score:.8f} **")
        else:
            print(f"  Mirror: {obj_md:.8f}")

    # Phase 4: Smooth optimization
    print(f"\n=== Phase 4: Smooth Opt ===")
    remaining = deadline - time.monotonic()
    p4_total = min(remaining * 0.08, 80.0)
    for T_val in [30.0, 80.0]:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        tb = min(p4_total / 2, remaining * 0.5)
        if tb < 8:
            continue
        a_sm = smooth_optimize(best_seq, time_budget=tb, T=T_val)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Smooth T={T_val}: {best_score:.8f} **")

    # Phase 5: Multi-n exploration
    print(f"\n=== Phase 5: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    n_opts = sorted(set([
        max(200, n_cur - 100), max(300, n_cur - 50),
        n_cur + 50, n_cur + 100, n_cur + 200,
        500, 600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_opts = [n for n in n_opts if 200 <= n <= 1500 and n != n_cur]
    n_opts.sort(key=lambda x: abs(x - n_cur))

    n_tried = 0
    for n_try in n_opts[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 30 or n_tried >= 5:
            break
        n_tried += 1
        a_try = resample_sequence(best_seq, n_try)
        alloc_ns = min(remaining * 0.08, 30.0)
        if alloc_ns > 8:
            a_try = nesterov_subgradient(a_try, time_budget=alloc_ns, K=4)
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.08, 40.0)
        if alloc_lp > 10:
            a_try = lp_alternating(a_try, time_budget=alloc_lp)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # Phase 6: Diverse restarts
    print(f"\n=== Phase 6: Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.15 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        rb = min(remaining * 0.12, 50.0)
        a_opt = lp_alternating(a_r, time_budget=rb)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  R{r} m{mode}: {best_score:.8f} **")
        else:
            print(f"  R{r} m{mode}: {obj_r:.8f}")

    # Phase 7: Final polish
    print(f"\n=== Phase 7: Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        tb = min(remaining * 0.3, 60.0)
        a_lp = lp_alternating(best_seq, time_budget=tb, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: {best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        tb = min(remaining * 0.2, 30.0)
        a_ns = nesterov_subgradient(best_seq, time_budget=tb, K=9)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  NS: {best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 12:
        tb = min(remaining * 0.2, 20.0)
        a_cd = coordinate_refine(best_seq, time_budget=tb)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: {best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.5, 25.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  LP2: {best_score:.8f} **")

    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
