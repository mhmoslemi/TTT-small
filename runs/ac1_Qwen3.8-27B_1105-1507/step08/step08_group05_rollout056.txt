"""
LP alternating with active-set constraints, diverse initializations,
n-sweeping, and multi-peak equalization. Uses FFT-based convolution,
warm-started LP solves, and aggressive line search to minimize
2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def _next_power_of_2(x):
    p = 1
    while p < x:
        p <<= 1
    return p


def compute_conv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = _next_power_of_2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full linear cross-convolution via FFT."""
    n = len(a)
    size = _next_power_of_2(2 * n - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return float(2.0 * n * np.max(conv))


def project_nonneg_sum1(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0)
    s = np.sum(v)
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s


def build_conv_rows(a, indices):
    """
    Build rows of the conv matrix for specific lag indices.
    Returns matrix A (len(indices) x n) such that A @ g = [conv(a,g)[k] for k in indices].
    """
    n = len(a)
    A = np.zeros((len(indices), n))
    for row_i, k in enumerate(indices):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for j in range(i_start, i_end):
            A[row_i, j] = a[k - j]
    return A


def solve_lp_active(a, k_active=60, max_iterations=15, solver=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for active k, sum(g)=1, g>=0.
    Uses active-set method: start with top peaks, add violated constraints.
    Returns (g_normalized, M_value) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    conv_aa = compute_conv(a_n)

    # Initial active set: top k peaks
    active_set = list(np.argsort(conv_aa)[::-1][:k_active])

    gv = None
    M_val = np.inf

    for it in range(max_iterations):
        A = build_conv_rows(a_n, active_set)

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        solved = False
        for sol in ([solver] if solver else [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]):
            try:
                prob.solve(solver=sol)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            return None, None

        gv = np.maximum(g.value, 0)
        gs = float(np.sum(gv))
        if gs < 1e-12:
            return None, None
        gv = gv / gs
        M_val = float(M.value)

        # Check for violated constraints
        conv_ag = cross_conv(a_n, gv)
        tolerance = 1e-9
        violated = np.where(conv_ag > M_val + tolerance)[0].tolist()

        if len(violated) == 0:
            return gv, M_val

        # Add violated constraints, keep top (k_active + len(violated)) by conv_ag value
        all_indices = list(set(active_set + violated))
        all_indices = sorted(all_indices, key=lambda k: -conv_ag[k])
        active_set = all_indices[:k_active + len(violated)]

    return gv, M_val


def lp_alternate(a0, time_budget=60.0, k_active=60, verbose=False):
    """LP alternating with 1D line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_active(a, k_active=k_active)
        if g_lp is None:
            break

        # Line search: minimize obj((1-t)*a + t*g_lp)
        def _line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-12:
                return np.inf
            return compute_obj(a_try / st)

        best_t = 0.5
        try:
            res = minimize_scalar(_line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 40, 'xatol': 0.005})
            if res.success:
                best_t = res.x
        except Exception:
            pass

        a_new = (1.0 - best_t) * a + best_t * g_lp
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-12:
            break
        a_new = a_new / sn
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-11:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose:
                print(f"        LP: obj={best_obj:.8f}, t={best_t:.3f}, t_elapsed={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 20:
                break

    return best_a


def subgradient_polish(a0, time_budget=30.0, top_k=5, verbose=False):
    """Subgradient descent on simplex using exact gradient formula."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05
    y = a.copy()
    patience = 0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]

        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                m_start = k - i_end
                m_end = k - i_start
                grad[m_start:m_end + 1] += 2.0 * a[i_start:i_end + 1][::-1]
        grad /= len(top_idx)
        grad -= np.mean(grad)

        y_new = y - step * grad
        a_new = project_nonneg_sum1(y_new)
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1.0 - t_k) * (a_new - a)
        y = project_nonneg_sum1(y)
        a = a_new

        if it % 20 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 300:
                step = max(step * 0.5, 0.002)
                patience = 0
        if it % 1000 == 999:
            step = max(step * 0.95, 0.003)

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def make_initializations(n, rng):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / np.sum(ramp))

    # Reverse ramp
    inits.append(ramp[::-1] / np.sum(ramp))

    # Triangle
    tri = np.minimum(np.arange(n) + 1.0, float(n) - np.arange(n))
    inits.append(tri / np.sum(tri))

    # Parabolic (peak in middle)
    x = np.linspace(-1, 1, n)
    para = np.maximum(1.0 - x**2, 0)
    inits.append(para / np.sum(para))

    # Gaussian
    mu = n / 2.0
    sigma = n / 4.0
    gauss = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
    inits.append(gauss / np.sum(gauss))

    # Hann window
    hw = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    inits.append(hw / np.sum(hw))

    # Bell-curve (narrower Gaussian)
    sigma2 = n / 6.0
    gauss2 = np.exp(-0.5 * ((np.arange(n) - mu) / sigma2) ** 2)
    inits.append(gauss2 / np.sum(gauss2))

    # Exponential decay
    expd = np.exp(-np.arange(n) / (n * 0.3))
    inits.append(expd / np.sum(expd))

    # Random
    rand = rng.random(n)
    inits.append(rand / np.sum(rand))

    # Two bumps
    b1 = np.exp(-0.5 * ((np.arange(n) - n * 0.25) / (n * 0.1)) ** 2)
    b2 = np.exp(-0.5 * ((np.arange(n) - n * 0.75) / (n * 0.1)) ** 2)
    two = b1 + b2
    inits.append(two / np.sum(two))

    # S-flat (approximately flat middle, tapered edges)
    sf = np.ones(n)
    taper = int(n * 0.15)
    for i in range(taper):
        sf[i] = 0.3 + 0.7 * i / taper
        sf[n - 1 - i] = 0.3 + 0.7 * i / taper
    inits.append(sf / np.sum(sf))

    return inits
