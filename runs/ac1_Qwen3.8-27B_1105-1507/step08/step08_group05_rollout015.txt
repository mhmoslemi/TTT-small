def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def logsumexp_max(conv, tau):
    """Smooth approximation of max using log-sum-exp."""
    shifted = conv - np.max(conv)
    return np.max(conv) + tau * np.log(np.sum(np.exp(shifted / tau)))

    top_idx = np.argsort(conv)[::-1][:top_k]
    top_vals = conv[top_idx]
    shifted = top_vals - np.max(top_vals)
    weights = np.exp(shifted / tau)

            if m != i:

    return grad
