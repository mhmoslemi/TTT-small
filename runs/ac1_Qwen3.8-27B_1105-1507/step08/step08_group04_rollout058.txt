Let me also think about the **simplex projection** more carefully:

The Euclidean projection of a point `v` onto `{x: x >= 0, sum(x) = 1}` is:
`x* = max(v - tau, 0)` where `tau` is chosen so that `sum(max(v - tau, 0)) = 1`.

To find tau, sort v in decreasing order: `v_(1) >= v_(2) >= ... >= v_(n)`.

from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    return compute_obj_normalized(a / s)

def softmax(x, beta=1.0):
    x_scaled = beta * x
    x_scaled = x_scaled - np.max(x_scaled)
    e = np.exp(x_scaled)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_hat = np.max(rho[cond])
    theta = (cssv[rho_hat - 1] - 1.0) / rho_hat
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, beta=10.0):
    """Compute smoothed gradient of max(conv(a,a)) for normalized a.
    Returns (gradient, convolution)."""
    n = len(a)
    conv = compute_conv(a)
    w = softmax(conv, beta)

    # = 2 * IFFT(FFT(w) * conj(FFT(a)))[:n] (with proper padding)
    size = 1
    while size < 3 * n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    corr = irfft(W * np.conj(A), size)[:n]
    grad = 2.0 * corr
    return grad, conv

def gradient_descent(a, time_budget=60.0, beta=10.0, verbose=False):
    """Nesterov-accelerated gradient descent on simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))

        return a

    best_obj = compute_obj_normalized(a)

    beta_mom = 0.9
    lr = 1.0 / n
    x = a.copy()  # Nesterov lookahead

        if time.monotonic() - start > time_budget:
            break
        
        grad, conv = compute_gradient(x, beta=beta)
        # Center gradient for simplex (remove mean to preserve sum)
        grad_centered = grad - np.mean(grad)

        a_new = simplex_project(a_new)

        a = a_new

"""
Optimized solver using FFT-based smoothed gradient descent with momentum,
proper simplex projection, pseudo-LP direction finding, and multi-phase strategy.
Key improvements: O(n log n) gradient via FFT, no matrix construction,
adaptive step size, and faster LP approximation via gradient steps on g.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ab(a, b):
    """FFT-based full linear convolution of a and b."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj_normalized(a):
    """Compute 2n*max(conv(a,a)) assuming sum(a)~1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute the full objective 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    return compute_obj_normalized(a / s)


def softmax_stable(x, beta=1.0):
    """Numerically stable softmax with temperature."""
    xs = beta * x
    xs = xs - np.max(xs)
    e = np.exp(xs)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def simplex_project(v):
    """Euclidean projection onto {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_hat = np.max(rho[cond])
    theta = (cssv[rho_hat - 1] - 1.0) / rho_hat
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(a, w):
    """Compute grad[m] = 2 * sum_k w[k]*a[k-m] using FFT.
    a has length n, w has length 2n-1. Returns array of length n."""
    n = len(a)
    size = 1
    while size < 3 * n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    corr = irfft(W * np.conj(A), size)[:n]
    return 2.0 * corr


def compute_gradient(a, beta=10.0):
    """Compute smoothed gradient of max(conv(a,a)) for normalized a."""
    n = len(a)
    conv = compute_conv(a)
    w = softmax_stable(conv, beta)
    grad = compute_gradient_fft(a, w)
    return grad, conv


def fast_gradient_descent(a, time_budget=60.0, beta=10.0, verbose=False):
    """Nesterov-accelerated gradient descent on the simplex with FFT gradients."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()

    if s < 1e-10:

    best_obj = compute_obj_normalized(a)

    beta_mom = 0.85
    lr = 2.0 / n
    x = a.copy()
    iteration = 0

    while time.monotonic() - start < time_budget:
        iteration += 1
        grad, conv = compute_gradient(x, beta=beta)
        grad_centered = grad - np.mean(grad)

        momentum = beta_mom * momentum - lr * grad_centered
        a_new = a + momentum
        a_new = simplex_project(a_new)

        obj = compute_obj_normalized(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        # Nesterov extrapolation
        t_k = 2.0 / (iteration + 2.0)
        x = a_new + (1.0 - t_k) * (a_new - a)
        x = simplex_project(x)
        a = a_new

            print(f"      GD: it={iteration}, obj={best_obj:.8f}, lr={lr:.2e}, beta={beta:.1f}, t={elapsed:.0f}s")

    return best_a


def pseudo_lp_direction(a, n_steps=20, lr_g=1.0, beta_g=5.0):
    """Approximate LP direction by gradient steps on g.
    Finds g that minimizes max(conv(a,g)) approx."""
    n = len(a)
    g = a.copy()
    for _ in range(n_steps):
        conv_ag = compute_conv_ab(a, g)
        w = softmax_stable(conv_ag, beta_g)
        # Gradient w.r.t. g: d/dg max(conv(a,g)) 
        # conv(a,g)[k] = sum_j a[j]*g[k-j], d/dg[m] = a[k-m] (symmetric)
        # Same formula: grad_g[m] = sum_k w[k]*a[k-m]
        grad_g = compute_gradient_fft(g, w)

        grad_g = compute_gradient_fft(a, w)

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = np.sum(a_new)
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

            for i in range(n):
                d = abs(i - center)
                a[i] += rng.standard_normal() * 0.5 * np.exp(-d / (width * 0.3))
        a = np.maximum(a, 0.001)
    elif choice == 4:
        # Multi-frequency
        a = np.ones(n)
        for _ in range(3):
            f = rng.uniform(1, 10)
            ph = rng.uniform(0, 2 * np.pi)
            a += 0.3 * np.cos(2 * np.pi * f * x + ph)
        a = np.maximum(a, 0.001)
    else:
        # Random smooth via Gaussian process-like
        a = np.ones(n)
        kernel_width = rng.uniform(n // 20, n // 5)
        for i in range(n):
            a[i] = np.exp(-i**2 / (2 * kernel_width**2))
        a = np.abs(a + rng.normal(0, 0.3, n))
        a = np.maximum(a, 0.001)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def spectral_flatten(a, n_iters=50, verbose=False):
    """Gerchberg-Saxton style: alternate between flattening spectrum and enforcing non-negativity."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    size = 1
    while size < 2 * n:
        size <<= 1

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    a_work = a.copy()
    for it in range(n_iters):
        if time.monotonic() - start > 30:
            break
        # Go to frequency domain
        A = rfft(a_work, size)
        power = np.abs(A) ** 2

        # Soft-flatten: reduce peaks in the power spectrum
        mean_p = np.mean(power)
        if mean_p > 0:
            # Compress dynamic range
            ratio = power / mean_p
            # Soft clip: ratio_new = ratio / (1 + alpha*(ratio-1)^2) for ratio>1
            alpha = 0.1 + 0.01 * it
            compressed = np.where(ratio > 1.0, ratio / (1.0 + alpha * (ratio - 1.0)**2), ratio)
            # Renormalize to preserve total power
            scale = np.mean(power) / np.mean(compressed * mean_p)
            new_power = compressed * mean_p * scale
            # Keep phase, adjust amplitude
            phase = np.angle(A * size)  # A is real FFT, handle carefully
            # Actually A is complex from rfft
            new_A = np.sqrt(np.maximum(new_power, 0)) * np.exp(1j * np.angle(A))
            a_work = irfft(new_A, size)[:n]

        # Enforce non-negativity and normalize
        a_work = np.maximum(a_work, 0)
        s_w = float(np.sum(a_work))
        if s_w < 1e-10:
            a_work = best_a.copy()
        else:
            a_work = a_work / s_w

        obj = compute_obj_normalized(a_work)
        if obj < best_obj:
            best_obj = obj
            best_a = a_work.copy()

        if verbose and it % 10 == 0:
            print(f"      Spectral: it={it}, obj={best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining FFT gradient descent,
    pseudo-LP direction finding, spectral flattening, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        n_base = len(a)
        print(f"Parent: n={n_base}, obj={compute_obj_normalized(a):.8f}")
    except Exception:
        a = np.ones(800) / 800
        n_base = 800
        print(f"No parent, using uniform n=800, obj={compute_obj_normalized(a):.8f}")

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)

    # === Phase 1: Fast gradient descent (main workhorse) ===
    print("\n=== Phase 1: FFT Gradient Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 250.0)
    a_p1 = fast_gradient_descent(best_a, time_budget=p1_budget, beta=10.0, verbose=True)
    obj_p1 = compute_obj_normalized(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1
        print(f"  Phase 1: obj={best_obj:.8f} **")
    else:
        print(f"  Phase 1: obj={obj_p1:.8f} (no improvement)")

    # === Phase 2: Pseudo-LP alternating ===
    print("\n=== Phase 2: Pseudo-LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = min(remaining * 0.20, 150.0)
        a_p2 = lp_alternate_pseudo(best_a, time_budget=p2_budget, verbose=True)
        obj_p2 = compute_obj_normalized(a_p2)
        if obj_p2 < best_obj:
            best_obj = obj_p2
            best_a = a_p2
            print(f"  Phase 2: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_p2:.8f} (no improvement)")

    # === Phase 3: More gradient descent with higher beta ===
    print("\n=== Phase 3: GD with Higher Beta ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.20, 150.0)
        a_p3 = fast_gradient_descent(best_a, time_budget=p3_budget, beta=25.0, verbose=True)
        obj_p3 = compute_obj_normalized(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3
            print(f"  Phase 3: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 3: obj={obj_p3:.8f} (no improvement)")

    # === Phase 4: Multi-n exploration ===
    print("\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([
        max(200, n_best - 100),
        n_best,
        min(2000, n_best + 100),
        min(3000, n_best + 200),
        min(2000, n_best + 400),
    ]))
    n_options = [x for x in n_options if 300 <= x <= 3000 and x != n_best]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_seq(best_a, n_try)

        # Quick optimization
        alloc = min(40.0, remaining * 0.25)
        a_opt = fast_gradient_descent(a_try, time_budget=alloc, beta=10.0)
        obj_try = compute_obj_normalized(a_opt)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Spectral flattening ===
    print("\n=== Phase 5: Spectral Flattening ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_sp = spectral_flatten(best_a, n_iters=100, verbose=True)
        obj_sp = compute_obj_normalized(a_sp)
        if obj_sp < best_obj:
            best_obj = obj_sp
            best_a = a_sp
            print(f"  Phase 5: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 5: obj={obj_sp:.8f} (no improvement)")
            # Polish spectral result with GD
            a_sp2 = fast_gradient_descent(a_sp, time_budget=min(20, remaining * 0.3), beta=15.0)
            obj_sp2 = compute_obj_normalized(a_sp2)
            if obj_sp2 < best_obj:
                best_obj = obj_sp2
                best_a = a_sp2
                print(f"  Phase 5 polish: obj={best_obj:.8f} **")

    # === Phase 6: Diverse restarts ===
    print("\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_use = len(best_a)

    for d in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_div = diverse_initialization(n_use, rng)
        # Blend with best to encourage proximity
        a_div = 0.5 * a_div + 0.5 * best_a
        a_div = simplex_project(a_div)

        alloc = min(30.0, remaining * 0.2)
        a_opt = fast_gradient_descent(a_div, time_budget=alloc, beta=10.0)
        obj_div = compute_obj_normalized(a_opt)

        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Diverse {d}: obj={obj_div:.8f}")

    # === Phase 7: Final polish ===
    print("\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        # GD with high beta
        alloc = min(remaining * 0.4, 60.0)
        a_fin = fast_gradient_descent(best_a, time_budget=alloc, beta=40.0)
        obj_fin = compute_obj_normalized(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin
            print(f"  Final GD: obj={best_obj:.8f} **")

    if remaining > 10:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.3, 30.0)
        a_fin2 = lp_alternate_pseudo(best_a, time_budget=alloc)
        obj_fin2 = compute_obj_normalized(a_fin2)
        if obj_fin2 < best_obj:
            best_obj = obj_fin2
            best_a = a_fin2
            print(f"  Final PseudoLP: obj={best_obj:.8f} **")

    # === Phase 8: One more GD round if time ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.5, 30.0)
        a_fin3 = fast_gradient_descent(best_a, time_budget=alloc, beta=50.0)
        obj_fin3 = compute_obj_normalized(a_fin3)
        if obj_fin3 < best_obj:
            best_obj = obj_fin3
            best_a = a_fin3
            print(f"  Final GD2: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
