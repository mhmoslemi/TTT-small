    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A

def subgradient_step(a, top_k=20):
    n = len(a)
    conv = compute_conv(a)
    if top_k >= len(conv):
        k_indices = np.arange(len(conv))
    else:
        k_indices = np.argsort(conv)[::-1][:top_k]
    
    w = np.zeros(2 * n - 1)
    for k in k_indices:
        w[int(k)] = 1.0 / len(k_indices)
    
    n_fft = 1
    while n_fft < 4 * n:
        n_fft <<= 1
    A_fft = rfft(a, n_fft)
    W_fft = rfft(w, n_fft)
    R = irfft(np.conj(A_fft) * W_fft, n_fft)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = R[(n_fft - j) % n_fft]
    grad *= 4.0 * n
    grad -= np.mean(grad)
    return grad

def bb_subgradient(a, time_budget=60.0, top_k=20):
    """Barzilai-Borwein adaptive subgradient method."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    grad_prev = None
    x_prev = None
    t_k = 1.0
    
    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break
        
        grad = subgradient_step(a, top_k=top_k)
        
        # BB step size
        if grad_prev is not None and x_prev is not None:
            dx = a - x_prev
            dy = grad - grad_prev
            denom = np.dot(dy, dy)
            if denom > 1e-20:
                t_bb = np.dot(dx, dy) / denom
                t_bb = max(1e-4, min(t_bb, 1.0))
                t_k = 0.7 * t_k + 0.3 * t_bb
        
        a_new = a - t_k * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            t_k *= 0.8
            a = a_new  # still accept but reduce step
        
        grad_prev = grad
        x_prev = a.copy()
        
        if it % 500 == 499:
            t_k *= 1.05  # occasional boost
    
    return best_a

def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

- Constraint-generation LP alternation (faster than full LP)

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_prod(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))

        solved = False

def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

    if remaining > 15:
        alloc = min(remaining * 0.25, 40.0)
        a_sg = bb_subgradient(best_a, time_budget=alloc, top_k=30)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"  Final BB: obj={best_obj:.8f} **")
    
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 30.0)
        a_last = lp_alternate_improved(best_a, time_budget=alloc)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Last LP: obj={best_obj:.8f} **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val == np.inf:
        result = [1.0] * 100
        val = 2.0
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={val:.8f}, time={elapsed:.1f}s ===")
    return result
