    size = 1
    while size < 2 * n - 1:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

    # df_2/da_1 = 2*a_1. And 2*a_{2-1} = 2*a_1. ✓
    # df_2/da_2 = 2*a_0. And 2*a_{2-2} = 2*a_0. ✓
    # 
    # Great, so the gradient of the smooth max is:
    # grad_j = sum_k weights_k * 2 * a_{k-j} (if 0 <= k-j < n, else 0)
    #        = 2 * sum_k weights_k * a_{k-j}
    # 
    # This is 2 * (cross-correlation of weights and a, evaluated at lag j)
    # where the cross-correlation is defined as sum_k weights[k] * a[k-j].
    
    # We need: grad_j = 2 * sum_k weights[k] * a[k-j] for k-j in [0, n-1]
    # This is 2 * (weights * a) at lag j, but we need to be careful about the correlation.
    # 
    # Let's define h_j = sum_k weights[k] * a[k-j] for valid k-j
    # This is the cross-correlation of weights and a at lag j.
    # 
    # Using FFT: cross-correlation can be computed as:
    # h = irfft(conj(rfft(a)) * rfft(weights_padded))
    # But we need to pad appropriately.
    # 
    # Actually, let's think of it as: h_j = sum_k weights[k] * a[k-j]
    # Let's define b[k] = weights[k] for k in 0..2n-2, 0 otherwise.
    # Then h_j = sum_k b[k] * a[k-j]
    # If we define c[m] = a[m] for m in 0..n-1, 0 otherwise
    # Then h_j = sum_k b[k] * c[k-j] = (b * c')_j where c' is the "reflected" version
    # 
    # Actually, this is the convolution of b with the reflected c:
    # h_j = sum_k b[k] * c[k-j] = (b * c_reflected)_j where c_reflected[m] = c[-m] = c_j for m = -j
    # Hmm, this is getting confusing. Let me just use a direct formula.
    # 
    # h_j = sum_{k=j}^{j+n-1} weights[k] * a[k-j] (clipping to valid range of weights: 0..2n-2)
    #     = sum_{m=0}^{n-1} weights[j+m] * a[m] (where weights is 0 outside [0, 2n-2])
    
    # This is a sliding dot product: h is the cross-correlation of weights and a.
    # We can compute it via FFT:
    # h = ifft(conj(fft(a)) * fft(weights))  -- but we need to handle lengths carefully.
    
    # Actually the simplest way: use np.correlate or manual FFT.
    # weights has length 2n-1, a has length n.
    # h_j = sum_m weights[j+m] * a[m] for m=0..n-1 (with weights 0 outside its range)
    # This is the cross-correlation of weights and a at lag j.
    # In signal processing: h = correlate(weights, a, mode='valid')  -- no, that gives the wrong thing.
    # 
    # Let me just use the direct FFT approach:
    # h_j = sum_m a[m] * weights[j+m]
    # If we zero-pad a to length 2n-1 (same as weights), then:
    # h = ifft(fft(weights) * fft(a_padded))  -- this gives the circular correlation
    # But we need the linear version. Since weights has length 2n-1 and a (padded) has length 2n-1,
    # and we only need h_j for j=0..n-1, we can use:
    
    size = 1
    while size < 3 * n:
        size <<= 1
    # h_j = sum_m a[m] * weights[j+m], m=0..n-1, j=0..n-1
    #这等价于 (a 和 reversed weights 的卷积) 的某部分
    # 或者直接用 correlate
    # 
    # Actually the simplest: h is the output of filtering 'a' with 'reversed weights'
    # h_j = sum_m a[m] * weights[j+m] = sum_m a[m] * (reverse_of_weights)[len(weights)-1-(j+m)]
    # This is getting complicated. Let me just use np.correlate.
    
    # np.correlate(weights, a, mode='full') gives:
    # out[k] = sum_m weights[m+k] * a[m]  (for the 'valid' range)
    # Actually, numpy's correlate definition:
    # v1[k] * v2[k] summed... let me just check.
    # np.correlate(a, v, mode='full')[k] = sum_n a[n] * v[n-k+L-1] where L = len(v)
    # 
    # I'll just compute it directly using the formula:
    # h_j = sum_{m=0}^{n-1} a[m] * weights[j+m] (with weights 0 outside [0, 2n-2])
    
    # Use FFT for efficiency:
    # h_j = sum_m a[m] * w[j+m] where w = weights (length 2n-1)
    # This is the cross-correlation. We can compute:
    # h = ifft(fft(w, size) * conj(fft(a, size))) and take the right part.
    # 
    # Actually, cross-correlation R[j] = sum_m a[m] * w[m+j]
    # = sum_m a[m] * conj(w[m+j]) (for real signals, conj is identity)
    # In Fourier domain: R = ifft(fft(w) * conj(fft(a)))  -- NO
    # 
    # Let me use the standard formula:
    # cross-correlation of a and w: R[j] = sum_m a[m] * w[m+j]
    # FFT: R = ifft(conj(FFT(a)) * FFT(w))  -- this gives circular cross-correlation
    # For linear: zero-pad to at least len(a) + len(w) - 1 = n + (2n-1) - 1 = 3n - 2
    
    size2 = 1
    while size2 < 3 * n:
        size2 <<= 1
    fa2 = rfft(a, size2)
    fw2 = rfft(np.zeros(size2))  # placeholder
    fw2 = rfft(np.concatenate([weights, np.zeros(size2 - len(weights))]), size2)
    h_full = irfft(fw2 * np.conj(fa2), size2)
    # h_full[j] = sum_m a[m] * w[m+j] for j = 0, 1, 2, ...
    # We want h[j] for j = 0, ..., n-1
    h = h_full[:n]
    
    grad = 2.0 * h
    smooth_val = max_c + T * np.log(total)
    return smooth_val, grad


def accelerated_smooth_max(a, time_budget=60.0, T_init=0.005, T_final=0.00005, 
                           n_iters=None, verbose=False, seed=0):
    """
    Nesterov-accelerated gradient descent on smooth max with adaptive temperature.
    Works on the simplex (sum(a)=1, a>=0).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if n_iters is None:
        n_iters = 5000
    
    T = T_init
    T_decay = (T_final / T_init) ** (1.0 / max(n_iters, 1))
    
    lr = 2.0 / (n + 1.0)  # standard step for smooth convex on simplex
    momentum = 0.9
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead
    
    stall = 0
    best_lr = lr
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Nesterov lookahead
        y = a + momentum * v
        
        # Compute gradient at y (project to simplex first)
        y_proj = simplex_project(y)
        _, grad = smooth_max_objective_and_grad(y_proj, T)
        
        # Center the gradient (since we're on the simplex, only the centered part matters)
        grad_centered = grad - np.mean(grad)
        
        # Update
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            if verbose and it % 200 == 0:
                print(f"            it={it}: obj={best_obj:.10f}, T={T:.2e}, lr={lr:.4f}")
        else:
            stall += 1
            if stall > 50:
                lr = max(lr * 0.8, 1e-6)
                a = best_a.copy()
                v = np.zeros(n)
                y = a.copy()
                stall = 0
        
        # Temperature decay
        T = max(T_final, T * T_decay)
        
        # Periodic lr adjustment
        if it % 100 == 0 and stall == 0:
            lr = min(lr * 1.02, 0.1)
    
    return best_a


def peak_targeted_refinement(a, time_budget=120.0, verbose=False):
    """
    Target the top convolution peaks and try to reduce them by adjusting
    the coordinates that contribute most to those peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for outer in range(50):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        
        # Find top peaks
        sort_idx = np.argsort(conv)[::-1]
        top_k = min(10, len(sort_idx))
        top_indices = sort_idx[:top_k]
        
        # For each top peak, find which coordinates contribute most
        for k in top_indices:
            if time.monotonic() - start > time_budget:
                break
            
            # Contribution of coordinate i to peak k: a[i] * a[k-i] (if valid)
            # Find the coordinate with maximum contribution
            contrib = np.zeros(n)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contrib[i] = a[i] * a[j]
            
            # Try reducing the top-contributing coordinate
            top_i = np.argmax(contrib)
            top_j = k - top_i
            
            # Try moving mass from top_i to a less consequential coordinate
            # Find a coordinate that minimally affects the top peaks
            min_contrib = np.zeros(n)
            for i in range(n):
                for k2 in top_indices:
                    j2 = k2 - i
                    if 0 <= j2 < n:
                        min_contrib[i] += a[i] * a[j2]
            
            # Choose a coordinate with minimal impact on all top peaks
            candidates = np.argsort(min_contrib)
            for dest in candidates[:5]:
                if dest == top_i:
                    continue
                
                # Try moving a small amount from top_i to dest
                for frac in [0.05, 0.1, 0.2, 0.01]:
                    move_amt = a[top_i] * frac
                    if move_amt < 1e-15:
                        continue
                    a_try = a.copy()
                    a_try[top_i] -= move_amt
                    a_try[dest] += move_amt
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        break
                if obj_try < best_obj + 1e-10:
                    break
        
        if verbose and outer % 5 == 0:
            print(f"            peak_refine iter={outer}: obj={best_obj:.10f}")
    
    return best_a


def fast_coordinate_descent(a, time_budget=100.0, verbose=False):
    """
    Faster coordinate descent using gradient-guided step sizes.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv = compute_conv(a)
    max_c = np.max(conv)
    
    # Compute initial gradient for direction finding
    T = 0.001
    _, grad = smooth_max_objective_and_grad(a, T)
    grad_centered = grad - np.mean(grad)
    
    delta = 1e-3
    pass_count = 0
    stall_passes = 0
    
    while time.monotonic() - start < time_budget:
        pass_count += 1
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        rng = np.random.default_rng(pass_count * 12345 + 67890)
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            # Use gradient to determine direction
            direction = -1 if grad_centered[i] > 0 else 1
            
            for d in [delta * direction, -delta * 0.5, delta * 0.5]:
                step = d
                if a[i] + step < 0:
                    continue
                a_try = a.copy()
                a_try[i] += step
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    # Update gradient approximately
                    conv = compute_conv(a)
                    max_c = np.max(conv)
                    T_curr = 0.001
                    _, grad = smooth_max_objective_and_grad(a, T_curr)
                    grad_centered = grad - np.mean(grad)
                    break
            
            if improved:
                continue
        
        if not improved:
            stall_passes += 1
            delta *= 0.7
            if delta < 1e-7 or stall_passes > 10:
                break
        else:
            stall_passes = 0
        
        if verbose and pass_count % 10 == 0:
            print(f"            CD pass {pass_count}: obj={best_obj:.10f}, delta={delta:.2e}")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def solve_lp_quick(a):
    """Quick LP solve to find optimal direction."""
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in ['CLARABEL', 'ECOS', 'SCS', 'MOSEK']:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def convex_line_search_a(a, g):
    """Find optimal blend between a and g using convexity."""
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)
    
    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)
    
    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return float(np.max(val))
    
    lo, hi = 0.0, 1.0
    for _ in range(12):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a_norm
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-start smooth-max gradient descent with Nesterov acceleration,
    peak-targeted refinement, and multi-n exploration.
    
    Strategy:
    1. Try diverse initializations (parent, uniform, Gaussian, random)
    2. Fast optimization of each with accelerated smooth-max
    3. Peak-targeted refinement of the best
    4. Multi-n exploration
    5. Final polish with coordinate descent and LP
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={len(parent)}): obj={best_score:.10f}")
    except Exception:
        pass

    # Phase 1: Multi-start exploration
    print("\n=== Phase 1: Multi-start Exploration ===")
    
    # Generate diverse initializations
    inits = []
    n_base = len(best_seq) if best_seq is not None else 800
    
    # Parent
    if best_seq is not None:
        inits.append(("parent", best_seq.copy()))
    
    # Uniform
    inits.append(("uniform", np.ones(n_base) / n_base))
    
    # Gaussian-like
    x = np.linspace(-1, 1, n_base)
    gauss = np.exp(-x**2 * 2)
    inits.append(("gaussian", gauss / np.sum(gauss)))
    
    # Cosine window
    cos_w = np.cos(np.pi * x) ** 2  # Hann-like
    cos_w = np.maximum(cos_w, 0)
    inits.append(("cosine", cos_w / np.sum(cos_w)))
    
    # Random (smoothed)
    r = rng.random(n_base)
    # Smooth it
    kernel = np.ones(10) / 10
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(("random_smooth", np.maximum(r_smooth, 0.01) / np.sum(r_smooth)))
    
    # Triangular
    tri = np.concatenate([np.linspace(0, 1, n_base//2), np.linspace(1, 0, n_base - n_base//2)])
    inits.append(("triangle", tri / np.sum(tri)))
    
    # Different n values with uniform
    for n_try in [700, 850, 900, 1000]:
        if n_try != n_base:
            inits.append((f"uniform_n{n_try}", np.ones(n_try) / n_try))
    
    # Try each initialization with fast optimization
    phase1_time = min(400.0, (deadline - start_time) * 0.3)
    per_init_time = phase1_time / len(inits)
    
    for name, init in inits:
        if time.monotonic() > deadline - 200:
            break
        
        alloc = min(per_init_time, 80.0)
        if alloc < 5:
            continue
        
        print(f"  Trying '{name}' (n={len(init)}, budget={alloc:.0f}s)...")
        result = accelerated_smooth_max(init.copy(), time_budget=alloc, 
                                         T_init=0.005, T_final=0.0001,
                                         verbose=False)
        obj = compute_obj(result)
        print(f"    → obj={obj:.10f}")
        
        if obj < best_score:
            best_score = obj
            best_seq = result.copy()
            print(f"    ** New best! **")
    
    n_best = len(best_seq)
    print(f"\nAfter Phase 1: n={n_best}, score={best_score:.10f}")
    
    # Phase 2: Intensive optimization of best
    print("\n=== Phase 2: Intensive Optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        phase2_budget = min(300.0, remaining * 0.3)
        
        a = best_seq.copy()
        # Continue with lower temperature
        result2 = accelerated_smooth_max(a, time_budget=phase2_budget,
                                          T_init=0.0005, T_final=0.00002,
                                          n_iters=8000, verbose=True)
        obj2 = compute_obj(result2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = result2.copy()
            a = result2.copy()
            print(f"  Phase 2 improved: obj={best_score:.10f}")
        else:
            print(f"  Phase 2: obj={obj2:.10f} (no improvement)")
    
    # Phase 3: LP alternating (a few iterations)
    print("\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        phase3_budget = min(120.0, remaining * 0.15)
        a = best_seq.copy()
        
        for lp_iter in range(20):
            if time.monotonic() > deadline - 150:
                break
            if time.monotonic() - (deadline - phase3_budget) > phase3_budget:
                break
            
            g_lp = solve_lp_quick(a)
            if g_lp is None:
                break
            
            a_new = convex_line_search_a(a, g_lp)
            obj_new = compute_obj(a_new)
            if obj_new < best_score - 1e-13:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                remaining_iter = deadline - 150 - time.monotonic()
                if remaining_iter < 5:
                    break
                print(f"  LP iter {lp_iter}: obj={best_score:.10f}")
            else:
                break
        
        print(f"  After LP: obj={best_score:.10f}")
    
    # Phase 4: Peak-targeted refinement
    print("\n=== Phase 4: Peak-Targeted Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        phase4_budget = min(150.0, remaining * 0.2)
        a_pt = peak_targeted_refinement(best_seq.copy(), time_budget=phase4_budget, verbose=True)
        obj_pt = compute_obj(a_pt)
        if obj_pt < best_score:
            best_score = obj_pt
            best_seq = a_pt.copy()
            print(f"  Phase 4 improved: obj={best_score:.10f}")
        else:
            print(f"  Phase 4: obj={obj_pt:.10f} (no improvement)")
    
    # Phase 5: Multi-n exploration
    print("\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(500, n_best - 100), max(500, n_best - 50),
        n_best + 25, n_best + 50, n_best + 100,
        600, 700, 750, 800, 850, 900, 950, 1000, 1100
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1200]
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        
        alloc = min(30.0, remaining * 0.1)
        a_opt = accelerated_smooth_max(a_try, time_budget=alloc,
                                        T_init=0.002, T_final=0.00005)
        obj_try = compute_obj(a_opt)
        
        if obj_try < best_score + 0.001 and remaining > 20:
            # Quick CD polish
            a_opt = fast_coordinate_descent(a_opt, time_budget=min(15.0, remaining * 0.2))
            obj_try = compute_obj(a_opt)
        
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try

"""
Multi-start accelerated smooth-max gradient descent with peak-targeted refinement,

Key improvements over previous approach:

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = 1
    while size < n_a + n_b - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n_a + n_b - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_grad(a, T):
    """
    Compute gradient of smooth max of conv(a,a) at a.
    Returns (smooth_value, gradient). Assumes a is on simplex (sum=1, a>=0).
    
    grad_j = 2 * sum_k p_k * a[k-j] where p_k are softmax weights on conv values.
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total  # length 2n-1

    # Compute h_j = sum_m a[m] * weights[j+m] via FFT cross-correlation
    # h_j = conv_aw[2n-2-j] where conv_aw = a * w_rev (linear convolution)
    w_rev = weights[::-1]  # length 2n-1
    size = 1
    while size < 3 * n:
        size <<= 1
    fa = rfft(a, size)
    fwr = rfft(w_rev, size)
    conv_aw = irfft(fa * fwr, size)
    # h_j = conv_aw[2n-2-j] for j=0..n-1
    # Indices: 2n-2, 2n-3, ..., n-1
    h = conv_aw[n - 1:2 * n - 1][::-1]

    grad = 2.0 * h
    smooth_val = max_c + T * np.log(total)
    return smooth_val, grad

    rng = np.random.default_rng(seed)

    T = T_init
    T_decay = (T_final / T_init) ** (1.0 / max(max_iters, 1))

    lr = 1.0 / (2.0 * n)
    momentum = 0.9
    v = np.zeros(n)
    y = a.copy()

    shake_threshold = 150

        a_new = simplex_project(a + v)

        sort_idx = np.argsort(conv)[::-1]
        top_k = min(15, len(sort_idx))
        top_indices = sort_idx[:top_k]

        improved_this_round = False

        for k in top_indices:
            if time.monotonic() - start > time_budget:
                break

            contrib = np.zeros(n)

                j = k - i
                if 0 <= j < n:
                    contrib[i] = a[i] * a[j]

            top_contrib = contrib[top_i]
            if top_contrib < 1e-15:

            # Find coordinates with minimal total contribution to all top peaks

        # Compute gradient for direction guidance
        T = 0.001
        _, grad = smooth_max_grad(a, T)
        grad_centered = grad - np.mean(grad)

        improved = False
        rng = np.random.default_rng(pass_count * 7919 + 104729)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            # Determine direction from gradient
            direction = -1.0 if grad_centered[i] > 0 else 1.0

            for d_mult in [1.0, 0.5, -0.5, 2.0]:
                step = delta * d_mult * direction
                if a[i] + step < -1e-12:
                    continue
                a_try = a.copy()
                a_try[i] += step
                a_try[i] = max(0.0, a_try[i])
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if improved:
                continue

        if not improved:
            stall_passes += 1
            delta *= 0.6
            if delta < 1e-7 or stall_passes > 15:
                break
        else:
            stall_passes = 0

        if verbose and pass_count % 5 == 0:
            print(f"              CD pass {pass_count}: obj={best_obj:.10f}, delta={delta:.2e}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def solve_lp_quick(a):
    """Quick LP solve to find optimal direction on simplex."""
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a = a / s
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in ['CLARABEL', 'ECOS', 'SCS', 'MOSEK']:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    return g_val / s_g
        except Exception:
            continue
    return None


def convex_line_search_blend(a, g):
    """Find optimal blend between a and g exploiting convexity in t."""
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(15):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a_norm
    return a_new / s


def lp_alternate_steps(a, n_steps=10, time_budget=60.0):
    """Run a few LP alternating steps."""
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        g_lp = solve_lp_quick(best_a)
        if g_lp is None:
            break

        a_new = convex_line_search_blend(best_a, g_lp)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-start accelerated smooth-max GD + peak-targeted + LP + multi-n.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 40.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={len(parent)}): obj={best_score:.10f}")
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 800

    # === Phase 1: Multi-start exploration ===
    print("\n=== Phase 1: Multi-start Exploration ===")
    inits = []

    if best_seq is not None:
        inits.append(("parent", best_seq.copy()))

    inits.append(("uniform", np.ones(n_base) / n_base))

    # Gaussian
    x = np.linspace(-1, 1, n_base)
    gauss = np.exp(-x ** 2 * 1.5)
    inits.append(("gauss15", gauss / np.sum(gauss)))

    gauss2 = np.exp(-x ** 2 * 0.5)
    inits.append(("gauss05", gauss2 / np.sum(gauss2)))

    # Cosine / Hann
    hann = 0.5 * (1 + np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 0)
    inits.append(("hann", hann / np.sum(hann)))

    # Triangular
    half = n_base // 2
    tri = np.concatenate([np.linspace(0.1, 1, half), np.linspace(1, 0.1, n_base - half)])
    inits.append(("triangle", tri / np.sum(tri)))

    # Smoothed random
    r = rng.random(n_base)
    kernel = np.ones(15) / 15
    r_s = np.convolve(r, kernel, mode='same')
    inits.append(("rand_smooth", np.maximum(r_s, 0.01) / np.sum(r_s)))

    # Different n uniforms
    for n_t in [700, 900, 1000]:
        if n_t != n_base:
            inits.append((f"uniform_{n_t}", np.ones(n_t) / n_t))

    phase1_total = min(350.0, (deadline - start_time) * 0.25)
    per_init = phase1_total / len(inits)

    for name, init in inits:
        remaining = deadline - time.monotonic()
        if remaining < 180:
            break
        alloc = min(per_init, 70.0)
        if alloc < 10:
            continue

        print(f"  [{name}] n={len(init)}, budget={alloc:.0f}s")
        res = accelerated_smooth_max(init.copy(), time_budget=alloc,
                                     T_init=0.004, T_final=0.00008,
                                     max_iters=4000, seed=hash(name) % 10000)
        obj = compute_obj(res)
        print(f"       obj={obj:.10f}")

        if obj < best_score:
            best_score = obj
            best_seq = res.copy()
            print(f"       ** New best! **")

    n_best = len(best_seq)
    print(f"\nPhase 1 done: n={n_best}, score={best_score:.10f}, elapsed={time.monotonic()-start_time:.0f}s")

    # === Phase 2: Intensive optimization ===
    print("\n=== Phase 2: Intensive Smooth-Max GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        phase2_budget = min(300.0, remaining * 0.3)
        a = best_seq.copy()
        res2 = accelerated_smooth_max(a, time_budget=phase2_budget,
                                      T_init=0.0004, T_final=0.00002,
                                      max_iters=10000, verbose=True, seed=12345)
        obj2 = compute_obj(res2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = res2.copy()
            print(f"  Phase 2 improved: obj={best_score:.10f}")
        else:
            print(f"  Phase 2: obj={obj2:.10f} (no improvement)")

    # === Phase 3: LP alternating ===
    print("\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        phase3_budget = min(120.0, remaining * 0.12)
        a_lp = lp_alternate_steps(best_seq.copy(), n_steps=15, time_budget=phase3_budget)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 3 improved: obj={best_score:.10f}")
        else:
            print(f"  Phase 3: obj={obj_lp:.10f} (no improvement)")

    # === Phase 4: Peak-targeted refinement ===
    print("\n=== Phase 4: Peak-Targeted Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        phase4_budget = min(150.0, remaining * 0.15)
        a_pt = peak_targeted_refine(best_seq.copy(), time_budget=phase4_budget, verbose=True)
        obj_pt = compute_obj(a_pt)
        if obj_pt < best_score:
            best_score = obj_pt
            best_seq = a_pt.copy()
            print(f"  Phase 4 improved: obj={best_score:.10f}")
        else:
            print(f"  Phase 4: obj={obj_pt:.10f} (no improvement)")

    # === Phase 5: Multi-n exploration ===
    print("\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(500, n_best - 100), max(500, n_best - 50), max(500, n_best - 25),
        n_best + 25, n_best + 50, n_best + 100,
        600, 650, 700, 750, 800, 850, 900, 950, 1000, 1050, 1100
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1200]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(25.0, remaining * 0.08)
        a_opt = accelerated_smooth_max(a_try, time_budget=alloc,
                                       T_init=0.002, T_final=0.00005, max_iters=2000)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score + 0.002 and remaining > 25:
            a_opt = fast_coord_descent(a_opt, time_budget=min(12.0, remaining * 0.15))
            obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 6: Final polish ===
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = best_seq.copy()

        # Low-T smooth max
        if remaining > 30:
            alloc = min(50.0, remaining * 0.25)
            a_sm = accelerated_smooth_max(a_final, time_budget=alloc,
                                          T_init=0.00008, T_final=0.00001,
                                          max_iters=3000, seed=999)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                a_final = a_sm.copy()
            print(f"  Final smooth-max: obj={best_score:.10f}")

        # Coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(40.0, remaining * 0.3)
            a_cd = fast_coord_descent(a_final, time_budget=alloc, verbose=True)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.10f}")

        # Final LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            g_lp = solve_lp_quick(best_seq)
            if g_lp is not None:
                a_new = convex_line_search_blend(best_seq, g_lp)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                print(f"  Final LP: obj={best_score:.10f}")

            # One more peak-targeted
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_pt2 = peak_targeted_refine(best_seq.copy(), time_budget=min(10.0, remaining * 0.3))
                obj_pt2 = compute_obj(a_pt2)
                if obj_pt2 < best_score:
                    best_score = obj_pt2
                    best_seq = a_pt2.copy()
                print(f"  Final peak: obj={best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
