```python
    while size < 2 * n - 1:

    fa = rfft(a, size)
```

```python
def mirror_descent_step(a, k_star, eta):
    """One step of mirror descent on the simplex."""
    n = len(a)
    # Subgradient: sg_j = a_{k*-j} + a_{j-k*} (with 0 for out of range)
    sg = np.zeros(n)
    for j in range(n):
        val = 0.0
        idx1 = k_star - j
        if 0 <= idx1 < n:
            val += a[idx1]
        idx2 = j - k_star
        if 0 <= idx2 < n:
            val += a[idx2]
        sg[j] = val
    # Mirror descent update
    logits = np.log(a + 1e-30) - eta * sg
    logits -= np.max(logits)  # for numerical stability
    new_a = np.exp(logits)
    new_a /= np.sum(new_a)
    return new_a
```

```python
import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_subgrad(a, k_star):
    """Compute subgradient of conv(a,a)_{k_star} w.r.t. a.
    sg[j] = 2*a[k_star - j] if 0 <= k_star - j < n, else 0."""
    n = len(a)
    sg = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        sg[lo:hi+1] = 2.0 * a[k_star - hi : k_star - lo + 1][::-1]
    return sg

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_inner(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt)."""
    n = len(a)
    A = build_conv_matrix(a)

    # Try CVXPY with multiple solvers
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ z <= M, cp.sum(z) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    return z_val, float(M.value)
        except Exception:
            continue

    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n])
    except Exception:
        pass

    return None, None

    n = len(a)

        # 2-cycle detection

                a_break = np.maximum(a_break, 0)

                if obj_break < best_obj - 1e-12:

                    best_a = a_break.copy()
                a = a_break
                result = solve_lp_inner(a)

                prev_a = None
                prev_prev_a = None

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use cubic interpolation for smoother resampling
    from scipy.interpolate import PchipInterpolator
    try:
        interp = PchipInterpolator(x_old, a)
        a_new = interp(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent_polish(a, time_budget=30.0):
    """Polish using coordinate descent: try small perturbations of individual coordinates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget:
        improved = False
        for _ in range(min(50, (time_budget - (time.monotonic() - start)) // 0.01)):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            for frac in [0.01, 0.03, 0.05, 0.1, -0.01, -0.03, -0.05]:
                delta = frac * a[i]
                if a[i] + delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try /= np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        if not improved:
            # Try pairwise
            for _ in range(20):
                if time.monotonic() - start > time_budget:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * min(a[i], a[j]) * 0.5
                    if delta >= 0.5 * min(a[i], a[j]):
                        continue
                    a_try = a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    a_try /= np.sum(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
            if not improved:
                break

    return best_a

        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 1 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 MD: obj={obj:.8f}")

    # === Phase 2: LP Alternating (250s) ===
    print(f"\n=== Phase 2: LP Alternating (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(250.0, remaining * 0.25)
    if alloc > 30:
        a_new = lp_alternating(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 LP: obj={obj:.8f}")

    # === Phase 3: Multi-n Exploration (350s) ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_candidates = sorted(set([
        n_parent - 100, n_parent - 50, n_parent - 200,
        n_parent + 50, n_parent + 100, n_parent + 200, n_parent + 400,
        n_parent + 600, n_parent + 800,
        400, 500, 600, 700, 800, 900, 1000, 1100, 1200, 1300, 1400, 1500,
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 2500]
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample_sequence(best_seq, n_try)
        # Quick mirror descent
        alloc_md = min(20.0, remaining * 0.12)
        a_opt = mirror_descent_optimize(a_try, time_budget=alloc_md)
        obj_try = compute_obj(a_opt)

        # If promising, do LP
        if obj_try < best_score + 0.01 and remaining > 50:
            alloc_lp = min(30.0, remaining * 0.15)
            a_lp = lp_alternating(a_opt, time_budget=alloc_lp)
            obj_try = compute_obj(a_lp)
            a_opt = a_lp

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Targeted LP on best (200s) ===
    print(f"\n=== Phase 4: Targeted LP (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(200.0, remaining * 0.25)
        a_new = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 4 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4 LP: obj={obj:.8f}")

        # Mirror descent follow-up
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(80.0, remaining * 0.2)
            a_new = mirror_descent_optimize(best_seq, time_budget=alloc)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  MD follow-up: obj={best_score:.8f} **")

    # === Phase 5: More Multi-n with best (150s) ===
    print(f"\n=== Phase 5: More Multi-n ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        n_best = len(best_seq)
        nearby_ns = [n_best + d for d in [-30, -20, -10, 10, 20, 30, 50, -50] if n_best + d >= 300]
        for n_try in nearby_ns:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_try = resample_sequence(best_seq, n_try)
            alloc_md = min(15.0, remaining * 0.1)
            a_opt = mirror_descent_optimize(a_try, time_budget=alloc_md)
            obj_try = compute_obj(a_opt)
            if obj_try < best_score + 0.005 and remaining > 30:
                alloc_lp = min(20.0, remaining * 0.15)
                a_lp = lp_alternating(a_opt, time_budget=alloc_lp)
                obj_try = compute_obj(a_lp)
                a_opt = a_lp
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Coordinate Descent Polish (60s) ===
    print(f"\n=== Phase 6: CD Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(60.0, remaining * 0.15)
        a_new = coordinate_descent_polish(best_seq, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  CD Polish: obj={best_score:.8f} **")
        else:
            print(f"  CD Polish: obj={obj:.8f}")

    # === Phase 7: Random Restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(5, int(remaining // 35))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            if r % 3 == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.03 + 0.01 * r, n_best))
            elif r % 3 == 1:
                a_r = best_seq.copy()
                n_blocks = max(5, n_best // 100)
                block_size = n_best // n_blocks
                blocks = rng.choice(n_blocks, min(3, n_blocks), replace=False)
                for b in blocks:
                    si = int(b) * block_size
                    ei = si + block_size
                    a_r[si:ei] *= rng.uniform(0.6, 1.4)
            else:
                a_r = best_seq.copy()
                n_sparse = max(5, n_best // 20)
                idx = rng.choice(n_best, n_sparse, replace=False)
                a_r[idx] *= rng.uniform(0.4, 1.8, n_sparse)

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr

            alloc = min(remaining * 0.3, 30.0)
            a_opt = mirror_descent_optimize(a_r, time_budget=alloc)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 8: Final LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 25.0)
        a_final = lp_alternating(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
