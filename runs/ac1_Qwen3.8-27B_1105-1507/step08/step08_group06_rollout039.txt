    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A

def lp_response(a, time_budget=5.0):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None
    a = a / s
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > time_budget:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs
        except Exception:
            continue
    
    # Fallback
    try:
        from scipy.optimize import linprog
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True, 'time_limit': time_budget})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except Exception:
        pass
    return None

def lp_annealing(a, time_budget=300.0, verbose=False):
    """LP alternating with simulated annealing."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    current_temp = 0.01
    base_temp = 0.01
    cool_rate = 0.97
    
    momentum = np.zeros(n)
    beta_mom = 0.5
    
    while time.monotonic() - start < time_budget:
        g_lp = lp_response(a)
        if g_lp is None:
            break
        
        # Try different step sizes with annealing
        direction = g_lp - a
        momentum = beta_mom * momentum + (1 - beta_mom) * direction
        
        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()
        
        t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for t in t_candidates:
            a_try = a + t * momentum
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            
            # Simulated annealing: accept worse solutions with probability exp(-delta/temp)
            if obj_try < best_t_obj or (rng.random() < np.exp(-(obj_try - best_t_obj) / (current_temp * best_t_obj + 1e-10))):
                if obj_try < best_t_obj or (obj_try - best_t_obj) / (best_t_obj + 1e-10) < current_temp:
                    best_t_obj = obj_try
                    best_t_val = t
                    best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
        else:
            # Even if not better than global best, update current if annealing accepts
            if best_a_local is not None and best_t_obj < compute_obj(a) + current_temp * best_obj:
                a = best_a_local.copy()
        
        current_temp *= cool_rate
        
        # Perturbation if stalled
        if int((time.monotonic() - start) * 2) % 20 == 0 and current_temp < 0.005:
            scale = 0.02
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            current_temp = base_temp
    
    return best_a

def gradient_descent(a, time_budget=100.0, verbose=False):
    """Adam gradient descent on the peak of the convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adam parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    while time.monotonic() - start < time_budget:
        t += 1
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Get active set
        threshold = max_val * 0.95
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = [int(np.argmax(conv))]
        if len(active) > 20:
            active = np.argsort(conv)[::-1][:20]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_max = min(n - 1, k)
            i_min = max(0, k - n + 1)
            indices = np.arange(i_min, i_max + 1)
            grad[indices] += a[k - indices]
        grad /= len(active)
        
        # Center gradient (to maintain sum=1)
        grad = grad - np.mean(grad)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Decay learning rate
        if t % 500 == 0:
            lr *= 0.95
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def solve_lp(a, time_limit=10.0):

    n = len(a)

        return None

    # Try CVXPY first

                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

"""
Optimized step function search v2: LP-annealing with aggressive length sweep,
diverse initializations, dual-guided updates, and adaptive polish.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key improvements: broader length exploration, dual-variable gradients,
multi-round deep optimization, and adaptive perturbation schedules.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, time_limit=10.0, need_dual=False):
    """
    Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0.
    Returns (g, y_dual) or (None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)
    m = 2 * n - 1

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            c1 = A @ g <= M
            c2 = cp.sum(g) == 1
            prob = cp.Problem(cp.Minimize(M), constraints=[c1, c2])
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > time_limit * 2:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    y_dual = None
                    if need_dual:
                        try:
                            if c1.dual_value is not None:
                                y_dual = np.maximum(np.asarray(c1.dual_value).flatten(), 0)
                                if np.sum(y_dual) < 1e-12:
                                    y_dual = None
                        except Exception:
                            y_dual = None
                    return g_val, y_dual
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, None
    except Exception:
        pass
    return None, None


def lp_anneal(a, time_budget=200.0, verbose=False, seed=42, use_dual=False):
    """
    LP alternating with simulated annealing, momentum, and optional dual guidance.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    momentum = np.zeros(n)
    beta_mom = 0.4
    temp = 0.004
    base_temp = 0.004
    cool_rate = 0.997
    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        need_dual = use_dual and (iter_count % 3 == 0)
        g_lp, y_dual = solve_lp(a, time_limit=8.0, need_dual=need_dual)
        if g_lp is None:
            break

        iter_count += 1
        lp_direction = g_lp - a

        # Build combined direction
        if y_dual is not None and np.sum(y_dual) > 1e-10:
            y_dual = y_dual / np.sum(y_dual)
            A_mat = build_conv_matrix(a)
            v = A_mat.T @ y_dual
            v_norm = np.max(np.abs(v))
            if v_norm > 1e-12:
                v = v / v_norm
                v = v - np.mean(v)
                direction = 0.65 * lp_direction + 0.35 * v
            else:
                direction = lp_direction
        else:
            direction = lp_direction

        momentum = beta_mom * momentum + (1 - beta_mom) * direction

        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = None

        t_candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for t in t_candidates:
            a_try = a + t * momentum
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_a_local is not None:
            if best_t_obj < best_obj - 1e-14:
                best_obj = best_t_obj
                best_a = best_a_local.copy()
                a = best_a
                temp = base_temp
                stall = 0
            else:
                stall += 1
                delta = best_t_obj - best_obj
                accept_prob = np.exp(-delta / (temp * (best_obj + 1e-10)))
                if rng.random() < accept_prob:
                    a = best_a_local.copy()
                else:
                    momentum *= 0.8
                temp *= cool_rate

            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP-ann iter {iter_count}: obj={best_obj:.8f}, stall={stall}, t={elapsed:.0f}s")

        # Periodic perturbation to escape local minima
        if stall >= 30 or iter_count % 80 == 0:
            scale = 0.01 + 0.005 * min(stall // 10, 6)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            momentum = np.zeros(n)
            temp = base_temp
            stall = 0

    return best_a


def subgradient_polish(a, time_budget=60.0, verbose=False):
    """Nesterov-accelerated subgradient method targeting active peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()
    y = a.copy()
    momentum = np.zeros(n)
    beta = 0.8
    lr = 0.02
    minibatch = 15

    for it in range(30000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(x)
        max_val = float(np.max(conv))
        threshold = max_val * 0.85
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = [int(np.argmax(conv))]
        if len(active) > minibatch:
            active = np.argsort(conv)[::-1][:minibatch]

        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_max = min(n - 1, k)
            i_min = max(0, k - n + 1)
            indices = np.arange(i_min, i_max + 1)
            grad[indices] += a[k - indices]
        grad /= len(active)
        grad = grad - np.mean(grad)

        momentum = beta * momentum - lr * grad
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
            s_new = 1.0
        y_new = y_new / s_new

        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        x = y_new + (1 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        y = y_new
        obj = compute_obj(y)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y.copy()

        if it % 1000 == 999:
            lr *= 0.92

    return best_a


def coordinate_refine(a, time_budget=30.0):

    n = len(a)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_diverse_init(kind, n, rng):
    """Generate diverse initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        s = np.sum(a)
        return a / s if s > 0 else np.ones(n) / n
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n, dtype=np.float64)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'bimodal':
        a = np.ones(n) * 0.3
        a[n // 4] = 1.5
        a[3 * n // 4] = 1.5
        kernel = np.ones(max(3, n // 50)) / max(3, n // 50)
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 8)
        a[:edge] = 0.2
        a[-edge:] = 0.2
        return a / np.sum(a)
    else:
        return np.ones(n) / n

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
