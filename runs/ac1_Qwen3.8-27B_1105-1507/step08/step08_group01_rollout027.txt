def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax(z):
    """Numerically stable softmax."""
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    return exp_z / np.sum(exp_z)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A_mat = build_conv_matrix(a)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_mat
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def convex_line_search(a, g, n_evals=12):
    """Find optimal t for a_new = (1-t)*a + t*g."""
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)
    
    conv_aa = compute_conv(a_norm)
    # conv_ag[k] = sum_j a_norm[j] * g_norm[k-j]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a_norm, size)
    fg = rfft(g_norm, size)
    conv_ag = irfft(fa * fg, size)[:2*n-1]
    conv_gg = compute_conv(g_norm)
    
    def obj_at_t(t):
        val = (1-t)**2 * conv_aa + 2*t*(1-t) * conv_ag + t**2 * conv_gg
        return 2.0 * n * float(np.max(val))
    
    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi-lo)/3.0
        m2 = hi - (hi-lo)/3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t_opt = (lo+hi)/2.0
    a_new = (1-t_opt)*a_norm + t_opt*g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    return t_opt, a_new / s
