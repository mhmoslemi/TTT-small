    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    nr = 2 * n - 1
    A = np.zeros((nr, n + 1))
    for j in range(n):
        ks, ke = j, min(2*n-2, j+n-1)
        A[ks:ke+1, j] = a_n[:ke-ks+1]

        g = np.maximum(res.x[:n], 0)

        if gs > 1e-10:

Enhanced LP alternating optimizer with dual-guided perturbation,
adaptive line search, multi-n exploration, and diverse restarts
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g, dual_array) or (None, None)."""
    n = len(a)
    a_n = a / float(np.sum(a))
    nr = 2 * n - 1
    A = np.zeros((nr, n + 1))
    for j in range(n):
        ks = j
        ke = min(2 * n - 2, j + n - 1)
        A[ks:ke + 1, j] = a_n[:ke - ks + 1]
    A[:, n] = -1.0
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    res = linprog(c, A_ub=A, b_ub=np.zeros(nr), A_eq=A_eq, b_eq=[1.0],
                  bounds=[(0, None)] * (n + 1), method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g))
        if gs > 1e-10:
            dual_vals = np.zeros(nr)
            try:
                if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                    dual_vals = np.abs(np.asarray(res.ineqlin.marginals))
            except Exception:
                pass
            return g / gs, dual_vals
    return None, None


def blend_search(a, g, n_try=40):
    """Line search for optimal blend parameter t in (0,1]."""
    best_obj = np.inf
    best_a = a.copy()
    ts = np.concatenate([np.linspace(0.01, 0.3, 10),
                         np.linspace(0.3, 0.7, 15),
                         np.linspace(0.7, 1.0, 15)])
    for t in ts:
        at = (1.0 - t) * a + t * g
        at = np.maximum(at, 0)
        s = float(np.sum(at))
        if s < 1e-10:
            continue
        at = at / s
        o = compute_obj(at)
        if o < best_obj:
            best_obj = o
            best_a = at.copy()
    return best_a, best_obj


def perturb_dual(a, dual_vals, rng, intensity=0.1):
    """Perturb using LP dual information to target expensive convolution positions."""
    n = len(a)
    penalty = np.zeros(n)
    for k in range(len(dual_vals)):
        dk = dual_vals[k]
        if dk < 1e-10:
            continue
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            idx = k - j
            if 0 <= idx < n:
                penalty[j] += dk * a[j] * a[idx]
    pm = float(np.max(penalty))
    if pm > 1e-10:
        penalty = penalty / pm
    a_new = a * np.exp(-intensity * penalty)
    a_new = a_new + rng.standard_normal(n) * intensity * 0.08 * a
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def perturb_peak(a, rng, intensity=0.1):
    """Perturb targeting the peak of the self-convolution."""
    n = len(a)
    conv = compute_conv(a)
    peak = float(np.max(conv))
    if peak < 1e-10:
        return a.copy()
    penalty = np.zeros(n)
    thresh = 0.8 * peak
    for k in range(len(conv)):
        if conv[k] < thresh:
            continue
        k = int(k)
        w = conv[k] / peak
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            idx = k - j
            if 0 <= idx < n:
                penalty[j] += w * a[j] * a[idx]
    pm = float(np.max(penalty))
    if pm > 1e-10:
        penalty = penalty / pm
    a_new = a * np.exp(-intensity * penalty)
    a_new = a_new + rng.standard_normal(n) * intensity * 0.1 * a
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))

def lp_alternate(a, time_budget, verbose=False, seed=42):
    """LP alternating with line search, dual-guided perturbation, and stall detection."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(seed)
    iteration = 0
    last_dual = None

    while time.monotonic() - start < time_budget:
        iteration += 1
        g, dual_vals = solve_lp(a)
        if g is None:
            break

        a_new, obj_new = blend_search(a, g, n_try=40)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            last_dual = dual_vals
            if verbose and iteration % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it={iteration} obj={best_obj:.8f} elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall == 1:
                # Slight nudge in direction of g even if not improvement
                a = 0.95 * a + 0.05 * g
                a = np.maximum(a, 0)
                s2 = float(np.sum(a))
                if s2 > 1e-10:

            elif stall >= 6:

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng):

        a = np.exp(-50 * (x - c1) ** 2) + np.exp(-50 * (x - c2) ** 2)
    elif choice == 3:
        a = 1.0 + rng.standard_normal(n) * 0.4
    elif choice == 4:
        a = x ** rng.uniform(0.3, 3.0)
    elif choice == 5:
        a = 1.0 / (1.0 + np.exp(-rng.uniform(4, 12) * (x - 0.5)))
    elif choice == 6:
        a = np.abs(np.sin(np.pi * x)) ** rng.uniform(0.5, 3.0)
    else:
        a = np.ones(n)
        for _ in range(3):
            center = float(rng.integers(0, n)) / n
            width = rng.uniform(0.05, 0.3)
            a *= 1.0 + rng.standard_normal() * np.exp(-(x - center) ** 2 / (2 * width ** 2))
    a = np.maximum(a, 0.001)
    return a / float(np.sum(a))


def coordinate_polish(a, time_budget=20.0):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    order = rng.permutation(n)
    it = 0

    while time.monotonic() - start < time_budget:
        j = int(order[it % n])
        a_j = a[j]
        candidates = [0.0, a_j * 0.3, a_j * 0.6, a_j, a_j * 1.4, a_j * 2.0, a_j * 3.0, a_j * 0.1]
        best_local = best_obj
        best_val = a_j
        for val in candidates:
            val = max(0.0, val)
            a[j] = val
            s_new = float(np.sum(a))
            if s_new < 1e-10:
                a[j] = a_j
                continue
            obj_try = compute_obj(a / s_new)
            if obj_try < best_local:
                best_local = obj_try
                best_val = val
        a[j] = best_val
        s_new = float(np.sum(a))
        if s_new > 1e-10:
            a = a / s_new
            if best_local < best_obj:
                best_obj = best_local
        it += 1
        if it % n == 0:
            obj_check = compute_obj(a)
            if obj_check < best_obj:
                best_obj = obj_check

    return a

        # Choose perturbation type
        if i % 3 == 0:
            a_p = perturb_peak(best_a, rng, intensity=0.08 + 0.04 * (i % 5))
        elif i % 3 == 1:
            a_p = random_perturb(best_a, rng, scale=0.1 + 0.05 * (i % 4))
        else:
            # Asymmetric shift: move mass from left to right (or vice versa)
            a_p = best_a.copy()
            shift = int(rng.integers(1, max(2, len(a_p) // 20)))
            direction = 1 if rng.random() > 0.5 else -1
            a_p = np.roll(a_p, direction * shift)
            a_p = np.maximum(a_p, 0)
            s_p = float(np.sum(a_p))
            if s_p > 1e-10:
                a_p = a_p / s_p
            else:
                a_p = best_a.copy()

        alloc = min(22.0, remaining * 0.2)
        a_opt = lp_alternate(a_p, alloc, seed=seed + 1000 + i * 13)
        o = compute_obj(a_opt)
        if o < best_obj:
            best_obj = o
            best_a = a_opt.copy()
            print(f"  Pert {i}: {best_obj:.8f} **")
        else:
            print(f"  Pert {i}: {o:.8f}")

    # === Phase 3: Diverse initializations ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 3: Diverse Inits (remaining={remaining:.0f}s) ===")
    n_best = len(best_a)
    max_div = min(20, int(remaining / 25))
    for i in range(max_div):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        a_d = diverse_init(n_best, rng)
        alloc = min(22.0, remaining * 0.2)
        a_opt = lp_alternate(a_d, alloc, seed=5000 + i * 31)
        o = compute_obj(a_opt)
        if o < best_obj:
            best_obj = o
            best_a = a_opt.copy()
            print(f"  Div {i}: {best_obj:.8f} **")
        else:
            print(f"  Div {i}: {o:.8f}")

    # === Phase 4: Multi-n exploration ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 4: Multi-n (remaining={remaining:.0f}s) ===")
    n_best = len(best_a)
    n_candidates = []
    for delta in [-400, -200, -100, -50, 0, 50, 100, 200, 400, 600, 1000]:
        nc = n_best + delta
        if 200 <= nc <= 4000:
            n_candidates.append(nc)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == n_best:
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
            else:
                a_try = np.ones(n_try) / n_try
        alloc = min(35.0, remaining * 0.25)
        a_opt = lp_alternate(a_try, alloc, seed=8000 + n_try)
        o = compute_obj(a_opt)
        if o < best_obj:
            best_obj = o
            best_a = a_opt.copy()
            print(f"  n={n_try}: {best_obj:.8f} **")
        else:
            print(f"  n={n_try}: {o:.8f}")

    # === Phase 5: Final polish ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Final Polish (remaining={remaining:.0f}s) ===")
    if remaining > 40:
        alloc_lp = min(remaining * 0.45, 120.0)
        a_final = lp_alternate(best_a, alloc_lp, verbose=True, seed=99999)
        o_final = compute_obj(a_final)
        if o_final < best_obj:
            best_obj = o_final
            best_a = a_final
            print(f"  Final LP: {best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_cd = min(remaining * 0.4, 30.0)
        a_cd = coordinate_polish(best_a, alloc_cd)
        o_cd = compute_obj(a_cd)
        if o_cd < best_obj:
            best_obj = o_cd
            best_a = a_cd
            print(f"  Final Coord: {best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_last = min(remaining * 0.5, 25.0)
        a_last = lp_alternate(best_a, alloc_last, seed=12345)
        o_last = compute_obj(a_last)
        if o_last < best_obj:
            best_obj = o_last
            best_a = a_last
            print(f"  Last LP: {best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")
    return result
