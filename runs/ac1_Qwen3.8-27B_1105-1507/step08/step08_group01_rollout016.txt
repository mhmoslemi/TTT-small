"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with adaptive active sets, spectral initialization,
pairwise coordinate descent, and multi-n exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:min(2*n-1, j+n), j] = a[:min(n, 2*n-1-j)]
    return A

def simplex_project(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_lp(a, active_indices=None):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A_full = build_conv_matrix(a)
    
    if active_indices is not None:
        A = A_full[active_indices, :]
    else:
        A = A_full
    
    m = A.shape[0]
    # Variables: [g_0,...,g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_norm = g_val / s
            return g_norm, float(res.x[n])
    return None, None

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

        S = np.array([w_cum[min(i+n, len(w))] - w_cum[i] for i in range(n)])
        
        grad = c1 + y * S  # grad[i] = c1[i] + y[i]*S[i]
        
        # Scale by 2n/T for the log-sum-exp objective (times 2n)
        # Actually, the objective is 2n * (1/T) * (log_sum_exp(T*conv) - T*max) 
        # The gradient of log(sum exp(T*conv[k])) w.r.t. a[i] is:
        # = T * sum_k exp(T*conv[k]) / sum exp(T*conv[k']) * d conv[k]/da[i]
        # = T * sum_k w[k] * d conv[k]/da[i]  (where w[k] = exp(T*(conv[k]-max))/Z)

        # = 2n * grad_as_computed_above
        # (the T and 1/T cancel)
        
        grad = 2.0 * n * grad
        
        # Center the gradient (for the simplex constraint)
        grad_centered = grad - np.mean(grad)  # or grad - sum(y*grad) for weighted centering
        
        # Nesterov step
        t_curr = 1.0 / (1.0 + 0.5 * np.sqrt(1.0 + 4.0 * it * it))
        v = t_curr * v + (1.0 - t_curr) * grad_centered  # not quite standard Nesterov

        # v = momentum * v - lr * grad
        # a_new = simplex_project(a + v)

        # (Nesterov can be unstable for non-strongly-convex problems)
        
        a_new = simplex_project(y - lr * grad)
        # Update Nesterov momentum
        v = v * (it / (it + 3.0)) + (-(grad) * lr) * (1.0 - it/(it+3.0))  # this is getting complicated
        
        # Let me just use plain momentum which is simpler and works well:
        # (I'll compute v separately)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            y = a + v  # Nesterov lookahead
        else:
            lr *= 0.9
            y = a.copy()
            if lr < 1e-8:
                break
        
        # Anneal temperature
        if it > 100:

    return best_a
