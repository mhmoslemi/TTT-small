"""
Optimized solver using FFT-accelerated Nesterov subgradient with log-sum-exp smoothing,
LP alternating, peak shaving, and multi-scale n-sweeping.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def next_pow2(n):
    return 1 << (n - 1).bit_length()


def compute_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def logsumexp_grad(a, beta):
    """
    Compute the gradient of the log-sum-exp approximation of max(conv(a,a)).
    Returns (grad, lse_value) where grad has length n and lse_value is the LSE.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    size = next_pow2(2 * n - 1)
    
    # Compute convolution
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)[:2 * n - 1]
    
    # Compute log-sum-exp weights
    max_conv = np.max(conv)
    exp_conv = np.exp(beta * (conv - max_conv))
    total = np.sum(exp_conv)
    p = exp_conv / total  # weights, length 2n-1
    
    # LSE value = max_conv + log(total) / beta
    lse_val = max_conv + np.log(total) / beta
    
    # Gradient: grad[m] = 2 * sum_k p[k] * a[k-m] for valid k-m
    # This is 2 * (p cross-correlate a)[m] = 2 * ifft(FFT(p) * conj(FFT(a)))[0:n]
    fp = rfft(p, size)
    grad = 2.0 * irfft(fp * np.conj(fa), size)[:n]
    
    return grad, lse_val


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

    a = a / s
    y = a.copy()
    best_a = a.copy()

        beta = np.exp(log_beta)

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    if n > 3000:
        # Too large for LP, use heuristic
        return None, None
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=60.0, verbose=False):
    """LP alternating with 1D line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))

        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        # 1D line search
        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))

            a_try = a_try / st
            return compute_obj(a_try)

                                  options={'maxiter': 30, 'xatol': 0.01})
            best_t = res.x if res.success else last_good_t
        except Exception:
            best_t = last_good_t
        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:

            best_a = a_try.copy()
            a = best_a
            last_good_t = best_t
            stall_count = 0
            if verbose:
                print(f"        LP: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall_count += 1
            if stall_count > 15:

        conv = compute_conv(a)
        top_k = np.argsort(conv)[::-1][:n_peaks]
        improved = False
        
        for k in top_k:
            k = int(k)
            # Find contributors
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            
            # Contribution of a[m] to conv[k]
            # conv[k] = sum_{i=i_start}^{i_end} a[i] * a[k-i]
            # a[m] contributes to conv[k] through: a[m]*a[k-m] (if k-m valid) + a[k-m]*a[m] (same term)
            # Actually, a[m] appears as a[i] when i=m (if m in [i_start, i_end]) contributing a[m]*a[k-m]
            # and as a[k-i] when k-i=m, i.e., i=k-m (if k-m in [i_start, i_end]) contributing a[k-m]*a[m]
            # So total contribution of a[m] to conv[k] is:
            #   if m in [i_start, i_end] and k-m in [0, n-1]: a[m]*a[k-m]
            #   if k-m in [i_start, i_end] and m in [0, n-1]: a[k-m]*a[m]
            # If m == k-m (i.e., k = 2m), these are the same term, so contribution is a[m]^2
            # Otherwise, contribution is 2*a[m]*a[k-m] (if both terms exist)
            
            # For simplicity, let's just try random pairs
            for trial in range(5):
                src = rng.integers(0, n)
                dst = rng.integers(0, n)
                if src == dst or a[src] < 1e-8:
                    continue
                # Try moving some mass from src to dst
                for frac in [0.01, 0.03, 0.05]:
                    delta = frac * a[src]
                    a[src] -= delta
                    a[dst] += delta
                    a = np.maximum(a, 0)
                    st = float(np.sum(a))
                    if st < 1e-10:
                        a[src] += delta
                        a[dst] -= delta
                        continue
                    a = a / st
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[dst] -= delta
                        a = a / st
                        break
        
        if not improved and outer > 30:
            break
        a = best_a.copy()
        if verbose and outer % 20 == 0:
            print(f"        PeakShave iter={outer}: obj={best_obj:.8f}")

    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def generalized_cosine_window(n, alpha):
    """Generalized cosine window: (1 + alpha*cos) / (1 + alpha) type."""
    k = np.arange(n)
    w = 1.0 + alpha * np.cos(2.0 * np.pi * k / max(1, n - 1))
    w = np.maximum(w, 0)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def raised_cosine_power(n, p):
    """(sin(pi*k/(n-1))^2)^p type window."""
    k = np.arange(n)
    w = np.sin(np.pi * k / max(1, n - 1)) ** (2 * p)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw
