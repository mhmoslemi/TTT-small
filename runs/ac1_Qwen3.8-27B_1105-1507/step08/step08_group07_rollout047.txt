    conv = compute_conv(a)
    max_val = np.max(conv)
    active_lags = np.where(np.abs(conv - max_val) < 1e-8 * max_val)[0]

    n = len(a)
    grad = np.zeros(n)
    lo = max(0, m - n + 1)
    hi = min(n - 1, m)
    indices = np.arange(lo, hi + 1)
    j_indices = m - indices
    grad[indices] = 2 * a[j_indices]
    # Fix the center point if m is even
    center = m // 2
    if m % 2 == 0 and 0 <= center < n:
        grad[center] = a[center]
    return grad

"""
Hybrid LP-alternating + fast coordinate descent optimizer for minimizing
2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key innovations over prior approach:
1. Fast 1D coordinate evaluation exploiting sparsity of affected lags
2. Convex-structured golden section for exact coordinate optimization
3. Hybrid LP + mirror descent for better global exploration
4. Systematic n-sweep with coordinated optimization
5. Peak-targeted local search for final squeeze
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt, dual_y) or (None, None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])

    dual_y = None
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    try:
                        dual_y = np.maximum(np.array(c1.dual_value), 0)
                    except:
                        dual_y = None
                    return z_val, float(M.value), dual_y
        except:
            continue

    # Fallback
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n]), None
    except:
        pass

    return None, None, None


def mirror_descent_step(a, eta, n_active=3):
    """
    One step of mirror descent with entropy regularizer.
    Uses subgradient from top n_active lags.
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = np.max(conv)
    if max_val < 1e-15:
        return a.copy()
    
    # Find active lags (top n_active)
    top_idx = np.argsort(conv)[-n_active:]
    
    # Compute subgradient: average over active lags
    grad = np.zeros(n)
    for m in top_idx:
        m = int(m)
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if lo > hi:
            continue
        indices = np.arange(lo, hi + 1)
        j_indices = m - indices
        contrib = 2.0 * a[j_indices]
        # Fix center for even m
        center = m // 2
        if m % 2 == 0 and lo <= center <= hi:
            contrib[center - lo] = a[center]
        grad[lo:hi + 1] += contrib
    
    grad /= n_active
    
    # Mirror descent: a_new = a * exp(-eta * grad) / sum
    log_update = np.exp(-eta * grad)
    a_new = a * log_update
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    return a_new


def lp_hybrid_alternating(a, time_budget=200.0, verbose=False):

    - Cycle detection (2-cycle and longer)
    - Adaptive step size
    - Dual-guided perturbations

    if s < 1e-10:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    history = []
    stall = 0
    eta = 1.0 / np.sqrt(n)
    use_md = False
    md_count = 0

    while time.monotonic() - start < time_budget:

        if stall < 5 or md_count % 3 == 0:
            result = solve_lp(a)
            if result[0] is None:
                break
            g_lp, M_lp, dual_y = result
            
            # Try mixing ratios
            best_cand_obj = best_obj
            best_cand = a.copy()
            for mix in [0.3, 0.5, 0.7, 1.0, 0.2, 0.4]:
                a_try = (1.0 - mix) * a + mix * g_lp
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_cand_obj - 1e-14:
                    best_cand_obj = obj_try
                    best_cand = a_try.copy()
        else:

            a_try = mirror_descent_step(a, eta)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_cand_obj = obj_try
                best_cand = a_try.copy()
            else:

                a_try = mirror_descent_step(a, eta * 0.5)

                    best_cand_obj = obj_try

                else:
                    best_cand_obj = best_obj
                    best_cand = a.copy()

                    break

            history.append(a.copy())
            if len(history) > 10:

                s2 = float(np.sum(a))

                best_a = a.copy()
                best_obj = compute_obj(a)
                history = [a.copy()]
                stall = 0
                eta = 1.0 / np.sqrt(n)

def compute_base_conv(a, i):
    """Compute base convolution (with a[i] removed) and unaffected max."""
    n = len(a)
    ai = a[i]
    if ai < 1e-15:
        conv = compute_conv(a)
        return conv, float(np.max(conv))
    
    # base_conv[m] = conv[m] - contribution of a[i]
    conv = compute_conv(a)
    base_conv = conv.copy()
    
    # Affected lags: m where 0 <= m-i < n
    m_lo = max(0, i)
    m_hi = min(2 * n - 2, i + n - 1)
    
    unaffected_max = 0.0
    for m in range(0, m_lo):
        unaffected_max = max(unaffected_max, base_conv[m])
    for m in range(m_hi + 1, 2 * n - 1):
        unaffected_max = max(unaffected_max, base_conv[m])
    
    for m in range(m_lo, m_hi + 1):
        j = m - i
        if j < 0 or j >= n:
            continue
        if i == j:
            base_conv[m] -= ai * ai
        else:
            base_conv[m] -= 2.0 * ai * a[j]
    
    return base_conv, unaffected_max


def obj_1d_fast(a, i, x, base_conv, unaffected_max, a_sum):
    """Fast evaluation of objective when a[i]=x."""
    n = len(a)
    S = a_sum - a[i] + x
    if S < 1e-10:
        return np.inf
    
    m_lo = max(0, i)
    m_hi = min(2 * n - 2, i + n - 1)
    
    max_conv = unaffected_max
    for m in range(m_lo, m_hi + 1):
        j = m - i
        if i == j:
            val = base_conv[m] + x * x
        else:
            val = base_conv[m] + 2.0 * x * a[j]
        if val > max_conv:
            max_conv = val
    
    if max_conv < 0:
        return np.inf
    return 2.0 * n * max_conv / (S * S)


def coordinate_descent_polish(a, time_budget=100.0, verbose=False):
    """

    no_improve_epochs = 0

        if time.monotonic() - start > time_budget:
            break
        
        epoch_improved = False
        epoch_start_obj = best_obj
        
        for idx in range(n):
            if time.monotonic() - start > time_budget:
                break
            i = order[idx]
            
            # Golden section search for optimal a[i]
            # Bracket: try expanding if needed
            ai = a[i]
            lo = 0.0
            hi = max(ai * 3.0, 1e-4)

            # Try a few test points to bracket the minimum
            test_points = [0.0, ai * 0.25, ai * 0.5, ai, ai * 1.5, ai * 2.0, ai * 4.0]
            test_points = [max(0.0, min(tp, 1.0)) for tp in test_points]

            best_test_x = ai
            for tp in test_points:

                if obj_tp < best_test_obj:
                    best_test_obj = obj_tp
                    best_test_x = tp
            lo = 0.0
            hi = max(best_test_x * 2.0 + 0.001, 0.01)
            
            # Golden section
            a1 = hi - gr * (hi - lo)
            a2 = lo + gr * (hi - lo)
            f1 = obj_1d_fast(a, i, a1, base_conv, unaff_max, a_sum)
            f2 = obj_1d_fast(a, i, a2, base_conv, unaff_max, a_sum)
            
            for _ in range(35):
                if f1 < f2:
                    hi = a2
                    a2 = a1
                    f2 = f1
                    a1 = hi - gr * (hi - lo)
                    f1 = obj_1d_fast(a, i, a1, base_conv, unaff_max, a_sum)
                else:
                    lo = a1
                    a1 = a2
                    f1 = f2
                    a2 = lo + gr * (hi - lo)
                    f2 = obj_1d_fast(a, i, a2, base_conv, unaff_max, a_sum)
            
            best_x = (lo + hi) / 2.0
            best_x_obj = obj_1d_fast(a, i, best_x, base_conv, unaff_max, a_sum)
            
            if best_x_obj < best_obj - 1e-13 and best_x > 1e-12:
                a[i] = best_x
                a_sum = float(np.sum(a))
                a = a / a_sum
                a_sum = 1.0
                best_obj = compute_obj(a)
                epoch_improved = True
        
        if verbose and epoch % 5 == 0:

            print(f"    CoordDesc epoch {epoch}: obj={best_obj:.8f}, t={elapsed:.0f}s")
        
        if not epoch_improved:
            no_improve_epochs += 1
            if no_improve_epochs >= 3:
                break
        else:
            no_improve_epochs = 0
        
        # Shuffle order each epoch
        order = rng.permutation(n)
    
    return a, best_obj


def pairwise_2d_polish(a, time_budget=60.0):
    """
    2D coordinate descent: optimize pairs of coordinates simultaneously.
    Uses Nelder-Mead on 2D subproblems.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)
    
    attempts = 0
    max_attempts = int(time_budget * 100)  # ~100 per second budget
    
    while time.monotonic() - start < time_budget and attempts < max_attempts:
        attempts += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        ai, aj = a[i], a[j]
        
        # Try a grid of (di, dj) perturbations
        best_local = best_obj
        best_di, best_dj = 0.0, 0.0
        
        for scale in [0.01, 0.02, 0.05, 0.1]:
            for fi in [-1, 0, 1]:
                for fj in [-1, 0, 1]:
                    if fi == 0 and fj == 0:
                        continue
                    di = fi * scale * ai
                    dj = fj * scale * aj
                    if ai + di < 0 or aj + dj < 0:
                        continue
                    a_try = a.copy()
                    a_try[i] += di
                    a_try[j] += dj
                    a_try = np.maximum(a_try, 0)
                    st = float(np.sum(a_try))
                    if st < 1e-10:
                        continue
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_local - 1e-13:
                        best_local = obj_try
                        best_di, best_dj = di, dj
        
        if best_local < best_obj - 1e-13:
            a[i] += best_di
            a[j] += best_dj
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_local
    
    return a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def peak_targeted_search(a, time_budget=60.0):
    """
    Directly target the peak lags: find moves that specifically reduce
    the highest autocorrelation lags.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(555)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = np.max(conv)
        # Find top lags
        top_lags = np.argsort(conv)[-5:]
        
        # For each top lag, find the coordinate that contributes most
        # and try to reduce it
        improved = False
        for m in top_lags:
            m = int(m)
            # Find i that maximizes a[i]*a[m-i] (contribution to lag m)
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            if lo > hi:
                continue
            
            best_contrib = 0
            best_i = -1
            for i in range(lo, hi + 1):
                j = m - i
                if 0 <= j < n:
                    if i == j:
                        c = a[i] * a[i]
                    else:
                        c = 2.0 * a[i] * a[j]
                    if c > best_contrib:
                        best_contrib = c
                        best_i = i
            
            if best_i >= 0 and best_contrib > 0.1 * max_val:
                # Try reducing a[best_i] and increasing a neighbor
                j = m - best_i
                for frac in [0.05, 0.1, 0.15, 0.2]:
                    delta = frac * a[best_i]
                    a_try = a.copy()
                    a_try[best_i] -= delta
                    if a_try[best_i] < 0:
                        continue
                    # Distribute to random neighbors
                    target = rng.integers(0, n)
                    a_try[target] += delta
                    a_try = np.maximum(a_try, 0)
                    st = float(np.sum(a_try))
                    if st < 1e-10:
                        continue
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        a = a_try.copy()
                        improved = True
                        break
        
        if not improved:
            # Random perturbation to escape
            noise = rng.normal(0, 0.02, n)
            a = a * np.exp(noise)
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            # Accept if not too bad
            obj_now = compute_obj(a)
            if obj_now > best_obj * 1.05:
                a = a * np.exp(-noise)
                a = np.maximum(a, 0)
                a = a / np.sum(a)
    
    return a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP hybrid + coordinate descent + n-sweep + peak targeting.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        # Structured initializations
        best_init = None
        best_init_obj = np.inf
        for n_init in [400, 600, 800, 1000, 1200, 1500]:
            t = np.linspace(0, 1, n_init)
            windows = [
                0.5 + 0.5 * np.cos(2 * np.pi * t),
                np.sin(np.pi * t) ** 2,
                1.0 - 0.3 * np.cos(4 * np.pi * t),
                np.exp(-2.0 * (t - 0.5) ** 2 / (0.1 * 0.1)),
            ]
            for w in windows:
                a_init = np.maximum(w, 0.01)
                a_init = a_init / np.sum(a_init)
                obj_init = compute_obj(a_init)
                if obj_init < best_init_obj:
                    best_init_obj = obj_init
                    best_init = a_init.copy()
        best_seq = best_init if best_init is not None else np.ones(800) / 800
        best_score = best_init_obj if best_init is not None else compute_obj(best_seq)
        print(f"Initial (n={len(best_seq)}): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: LP Hybrid Alternating ===
    print(f"\n=== Phase 1: LP Hybrid (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(400.0, remaining * 0.30)
    a1 = lp_hybrid_alternating(best_seq, time_budget=alloc, verbose=True)
    obj1 = compute_obj(a1)
    if obj1 < best_score:
        best_score = obj1
        best_seq = a1.copy()
        print(f"  Phase 1 done: obj={best_score:.8f} **")
    else:
        print(f"  Phase 1 done: obj={obj1:.8f}")

    # === Phase 2: Coordinate Descent Polish ===
    print(f"\n=== Phase 2: Coord Descent (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(200.0, remaining * 0.18)
        a2, obj2 = coordinate_descent_polish(best_seq, time_budget=alloc, verbose=True)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase 2 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 done: obj={obj2:.8f}")

    # === Phase 3: 2D Polish ===
    print(f"\n=== Phase 3: 2D Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(60.0, remaining * 0.05)
        a3, obj3 = pairwise_2d_polish(best_seq, time_budget=alloc)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase 3 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 done: obj={obj3:.8f}")

    # === Phase 4: N-sweep ===
    print(f"\n=== Phase 4: N-Sweep ===")
    remaining = deadline - time.monotonic()
    n_candidates = sorted(set([
        max(200, n_parent - 200), max(200, n_parent - 100),
        n_parent,
        n_parent + 100, n_parent + 200, n_parent + 400, n_parent + 600,
        500, 700, 900, 1100, 1300, 1500, 1800,
    ]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 2500]
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample_sequence(best_seq, n_try)
        # Quick LP + coord descent
        alloc_lp = min(40.0, remaining * 0.12)
        alloc_cd = min(30.0, remaining * 0.08)
        
        a_opt = lp_hybrid_alternating(a_try, time_budget=alloc_lp)
        a_opt, obj_opt = coordinate_descent_polish(a_opt, time_budget=alloc_cd)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 5: Peak Targeted Search ===
    print(f"\n=== Phase 5: Peak Targeted ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(80.0, remaining * 0.10)
        a5, obj5 = peak_targeted_search(best_seq, time_budget=alloc)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Phase 5 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 5 done: obj={obj5:.8f}")

    # === Phase 6: Final Coord Descent ===
    print(f"\n=== Phase 6: Final Coord Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 150.0)
        a6, obj6 = coordinate_descent_polish(best_seq, time_budget=alloc, verbose=True)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Phase 6 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 6 done: obj={obj6:.8f}")

    # === Phase 7: Random Restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 7: Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(5, int(remaining // 50))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 35:
                break

            if r % 2 == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.03, n_best))
            else:
                # Block perturbation
                a_r = best_seq.copy()
                n_blocks = max(5, n_best // 80)
                block_size = n_best // n_blocks
                blocks = rng.choice(n_blocks, min(3, n_blocks), replace=False)
                for b in blocks:
                    si = int(b) * block_size
                    ei = min(si + block_size, n_best)
                    a_r[si:ei] *= rng.uniform(0.6, 1.4)

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr

            alloc = min(remaining * 0.35, 40.0)
            a_opt = lp_hybrid_alternating(a_r, time_budget=alloc * 0.5)
            a_opt, obj_r = coordinate_descent_polish(a_opt, time_budget=alloc * 0.5)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 8: Final LP + Coord ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 40.0)
        a_final = lp_hybrid_alternating(best_seq, time_budget=alloc * 0.5)
        a_final, obj_final = coordinate_descent_polish(a_final, time_budget=alloc * 0.5)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
