def project_simplex(v):

        grad = smoothed_grad(y, tau)
        
        # FISTA update
        t_k = (1.0 + math.sqrt(1.0 + 4.0 * k * k)) / 2.0
        x_new = project_simplex(y - grad / L)
        y = x_new + ((k - 1) / t_k) * (x_new - x)
        y = project_simplex(y)
        
        x = x_new
        k += 1
        it += 1
        
        # Evaluate
        if it % 10 == 0:
            obj = compute_obj(x)
            if obj < best_obj:
                best_obj = obj
                best_a = x.copy()
                if verbose:
                    elapsed = time.monotonic() - start
                    print(f"    FISTA: it={it}, tau={tau:.5f}, obj={best_obj:.8f}, time={elapsed:.0f}s")
    
    return best_a


def coordinate_descent(a, time_budget=60.0, verbose=False):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Precompute: for each coordinate i, the "base" convolution (without a[i] contribution)
    # a*a without a[i]: conv_base[k] = (full conv)[k] - 2*a[i]*a[k-i]*1[valid] - a[i]^2*1[k=2i]
    
    it = 0
    while time.monotonic() - start < time_budget:
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                    break
            
            a_old = a[i]
            # Find optimal a[i] using ternary search
            # The function to minimize: max_k (a*a)[k] as a function of t = a[i]
            # = max_k (C_k + 2*t*a[k-i]*1[valid, k-i!=i] + t^2*1[k=2i])
            # subject to: sum_j≠i a[j] + t = 1, t >= 0
            # i.e., 0 <= t <= 1 - sum_j≠i a[j] = 1 - (1 - a_old) = a_old... 
            # Wait, sum a = 1, so sum_{j≠i} a[j] = 1 - a[i]. If we change a[i] to t,
            # we need to redistribute: sum_{j≠i} a[j] + t = 1, so t = 1 - sum_{j≠i} a[j].
            # But in coordinate descent, we fix other coordinates and only change a[i].
            # So the constraint is: t + (1 - a_old) = 1, meaning t = a_old.
            # That doesn't make sense for coordinate descent...
            
            # Actually, for coordinate descent on the simplex, we fix all other coordinates
            # and optimize a[i] subject to a[i] >= 0 and sum = 1.
            # Since sum_{j≠i} a[j] = 1 - a[i], if we change a[i] to t, we need
            # t + sum_{j≠i} a[j] = 1, so t + (1 - a_old) = 1, giving t = a_old.
            # This means we can't change a[i] independently!
            
            # The correct approach: we're on the simplex, so changing a[i] requires
            # adjusting some other coordinate. In standard coordinate descent on the
            # simplex, we pair coordinate i with coordinate j and optimize the
            # 2D problem.
            
            # Alternatively, we can parameterize: a[i] = t, a[j] = a[j]_old + (a_old - t)
            # for some j (e.g., j with largest a[j] or j = (i+1) % n).
            
            # Let me use the approach: pair i with the coordinate that has the
            # largest value (to maintain non-negativity).
            
            j = int(np.argmax(a))
            if j == i:
                j = (i + 1) % n
            
            a_j_old = a[j]
            # t + (1 - a_old - a_j_old + (a_j_old + a_old - t)) = 1 ✓
            # New: a[i] = t, a[j] = a_j_old + a_old - t
            # Constraint: t >= 0, a_j_old + a_old - t >= 0, so 0 <= t <= a_old + a_j_old
            
            # Objective as function of t:
            # (a*a)[k](t) = C_k + (t + a_j_old + a_old - t) * (...) + ...
            # This gets complicated. Let me just use a simple approach:
            # Try a few candidate values of t around a_old and pick the best.
            
            # Simple approach: try t = a_old * (1 ± 0.1, ± 0.05, ± 0.01) and also
            # the boundaries t = 0 and t = a_old + a_j_old.
            
            candidates = [
                a_old * 0.8, a_old * 0.9, a_old * 0.95, a_old,
                a_old * 1.05, a_old * 1.1, a_old * 1.2,
            ]
            # Also try the boundaries
            candidates.append(0.0)
            candidates.append(a_old + a_j_old)
            
            # Filter valid candidates
            best_t = a_old
            best_val = compute_obj(a)
            
            for t in candidates:
                if t < 0 or t > a_old + a_j_old:
                    continue
                a[i] = t
                a[j] = a_j_old + a_old - t
                # Ensure non-negativity
                if a[j] < 0:
                    a[i] = a_old
                    a[j] = a_j_old
                    continue
                # Renormalize (should already be sum=1)
                val = compute_obj(a)
                if val < best_val:
                    best_val = val
                    best_t = t
            
            if best_t != a_old:
                a[i] = best_t
                a[j] = a_j_old + a_old - best_t
                improved = True
                if best_val < best_obj:
                    best_obj = best_val
                    best_a = a.copy()
        
        it += 1
        if not improved:
            break
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD: sweep={it}, obj={best_obj:.8f}, time={elapsed:.0f}s")
    
    return best_a


def active_set_subgrad(a, time_budget=100.0, top_k=20, verbose=False):
    """Subgradient method focused on top-k active constraints."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0 / n
    momentum = np.zeros(n)
    beta = 0.5
    
    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Get top-k active positions
        k_indices = np.argsort(conv)[::-1][:top_k]
        
        # Compute subgradient: average gradient over active set
        # grad[m] = (2/|S|) * sum_{k in S} a[k-m] * 1[0 <= k-m < n]
        n_fft = 1
        while n_fft < 3 * n:
            n_fft <<= 1
        
        w = np.zeros(2 * n - 1)
        for k in k_indices:
            w[int(k)] = 1.0 / len(k_indices)
        
        A_fft = rfft(a, n_fft)
        W_fft = rfft(w, n_fft)
        R = irfft(np.conj(A_fft) * W_fft, n_fft)
        grad = R[:n] * 2.0  # Factor of 2 from the gradient formula
        
        # Scale: the objective is 2n * max_conv, so multiply by 2n
        # But since we're on the simplex, the scaling just affects step size
        grad *= 2.0 * n
        
        # Center gradient (project to tangent space of simplex)
        grad -= np.mean(grad)
        
        # Momentum update
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = project_simplex(a_new)
        
        a = a_new
        it += 1
        
        if it % 5 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                if verbose:
                    elapsed = time.monotonic() - start
                    print(f"    SubGrad: it={it}, obj={best_obj:.8f}, time={elapsed:.0f}s")
        
        # Decay learning rate
        if it % 200 == 0:
            lr *= 0.95
        if lr < 1e-6 / n:
            break
    
    return best_a


def lp_step(a, top_k=50, verbose=False):
    """LP step: find best direction g that reduces max conv."""
    n = len(a)
    a = a / np.sum(a)
    
    # Build the constraint matrix for top-k active constraints
    conv = compute_conv(a)
    active_k = np.argsort(conv)[::-1][:top_k]
    M_target = float(conv[active_k[0]])  # current max
    
    # LP: min M s.t. conv(a, g)[k] <= M for k in active_k, sum g = 1, g >= 0
    # conv(a, g)[k] = sum_j a[j] * g[k-j] for valid j, k-j
    # This is linear in g!
    
    # Build A matrix: A[k_idx, j] = a[j'] where j' is such that the term a[j]*g[?] contributes to conv[k]
    # conv(a, g)[k] = sum_j a[j] * g[k-j] * 1[0<=j<n, 0<=k-j<n]
    # = sum_j g[j] * a[k-j] * 1[0<=j<n, 0<=k-j<n]  (reindexing)
    # So the coefficient of g[j] in conv(a,g)[k] is a[k-j] if 0 <= k-j < n.
    
    # For efficiency, only include active constraints
    m_constraints = len(active_k)
    A = np.zeros((m_constraints, n))
    for idx, k in enumerate(active_k):
        k_int = int(k)
        for j in range(n):
            diff = k_int - j
            if 0 <= diff < n:
                A[idx, j] = a[diff]
    
    # Solve: min M s.t. A @ g <= M, sum g = 1, g >= 0, M >= 0
    # Variables: g (n) + M (1) = n+1 variables
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((m_constraints, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # A@g - M <= 0
    b_ub = np.zeros(m_constraints)
    
    A_eq = np.ones((1, n + 1))
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def lp_alternate(a, time_budget=200.0, top_k=50, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = lp_step(a, top_k=top_k)
        if g_lp is None:
            break
        
        # Try different blending factors
        best_local = np.inf
        best_local_a = best_a.copy()
        
        t_values = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local:
                best_local = obj_try
                best_local_a = a_try.copy()
        
        if best_local < best_obj - 1e-12:
            best_obj = best_local
            best_a = best_local_a.copy()
            a = best_a.copy()
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    LP: obj={best_obj:.8f}, time={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 5:
                # Perturbation
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
    
    return best_a


def diverse_init(n, rng):
    """Generate diverse initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.exp(-50 * (x - 0.5) ** 2)
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 3 * x)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-80 * (x - 0.3) ** 2) + np.exp(-80 * (x - 0.7) ** 2)
    elif choice == 3:
        a = np.ones(n)
        for _ in range(3):
            center = rng.integers(0, n)
            width = max(1, n // 10)
            for i in range(n):
                d = min(abs(i - center), n - abs(i - center))
                a[i] *= 1.0 + rng.standard_normal() * np.exp(-d / (width * 0.5))
        a = np.maximum(a, 0.01)
    elif choice == 4:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(0.5, 3.0)
    else:
        # Multi-bump
        n_bumps = rng.integers(2, 6)
        a = np.zeros(n)
        for _ in range(n_bumps):
            center = rng.uniform(0, 1)
            width = rng.uniform(0.05, 0.3)
            x = np.linspace(0, 1, n)
            a += np.exp(-((x - center) / width) ** 2)
        a = np.maximum(a, 0.001)
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_seq(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
