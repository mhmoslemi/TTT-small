def compute_conv(a):

    n = len(a)
    size = 1
    while size < 2 * n - 1:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return float(2.0 * n * np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(i + n, 2 * n - 1)
        A[k_start:k_end, i] = a[:k_end - k_start]
    return A

def solve_lp(a):
    n = len(a)
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.column_stack([A, -np.ones(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        z_val = np.maximum(res.x[:n], 0)
        zs = np.sum(z_val)
        if zs > 1e-10:
            z_val = z_val / zs
            M_val = float(res.x[n])

                dual = np.maximum(res.ineqlin.marg, 0)
            return z_val, M_val, dual
    return None, None, None

    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return float(2.0 * n * np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ z gives cross-conv a*z."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(i + n, 2 * n - 1)
        A[k_start:k_end, i] = a[:k_end - k_start]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. (a*z)[k] <= M for all k, sum(z)=1, z>=0.
    Returns (z, M_opt, dual) or (None, None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.column_stack([A, -np.ones(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status == 0:
        z_val = np.maximum(res.x[:n], 0)
        zs = np.sum(z_val)
        if zs > 1e-10:
            z_val = z_val / zs
            M_val = float(res.x[n])
            dual = None
            try:
                dual = np.maximum(res.ineqlin.marg, 0)
            except Exception:
                pass
            return z_val, M_val, dual
    return None, None, None

                if obj_try < best_obj - 1e-14:

                # Try reverse direction

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    alloc = min(200.0, remaining * 0.25)
    a_new = lp_alternating(best_seq, time_budget=alloc, verbose=True, rng=rng)
    obj = compute_obj(a_new)
    if obj < best_score:
        best_score = obj
        best_seq = a_new.copy()
        print(f"  Phase 1 done: obj={best_score:.8f} **")
    else:
        print(f"  Phase 1 done: obj={obj:.8f}")

    # === Phase 2: Multi-n exploration (450s) ===
    print(f"\n=== Phase 2: Multi-n exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_base = len(best_seq)
        # Wide range of n values
        n_candidates = []
        for n_try in range(200, 2501, 50):
            n_candidates.append(n_try)
        # Add values near n_base
        for offset in [-100, -75, -50, -25, 25, 50, 75, 100, 150, 200, 300, 500]:
            n_try = n_base + offset
            if 200 <= n_try <= 2500:
                n_candidates.append(n_try)
        n_candidates = sorted(set(n_candidates))
        # Sort by proximity to n_base
        n_candidates.sort(key=lambda x: abs(x - n_base))

        n_tried = set()
        for n_try in n_candidates:
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            if n_try in n_tried:
                continue
            n_tried.add(n_try)

            if n_try == len(best_seq):
                alloc = min(30.0, remaining * 0.06)
            else:
                a_try = resample_seq(best_seq, n_try)
                alloc = min(40.0, remaining * 0.08)

            a_opt = lp_alternating(a_try, time_budget=alloc, rng=rng)
            obj_try = compute_obj(a_opt)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            elif abs(n_try - n_base) <= 100 or n_try % 200 == 0:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 3: Refine best n with more LP iterations (250s) ===
    print(f"\n=== Phase 3: Refine n={len(best_seq)} ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(200.0, remaining * 0.35)
        a_new = lp_alternating(best_seq, time_budget=alloc, verbose=True, rng=rng)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 3 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 LP: obj={obj:.8f}")

    # === Phase 4: Pairwise + coordinate polish (120s) ===
    print(f"\n=== Phase 4: Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc_pw = min(70.0, remaining * 0.4)
        a_new = pairwise_polish(best_seq, time_budget=alloc_pw, rng=rng)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Pairwise polish: obj={best_score:.8f} **")
        else:
            print(f"  Pairwise polish: obj={obj:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc_cd = min(50.0, remaining * 0.4)
            a_new = coordinate_polish(best_seq, time_budget=alloc_cd, rng=rng)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Coord polish: obj={best_score:.8f} **")
            else:
                print(f"  Coord polish: obj={obj:.8f}")

    # === Phase 5: Second LP pass after polishing (100s) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 5: Post-polish LP ===")
        alloc = min(100.0, remaining * 0.4)
        a_new = lp_alternating(best_seq, time_budget=alloc, verbose=False, rng=rng)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Post-polish LP: obj={best_score:.8f} **")
        else:
            print(f"  Post-polish LP: obj={obj:.8f}")

        # Quick coordinate polish
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_new = coordinate_polish(best_seq, time_budget=min(20.0, remaining * 0.5), rng=rng)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Final coord: obj={best_score:.8f} **")

    # === Phase 6: Diverse restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Diverse restarts ===")
        n_best = len(best_seq)
        max_restarts = min(6, int(remaining // 35))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            if r % 3 == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.04 + 0.01 * r, n_best))
            elif r % 3 == 1:
                # Block perturbation
                a_r = best_seq.copy()
                n_blocks = max(5, n_best // 80)
                block_size = max(1, n_best // n_blocks)
                n_choose = min(4, n_blocks)
                blocks = rng.choice(n_blocks, n_choose, replace=False)
                for b in blocks:
                    si = int(b) * block_size
                    ei = min(si + block_size, n_best)
                    a_r[si:ei] *= rng.uniform(0.4, 1.6)
            else:
                # Try different n with perturbation
                n_r = int(rng.integers(max(200, n_best - 150), min(2200, n_best + 200)))
                a_r = resample_seq(best_seq, n_r)
                a_r = a_r * np.exp(rng.normal(0, 0.03, n_r))

            a_r = np.maximum(a_r, 0)
            s_r = np.sum(a_r)
            if s_r > 1e-10:
                a_r = a_r / s_r

            alloc = min(remaining * 0.35, 30.0)
            a_opt = lp_alternating(a_r, time_budget=alloc, rng=rng)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r} (n={len(a_opt)}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (n={len(a_opt)}): obj={obj_r:.8f}")

    # === Phase 7: Final micro-polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_final = coordinate_polish(best_seq, time_budget=min(remaining * 0.6, 15.0), rng=rng)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"\n  Final micro-polish: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
