Minimize 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP alternating with convex line search
2. Smooth max gradient descent with adaptive temperature

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)

            abs_A_new = abs_A ** alpha
            A_new = abs_A_new * np.exp(1j * phase_A)
            a_new = np.real(ifft(A_new))[:n]
            a_new = np.maximum(a_new, 0)
            s = float(np.sum(a_new))
            if s < 1e-10:
                break
            a_new /= s
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best = a_new.copy()
                a_cur = a_new
            else:
                break
        if verbose:
            print(f"    Spectral alpha={alpha:.2f}: obj={best_obj:.8f}, time={time.monotonic()-start:.1f}s")
    return best


def coordinate_descent_fine(a, time_budget=250.0, delta_init=5e-6, verbose=False):
    """Very fine coordinate descent with FFT-based fast evaluation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64) / np.sum(a)
    best = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    delta = delta_init
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    pass_count = 0
    while time.monotonic() - start < time_budget:
        pass_count += 1
        rng = np.random.default_rng(pass_count * 31 + 999)
        order = rng.permutation(n)
        pass_improved = False
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            found = False
            for d in [delta, delta * 0.5, delta * 2.0]:
                # Try increase
                a_try = a.copy()
                a_try[i] += d
                a_try = np.maximum(a_try, 0)
                a_try /= np.sum(a_try)
                fa = rfft(a_try, size)
                conv = irfft(fa * fa, size)[:2 * n - 1]
                obj_try = 2.0 * n * float(np.max(conv))
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    best = a_try.copy()
                    a = a_try
                    pass_improved = True
                    found = True
                    break
                # Try decrease (if a[i] is large enough)
                if a[i] > d * 2:
                    a_try2 = a.copy()
                    a_try2[i] -= d
                    a_try2 = np.maximum(a_try2, 0)
                    a_try2 /= np.sum(a_try2)
                    fa2 = rfft(a_try2, size)
                    conv2 = irfft(fa2 * fa2, size)[:2 * n - 1]
                    obj_try2 = 2.0 * n * float(np.max(conv2))
                    if obj_try2 < best_obj - 1e-15:
                        best_obj = obj_try2
                        best = a_try2.copy()
                        a = a_try2
                        pass_improved = True
                        found = True
                        break
            if found:
                continue
        if verbose and pass_count % 100 == 0:
            print(f"    CD pass {pass_count}: obj={best_obj:.8f}, delta={delta:.2e}, time={time.monotonic()-start:.1f}s")
        if not pass_improved:
            delta *= 0.7
            if delta < 1e-9:
                break
    return best
