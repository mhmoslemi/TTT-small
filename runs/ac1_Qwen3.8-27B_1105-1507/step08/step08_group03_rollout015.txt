Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast LP alternating

efficient FFT-based convolution, and analytical initialization.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp

def compute_conv(a):
    """Full convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    # g[m] = 2*a[k-m] if 0 <= k-m < n
    i_start = max(0, k - (n-1))
    i_end = min(n-1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end+1] = 2.0 * a[j_start:j_end+1][::-1]
    return g

def build_conv_matrix(a):
    """Build A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        row_end = min(2*n-1, j+n)
        A[j:row_end, j] = a[:row_end-j]

def nesterov_subgradient(a, time_budget=120.0, K=5, verbose=False):
    """
    Nesterov-accelerated subgradient method on the simplex.
    Uses dual-guided peak weighting.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 100
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        # Adaptive temperature weighting
        spread = (peak_vals[0] - peak_vals[-1]) / (max_val + 1e-30)
        temp = 5.0 / (1.0 + spread * 5.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / (max_val + 1e-30))
        weights /= weights.sum()

        # Compute weighted subgradient
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                g += w * grad_conv_at_peak(y, int(k))

        # Nesterov step
        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)
        y_new = np.maximum(y_new, 0)
        s_y = float(np.sum(y_new))
        if s_y > 1e-10:
            y_new = y_new / s_y

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                t_k = 1.0
                y = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.4)

        a_prev = a_new.copy()
        y = y_new.copy()
        t_k = t_new
        it += 1

        if verbose and it % 500 == 0:
            print(f"    NS it{it}: obj={best_obj:.8f}, t_k={t_k:.1f}, "

def mirror_descent(a, time_budget=80.0, verbose=False):

    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)

    best_a = a.copy()
    best_obj = compute_obj(a)

    min_lr = 1e-5
    stall = 0
    momentum = 0.3
    v = np.zeros(n)
    rng = np.random.default_rng(99)
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()
        if max_val < 1e-15:
            break

        # Smooth weighting over all peaks
        T = 5.0 * max_val
        w = np.exp((conv - max_val) / T)
        w /= w.sum()

        # Only use significant weights
        threshold = np.percentile(w, 90)
        active = w > threshold
        if not active.any():
            active[np.argmax(conv)] = True
        active_w = w[active]
        active_w /= active_w.sum()
        active_idx = np.where(active)[0]

        g = np.zeros(n)
        for wi, k in zip(active_w, active_idx):
            if wi > 1e-10:
                g += wi * grad_conv_at_peak(a, int(k))

        # Momentum
        v = momentum * v + g

        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new /= a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = max(min_lr, lr * 0.998)
            momentum = min(0.8, momentum + 0.003)
        else:
            stall += 1
            if stall > 150:
                a = best_a * np.exp(rng.normal(0, 0.2, n))
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                momentum = 0.3
                lr = max(min_lr, lr * 0.4)
                stall = 0

        if verbose and it % 200 == 0:

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_k = np.argsort(conv)[::-1][:5]

        improved = False
        for k in top_k:
            peak_val = conv[k]
            # Find contributors to this peak
            contrib = []
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contrib.append((a[i] * a[j], i, j))
            contrib.sort(reverse=True)

            # Find non-contributors (indices that don't affect this peak)
            non_contrib = [i for i in range(n) if not (0 <= k - i < n)]

            if not non_contrib:
                continue

            for val, i, j in contrib[:10]:
                if val < 1e-12:
                    break
                for target in non_contrib[:5]:
                    for frac in [0.005, 0.01, 0.02, 0.05]:
                        delta = frac * min(a[i], a[j])
                        if delta < 1e-15:
                            continue
                        src = i if a[i] >= a[j] else j
                        a[src] -= delta
                        a[target] += delta
                        a = np.maximum(a, 0)
                        s = float(np.sum(a))
                        if s < 1e-10:
                            a[src] += delta
                            a[target] -= delta
                            break
                        a = a / s
                        new_obj = compute_obj(a)
                        if new_obj < best_obj - 1e-13:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[target] -= delta
                            a = a / a.sum()
                            break
                if improved:
                    break
            if improved:
                break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
        a = a / a.sum()
    elif mode == 3:
        # Exponential decay
        r = 0.99
        a = np.array([r ** i for i in range(n)])
        a = a / a.sum()
    elif mode == 4:
        # Two bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Sinusoidal
        freqs = rng.integers(1, 5, size=2)
        phases = rng.uniform(0, 2 * np.pi, size=2)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Trapezoidal
        x = np.linspace(0, 1, n)
        a = np.clip(1 - 2 * np.abs(x - 0.5), 0.01, 1)
        a = a / a.sum()
    elif mode == 7:
        # Anti-symmetric emphasis (asymmetric)
        x = np.linspace(0, 1, n)
        a = 1 + 0.3 * x + 0.2 * np.sin(6 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Hybrid optimization combining fast LP alternating,
    Nesterov subgradient, mirror descent, peak reduction, and multi-scale
    n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    n_curr = len(best_seq)

    # === Phase 1: Quick LP alternating to get a good starting point (~100s) ===
    print(f"\n=== Phase 1: Initial LP Alternating ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.1, 100.0)
    if p1_budget > 20:
        a_lp1 = lp_alternating(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp1 = compute_obj(a_lp1)
        if obj_lp1 < best_score:
            best_score = obj_lp1
            best_seq = a_lp1.copy()
            print(f"  Phase 1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 LP: obj={obj_lp1:.8f} (no improvement)")

    # === Phase 2: Nesterov subgradient (~80s) ===
    print(f"\n=== Phase 2: Nesterov Subgradient ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.08, 80.0)
    if p2_budget > 15:
        a_ns = nesterov_subgradient(best_seq, time_budget=p2_budget, K=7, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Phase 2 NS: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 NS: obj={obj_ns:.8f} (no improvement)")

    # === Phase 3: Main LP alternating with multiple rounds (~500s) ===
    print(f"\n=== Phase 3: Main LP Alternating ===")
    remaining = deadline - time.monotonic()
    p3_total = min(remaining * 0.45, 500.0)

    n_rounds = 4
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p3_total / (n_rounds - rnd) * 0.8, remaining * 0.5)
        if alloc < 20:
            break

        if rnd > 0:
            # Perturb to escape potential stagnation
            noise_scale = 0.02 + 0.02 * rnd
            a_pert = best_seq * np.exp(rng.normal(0, noise_scale, len(best_seq)))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
        else:
            a_pert = best_seq.copy()

        a_lp = lp_alternating(a_pert, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 4: Mirror descent (~80s) ===
    print(f"\n=== Phase 4: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 80.0)
    if p4_budget > 15:
        a_md = mirror_descent(best_seq, time_budget=p4_budget, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Phase 4 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4 MD: obj={obj_md:.8f} (no improvement)")

    # === Phase 5: Peak reduction (~50s) ===
    print(f"\n=== Phase 5: Peak Reduction ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.05, 50.0)
    if p5_budget > 10:
        a_pr = peak_reduction(best_seq, time_budget=p5_budget)
        obj_pr = compute_obj(a_pr)
        if obj_pr < best_score:
            best_score = obj_pr
            best_seq = a_pr.copy()
            print(f"  Phase 5 PR: obj={best_score:.8f} **")
        else:
            print(f"  Phase 5 PR: obj={obj_pr:.8f} (no improvement)")

    # === Phase 6: Multi-n exploration (~200s) ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(300, n_current - 200),
        max(300, n_current - 100),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        400, 500, 600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40 or n_tried >= 5:
            break
        if abs(n_try - n_current) < 50 and n_tried < 2:
            continue

        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1

        # Quick LP optimization
        alloc = min(remaining * 0.12, 60.0)
        if alloc > 15:
            a_try = lp_alternating(a_try, time_budget=alloc)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse restarts (~150s) ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        restart_budget = min(remaining * 0.15, 50.0)
        a_opt = lp_alternating(a_r, time_budget=restart_budget)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        lp_alloc = min(remaining * 0.3, 60.0)
        a_lp = lp_alternating(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        ns_alloc = min(remaining * 0.25, 30.0)
        a_ns = nesterov_subgradient(best_seq, time_budget=ns_alloc, K=10)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Final NS: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_pr = peak_reduction(best_seq, time_budget=min(remaining * 0.3, 20.0))
        obj_pr = compute_obj(a_pr)
        if obj_pr < best_score:
            best_score = obj_pr
            best_seq = a_pr.copy()
            print(f"  Final PR: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.5, 25.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
