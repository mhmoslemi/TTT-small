"""

adaptive constraint subsets, and coordinate descent polish.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    na = len(a)
    nb = len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:na + nb - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(max(0, j - n + 1), min(2 * n - 1, j + n)):
            A[k, j] = a[k - j] if 0 <= k - j < n else 0.0
    return A

def build_conv_matrix_fast(a):
    """Build matrix A more efficiently."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # A[k, j] = a[k-j] if 0 <= k-j < n
        # k ranges from j to j+n-1 (but bounded to [0, m-1])
        k_start = max(0, j)
        k_end = min(m, j + n)
        a_slice = a[:k_end - k_start]
        A[k_start:k_end, j] = a_slice
    return A

def solve_lp_for_g(a, active_indices=None, warm_start=None):
    """
    Solve LP: min M s.t. conv(a, g) <= M (for active constraints), sum(g)=1, g>=0.
    Returns (g, M).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a = a / s
    
    m = 2 * n - 1
    A = build_conv_matrix_fast(a)
    
    if active_indices is not None:
        A_sub = A[active_indices, :]
    else:
        A_sub = A
    
    p = A_sub.shape[0]
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((p, n + 1))
    A_ub[:, :n] = A_sub
    A_ub[:, n] = -1.0
    b_ub = np.zeros(p)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s_g = float(np.sum(g_val))
        if s_g > 1e-15:
            g_val = g_val / s_g
            M_val = float(res.x[n])
            # Verify against ALL constraints
            conv_check = A @ g_val
            true_M = float(np.max(conv_check))
            return g_val, true_M
    return None, None

    return solve_lp_for_g(g, active_indices)  # Same structure due to symmetry of the problem

def get_active_set(a, top_k=50):
    """Get indices of top-k convolution peaks."""
    conv = compute_conv(a)
    sort_idx = np.argsort(conv)[::-1]

def convex_blend_search(a, g, n_trials=12):
    """
    Find best blend t in [0,1] for a_new = (1-t)*a + t*g.
    Uses the quadratic form of the objective in t.
    """
    n = len(a)
    a = a / np.sum(a)
    g = g / np.sum(g)
    
    conv_aa = compute_conv(a)
    conv_ag = compute_conv_cross(a, g)
    conv_gg = compute_conv(g)
    
    def obj_at_t(t):
        val = (1 - t)**2 * conv_aa + 2 * t * (1 - t) * conv_ag + t**2 * conv_gg
        return 2.0 * n * float(np.max(val))
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_at_t(c)
    fd = obj_at_t(d)
    
    for _ in range(n_trials):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at_t(d)
    
    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a + t_opt * g
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-15:
        return 0.5, a
    a_new = a_new / s
    return t_opt, a_new

def subgradient_polish(a, time_budget=60.0, verbose=False):

    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    a = a / s
    
    # Lower bound: by averaging, max >= (sum a)^2 / (2n-1) = 1/(2n-1)
    lb = 1.0 / (2 * n - 1)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr_scale = 1.0
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        if time.monotonic() - start > time_budget - 5:
            break
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        obj_cur = 2.0 * n * max_conv
        
        # Find active peaks (within 1% of max)
        threshold = max_conv * 0.99
        active = conv >= threshold
        
        # Subgradient: for the max, use weighted average of gradients at active points
        # grad_i = sum_k active_k * (a shifted)[k,i]
        # (a*a)[k] = sum_l a[l]*a[k-l], d/d a[i] = sum_l where k-l=i => a[k-i] + a[i-k] (if valid)
        # More carefully: d(a*a)[k]/da[i] = a[k-i] if 0<=k-i<n, plus a[i-k] if 0<=i-k<n
        # Actually: (a*a)[k] = sum_j a[j]*a[k-j], so d/da[i] = a[k-i] (if 0<=k-i<n)
        # Wait, that's not right either. Let me be more careful.
        # (a*a)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
        # d/da[i] = sum_j [delta(j,i)*a[k-j] + delta(k-j,i)*a[j]]
        #         = a[k-i] (if 0<=i<n and 0<=k-i<n) + a[k-i] (if 0<=k-i<n and 0<=k-(k-i)<n)
        #         = 2*a[k-i] (if 0<=k-i<n) 
        # Wait no. If i appears as j: a[i]*a[k-i], derivative w.r.t. a[i] is a[k-i].
        # If i appears as k-j: a[j]*a[i], derivative w.r.t. a[i] is a[j] where j=k-i, so a[k-i].
        # So total: a[k-i] + a[k-i] = 2*a[k-i] if 0 <= k-i < n.
        # But wait, this double counts when i = k-i, i.e., k = 2i. In that case it's just a[i].
        # Actually no, in the convolution sum, each pair (j, k-j) with j != k-j appears once.
        # When j = k-j (i.e., k = 2j), the term is a[j]^2, and d/da[j] = 2*a[j].
        # When j != k-j, the term a[j]*a[k-j] appears once, and d/da[i] for i=j is a[k-j], for i=k-j is a[j].
        # So: d(a*a)[k]/da[i] = a[k-i] if 0 <= k-i < n (accounting for the single occurrence in the sum).
        # Actually this is correct: in the sum sum_j a[j]*a[k-j], the coefficient of a[i] is a[k-i] if 0<=k-i<n.
        
        # So the gradient of (a*a)[k] w.r.t. a is: g_k[i] = a[k-i] if 0<=k-i<n, else 0.
        # This is just a shifted and reversed version of a!
        
        # For the max over active k, the subgradient is the average:
        num_active = int(np.sum(active))
        if num_active == 0:
            active = np.argmax(conv)
            grad_sub = np.zeros(n)
            k_star = active
            shift = k_star
            if shift < n:
                grad_sub[:n - shift] = a[shift:]
            if shift >= 1:
                grad_sub[shift:] += a[:n - shift]
            # Wait, I need to be more careful
            # (a*a)[k] = sum_j a[j] a[k-j]
            # d/da[i] = a[k-i] if 0 <= k-i < n
            # So grad_sub[i] = a[k-i] if 0 <= k-i < n, else 0
            # This means grad_sub = a reversed and shifted
            grad_sub = np.zeros(n)
            # For each i in [0, n-1], grad_sub[i] = a[k-i] if 0<=k-i<n
            # Let j = k-i, so i = k-j. j ranges over [max(0,k-n+1), min(n-1,k)]
            # grad_sub[k-j] = a[j]
            j_start = max(0, k_star - n + 1)
            j_end = min(n - 1, k_star)
            for j in range(j_start, j_end + 1):
                grad_sub[k_star - j] = a[j]
        else:
            # Weighted average of subgradients at active points
            # For efficiency, compute the gradient as a convolution-like operation
            # grad[i] = (1/num_active) * sum_{k active} a[k-i] (if 0<=k-i<n)
            # This is: (1/num_active) * sum_{k active} a[k-i]
            # = (1/num_active) * (indicator_active * a_reversed flipped)[n-1-i]... 
            # Let me just compute it directly using the structure
            # For active set A: grad[i] = (1/|A|) * sum_{k in A} a[k-i] (if 0<=k-i<n)
            # = (1/|A|) * sum_{j: k=i+j in A, 0<=j<n} a[j]
            # Let's define b = indicator of active set (length 2n-1)
            # Then grad[i] = (1/|A|) * sum_j b[i+j] * a[j] for j where i+j < 2n-1
            # This is a cross-correlation!
            b = active.astype(np.float64) / num_active
            # grad[i] = sum_j b[i+j] * a[j] = cross-correlation of b and a
            # cross_corr[i] = sum_j b[i+j] * a[j]
            # = sum_j b[j+i] * a[j]
            # Using FFT: this is the "valid" cross-correlation
            # Actually, let's use np.correlate or manual FFT
            size_corr = 2 * n
            size = 1
            while size < size_corr:
                size <<= 1
            fb = rfft(b, size)
            fa = rfft(a, size)
            # cross_corr[i] = sum_j b[j+i] * a[j] = (a * flip(b))[i + n - 1] in convolution terms
            # Let's use: sum_j b[i+j] a[j] = sum_j a[j] b[i+j]
            # This is the cross-correlation of a and b at lag i
            # In terms of convolution: cross_corr = irfft(fa * conj(fb)) but we need to handle the shift
            # Actually: (a ⋆ b)[i] = sum_j a[j] b[j+i] = sum_j a[j+i] b[j] (by change of variable)
            # Let me just compute it with the FFT directly
            # sum_j a[j] b[j+i]: this is the cross-correlation at lag i
            # = ifft(conj(fft(a)) * fft(b))[i] (with appropriate normalization)
            # Or: = ifft(fft(a) * conj(fft(b)))[n-1-i] (circular cross-correlation)
            # For linear cross-correlation, we need zero-padding
            cross = irfft(fa * np.conj(fb), size)[:2*n-1]
            # cross[k] = sum_j a[j] b[j-k] (or sum_j a[j+k] b[j], depending on convention)
            # Let me verify: cross[k] = sum_j a[j] b[j+k-n+1]... this is getting confusing
            # Let me just do it directly for now (it's O(n^2) but n is at most ~1500)
            grad_sub = np.zeros(n)
            active_indices_list = np.where(active)[0]
            for k in active_indices_list:
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                grad_sub[k - j_start:k - j_end + 1] += a[j_start:j_end + 1]
            grad_sub /= num_active
        
        # Projected subgradient step
        # We want to minimize f(a) = 2n * max_k (a*a)[k] on the simplex
        # Subgradient step: a_new = project_simplex(a - lr * grad_sub)
        # But we need to be careful: the subgradient gives us the direction to DECREASE the function
        # Actually, we want to move in the direction that reduces the peak
        # The subgradient of max_k (a*a)[k] points in the direction of INCREASE
        # So we subtract: a_new = a - lr * grad_sub (then project)
        
        # Polyak step: lr = (f(a) - f*) / ||grad||^2
        # f* >= lb (lower bound)
        f_gap = obj_cur - 2.0 * n * lb  # This is 2n * (max_conv - 1/(2n-1))
        grad_norm_sq = float(np.sum(grad_sub**2))
        
        if grad_norm_sq < 1e-30:
            break
        lr = lr_scale * f_gap / grad_norm_sq
        lr = min(lr, 0.5)  # Cap step size
        
        a_new = a - lr * grad_sub
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = simplex_project(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr_scale *= 0.8
            if lr_scale < 0.01:
                lr_scale = 0.5  # Reset
                break
        if verbose and it % 200 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}, lr={lr:.6f}, time={time.monotonic()-start:.1f}s")

    return best_a

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def coordinate_descent_convex(a, time_budget=60.0, verbose=False):
    """
    Coordinate descent on the convex problem: min max_k (a*a)[k] s.t. sum(a)=1, a>=0.
    For each coordinate i, find the optimal value by 1D search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Precompute: when we change a[i] to a[i]+d, the convolution changes by:
    # delta_conv[k] = d * (a[k-i] + a[i-k]) (with appropriate bounds)
    # Actually: new_conv[k] = old_conv[k] + d*(a[k-i] + a[i-k]) - d^2 * (something)
    # Wait, (a*a)[k] = sum_j a[j] a[k-j]
    # If we change a[i] to a[i]+d, then:
    # new_conv[k] = sum_j a'[j] a'[k-j]
    # = old_conv[k] + d*a[k-i] + d*a[i-k] + d^2*delta(k, 2i)
    # where a[k-i] means a[k-i] if 0<=k-i<n else 0
    # and a[i-k] means a[i-k] if 0<=i-k<n else 0 (which is a[k-i] with i and k swapped)
    # Actually a[i-k] is defined as a[i-k] if 0<=i-k<n else 0 = a[k-i] if 0<=k-i<n... no
    # a[i-k] is the value at index (i-k), which is valid if 0 <= i-k < n, i.e., k <= i and k >= i-n+1
    # So: new_conv[k] = old_conv[k] + d * (a[max(0,k-i):] ... ) 
    # Let me be more careful:
    # (a*a)[k] = sum_j a[j]*a[k-j] where j ranges over [max(0,k-n+1), min(n-1,k)]
    # d/da[i] (a*a)[k] = a[k-i] (if 0<=k-i<n) + a[k-i] (if i=k-j for some valid j, i.e., j=k-i, 0<=k-i<n)
    # Wait, I keep going in circles. Let me just use:
    # (a*a)[k] is quadratic in a. If we change a[i] by d:
    # d(a*a)[k] = d * (partial (a*a)[k] / partial a[i]) + d^2 * (partial^2 (a*a)[k] / partial a[i]^2)
    # partial (a*a)[k] / partial a[i] = a[k-i] (if 0<=k-i<n) -- this is because a[i] appears in the sum as a[j] when j=i, giving a[k-i], AND as a[k-j] when k-j=i, i.e., j=k-i, giving a[i].
    # So partial = a[k-i] (if 0<=k-i<n) + a[i] (if i is in the valid range for j, which it always is since 0<=i<n)
    # Wait no. Let me restart.
    # (a*a)[k] = sum_{j=0}^{n-1} a[j] * a[k-j] (with the convention that a[m]=0 if m<0 or m>=n)
    # = sum over all j where both j and k-j are in [0,n-1]
    # d/da[i]: a[i] appears when j=i (term: a[i]*a[k-i], requires 0<=k-i<n) 
    #           AND when k-j=i, i.e., j=k-i (term: a[k-i]*a[i], requires 0<=k-i<n)
    # These are the SAME term when j=i AND k-j=i, i.e., k=2i. In that case, the term is a[i]^2 and d/da[i] = 2*a[i].
    # So: d(a*a)[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] if 0<=k-i<n and i != k-i (i.e., k != 2i)
    #                        = 2*a[i] if k = 2i (which equals 2*a[k-i] since k-i=i)
    # In all cases: d(a*a)[k]/da[i] = 2*a[k-i] if 0<=k-i<n, else 0.
    # Wait, that's not right either. When j=i: the term is a[i]*a[k-i], derivative is a[k-i].
    # When j=k-i (and k-i != i): the term is a[k-i]*a[i], derivative is a[k-i].
    # Total: 2*a[k-i] if 0<=k-i<n and k-i != i.
    # When k-i = i (k=2i): the term is a[i]^2, derivative is 2*a[i] = 2*a[k-i].
    # So in ALL cases: d(a*a)[k]/da[i] = 2*a[k-i] if 0<=k-i<n, else 0.
    
    # So if we change a[i] by d:
    # new_conv[k] = old_conv[k] + 2*d*a[k-i] (if 0<=k-i<n) + d^2 (if k=2i, from the a[i]^2 term)
    # Actually, the second-order term: (a[i]+d)^2 - a[i]^2 = 2da[i] + d^2. The 2da[i] is already captured.
    # So new_conv[2i] gets an extra d^2.
    # More precisely: new_conv[k] = old_conv[k] + d * grad_k for all k, where grad_k = 2*a[k-i] if 0<=k-i<n else 0
    # EXCEPT new_conv[2i] also gets + d^2 if 2i < 2n-1.
    
    # This means for fixed i, the objective as a function of d (the change in a[i], after renormalization)
    # is complex because of the renormalization. Let's skip the exact 1D solve and use bisection.
    
    # Simpler approach: try small perturbations
    delta = 1e-4
    for sweep in range(200):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        rng = np.random.default_rng(sweep * 7919 + n)
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            # Try increasing a[i]
            for d in [delta, delta * 3, delta * 10]:
                a_try = a.copy()
                a_try[i] += d
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            
            if improved:
                continue
            
            # Try decreasing a[i]
            for d in [delta, delta * 3]:
                if a[i] < d:
                    continue
                a_try = a.copy()
                a_try[i] -= d
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        
        if not improved:
            delta *= 0.7
            if delta < 1e-7:
                break
        if verbose and sweep % 20 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, delta={delta:.2e}, time={time.monotonic()-start:.1f}s")
    
    return best_a

def spectral_init(n, rng):
    """
    Initialize using spectral method: create a sequence with approximately flat
    power spectrum, which tends to give flatter autocorrelation.
    """
    # Start with random phase spectrum (but real-valued, so use cos)
    freqs = np.arange(n)
    phases = rng.uniform(0, 2 * np.pi, n)
    # Symmetric spectrum for real-valued sequence
    spec = np.ones(n)
    for k in range(1, n // 2 + 1):
        spec[k] = np.abs(rng.normal(1, 0.3))
        spec[n-k] = spec[k] if n-k < n else spec[k]
    # Actually, let's use a simpler approach: random sequence with some structure
    a = rng.exponential(1, n)
    # Apply a window to encourage edge concentration
    window = 1.0 / (1.0 + 0.5 * np.exp(-np.abs(np.arange(n) - n//2) / (n//4)))
    a = a * window
    return a / np.sum(a)

def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1, n)
    return a / np.sum(a)

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using LP alternating with warm starts,
adaptive constraint subsets, subgradient polishing, and coordinate descent.
Key improvements: efficient HiGHS LP solves, vectorized subgradient, multi-n search,
multiple initializations, and intensive final polishing.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    na = len(a)
    nb = len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:na + nb - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_fast(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_start = max(0, j)
        k_end = min(m, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_for_g(a, active_indices=None):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M for k in active_indices (or all),
    sum(g)=1, g>=0. Returns (g, M).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a = a / s

    m = 2 * n - 1
    A_full = build_conv_matrix_fast(a)

    if active_indices is not None and len(active_indices) < m:
        A_sub = A_full[active_indices, :].copy()
    else:
        A_sub = A_full

    p = A_sub.shape[0]

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((p, n + 1))
    A_ub[:, :n] = A_sub
    A_ub[:, n] = -1.0
    b_ub = np.zeros(p)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_g = float(np.sum(g_val))
            if s_g > 1e-15:
                g_val = g_val / s_g
                conv_check = A_full @ g_val
                true_M = float(np.max(conv_check))
                return g_val, true_M
    except Exception:
        pass
    return None, None


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def convex_blend_search(a, g, n_trials=12):
    """Find best blend t in [0,1] for a_new = (1-t)*a + t*g using golden section."""
    n = len(a)
    a_n = a / np.sum(a)
    g_n = g / np.sum(g)

    conv_aa = compute_conv(a_n)
    conv_ag = compute_conv_cross(a_n, g_n)
    conv_gg = compute_conv(g_n)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c_val = hi - gr * (hi - lo)
    d_val = lo + gr * (hi - lo)
    fc = obj_at_t(c_val)
    fd = obj_at_t(d_val)

    for _ in range(n_trials):
        if fc < fd:
            hi = d_val
            d_val = c_val
            fd = fc
            c_val = hi - gr * (hi - lo)
            fc = obj_at_t(c_val)
        else:
            lo = c_val
            c_val = d_val
            fc = fd
            d_val = lo + gr * (hi - lo)
            fd = obj_at_t(d_val)

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_n + t_opt * g_n
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-15:
        return 0.5, a_n
    return t_opt, a_new / s


def lp_alternation(a, time_budget=100.0, verbose=False, top_k=60):
    """LP alternating with adaptive constraints and both directions."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(5000):

            break

        max_conv = float(np.max(conv_cur))
        sort_idx = np.argsort(conv_cur)[::-1]
        active = sort_idx[:min(2 * n - 1, top_k)]

def subgradient_polish(a, time_budget=60.0, verbose=False):
    """
    Projected subgradient method on simplex for min 2n*max_k(a*a)_k.
    Uses weighted subgradient from top peaks and Polyak-like step.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lb = 1.0 / (2 * n - 1)
    lr_scale = 1.0

    for it in range(200000):
        if time.monotonic() - start > time_budget - 2:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        obj_cur = 2.0 * n * max_conv

        # Active set: top peaks within 5% of max
        threshold = max_conv * 0.95
        active_mask = conv >= threshold
        num_active = int(np.sum(active_mask))
        if num_active == 0:
            active_mask = np.zeros(2 * n - 1, dtype=bool)
            active_mask[int(np.argmax(conv))] = True
            num_active = 1

        # Compute subgradient: grad[i] = (1/|A|) * sum_{k in A} 2*a[k-i] (if 0<=k-i<n)
        # = (2/|A|) * sum_{k in A} a[k-i] for valid k-i
        # Vectorized: for each valid shift d = k-i, accumulate a values
        active_indices = np.where(active_mask)[0]
        grad = np.zeros(n)
        # For each k in active: grad[k-j] += a[j] for j in valid range
        # Equivalent: grad[i] = sum_{k in active} a[k-i] (if 0<=k-i<n)
        # = sum_j a[j] * (number of k in active such that k=i+j)
        # = sum_j a[j] * active_mask[i+j] (if i+j < 2n-1)
        # This is a cross-correlation of a with active_mask
        size_cc = 1
        while size_cc < 2 * n:
            size_cc <<= 1
        fa_cc = rfft(a, size_cc)
        fm_cc = rfft(active_mask.astype(np.float64), size_cc)
        # cross[i] = sum_j a[j] * active_mask[i+j]
        # = ifft(fft(a) * fft(active_mask))[i] but with circular convention
        # Actually: sum_j a[j] * b[j+i] = circular cross-correlation
        # = ifft(conj(fft(a)) * fft(b)) ... let me just use the direct approach
        # For efficiency with n up to ~1500, direct O(n * num_active) is fine
        for k in active_indices:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            grad[k - j_start:k - j_end + 1] += a[j_start:j_end + 1]
        grad *= 2.0 / num_active

        grad_norm_sq = float(np.sum(grad ** 2))
        if grad_norm_sq < 1e-30:
            break

        f_gap = obj_cur - 2.0 * n * lb
        lr = lr_scale * f_gap / grad_norm_sq
        lr = min(lr, 1.0)

        a_new = simplex_project(a - lr * grad)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr_scale *= 0.85
            if lr_scale < 0.01:
                break

        if verbose and it % 500 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}, lr={lr:.6f}, time={time.monotonic()-start:.1f}s")

    return best_a


def coordinate_descent_convex(a, time_budget=60.0, verbose=False):
    """Coordinate descent with adaptive step sizes."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    delta = 2e-4

    for sweep in range(500):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        rng = np.random.default_rng(sweep * 7919 + n)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            for d in [delta, delta * 3, delta * 0.3]:
                a_try = a.copy()
                a_try[i] += d
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-15:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if improved:
                continue

            for d in [delta, delta * 0.3]:
                if a[i] < d * 0.5:
                    continue
                a_try = a.copy()
                a_try[i] -= d
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-15:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            delta *= 0.6
            if delta < 1e-8:
                break
        if verbose and sweep % 20 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, delta={delta:.2e}, time={time.monotonic()-start:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, rng):
    """Random exponential initialization."""
    a = rng.exponential(1.0, n)
    return a / np.sum(a)


def edge_weighted_init(n, rng):
    """Initialization with edge concentration."""
    a = rng.exponential(1.0, n)
    dist_from_center = np.abs(np.arange(n) - n // 2) / (n // 2 + 1)
    a *= (1.0 + 0.5 * dist_from_center)
    return a / np.sum(a)


def peak_equalization(a, time_budget=30.0):
    """Targeted LP on top peaks only."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        sort_idx = np.argsort(conv)[::-1]
        top_k = min(30, 2 * n - 1)
        active = sort_idx[:top_k]
        g, _ = solve_lp_for_g(a, active)
        if g is None:
            break
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-15:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating + subgradient + coordinate descent + multi-n.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={len(parent)}): obj={best_score:.8f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    n_cur = len(best_seq)

    # === Phase 1: Intensive LP Alternating ===
    print(f"\n=== Phase 1: LP Alternating (n={n_cur}) ===")
    remaining = deadline - time.monotonic()
    p1 = min(300.0, remaining * 0.30)
    a_opt = lp_alternation(best_seq.copy(), time_budget=p1, verbose=True, top_k=50)
    obj_opt = compute_obj(a_opt)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        print(f"  P1 done: obj={best_score:.8f}")
    else:
        print(f"  P1 done: obj={obj_opt:.8f}")

    # === Phase 2: Peak Equalization ===
    print(f"\n=== Phase 2: Peak Equalization ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p2 = min(50.0, remaining * 0.08)
        a_pe = peak_equalization(best_seq.copy(), time_budget=p2)
        obj_pe = compute_obj(a_pe)
        if obj_pe < best_score:
            best_score = obj_pe
            best_seq = a_pe.copy()
            print(f"  P2 done: obj={best_score:.8f}")
        else:
            print(f"  P2 done: obj={obj_pe:.8f}")

    # === Phase 3: Subgradient Polish ===
    print(f"\n=== Phase 3: Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p3 = min(60.0, remaining * 0.10)
        a_sg = subgradient_polish(best_seq.copy(), time_budget=p3, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  P3 done: obj={best_score:.8f}")
        else:
            print(f"  P3 done: obj={obj_sg:.8f}")

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coord Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4 = min(80.0, remaining * 0.12)
        a_cd = coordinate_descent_convex(best_seq.copy(), time_budget=p4, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  P4 done: obj={best_score:.8f}")
        else:
            print(f"  P4 done: obj={obj_cd:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(400, n_best - 100), max(400, n_best - 50), max(400, n_best - 25),
        n_best, n_best + 25, n_best + 50, n_best + 100, n_best + 150,
        500, 600, 700, 750, 800, 850, 900, 950, 1000, 1100, 1200
    ]))
    n_options = [x for x in n_options if 400 <= x <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.08, 25.0)
        a_opt_n = lp_alternation(a_try, time_budget=alloc, top_k=40)
        obj_try = compute_obj(a_opt_n)

        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_try < best_score + 0.005:
            a_opt_n = coordinate_descent_convex(a_opt_n, time_budget=min(15.0, remaining * 0.4))
            obj_try = compute_obj(a_opt_n)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt_n.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Alternative Initializations ===
    print(f"\n=== Phase 6: Alt Inits ===")
    n_final = len(best_seq)
    for init_fn, name in [(random_init, 'random'), (edge_weighted_init, 'edge')]:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break
        a_try = init_fn(n_final, rng)
        alloc = min(remaining * 0.10, 35.0)
        a_opt = lp_alternation(a_try, time_budget=alloc, top_k=50)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** {name} init: obj={best_score:.8f} **")
        else:
            print(f"  {name} init: obj={obj_try:.8f}")

    # === Phase 7: Final Polishing ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(50.0, remaining * 0.25)
        a_lp = lp_alternation(best_seq.copy(), time_budget=alloc, verbose=True, top_k=40)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(30.0, remaining * 0.25)
        a_sg = subgradient_polish(best_seq.copy(), time_budget=alloc)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(30.0, remaining * 0.4)
        a_cd = coordinate_descent_convex(best_seq.copy(), time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f}")

    # Validation
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if np.isfinite(val) and val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
