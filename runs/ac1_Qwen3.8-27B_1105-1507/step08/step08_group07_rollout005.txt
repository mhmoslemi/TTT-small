def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_conv_and_obj(a):
    """Compute conv and objective together."""
    n = len(a)
    s = float(np.sum(a))
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_b = float(np.max(conv))
    obj = 2.0 * n * max_b  # since sum(a_norm) = 1
    return conv, obj

    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

            best_a = a_new.copy()
            a = a_new
        else:
            # Backtracking
            for factor in [0.5, 0.25, 0.1]:
                a_try = a + factor * v
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st > 1e-10:
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        break

    return best_a


def lp_alternating_improved(a, time_budget=200.0, verbose=False):
    """
    Improved LP-alternating with:
    - Nesterov momentum
    - Adaptive step size
    - Cycle detection
    - Periodic smoothed gradient steps
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)  # momentum velocity
    beta = 0.9  # momentum coefficient
    stall = 0
    history = []  # for cycle detection

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        # Every 20 iterations, try a smoothed gradient step to escape local minima
        if len(history) > 0 and len(history) % 20 == 0 and stall > 5:
            # Try smoothed gradient for a short time
            a_smooth = smooth_gradient_descent(a, time_budget=min(30, time_budget - elapsed), 
                                               tau_start=20, tau_end=200)
            obj_smooth = compute_obj(a_smooth)
            if obj_smooth < best_obj:
                best_obj = obj_smooth
                best_a = a_smooth.copy()
                a = a_smooth
                v = np.zeros(n)
                stall = 0
                history = []
                continue

        result = solve_lp(a)
        if result[0] is None:
            break
        g_lp, M_lp, dual_y = result

        # Nesterov momentum
        v = beta * v + (g_lp - a)
        v = np.maximum(v, -a)  # prevent going too negative
        # Actually, let's use a simpler momentum:
        # v = beta * v + (1-beta) * (g_lp - a)
        # a_new = a + step * v

        # Try multiple step sizes
        best_step = 0
        best_obj_step = best_obj
        for step in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            a_try = a + step * v
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_step:
                best_obj_step = obj_try
                best_step = step

        if best_step > 0 and best_obj_step < best_obj - 1e-13:
            a_new = a + best_step * v
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            best_obj = best_obj_step
            best_a = a_new.copy()

            # Cycle detection
            if len(history) >= 2:
                if np.max(np.abs(a_new - history[-2])) < 1e-6:
                    # 3-cycle or longer: use average of recent points
                    recent = np.mean(history[-4:], axis=0)
                    recent = np.maximum(recent, 0)
                    recent = recent / np.sum(recent)
                    obj_recent = compute_obj(recent)
                    if obj_recent < best_obj:
                        best_obj = obj_recent
                        best_a = recent.copy()
                        a_new = recent
                    v = np.zeros(n)
                    history = []
                    stall = 0

            a = a_new
            history.append(a_new.copy())
            if len(history) > 10:
                history = history[-10:]
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Randomized escape
                rng = np.random.default_rng(int(time.monotonic() * 1e9) % (2**31))
                if dual_y is not None and np.max(dual_y) > 0:
                    # Dual-guided: perturb based on which constraints are active
                    A = build_conv_matrix(a)
                    active = dual_y > 0.01
                    if np.sum(active) > 0:
                        col_imp = np.sum(A[active, :], axis=0)
                        col_imp = col_imp - np.mean(col_imp)
                        pert = -0.03 * np.sign(col_imp) + rng.normal(0, 0.02, n)
                    else:
                        pert = rng.normal(0, 0.03, n)
                else:
                    pert = rng.normal(0, 0.03, n)

                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                history = []
                stall = 0
                if verbose:
                    print(f"    [Escape] obj={best_obj:.8f}")

        if verbose and int(elapsed * 2) % 10 == 0:
            print(f"    LP-improved: obj={best_obj:.8f}, t={elapsed:.0f}s, stall={stall}")

    return best_a


def coordinate_descent_polish(a, time_budget=60.0, verbose=False):
    """
    Coordinate descent: try changing one coordinate at a time.
    Efficiently updates only affected convolution coefficients.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    conv = compute_conv(a)
    current_max = float(np.max(conv))

    start = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0

    while time.monotonic() - start < time_budget:
        improved = False

        # Try a set of random coordinates
        for _ in range(50):
            if time.monotonic() - start > time_budget:
                break
            if no_improve > 200:
                break

            m = rng.integers(0, n)

            # Find best delta for coordinate m
            # conv[k] for k in [m, min(m+n-1, 2n-2)] is affected
            k_min = m
            k_max = min(m + n - 1, 2 * n - 2)

            # Try several deltas
            best_delta = 0
            best_new_max = current_max
            for frac in [0.01, 0.02, 0.05, 0.1, -0.01, -0.02, -0.05, -0.1]:
                delta = frac * a[m]
                new_val = a[m] + delta
                if new_val < 0:
                    continue

                # Compute new max
                new_max = current_max
                for k in range(k_min, k_max + 1):
                    if k == 2 * m:
                        new_conv_k = conv[k] + 2 * a[m] * delta + delta * delta
                    else:
                        idx = k - m
                        if 0 <= idx < n:
                            new_conv_k = conv[k] + 2 * delta * a[idx]
                        else:
                            new_conv_k = conv[k]
                    if new_conv_k > new_max:
                        new_max = new_conv_k

                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0 and best_new_max < current_max - 1e-13:
                # Apply the change
                a[m] += best_delta
                a[m] = max(a[m], 0)
                a = a / np.sum(a)

                # Recompute conv (simpler than partial update for now)
                conv = compute_conv(a)
                current_max = float(np.max(conv))
                obj = 2.0 * n * current_max
                best_obj = obj
                best_a = a.copy()
                improved = True
                no_improve = 0
            else:
                no_improve += 1

        if not improved and no_improve > 200:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
