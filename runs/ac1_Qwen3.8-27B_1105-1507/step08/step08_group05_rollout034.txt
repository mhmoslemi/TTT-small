            best_p = p_new.copy()
        
        p = p_new
        
        if verbose and it % 5000 == 0:
            print(f"    MD it={it}: beta={beta:.1f}, best={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_p

**Key insight**: The problem is to minimize `max_k conv(a,a)[k]` subject to `a >= 0, sum(a) = 1`. This is a convex optimization problem in disguise! Wait, is it?

`conv(a,a)[k] = sum_i a[i] * a[k-i]` is a quadratic form in `a`. The constraint `conv(a,a)[k] <= M` is a quadratic inequality. For this to define a convex set, the quadratic form must be positive semi-definite. 

The matrix of the quadratic form for `conv(a,a)[k]` is a shift matrix (banded). Specifically, `conv(a,a)[k] = a^T Q_k a` where `Q_k` is a matrix with `Q_k[i, k-i] = 1/2` and `Q_k[k-i, i] = 1/2` (with appropriate zeroing for out-of-bounds indices), and `Q_k[j, j] = 1` if `k = 2j`.

Is `Q_k` positive semi-definite? For `k=0`: `Q_0` is the identity, which is PSD. For `k=1`: `Q_1` has `1/2` on the first super/sub diagonal, which has eigenvalues `cos(πj/(n+1))` for j=1,...,n-1... actually it's a bidiagonal-like matrix. Let me think...

For `k=1`, `Q_1` is an n×n matrix with `Q_1[i, i+1] = 1/2` and `Q_1[i+1, i] = 1/2` for i=0,...,n-2, and zeros elsewhere. This is `(1/2) * (E + E^T)` where E is the shift matrix. The eigenvalues of E are `e^{2πij/n}` for j=0,...,n-1, so the eigenvalues of `(E+E^T)/2` are `cos(2πj/n)` for j=0,...,n-1. Since `cos(2πj/n)` can be negative (for j > n/4), `Q_1` is NOT positive semi-definite!

So the problem is NOT convex. The constraints `conv(a,a)[k] <= M` are non-convex. This makes it a non-convex QCQP, which is NP-hard in general.

                mask[k_start:k_start+n_lin] = False

                denom = (1.0 - old_val + x) ** 2

                    return np.inf

            best_f = eval_1d(old_val)
            for trial_x in [0.0, old_val*0.5, old_val*1.5, 0.01, 0.1, 0.5]:

                f_trial = eval_1d(trial_x)
                if f_trial < best_f:
                    best_f = f_trial
                    best_x = trial_x

            res = minimize_scalar(eval_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 40})

                best_f = res.fun
                best_x = res.x

                a[j] = best_x

Uses logsumexp smooth approximation with adaptive beta for mirror descent,
FFT-based gradient computation, and exact 1D coordinate minimization.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_grad_fft(a, beta):
    """
    Compute logsumexp(beta*conv(a,a))/beta and its gradient using FFT.
    Returns (smooth_value, gradient) where gradient has same shape as a.
    """
    n = len(a)
    conv = compute_conv(a)  # length 2n-1
    
    # Logsumexp
    v = beta * conv
    v_max = np.max(v)
    exp_v = np.exp(v - v_max)
    w_sum = np.sum(exp_v)
    weights = exp_v / w_sum
    smooth_val = (v_max + np.log(w_sum)) / beta
    
    # Gradient: grad[j] = 2 * conv(a, weights_reversed)[2n-2-j]
    w_rev = weights[::-1]  # length 2n-1
    
    # Linear convolution of a (length n) and w_rev (length 2n-1)
    # Result length: n + (2n-1) - 1 = 3n-2
    fft_size = 1
    while fft_size < 3 * n - 2:
        fft_size <<= 1
    fa = rfft(a, fft_size)
    fw = rfft(w_rev, fft_size)
    conv_aw = irfft(fa * fw, fft_size)[:3*n-2]
    
    # grad[j] = 2 * conv_aw[2n-2-j] for j = 0, ..., n-1
    # Indices: 2n-2, 2n-3, ..., n-1
    grad = 2.0 * conv_aw[n-1:2*n-1][::-1]
    
    return smooth_val, grad


def mirror_descent_smooth(a0, time_budget=100.0, beta_start=20.0, beta_end=500.0,
                          eta_init=0.5, verbose=False):
    """
    Mirror descent (multiplicative weights) with smooth logsumexp approximation.
    Works on the probability simplex (sum(a)=1, a>=0).
    """
    n = len(a0)
    p = a0.astype(np.float64).copy()
    s = float(np.sum(p))
    if s < 1e-10:
        return a0.copy()
    p = p / s
    
    best_p = p.copy()
    best_obj = compute_obj(p)
    start = time.monotonic()
    
    # Warmup iterations
    warmup = max(100, int(0.1 * time_budget * 100))
    total_iters = max(1000, int(time_budget * 200))
    it = 0
    stall_count = 0
    last_improve_it = 0
    
    while it < total_iters:
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        
        # Adaptive beta
        progress = min(1.0, elapsed / max(1.0, time_budget))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute smooth gradient
        smooth_val, grad = smooth_grad_fft(p, beta)
        
        # Center gradient for simplex
        grad_centered = grad - np.mean(grad)
        
        # Adaptive step size
        eta = eta_init / (1.0 + 0.01 * it)
        # Also scale by 1/beta since the gradient of logsumexp/beta scales with beta
        # Actually, the gradient of (1/beta)*logsumexp(beta*conv) is:
        # sum_k w_k * d(conv[k])/da where w_k are the softmax weights
        # This doesn't scale with beta (the 1/beta and beta cancel in the derivative).
        # So the magnitude of grad is O(1) regardless of beta.
        # But for large beta, the gradient concentrates on the top peak,
        # so the effective "direction" changes.
        
        # Multiplicative update
        p_new = p * np.exp(-eta * grad_centered)
        p_new = np.maximum(p_new, 1e-30)
        s_new = float(np.sum(p_new))
        if s_new < 1e-10:
            p_new = p.copy()
        else:
            p_new = p_new / s_new
        
        # Evaluate

                stall_count = 0
            else:
                stall_count += 1
        else:
            stall_count += 1
        
        p = p_new
        it += 1
        
        if stall_count > 2000:
            # Reduce step size

            stall_count = 0
            # Or try a perturbation

        if verbose and it % 5000 == 0:

    return best_p

def build_conv_matrix_rows(a):
    """
    Build the matrix A such that A @ g = conv(a, g) for any g.
    A has shape (2n-1, n).
    """
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # conv(a, g)[k] = sum_i a[i] * g[k-i]
        # For fixed j (index in g), g[j] appears in conv[k] when k-j = i, i.e., k = i+j
        # So A[i+j, j] = a[i] for i in [0, n-1] and i+j in [0, 2n-2]
        for i in range(n):
            k = i + j
            if k < m:
                A[k, j] = a[i]
    return A


def solve_lp_minpeak(a):
    """
    LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix_rows(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None

def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw

    """Golden section search for 1D minimization."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(maxiter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2.0
