"""
Smooth gradient descent with tau annealing + LP alternating for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses multiplicative weights (mirror descent)
on the simplex with adaptive step sizing and softmax-based gradient
approximation of the max.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_obj_grad(a, tau):
    """
    Compute the smooth surrogate objective and its gradient.
    f_tau(a) = (1/tau) * log(sum_k exp(tau * c[k])) + c_max
    where c = conv(a, a), and a is normalized to sum=1.
    Returns (objective_value, gradient_vector).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf, np.zeros(n)
    a = a / s
    c = compute_conv(a)
    c_max = float(np.max(c))
    exp_c = np.exp(tau * (c - c_max))
    sum_exp = float(np.sum(exp_c))
    if sum_exp < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_c / sum_exp
    obj_val = (1.0 / tau) * (np.log(sum_exp) + tau * c_max)
    # Gradient computation
    # grad[m] = 2 * sum_k p[k] * a[k-m] for valid k-m in [0, n-1]
    grad = np.zeros(n)
    for m in range(n):
        k_lo = m
        k_hi = min(m + n, 2 * n - 1)
        if k_lo < k_hi:
            grad[m] = 2.0 * float(np.dot(p[k_lo:k_hi], a[:k_hi - k_lo]))
    return obj_val, grad


def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update on the simplex."""
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s


def smooth_gd(a0, time_budget=100.0, tau_start=5.0, tau_end=500.0,
              tau_mult=1.4, verbose=False):
    """
    Smooth gradient descent with tau annealing and multiplicative weights.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    tau = tau_start
    eta = 1.0 / (tau * max(float(np.max(a)), 1e-6))
    
    while tau < tau_end and time.monotonic() - start < time_budget:
        tau_iter_start = time.monotonic()
        no_improve = 0
        max_no_improve = 50
        
        while time.monotonic() - start < time_budget and no_improve < max_no_improve:
            obj_s, grad = smooth_obj_grad(a, tau)
            # Center the gradient (remove mean for better behavior on simplex)
            grad = grad - np.mean(grad)
            
            # Adaptive step size with backtracking
            step_accepted = False
            for attempt in range(20):
                a_try = mirror_descent_step(a, grad, eta)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj or obj_try < obj_s - 1e-10:
                    a = a_try
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                    step_accepted = True
                    break
                else:
                    eta *= 0.5
            
            if not step_accepted:
                # Increase tau if we can't make progress
                break
            
            # Slowly increase eta
            eta *= 1.01
            
            # Check time
            if time.monotonic() - tau_iter_start > 30 and no_improve == 0:
                # Allow more iterations at this tau
                no_improve = 0
        
        # Increase tau
        tau *= tau_mult
        # Reset eta for new tau
        eta = 1.0 / (tau * max(float(np.max(a)), 1e-6))
        
        if verbose:
            print(f"      SmoothGD tau={tau:.1f}: best={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    
    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_norm, M_opt) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=100.0, verbose=False):
    """LP alternating with 1D line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_good_t = 0.5

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)
        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 50, 'xatol': 0.005})
            best_t = res.x if res.success else last_good_t
        except Exception:
            best_t = last_good_t
        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            last_good_t = best_t
            stall_count = 0
            if verbose:
                print(f"      LP: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall_count += 1
            if stall_count > 20:
                break

    return best_a


def coordinate_descent(a0, time_budget=60.0, verbose=False):
    """
    Coordinate-wise descent: try adjusting pairs of coordinates to reduce the peak.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv = compute_conv(a)
    peak_lags = np.argsort(conv)[::-1][:10]
    
    for outer in range(1000):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for peak_k in peak_lags:
            peak_k = int(peak_k)
            # Find pairs (i, j) where moving mass from i to j reduces the peak
            # Contribution of a[m] to conv[k] is a[k-m] (times 2)
            # We want to move mass from high contributors to low contributors
            # Try all pairs where i is a high contributor and j is a low one
            
            contrib = np.zeros(n)
            m_start = max(0, peak_k - n + 1)
            m_end = min(n - 1, peak_k)
            for m in range(m_start, m_end + 1):
                contrib[m] = a[peak_k - m]
            
            # Top contributors and bottom contributors
            top_c = np.argsort(contrib)[::-1][:5]
            bot_c = np.argsort(contrib)[:5]
            
            for src in top_c:
                if a[src] < 1e-10:
                    continue
                for dst in bot_c:
                    if src == dst:
                        continue
                    for frac in [0.02, 0.05, 0.1, 0.2, 0.5]:
                        delta = frac * a[src]
                        if delta > a[src]:
                            continue
                        a[src] -= delta
                        a[dst] += delta
                        st = float(np.sum(a))
                        if st < 1e-10:
                            a[src] += delta
                            a[dst] -= delta
                            continue
                        a_norm = a / st
                        obj_try = compute_obj(a_norm)
                        if obj_try < best_obj - 1e-13:
                            best_obj = obj_try
                            best_a = a_norm.copy()
                            a = a_norm
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[dst] -= delta
                            a = a / st
                            break
            if improved:
                conv = compute_conv(a)
                peak_lags = np.argsort(conv)[::-1][:10]
        
        if not improved and outer > 20:
            break
    
    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

"""
Smooth gradient descent with tau annealing + LP alternating + coordinate descent
for minimizing 2*n*max(conv(a,a))/(sum(a))^2. Uses multiplicative weights (mirror
descent) on the simplex with softmax-based smooth approximation of the max,
FFT-accelerated gradient computation, and adaptive step sizing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_obj_grad(a, tau):
    """
    Compute the smooth surrogate objective and its gradient using FFT-accelerated
    cross-correlation. f_tau(a) = (1/tau)*log(sum exp(tau*c)) + c_max.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf, np.zeros(n)
    a = a / s
    c = compute_conv(a)
    c_max = float(np.max(c))
    exp_c = np.exp(tau * (c - c_max))
    sum_exp = float(np.sum(exp_c))
    if sum_exp < 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_c / sum_exp
    obj_val = (1.0 / tau) * (np.log(sum_exp) + tau * c_max)
    # Gradient via FFT cross-correlation: grad[m] = 2 * sum_k p[k] * a[k-m]
    m_len = len(p)
    N = 1
    while N < m_len + n - 1:
        N <<= 1
    Pf = rfft(p, N)
    Af = rfft(a, N)
    cross = irfft(np.conj(Pf) * Af, N)
    grad = 2.0 * cross[:n]
    return obj_val, grad


def mirror_step(a, grad, eta):
    """Multiplicative weights (mirror descent) step on the simplex."""
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s


def smooth_gd(a0, time_budget=100.0, tau_start=5.0, tau_end=500.0,
              tau_mult=1.3, verbose=False):
    """
    Smooth gradient descent with tau annealing and multiplicative weights.
    Uses backtracking line search for adaptive step sizing.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = tau_start
    while tau <= tau_end and time.monotonic() - start < time_budget:
        tau_start_time = time.monotonic()
        eta = 0.5 / (tau * max(float(np.max(a)), 1e-6))
        no_improve = 0
        max_stall = 80

        while time.monotonic() - start < time_budget and no_improve < max_stall:
            obj_s, grad = smooth_obj_grad(a, tau)
            grad = grad - np.mean(grad)

            step_accepted = False
            for attempt in range(25):

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def line_obj_func(a, g_lp, t):
    """Objective for line search between a and g_lp."""
    a_try = (1.0 - t) * a + t * g_lp
    a_try = np.maximum(a_try, 0)
    st = float(np.sum(a_try))
    if st < 1e-10:
        return np.inf
    a_try = a_try / st
    return compute_obj(a_try)

def coordinate_descent(a0, time_budget=60.0, verbose=False):
    """
    Coordinate-wise descent: adjust pairs to reduce peak convolution values.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = compute_conv(a)
    peak_lags = np.argsort(conv)[::-1][:12]

    for outer in range(2000):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for pk in peak_lags:
            pk = int(pk)
            m_lo = max(0, pk - n + 1)
            m_hi = min(n - 1, pk)
            contrib = np.zeros(n)
            for m in range(m_lo, m_hi + 1):
                contrib[m] = a[pk - m]
            top_c = np.argsort(contrib)[::-1][:6]
            bot_c = np.argsort(contrib)[:6]
            for src in top_c:
                if a[src] < 1e-12:
                    continue
                for dst in bot_c:
                    if src == dst:
                        continue
                    for frac in [0.03, 0.08, 0.15, 0.3, 0.6]:
                        delta = frac * a[src]
                        a[src] -= delta
                        a[dst] += delta
                        st = float(np.sum(a))
                        a_n = a / st

                            a = a_n
                            improved = True
                            break

                            a[src] += delta
                            a[dst] -= delta
                            a = a / st
                            break
            if improved:
                conv = compute_conv(a)
                peak_lags = np.argsort(conv)[::-1][:12]
        if not improved and outer > 30:
            break
        if verbose and outer % 100 == 0:
            print(f"      CD iter={outer}: obj={best_obj:.8f}")

    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

    # Gaussian bump
    gauss = np.exp(-0.5 * ((np.arange(n_base) - n_base / 2.0) / (n_base / 4.0)) ** 2)
    inits.append(gauss / np.sum(gauss))

    # Flat with cosine edges
    edge = max(5, n_base // 12)
    sf = np.ones(n_base)
    for i in range(edge):
        sf[i] = 0.5 * (1.0 - np.cos(np.pi * i / edge))
        sf[n_base - 1 - i] = sf[i]
    inits.append(sf / np.sum(sf))

    # Broad Gaussian (sigma = n/3)
    bg = np.exp(-0.5 * ((np.arange(n_base) - n_base / 2.0) / (n_base / 3.0)) ** 2)
    inits.append(bg / np.sum(bg))

    # Random smooth (boxcar-smoothed)
    rr = rng.random(n_base)
    k = max(3, n_base // 25)
    kk = k if k % 2 == 1 else k + 1
    kernel = np.ones(kk) / kk
    rr_s = np.convolve(rr, kernel, mode='same')
    rr_s = np.maximum(rr_s, 0.01)
    inits.append(rr_s / np.sum(rr_s))

    for i, init in enumerate(inits):
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        obj = compute_obj(init)
        if obj < best_score:
            best_score = obj
            best_seq = init.copy()
            print(f"  Init {i}: obj={best_score:.8f} **")
        else:
            print(f"  Init {i}: obj={obj:.8f}")
        if remaining > 20:
            alloc = min(15.0, remaining * 0.18)
            a_opt = smooth_gd(init, time_budget=alloc, tau_start=5.0, tau_end=150.0)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Init {i}+SGD: obj={best_score:.8f} **")

    # === Phase 2: n-sweep (120s) ===
    print(f"\n=== Phase 2: n-Sweep ===")
    n_options = [200, 300, 400, 500, 600, 800, 1000, 1200, 1500, 2000]
    n_options = sorted(set(n_options + [n_base]))
    n_options = [x for x in n_options if 100 <= x <= 2500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)
        alloc = min(10.0, remaining * 0.15)
        a_opt = smooth_gd(a_try, time_budget=alloc, tau_start=5.0, tau_end=200.0)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 3: Main smooth GD (full tau range) (450s) ===
    print(f"\n=== Phase 3: Main Smooth GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(250.0, remaining * 0.45)
        a_new = smooth_gd(best_seq, time_budget=alloc, tau_start=3.0,
                          tau_end=2000.0, tau_mult=1.25, verbose=True)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Main SGD: obj={best_score:.8f} **")
        else:
            print(f"  Main SGD: obj={obj_new:.8f}")

    # Second smooth GD pass with different params
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(150.0, remaining * 0.3)
        a_new2 = smooth_gd(best_seq, time_budget=alloc, tau_start=50.0,
                           tau_end=3000.0, tau_mult=1.2, verbose=True)
        obj_new2 = compute_obj(a_new2)
        if obj_new2 < best_score - 1e-12:
            best_score = obj_new2
            best_seq = a_new2.copy()
            print(f"  Main SGD2: obj={best_score:.8f} **")
        else:
            print(f"  Main SGD2: obj={obj_new2:.8f}")

    # === Phase 4: LP Alternating (250s) ===
    print(f"\n=== Phase 4: LP Alternating ===")
    for round_idx in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(50.0, remaining * 0.22)
        a_new = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  LP R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP R{round_idx}: obj={obj_new:.8f}")
            # Perturb to escape
            n_cur = len(best_seq)
            best_seq = best_seq * np.exp(rng.normal(0, 0.01, n_cur))
            best_seq = np.maximum(best_seq, 0)
            best_seq = best_seq / np.sum(best_seq)
            if round_idx >= 3:
                break

    # === Phase 5: Coordinate descent (80s) ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(50.0, remaining * 0.35)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score - 1e-12:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 6: LP after CD (60s) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(40.0, remaining * 0.35)
        a_lp = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-12:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP after CD: obj={best_score:.8f} **")
        else:
            print(f"  LP after CD: obj={obj_lp:.8f}")

    # === Phase 7: Smooth GD final polish (60s) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(40.0, remaining * 0.35)
        a_sg = smooth_gd(best_seq, time_budget=alloc, tau_start=100.0,
                         tau_end=5000.0, tau_mult=1.15)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score - 1e-12:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SGD: obj={best_score:.8f} **")
        else:
            print(f"  Final SGD: obj={obj_sg:.8f}")

    # === Phase 8: Random restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        print(f"\n=== Phase 8: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(20):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            scale = 0.008 + 0.004 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr < 1e-10:
                continue
            a_r = a_r / sr
            alloc = min(remaining * 0.25, 18.0)
            if r % 2 == 0:
                a_opt = smooth_gd(a_r, time_budget=alloc * 0.6,
                                  tau_start=20.0, tau_end=1000.0)
                a_opt = lp_alternate(a_opt, time_budget=alloc * 0.3)
            else:
                a_opt = lp_alternate(a_r, time_budget=alloc * 0.5)
                a_opt = smooth_gd(a_opt, time_budget=alloc * 0.4,
                                  tau_start=50.0, tau_end=2000.0)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score - 1e-12:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  R{r}: obj={best_score:.8f} **")
            else:
                print(f"  R{r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
