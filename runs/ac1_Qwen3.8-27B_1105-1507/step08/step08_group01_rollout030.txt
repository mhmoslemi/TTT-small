"""
Multi-n LP-alternation + coordinate descent for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Strategy: explore multiple sequence lengths (emphasizing small n), use LP alternation
with full constraints, and fine coordinate descent polish.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return np.real(irfft(fa * fa, size)[:2 * n - 1])


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    p = a / s
    conv = compute_conv(p)
    return 2.0 * n * float(np.max(conv))


def build_A(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g) (normalized a)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a_norm):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a_norm)
    A = build_A(a_norm)
    
    # Variables: g_0, ..., g_{n-1}, M
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-12:
            return g / s, float(res.x[n])
    return None, None


def lp_alternate(a, max_iters=200, time_budget=100.0, verbose=False):
    """LP alternation with convex line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Precompute for line search
    def obj_blend(p, q, t):
        """Objective of (1-t)*p + t*q, normalized."""
        x = (1 - t) * p + t * q
        x = np.maximum(x, 0)
        s = float(np.sum(x))
        if s < 1e-12:
            return np.inf
        x = x / s
        conv = compute_conv(x)
        return 2.0 * n * float(np.max(conv))
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g, M_lp = solve_lp(a)
        if g is None:
            break
        
        # Line search for optimal t
        # f(t) = obj((1-t)*a + t*g) is convex in t
        lo, hi = 0.0, 1.0
        for _ in range(20):
            m1 = lo + (hi - lo) * 0.382
            m2 = lo + (hi - lo) * 0.618
            f1 = obj_blend(a, g, m1)
            f2 = obj_blend(a, g, m2)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        t_opt = (lo + hi) / 2.0
        
        a_new = (1 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-12:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 10 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}")
        else:
            # Try smaller step
            for t_small in [0.05, 0.02, 0.01]:
                a_try = (1 - t_small) * a + t_small * g
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-12:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                # Stuck, perturb
                rng = np.random.default_rng(it * 7919 + n)
                a_pert = a * np.exp(rng.normal(0, 0.01, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-12:
                    a = a_pert / s_p
                if it > 50:
                    break
    
    return best_a


def coordinate_descent(a, time_budget=60.0, verbose=False):
    """Fine coordinate descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    def try_update(i, delta):
        nonlocal a, best_obj, best_a
        a_new = a.copy()
        a_new[i] = max(0, a_new[i] + delta)
        s_new = float(np.sum(a_new))
        if s_new < 1e-12:
            return False
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            return True
        return False
    
    delta = 1e-4
    pass_count = 0
    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count * 104729 + n)
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            # Binary search for best delta
            for d in [delta, delta * 0.1, delta * 10]:
                if try_update(i, d):
                    improved = True
                    break
                if try_update(i, -d):
                    improved = True
                    break
        
        if verbose and pass_count % 5 == 0:
            print(f"    CD pass {pass_count}: obj={best_obj:.10f}, delta={delta:.2e}")
        
        if not improved:
            delta *= 0.5
            if delta < 1e-8:
                break
    
    return best_a


def init_sequence(n, method='raised_cosine', rng=None):
    """Create initial sequence of length n."""
    if method == 'uniform':
        return np.ones(n)
    elif method == 'raised_cosine':
        j = np.arange(n)
        return np.cos(np.pi * j / n) ** 2 + 0.1
    elif method == 'bimodal':
        j = np.arange(n)
        center = (n - 1) / 2.0
        return 1.0 + 0.5 * np.cos(2 * np.pi * j / n)
    elif method == 'edge_peak':
        j = np.arange(n)
        center = (n - 1) / 2.0
        return 1.0 / (1.0 + ((j - center) / (n * 0.3)) ** 2)
    elif method == 'triangular':
        j = np.arange(n)
        center = (n - 1) / 2.0
        return 1.0 - np.abs(j - center) / (center + 1)
    elif method == 'random':
        if rng is None:
            rng = np.random.default_rng(42)
        return rng.dirichlet(np.ones(n))
    else:
        return np.ones(n)

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return np.real(irfft(fa * fa, size)[:2 * n - 1])


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    p = a / s
    conv = compute_conv(p)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a_norm):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a_norm)
    a_norm = np.asarray(a_norm, dtype=np.float64)
    s = float(np.sum(a_norm))
    if s < 1e-12:
        return None, None

    A = build_conv_matrix(a_norm)
