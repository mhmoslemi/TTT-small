def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

        return v

    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

    # Actually: sum_i a[i] * w[i+j] = sum_i a[i] * w[i+j]
    # Let's define b[i] = w[i] for i = 0,...,2n-2, and 0 otherwise
    # sum_i a[i] * b[i+j] = (a * b_rev)(len(b)-1-j) where * is convolution
    # Hmm, this is getting complicated. Let me just use the direct sum for now.
    # For n up to 2000, O(n^2) = 4M operations per iteration, which is fine.
    
    # Actually, let me use numpy's correlate:
    # np.correlate(a, w, mode='valid') gives sum_i a[i+n-1-k] * w[k] for k=0,...,n-1
    # That's not quite what we want.
    
    # What we want: for each j in 0..n-1: sum_{i=0}^{n-1} a[i] * w[i+j]
    # = dot(a, w[j:j+n])
    
    # We can compute all of these at once using the fact that:
    # result[j] = dot(a, w[j:j+n]) for j = 0, ..., n-1
    # This is like a "sliding window dot product" or "cross-correlation"
    
    # Using FFT:
    # result = ifft(fft(a, 2n) * conj(fft(w, 2n))) ... no, that's not right either.
    
    # Let me think again. We want:
    # result[j] = sum_{i=0}^{n-1} a[i] * w[i+j]
    # This is the (j-th) element of the cross-correlation of a and w.
    # scipy.signal.correlate(w, a, mode='valid') would give:
    # sum_i w[i+j] * a[i] for j = 0, ..., (len(w)-len(a)) = n-1
    # Wait: correlate(w, a, mode='valid') has length len(w) - len(a) + 1 = (2n-1) - n + 1 = n
    # And the j-th element is sum_i w[i+j] * a[i] (for i where both are valid)
    # Actually, scipy.correlate(in1, in2, mode='valid') computes:
    # out[k] = sum_i in1[i+k] * conj(in2[i]) for k = 0, ..., len(in1)-len(in2)
    # So correlate(w, a, mode='valid')[k] = sum_i w[i+k] * a[i]
    # for k = 0, ..., (2n-1) - n = n-1
    # That's exactly what we want!
    
    from scipy.signal import correlate
    result = correlate(w, a, mode='valid')
    return 2.0 * result


def soft_peak_multiplicative_descent(a0, time_budget=60.0, tau_init=50.0, verbose=False):
    """
    Multiplicative descent using log-sum-exp smoothed peak objective.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s  # Normalize to sum=1
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    tau = tau_init
    eta = 0.5  # Step size for multiplicative update
    patience = 0
    no_improve_iter = 0
    
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        conv_max = np.max(conv)
        if conv_max < 1e-15:
            break
        
        # Compute softmax weights
        scaled = tau * (conv - conv_max)
        w = np.exp(scaled)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            w = w / w_sum
        
        # Compute gradient
        grad = soft_peak_grad_efficient(a, w)
        
        # Multiplicative update: a[i] *= exp(-eta * grad[i] / max(grad))
        grad_max = np.max(grad)
        if grad_max < 1e-15:
            break
        a = a * np.exp(-eta * grad / grad_max)
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s
        
        # Check objective periodically
        if it % 5 == 4:
            obj = compute_obj(a)
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_a = a.copy()
                no_improve_iter = 0
                patience = 0
            else:
                no_improve_iter += 1
                patience += 1
            
            if patience > 100:
                # Reduce step size and increase temperature
                eta = max(eta * 0.7, 0.01)
                tau = min(tau * 1.3, 5000.0)
                patience = 0
        
        # Anneal temperature and step size
        if it % 500 == 499:
            tau = min(tau * 1.1, 10000.0)
            eta = max(eta * 0.98, 0.05)
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            obj = compute_obj(a)
            print(f"      SPD it={it}: obj={obj:.8f}, best={best_obj:.8f}, tau={tau:.0f}, eta={eta:.3f}, t={elapsed:.1f}s")
    
    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_norm, M_opt) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate_advanced(a0, time_budget=100.0, verbose=False):
    """
    LP alternating with adaptive update rule.
    Instead of simple convex combination, uses power mapping.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Try different update rules
        best_update_obj = best_obj
        best_update_a = a.copy()
        
        # Rule 1: Convex combination
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_update_obj:
                best_update_obj = obj_try
                best_update_a = a_try.copy()
        
        # Rule 2: Geometric mean
        for t in [0.3, 0.5, 0.7]:
            a_try = np.power(a, 1.0 - t) * np.power(g_lp, t)
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_update_obj:
                best_update_obj = obj_try
                best_update_a = a_try.copy()
        
        # Rule 3: Multiplicative
        for t in [0.1, 0.3, 0.5]:
            a_try = a * np.power(g_lp / (a + 1e-15), t * 0.1)
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_update_obj:
                best_update_obj = obj_try
                best_update_a = a_try.copy()
        
        if best_update_obj < best_obj - 1e-12:
            best_obj = best_update_obj
            best_a = best_update_a.copy()
            a = best_a
            stall_count = 0
            if verbose:
                print(f"      LP-Adv: obj={best_obj:.8f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall_count += 1
            if stall_count > 10:
                break
    
    return best_a


def local_search(a0, time_budget=60.0, verbose=False):
    """
    Local search: try coordinate moves and pairwise transfers.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget:
        improved = False
        # Try random coordinate moves
        for _ in range(100):
            if time.monotonic() - start > time_budget:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            for frac in [0.01, 0.05, 0.1, 0.2]:
                delta = frac * a[i]
                if a[i] < delta + 1e-15:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                st = float(np.sum(a))
                if st < 1e-10:
                    a[i] += delta
                    a[j] -= delta
                    continue
                a = a / st
                obj = compute_obj(a)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    a = a / st
        
        if not improved:
            # Try larger perturbations
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and a[i] > 0.01:
                for frac in [0.3, 0.5, 0.8]:
                    delta = frac * a[i]
                    if a[i] < delta + 1e-15:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    a = np.maximum(a, 0)
                    st = float(np.sum(a))
                    if st < 1e-10:
                        continue
                    a = a / st
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a = best_a.copy()
        
        if not improved:
            # Reset to best and try harder
            a = best_a.copy()
            # Try a structural perturbation: flip a portion
            if n > 10:
                start_idx = int(rng.integers(0, n // 2))
                end_idx = int(rng.integers(n // 2, n))
                seg = a[start_idx:end_idx].copy()
                a[start_idx:end_idx] = seg[::-1]
                a = np.maximum(a, 0)
                st = float(np.sum(a))
                if st > 1e-10:
                    a = a / st
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
        
        if not improved and (it := int((time.monotonic() - start) * 10)) % 5000 < 10:
            # Simulated annealing style: accept worse with small probability
            pass
    
    if verbose:
        print(f"      LocalSearch: obj={best_obj:.8f}, elapsed={time.monotonic()-start:.1f}s")
    return best_a


def resample_seq(a, new_n):
    """Cubic interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if new_n == 1:
        return np.array([a[n // 2]])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use cubic interpolation
    from scipy.interpolate import interp1d
    f = interp1d(x_old, a, kind='cubic', bounds_error=False, fill_value=0)
    result = f(x_new)
    return np.maximum(result, 0)


def construct_u_shape(n, alpha=1.0):
    """Construct a U-shaped distribution."""
    x = np.linspace(0, 1, n)
    # U-shape: high at edges, low in middle
    a = 1.0 + alpha * (2.0 * x - 1.0) ** 2
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def construct_weighted(n, weights_type='cos'):
    """Construct various shaped distributions."""
    if weights_type == 'cos':
        x = np.arange(n) / max(1, n - 1)
        a = 0.5 + 0.5 * np.cos(np.pi * x)
    elif weights_type == 'u_quad':
        x = np.arange(n) / max(1, n - 1)
        a = 1.0 + 4.0 * (x - 0.5) ** 2
    elif weights_type == 'u_cubic':
        x = np.arange(n) / max(1, n - 1)
        a = 1.0 + 8.0 * (x - 0.5) ** 3
        a = np.abs(a)
    elif weights_type == 'sinc':
        x = np.arange(n) - n / 2.0
        x = np.where(x == 0, 1e-10, x)
        a = np.sinc(x / (n / 4.0))
        a = np.maximum(a, 0)
    elif weights_type == 'exp_edges':
        x = np.arange(n) / max(1, n - 1)
        a = np.exp(3.0 * (2.0 * x - 1.0) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)
