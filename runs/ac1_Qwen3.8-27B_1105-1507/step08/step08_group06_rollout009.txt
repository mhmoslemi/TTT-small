    return 2.0 * n * float(np.max(conv))

def compute_cross_corr_with_p(a, p):
    """
    Compute g[i] = sum_j a[j] * p[i+j] for i=0,...,n-1.
    Uses FFT for efficiency.
    """
    n = len(a)
    m = len(p)  # m = 2n-1
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.fft(a, size)
    fp = np.fft.fft(p, size)
    # IFFT(conj(FFT(a)) * FFT(p))[x] = sum_n a[n] * p[n+x]
    g_full = np.fft.ifft(np.conj(fa) * fp).real
    return g_full[:n]

    """
    Compute the projected gradient of the smooth objective.
    Returns the direction to move (subtract from a).
    """
    n = len(a)
    s = float(np.sum(a))
    a_norm = a / s
    conv = compute_conv(a_norm)
    
    # Compute softmax probabilities
    max_c = float(np.max(conv))
    exp_vals = np.exp((conv - max_c) / epsilon)
    p = exp_vals / np.sum(exp_vals)

    g = compute_cross_corr_with_p(a_norm, p)

    grad = g - np.mean(g)
    return grad, conv, s

"""
Smooth optimization for step function minimization using:
1. Log-sum-exp smoothed objective with epsilon annealing
2. Multiplicative weights (exponentiated gradient) updates - O(n log n) per iteration
3. FFT-based gradient computation via cross-correlation
4. LP-based refinement for final polish
5. Multi-restart with diverse initializations and length exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_cross_corr(a, p):
    """
    Compute g[i] = sum_j a[j] * p[i+j] for i=0,...,n-1.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    m = len(p)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.fft(a, size)
    fp = np.fft.fft(p, size)
    g_full = np.fft.ifft(np.conj(fa) * fp).real
    return g_full[:n]


def smooth_gradient(a, epsilon):
    """
    Compute the projected gradient direction for the smooth objective.
    Returns (gradient_direction, conv, sum_a) where gradient_direction = g - mean(g).
    """
    n = len(a)
    s = float(np.sum(a))
    a_norm = a / s
    conv = compute_conv(a_norm)

    # Compute softmax probabilities (numerically stable)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / epsilon
    exp_vals = np.exp(shifted)
    p = exp_vals / np.sum(exp_vals)

    # Compute cross-correlation: g[i] = sum_j a_norm[j] * p[i+j]
    g = compute_cross_corr(a_norm, p)

    # Project gradient to hyperplane sum=0
    grad = g - np.mean(g)
    return grad, conv, s


def mw_optimize(a, time_budget=300.0, epsilon_start=0.5, epsilon_end=0.005,

        # Normalize

    m = 2 * n - 1
    A = np.zeros((m, n))

        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A

        A_eq = np.zeros((1, n + 1))

        if res.status == 0:

            gs = float(np.sum(g_val))

    except Exception:

            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            if verbose:
                print(f"    LP: obj={best_obj:.8f}")
        else:
            # No improvement, try perturbing
            rng = np.random.default_rng(int(time.monotonic() * 1e4) % (2**31))
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s

        # Check time
        if time.monotonic() - start > time_budget:
            break

    return best_a


def generate_init(kind, n, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(0)
    if kind == 'uniform':
        return np.ones(n)
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        return 0.5 * (1 + np.cos(x))
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        return np.sin(x) ** 2 + 0.05
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        return np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif kind == 'random':
        return rng.random(n) + 0.01
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 8)
        a[:edge] = 0.3
        a[-edge:] = 0.3
        return a
    elif kind == 'two_peaks':
        a = np.ones(n) * 0.3
        a[n // 3] = 1.5
        a[2 * n // 3] = 1.5
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        return a
    else:
        return np.ones(n)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining smooth optimization and LP refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0  # 20s buffer

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800)
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"\nStarting search with n={n_parent}, initial obj={best_score:.8f}")

    # === Phase 1: Quick initialization scan (30s) ===
    print(f"\n=== Phase 1: Initialization Scan ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        init_budget = min(30.0, remaining * 0.03)
        init_kinds = ['cosine', 'sine_taper', 'gaussian', 'plateau', 'two_peaks']
        for kind in init_kinds:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            a_init = generate_init(kind, n_parent, rng)
            a_init = a_init / np.sum(a_init)
            obj_init = compute_obj(a_init)
            if obj_init < best_score:
                best_score = obj_init
                best_seq = a_init.copy()
                print(f"  Init '{kind}': obj={obj_init:.8f} **")
            else:
                print(f"  Init '{kind}': obj={obj_init:.8f}")

    # === Phase 2: Main MW optimization with multiple cycles (900s) ===
    print(f"\n=== Phase 2: MW Optimization (Multi-Cycle) ===")
    remaining = deadline - time.monotonic()
    mw_budget = min(900.0, remaining * 0.6)
    a_mw = mw_cycles(best_seq, time_budget=mw_budget, num_cycles=4, verbose=True)
    obj_mw = compute_obj(a_mw)
    if obj_mw < best_score:
        best_score = obj_mw
        best_seq = a_mw.copy()
        print(f"  ** After MW cycles: obj={best_score:.8f} **")
    else:
        print(f"  After MW cycles: obj={obj_mw:.8f} (no improvement)")

    # === Phase 3: More MW with lower epsilon range (200s) ===
    print(f"\n=== Phase 3: Fine MW Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        fine_budget = min(200.0, remaining * 0.15)
        a_fine = mw_optimize(best_seq, time_budget=fine_budget,
                           epsilon_start=0.05, epsilon_end=0.002,
                           anneal_rate=0.998, lr_start=0.5,
                           verbose=True, label="Fine")
        obj_fine = compute_obj(a_fine)
        if obj_fine < best_score:
            best_score = obj_fine
            best_seq = a_fine.copy()
            print(f"  ** Fine MW: obj={best_score:.8f} **")
        else:
            print(f"  Fine MW: obj={obj_fine:.8f}")

    # === Phase 4: LP Refinement (200s) ===
    print(f"\n=== Phase 4: LP Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        lp_budget = min(200.0, remaining * 0.2)
        a_lp = lp_refine(best_seq, time_budget=lp_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 5: Another MW round from LP result (100s) ===
    print(f"\n=== Phase 5: Post-LP MW Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        post_budget = min(100.0, remaining * 0.1)
        a_post = mw_optimize(best_seq, time_budget=post_budget,
                           epsilon_start=0.03, epsilon_end=0.003,
                           anneal_rate=0.998, lr_start=0.3,
                           verbose=False, label="Post")
        obj_post = compute_obj(a_post)
        if obj_post < best_score:
            best_score = obj_post
            best_seq = a_post.copy()
            print(f"  ** Post-LP MW: obj={best_score:.8f} **")
        else:
            print(f"  Post-LP MW: obj={obj_post:.8f}")

    # === Phase 6: Length exploration (80s) ===
    print(f"\n=== Phase 6: Length Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_best = len(best_seq)
        n_options = [n_best - 50, n_best, n_best + 50, n_best + 100]
        n_options = [n for n in n_options if 200 <= n <= 3000 and n != n_best]
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_try = resample_sequence(best_seq, n_try)
            # Quick MW optimization
            a_opt = mw_optimize(a_try, time_budget=min(20.0, remaining * 0.4),
                               epsilon_start=0.1, epsilon_end=0.01,
                               anneal_rate=0.997, lr_start=0.8, verbose=False, label=f"n{n_try}")
            obj_try = compute_obj(a_opt)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final ultra-fine polish (remaining time) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        final_budget = min(remaining * 0.8, 60.0)
        a_final = mw_optimize(best_seq, time_budget=final_budget,
                             epsilon_start=0.02, epsilon_end=0.002,
                             anneal_rate=0.999, lr_start=0.2,
                             verbose=False, label="Final")
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final: obj={best_score:.8f} **")
        else:
            print(f"  Final: obj={obj_final:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
