```python
def spectral_flatten(a, alpha=0.5):
    """
    Apply spectral flattening: reduce peaks in |A|^2 (power spectrum)
    to make b = a*a more uniform.
    """
    n = len(a)
    size = 1
    while size < 2*n:
        size <<= 1
    A = rfft(a, size)
    power = np.abs(A)**2
    # Flatten: power_new = power^(1/alpha) for alpha > 1 reduces peaks
    # Or: power_new = power / (1 + power/max_power * beta)
    max_power = np.max(power)
    if max_power < 1e-15:
        return a.copy()
    # Soft clip
    power_flat = power / (1 + alpha * power / max_power)
    # Reconstruct: we need to find a' such that |A'|^2 ≈ power_flat and a' >= 0
    # Simple approach: scale a by power_flat / power in frequency, then project
    # A_new = A * sqrt(power_flat / power) (preserving phase)
    with np.errstate(divide='ignore', invalid='ignore'):
        ratio = np.sqrt(power_flat / np.maximum(power, 1e-15))
    A_new = A * ratio
    a_new = irfft(A_new, size)[:n]
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new /= s
    return a_new
```

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2n * max_conv / (sum)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def mirror_descent(a, time_budget=300, mu_init=50, mu_max=500, eta_init=0.1,
                   eta_decay=0.999, reheat_every=2000, verbose=True):
    """
    Mirror descent with adaptive temperature for the smoothed max objective.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    mu = mu_init
    eta = eta_init
    momentum = np.zeros(n)
    beta = 0.5  # momentum coefficient
    
    for it in range(100000):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        
        # Compute convolution
        b = compute_conv(a)
        b_max = float(np.max(b))
        
        # Compute softmax weights
        # Use stable log-sum-exp
        b_scaled = mu * (b - b_max)
        w = np.exp(b_scaled)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.zeros(len(b))
            w[int(np.argmax(b))] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient: grad_j = 2 * sum_k w_k * a_{k-j} for valid k-j
        # = 2 * correlate(w, a, 'valid')
        grad = 2.0 * np.correlate(w, a, mode='valid')
        
        # Center gradient (project to sum=0 hyperplane)
        grad_centered = grad - np.mean(grad)
        
        # Momentum
        momentum = beta * momentum - eta * grad_centered
        
        # Multiplicative update (mirror descent step)
        a_new = a * np.exp(momentum)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new
        
        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Adapt parameters
        if it % 100 == 0:
            mu = min(mu * 1.01, mu_max)  # slowly increase temperature
            eta *= eta_decay  # decay learning rate
        
        if it % reheat_every == reheat_every - 1:
            # Reheat: increase eta temporarily
            eta = eta_init * 0.5
            mu = mu_init * 2
            # Perturb
            rng = np.random.default_rng(it)
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            momentum = np.zeros(n)
        
        if verbose and it % 5000 == 0:
            print(f"    MD it={it}, obj={best_obj:.8f}, mu={mu:.1f}, eta={eta:.5f}, time={elapsed:.1f}s")
    
    return best_a

def lp_solve(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    import cvxpy as cp
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    
    # Build constraint matrix
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val /= gs
                    M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                    return g_val, M_val
        except Exception:
            continue
    
    # Fallback: scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val /= gs
            M_val = float(res.x[n])
            return g_val, M_val
    except Exception:
        pass
    
    return None, None

def lp_kick(a, time_budget=60, verbose=True):
    """
    Use LP to get direction, then do multiplicative updates in that direction.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = lp_solve(a)
        if g_lp is None:
            break
        
        # Interpolate towards LP solution
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break
        else:
            # No improvement, perturb
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
    
    if verbose:
        print(f"    LP kick: obj={best_obj:.8f}, time={time.monotonic()-start:.1f}s")
    
    return best_a

def coordinate_descent_polish(a, time_budget=60):
    """
    Polish using pairwise coordinate moves targeting the peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        b = compute_conv(a)
        k_star = int(np.argmax(b))
        
        # Find top contributors to the peak
        contrib = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                contrib[j] = a[j] * a[k_j]
        
        top_idx = np.argsort(contrib)[::-1][:10]
        
        improved = False
        rng = np.random.default_rng(int(time.monotonic() * 1e4) % (2**31))
        
        # Try moving mass from top contributors to random positions
        for frac in [0.01, 0.02, 0.05, 0.1]:
            for i in top_idx:
                if time.monotonic() - start > time_budget:
                    break
                # Move to a position that doesn't contribute to the peak
                target = (int(i) + n // 2) % n
                delta = frac * a[int(i)]
                if delta <= 0 or delta >= a[int(i)]:
                    continue
                a[int(i)] -= delta
                a[target] += delta
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                else:
                    a[int(i)] += delta
                    a[target] -= delta
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
            if improved:
                break
        
        if not improved:
            # Random swap
            for _ in range(10):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.02 * min(a[i], a[j])
                if delta <= 0:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved and (time.monotonic() - start) > time_budget * 0.5:
            # Stagnation: bigger perturbation
            scale = 0.05
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def generate_init(kind, n, rng):
    """Generate initialization."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 10)
        a[:edge] = 0.3
        a[-edge:] = 0.3
        return a / np.sum(a)
    elif kind == 'bimodal':
        a = np.ones(n) * 0.5
        a[n//4] = 2.0
        a[3*n//4] = 2.0
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    else:
        return np.ones(n) / n

```python
"""
Optimized step function search using mirror descent (multiplicative gradient)
with adaptive temperature, LP direction kicks, coordinate polish, and length sweep.
Key innovation: smooth log-sum-exp gradient with multiplicative updates preserves
non-negativity and enables fast convergence via FFT-based convolution and
correlation operations.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv / (sum)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_gradient_smoothed(a, mu=100.0):
    """
    Compute gradient of smoothed max of conv(a,a) using log-sum-exp.
    Returns gradient vector of length n.
    """
    n = len(a)
    b = compute_conv(a)
    b_max = float(np.max(b))
    b_scaled = mu * (b - b_max)
    # Clip for numerical stability
    b_scaled = np.clip(b_scaled, -50, 50)
    w = np.exp(b_scaled)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros(len(b))
        w[int(np.argmax(b))] = 1.0
    else:
        w /= w_sum
    # grad_j = 2 * sum_k w_k * a_{k-j} for 0 <= k-j < n
    # = 2 * correlate(w, a, 'valid')
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad, b


def mirror_descent(a, time_budget=300, mu_init=50, mu_max=2000,
                   eta_init=0.05, eta_min=0.0001, verbose=True):
    """
    Mirror descent (multiplicative gradient) with adaptive temperature.
    Uses log-sum-exp smoothing of the max to get a smooth gradient.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    mu = float(mu_init)
    eta = float(eta_init)
    momentum = np.zeros(n)
    beta = 0.5
    no_improve = 0

    for it in range(500000):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break

        grad, b = compute_gradient_smoothed(a, mu)

        # Center gradient
        grad_c = grad - np.mean(grad)

        # Momentum update
        momentum = beta * momentum - eta * grad_c

        # Multiplicative update (preserves non-negativity)
        step = np.clip(momentum, -2.0, 2.0)
        a_new = a * np.exp(step)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive parameter updates
        if it % 500 == 0:
            # Increase temperature (sharper focus on peaks)
            mu = min(mu * 1.02, mu_max)
            # Decay learning rate
            eta = max(eta * 0.995, eta_min)

        if no_improve > 2000:
            # Reheat with perturbation
            rng = np.random.default_rng(it * 7 + 13)
            a = best_a * np.exp(rng.normal(0, 0.02 + 0.01 * min(no_improve // 2000, 5), n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            momentum = np.zeros(n)
            mu = mu_init * 2
            eta = eta_init * 0.5
            no_improve = 0

        if verbose and it % 2000 == 0:
            print(f"    MD it={it}, best_obj={best_obj:.8f}, mu={mu:.1f}, "
                  f"eta={eta:.6f}, no_imp={no_improve}, t={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def lp_solve(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        g_val /= gs
                        M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                        return g_val, M_val
            except Exception:
                continue
    except Exception:
        pass

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val /= gs
            M_val = float(res.x[n])
            return g_val, M_val
    except Exception:
        pass

    return None, None


def lp_kick(a, time_budget=100, verbose=True):
    """
    Use LP to get optimal response direction, interpolate towards it.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()

def coordinate_polish(a, time_budget=60):
    """
    Coordinate descent: move mass from peak contributors to non-contributing positions.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0

    while time.monotonic() - start < time_budget and stall < 50:
        b = compute_conv(a)
        k_star = int(np.argmax(b))

        # Compute contributions to peak
        contrib = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                contrib[j] = a[j] * a[k_j]

        top_idx = np.argsort(contrib)[::-1][:15]
        improved = False

        for frac in [0.005, 0.01, 0.02, 0.05]:
            if improved:
                break
            for i in top_idx:
                if time.monotonic() - start > time_budget:
                    break
                i = int(i)
                # Target: position far from peak contributor
                target = (i + n // 3) % n
                if abs(target - i) < n // 5:
                    target = (i + 2 * n // 3) % n
                if target == i:
                    target = (i + 1) % n

                delta = frac * a[i]
                if delta <= 0:
                    continue

                a[i] -= delta
                a[target] += delta
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                new_obj = compute_obj(a)

                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] += delta
                    a[target] -= delta
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)

        if not improved:
            # Random pair swap
            for _ in range(20):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.01 * a[i]
                if delta <= 0 or delta > a[i]:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            # Perturb to escape
            if stall % 5 == 0:
                scale = 0.02 + 0.01 * stall
                a = best_a * np.exp(rng.normal(0, min(scale, 0.1), n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_init(kind, n, rng):
    """Generate diverse initialization."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 10)
        a[:edge] = 0.3
        a[-edge:] = 0.3
        return a / np.sum(a)
    elif kind == 'bimodal':
        a = np.ones(n) * 0.5
        a[n // 4] = 2.0
        a[3 * n // 4] = 2.0
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'tapered':
        x = np.linspace(-1, 1, n)
        a = np.exp(-2.0 * x ** 2)
        return a / np.sum(a)
    else:
        return np.ones(n) / n

    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
