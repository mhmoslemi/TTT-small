import numpy as np

def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_grad_at_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        m_start = k - i_end
        m_end = k - i_start
        grad[m_start:m_end + 1] = 2.0 * a[i_start:i_end + 1][::-1]
    return grad


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None

"""
Improved optimizer: Coordinate descent + LP alternating with multi-point line search,
all-peak subgradient, and adaptive n-sweep. Uses stale-convolution coordinate descent
for fast local optimization, combined with LP-based global moves.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_grad_at_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        m_start = k - i_end
        m_end = k - i_start
        grad[m_start:m_end + 1] = 2.0 * a[i_start:i_end + 1][::-1]
    return grad


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None

                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a = best_a.copy()
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved and outer > 20:
            break
        if verbose and outer % 10 == 0:
            print(f"      PR outer={outer}: obj={best_obj:.8f}, el={time.monotonic()-start:.1f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating + coordinate descent + Nesterov + peak reduction.
    Strategy: aggressive LP for global moves, coordinate descent for local polish,
    with n-sweep and random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 600
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n_init) / max(1, n_init - 1)))
        best_seq = a / np.sum(a)
        best_score = compute_obj(best_seq)
        print(f"Hann init (n={n_init}): obj={best_score:.8f}")

    n_cur = len(best_seq)

    # === Phase 1: Initial LP alternating (250s) ===
    print(f"\n=== Phase 1: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(250.0, remaining * 0.4)
        a_lp = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-12:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  After Phase 1 LP: obj={best_score:.8f} **")
        else:
            print(f"  After Phase 1 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Coordinate descent polish (200s) ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(200.0, remaining * 0.35)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score - 1e-12:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  After Phase 2 CD: obj={best_score:.8f} **")
        else:
            print(f"  After Phase 2 CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 3: More LP alternating (250s) ===
    print(f"\n=== Phase 3: LP Alternating (Round 2) ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(250.0, remaining * 0.35)
        a_lp2 = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score - 1e-12:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  After Phase 3 LP: obj={best_score:.8f} **")
        else:
            print(f"  After Phase 3 LP: obj={obj_lp2:.8f} (no improvement)")
            # Perturb and retry
            a_pert = best_seq * np.exp(rng.normal(0, 0.02, n_cur))
            a_pert = np.maximum(a_pert, 0)
            a_pert /= np.sum(a_pert)
            alloc2 = min(100.0, remaining * 0.15)
            a_lp3 = lp_alternate(a_pert, time_budget=alloc2, verbose=True)
            obj_lp3 = compute_obj(a_lp3)
            if obj_lp3 < best_score - 1e-12:
                best_score = obj_lp3
                best_seq = a_lp3.copy()
                print(f"  After Phase 3 LP (perturbed): obj={best_score:.8f} **")

    # === Phase 4: n-sweep (200s) ===
    print(f"\n=== Phase 4: n-Sweep ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        n_cur,
        max(200, n_cur - 100),
        n_cur + 100,
        max(200, n_cur - 200),
        n_cur + 200,
        400, 500, 600, 700, 800, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)
        alloc = min(20.0, remaining * 0.15)
        a_opt = coordinate_descent(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score - 1e-12:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Peak reduction (150s) ===
    print(f"\n=== Phase 5: Peak Reduction ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(150.0, remaining * 0.3)
        a_pr = peak_reduction(best_seq, time_budget=alloc, verbose=True)
        obj_pr = compute_obj(a_pr)
        if obj_pr < best_score - 1e-12:
            best_score = obj_pr
            best_seq = a_pr.copy()
            print(f"  After Phase 5 PR: obj={best_score:.8f} **")
        else:
            print(f"  After Phase 5 PR: obj={obj_pr:.8f} (no improvement)")

    # === Phase 6: Nesterov polish (100s) ===
    print(f"\n=== Phase 6: Nesterov Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for k_val in [3, 5, 10]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(40.0, remaining * 0.3)
            a_ns = nesterov_subgrad(best_seq, time_budget=alloc, top_k=k_val, verbose=True)
            obj_ns = compute_obj(a_ns)
            if obj_ns < best_score - 1e-12:
                best_score = obj_ns
                best_seq = a_ns.copy()
                print(f"  Nest k={k_val}: obj={best_score:.8f} **")
            else:
                print(f"  Nest k={k_val}: obj={obj_ns:.8f}")

    # === Phase 7: Final LP + CD (remaining) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(80.0, remaining * 0.5)
        a_f1 = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score - 1e-12:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc2 = min(50.0, remaining * 0.5)
            a_f2 = coordinate_descent(best_seq, time_budget=alloc2, verbose=True)
            obj_f2 = compute_obj(a_f2)
            if obj_f2 < best_score - 1e-12:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

    # === Phase 8: Random restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 8: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(10):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            scale = 0.01 + 0.003 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr < 1e-10:
                continue
            a_r = a_r / sr
            alloc_cd = min(remaining * 0.4, 25.0)
            a_opt = coordinate_descent(a_r, time_budget=alloc_cd)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score - 1e-12:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  R{r}: obj={best_score:.8f} **")
            else:
                print(f"  R{r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
