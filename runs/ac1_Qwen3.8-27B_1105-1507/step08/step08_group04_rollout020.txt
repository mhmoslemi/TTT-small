"""
Frank-Wolfe (conditional gradient) optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses backtracking line search, multiple initializations, multi-n exploration,
and subgradient polishing with adaptive peak selection.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A

def solve_lp_fast(a, solver=None):
    """Solve the FW subproblem: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    if solver:
        solvers = [solver] + solvers
    
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.round(g.value, 10), 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except Exception:
        pass
    return None

def line_search(a, g, max_evals=30):
    """Backtracking/golden-section line search along a -> g on the simplex."""
    best_t = 2.0 / 4.0  # default FW step
    best_obj = compute_obj(a)
    
    # Try a grid of t values first
    t_candidates = np.concatenate([
        np.linspace(0, 0.5, 11),
        np.linspace(0.5, 1.0, 21)
    ])
    
    best_local_obj = best_obj
    best_local_t = 0.0
    
    for t in t_candidates:
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_local_obj:
            best_local_obj = obj_try
            best_local_t = t
    
    return best_local_t, best_local_obj

def fw_optimize(a, time_budget=100.0, verbose=False, stall_limit=30):
    """Frank-Wolfe with adaptive line search and stall detection."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    no_improve_count = 0
    
    iteration = 0
    while time.monotonic() - start < time_budget:
        g = solve_lp_fast(a)
        if g is None:
            break

def subgradient_optimize(a, time_budget=60.0, top_k=10, verbose=False):
    """Subgradient method with Nesterov acceleration."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = np.zeros(n)
    beta = 0.85
    lr = 1.0 / np.sqrt(n)
    x = a.copy()
    
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(x)
        k_indices = np.argsort(conv)[::-1][:top_k]
        
        # Compute subgradient
        w = np.zeros(2 * n - 1)
        for k in k_indices:
            w[int(k)] = 1.0 / top_k
        
        # grad[j] = (4n/K) * sum_k w[k] * a[k-j] (for valid k-j)
        n_fft = 1
        while n_fft < 4 * n:
            n_fft <<= 1
        A_fft = rfft(a, n_fft)
        W_fft = rfft(w, n_fft)
        R = irfft(np.conj(A_fft) * W_fft, n_fft)
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = R[(n_fft - j) % n_fft]
        grad *= 4.0 * n
        grad -= np.mean(grad)
        
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        t_k = 2.0 / (it + 2.0)
        x = a_new + (1 - t_k) * (a_new - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx
        
        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
        
        if it % 500 == 499:
            lr *= 0.95
    
    return best_a

def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_initialization(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)

        a = np.exp(-rng.uniform(2, 8) * x)

        # Cosine bell
        a = 0.5 + 0.5 * np.cos(np.pi * x)

        # Gaussian centered

        a = np.exp(-((x - center) / width) ** 2)
    elif choice == 4:
        # Two bumps
        c1, c2 = sorted(rng.uniform(0, 1, 2))
        a = np.exp(-((x - c1)/0.2)**2) + np.exp(-((x - c2)/0.2)**2)

        freq = rng.uniform(1, 10)
        a = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x)
        a = np.maximum(a, 0.01)
    elif choice == 6:
        # Power law
        a = (x + 0.01) ** rng.uniform(-2, 3)
        a = np.maximum(a, 0)
    else:
        # Random smooth
        a = np.ones(n)
        for _ in range(rng.integers(2, 6)):
            center = rng.integers(0, n)
            width = max(1, rng.integers(n // 8, n // 3))
            for i in range(n):
                d = min(abs(i - center), n - abs(i - center))
                a[i] *= 1.0 + rng.standard_normal() * 0.5 * np.exp(-d / (width * 0.5))

    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

"""
Frank-Wolfe (conditional gradient) optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous approach:
- Adaptive golden-section line search for FW steps
- Multi-phase strategy with diverse initializations (exponential decay, power law, etc.)
- Aggressive stall detection with structural perturbations
- Multi-n exploration with early termination
- Subgradient polishing with adaptive top-k
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2. Assumes a is non-negative."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        lengths = k_end - k_start + 1
        A[k_start:k_end + 1, j] = a[:lengths]
    return A


def solve_lp_fast(a):
    """Solve FW subproblem: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns normalized g or None."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs
        except Exception:
            continue

    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * (n + 1)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0.0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except Exception:
        pass
    return None


def line_search(a, g, n_evals=40):
    """Adaptive line search: coarse grid then golden-section refinement."""
    best_t = 0.0
    best_obj = compute_obj(a)

    # Coarse grid
    t_grid = np.linspace(0.0, 1.0, n_evals)
    coarse_best_t = 0.0
    coarse_best_obj = best_obj
    for t in t_grid:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0.0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < coarse_best_obj:
            coarse_best_obj = obj_try
            coarse_best_t = t

    if coarse_best_obj >= best_obj:
        return 0.0, best_obj

    # Golden section refinement around the best grid point
    idx = int(coarse_best_t * (n_evals - 1))
    lo = t_grid[max(0, idx - 2)]
    hi = t_grid[min(n_evals - 1, idx + 2)]

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(20):
        if hi - lo < 1e-6:
            break
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        a_c1 = np.maximum((1.0 - c1) * a + c1 * g, 0.0)
        s_c1 = float(np.sum(a_c1))
        obj_c1 = compute_obj(a_c1 / s_c1) if s_c1 > 1e-10 else np.inf
        a_c2 = np.maximum((1.0 - c2) * a + c2 * g, 0.0)
        s_c2 = float(np.sum(a_c2))
        obj_c2 = compute_obj(a_c2 / s_c2) if s_c2 > 1e-10 else np.inf
        if obj_c1 < obj_c2:
            hi = c2
        else:
            lo = c1

    t_refined = (lo + hi) / 2.0
    a_ref = np.maximum((1.0 - t_refined) * a + t_refined * g, 0.0)
    s_ref = float(np.sum(a_ref))
    obj_ref = compute_obj(a_ref / s_ref) if s_ref > 1e-10 else np.inf

    if obj_ref < coarse_best_obj:
        return t_refined, obj_ref
    return coarse_best_t, coarse_best_obj


def fw_optimize(a, time_budget=100.0, verbose=False, stall_limit=25):
    """Frank-Wolfe with golden-section line search and stall detection."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    iteration = 0
    rng = np.random.default_rng(12345 + n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if time_budget - elapsed < 2.0:
            break

        g = solve_lp_fast(a)
        if g is None:
            break

        t_best, obj_best = line_search(a, g)

        if obj_best < best_obj - 1e-14:
            a_try = (1.0 - t_best) * a + t_best * g
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                best_obj = obj_best
                best_a = a_try.copy()
                a = a_try
            stall = 0
        else:
            stall += 1

        iteration += 1

        if verbose and iteration % 100 == 0:
            print(f"    FW: it={iteration}, obj={best_obj:.8f}, stall={stall}, t={time.monotonic()-start:.0f}s")

        if stall > stall_limit:
            # Aggressive perturbation
            pert_type = stall // stall_limit
            if pert_type <= 1:
                scale = 0.08
                a = best_a * np.exp(rng.normal(0, scale, n))
            elif pert_type == 2:
                # Structural: blend with a random shape
                x = np.linspace(0, 1, n)
                shape = np.exp(-rng.uniform(2, 8) * x)
                a = 0.5 * best_a + 0.5 * shape
            else:
                # Heavy restart
                a = diverse_initialization(n, rng)
            a = np.maximum(a, 0.0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            stall = 0

    return best_a


def subgradient_optimize(a, time_budget=60.0, top_k=10, verbose=False):
    """Nesterov-accelerated subgradient on the simplex."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.85
    lr = 1.5 / np.sqrt(n)
    x = a.copy()

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(x)
        k_indices = np.argsort(conv)[::-1][:top_k]

        w = np.zeros(2 * n - 1)
        for k in k_indices:
            w[int(k)] = 1.0 / top_k

        n_fft = 1
        while n_fft < 4 * n:
            n_fft <<= 1
        A_fft = rfft(a, n_fft)
        W_fft = rfft(w, n_fft)
        R = irfft(np.conj(A_fft) * W_fft, n_fft)
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = R[(n_fft - j) % n_fft]
        grad *= 4.0 * n
        grad -= np.mean(grad)

        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        t_k = 2.0 / (it + 2.0)
        x = a_new + (1.0 - t_k) * (a_new - a)
        x = np.maximum(x, 0.0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        if it % 1000 == 999:
            lr *= 0.92

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def diverse_initialization(n, rng):
    """Generate diverse structured initializations motivated by low-peak-autocorrelation."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-rng.uniform(3.0, 10.0) * x)
    elif choice == 1:
        alpha = rng.uniform(0.5, 4.0)
        a = (1.0 - x) ** alpha
    elif choice == 2:
        a = 0.5 + 0.5 * np.cos(np.pi * x)
    elif choice == 3:
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        a = np.exp(-((x - center) / width) ** 2)
    elif choice == 4:
        c1, c2 = sorted(rng.uniform(0.1, 0.9, 2))
        a = np.exp(-((x - c1) / 0.15) ** 2) + np.exp(-((x - c2) / 0.15) ** 2)
    elif choice == 5:
        freq = rng.uniform(0.5, 5.0)
        a = 0.6 + 0.4 * np.cos(2.0 * np.pi * freq * x)
        a = np.maximum(a, 0.01)
    elif choice == 6:
        gamma = rng.uniform(1.0, 5.0)
        a = np.exp(-gamma * x) + 0.1
    elif choice == 7:
        # Smooth random
        a = np.ones(n)
        for _ in range(rng.integers(3, 7)):
            center = rng.integers(0, n)
            width = max(2, int(rng.integers(n // 8, n // 3)))
            amp = rng.uniform(-0.5, 0.5)
            for i in range(n):
                d = min(abs(i - center), n - abs(i - center))
                a[i] += amp * np.exp(-d / (width * 0.5))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        # Power law decay
        beta = rng.uniform(0.5, 3.0)
        a = (1.0 - x + 0.01) ** (-beta)
        a = a / np.max(a)
    else:
        # Exponential with oscillation
        gamma = rng.uniform(1.0, 6.0)
        freq = rng.uniform(1.0, 8.0)
        a = np.exp(-gamma * x) * (1.0 + 0.3 * np.sin(2.0 * np.pi * freq * x))
        a = np.maximum(a, 0.0)

    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase Frank-Wolfe with diverse initializations,
    line search, multi-n exploration, and subgradient polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 35.0

    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        print(f"Parent loaded: n={len(a)}, obj={compute_obj(a):.8f}")
    except Exception:
        a = np.ones(800) / 800.0
        print(f"No parent available, using uniform n=800, obj={compute_obj(a):.8f}")

    best_a = a.copy()
    best_obj = compute_obj(a)
    n_base = len(best_a)

    # === Phase 1: FW on parent ===
    print(f"\n=== Phase 1: FW on parent (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.28, 320.0)
    a_p1 = fw_optimize(best_a, time_budget=p1_budget, verbose=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1
        print(f"  Phase 1: obj={best_obj:.8f} **")
    else:
        print(f"  Phase 1: no improvement (obj={obj_p1:.8f})")

    # === Phase 2: Subgradient polish ===
    print(f"\n=== Phase 2: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for tk in [5, 15, 25]:
            if time.monotonic() > deadline - 25:
                break
            alloc = min(50.0, remaining * 0.08)
            a_sg = subgradient_optimize(best_a, time_budget=alloc, top_k=tk)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg
                print(f"  SubGrad k={tk}: obj={best_obj:.8f} **")
            else:
                print(f"  SubGrad k={tk}: obj={obj_sg:.8f}")

    # === Phase 3: FW Round 2 from improved point ===
    print(f"\n=== Phase 3: FW Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p3_budget = min(remaining * 0.22, 200.0)
        a_p3 = fw_optimize(best_a, time_budget=p3_budget, verbose=True)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3
            print(f"  Phase 3: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 3: no improvement (obj={obj_p3:.8f})")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        max(200, n_base - 200), max(200, n_base - 100),
        n_base, n_base + 100, n_base + 200,
        n_base + 400, n_base + 600,
    ]))
    n_options = [x for x in n_options if 200 <= x <= 6000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        if n_try == len(best_a):
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try

        alloc = min(55.0, remaining * 0.20)
        a_opt = fw_optimize(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse initializations ===
    print(f"\n=== Phase 5: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    for d in range(15):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break
        a_div = diverse_initialization(n_best, rng)
        alloc = min(55.0, remaining * 0.18)
        a_opt = fw_optimize(a_div, time_budget=alloc)
        obj_div = compute_obj(a_opt)
        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Init {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Init {d}: obj={obj_div:.8f}")

    # === Phase 6: Try different n from best diverse result ===
    print(f"\n=== Phase 6: Cross-n Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_current = len(best_a)
        for delta in [-100, 100, 200]:
            n_try = n_current + delta
            if n_try < 200 or n_try > 6000:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            a_try = resample_seq(best_a, n_try)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
            alloc = min(45.0, remaining * 0.3)
            a_opt = fw_optimize(a_try, time_budget=alloc)
            obj_try = compute_obj(a_opt)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_opt.copy()
                print(f"  ** Cross-n {n_try}: obj={best_obj:.8f} **")
            else:
                print(f"  Cross-n {n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.35, 70.0)
        a_final = fw_optimize(best_a, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final FW: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 35.0)
        a_sg = subgradient_optimize(best_a, time_budget=alloc, top_k=8)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"  Final SG: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 25.0)
        a_last = fw_optimize(best_a, time_budget=alloc)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Last FW: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100) / 100.0
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")
    return result
