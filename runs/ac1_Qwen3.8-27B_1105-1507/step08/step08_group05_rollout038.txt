"""
Refined optimizer for self-convolution peak minimization.
Uses adaptive active-set subgradient with backtracking line search,
Nesterov momentum with restarts, n-sweeping, and local coordinate refinement.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
from scipy.optimize import minimize


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def conv_gradient(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a. Returns vector g where g[m] = 2*a[k-m] if valid."""
    n = len(a)
    g = np.zeros(n)
    m_start = max(0, k - n + 1)
    m_end = min(n - 1, k)
    if m_start <= m_end:
        # g[m] = 2*a[k-m] for m in [m_start, m_end]
        # When m = m_start, k-m = k-m_start; when m = m_end, k-m = k-m_end
        # So g[m_start:m_end+1] = 2*a[k-m_end : k-m_start+1][::-1]
        idx_start = k - m_end
        idx_end = k - m_start
        g[m_start:m_end + 1] = 2.0 * a[idx_start:idx_end + 1][::-1]
    return g

def coordinate_descent(a0, time_budget=60.0, verbose=False):
    """
    Coordinate descent: optimize one element at a time.
    For each coordinate i, find the best value of a[i] (with others fixed).

    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:

    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Precompute: for each coordinate i, the effect on conv
    # conv(a,a)[k] depends on a[i] through terms involving a[i]
    # Specifically, conv[k] = sum_j a[j]*a[k-j], and a[i] appears when
    # j=i (giving a[i]*a[k-i]) or k-j=i (giving a[k-i]*a[i])
    
    # For coordinate descent on a[i]:
    # conv[k] = base_k + a[i]*(a[k-i] + a[k-i] if k-2i != k... )
    # Actually it's cleaner to just recompute conv after each change
    
    # Strategy: for each i, try several candidate values and pick the best
    no_improve = 0
    for sweep in range(200):
        if time.monotonic() - start > time_budget:
            break
        sweep_improved = False
        # Random order for diversity
        indices = np.random.permutation(n) if sweep % 3 == 0 else np.arange(n)
        
        for i in indices:
            if time.monotonic() - start > time_budget:
                break
            # Try several candidate values for a[i]
            current_val = a[i]
            candidates = [
                current_val * 0.5,

                current_val * 2.0,
                current_val + 0.001,

            best_candidate = current_val

def slsqp_polish(a0, time_budget=30.0):
    """
    Use SLSQP to polish the solution. Only effective for small n.
    """
    n = len(a0)

    s = float(np.sum(a))

        return a0.copy()
    a = a / s
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Define objective and gradient

        if time.monotonic() - start > time_budget:
            return best_obj, np.zeros(n)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx < 1e-10:
            return np.inf, np.zeros(n)
        x = x / sx
        conv = compute_conv(x)
        max_val = conv.max()
        obj = 2.0 * n * max_val
        # Subgradient
        max_idx = np.argmax(conv)
        g = conv_gradient(x, int(max_idx))
        # Adjust for the normalization (chain rule)
        g = g - obj / sx * np.ones(n)  # approximate
        return obj, g
    
    from scipy.optimize import minimize as sp_minimize
    try:
        result = sp_minimize(
            obj_and_grad, a, method='SLSQP',
            bounds=[(0, None)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - 1}],
            options={'maxiter': 200, 'ftol': 1e-10}
        )
        if result.x is not None:
            a_new = np.maximum(result.x, 0)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                return a_new
    except Exception as e:
        pass
    
    return a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

"""
Refined optimizer for self-convolution peak minimization.
Uses adaptive active-set subgradient with backtracking line search,
Nesterov momentum with restarts, active-set Newton steps, coordinate descent,
and n-sweeping to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
from scipy.optimize import minimize as sp_minimize, minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_gradient(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a: g[m] = 2*a[k-m] if 0<=k-m<n."""
    n = len(a)
    g = np.zeros(n)
    m_start = max(0, k - n + 1)
    m_end = min(n - 1, k)
    if m_start <= m_end:
        idx_start = k - m_end
        idx_end = k - m_start
        g[m_start:m_end + 1] = 2.0 * a[idx_start:idx_end + 1][::-1]
    return g


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def fast_subgradient(a0, time_budget=60.0, verbose=False, seed=42, peak_tol=0.02):
    """
    Nesterov-accelerated subgradient with adaptive peak selection and line search.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    step_size = 0.1
    patience = 0
    max_patience = 300
    prev_grad = None

    for it in range(500000):
        if time.monotonic() - start > time_budget:
            break

        conv_y = compute_conv(y)
        max_val = conv_y.max()

        # Adaptive active set
        threshold = max_val * (1.0 - peak_tol)
        active_idx = np.where(conv_y >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(conv_y)])
        if len(active_idx) > 40:
            vals = conv_y[active_idx]
            top_n = min(20, len(active_idx))
            order = np.argsort(vals)[::-1][:top_n]
            active_idx = active_idx[order]

        # Compute subgradient
        grad = np.zeros(n)
        for k in active_idx:
            grad += conv_gradient(y, int(k))
        grad /= len(active_idx)
        grad = grad - np.mean(grad)

        # Momentum restart: reset if direction reversed
        if prev_grad is not None and np.dot(grad, prev_grad) < 0:
            y = a.copy()
        prev_grad = grad.copy()

        # Backtracking line search
        t = step_size
        found = False
        for _ in range(8):
            a_trial = project_simplex(y - t * grad)
            obj_trial = compute_obj(a_trial)
            if obj_trial < best_obj:
                found = True
                break
            t *= 0.5

        if found:
            a_new = a_trial
            obj_new = obj_trial
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                patience = 0
            else:
                patience += 1
            t_k = 2.0 / (it + 2.0)
            y = a_new + (1.0 - t_k) * (a_new - a)
            y = project_simplex(y)
            a = a_new
        else:
            step_size = max(step_size * 0.7, 0.003)
            patience += 1
            if patience > max_patience:
                break

        # Adapt parameters
        if it % 200 == 0:
            peak_tol = max(0.005, peak_tol * 0.97)
        if it % 1000 == 0:
            step_size = min(step_size * 1.1, 0.15)

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"      SG it={it}: best={best_obj:.8f}, step={step_size:.4f}, "
                  f"peaks={len(active_idx)}, t={elapsed:.1f}s")

    return best_a


def active_set_newton(a0, time_budget=30.0, verbose=False):
    """
    Active-set Newton: for the current set of active peaks, solve the
    linearized system to find a Newton direction that equalizes peaks.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for outer in range(100):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = conv.max()
        # Active set: peaks within 0.1% of max
        active_idx = np.where(conv >= max_val * 0.999)[0]
        if len(active_idx) < 2:
            active_idx = np.argsort(conv)[::-1][:3]
        active_idx = active_idx[:min(len(active_idx), n)]  # Cap at n

        # Build linearized system: for each active peak k, 
        # the gradient direction P_k @ a should be in null space
        # We want delta such that:
        #   sum(delta) = 0 (maintain normalization)
        #   delta^T (P_k a) = 0 for k in active set (first-order peak preservation)
        # And delta should reduce the quadratic (second-order) term

        # P_k @ a gives the gradient of conv[k] / 2
        # So g_k = conv_gradient(a, k) / 2 = the "autocorrelation-like" vector
        G = np.zeros((len(active_idx) + 1, n))
        for i, k in enumerate(active_idx):
            G[i] = conv_gradient(a, int(k))  # This is 2 * (P_k a)
        G[-1] = 1.0  # sum constraint

        # Find null space of G: delta in null(G)
        # Use SVD
        try:
            U, svals, Vt = np.linalg.svd(G, full_matrices=True)
            # Null space vectors are the last rows of Vt
            rank = np.sum(svals > 1e-10 * svals[0])
            null_dim = n - rank
            if null_dim <= 0:
                break
            null_vecs = Vt[rank:, :]  # shape (null_dim, n)

            # Among null space vectors, find the one that best reduces the objective
            # The second-order effect of delta on conv[k] is delta^T P_k delta
            # We want to minimize max_k (conv[k] + 2*delta^T P_k a + delta^T P_k delta)
            # Since delta is in null space, the first-order term vanishes
            # So we want to minimize max_k delta^T P_k delta (the second-order term)

            best_dir = None
            best_score = np.inf
            for j in range(null_dim):
                d = null_vecs[j]
                d = d - np.mean(d)  # Ensure zero sum (redundant but safe)
                if np.max(np.abs(d)) < 1e-15:
                    continue
                # Evaluate second-order effect
                a2 = a + 0.01 * d
                a2 = np.maximum(a2, 0)
                a2 = a2 / np.sum(a2)
                obj2 = compute_obj(a2)
                if obj2 < best_score:
                    best_score = obj2
                    best_dir = d
                # Also try negative direction
                a2_neg = a - 0.01 * d
                a2_neg = np.maximum(a2_neg, 0)
                a2_neg = a2_neg / np.sum(a2_neg)
                obj2_neg = compute_obj(a2_neg)
                if obj2_neg < best_score:
                    best_score = obj2_neg
                    best_dir = -d

            if best_dir is None:
                break

            # Line search along best_dir
            best_dir = best_dir / np.linalg.norm(best_dir)

            def line_obj(t):
                a_t = a + t * best_dir
                a_t = np.maximum(a_t, 0)
                st = np.sum(a_t)
                if st < 1e-10:
                    return np.inf
                a_t = a_t / st
                return compute_obj(a_t)

            try:
                res = minimize_scalar(line_obj, bounds=(-1.0, 1.0), method='bounded',
                                      options={'maxiter': 30, 'xatol': 0.01})
                t_opt = res.x
            except Exception:
                t_opt = 0.01

            a_new = a + t_opt * best_dir
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)

            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                if verbose:
                    print(f"      ASN iter={outer}: obj={best_obj:.8f}, |delta|={abs(t_opt):.4f}")
            else:
                break

        if verbose and outer % 20 == 0:
            print(f"      ASN iter={outer}: best={best_obj:.8f}")

    return best_a


def coordinate_descent(a0, time_budget=60.0, verbose=False, seed=42):
    """
    Coordinate descent: for each position, try candidate values and keep the best.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_obj = compute_obj(a)
    start = time.monotonic()

    no_improve_sweeps = 0
    for sweep in range(500):
        if time.monotonic() - start > time_budget:
            break
        improved = False

        # Use sequential order for first few sweeps, random after
        if sweep < 3:
            indices = np.arange(n)
        else:
            indices = rng.permutation(n)

        for i in indices:
            if time.monotonic() - start > time_budget:
                break
            current = a[i]
            if current < 1e-12:
                # Try setting to small positive
                candidates = [1e-4, 1e-3, 5e-3]
            else:
                candidates = [current * 0.5, current * 0.7, current * 1.3,
                              current * 2.0, 0.0, current * np.exp(rng.normal(0, 0.5))]

            best_cand = current
            best_local = best_obj
            rest_sum = s - current

            for cv in candidates:
                cv = max(0.0, cv)
                new_sum = rest_sum + cv
                if new_sum < 1e-10:
                    continue
                a[i] = cv
                a_norm = a / new_sum
                obj_try = compute_obj(a_norm)
                if obj_try < best_local - 1e-13:
                    best_local = obj_try
                    best_cand = cv

            if best_cand != current:
                a[i] = best_cand
                a = a / np.sum(a)
                # s is still 1 after normalization
                best_obj = best_local
                improved = True

        if improved:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 3:
                break

        if verbose and sweep % 10 == 0:
            print(f"      CD sweep={sweep}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return a


def resample_seq(a, new_n):
    """Linear interpolation resampling to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def slsqp_polish(a0, time_budget=30.0):
    """SLSQP polish for small sequences (n <= 200)."""
    n = len(a0)
    if n > 200:
        return a0.copy()

    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_obj = compute_obj(a)
    start = time.monotonic()

    def obj_fn(x):
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx < 1e-10:
            return 1e10
        x = x / sx
        conv = compute_conv(x)
        return 2.0 * n * conv.max()

    def obj_grad(x):
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx < 1e-10:
            return 1e10, np.zeros(n)
        x = x / sx
        conv = compute_conv(x)
        max_k = np.argmax(conv)
        g = conv_gradient(x, int(max_k))
        # Account for normalization: d/da [2n*max(c)/s^2] where s=sum(a)
        # = 2n * (max(c)'*s - max(c)*1) / s^3 (roughly)
        # Simpler: just use the subgradient of the unnormalized version
        g = g / sx  # approximate chain rule for normalization
        return 2.0 * n * conv.max() / sx, g

    try:
        result = sp_minimize(
            obj_grad, a, method='SLSQP',
            jac=True,
            bounds=[(0, None)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0}],
            options={'maxiter': 100, 'ftol': 1e-12}
        )
        if result.x is not None:
            a_new = np.maximum(result.x, 0)
            a_new = a_new / np.sum(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                return a_new
    except Exception:
        pass

    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining adaptive subgradient, active-set Newton,
    coordinate descent, n-sweeping, and random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n_init) / (n_init - 1)))
        best_seq = w / np.sum(w)
        best_score = compute_obj(best_seq)
        print(f"Hann init (n=800): obj={best_score:.8f}")

    print(f"\n=== Phase 1: n-Exploration ===")

            break

        a_try = a_try / np.sum(a_try)
        alloc = min(12.0, remaining - 5)
        a_opt = fast_subgradient(a_try, time_budget=alloc, seed=seed + n_try)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
