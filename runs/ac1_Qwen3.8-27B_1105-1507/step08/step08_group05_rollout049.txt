def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_grad_at_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a[m]."""

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None

def resample_seq(a, new_n):
    """Cubic interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if new_n == 1:
        return np.array([a.sum()])
    # Use scipy interp1d for cubic
    from scipy.interpolate import interp1d
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    return np.maximum(f(x_new), 0)


def smooth_obj(a, beta=50.0):
    """Smooth approximation of max using log-sum-exp."""
    conv = compute_conv(a)
    # Shift for numerical stability
    conv_shifted = conv - conv.max()
    return 2.0 * len(a) * (conv.max() + np.log(np.sum(np.exp(beta * conv_shifted))) / beta)

def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:

def conv_grad_at_k(a, k):

def project_simplex(v):

    if not np.any(cond):

    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

    return A


def solve_lp_peak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_topk(a, top_k=5):
    """LP: minimize weighted sum of top-k peaks of conv(a,g)."""
    n = len(a)
    a_n = a / float(np.sum(a))
    # Find current top peaks
    conv_a = compute_conv(a_n)
    top_idx = np.argsort(conv_a)[::-1][:top_k]
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(g) == 1]
    # Minimize sum of top-k correlation values
    obj = cp.sum([A[k] @ g for k in top_idx]) / top_k
    prob = cp.Problem(cp.Minimize(obj), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs
        except Exception:
            continue
    return None


def lp_alternate(a0, time_budget=100.0, use_topk=False, top_k=5):
    """LP alternating with 1D line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - start < time_budget:
        if use_topk:
            g_lp = solve_lp_topk(a, top_k=top_k)
        else:
            g_lp, _ = solve_lp_peak(a)
        if g_lp is None:
            break

        # 1D line search with golden section refinement
        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)

        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 60, 'xatol': 0.0005})
            best_t = res.x
        except Exception:
            best_t = 0.3

        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 8:
                # Try smaller steps
                found = False
                for t_small in [0.1, 0.2, 0.05, 0.3]:
                    a_try2 = (1.0 - t_small) * a + t_small * g_lp
                    a_try2 = np.maximum(a_try2, 0)
                    st2 = float(np.sum(a_try2))
                    if st2 < 1e-10:
                        continue
                    a_try2 = a_try2 / st2
                    obj2 = compute_obj(a_try2)
                    if obj2 < best_obj - 1e-14:
                        best_obj = obj2
                        best_a = a_try2.copy()
                        a = best_a
                        stall_count = 0
                        found = True
                        break
                if not found and stall_count > 15:
                    break
                if not found:
                    # Perturb and continue
                    a = best_a * np.exp(rng.normal(0, 0.008, n))
                    a = np.maximum(a, 0)
                    a /= float(np.sum(a))

    return best_a


def resample_seq(a, new_n):
    """Cubic interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    if new_n == 1:
        return np.array([float(a.sum())])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    return np.maximum(f(x_new), 0)


def coordinate_descent(a0, time_budget=60.0, n_pairs=200):
    """Coordinate descent: try moving mass between pairs of coordinates."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        # Random pairs
        improved = False
        for _ in range(n_pairs):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-12:
                continue
            for frac in [0.02, 0.05, 0.1, 0.01]:
                delta = frac * a[i]
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                st = float(np.sum(a))
                if st < 1e-10:
                    a[i] += delta
                    a[j] -= delta
                    continue
                a = a / st
                obj_try = compute_obj(a)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    a = a / st
                    break
        if not improved and it > 50:
            break
        a = best_a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimizer: LP alternating (peak + topk), Nesterov subgradient,
    coordinate descent, and adaptive n-selection.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Uniform init (n=800): obj={best_score:.8f}")

    # === Phase 1: n-sweep with quick Nesterov (100s) ===
    print(f"\n=== Phase 1: n-Sweep ===")
    n_base = len(best_seq)
    n_options = sorted(set([
        200, 300, 400, 500, 600, 700, 800, 900, 1000, 1200, 1500, 2000,
        n_base, max(100, n_base - 100), max(100, n_base - 200),
        min(3000, n_base + 100), min(3000, n_base + 200)
    ]))
    n_options = [n for n in n_options if 80 <= n <= 3000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 8:
            break
        if n_try == n_base:
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)
        alloc = min(6.0, remaining * 0.12)
        a_opt = nesterov_subgrad(a_try, time_budget=alloc, top_k=3, seed=n_try)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score - 1e-14:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 2: Main LP Alternating - Peak (400s) ===
    print(f"\n=== Phase 2: LP Peak Alternating ===")
    a = best_seq.copy()
    n_cur = len(a)

    for round_idx in range(12):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = min(50.0, remaining * 0.2)
        a_new = lp_alternate(a, time_budget=alloc, use_topk=False)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-14:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new
            print(f"  LP-Peak R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP-Peak R{round_idx}: obj={obj:.8f}")
            # Perturb to escape
            a = best_seq * np.exp(rng.normal(0, 0.008, n_cur))
            a = np.maximum(a, 0)
            a /= float(np.sum(a))

    # === Phase 3: LP Alternating - TopK (250s) ===
    print(f"\n=== Phase 3: LP TopK Alternating ===")
    a = best_seq.copy()
    n_cur = len(a)

    for round_idx in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = min(50.0, remaining * 0.2)
        a_new = lp_alternate(a, time_budget=alloc, use_topk=True, top_k=5)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-14:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new
            print(f"  LP-TopK R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP-TopK R{round_idx}: obj={obj:.8f}")
            a = best_seq * np.exp(rng.normal(0, 0.006, n_cur))
            a = np.maximum(a, 0)
            a /= float(np.sum(a))

    # === Phase 4: Nesterov polish with various top_k (200s) ===
    print(f"\n=== Phase 4: Nesterov Polish ===")
    for k_val in [3, 5, 8, 12]:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(50.0, remaining * 0.25)
        a_ns = nesterov_subgrad(best_seq, time_budget=alloc, top_k=k_val, seed=100+k_val)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score - 1e-14:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nest k={k_val}: obj={best_score:.8f} **")
        else:
            print(f"  Nest k={k_val}: obj={obj_ns:.8f}")

    # === Phase 5: Coordinate Descent (100s) ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(60.0, remaining * 0.3)
        a_cd = coordinate_descent(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score - 1e-14:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 6: LP + Nesterov interleaved (100s) ===
    print(f"\n=== Phase 6: Interleaved LP+Nest ===")
    a = best_seq.copy()
    for it in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc_lp = min(25.0, remaining * 0.3)
        alloc_ns = min(15.0, remaining * 0.2)
        a_new = lp_alternate(a, time_budget=alloc_lp, use_topk=(it % 2 == 1))
        a_new = nesterov_subgrad(a_new, time_budget=alloc_ns, top_k=5, seed=200+it)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-14:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new
            print(f"  Inter R{it}: obj={best_score:.8f} **")
        else:
            print(f"  Inter R{it}: obj={obj:.8f}")
            a = best_seq * np.exp(rng.normal(0, 0.005, len(a)))
            a = np.maximum(a, 0)
            a /= float(np.sum(a))

    # === Phase 7: Final n fine-tuning (60s) ===
    print(f"\n=== Phase 7: n Fine-tuning ===")
    n_best = len(best_seq)
    n_fine = sorted(set([n_best - 20, n_best - 10, n_best + 10, n_best + 20]))
    n_fine = [n for n in n_fine if 50 <= n <= 5000]
    for n_try in n_fine:
        remaining = deadline - time.monotonic()
        if remaining < 8:
            break
        a_try = resample_seq(best_seq, n_try)
        alloc = min(12.0, remaining * 0.3)
        a_opt = nesterov_subgrad(a_try, time_budget=alloc, top_k=3, seed=n_try)
        a_opt = lp_alternate(a_opt, time_budget=min(10.0, remaining * 0.2))
        obj_try = compute_obj(a_opt)
        if obj_try < best_score - 1e-14:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final LP squeeze (remaining) ===
    print(f"\n=== Phase 8: Final LP Squeeze ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining - 10, 60.0)
        a_f = lp_alternate(best_seq, time_budget=alloc, use_topk=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score - 1e-14:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_f:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
