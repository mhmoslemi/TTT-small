def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=5.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    if res.status == 0:

        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

            try:

                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:

            except Exception:

    b_ub = np.full(2 * n - 1, M_target)

    if res.status == 0:
        g_val = np.maximum(res.x, 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None


def anderson_accelerate(history_x, history_Gx, m):
    """
    Anderson acceleration.
    history_x: list of last (m+1) iterates x_k
    history_Gx: list of corresponding G(x_k)
    Returns new point via Anderson acceleration.
    """
    m = min(m, len(history_x) - 1)
    if m == 0:
        return history_Gx[-1].copy()

    # Build the m x m system for Anderson acceleration
    xs = history_x[-(m + 1):]
    Gxs = history_Gx[-(m +1):]

    # r_i = x_i - G(x_i)
    residuals = [xs[i] - Gxs[i] for i in range(m + 1)]

    # Build matrix: rows are (x_i - G(x_i)), we want weights minimizing ||sum w_i r_i||^2
    # subject to sum w_i = 1
    # This is a least squares problem with linear constraint

    # Shape: (m+1) x n
    R = np.array(residuals)  # (m+1, n)

    # We want to find w (m+1,) with sum(w)=1 minimizing ||R^T w||^2
    # Unconstrained: w = R @ (R^T R)^{-1} @ 1 ... not quite
    # Actually: minimize ||R^T w||^2 s.t. 1^T w = 1
    # KKT: 2 R R^T w + lambda 1 = 0, 1^T w = 1

    RRt = R @ R.T  # (m+1) x (m+1)
    one = np.ones(m + 1)

    # Solve the KKT system
    # [2*RRt  1] [w]   [0]
    # [1^T    0] [lam] = [1]
    dim = m + 1
    KKT = np.zeros((dim + 1, dim + 1))
    KKT[:dim, :dim] = 2.0 * RRt
    KKT[:dim, dim] = 1.0
    KKT[dim, :dim] = 1.0
    rhs = np.zeros(dim + 1)
    rhs[dim] = 1.0

    try:
        sol = np.linalg.solve(KKT + 1e-12 * np.eye(dim + 1), rhs)
    except np.linalg.LinAlgError:
        return Gxs[-1].copy()

    w = sol[:dim]
    # Clip negative weights (optional, for stability)
    if np.min(w) < -0.5:
        return Gxs[-1].copy()

    x_new = np.zeros_like(xs[0])
    for i in range(dim):
        x_new += w[i] * Gxs[i]
    return x_new


def lp_alternating_anderson(a, time_budget=200.0, m_anderson=5, verbose=False):
    """
    LP alternating with Anderson acceleration for faster convergence.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345)

    # Anderson acceleration history
    history_x = []
    history_Gx = []

    # Best step size tracking
    best_t = 0.3
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        # Solve LP
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            conv_a = compute_conv(a)
            M_cur = float(np.max(conv_a))
            g_lp = solve_lp_maxsum(a, M_cur)
            if g_lp is None:
                break

        # Anderson acceleration: find optimal mixing
        history_x.append(a.copy())
        history_Gx.append(g_lp.copy())
        if len(history_x) > m_anderson + 1:
            history_x = history_x[-(m_anderson + 1):]
            history_Gx = history_Gx[-(m_anderson + 1):]

        if len(history_x) >= 3:
            # Anderson accelerated step
            x_aa = anderson_accelerate(history_x, history_Gx, m_anderson)
            x_aa = np.maximum(x_aa, 0)
            s_aa = float(np.sum(x_aa))
            if s_aa > 1e-10:
                x_aa = x_aa / s_aa
                obj_aa = compute_obj(x_aa)
                if obj_aa < best_obj:
                    best_obj = obj_aa
                    best_a = x_aa.copy()
                    a = x_aa.copy()
                    history_x = [a.copy()]
                    history_Gx = []
                    stall = 0
                    continue

        # Standard line search
        ts = [best_t, best_t * 0.5, best_t * 1.5, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        ts = [t for t in ts if 0.01 < t <= 1.0]
        ts = list(dict.fromkeys(ts))

        best_t_obj = np.inf
        best_t_val = best_t
        best_a_local = best_a.copy()

        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            best_t = best_t_val
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP-AA it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 20:
                # Escape: perturb and restart Anderson history
                noise_scale = min(0.2, 0.04 + 0.03 * (stall // 20))
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                history_x = []
                history_Gx = []
                stall = 0
                best_t = 0.3

    return best_a


def smooth_optimize_exact(a, time_budget=60.0, T=None, verbose=False):
    """
    Smooth optimization using exact gradient of log-sum-exp approximation.
    Uses L-BFGS-B with careful handling.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if T is None:
        T = 50.0 / np.log(2 * n)

    call_count = [0]
    time_up = [False]

    def obj_grad(x):
        call_count[0] += 1
        if time.monotonic() - start > time_budget * 0.9:
            time_up[0] = True
        x_safe = np.maximum(x, 1e-15)
        s = x_safe.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x_safe = x_safe / s
        conv = compute_conv(x_safe)
        max_c = conv.max()
        z = T * (conv - max_c)
        # Numerical stability
        e = np.exp(z - z.max())
        sum_e = e.sum()
        log_sum_exp = z.max() + np.log(sum_e)
        obj_val = 2.0 * n * (max_c + log_sum_exp / T)

        # Gradient
        w = e / sum_e
        g = np.zeros(n)
        # Only compute gradient for significant weights
        threshold = 1e-6
        active = w > threshold
        for k in np.where(active)[0]:
            g += w[k] * grad_conv_at_peak(x_safe, int(k))
        g *= 2.0 * n
        return obj_val, g

    # Run L-BFGS-B
    try:
        max_iter = 500
        res = sp_minimize(obj_grad, a, method='L-BFGS-B',
                          bounds=[(1e-12, None)] * n,
                          options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                   'gtol': 1e-8, 'disp': False})
        if not time_up[0]:
            a_opt = np.maximum(res.x, 0)
            s_opt = float(np.sum(a_opt))
            if s_opt > 1e-10:
                a_opt = a_opt / s_opt
                obj_opt = compute_obj(a_opt)
                if obj_opt < best_obj:
                    best_obj = obj_opt
                    best_a = a_opt.copy()
                    if verbose:
                        print(f"    Smooth (T={T:.2f}): obj={best_obj:.8f}")
    except Exception as e:
        if verbose:
            print(f"    Smooth error: {e}")

    return best_a


def pairwise_exchange(a, time_budget=30.0, verbose=False):
    """
    Local search via pairwise mass exchange: move small amounts from one position to another
    to reduce the maximum autocorrelation peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    rng = np.random.default_rng(999)

    improved = True
    while improved and time.monotonic() - start < time_budget:
        improved = False
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Identify positions contributing to the peak
        contrib_positions = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib_positions.append((a[i] * a[j], i, j))
        contrib_positions.sort(reverse=True)

        # Try moving mass from contributing positions to non-contributing ones
        for val, i, j in contrib_positions[:30]:
            if time.monotonic() - start > time_budget:
                break
            if val < 1e-14:
                continue
            # Try moving mass from i (or j) to a random non-contributing position
            src = i if a[i] >= a[j] else j
            if a[src] < 1e-10:
                continue

            for attempt in range(5):
                target = rng.integers(0, n)
                # Check if target doesn't contribute to the peak
                tj = k_star - target
                if 0 <= tj < n:
                    continue  # target contributes to peak, skip

                for frac in [0.02, 0.05, 0.1, 0.2, 0.3]:
                    delta = frac * a[src]
                    if delta < 1e-15:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-15:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break

    return best_a


def nesterov_subgradient(a, time_budget=60.0, K=5, verbose=False):
    """Nesterov-accelerated subgradient method on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 100

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        spread = (peak_vals[0] - peak_vals[-1]) / max_val
        temp = 3.0 / (1.0 + spread * 10.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / max_val)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                g += w * grad_conv_at_peak(y, int(k))

        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                t_k = 1.0
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.2)

        a_prev = a_new.copy()
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        y = y_new.copy()
        t_k = t_new
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.01))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # "U-shaped" - high at edges, low in middle
        x = np.linspace(0, 1, n)
        a = x**2 + (1 - x)**2 + 0.02
        a = a / a.sum()
    elif mode == 6:
        # Multi-modal
        n_modal = 3
        centers = rng.uniform(0.1, 0.9, n_modal)
        widths = rng.uniform(0.05, 0.15, n_modal)
        x = np.linspace(0, 1, n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses Anderson-accelerated LP alternating,
    exact-gradient smooth optimization, pairwise exchange, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    def check_time():
        return time.monotonic() < deadline - 5.0

    # === Phase 1: Quick smooth optimization to refine parent (120s) ===
    print(f"\n=== Phase 1: Smooth Optimization (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 120.0)
    if p1_budget > 15:
        for T_val in [30.0, 80.0]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            t_budget = p1_budget / 2.0
            if t_budget < 8:
                break
            a_sm = smooth_optimize_exact(best_seq, time_budget=t_budget, T=T_val, verbose=True)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Smooth T={T_val}: obj={best_score:.8f} **")
            else:
                print(f"  Smooth T={T_val}: obj={obj_sm:.8f} (no improvement)")

    # === Phase 2: Anderson-accelerated LP (main workhorse, 400s) ===
    print(f"\n=== Phase 2: LP with Anderson Acceleration ===")
    remaining = deadline - time.monotonic()
    p2_total = min(remaining * 0.3, 400.0)

    # Multiple rounds with different Anderson memory sizes
    for m_aa in [5, 8, 3]:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = p2_total / 3.0
        alloc = min(alloc, remaining * 0.5)
        if alloc < 20:
            break

        a_lp = lp_alternating_anderson(best_seq, time_budget=alloc, m_anderson=m_aa, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP-AA (m={m_aa}): obj={best_score:.8f} **")
        else:
            print(f"  LP-AA (m={m_aa}): obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: More smooth optimization with different T (200s) ===
    print(f"\n=== Phase 3: Extended Smooth Optimization ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)
    if p3_budget > 20:
        for T_val in [15.0, 50.0, 100.0, 200.0]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            t_budget = p3_budget / 4.0
            if t_budget < 8:
                break
            a_sm = smooth_optimize_exact(best_seq, time_budget=t_budget, T=T_val, verbose=True)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Smooth T={T_val}: obj={best_score:.8f} **")

    # === Phase 4: Nesterov subgradient (100s) ===
    print(f"\n=== Phase 4: Nesterov Subgradient ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 100.0)
    if p4_budget > 15:
        a_ns = nesterov_subgradient(best_seq, time_budget=p4_budget, K=7)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")

    # === Phase 5: Multi-n exploration (300s) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p5_total = min(remaining * 0.2, 300.0)
    n_current = len(best_seq)

    # Generate candidate n values
    n_candidates = set()
    n_candidates.add(n_current)
    for delta in [-200, -100, -50, 50, 100, 200, 300, 400, 500]:
        n_try = n_current + delta
        if 100 <= n_try <= 3000:
            n_candidates.add(n_try)
    # Also try some fixed values
    for n_fixed in [300, 400, 500, 600, 700, 900, 1000, 1200, 1500, 2000]:
        n_candidates.add(n_fixed)

    n_candidates = sorted(n_candidates, key=lambda x: abs(x - n_current))
    n_tried = 0
    max_n_tries = 8

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40 or n_tried >= max_n_tries:
            break
        if n_try == n_current:
            continue

        alloc = p5_total / max(2, max_n_tries - n_tried)
        alloc = min(alloc, remaining * 0.4)
        if alloc < 20:
            break

        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1
        print(f"  Trying n={n_try} (budget={alloc:.0f}s)...")

        # Quick optimization: smooth + LP
        half = alloc * 0.4
        if half > 10:
            a_try = smooth_optimize_exact(a_try, time_budget=half, T=50.0/np.log(2*n_try))

        remaining = deadline - time.monotonic()
        half2 = min(alloc * 0.5, remaining * 0.4)
        if half2 > 15:
            a_try = lp_alternating_anderson(a_try, time_budget=half2, m_anderson=4)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"     n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Pairwise exchange local search (150s) ===
    print(f"\n=== Phase 6: Pairwise Exchange ===")
    remaining = deadline - time.monotonic()
    p6_budget = min(remaining * 0.1, 150.0)
    if p6_budget > 10:
        a_pe = pairwise_exchange(best_seq, time_budget=p6_budget, verbose=True)
        obj_pe = compute_obj(a_pe)
        if obj_pe < best_score:
            best_score = obj_pe
            best_seq = a_pe.copy()
            print(f"  Pairwise: obj={best_score:.8f} **")
        else:
            print(f"  Pairwise: obj={obj_pe:.8f} (no improvement)")

    # === Phase 7: Diverse restarts (100s) ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p7_total = min(remaining * 0.1, 100.0)
    n_best = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        mode = r % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.15 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = p7_total / 3.0
        restart_budget = min(restart_budget, remaining * 0.4)
        if restart_budget < 10:
            break

        a_opt = lp_alternating_anderson(a_r, time_budget=restart_budget, m_anderson=4)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final LP
    if remaining > 30:
        lp_alloc = min(remaining * 0.25, 80.0)
        a_lp = lp_alternating_anderson(best_seq, time_budget=lp_alloc, m_anderson=6, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final smooth
    remaining = deadline - time.monotonic()
    if remaining > 20:
        sm_alloc = min(remaining * 0.2, 40.0)
        a_sm = smooth_optimize_exact(best_seq, time_budget=sm_alloc, T=100.0)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Final Smooth: obj={best_score:.8f} **")

    # Final pairwise
    remaining = deadline - time.monotonic()
    if remaining > 15:
        pe_alloc = min(remaining * 0.2, 30.0)
        a_pe = pairwise_exchange(best_seq, time_budget=pe_alloc)
        obj_pe = compute_obj(a_pe)
        if obj_pe < best_score:
            best_score = obj_pe
            best_seq = a_pe.copy()
            print(f"  Final PE: obj={best_score:.8f} **")

    # One more LP if time
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_alternating_anderson(best_seq, time_budget=min(remaining * 0.5, 30.0), m_anderson=5)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
