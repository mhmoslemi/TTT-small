```python
def subgradient_step(a, step_size, n_iter_max=100000):
    """
    Perform subgradient descent on f(a) = 2n * max(conv(a,a)) on the simplex.
    Uses projected subgradient method.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * np.max(compute_conv(a))
    
    for it in range(n_iter_max):
        b = compute_conv(a)
        max_b = np.max(b)
        obj = 2.0 * n * max_b
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Active set
        K = b == max_b  # boolean vector of length 2n-1
        n_active = np.sum(K)
        
        # Subgradient: ∇f[i] = (4n/n_active) * sum_j a[j] * K[j+i]
        # This is (4n/n_active) * correlate(a, K)
        # correlate(a, K)[i] = sum_j a[j] * K[j+i]
        
        # Using FFT: correlate(a, K) = irfft(rfft(a, size) * conj(rfft(K, size)))
        size = 1
        while size < 2 * n:
            size <<= 1
        fa = rfft(a, size)
        fK = rfft(K.astype(np.float64), size)
        corr = irfft(fa * np.conj(fK), size)[:n]
        
        grad = (4.0 * n / n_active) * corr
        # Project: subtract mean to stay on simplex tangent
        grad = grad - np.mean(grad)
        
        # Step
        a_new = a - step_size * grad
        # Project onto simplex: a_i >= 0, sum(a_i) = 1
        a_new = project_simplex(a_new)
        
        a = a_new
    
    return best_a, best_obj

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Algorithm from Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

Now, the challenge is choosing the step size. For subgradient methods, the standard choice is O(1/sqrt(t)) where t is the iteration number. But this converges slowly (O(1/sqrt(t))).

Better approaches:
1. **Fester's method**: Use a decreasing step size and average the iterates
2. **Mirror descent**: Use an entropic step (multiplicative updates) which is better for the simplex
3. **Accelerated methods**: Nesterov acceleration

Actually, for this problem, **mirror descent with entropic barrier** is ideal because:
- The domain is the simplex
- The entropic regularizer D(a) = sum a[i] * log(a[i])
- The update is: a_new[i] = a[i] * exp(-eta * grad[i]) / Z (where Z normalizes)
- This is a multiplicative (exponential) update, which naturally stays on the positive simplex

This is also known as the "multiplicative weights" or "exponential" update, and it's equivalent to the "Hedge" algorithm in online learning.

The convergence rate of mirror descent is O(1/t) (better than O(1/sqrt(t)) for subgradient), and with Nesterov acceleration, O(1/t²).

```python
def mirror_descent(a, time_budget=100.0, verbose=False):
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * np.max(compute_conv(a))
    
    start = time.monotonic()
    eta = 1.0  # initial learning rate
    
    for t in range(1, 1000000):
        if time.monotonic() - start > time_budget:
            break
        
        b = compute_conv(a)
        max_b = np.max(b)
        obj = 2.0 * n * max_b
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Active set
        tol = 0.05 * max_b  # include lags within 5% of max
        K = b >= (max_b - tol)
        n_active = np.sum(K)
        
        # Subgradient of max at a (using smoothed version)
        # ∇_smooth[i] = (4n/n_active) * sum_j a[j] * K[j+i]
        size = 1
        while size < 2 * n:
            size <<= 1
        fa = rfft(a, size)
        fK = rfft(K.astype(np.float64), size)
        corr = irfft(fa * np.conj(fK), size)[:n]
        grad = (4.0 * n / n_active) * corr
        
        # Mirror descent update (entropic)
        # a_new[i] = a[i] * exp(-eta * grad[i]) / Z
        # For numerical stability, subtract max of grad
        grad_centered = grad - np.max(grad)
        a_new = a * np.exp(-eta * grad_centered)
        s = np.sum(a_new)
        if s > 0:
            a = a_new / s
        else:
            break
        
        # Adaptive learning rate
        eta = 1.0 / np.sqrt(t)  # standard subgradient rate
    
    return best_a, best_obj
```

```python
            # b_zero[k] = b[k] - ai * d[k] - ai^2 * I(k == 2i)
            b_zero = b - ai * d
            k_quad = 2 * i
            if 0 <= k_quad <= 2 * n - 2:
                b_zero[k_quad] -= ai * ai
            
            # Now find x >= 0 that minimizes g(x) = max_k (b_zero[k] + x * d[k] + x^2 * I(k=k_quad))
            
            # The quadratic term at k_quad:
            has_quad = (0 <= k_quad <= 2 * n - 2)
            
            # For the 1D optimization, use bisection on the subgradient
            # g(x) is convex for x >= 0
            
            # Find bracket: g(0) and check direction
            g0 = np.max(b_zero)
            
            # Subgradient at x:
            # sg(x) = d[k*] + 2x * I(k* == k_quad) where k* = argmax_k (b_zero[k] + x*d[k] + x^2*I(k=k_quad))
            
            # Check if g is decreasing at x=0
            vals0 = b_zero.copy()
            k_max0 = np.argmax(vals0)
            sg0 = d[k_max0]
            if has_quad and k_max0 == k_quad:
                sg0 += 0  # 2*0 = 0
            # If sg0 > 0, the function is increasing at 0, so optimum is at x=0
            # If sg0 < 0, the function is decreasing at 0, so optimum is at x > 0
            
            if sg0 >= 0:
                x_opt = 0.0
            else:
                # Find upper bound
                hi = ai * 2 + 1.0 / n
                # Check if g is still decreasing at hi
                for _ in range(10):
                    vals_hi = b_zero + hi * d
                    if has_quad:
                        vals_hi[k_quad] += hi * hi
                    k_max_hi = np.argmax(vals_hi)
                    sg_hi = d[k_max_hi]
                    if has_quad and k_max_hi == k_quad:
                        sg_hi += 2 * hi
                    if sg_hi >= 0:
                        break
                    hi *= 2
                
                # Bisection on subgradient
                lo = 0.0
                for _ in range(60):
                    mid = (lo + hi) / 2
                    vals_mid = b_zero + mid * d
                    if has_quad:
                        vals_mid[k_quad] += mid * mid
                    k_max_mid = np.argmax(vals_mid)
                    sg_mid = d[k_max_mid]
                    if has_quad and k_max_mid == k_quad:
                        sg_mid += 2 * mid
                    if sg_mid > 0:
                        hi = mid
                    else:
                        lo = mid
                x_opt = (lo + hi) / 2
            
            if x_opt < 0:
                x_opt = 0.0
            
            if abs(x_opt - ai) > 1e-14:
                # Update b
                b = b_zero + x_opt * d
                if has_quad:
                    b[k_quad] += x_opt * x_opt
                a[i] = x_opt
                any_improved = True
        
        iterations += 1
        max_b = np.max(b)
        if max_b < best_max_b - 1e-12:
            best_max_b = max_b
            best_a = a.copy()
        
        if not any_improved and iterations > 10:
            break
        
        if verbose and iterations % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"      CD: iter={iterations}, max_b={max_b:.10f}, t={elapsed:.1f}s")
    
    return best_a, 2.0 * n * best_max_b
```

Wait, there's an issue with my bisection. The subgradient sg(x) is not necessarily monotone in the way I'm using it. At a kink point, the subgradient is a set, and I'm only evaluating one element of that set. This means the bisection might not converge correctly.

A better approach: use the fact that g(x) is convex and unimodal, and use **ternary search** or **golden section search**.

For ternary search on [lo, hi]:
- Evaluate g(m1) and g(m2) where m1 = lo + (hi-lo)/3, m2 = hi - (hi-lo)/3
- If g(m1) < g(m2): minimum is in [lo, m2], so hi = m2
- If g(m1) > g(m2): minimum is in [m1, hi], so lo = m1
- If g(m1) == g(m2): minimum is in [m1, m2]

This works for any unimodal function (convex functions are unimodal).

Each evaluation of g(x) is O(n). With 80 iterations of ternary search, that's O(80n) per coordinate. Full pass: O(80n²). For n=800: 80*640000 = 51.2M. At 100M ops/sec: ~500ms per pass. In 100 seconds: ~200 passes. That's still good.

But we can do better. Since g(x) is piecewise linear (plus one quadratic term), the minimum occurs at either x=0 or at an intersection of two "lines" (or the quadratic). We can find this more efficiently.

Actually, for the bisection approach, let me fix the subgradient issue. The key insight is:
- If the subgradient at x contains 0 (i.e., min_sg ≤ 0 ≤ max_sg), then x is optimal.
- Otherwise, if all subgradients > 0, x is too large.
- If all subgradients < 0, x is too small.

So for the bisection:
- Compute all active k's (those achieving the max)
- Compute the range of subgradients: [min(d[k] + 2x*I(k=k_quad)), max(d[k] + 2x*I(k=k_quad))] for k in active set
- If 0 is in this range: we're at the optimum
- If min > 0: x is too large, hi = x
- If max < 0: x is too small, lo = x

Let me implement this:

```python
def optimize_coordinate_1d(b_zero, d, k_quad, has_quad, ai):
    """Find x >= 0 minimizing g(x) = max_k(b_zero[k] + x*d[k] + x^2*I(k=k_quad))."""
    # Check if x=0 is optimal
    vals = b_zero.copy()
    max_val = np.max(vals)
    active = np.where(vals >= max_val - 1e-12)[0]
    # Subgradient range at x=0
    sgs = d[active]

        active = np.where(vals >= max_val - 1e-12)[0]
        sgs = d[active]
        if has_quad:
            # Add 2*hi for k_quad if it's active
            for idx in active:
                if idx == k_quad:
                    sgs = sgs  # handle below
        # Check max subgradient
        max_sg = np.max(d[active])
        if has_quad and k_quad in active:
            max_sg = max(max_sg, d[k_quad] + 2*hi)
        if max_sg >= 0:
            break
        hi *= 2

    lo = 0.0
    for _ in range(80):
        mid = (lo + hi) / 2
        vals = b_zero + mid * d
        if has_quad:
            vals[k_quad] += mid * mid
        max_val = np.max(vals)
        active = np.where(vals >= max_val - 1e-12)[0]
        # Subgradient range
        min_sg = np.min(d[active])
        max_sg = np.max(d[active])
        if has_quad:
            # For k_quad, subgradient includes d[k_quad] + 2*mid
            if k_quad in active:
                min_sg = min(min_sg, d[k_quad] + 2*mid)
                max_sg = max(max_sg, d[k_quad] + 2*mid)
        
        if min_sg <= 0 <= max_sg:
            # 0 is in the subgradient set: optimal!
            break
        elif min_sg > 0:
            hi = mid
        else:  # max_sg < 0
            lo = mid
    
    return (lo + hi) / 2
```

Hmm, this is getting complicated with the `k_quad in active` check (which is a Python loop). Let me simplify by just using ternary search, which is simpler and guaranteed to work:

```python
def optimize_coordinate_1d(b_zero, d, k_quad, has_quad):
    """Ternary search for x >= 0 minimizing g(x)."""
    def g(x):
        if x < 0:
            return np.inf
        vals = b_zero + x * d
        if has_quad:
            vals = vals.copy()
            vals[k_quad] += x * x
        return np.max(vals)
    
    # Find bracket
    lo = 0.0
    hi = 1.0 / len(b_zero) * 4  # initial guess
    g_lo = g(lo)
    g_hi = g(hi)
    
    if g_lo <= g_hi:
        # Optimum is at or very near 0
        # Check derivative direction
        # For small x, g(x) ≈ g(0) + x * sg(0)
        # If sg(0) >= 0, optimum is at 0
        return 0.0
    
    # Expand bracket
    while g_hi > g_lo and hi < 1.0:
        hi *= 2
        g_hi = g(hi)
    
    if g_lo <= g_hi:
        hi = 1.0 / len(b_zero)
        return 0.0
    
    # Ternary search
    for _ in range(100):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        g1 = g(m1)
        g2 = g(m2)
        if g1 < g2:
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-15:
            break
    
    x_opt = (lo + hi) / 2
    return max(0.0, x_opt)
```

The issue is that each g(x) evaluation involves creating a temporary array of size 2n-1, which is O(n). With 100 iterations, that's 100n per coordinate. Full pass: 100n². For n=800: 64M operations. In numpy, this might take a few hundred ms per pass.

Let me optimize the g(x) evaluation:
- Precompute b_zero and d
- g(x) = max(b_zero + x*d + x²*e) where e is a vector with e[k_quad]=1 and 0 elsewhere
- = max(b_zero + x*d) if not has_quad, or max(b_zero + x*d + x²*e) if has_quad

For the has_quad case, we can compute:
vals = b_zero + x*d  (this is a vector of length 2n-1)
vals[k_quad] += x*x
return max(vals)

This is just a few numpy operations on a vector of length 2n-1 ≈ 1600. Very fast.

Let me estimate: 100 iterations * 1600 elements * a few ops = ~50K operations per coordinate. Full pass: 800 * 50K = 40M. In numpy, maybe 20-50ms per pass. In 100 seconds: 2000-5000 passes. Good.

1. **Fast coordinate descent** as the main workhorse
2. **Smoothed gradient descent** (mirror descent with log-sum-exp) for global shaping
3. **Multi-n exploration** to find the best length
4. **Random restarts** with diverse initializations

Let me now write the complete code. I'll focus on making the coordinate descent as fast as possible since it's the core.

One more optimization: instead of random permutation each sweep, I can use a **Gauss-Seidel** style (fixed order) which sometimes converges faster. Or I can try both.

Also, I should add **momentum** to the coordinate descent: instead of setting a[i] = x_opt, use a[i] = (1-β)*a[i] + β*x_opt with β < 1. This can help escape local minima in the coordinate descent landscape.

Hybrid optimizer combining:
1. Fast cyclic coordinate descent (exact 1D subproblems)
2. Smoothed mirror descent (log-sum-exp + entropic updates)
3. Multi-n exploration with resampling
4. Diverse random restarts
5. Simulated annealing on the smoothing parameter

import numpy as np
from scipy.fft import rfft, irfft
import time

# === Core computation functions ===

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

# === Coordinate Descent (fast) ===

def cd_optimize_coordinate(b, a, i, n):
    """Optimize coordinate i given current b and a. Returns (new_ai, updated_b)."""
    ai = a[i]
    
    # d[k] = a[k-i] for k in [i, i+n-1] (clamped to [0, 2n-2])
    d = np.zeros(2 * n - 1)
    d[i:i+n] = a
    
    # b_zero[k] = b[k] - ai*d[k] - ai²*I(k=2i)
    b_zero = b - ai * d
    k_quad = 2 * i
    has_quad = (0 <= k_quad <= 2 * n - 2)
    if has_quad:
        b_zero[k_quad] -= ai * ai
    
    # Ternary search for x >= 0 minimizing max(b_zero + x*d + x²*e)
    # where e[k] = 1 if k == k_quad else 0
    
    def g(x):
        if x < 0:
            return np.inf
        vals = b_zero + x * d
        if has_quad:
            vals = vals.copy()
            vals[k_quad] += x * x
        return np.max(vals)
    
    # Quick check: is x=0 optimal?
    # Subgradient at 0: d[k*] where k* = argmax(b_zero)
    k_star = np.argmax(b_zero)
    if d[k_star] >= 0:
        x_opt = 0.0
    else:
        # Need x > 0
        # Find upper bracket
        hi = ai * 2.0 + 1.0 / n
        for _ in range(20):
            if g(hi) > g(0):
                break
            hi *= 2
        else:
            hi = 10.0 / n
        
        # Ternary search
        lo = 0.0
        for _ in range(80):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if g(m1) < g(m2):
                hi = m2
            else:
                lo = m1
            if hi - lo < 1e-14:
                break
        x_opt = (lo + hi) / 2.0
    
    x_opt = max(0.0, x_opt)
    
    # Update b
    b_new = b_zero + x_opt * d
    if has_quad:
        b_new[k_quad] += x_opt * x_opt
    
    return x_opt, b_new

def coordinate_descent(a, time_budget=100.0, verbose=False):
    """Cyclic coordinate descent with random restarts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    b = compute_conv(a)
    best_max_b = np.max(b)
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    
    sweep = 0
    while time.monotonic() - start < time_budget:
        # Random order
        order = rng.permutation(n)
        
        sweep_improved = False
        for i in order:
            x_opt, b_new = cd_optimize_coordinate(b, a, i, n)
            if abs(x_opt - a[i]) > 1e-14:
                a[i] = x_opt
                b = b_new
                sweep_improved = True
        
        sweep += 1
        max_b = np.max(b)
        if max_b < best_max_b - 1e-12:
            best_max_b = max_b
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 20:
            # Perturb and restart
            pert = rng.normal(0, 0.01 + 0.005 * no_improve / 10, n)
            a = best_a * np.exp(pert)
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            b = compute_conv(a)
            no_improve = 0
        
        if verbose and sweep % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"      CD: sweep={sweep}, max_b={max_b:.10f}, obj={2*n*max_b:.8f}, t={elapsed:.1f}s")
    
    return best_a

# === Smoothed Mirror Descent ===

def smoothed_mirror_descent(a, time_budget=50.0, verbose=False):
    """Mirror descent with log-sum-exp smoothing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    start = time.monotonic()
    
    # Annealing schedule for tau
    taus = [1.0/n * 10, 1.0/n * 5, 1.0/n * 2, 1.0/n, 1.0/n * 0.5, 1.0/n * 0.2, 1.0/n * 0.1]
    
    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        
        n_iter = int((time_budget / len(taus)) * 1000)
        eta = 5.0  # learning rate (will be scaled)
        
        for t in range(n_iter):
            if time.monotonic() - start > time_budget:
                break
            
            b = compute_conv(a)
            b_max = np.max(b)
            
            # Softmax weights
            weights = np.exp((b - b_max) / tau)
            weights = weights / np.sum(weights)
            
            # Gradient: 4n * correlate(a, weights)
            size = 1
            while size < 2 * n:
                size <<= 1
            fa = rfft(a, size)
            fw = rfft(weights, size)
            corr = irfft(fa * np.conj(fw), size)[:n]
            grad = 4.0 * n * corr
            grad = grad - np.mean(grad)  # project to simplex tangent
            
            # Entropic update
            eta_t = eta / np.sqrt(t + 1)
            log_a = np.log(a + 1e-30) - eta_t * grad
            a_new = np.exp(log_a - np.max(log_a))
            a_new = a_new / np.sum(a_new)
            a = a_new
            
            # Track best using exact objective
            obj = 2.0 * n * np.max(compute_conv(a))
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        
        if verbose:
            print(f"      MD: tau={tau:.6f}, obj={best_obj:.8f}")
    
    return best_a

# === Resampling ===

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
