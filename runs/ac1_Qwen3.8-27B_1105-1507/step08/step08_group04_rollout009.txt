"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using LP alternating,
peak-directed deflation, Fourier perturbation, and multi-scale exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A


def solve_lp(a):
    """Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_value) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    # Try scipy HiGHS (fastest for medium-size LPs)
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    # Fallback: try cvxpy
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.HIGHS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        return g_val / gs, float(M.value)
            except Exception:
                continue
    except Exception:
        pass

    return None, None

def gradient_scale(grad, a):
    """Compute a scale factor for the gradient step."""
    n = len(a)
    return np.linalg.norm(a) / np.sqrt(n)


def fourier_perturb(a, rng, num_modes=3):
    """Perturb a using random Fourier modes."""
    n = len(a)
    log_a = np.log(np.maximum(a, 1e-15))
    perturbation = np.zeros(n)
    x = np.arange(n)
    for _ in range(num_modes):
        freq = rng.uniform(1, n // 4)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.5)
        perturbation += amp * np.cos(2 * np.pi * freq * x / n + phase)
    log_a += perturbation
    a_new = np.exp(log_a - np.max(log_a))
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new

def subgradient_optimize(a, time_budget=60.0, verbose=False):
    """Subgradient method with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / n
    beta = 0.85
    momentum = np.zeros(n)
    top_k = max(5, n // 50)
    n_fft = 1
    while n_fft < 4 * n:
        n_fft <<= 1

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_indices = np.argsort(conv)[::-1][:top_k]

        # Compute subgradient using FFT
        w = np.zeros(2 * n - 1)
        for k in k_indices:
            w[int(k)] = 1.0 / len(k_indices)

        A_fft = rfft(a, n_fft)
        W_fft = rfft(w, n_fft)
        R = irfft(np.conj(A_fft) * W_fft, n_fft)
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = R[(n_fft - j) % n_fft]
        grad *= 4.0 * n
        grad -= np.mean(grad)

        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        if it % 100 == 99:
            lr = max(lr * 0.995, 0.01 / n)

        if verbose and it % 5000 == 0:
            print(f"    SubGrad: it={it}, obj={best_obj:.8f}")

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                best_a = a_pd2
                print(f"  Peak deflation (8 peaks): obj={best_obj:.8f} **")

    # === Phase 3: LP again from improved point ===
    print(f"\n=== Phase 3: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.25, 250.0)
        a_p3 = lp_alternate(best_a, time_budget=p3_budget, verbose=True)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3
            print(f"  Phase 3 result: obj={best_obj:.8f} **")

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = min(remaining * 0.10, 100.0)
        a_sg = subgradient_optimize(best_a, time_budget=p4_budget, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"  SubGrad: obj={best_obj:.8f} **")
        else:
            print(f"  SubGrad: obj={obj_sg:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_base_best = len(best_a)
    n_options = sorted(set([
        max(100, n_base_best - 300),
        max(100, n_base_best - 150),
        max(100, n_base_best - 50),
        n_base_best,
        n_base_best + 50,
        n_base_best + 150,
        n_base_best + 300,
        n_base_best + 500,
    ]))
    n_options = [x for x in n_options if 100 <= x <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == len(best_a):
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            a_try = a_try / np.sum(a_try)

        alloc = min(40.0, remaining * 0.25)
        a_opt = lp_alternate(a_try, time_budget=alloc)
        # Quick peak deflation
        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            a_opt = peak_deflation(a_opt, n_peaks=3, step_size=0.4, n_iters=50)
        obj_try = compute_obj(a_opt)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Coarse-to-fine from diverse inits ===
    print(f"\n=== Phase 6: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    for d in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_div = diverse_initialization(n_best, rng)
        alloc = min(35.0, remaining * 0.2)
        a_opt = lp_alternate(a_div, time_budget=alloc)
        obj_div = compute_obj(a_opt)

        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse init {d}: obj={best_obj:.8f} **")
        else:
            if d < 3:
                print(f"  Diverse init {d}: obj={obj_div:.8f}")

    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 20:
        alloc_lp = min(remaining * 0.35, 100.0)
        a_final = lp_alternate(best_a, time_budget=alloc_lp, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final LP: obj={best_obj:.8f} **")

    # Peak deflation
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_pd = peak_deflation(best_a, n_peaks=5, step_size=0.4, n_iters=80)
        obj_pd = compute_obj(a_pd)
        if obj_pd < best_obj:
            best_obj = obj_pd
            best_a = a_pd
            print(f"  Final peak deflation: obj={best_obj:.8f} **")

    # Subgradient micro-polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_sg = min(remaining * 0.3, 40.0)
        a_sg = subgradient_optimize(best_a, time_budget=alloc_sg)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"  Final SubGrad: obj={best_obj:.8f} **")

    # One more LP if time allows
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_last = min(remaining * 0.5, 30.0)
        a_last = lp_alternate(best_a, time_budget=alloc_last)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Last LP: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
