"""
Optimized step function search using mirror descent (multiplicative weight updates),

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_grad_at_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a[m] = 2*a[k-m] if 0 <= k-m < n."""
    n = len(a)
    grad = np.zeros(n)
    # conv(k) = sum_i a[i]*a[k-i], so d/d(a[m]) = a[k-m] + a[k-m] = 2*a[k-m]
    # (when m != k-m, i.e., m != k/2; when m == k/2, it's 2*a[m] anyway)
    i = k - np.arange(n)  # i[k-m] gives k-m
    valid = (i >= 0) & (i < n)
    if np.any(valid):
        grad[valid] = 2.0 * a[i[valid]]
    return grad


def conv_grad_multi(a, ks):
    """Sum of gradients at multiple peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k in ks:
        k = int(k)
        idx = k - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        if np.any(valid):
            grad[valid] += 2.0 * a[idx[valid]]
    return grad


def mirror_descent(a0, time_budget=60.0, top_k=5, verbose=False):
    """Mirror descent with multiplicative weight updates on the simplex."""
    n = len(a0)

        return a0.copy()

            break
        
        conv_a = compute_conv(a)
        # Get top k peaks
        top_idx = np.argsort(conv_a)[::-1][:top_k]
        grad = conv_grad_multi(a, top_idx)
        grad /= len(top_idx)
        
        # Center the gradient (important for mirror descent on simplex)
        grad_centered = grad - np.mean(grad)
        
        # Multiplicative update
        log_update = -lr * grad_centered
        # Clamp to avoid overflow
        log_update = np.clip(log_update, -5.0, 5.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)

                lr *= decay
                patience = 0
                if lr < 0.001:

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_weighted(a, weights):
    """LP: min sum(weights * conv(a,g)) s.t. sum(g)=1, g>=0, conv(a,g) >= 0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    # Objective: weights @ (A @ g) = (A.T @ weights) @ g
    obj_vec = A.T @ weights
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(cp.sum(cp.multiply(obj_vec, g))), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs
        except Exception:
            continue
    return None

def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def flat_top_window(n):
    """Flat-top window normalized to sum=1."""
    x = np.arange(n)
    # Flat-top: combination of cosines
    a1, a2, a3, a4 = 0.21557895, 0.41663158, 0.277263158, 0.083578947
    w = (a1 - a2 * np.cos(2*np.pi*x/n) + a3 * np.cos(4*np.pi*x/n) 
         - a4 * np.cos(6*np.pi*x/n) + a1 * np.cos(8*np.pi*x/n))
    w = np.maximum(w, 0)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def raised_cosine(n, alpha=0.5):
    """Raised cosine window."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    width = n * alpha
    w = np.cos(np.pi * (x - center) / width) ** 2
    w[x - center > width] = 0
    w = np.maximum(w, 0)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw

    """Coordinate descent targeting specific peaks."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)

        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        top_idxs = np.argsort(conv)[::-1][:n_peaks]
        improved = False
        
        for k in top_idxs:
            k = int(k)
            # For peak at position k: conv[k] = sum_i a[i]*a[k-i]
            # To reduce this, we can reduce a[i] for i that contribute to this peak
            # and redistribute to positions that don't contribute
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            
            # Find the contributor with the largest product
            products = np.array([a[i] * a[k-i] for i in range(i_start, i_end+1)])
            if len(products) == 0:
                continue
            max_i = i_start + int(np.argmax(products))
            
            # Try to reduce a[max_i] and add to a position that doesn't contribute to peak k
            # A position m that doesn't contribute to peak k means k-m is out of [0, n-1]
            # i.e., m < k-(n-1) or m > k
            candidate_dsts = []
            for m in range(n):
                if m == max_i:
                    continue
                if not (i_start <= m <= i_end) and not (i_start <= k-m <= i_end):
                    candidate_dsts.append(m)
            
            if not candidate_dsts:
                continue
            
            # Pick the dst that minimizes the impact on other peaks
            # (approximation: pick one with smallest current value)
            dst = min(candidate_dsts, key=lambda m: a[m])
            
            # Try different fractions
            for frac in [0.02, 0.05, 0.1, 0.15]:
                delta = frac * a[max_i]
                if a[max_i] - delta < 0:
                    continue
                a[max_i] -= delta
                a[dst] += delta
                a = np.maximum(a, 0)
                st = float(np.sum(a))
                a = a / st
                obj_try = compute_obj(a)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try

                    improved = True
                    break
                else:
                    # Revert
                    a[max_i] += delta
                    a[dst] -= delta
                    a = a / st
        
        if not improved and outer > 20:

"""
Optimized step function search using mirror descent (multiplicative weight updates),
LP alternating, softmax-weighted LP, and adaptive n-sweeping.
Key innovation: multiplicative updates naturally handle non-negativity and provide
better exploration than Euclidean subgradient. Combined with LP alternating and
peak coordinate descent for polishing.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Compute full linear convolution of a with b using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_grad_at_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a. grad[m] = 2*a[k-m] if 0<=k-m<n."""
    n = len(a)
    grad = np.zeros(n)
    idx = k - np.arange(n)
    valid = (idx >= 0) & (idx < n)
    if np.any(valid):
        grad[valid] = 2.0 * a[idx[valid]]
    return grad


def conv_grad_multi(a, ks):
    """Sum of gradients at multiple peak positions."""
    n = len(a)
    grad = np.zeros(n)
    for k in ks:
        k = int(k)
        idx = k - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        if np.any(valid):
            grad[valid] += 2.0 * a[idx[valid]]
    return grad


def mirror_descent(a0, time_budget=60.0, top_k=5, verbose=False):
    """Mirror descent with multiplicative weight updates on the simplex."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.15
    decay = 0.995
    patience = 0
    max_patience = 80

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break

        conv_a = compute_conv(a)
        top_idx = np.argsort(conv_a)[::-1][:top_k]
        grad = conv_grad_multi(a, top_idx)
        grad /= len(top_idx)
        grad_centered = grad - np.mean(grad)

        log_update = -lr * grad_centered
        log_update = np.clip(log_update, -4.0, 4.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)
        a = a_new

        if it % 5 == 0 or it < 10:
            obj = compute_obj(a)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > max_patience:
                lr *= decay
                patience = 0
                if lr < 0.002:
                    break

        if it % 2000 == 1999:
            lr = max(lr * 0.97, 0.003)

    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_weighted(a, weights):
    """LP: min (weights . conv(a,g)) s.t. sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    obj_vec = A.T @ weights
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(cp.sum(cp.multiply(obj_vec, g))), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs
        except Exception:
            continue
    return None


def lp_alternate(a0, time_budget=100.0, verbose=False):
    """LP alternating with 1D line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_good_t = 0.4

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)

        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 50, 'xatol': 0.003})
            best_t = res.x if res.success else last_good_t
        except Exception:
            best_t = last_good_t

        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            last_good_t = best_t
            stall_count = 0
            if verbose:
                print(f"      LP: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall_count += 1
            if stall_count > 25:
                break

    return best_a


def softmax_lp_alternate(a0, time_budget=60.0, temperature=0.5, verbose=False):
    """Alternating method using softmax-weighted peaks for smoother optimization."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    temp = temperature
    stall = 0

    while time.monotonic() - start < time_budget:
        conv_a = compute_conv(a)
        shift = conv_a - np.max(conv_a)
        scale = np.max(conv_a) + 1e-15
        weights = np.exp(shift / (temp * scale))
        weights = weights / np.sum(weights)

        g = solve_lp_weighted(a, weights)
        if g is None:
            break

        def line_obj(t):
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)

        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 40, 'xatol': 0.01})
            best_t = res.x if res.success else 0.3
        except Exception:
            best_t = 0.3

        a_try = (1.0 - best_t) * a + best_t * g
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            stall = 0
            temp *= 0.995
            if verbose:
                print(f"      SoftLP: obj={best_obj:.8f}, t={best_t:.3f}, temp={temp:.3f}")
        else:
            stall += 1
            temp *= 1.03
            if stall > 40:
                break

    return best_a


def peak_coordinate_descent(a0, time_budget=60.0, n_peaks=5, verbose=False):
    """Targeted coordinate descent on peak-contributing elements."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for outer in range(500):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        top_idxs = np.argsort(conv)[::-1][:n_peaks]
        improved = False

        for k in top_idxs:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue

            products = np.array([a[i] * a[k - i] for i in range(i_start, i_end + 1)])
            if len(products) == 0:
                continue
            max_pos = i_start + int(np.argmax(products))
            max_i = max_pos
            max_j = k - max_i

            if a[max_i] < 1e-10:
                continue

            candidate_dsts = []
            for m in range(n):
                if m == max_i or m == max_j:
                    continue
                in_peak_i = (i_start <= m <= i_end)
                in_peak_j = (i_start <= k - m <= i_end)
                if not in_peak_i and not in_peak_j:
                    candidate_dsts.append(m)

            if not candidate_dsts:
                continue
            dst = min(candidate_dsts, key=lambda m: a[m])

            for frac in [0.03, 0.08, 0.15]:
                delta = frac * a[max_i]
                if a[max_i] - delta < 0:
                    continue
                a[max_i] -= delta
                a[dst] += delta
                a = np.maximum(a, 0)
                st = float(np.sum(a))
                if st < 1e-10:
                    a[max_i] += delta
                    a[dst] -= delta
                    continue
                a = a / st
                obj_try = compute_obj(a)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[max_i] += delta
                    a[dst] -= delta
                    a = a / st

        if not improved and outer > 30:
            break
        a = best_a.copy()
        if verbose:
            print(f"      PCD iter={outer}: obj={best_obj:.8f}")

    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def flat_top_window(n):
    """Flat-top window normalized to sum=1."""
    x = np.arange(n)
    a1, a2, a3, a4 = 0.21557895, 0.41663158, 0.277263158, 0.083578947
    w = (a1 - a2 * np.cos(2 * np.pi * x / n) + a3 * np.cos(4 * np.pi * x / n)
         - a4 * np.cos(6 * np.pi * x / n) + a1 * np.cos(8 * np.pi * x / n))
    w = np.maximum(w, 0)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


defRaised_cosine_window(n, alpha=0.6):
    """Raised cosine window."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    width = n * alpha / 2.0
    dist = np.abs(x - center) / width
    w = np.ones(n)
    mask = dist > 1.0
    w[mask] = 0.5 * (1.0 + np.cos(np.pi * (dist[mask] - 1.0)))
    w = np.maximum(w, 0)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def gaussian_window(n, sigma_frac=0.3):
    """Gaussian window."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = n * sigma_frac
    w = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    w = np.maximum(w, 0)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: mirror descent + LP alternating + softmax LP + peak coordinate descent.
    Uses multiplicative updates for better simplex navigation, diverse initializations,
    and adaptive n-sweeping.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)

            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = hann_window(800)
        best_score = compute_obj(best_seq)
        print(f"Hann init (n=800): obj={best_score:.8f}")

    # === Phase 1: n-Sweep with mirror descent ===
    print(f"\n=== Phase 1: n-Sweep ===")
    n_base = len(best_seq)
    n_options = list(range(200, 2100, 100)) + [n_base]
    n_options = sorted(set(n for n in n_options if 100 <= n <= 2500))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        a_try = resample_seq(best_seq, n_try)
        alloc = min(6.0, remaining * 0.1)
        a_opt = mirror_descent(a_try, time_budget=alloc, top_k=5)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 2: Diverse initializations ===
    print(f"\n=== Phase 2: Diverse Initializations ===")
    n_cur = len(best_seq)
    inits = [best_seq.copy()]
    inits.append(hann_window(n_cur))
    inits.append(flat_top_window(n_cur))
    inits.append(Raised_cosine_window(n_cur, 0.5))

    inits.append(gaussian_window(n_cur, 0.35))
    inits.append(gaussian_window(n_cur, 0.45))

    # Smoothed random
    for _ in range(4):
        r = rng.random(n_cur)
        kernel_size = max(3, n_cur // 50)
        kernel = np.ones(kernel_size) / kernel_size
        r = np.convolve(r, kernel, mode='same')
        r = np.maximum(r, 0)
        inits.append(r / np.sum(r))

    # Super-Gaussian (sharper)
    x = np.arange(n_cur)
    c = (n_cur - 1) / 2.0
    sg = np.exp(-((x - c) / (n_cur * 0.2)) ** 4)
    inits.append(np.maximum(sg, 0) / np.sum(np.maximum(sg, 0)))

    for i, init in enumerate(inits):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        obj = compute_obj(init)
        if obj < best_score:
            best_score = obj
            best_seq = init.copy()
            print(f"  Init {i}: obj={best_score:.8f} **")
        else:
            print(f"  Init {i}: obj={obj:.8f}")

        if remaining > 20:
            alloc = min(10.0, remaining * 0.15)
            a_opt = mirror_descent(init, time_budget=alloc * 0.5, top_k=5)
