"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. Fast LP with adaptive constraint subset (cutting-plane style, only top-K active constraints)
2. Golden-section line search exploiting convexity of the 1D blend objective
3. Alternating min-max LP and min-sum-of-peaks LP for diversity in search directions
4. Multiplicative (mirror) descent as complementary optimizer
5. Systematic multi-n exploration with resampling
6. Periodic "kick" perturbations to escape local minima
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_rows(a, constraint_indices, n):
    """
    Build rows of the convolution matrix for specific constraint indices.
    Returns A such that A @ g = conv(a, g)[constraint_indices].
    constraint_indices: array of lag positions k where we need convolution values.
    """
    rows = len(constraint_indices)
    A = np.zeros((rows, n))
    for row_idx, k in enumerate(constraint_indices):
        k = int(k)
        # conv(a,g)[k] = sum_i a[i] * g[k-i] for valid i, k-i
        # g[j] appears in conv[k] when j = k - i, i.e., i = k - j
        # So conv[k] = sum_j a[k-j] * g[j] for valid indices
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[row_idx, j] = a[i]
    return A


def build_conv_matrix_sparse(a, constraint_indices):
    """
    Efficiently build convolution matrix for subset of constraints.
    Uses the structure: row k has a shifted copy of a.
    """
    n = len(a)
    rows = len(constraint_indices)
    A = np.zeros((rows, n))
    for row_idx, k in enumerate(constraint_indices):
        k = int(k)
        # conv(a,g)[k] = sum_j a[k-j] * g[j] where 0<=k-j<n and 0<=j<n
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_min <= j_max:
            i_vals = k - np.arange(j_min, j_max + 1)
            A[row_idx, j_min:j_max + 1] = a[i_vals]
    return A


def solve_lp_minmax(a, constraint_indices, max_iters=20):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for k in constraint_indices, sum(g)=1, g>=0.
    Uses cutting-plane: starts with given constraints, adds violated ones.
    """
    n = len(a)
    a_norm = a / np.sum(a)

    current_constraints = list(constraint_indices)
    g_val = None
    M_val = None

    for iter_idx in range(max_iters):
        A = build_conv_matrix_sparse(a_norm, current_constraints)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        g_val = g_val / s
                        M_val = float(M_var.value)
                        solved = True
                        break
            except Exception:
                continue

        if not solved or g_val is None:
            break

        # Check for violated constraints (cutting plane)
        conv_ag = compute_conv_cross(a_norm, g_val)
        max_pos = np.argmax(conv_ag)
        if conv_ag[max_pos] > M_val * (1 + 1e-6) + 1e-10:
            if max_pos not in current_constraints and len(current_constraints) < 500:
                current_constraints.append(int(max_pos))
                continue
        break

    return g_val, M_val


def solve_lp_minsum_peaks(a, constraint_indices):
    """
    Solve LP: min sum of conv(a,g)[k] for k in constraint_indices,
    subject to max conv(a,g)[k] <= current_max * 1.1, sum(g)=1, g>=0.
    This spreads out the peaks.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)
    current_max = float(np.max(conv_aa))

    A = build_conv_matrix_sparse(a_norm, constraint_indices)
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [
        A @ g_var <= M_var,
        M_var <= current_max * 1.05,
        cp.sum(g_var) == 1
    ]
    # Minimize sum of the peak values
    obj = cp.sum(A @ g_var)
    prob = cp.Problem(cp.Minimize(obj), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val
        except Exception:
            continue
    return None


def golden_section_line_search(a, g, iterations=30):
    """
    Find t in [0, 1] that minimizes max(conv((1-t)a + t*g, (1-t)a + t*g)).
    The function is convex in t (max of quadratics), so golden section works.
    """
    inv_phi = (np.sqrt(5) - 1) / 2
    a_lo, a_hi = 0.0, 1.0

    c = a_hi - inv_phi * (a_hi - a_lo)
    d = a_lo + inv_phi * (a_hi - a_lo)

    def obj_at_t(t):
        blend = (1 - t) * a + t * g
        conv = compute_conv(blend)
        return float(np.max(conv))

    fc = obj_at_t(c)
    fd = obj_at_t(d)

    for _ in range(iterations):
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - inv_phi * (a_hi - a_lo)
            fc = obj_at_t(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + inv_phi * (a_hi - a_lo)
            fd = obj_at_t(d)

    t_best = (a_lo + a_hi) / 2
    # Also check endpoints
    candidates = [0.0, 0.1, 0.3, 0.5, t_best, 0.7, 1.0]
    best_t = 0.0
    best_val = np.inf
    for t in candidates:
        v = obj_at_t(t)
        if v < best_val:
            best_val = v
            best_t = t
    return best_t

    Alternates between min-max and min-sum LPs for diversity.

    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iter_count = 0
    stall_count = 0
    rng = np.random.default_rng(999)

    # Initial constraint set: top 150 peaks
    conv = compute_conv(a)
    top_indices = np.argsort(conv)[::-1][:150]

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if verbose and iter_count % 10 == 0:
            print(f"      LP iter {iter_count}: obj={best_obj:.8f}, t={elapsed:.0f}s")

        # Update constraint set periodically
        if iter_count % 20 == 0:
            conv = compute_conv(a)
            top_indices = np.argsort(conv)[::-1][:min(200, len(conv))]

        # Alternate between LP types
        if iter_count % 4 < 3:
            # Min-max LP (3 out of 4 iterations)
            g_lp, M_lp = solve_lp_minmax(a, top_indices[:100], max_iters=5)
        else:
            # Min-sum LP (1 out of 4)
            g_lp = solve_lp_minsum_peaks(a, top_indices[:100])
            M_lp = None

        if g_lp is None:
            # Fallback: use uniform
            g_lp = np.ones(n) / n

        # Line search for optimal blend
        t_best = golden_section_line_search(a, g_lp, iterations=20)

        # Apply update
        a_new = (1 - t_best) * a + t_best * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            iter_count += 1
        else:
            stall_count += 1
            if stall_count > 15:
                # Kick: random perturbation
                scale = 0.01 + 0.005 * min(stall_count // 15, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0
                # Update constraints
                conv = compute_conv(a)
                top_indices = np.argsort(conv)[::-1][:150]
            iter_count += 1

    return best_a


def mirror_descent(a, time_budget=200.0, verbose=False):
    """
    Multiplicative (mirror) descent with FFT-accelerated gradient.
    Uses log-sum-exp smoothing with adaptive temperature.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    lr = 1.0 / n
    momentum = 0.5
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = 20.0 + 300.0 * progress ** 0.5

        # Compute smoothed gradient
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        # Cross-correlation gradient via FFT
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        # Momentum
        v = momentum * v + g

        # Multiplicative update
        a_new = a * np.exp(-lr * v)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.95
            if lr < 1e-10:
                break

        # Adaptive lr
        lr = lr * (1 + 0.01 * (1 - progress))
        it += 1

        if verbose and it % 500 == 0:
            print(f"      MD iter {it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Cauchy-like distribution
        x = np.linspace(-1, 1, n)
        a = 1.0 / (1 + x**2)
        a = a / a.sum()
    elif mode == 7:
        # Bimodal
        c1 = rng.integers(0, n // 3)
        c2 = rng.integers(2 * n // 3, n)
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def peak_shaving(a, time_budget=100.0, verbose=False):
    """
    Peak shaving: iteratively identify the dominant peak and
    solve a constrained LP that specifically reduces it while
    not increasing others.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(7)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_peak = float(np.max(conv))
        peak_idx = int(np.argmax(conv))

        # Build constraint set: top peaks excluding the max (to force reduction)
        sorted_idx = np.argsort(conv)[::-1]
        constraint_set = sorted_idx[:100]

        # LP: minimize conv(a,g)[peak_idx] subject to
        # conv(a,g)[k] <= conv(a,a)[k] * 1.01 for other top peaks, sum(g)=1, g>=0
        a_norm = a / np.sum(a)
        g_var = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(g_var) == 1]

        # Add constraints for non-target peaks
        for k in constraint_set:
            if int(k) == peak_idx:
                continue
            row = build_conv_matrix_sparse(a_norm, [int(k)])
            if row.shape[0] > 0:
                constraints.append(row[0] @ g_var <= conv[int(k)] * 1.02 + 1e-10)

        # Objective: minimize the target peak
        target_row = build_conv_matrix_sparse(a_norm, [peak_idx])
        if target_row.shape[0] > 0:
            prob = cp.Problem(cp.Minimize(target_row[0] @ g_var), constraints)
            solved = False
            for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
                try:
                    prob.solve(solver=solver, verbose=False)
                    if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                        g_val = np.maximum(g_var.value, 0)
                        sv = float(np.sum(g_val))
                        if sv > 1e-10:
                            g_val = g_val / sv
                            solved = True
                            break
                except Exception:
                    continue

            if not solved:
                break

            # Line search
            t_best = golden_section_line_search(a, g_val, iterations=15)
            a_new = (1 - t_best) * a + t_best * g_val
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 10:
                    a = best_a * np.exp(rng.normal(0, 0.02, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    stall = 0

        if verbose:
            elapsed = time.monotonic() - start
            print(f"      PeakShave: obj={best_obj:.8f}, peak={max_peak:.6f}, t={elapsed:.0f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Fast LP + mirror descent + peak shaving + multi-n + diversification.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Fast LP iterations (main workhorse) ===
    print(f"\n=== Phase 1: Fast LP (budget-dependent) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a_lp = fast_lp_iteration(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Mirror descent ===
    print(f"\n=== Phase 2: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 150.0)
    if p2_budget > 20:
        a_md = mirror_descent(best_seq, time_budget=p2_budget, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Phase2 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 MD: obj={obj_md:.8f} (no improvement)")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 120.0)
    if p3_budget > 20:
        a_ps = peak_shaving(best_seq, time_budget=p3_budget, verbose=True)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Phase3 PS: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 PS: obj={obj_ps:.8f} (no improvement)")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = []
    for dn in [-300, -200, -100, 100, 200, 300, 400]:
        nt = n_current + dn
        if 400 <= nt <= 1600:
            n_options.append(nt)
    # Add some fixed values
    for nv in [500, 600, 700, 800, 900, 1000, 1100, 1200, 1400]:
        if nv not in n_options and 400 <= nv <= 1600 and abs(nv - n_current) > 80:
            n_options.append(nv)
    n_options = sorted(set(n_options))

    # Sort by distance from current (try closest first for refinement)
    n_options_sorted = sorted(n_options, key=lambda x: abs(x - n_current))

    for n_try in n_options_sorted[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # Quick LP optimization
        alloc_lp = min(remaining * 0.2, 60.0)
        if alloc_lp > 15:
            a_try = fast_lp_iteration(a_try, time_budget=alloc_lp)

        # Quick mirror descent
        remaining = deadline - time.monotonic()
        alloc_md = min(remaining * 0.15, 40.0)
        if alloc_md > 10:
            a_try = mirror_descent(a_try, time_budget=alloc_md)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diversification with kick + re-optimize ===
    print(f"\n=== Phase 5: Diversification ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_kicks = 4

    for r in range(n_kicks):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        # Different perturbation strategies
        if r == 0:
            # Gaussian noise
            a_kick = best_seq * np.exp(rng.normal(0, 0.03, n_best))
        elif r == 1:
            # Sparse randomDir perturbation
            a_kick = best_seq.copy()
            n_perturb = max(10, n_best // 10)
            idx = rng.choice(n_best, n_perturb, replace=False)
            a_kick[idx] *= np.exp(rng.normal(0, 0.1, n_perturb))
        elif r == 2:
            # Blend with diverse init
            mode = r + 1
            a_div = diverse_init(n_best, rng, mode=mode)
            a_kick = 0.8 * best_seq + 0.2 * a_div
        else:
            # Stronger perturbation
            a_kick = best_seq * np.exp(rng.normal(0, 0.06, n_best))

        a_kick = np.maximum(a_kick, 0)
        a_kick = a_kick / a_kick.sum()

        # Re-optimize
        alloc = min(remaining * 0.3, 100.0)
        if alloc > 30:
            half = alloc / 2
            a_kick = fast_lp_iteration(a_kick, time_budget=half)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_kick = mirror_descent(a_kick, time_budget=min(remaining * 0.3, 40.0))

        obj_kick = compute_obj(a_kick)
        if obj_kick < best_score:
            best_score = obj_kick
            best_seq = a_kick.copy()
            print(f"  Kick {r}: obj={best_score:.8f} **")
        else:
            print(f"  Kick {r}: obj={obj_kick:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Extended LP
    if remaining > 40:
        alloc = min(remaining * 0.3, 150.0)
        a_final = fast_lp_iteration(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Peak shaving
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 80.0)
        a_final = peak_shaving(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final PS: obj={best_score:.8f} **")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 100.0)
        a_final = fast_lp_iteration(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Mirror descent final
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 50.0)
        a_final = mirror_descent(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
