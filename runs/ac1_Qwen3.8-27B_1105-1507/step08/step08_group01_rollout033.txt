import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A

def solve_lp(a, active_indices=None, solver_method='highs'):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A_full = build_conv_matrix(a)
    
    if active_indices is not None and len(active_indices) < 2 * n - 1:
        A = A_full[active_indices, :]
    else:
        A = A_full
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((A.shape[0], n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(A.shape[0])
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

def convex_line_search(a, g, n_evals=12):
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)
    
    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)
    
    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))
    
    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    a_new = a_new / s
    return t_opt, a_new

def compute_conv(a):

    size = 1
    while size < 2 * n - 1:

Now let me think about the overall algorithm more carefully.

The main loop:
1. Start with a sequence a (normalized to sum 1).
2. Compute conv(a, a) and find the peak location(s).
3. Solve LP: min M s.t. conv(a, g)_k <= M for all k (or active k), sum(g) = 1, g >= 0.
4. Blend: a_new = (1-t)*a + t*g for optimal t.
5. Repeat.

The key question is: how to make this converge faster?

Idea: **Simultaneous LP**. Instead of fixing a and solving for g, what if we solve a joint problem?

The joint problem is: min max_k (a*g)_k s.t. sum(a) = sum(g) = 1, a, g >= 0.
This is still non-convex (bilinear), but we could use a biconvex approach.

Alternative: **Penalty method**. Add a penalty for deviating from g = a, and solve:
min max_k (a*g)_k + ρ * ||a - g||^2 s.t. sum(a) = sum(g) = 1, a, g >= 0

As ρ → ∞, g → a, and we get the original problem. For finite ρ, this is a non-convex problem but might be easier to solve.

Actually, I think the most practical improvement is to:
1. Speed up the LP solves (use only active constraints, which are typically much fewer than 2n-1).
2. Use a better step size (the convex line search is good).
3. Try more diverse starting points.
4. Be more systematic about n.

Let me also consider: **what if we use a different norm than max?** For example, if we minimize the sum of the top-k peaks instead of just the max, we might get a flatter autocorrelation.

"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-alternating with adaptive active-set management, convex line search,
coordinate descent with 1D line search, multi-n exploration, and structured
perturbation to escape local optima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def build_conv_rows(a, indices):
    """Build only specified rows of the convolution matrix."""
    n = len(a)
    m = len(indices)
    A = np.zeros((m, n))
    for idx in range(m):
        k = indices[idx]
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        A[idx, lo:hi] = a[k - hi + 1:k - lo + 1][::-1]
    return A


def solve_lp(a, active_indices=None):
    """Solve LP: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0.
    If active_indices given, only use those constraints."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)

    if active_indices is not None and len(active_indices) < 2 * n - 1:
        A = build_conv_rows(a, active_indices)
    else:
        A = build_conv_matrix(a)

    m = A.shape[0]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def convex_line_search(a, g, n_evals=12):
    """Find optimal t for a_new = (1-t)*a + t*g exploiting convexity in t."""
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    a_new = a_new / s
    return t_opt, a_new


def lp_alternating(a, time_budget=200.0, verbose=False):
    """LP alternating with adaptive active set and convex line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    active_threshold = 0.99  # starts wide, narrows over time

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        conv_cur = compute_conv(a)
        max_conv = float(np.max(conv_cur))

        # Adaptive active set: thresholds get tighter as we converge
        active_threshold_current = 0.995 + 0.005 * min(1.0, it / 50.0)
        threshold = max_conv * active_threshold_current
        active_set = np.where(conv_cur >= threshold)[0]

        # Solve LP with active set
        g_lp, M_lp = solve_lp(a, active_indices=active_set)

        # Verify against full constraints
        if g_lp is not None and len(active_set) < 2 * n - 1:
            A_full = build_conv_matrix(a)
            conv_check = A_full @ g_lp
            true_M = float(np.max(conv_check))
            if true_M > M_lp * 1.005:
                # Active set too restrictive, re-solve with more constraints
                threshold2 = max_conv * 0.98
                active_set2 = np.where(conv_cur >= threshold2)[0]
                g_lp2, M_lp2 = solve_lp(a, active_indices=active_set2)
                if g_lp2 is not None:
                    A_full2 = build_conv_matrix(a)
                    conv_check2 = A_full2 @ g_lp2
                    true_M2 = float(np.max(conv_check2))
                    if true_M2 <= M_lp2 * 1.005:
                        g_lp, M_lp = g_lp2, true_M2
                    else:
                        g_lp, M_lp = solve_lp(a, active_indices=None)
                else:
                    g_lp, M_lp = solve_lp(a, active_indices=None)

        if g_lp is None:
            stall_count += 1
            if stall_count > 10:
                break
            continue

        # Convex line search
        t_opt, a_new = convex_line_search(a, g_lp, n_evals=10)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, M={M_lp:.6f}, active={len(active_set)}, time={elapsed:.1f}s")
        else:
            stall_count += 1
            if stall_count > 12:
                # Structured perturbation
                rng = np.random.default_rng((it * 31337 + n * 7) % (2**31))
                # Smooth perturbation: add low-frequency noise
                freq = rng.integers(1, max(2, n // 10))
                phase = rng.uniform(0, 2 * np.pi)
                amp = 0.002 * (1 + rng.random())
                jitter = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
                a_pert = best_a + jitter
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a = a_pert / s_p
                stall_count = 0
                it_offset = 0

    return best_a


def coordinate_descent(a, time_budget=100.0, delta_init=1e-4, verbose=False):
    """Coordinate descent with adaptive step size and 1D line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    delta = delta_init

    pass_count = 0
    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count * 104729 + n)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            best_local = best_obj
            best_local_a = a.copy()

            # Try several step sizes in both directions
            for sign in [+1, -1]:
                for mult in [1.0, 0.5, 2.0, 0.25, 4.0]:
                    d = delta * mult
                    if sign < 0 and a[i] < d * 0.5:
                        continue
                    a_try = a.copy()
                    a_try[i] += sign * d
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_local - 1e-15:
                        best_local = obj_try
                        best_local_a = a_try.copy()

            if best_local < best_obj - 1e-15:
                best_obj = best_local
                best_a = best_local_a
                a = best_local_a
                improved = True

        if verbose and pass_count % 10 == 0:
            print(f"      CD pass {pass_count}: obj={best_obj:.8f}, time={time.monotonic()-start:.1f}s")

        if not improved:
            delta *= 0.5
            if delta < 1e-8:
                break

    return best_a


def smooth_max_descent(a, time_budget=60.0, T_init=0.0005, T_min=0.00003):
    """Gradient descent using smooth maximum approximation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.003
    momentum = 0.8
    v = np.zeros(n)
    stall = 0

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total

        # Gradient: 2 * correlate(weights, a, 'valid')
        grad = 2.0 * np.correlate(weights, a, mode='valid')
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            stall += 1
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                lr *= 0.7
                if lr < 1e-7:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0

        T = max(T_min, T_init * (0.997 ** it))

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def parametric_init(n, family):
    """Generate a parametric initial sequence of length n."""
    if family == 'uniform':
        return np.ones(n)
    elif family == 'cosine':
        t = np.arange(n) / max(1, n - 1)
        return np.cos(np.pi * (t - 0.5)) ** 2
    elif family == 'parabolic':
        t = np.arange(n) / max(1, n - 1)
        return np.maximum(1.0 - 0.4 * (2 * t - 1) ** 2, 0.01)
    elif family == 'sine':
        return np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    elif family == 'gaussian':
        t = np.arange(n) / max(1, n - 1)
        return np.exp(-2.0 * (t - 0.5) ** 2)
    elif family == 'flat_edge':
        t = np.arange(n) / max(1, n - 1)
        # Flat in middle, tapered at edges
        edge = min(n // 5, 50)
        result = np.ones(n)
        for i in range(edge):
            frac = i / (edge + 1)
            result[i] = 0.5 + 0.5 * frac
            result[n - 1 - i] = 0.5 + 0.5 * frac
        return result
    else:
        return np.ones(n)

        print(f"  Budget: {budget:.0f}s")
        a_opt = lp_alternating(best_seq, time_budget=budget, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Continued: obj={best_score:.8f}")

    # === Phase 4: Smooth max gradient descent ===
    print(f"\n=== Phase 4: Smooth Max Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        budget = min(70.0, remaining * 0.15)
        a_sm = smooth_max_descent(best_seq, time_budget=budget)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Smooth max: obj={best_score:.8f}")
        else:
            print(f"  Smooth max: obj={obj_sm:.8f} (no improvement)")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        budget = min(120.0, remaining * 0.25)
        a_cd = coordinate_descent(best_seq, time_budget=budget, delta_init=1e-4, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f}")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 6: Fine multi-n search ===
    print(f"\n=== Phase 6: Fine Multi-n Search ===")
    n_best = len(best_seq)
    n_fine = sorted(set([n_best + d for d in range(-30, 31, 5) if 400 <= n_best + d <= 1500]))

    for n_try in n_fine:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        if n_try == n_best:
            continue
        a_try = resample_sequence(best_seq, n_try)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        budget = min(25.0, remaining * 0.06)
        a_opt = lp_alternating(a_try, time_budget=budget)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Quick LP
        if remaining > 40:
            budget = min(50.0, remaining * 0.3)
            a_LP = lp_alternating(best_seq, time_budget=budget)
            obj_L = compute_obj(a_LP)
            if obj_L < best_score:
                best_score = obj_L
                best_seq = a_LP.copy()
                print(f"  Final LP: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        # CD with smaller delta
        if remaining > 25:
            budget = min(50.0, remaining * 0.35)
            a_cd = coordinate_descent(best_seq, time_budget=budget, delta_init=3e-5)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final CD: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        # Smooth max with finer T
        if remaining > 20:
            budget = min(30.0, remaining * 0.4)
            a_sm = smooth_max_descent(best_seq, time_budget=budget, T_init=0.0002, T_min=0.00002)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Final SmoothMax: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
