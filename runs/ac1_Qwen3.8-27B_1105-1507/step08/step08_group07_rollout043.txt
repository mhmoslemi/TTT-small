def lp_polish(a, budget, max_iter=30):
   current=normalize_prob(a); best=current.copy(); best_obj=internal_objective(best)
   start=time.monotonic(); iters=0
   n=len(current)
   top_k = min(64, max(16, n//32))
   while time.monotonic()-start < budget and iters < max_iter:
      z = solve_active_lp(current, top_k, max_outer=4)
      iters += 1
      if z is None: break
      improved=False
      for mix in (1.0, 0.8, 0.5, 0.3):
         cand=(1-mix)*current + mix*z
         cand=normalize_prob(cand)
         obj=internal_objective(cand)
         if obj < best_obj - 1e-12:
            current=cand; best=cand.copy(); best_obj=obj; improved=True

      if not improved:

         rng=np.random.default_rng(int(time.monotonic()*1e6)+iters)

         if obj < best_obj - 1e-12:
            current=cand; best=cand.copy(); best_obj=obj

      if verbose? can't pass? no.

def active_gradient(a, c, beta=1e6, top_count=128):
   n=a.size; L=c.size
   if L==0: return None
   m=float(np.max(c))
   if not np.isfinite(m) or m <= 0.0:
       return np.zeros(n, dtype=float)
   z=beta*(c-m)
   z=np.where(np.isfinite(z), z, 0.0)
   # Avoid underflow not issue, but for extremely negative, exp 0.
   expz=np.exp(z)
   sw=float(np.sum(expz))
   if sw <= 1e-300:
       w=np.zeros(L); w[int(np.argmax(c))]=1.0
   else:
       w=expz/sw
   if L > top_count:
       top = np.argpartition(w, -top_count)[-top_count:]
       w_top = w[top]
       # maybe recompute on top? We'll use full w and indices top
   else:
       top = np.arange(L)
   # select significant weights
   threshold = max(1e-14, float(np.max(w)) * 1e-8)
   keep = top[w[top] > threshold]
   grad = np.zeros(n, dtype=float)
   for t in keep:
       t=int(t); wt=float(w[t])
       i_lo = max(0, t - (n - 1)); i_hi = min(n - 1, t)
       if i_lo > i_hi: continue
       j_lo = t - i_hi; j_hi = t - i_lo
       if j_hi == j_lo:
           grad[i_lo] += 2.0 * wt * a[j_lo]
       else:
           if j_lo == 0:
               seg = a[j_hi::-1]
           else:
               seg = a[j_hi:j_lo - 1:-1]
           grad[i_lo:i_hi + 1] += 2.0 * wt * seg
   return grad

```python
"""
Black-box + active-set optimizer for non-negative step-function heights.

The solver minimizes the provided evaluate_sequence score using a mixture of:
  - direct black-box local search in the original sequence space,
  - smooth feature-space perturbations,
  - active-lag smooth-max Adam in log-simplex coordinates,
  - cutting-plane LP linearization of active convolution peaks,
  - multi-length resampling and random restarts.

The function propose_candidate returns a plain Python list of non-negative
floats.
"""

import numpy as np
import time
import math

from scipy.fft import rfft, irfft

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def safe_sequence(a):
    """Convert a sequence to a list of finite non-negative floats."""
    arr = np.asarray(a, dtype=np.float64)
    if arr.size == 0:
        return []
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    arr = np.maximum(arr, 1e-15)
    return [float(x) for x in arr]


def true_score(seq):
    """Call the external evaluator safely."""
    try:
        val = evaluate_sequence(seq)
        if val is None:
            return np.inf
        val = float(val)
        if not np.isfinite(val):
            return np.inf
        return val
    except Exception:
        return np.inf


def consider_candidate(a, best_seq, best_score, label=""):
    """Normalize, score, and possibly update the current best sequence."""
    if a is None:
        return best_seq, best_score

    arr = np.asarray(a, dtype=np.float64)
    if arr.size == 0 or not np.isfinite(arr).all():
        return best_seq, best_score

    arr = np.clip(arr, 0.0, 1000.0)
    s = float(np.sum(arr))
    if s < 1e-12:
        return best_seq, best_score

    arr = arr / s
    arr = np.maximum(arr, 1e-15)
    arr = arr / float(np.sum(arr))

    seq = safe_sequence(arr)
    val = true_score(seq)

    if np.isfinite(val):
        print(f"[{label}] n={arr.size} score={val:.10f}")
        if val < best_score:
            return seq, val

    return best_seq, best_score


def normalize_prob(a):
    """Normalize a non-negative sequence to sum 1 as a numpy array."""
    arr = np.asarray(a, dtype=np.float64)
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(np.sum(arr))
    if s <= 1e-15:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    arr = arr / s
    arr = np.maximum(arr, 1e-15)
    arr = arr / float(np.sum(arr))
    return arr


def convolution_fast(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    size = 1
    while size < L:
        size <<= 1
    fa = rfft(a, size)
    c = irfft(fa * fa, size)[:L]
    c[c < 0.0] = 0.0
    return c


def cross_convolution_fast(a, b, fa=None, size=None):
    """Linear convolution a*b via FFT.  Optional precomputed FFT of a."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    L = 2 * n - 1
    if size is None:
        size = 1
        while size < L:
            size <<= 1
    if fa is None:
        fa = rfft(a, size)
    fb = rfft(b, size)
    c = irfft(fa * fb, size)[:L]
    c[c < 0.0] = 0.0
    return c


def internal_objective(a):
    """Surrogate objective: n * max(auto_convolution) for sum(a)=1."""
    c = convolution_fast(a)
    if c.size == 0:
        return np.inf
    return float(len(a) * np.max(c))


def active_gradient(a, c, beta=1e6, top_count=128):
    """
    Gradient of a smoothed max of the convolution entries.

    c[t] corresponds to convolution index t.  The exact gradient of c[t]
    with respect to a[i] is 2*a[t-i] when 0 <= t-i < len(a).
    """
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = a.size
    L = c.size
    if L == 0:
        return np.zeros(n, dtype=np.float64)

    m = float(np.max(c))
    if not np.isfinite(m) or m <= 0.0:
        return np.zeros(n, dtype=np.float64)

    top_count = max(1, min(int(top_count), L))

    z = beta * (c - m)
    z = np.where(np.isfinite(z), z, 0.0)
    expz = np.exp(z)
    sw = float(np.sum(expz))

    if sw <= 1e-300:
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w = expz / sw

    if L > top_count:
        idx = np.argpartition(w, -top_count)[-top_count:]
    else:
        idx = np.arange(L)

    threshold = max(1e-14, float(np.max(w)) * 1e-8)
    keep = idx[w[idx] > threshold]

    grad = np.zeros(n, dtype=np.float64)

    for t in keep:
        t = int(t)
        wt = float(w[t])
        i_lo = max(0, t - (n - 1))
        i_hi = min(n - 1, t)
        if i_lo > i_hi:
            continue

        j_lo = t - i_hi
        j_hi = t - i_lo

        if j_hi == j_lo:
            grad[i_lo] += 2.0 * wt * a[j_lo]
        else:
            if j_lo == 0:
                seg = a[j_hi::-1]
            else:
                seg = a[j_hi:j_lo - 1:-1]
            grad[i_lo:i_hi + 1] += 2.0 * wt * seg

    return grad


def softmax_from_u(u):
    """Exponential-normalize a log-coordinate vector."""
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return np.array([], dtype=np.float64)
    u = u - np.mean(u)
    u = u - np.max(u)
    out = np.exp(u)
    s = float(np.sum(out))
    if s <= 1e-300:
        return np.ones(u.size, dtype=np.float64) / float(u.size)
    return out / s


def adam_smooth_search(a, budget, beta_init=1e6, lr_init=1.0, verbose=False):
    """
    Adam ascent/descent on log-simplex coordinates for the active-lag
    smooth-max of the auto-convolution.
    """
    a = normalize_prob(a)
    n = a.size
    if n < 2 or budget <= 0:
        return a

    start = time.monotonic()
    rng = np.random.default_rng(987654321)

    u = np.log(np.maximum(a, 1e-300))
    u -= np.mean(u)

    current = a.copy()
    current_obj = internal_objective(current)
    best = current.copy()
    best_obj = current_obj

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1 = 0.9
    b2 = 0.99
    lr = float(lr_init)
    beta = float(beta_init)
    betas = [1e6, 1e5, 1e4, 1e3, 1e2]
    beta_idx = 0
    stall = 0
    it = 0
    max_steps = 60000

    while time.monotonic() - start < budget and it < max_steps:
        it += 1
        c = convolution_fast(current)
        top_count = max(8, min(128, n // 25))
        grad = active_gradient(current, c, beta=beta, top_count=top_count)
        if grad is None or np.all(grad == 0.0):
            break

        gu = current * grad
        gu -= current * float(np.sum(current * grad))
        gu -= np.mean(gu)

        m = b1 * m + (1.0 - b1) * gu
        v = b2 * v + (1.0 - b2) * (gu * gu)

        corr1 = 1.0 - b1 ** it
        corr2 = 1.0 - b2 ** it
        step_vec = (m / corr1) / (np.sqrt(v / corr2) + 1e-8)

        accepted = False
        for fac in (1.0, 0.6, 0.35, 0.15, 0.05):
            u_try = u - fac * lr * step_vec
            u_try -= np.mean(u_try)
            a_try = softmax_from_u(u_try)
            obj_try = internal_objective(a_try)

            if obj_try < current_obj - 1e-12:
                current = a_try
                u = u_try
                current_obj = obj_try
                if obj_try < best_obj - 1e-12:
                    best = a_try.copy()
                    best_obj = obj_try
                    stall = 0
                accepted = True
                break

        if accepted:
            lr = min(8.0, lr * 1.03)
        else:
            stall += 1
            lr = max(0.05, lr * 0.75)

            if stall > 70 and beta_idx < len(betas) - 1:
                beta_idx += 1
                beta = betas[beta_idx]

            if stall > 180:
                u = u + rng.normal(0.0, 0.08, n)
                u -= np.mean(u)
                current = softmax_from_u(u)
                current_obj = internal_objective(current)
                if current_obj < best_obj - 1e-12:
                    best = current.copy()
                    best_obj = current_obj
                stall = 0
                lr = 1.0
                m = 0.5 * m
                v = 0.5 * v
                beta_idx = min(beta_idx + 1, len(betas) - 1)
                beta = betas[beta_idx]

        if verbose and it % 2000 == 0:
            print(f"    adam step {it}: current={current_obj:.8f} best={best_obj:.8f} beta={beta:.3g}")

    return best


def solve_active_lp(a, top_k=64, max_outer=5):
    """
    Solve min M s.t. (a*z)[active_lags] <= M, sum(z)=1, z>=0.

    Uses cutting-plane: start from the current top convolution lags,
    then add the worst violating lag from the new cross-convolution.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return None
    if linprog is None:
        return None

    c = convolution_fast(a)
    L = c.size
    top_k = max(1, min(int(top_k), L))
    active = np.argsort(c)[-top_k:]
    max_active = 160

    Lconv = 2 * n - 1
    size = 1
    while size < Lconv:
        size <<= 1
    fa = rfft(a, size)

    z = None

    for _ in range(int(max_outer)):
        active = np.unique(active)
        active = np.asarray(active, dtype=int)
        k = len(active)
        if k <= 0:
            break

        A = np.zeros((k, n), dtype=np.float64)
        for r, t in enumerate(active):
            t = int(t)
            i_lo = max(0, t - (n - 1))
            i_hi = min(n - 1, t)
            if i_lo <= i_hi:
                j = np.arange(i_lo, i_hi + 1, dtype=int)
                A[r, j] = a[t - j]

        cvec = np.zeros(n + 1, dtype=np.float64)
        cvec[-1] = 1.0

        A_ub = np.zeros((k, n + 1), dtype=np.float64)
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(k, dtype=np.float64)

        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0], dtype=np.float64)

        bounds = [(0.0, None)] * n + [(None, None)]

        try:
            res = linprog(
                cvec,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs"
            )
        except Exception:
            return z

        if res is None or res.x is None:
            if z is not None:
                return z
            break

        z_tmp = np.maximum(res.x[:n], 0.0)
        s_tmp = float(np.sum(z_tmp))
        if s_tmp <= 1e-15:
            z_tmp = np.ones(n, dtype=np.float64) / float(n)
        else:
            z_tmp = z_tmp / s_tmp

        cross = cross_convolution_fast(a, z_tmp, fa=fa, size=size)
        max_cross = float(np.max(cross))
        M_val = float(res.x[-1])
        if not np.isfinite(M_val) or M_val <= 0.0:
            M_val = max_cross

        if max_cross <= M_val * (1.0 + 1e-5) + 1e-12:
            return z_tmp

        worst = int(np.argmax(cross))

        if len(active) == L:
            return z_tmp

        if np.count_nonzero(active == worst) == 0:
            active = np.append(active, worst)
            if len(active) > max_active:
                order = np.argsort(c[active])[::-1]
                active = active[order[:max_active]]
                if np.count_nonzero(active == worst) == 0:
                    active[-1] = worst
        else:
            if max_cross <= M_val * 1.1 + 1e-10:
                return z_tmp
            break

        z = z_tmp

    return z


def lp_polish(a, budget, max_iter=30):
    """
    LP alternating polish using active-set cutting-plane subproblems.
    Minimize the auto-convolution surrogate.
    """
    a = normalize_prob(a)
    n = a.size
    if n < 2 or budget <= 0:
        return a
    if linprog is None:
        return a
    if n > 1200:
        return a

    start = time.monotonic()
    current = a.copy()
    current_obj = internal_objective(current)
    best = current.copy()
    best_obj = current_obj
    rng = np.random.default_rng(24601)

    top_k = max(16, min(80, n // 20))
    it = 0

    while time.monotonic() - start < budget and it < int(max_iter):
        it += 1
        z = solve_active_lp(current, top_k=top_k, max_outer=4)
        if z is None:
            break

        improved = False
        for mix in (1.0, 0.8, 0.55, 0.35, 0.2):
            cand = (1.0 - mix) * current + mix * z
            cand = normalize_prob(cand)
            obj = internal_objective(cand)
            if obj < current_obj - 1e-12:
                current = cand
                current_obj = obj
                if obj < best_obj - 1e-12:
                    best = cand.copy()
                    best_obj = obj
                improved = True
                break

        if not improved:
            cand = current * np.exp(rng.normal(0.0, 0.035, n))
            cand = normalize_prob(cand)
            obj = internal_objective(cand)
            if obj < current_obj - 1e-12:
                current = cand
                current_obj = obj
                if obj < best_obj - 1e-12:
                    best = cand.copy()
                    best_obj = obj
            else:
                break

    return best


def resample_sequence(a, new_n):
    """Resample a sequence to a new length by linear interpolation."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = max(1, int(new_n))
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    out = np.interp(x_new, x_old, a)
    return normalize_prob(out)


def make_shape(n, kind):
    """Generate a positive smooth shape of length n."""
    n = max(1, int(n))
    t = np.linspace(0.0, 1.0, n)
    if kind == 0:
        s = np.ones(n, dtype=np.float64)
    elif kind == 1:
        s = 0.5 + 0.5 * np.cos(2.0 * math.pi * t)
    elif kind == 2:
        s = np.sin(math.pi * t) ** 2
    elif kind == 3:
        s = np.exp(-0.5 * ((t - 0.5) / 0.20) ** 2)
    elif kind == 4:
        s = 1.0 - np.abs(2.0 * t - 1.0)
    elif kind == 5:
        s = 1.0 + 0.25 * np.cos(4.0 * math.pi * t)
    elif kind == 6:
        s = 1.0 + 0.20 * np.cos(6.0 * math.pi * t)
    else:
        s = np.exp(-((t - 0.3) ** 2) / 0.02) + np.exp(-((t - 0.7) ** 2) / 0.02) + 0.1
    s = np.where(np.isfinite(s), s, 1.0)
    s = np.maximum(s, 1e-6)
    return s


def build_feature_bases(n):
    """Build zero-mean unit-norm smooth basis functions."""
    n = max(1, int(n))
    if n <= 1:
        return []
    t = np.linspace(0.0, 1.0, n)
    raw = []

    b = t - 0.5
    raw.append(b)

    b = (t - 0.5) ** 2
    raw.append(b)

    for f in range(1, 7):
        raw.append(np.sin(2.0 * math.pi * f * t))
        raw.append(np.cos(2.0 * math.pi * f * t))

    centers = (0.15, 0.3, 0.5, 0.7, 0.85)
    widths = (0.05, 0.10, 0.18)
    for c in centers:
        for w in widths:
            raw.append(np.exp(-0.5 * ((t - c) / w) ** 2))

    bases = []
    for b in raw:
        b = np.asarray(b, dtype=np.float64)
        b = b - np.mean(b)
        nm = float(np.linalg.norm(b))
        if nm > 1e-12:
            bases.append(b / nm)
    return bases


def blackbox_local_search(base, budget, rng, initial_score=np.inf):
    """
    Generic black-box local search using true evaluate_sequence.

    Uses smooth feature perturbations, raw multiplicative noise, block noise,
    and small mass transfers.  Includes mild simulated annealing.
    """
    a = normalize_prob(base)
    n = a.size
    if n == 0:
        return np.ones(1, dtype=np.float64), np.inf

    best = a.copy()
    if np.isfinite(initial_score):
        best_score = float(initial_score)
    else:
        best_score = true_score(safe_sequence(best))
        if not np.isfinite(best_score):
            return best, np.inf

    current = a.copy()
    current_score = best_score

    bases = build_feature_bases(n)

    if n <= 500:
        max_evals = 1200
    elif n <= 800:
        max_evals = 700
    elif n <= 1200:
        max_evals = 350
    elif n <= 2000:
        max_evals = 160
    else:
        max_evals = 70

    step = 0.05
    temp = 0.0025
    no_improve = 0
    evals = 0
    start = time.monotonic()

    while time.monotonic() - start < budget and evals < max_evals:
        mode = int(rng.integers(0, 4))
        cand = current.copy()

        if mode == 0 and len(bases) > 0:
            k = int(rng.integers(1, 4))
            idxs = rng.integers(0, len(bases), k)
            coeffs = rng.uniform(0.5, 1.5, k) * step
            signs = np.ones(k, dtype=np.float64)
            signs[rng.random(k) < 0.5] = -1.0
            logp = np.zeros(n, dtype=np.float64)
            for ii in range(k):
                logp += signs[ii] * coeffs[ii] * bases[int(idxs[ii])]
            cand *= np.exp(logp)

        elif mode == 1:
            cand *= np.exp(rng.normal(0.0, step, n))

        elif mode == 2:
            nblocks = max(5, n // 80)
            bs = max(1, n // nblocks)
            for _ in range(4):
                b = int(rng.integers(0, nblocks))
                scale = 1.0 + step * rng.uniform(-1.0, 1.0)
                if scale < 1e-6:
                    scale = 1e-6
                lo = int(b * bs)
                hi = int(min(n, (b + 1) * bs))
                cand[lo:hi] *= scale

        else:
            if n >= 2:
                for _ in range(6):
                    i = int(rng.integers(0, n))
                    j = int(rng.integers(0, n))
                    if i == j:
                        continue
                    delta = step * min(float(cand[i]), float(cand[j])) * rng.uniform(0.1, 1.0)
                    if delta <= 0.0:
                        continue
                    if rng.random() < 0.5:
                        if cand[i] > delta:
                            cand[i] -= delta
                            cand[j] += delta
                    else:
                        if cand[j] > delta:
                            cand[j] -= delta
                            cand[i] += delta

        cand = normalize_prob(cand)
        val = true_score(safe_sequence(cand))
        evals += 1
        temp = max(1e-7, temp * 0.995)

        if np.isfinite(val) and val < best_score - 1e-12:
            best = cand.copy()
            best_score = val
            current = cand.copy()
            current_score = val
            step = min(1.0, step * 1.15)
            no_improve = 0
            print(f"    blackbox improve n={n} score={val:.10f} evals={evals}")
        else:
            if np.isfinite(val):
                delta = val - current_score
                if delta < 0.0:
                    current = cand.copy()
                    current_score = val
                    step = min(1.0, step * 1.03)
                else:
                    prob = math.exp(-min(20.0, max(0.0, delta) / max(temp, 1e-8)))
                    if rng.random() < prob:
                        current = cand.copy()
                        current_score = val
            no_improve += 1
            step = max(0.005, step * 0.97)

            if no_improve > 120:
                current = best.copy()
                current_score = best_score
                step = 0.08
                temp = max(temp, 0.002)
                no_improve = 0

    return best, best_score


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search entry point.

    Returns a Python list of non-negative floats.
    """
    start_time = time.monotonic()
    total_budget = max(120.0, float(budget_s))
    deadline = start_time + total_budget - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf
    parent_n = 800

    # Load parent construction if available.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_n = int(parent.size)
            best_seq, best_score = consider_candidate(parent, best_seq, best_score, "parent")
    except Exception:
        pass

    # Initial uniform candidates.
    init_end = start_time + min(60.0, total_budget * 0.05)
    n_candidates = []
    for n in (100, 200, 300, 400, 500, 600, 700, 800, 900, 1000, 1200, 1500, 2000, 2500):
        if 50 <= n <= 3000:
            n_candidates.append((abs(n - parent_n), n))
    n_candidates.sort()
    for _, n in n_candidates:
        if time.monotonic() > init_end:
            break
        best_seq, best_score = consider_candidate(np.full(n, 1.0), best_seq, best_score, f"uniform {n}")

    if best_seq is None:
        best_seq, best_score = consider_candidate(np.full(parent_n, 1.0), best_seq, best_score, "fallback uniform")

    base_arr = normalize_prob(best_seq)
    base_n = int(base_arr.size)
    print(f"Initial best n={base_n} score={best_score:.10f}")

    # Initial smooth shape candidates.
    shape_end = start_time + min(90.0, total_budget * 0.08)
    for kind in range(1, 8):
        if time.monotonic() > shape_end:
            break
        best_seq, best_score = consider_candidate(make_shape(base_n, kind), best_seq, best_score, f"shape{kind}")
        other_n = min(3000, base_n + 200)
        if other_n != base_n:
            best_seq, best_score = consider_candidate(make_shape(other_n, kind), best_seq, best_score, f"shape{kind}_{other_n}")
        base_arr = normalize_prob(best_seq)
        base_n = int(base_arr.size)

    # Phase 1: generic black-box local search.
    if deadline - time.monotonic() > 120:
        alloc = min(250.0, (deadline - time.monotonic()) * 0.30)
        arr_opt, _ = blackbox_local_search(base_arr, alloc, rng, best_score)
        best_seq, best_score = consider_candidate(arr_opt, best_seq, best_score, "blackbox1")
        base_arr = normalize_prob(best_seq)

    # Phase 2: Adam smooth-max internal optimizer.
    if deadline - time.monotonic() > 150:
        alloc = min(250.0, (deadline - time.monotonic()) * 0.30)
        arr_opt = adam_smooth_search(base_arr, alloc, beta_init=1e6, lr_init=1.0, verbose=True)
        best_seq, best_score = consider_candidate(arr_opt, best_seq, best_score, "adam1")
        base_arr = normalize_prob(best_seq)

    # Phase 3: LP active-set polish.
    if deadline - time.monotonic() > 120 and len(base_arr) <= 1000:
        alloc = min(200.0, (deadline - time.monotonic()) * 0.25)
        arr_opt = lp_polish(base_arr, alloc, max_iter=20)
        best_seq, best_score = consider_candidate(arr_opt, best_seq, best_score, "lp1")
        base_arr = normalize_prob(best_seq)

    # Phase 4: black-box again after internal improvements.
    if deadline - time.monotonic() > 120:
        alloc = min(200.0, (deadline - time.monotonic()) * 0.30)
        arr_opt, _ = blackbox_local_search(base_arr, alloc, rng, best_score)
        best_seq, best_score = consider_candidate(arr_opt, best_seq, best_score, "blackbox2")
        base_arr = normalize_prob(best_seq)

    # Phase 5: length exploration.
    multi_end = time.monotonic() + min(300.0, max(60.0, (deadline - time.monotonic()) * 0.35))
    current_base = int(base_arr.size)
    n_set = set()
    for n in (
        current_base - 400, current_base - 200, current_base - 100, current_base - 50,
        current_base, current_base + 50, current_base + 100, current_base + 200, current_base + 400,
        300, 400, 500, 600, 700, 800, 900, 1000, 1200, 1500, 2000
    ):
        if 100 <= n <= 3000:
            n_set.add(int(n))

    n_order = []
    for n in n_set:
        n_order.append((abs(n - current_base), n))
    n_order.sort()

    tried = 0
    for _, n in n_order:
        if time.monotonic() > multi_end:
            break
        if tried >= 8:
            break
        if time.monotonic() > deadline - 60:
            break
        tried += 1

        arr_try = resample_sequence(base_arr, n)
        best_seq, best_score = consider_candidate(arr_try, best_seq, best_score, f"resample {n}")
        base_arr = normalize_prob(best_seq)

        if deadline - time.monotonic() > 100:
            alloc = min(70.0, (deadline - time.monotonic()) * 0.25)
            arr_opt = adam_smooth_search(arr_try, alloc, beta_init=1e6, lr_init=1.0, verbose=False)
            best_seq, best_score = consider_candidate(arr_opt, best_seq, best_score, f"adam n={n}")
            base_arr = normalize_prob(best_seq)

        if deadline - time.monotonic() > 70:
            alloc = min(35.0, (deadline - time.monotonic()) * 0.20)
            arr_opt, _ = blackbox_local_search(arr_try, alloc, rng, best_score)
            best_seq, best_score = consider_candidate(arr_opt, best_seq, best_score, f"bb n={n}")
            base_arr = normalize_prob(best_seq)

    # Phase 6: structured restarts from the current best.
    restart_end = time.monotonic() + min(300.0, max(60.0, (deadline - time.monotonic()) * 0.35))
    restarts = 0
    n_best = int(base_arr.size)

    while time.monotonic() < restart_end and restarts < 5:
        if time.monotonic() > deadline - 80:
            break

        if restarts % 3 == 0:
            cand = base_arr * np.exp(rng.normal(0.0, 0.04 + 0.015 * restarts, n_best))
        elif restarts % 3 == 1:
            cand = base_arr.copy()
            nblocks = max(5, n_best // 100)
            bs = max(1, n_best // nblocks)
            for _ in range(3):
                b = int(rng.integers(0, nblocks))
                scale = rng.uniform(0.6, 1.4)
                lo = int(b * bs)
                hi = int(min(n_best, (b + 1) * bs))
                cand[lo:hi] *= scale
        else:
            cand = base_arr.copy()
            k = max(5, n_best // 40)
            idx = rng.integers(0, n_best, k)
            scales = rng.uniform(0.5, 1.8, k)
            cand[idx] *= scales

        cand = normalize_prob(cand)
        best_seq, best_score = consider_candidate(cand, best_seq, best_score, f"restart{restarts}")

        if deadline - time.monotonic() > 80:
            alloc = min(55.0, (deadline - time.monotonic()) * 0.35)
            arr_opt, _ = blackbox_local_search(cand, alloc, rng, best_score)
            best_seq, best_score = consider_candidate(arr_opt, best_seq, best_score, f"restart{restarts}bb")
            base_arr = normalize_prob(best_seq)

        if deadline - time.monotonic() > 60:
            alloc = min(45.0, (deadline - time.monotonic()) * 0.30)
            arr_opt = adam_smooth_search(cand, alloc, beta_init=1e6, lr_init=1.0, verbose=False)
            best_seq, best_score = consider_candidate(arr_opt, best_seq, best_score, f"restart{restarts}adam")
            base_arr = normalize_prob(best_seq)

        restarts += 1

    # Final polishing.
    if deadline - time.monotonic() > 30:
        base_arr = normalize_prob(best_seq)
        alloc = min(35.0, (deadline - time.monotonic()) * 0.50)
        arr_opt = adam_smooth_search(base_arr, alloc, beta_init=1e6, lr_init=1.0, verbose=False)
        best_seq, best_score = consider_candidate(arr_opt, best_seq, best_score, "final adam")

    if deadline - time.monotonic() > 20:
        base_arr = normalize_prob(best_seq)
        alloc = min(25.0, (deadline - time.monotonic()) * 0.50)
        arr_opt, _ = blackbox_local_search(base_arr, alloc, rng, best_score)
        best_seq, best_score = consider_candidate(arr_opt, best_seq, best_score, "final bb")

    # Final verification and fallback.
    if best_seq is None:
        final = safe_sequence(np.full(parent_n, 1.0))
    else:
        final = safe_sequence(best_seq)

    val = true_score(final)
    if not np.isfinite(val):
        final = [1.0] * 100
        val = true_score(final)

    elapsed = time.monotonic() - start_time
    print(f"FINAL n={len(final)} score={val:.10f} elapsed={elapsed:.1f}s")

    return final
```
