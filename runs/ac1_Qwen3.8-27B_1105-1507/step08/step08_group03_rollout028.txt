Let me also consider a completely different approach: **alternating minimization with Bregman projections**.

The problem is: min max_k (a*a)[k] s.t. sum(a)=1, a>=0.

This can be seen as a two-player zero-sum game:
- Player 1 chooses a (on the simplex)
- Player 2 chooses k (the index)
- Payoff is (a*a)[k]

The value of this game is what we want to minimize.

By the minimax theorem (for the relaxed game where Player 2 can choose a mixed strategy over k), the value is:
min_a max_k (a*a)[k] = min_a max_{p in simplex} sum_k p_k (a*a)[k] = min_a max_{p} p^T A(a) 1 (where A(a) is the convolution matrix)

Let me think differently. Fix a. Then max_k (a*a)[k] is a function of a. We want to minimize it.

The subgradient of max_k (a*a)[k] with respect to a is (a*a)[k*] backpropagated, which gives the vector where the i-th component is a[k*-i] (summed over relevant k*).

Actually, d(a*a)[k] / da[i] = a[k-i] (if 0 <= k-i < n, else 0).

So the subgradient of max_k (a*a)[k] at a is the vector g where g[i] = sum_{k in S} a[k-i] / |S| (approximately, where S is the set of active indices).

For the max, the subgradient is a convex combination of the gradients at the active points. If only one point is active (k*), then g[i] = a[k*-i] (for valid i).

This is exactly what the Nesterov subgradient method in the previous code does!

```python
import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def objective(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    a = a/s
    return 2.0*n*float(np.max(compute_conv(a)))

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u*idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n)/n
    rho = rho[-1]
    theta = (cssv[rho]-1.0)/(rho+1.0)
    return np.maximum(v-theta, 0.0)

def grad_conv_k(a, k):
    """Gradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    i_lo = max(0, k-(n-1))
    i_hi = min(n-1, k)
    for i in range(i_lo, i_hi+1):
        g[i] = a[k-i]
    return g
```

"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses accelerated LP alternating with HiGHS, adaptive step sizes,
Nesterov subgradient refinement, multi-n exploration, and structured restarts.
Key improvements over previous: faster LP via constraint ordering, momentum-based
updates, smarter restarts using polynomial shapes, and systematic n-sweeping.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_conv_at_k(a, k):
    """Gradient of (a*a)[k] with respect to a. Vector g where g[i] = a[k-i] for valid i."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    # Use scipy HiGHS
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_reduced(a, top_k=80):
    """
    Solve LP with only the top constraints for speed.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / float(np.sum(a))
    A_full = build_conv_matrix(a)

    # Use current a as proxy for which constraints are tight
    conv_a = A_full @ a
    top_idx = np.argsort(conv_a)[::-1][:top_k]

    A = A_full[top_idx, :]
    b_ub = np.zeros(len(top_idx))

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(top_idx), n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                g_val = g_val / s
                actual_conv = A_full @ g_val
                M_actual = float(np.max(actual_conv))
                return g_val, M_actual
    except Exception:
        pass
    return None, None


def lp_alternate(a, time_budget=200.0, verbose=False):
    """
    LP alternating method with adaptive step size and momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.3
    best_t = 0.4
    stall = 0
    rng = np.random.default_rng(12345)
    it = 0
    use_reduced = (n > 400)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break

        # Solve LP
        if use_reduced:
            g_lp, M_lp = solve_lp_reduced(a, top_k=min(100, 2 * n // 3))
            if g_lp is None:
                g_lp, M_lp = solve_lp_minpeak(a)
        else:
            g_lp, M_lp = solve_lp_minpeak(a)

        if g_lp is None:
            break

        # Adaptive line search
        ts = [best_t, best_t * 0.5, best_t * 1.5, 0.1, 0.3, 0.5, 0.7, 1.0]
        ts = sorted(set([t for t in ts if 0.01 < t <= 1.0]))

        # Add momentum direction
        a_mom = a + 0.2 * momentum
        a_mom = np.maximum(a_mom, 0)
        sm = float(np.sum(a_mom))
        if sm > 1e-10:
            a_mom = a_mom / sm

        best_t_obj = np.inf
        best_t_val = best_t
        best_a_local = a.copy()

        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        # Also try momentum-blended step
        for t in [0.3, 0.5, 0.7]:
            a_try = (1.0 - t) * a_mom + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-12:
            momentum = beta * momentum + (best_a_local - a)
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            best_t = best_t_val
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 12:
                # Escape: structured perturbation
                scale = min(0.25, 0.04 + 0.03 * (stall // 12))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                best_a = a.copy()
                best_obj = compute_obj(a)
                momentum = np.zeros(n)
                stall = 0
                best_t = 0.4

    return best_a


def nesterov_subgrad(a, time_budget=100.0, K=5, verbose=False):
    """Nesterov-accelerated subgradient on simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 3.0 / np.sqrt(n)
    stall = 0

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        # Weighted subgradient with exponential weighting
        spread = max(peak_vals[0] - peak_vals[-1], 1e-15)
        temp = 5.0 / spread
        weights = np.exp(-(peak_vals - peak_vals[-1]) * temp)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                g += w * grad_conv_at_k(y, int(k))

        lr = lr0 / np.sqrt(t_k)
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 60:
                t_k = 1.0
                y = best_a.copy()
                a_prev = y.copy()
                y_new = y.copy()
                stall = 0

        a_prev = a_new.copy()
        y = y_new.copy()
        t_k = t_new
        it += 1

        if verbose and it % 500 == 0:
            print(f"    NS it{it}: obj={best_obj:.8f}, t_k={t_k:.1f}, el={time.monotonic()-start:.1f}s")

    return best_a


def coordinate_descent(a, time_budget=30.0):
    """Coordinate descent: move mass away from peak-contributing positions."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Find top contributors to peak
        contrib = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib.append((a[i] * a[j], i, j))
        contrib.sort(reverse=True)

        # Find non-contributing positions
        non_contrib = []
        for i in range(n):
            j = k_star - i
            if j < 0 or j >= n:
                non_contrib.append(i)

        if not non_contrib:
            # Fallback: use tail positions
            non_contrib = list(range(max(0, n - 20), n)) + list(range(0, min(20, n)))
            non_contrib = [x for x in non_contrib if x != k_star]

        improved = False
        for (val, i, j) in contrib[:20]:
            if val < 1e-14:
                break
            src = i if a[i] >= a[j] else j
            for target in non_contrib[:10]:
                if target == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = frac * a[src]
                    if delta < 1e-15:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    s2 = float(np.sum(a))
                    if s2 < 1e-10:
                        a[src] += delta
                        a[target] -= delta
                        break
                    obj2 = compute_obj(a)
                    if obj2 < best_obj - 1e-13:
                        best_obj = obj2
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break

        if not improved:
            # Try random coordinate moves
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                delta = 0.01 * a[i]
                if delta < 1e-15:
                    continue
                a[i] -= delta
                a[j] += delta
                s2 = float(np.sum(a))
                if s2 > 1e-10:
                    obj2 = compute_obj(a)
                    if obj2 < best_obj - 1e-13:
                        best_obj = obj2
                        best_a = a.copy()
                        improved = True
                    else:
                        a[i] += delta
                        a[j] -= delta
            if not improved:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Power ramp
        a = np.arange(1, n + 1, dtype=np.float64) ** 1.5
        a = a / a.sum()
    elif mode == 3:
        # Inverse ramp
        a = np.arange(n, 0, -1, dtype=np.float64) ** 1.5
        a = a / a.sum()
    elif mode == 4:
        # Half-ramp (increasing only)
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 5:
        # Gaussian centered
        x = np.linspace(-1, 1, n)
        a = np.exp(-x**2 * 2)
        a = a / a.sum()
    elif mode == 6:
        # U-shaped
        x = np.linspace(-1, 1, n)
        a = x**2 + 0.1
        a = a / a.sum()
    elif mode == 7:
        # Random smooth
        a = rng.uniform(0, 1, n)
        from scipy.ndimage import uniform_filter1d
        a = uniform_filter1d(a, size=max(2, n // 50))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a
