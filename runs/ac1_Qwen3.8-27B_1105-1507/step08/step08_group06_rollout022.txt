"""
Optimized step function search combining:
1. LP alternating with Anderson acceleration
2. Parametric family optimization
3. Smooth max + L-BFGS-B
4. Aggressive length sweep
5. Coordinate-level polish
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar
import time
import cvxpy as cp


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a, g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.HIGHS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                    return g_val, M_val
        except Exception:
            continue

    return None, None


def anderson_lp_alternating(a, time_budget=300.0, m_anderson=5, verbose=False):
    """LP alternating with Anderson acceleration."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Anderson acceleration history
    history_x = [a.copy()]
    history_F = [a.copy()]  # F(x) = best response - x = g - x

    alpha_anderson = 1.0  # weight for Anderson combination
    gamma = 0.5  # weight for base iteration

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        # Compute best response
        g, M_lp = solve_lp(a)
        if g is None:
            break

        # Base residual
        F = g - a

        if len(history_x) < 2:
            # Simple step
            a_new = a + alpha_anderson * F
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
        else:
            # Anderson acceleration
            m = min(m_anderson, len(history_x))
            X = np.stack(history_x[-m:])  # (m, n)
            F_h = np.stack(history_F[-m:])  # (m, n)

            # Solve for omega coefficients
            # We want: x_new = (1-tau)*sum(omega_i * x_i) + tau * (x + F)
            # where omega is from: (F_h - F_h[0]) @ omega = F_h[0] - F
            # Simplified Anderson:
            diff_F = F_h[:-1] - F_h[-1:]  # (m-1, n)
            target = F_h[-1] - F  # (n,)
            
            # Solve least squares: diff_F @ omega = target
            try:
                omega, _, _, _ = np.linalg.lstsq(diff_F, target, rcond=None)
                omega = np.clip(omega, 0, None)
                if np.sum(omega) > 1:
                    omega = omega / np.sum(omega)
            except:
                omega = np.ones(m - 1) / (m - 1)

            # Anderson combination
            x_combined = np.sum(omega * history_x[-m-1:-1], axis=0)
            x_base = a + F  # = g
            
            a_new = (1 - gamma) * x_combined + gamma * x_base
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s_new

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            if verbose and it % 20 == 0:
                print(f"    Anderson LP it={it}: obj={best_obj:.8f}")
        else:
            # Try a more conservative step
            for shrink in [0.5, 0.25, 0.1]:
                a_try = a + shrink * F
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    F = g - a
                    break
            else:
                # No improvement, perturb
                if it % 50 == 49:
                    rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                    a = best_a * np.exp(rng.normal(0, 0.02, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    history_x = [a.copy()]
                    history_F = [a.copy()]
                    continue

        # Update history
        history_x.append(a.copy())
        history_F.append((g - a).copy())
        if len(history_x) > m_anderson + 1:
            history_x = history_x[-(m_anderson + 1):]
            history_F = history_F[-(m_anderson + 1):]

    return best_a


def smooth_max_conv(a, temperature):
    """Smooth max of convolution using log-sum-exp."""
    conv = compute_conv(a)
    scaled = conv / temperature
    m = np.max(scaled)
    return temperature * (m + np.log(np.sum(np.exp(scaled - m))))


def parametric_search(n, time_budget=100.0):
    """Search over parametric families."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    # Family 1: Raised cosine with parameter
    def obj_cosine(param):
        beta = param
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x) ** (1/beta if beta > 0.01 else 100))
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-10:
            return 1e6
        a = a / s
        return compute_obj(a)

    # Family 2: Gaussian
    def obj_gaussian(param):
        sigma = param * n / 2
        x = np.arange(n)
        center = (n - 1) / 2.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        a = np.maximum(a, 1e-10)
        a = a / np.sum(a)
        return compute_obj(a)

    # Family 3: Power law (symmetric)
    def obj_power(param):
        alpha = param
        x = np.arange(n)
        center = (n - 1) / 2.0
        d = np.abs(x - center) + 1
        a = d ** (-alpha)
        a = a / np.sum(a)
        return compute_obj(a)

    # Family 4: Sine window
    def obj_sine(param):
        beta = param
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** beta
        a = np.maximum(a, 1e-10)
        a = a / np.sum(a)
        return compute_obj(a)

    # Family 5: Double peak
    def obj_double(param):
        # Two Gaussian peaks at param*n and (1-param)*n
        x = np.arange(n)
        p = param
        a = np.exp(-2*((x - p*n)**2)/(n*0.1)) + np.exp(-2*((x - (1-p)*n)**2)/(n*0.1))
        a = a / np.sum(a)
        return compute_obj(a)

    families = [
        (obj_cosine, 0.1, 10),
        (obj_gaussian, 0.05, 1.0),
        (obj_power, 0.1, 5.0),
        (obj_sine, 0.1, 10),
        (obj_double, 0.1, 0.45),
    ]

    for func, lo, hi in families:
        if time.monotonic() - start > time_budget:
            break
        try:
            res = minimize_scalar(func, bounds=(lo, hi), method='bounded',
                                  options={'maxiter': 50})
            if res.success:
                obj_val = res.fun
                if obj_val < best_obj:
                    # Recover the sequence
                    # (We'd need to reconstruct a from the parameter)
                    best_obj = obj_val
                    print(f"  Parametric: obj={obj_val:.8f} at param={res.x:.4f}")
        except Exception as e:
            pass

    return best_obj


def lbfgs_smooth(a, time_budget=200.0, verbose=False):
    """Optimize using smooth max approximation with L-BFGS-B."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Decreasing temperature schedule
    temps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    for temp in temps:
        if time.monotonic() - start > time_budget:
            break
        alloc = min(30.0, (time_budget - (time.monotonic() - start)) / len(temps))

        def obj_func(x):
            x = np.maximum(x, 0)
            s = np.sum(x)
            if s < 1e-10:
                return 1e6
            x = x / s
            conv = compute_conv(x)
            sm = smooth_max_conv(x, temp)
            return 2.0 * n * sm

        def grad_func(x):
            x = np.maximum(x, 0)
            s = np.sum(x)
            if s < 1e-10:
                return np.zeros(n)
            x = x / s
            conv = compute_conv(x)
            scaled = conv / temp
            m = np.max(scaled)
            weights = np.exp(scaled - m)
            weights /= np.sum(weights)

            # Compute gradient
            grad = np.zeros(n)
            for k in range(min(len(conv), 2*n-1)):
                if weights[k] < 1e-8:
                    continue
                k_int = int(k)
                # Contribution from a[i] to conv[k_int]:
                # a[i] * a[i-k_int] if i >= k_int
                # a[i+k_int] * a[i] if i + k_int < n
                for i in range(max(0, k_int), n):
                    # i >= k_int: a[i]*a[i-k_int], derivative w.r.t. a[i] is a[i-k_int]
                    if i >= k_int:
                        grad[i] += weights[k_int] * a[i - k_int]
                    # i + k_int < n: a[i+k_int]*a[i], derivative w.r.t. a[i] is a[i+k_int]
                    if i + k_int < n:
                        grad[i] += weights[k_int] * a[i + k_int]

            grad *= 2.0 * n
            grad = grad - np.mean(grad)
            return grad

        # Run L-BFGS-B
        try:
            res = minimize(obj_func, a, method='L-BFGS-B', jac=grad_func,
                          options={'maxiter': 200, 'maxfun': 500,
                                   'maxtime': max(1, alloc)})
            if res.x is not None:
                a_new = np.maximum(res.x, 0)
                s_new = np.sum(a_new)
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new.copy()
        except Exception:
            pass

    return best_a

                    omega = omega / wsum
                else:
                    omega = np.ones(m - 1) / (m - 1)
            except Exception:
                omega = np.ones(m - 1) / (m - 1)

            x_comb = np.sum(omega * X_h[:-1], axis=0)
            x_base = a + F
            a_new = (1 - gamma) * x_comb + gamma * x_base
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s_new

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            if verbose and it % 50 == 0:
                print(f"      Anderson it={it}: obj={best_obj:.10f}")
        else:
            improved = False
            for shrink in [0.5, 0.3, 0.1, 0.05]:
                a_try = a + shrink * F
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    stall = 0
                    improved = True
                    break

            if not improved:
                stall += 1
                if stall >= 30:
                    rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                    scale = 0.01 + 0.01 * min(stall // 30, 5)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    hist_x = []
                    hist_F = []
                    stall = 0
                    continue

        hist_x.append(a.copy())
        hist_F.append((g - a).copy())
        if len(hist_x) > m_hist:
            hist_x = hist_x[-m_hist:]
            hist_F = hist_F[-m_hist:]

    return best_a


def compute_smooth_obj_grad(x, temp, n):
    x = np.maximum(x, 0)
    s = np.sum(x)
    if s < 1e-10:
        return np.zeros(n)
    x = x / s
    conv = compute_conv(x)
    scaled = conv / temp
    m = np.max(scaled)
    weights = np.exp(scaled - m)
    weights /= np.sum(weights)

    obj_val = 2.0 * n * temp * (m + np.log(np.sum(weights)))

    grad = np.zeros(n)
    max_k = min(len(conv), 2 * n - 1)
    for k in range(max_k):
        wk = weights[k]
        if wk < 1e-10:
            continue
        k_int = int(k)
        # term1: a[i-k] for i >= k_int
        if k_int < n:
            grad[k_int:] += wk * x[:n - k_int]
        # term2: a[i+k] for i + k < n, i.e., i < n - k_int
        if k_int < n:
            grad[:n - k_int] += wk * x[k_int:]

    grad *= 2.0 * n
    grad = grad - np.mean(grad)
    return obj_val, grad


def lbfgs_smooth_optimize(a, time_budget=200.0, verbose=False):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    temps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    for idx, temp in enumerate(temps):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        alloc = max(2.0, remaining / (len(temps) - idx))

        def obj_wrapper(x):
            x = np.maximum(x, 0)
            s_x = np.sum(x)
            if s_x < 1e-10:
                return 1e6
            x_n = x / s_x
            conv = compute_conv(x_n)
            scaled = conv / temp
            m = np.max(scaled)
            return 2.0 * n * temp * (m + np.log(np.sum(np.exp(scaled - m))))

        def grad_wrapper(x):
            _, g = compute_smooth_obj_grad(x, temp, n)
            return g

        try:
            res = minimize(obj_wrapper, a, method='L-BFGS-B', jac=grad_wrapper,
                           options={'maxiter': 100, 'maxfun': 300, 'maxtime': alloc})
            if res.x is not None:
                a_new = np.maximum(res.x, 0)
                s_new = np.sum(a_new)
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new.copy()
                        if verbose:
                            print(f"    L-BFGS temp={temp}: obj={best_obj:.10f}")
        except Exception:
            pass

    return best_a


def subgradient_polish(a, time_budget=100.0, verbose=False):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    momentum = np.zeros(n)
    beta = 0.7
    lr = 0.03

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(y)
        max_val = float(np.max(conv))
        threshold = max_val * 0.85
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 40:
            active = np.argsort(conv)[::-1][:40]

        grad = np.zeros(n)
        for k in active:
            k_int = int(k)
            for i in range(max(0, k_int), n):
                if i >= k_int:
                    grad[i] += y[i - k_int]
                if i + k_int < n:
                    grad[i] += y[i + k_int]
        grad /= len(active)
        grad = grad - np.mean(grad)

        momentum = beta * momentum - lr * grad
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
            s_new = 1.0
        y_new = y_new / s_new
        y = y_new

        obj = compute_obj(y)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = y.copy()

        if it % 100 == 99:
            lr = max(lr * 0.95, 0.001)

    return best_a


def coordinate_polish(a, time_budget=60.0):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    for iteration in range(300):
        if time.monotonic() - start > time_budget:
            break

        k_star = int(np.argmax(conv))
        improved = False

        contributions = np.zeros(n)
        for j in range(n):
            jk = k_star - j
            if 0 <= jk < n:
                contributions[j] = a[j] * a[jk]

        sorted_idx = np.argsort(contributions)[::-1]

        for frac in [0.003, 0.008, 0.015, 0.03]:
            done = False
            for i in sorted_idx[:30]:
                if done or time.monotonic() - start > time_budget:
                    break
                target = (k_star - int(i) + n // 2) % n
                if target == int(i):
                    target = (int(i) + 1) % n
                delta = frac * a[int(i)]
                if delta <= 0 or delta > a[int(i)]:
                    continue
                a[int(i)] -= delta
                a[target] += delta
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                    done = True
                else:
                    a[int(i)] += delta
                    a[target] -= delta
            if done:
                improved = True

        if not improved:
            rng = np.random.default_rng(int(time.monotonic() * 1e3) % (2**31))
            found = False
            for _ in range(10):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                delta = 0.005 * min(a[i], a[j])
                if delta <= 0:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s2 = float(np.sum(a))
                a = a / s2
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                    found = True
                    break
            if not found:
                a = best_a.copy()
                conv = compute_conv(a)

        if not improved and iteration > 80:
            break

    return best_a


def generate_init(kind, n, rng):
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 8)
        a[:edge] = 0.4
        a[-edge:] = 0.4
        return a / np.sum(a)
    elif kind == 'aliased':
        a = np.ones(n) / n
        period = max(3, n // 12)
        for k in range(period, n, period):
            a[k] *= 0.3
        return a / np.sum(a)
    elif kind == 'log_taper':
        x = np.arange(n)
        center = (n - 1) / 2.0
        d = np.abs(x - center) / center
        a = np.exp(-2 * d)
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def quick_optimize(a, time_budget=30.0):
    n = len(a)
    a = lp_alternating_anderson(a, time_budget=min(time_budget * 0.6, 20.0), m_hist=4)
    a = subgradient_polish(a, time_budget=min(time_budget * 0.3, 10.0))
    return a


def propose_candidate(seed=42, budget_s=1500):

    deadline = start_time + float(budget_s) - 20.0

        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.10f}")

    n_parent = len(best_seq)
    print(f"Starting n={n_parent}, best_score={best_score:.10f}")

    # === Phase 1: Quick LP alternating from parent (80s) ===
    print("\n=== Phase 1: LP Anderson from parent ===")
    remaining = deadline - time.monotonic()

        a1 = lp_alternating_anderson(best_seq.copy(), time_budget=min(80.0, remaining * 0.06),
                                     m_hist=6, verbose=True)

            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase 1: obj={best_score:.10f} **")
        else:
            print(f"  Phase 1: obj={obj1:.10f}")

    # === Phase 2: L-BFGS smooth optimization (200s) ===

            best_seq = a2.copy()

            print(f"  Phase 2: obj={obj2:.10f}")

    # === Phase 3: Main LP Anderson (400s) ===
    print("\n=== Phase 3: Main LP Anderson ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        # Multiple rounds with perturbations
        a3 = best_seq.copy()
        for round_idx in range(3):

            if remaining < 50:
                break
            alloc = min(150.0, remaining * 0.4)

            obj3 = compute_obj(a3)
            if obj3 < best_score:
                best_score = obj3
                best_seq = a3.copy()
                print(f"  Phase 3 R{round_idx}: obj={best_score:.10f} **")
            else:
                print(f"  Phase 3 R{round_idx}: obj={obj3:.10f}")
                if round_idx < 2:
                    a3 = best_seq * np.exp(rng.normal(0, 0.02, len(best_seq)))
                    a3 = np.maximum(a3, 0)
                    a3 = a3 / np.sum(a3)

    # === Phase 4: L-BFGS again from best (150s) ===
    print("\n=== Phase 4: L-BFGS from best ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        a4 = lbfgs_smooth_optimize(best_seq.copy(), time_budget=min(150.0, remaining * 0.12),
                                   verbose=True)

            best_seq = a7.copy()
