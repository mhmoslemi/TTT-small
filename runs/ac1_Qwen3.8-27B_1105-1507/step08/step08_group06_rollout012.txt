Optimized step function search using fast active-set LP alternating,
spectral initialization, Nesterov subgradient polishing, and multi-length sweep.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time
import sys

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max_conv / (sum_a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_active_conv_matrix(a, active_indices):
    """Build matrix A (len(active) x n) such that A @ g = conv(a, g) at active indices."""
    n = len(a)
    m = len(active_indices)
    A = np.zeros((m, n))
    for i, k in enumerate(active_indices):
        k = int(k)
        # (a*g)_k = sum_j a[j] * g[k-j] where 0 <= j < n and 0 <= k-j < n
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[i, j] = a[j]
    return A

def nesterov_subgradient(a, time_budget=50.0, verbosity=0):
    """
    Nesterov-accelerated subgradient method.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = a.copy()  # Nesterov lookahead
    y = a.copy()  # current point
    momentum = np.zeros(n)
    beta = 0.8
    lr = 0.03
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        conv_x = compute_conv(x)
        max_val = float(np.max(conv_x))
        
        # Get active peaks
        threshold = max_val * 0.85
        active = np.where(conv_x >= threshold)[0]
        if len(active) == 0:
            active = [int(np.argmax(conv_x))]
        if len(active) > 40:
            active = np.argsort(conv_x)[::-1][:40]
        
        # Subgradient of max over active peaks
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            indices = np.arange(j_min, j_max + 1)
            grad[indices] += a[k - indices]
        grad /= len(active)
        
        # Project gradient to sum=0 subspace
        grad_centered = grad - np.mean(grad)
        
        # Nesterov momentum
        momentum = beta * momentum - lr * grad_centered
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
            s_new = 1.0
        y_new = y_new / s_new
        
        # Nesterov update
        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        x = y_new + (1 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx
        
        y = y_new
        obj = compute_obj(y)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = y.copy()
        
        # Adaptive learning rate
        if it % 500 == 499:
            lr = max(lr * 0.92, 0.001)
            beta = min(beta + 0.005, 0.95)
    
    return best_a

def coordinate_polish(a, time_budget=30.0):
    """
    Coordinate descent polish: adjust pairs to reduce peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv = compute_conv(a)
    
    for iteration in range(1000):
        if time.monotonic() - start > time_budget:
            break
        
        k_star = int(np.argmax(conv))
        improved = False
        
        # Find top contributors to the peak
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        contributions = np.zeros(n)
        for j in range(j_min, j_max + 1):
            contributions[j] = a[j] * a[k_star - j]
        
        # Try moving mass from high contributors to low contributors
        sorted_idx = np.argsort(contributions)[::-1]
        
        for frac in [0.002, 0.005, 0.01, 0.02]:
            done = False
            for i in sorted_idx[:15]:
                if done or time.monotonic() - start > time_budget:
                    break
                # Move mass to a position that doesn't contribute to the peak
                # Choose a position far from the peak
                target = (int(i) + n // 2) % n
                if target == i:
                    target = (i + 1) % n
                
                delta = frac * a[i]
                if delta <= 0 or delta >= a[i]:
                    continue
                
                a[i] -= delta
                a[target] += delta
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                    done = True
                else:
                    a[i] += delta
                    a[target] -= delta
            
            if done:
                improved = True
        
        if not improved:
            # Random perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            for _ in range(3):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.005 * a[i]
                if delta <= 0:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                    break
        
        if not improved and iteration > 200:
            break
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def make_symmetric(a):
    """Make a sequence symmetric (palindromic)."""
    n = len(a)
    a_sym = np.zeros_like(a)
    for i in range(n):
        a_sym[i] = 0.5 * (a[i] + a[n - 1 - i])
    return a_sym
