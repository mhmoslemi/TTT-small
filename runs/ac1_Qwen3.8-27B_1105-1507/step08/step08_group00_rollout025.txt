    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smoothed_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*conv_k)) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    p = np.exp(tau * (conv - max_c))
    p /= p.sum()
    # grad[i] = 2 * sum_k p[k] * a[k-i] (cross-correlation)
    N = 1
    while N < 3*n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g = ifft(P * np.conj(A)).real[:n] * 2.0
    return g, conv

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
