Minimizes 2*n*max(conv(a,a))/(sum(a))^2.

from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy import sparse
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_lp(a):
    """Solve: min M s.t. A@g <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build sparse constraint matrix
    # A[k, j] = a_norm[k-j] if 0 <= k-j < n else 0
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            if k < 2 * n - 1:
                rows.append(k)
                cols.append(j)
                vals.append(a_norm[k - j])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Full matrix for linprog: [A, -1] g,M <= 0; [1,...,1, 0] g,M = 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = sparse.hstack([A, sparse.csc_matrix((-np.ones(2*n-1), (range(2*n-1), [n]*(2*n-1)), shape=(2*n-1, 1)))], format='coo')
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = g.sum()
        if s > 1e-12:
            g /= s
            return g, float(res.x[n])
    return None, None

        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_A(a):
    """Build dense convolution matrix A where (A@g)[k] = (a*g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        A[j:j + length, j] = a[:length]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_A(a_norm)

    # Full problem matrix
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})

    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = g.sum()
        if s > 1e-12:
            g /= s
            return g, float(max(res.x[n], 0))
    return None, None


def solve_lp_smooth(a, tau=50.0):
    """
    Smooth LP approximation using log-sum-exp.
    Minimizes (1/tau)*log(sum(exp(tau*(A@g)))) over the simplex.
    Returns the optimal g using mirrored gradient descent.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_A(a_norm)
    one = np.ones(n)

    # Start from uniform
    g = one / n
    lr = 1.0 / np.sqrt(n)
    momentum_v = np.zeros(n)
    momentum = 0.7

    for it in range(200):
        Ag = A @ g
        m = Ag.max()
        shifted = tau * (Ag - m)
        weights = np.exp(shifted)
        weights /= weights.sum()

        # Gradient: tau * A^T @ weights (projection onto tangent space)
        grad = tau * (A.T @ weights)
        grad -= one * (grad @ one) / n  # project to tangent of simplex

        momentum_v = momentum * momentum_v + grad
        log_g = np.log(np.maximum(g, 1e-30)) - lr * momentum_v
        log_g -= log_g.max()
        g_new = np.exp(log_g)
        g_new /= g_new.sum()

        if np.max(np.abs(g_new - g)) < 1e-10:
            g = g_new
            break
        g = g_new
        lr *= 0.995

    return g


def line_search_parabolic(a, g_dir, a0_obj):
    """
    Parabolic line search for optimal t in a_new = (1-t)*a + t*g_dir.
    Evaluates at several points and fits a parabola.
    """
    a = a / a.sum()
    g_dir = g_dir / g_dir.sum()

    best_t = 0.0
    best_obj = a0_obj

    # Initial evaluations
    evals = []
    for t in [0.1, 0.3, 0.5, 0.7, 0.9]:
        a_try = (1.0 - t) * a + t * g_dir
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try /= s
        obj = compute_obj(a_try)
        evals.append((t, obj))

    if not evals:
        return a, a0_obj, 0.0

    # Find best in initial sweep
    best_t = min(evals, key=lambda x: x[1])[0]
    best_obj = min(evals, key=lambda x: x[1])[1]

    if best_obj >= a0_obj:
        # Try smaller t values
        for t in [0.02, 0.05, 0.1]:
            a_try = (1.0 - t) * a + t * g_dir
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try /= s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_t = t
                best_obj = obj
        if best_obj >= a0_obj:
            return a, a0_obj, 0.0

    # Parabolic refinement around best_t
    if len(evals) >= 3:
        # Fit parabola to 3 closest points
        evals.sort()
        idx = min(len(evals) - 1, max(0, [e[0] for e in evals].index(best_t)))
        # Use points around best
        candidates = []
        for i in range(max(0, idx - 1), min(len(evals), idx + 2)):
            candidates.append(evals[i])
        if len(candidates) >= 3:
            ts = np.array([c[0] for c in candidates])
            os_ = np.array([c[1] for c in candidates])
            try:
                coef = np.polyfit(ts, os_, 2)
                t_opt = -coef[1] / (2 * coef[0])
                if 0 < t_opt < 1:
                    a_try = (1.0 - t_opt) * a + t_opt * g_dir
                    a_try = np.maximum(a_try, 0)
                    a_try /= a_try.sum()
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_t = t_opt
                        best_obj = obj_try
            except Exception:
                pass

    a_new = (1.0 - best_t) * a + best_t * g_dir
    a_new = np.maximum(a_new, 0)
    a_new /= a_new.sum()
    return a_new, best_obj, best_t


def freq_perturb(a, rng, strength=0.1):
    """Apply a frequency-domain perturbation for diversity."""
    n = len(a)
    # Go to frequency domain, add random phase, come back
    fa = rfft(a)
    # Random phase shift (preserve DC)
    phases = rng.uniform(-strength, strength, len(fa))
    phases[0] = 0  # preserve DC (sum)
    fa_new = fa * np.exp(1j * phases)
    a_new = irfft(fa_new, 2 * n - 1)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new


def lp_main_loop(a, time_budget, verbose=True):
    """
    Main LP alternating loop with adaptive line search and perturbations.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a /= a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    rng = np.random.default_rng(42 + n)
    stall = 0
    patience = 30
    perturbation_level = 0.0
    momentum_dir = np.zeros(n)
    it = 0

    # Track line search history for adaptive step
    recent_ts = []

    while time.monotonic() - start < time_budget:
        # Solve LP
        g_lp, M_lp = solve_lp(a)

        if g_lp is None:
            # Fallback to smooth LP
            g_lp = solve_lp_smooth(a, tau=50)

        # Compute similarity to current
        diff = np.linalg.norm(a - g_lp)
        if diff < 1e-6:
            # LP direction is essentially the current point - stalled
            stall += 2
        else:
            # Check if LP gives improvement
            obj_lp = compute_obj(g_lp)
            if obj_lp < best_obj * (1 - 0.001):
                stall = 0
            else:
                stall += 1

        # Line search
        a_new, obj_new, t_used = line_search_parabolic(a, g_lp, best_obj)

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            recent_ts.append(t_used)
            if len(recent_ts) > 10:
                recent_ts.pop(0)
            it += 1
        else:
            stall += 1
            if stall >= patience:
                # Apply perturbation
                perturbation_level = min(0.5, perturbation_level * 1.5 + 0.02)

                # Choose perturbation type
                r = rng.random()
                if r < 0.4:
                    # Frequency domain perturbation
                    a = freq_perturb(best_a, rng, strength=perturbation_level)
                elif r < 0.7:
                    # Multiplicative noise
                    noise = rng.normal(0, perturbation_level, n)
                    a = best_a * np.exp(noise)
                    a = np.maximum(a, 0)
                    a /= a.sum()
                else:
                    # Blend with random Dirichlet
                    random_v = rng.dirichlet(np.ones(n) * 0.1)
                    blend = perturbation_level * 2
                    a = (1 - blend) * best_a + blend * random_v
                    a = np.maximum(a, 0)
                    a /= a.sum()

                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                patience = max(15, patience - 2)
                recent_ts = []
                it += 1

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.10f}, stall={stall}, "
                  f"pert={perturbation_level:.3f}, t={elapsed:.1f}s")

    return best_a, best_obj


def smooth_mirror_descent(a, time_budget, verbose=False):
    """
    Mirror descent with log-sum-exp smoothing of the max-convolution objective.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 20.0
    lr = 2.0 / np.sqrt(n)
    momentum = 0.0
    v = np.zeros(n)
    stall = 0
    patience = 100
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        m = conv.max()
        shifted = beta * (conv - m)
        weights = np.exp(shifted)
        weights /= weights.sum()

        # Gradient: g_i = 2 * sum_{m: i+m valid} a_m * w_{i+m}
        g = np.zeros(n)
        for i in range(n):
            m_lo = max(0, 0 - i)
            m_hi = min(n - 1, 2 * n - 2 - i)
            if m_lo <= m_hi:
                g[i] = 2.0 * np.dot(a[m_lo:m_hi + 1], weights[i + m_lo:i + m_hi + 1])

        v = momentum * v + g
        log_a = np.log(a) - lr * v
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            momentum = min(0.8, momentum + 0.001)
            beta *= 1.01
            lr *= 0.999
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(it * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                momentum = 0.0
                beta = max(10, beta * 0.5)
                lr = max(1e-4, lr * 0.5)
                stall = 0
                patience = max(50, patience - 10)
            else:
                lr *= 0.98

        it += 1
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it{it}: obj={best_obj:.10f}, beta={beta:.1f}, t={elapsed:.1f}s")

    return best_a, best_obj


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    return a_new


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Triangle
        x = np.linspace(-1, 1, n)
        a = np.maximum(1.0 - np.abs(x), 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Frequency content
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = a / a.sum()
    elif mode == 7:
        # Chirp
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(2 * np.pi * (2 + 6 * x) * x)) + 0.01
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def coordinate_refine(a, time_budget=30.0):
    """
    Coordinate descent: move mass from peak-contributing positions to non-peak positions.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for _ in range(50):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to peak at k_star
        contribs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(reverse=True)

        non_contrib = set(range(n))
        for _, i, j in contribs[:20]:
            non_contrib.discard(i)
            if i != j:
                non_contrib.discard(j)
        non_contrib_list = sorted(non_contrib)

        improved = False
        for contrib, i, j in contribs[:15]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break
            src = i if a[i] >= a[j if i != j else i] else j
            for target in non_contrib_list[:5]:
                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * a[src]
                    if delta >= a[src] or delta < 1e-15:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            break

    return best_a, best_obj

        best_score = compute_obj(best_seq)

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val != np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
