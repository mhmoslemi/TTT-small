def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):

    n = len(a)

    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]

    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Optimized step function search using FFT-accelerated Nesterov gradient on
log-sum-exp smoothed objective, active-set LP alternating with warm starts,
multi-dimension exploration, and structured initializations.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_grad_fft(a, weights, N):
    """Compute gradient of smooth max(conv) using FFT cross-correlation.
    g_i = 2 * sum_m a_m * w_{i+m}
    """
    n = len(a)
    fa = fft(a, N)
    fw = fft(weights, N)
    cross = ifft(np.conj(fa) * fw).real[:n]
    return 2.0 * cross


def get_softmax_weights(conv, beta):
    """Compute softmax weights over convolution values."""
    m = conv.max()
    shifted = np.clip(beta * (conv - m), -60, 60)
    weights = np.exp(shifted)
    total = weights.sum()
    if total < 1e-300:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= total
    return weights


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_optimize(a, time_budget=300.0, beta_start=50.0, beta_end=8000.0, verbose=False):
    """
    Nesterov-accelerated projected gradient on log-sum-exp smoothed max(conv).
    Minimizes smooth_max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    Uses FFT-accelerated gradient computation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov state
    y = a.copy()
    x_prev = a.copy()
    t_k = 1.0
    lr = 0.5
    stall = 0
    patience = 200
    it = 0
    best_lr = lr

    # Precompute N for FFT cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1

    while time.monotonic() - start < time_budget:
        # Adaptive beta: geometric increase
        frac = min(1.0, float(it) / max(1.0, time_budget * 0.8))
        beta = beta_start * (beta_end / beta_start) ** frac

        # Compute gradient at Nesterov point y
        conv_y = compute_conv(y)
        weights = get_softmax_weights(conv_y, beta)
        grad = smooth_grad_fft(y, weights, N)

        # Estimate Lipschitz constant
        grad_norm = np.linalg.norm(grad)
        L_est = beta * 4.0 * np.max(conv_y) + 1e-8
        lr_opt = 1.0 / L_est

        # Adaptive learning rate
        lr = 0.7 * lr + 0.3 * lr_opt
        lr = max(1e-7, min(lr, 2.0 * lr_opt))

        # Nesterov step
        x_new = y - lr * grad
        x_new = project_simplex(x_new)

        # Nesterov extrapolation
        t_k1 = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = x_new + (t_k - 1.0) / t_k1 * (x_new - x_prev)
        y_new = np.maximum(y_new, 0)
        s_y = float(np.sum(y_new))
        if s_y > 1e-10:
            y_new = y_new / s_y

        # Evaluate
        obj_new = compute_obj(x_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = x_new.copy()
            x_prev = a.copy()
            a = x_new.copy()
            y = y_new.copy()
            t_k = t_k1
            stall = 0
            best_lr = lr
        else:

            for shrink in [0.5, 0.25, 0.125]:
                x_try = a + shrink * (x_new - a)

                obj_try = compute_obj(x_try)

                    x_prev = a.copy()
                    a = x_try.copy()
                    y = x_try.copy()
                    t_k = t_k1
                    stall = 0
                    lr *= shrink
                    found = True
                    break
            if not found:
                stall += 1
                if stall > patience:
                    rng = np.random.default_rng(it * 2654435761 + n)
                    noise = rng.normal(0, 0.03, n)
                    a = best_a * (1.0 + noise)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    y = a.copy()
                    x_prev = a.copy()
                    t_k = 1.0
                    lr = best_lr * 0.5

        it += 1
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"      Nest it{it}: obj={best_obj:.8f}, beta={beta:.0f}, lr={lr:.6f}, stall={stall}, t={elapsed:.1f}s")

def build_conv_matrix(a_norm):
    """Build the convolution matrix A such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_active(a, top_k=None, verbose=False):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0.
    Uses active-set: only keep top_k convolution constraints for speed.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A_full = build_conv_matrix(a_norm)

    # Determine active constraints: top-k values of conv(a, a)
    conv_aa = compute_conv(a_norm)
    if top_k is not None and top_k < len(conv_aa):
        # Keep top-k constraints plus a few random ones for diversity
        top_idx = np.argsort(conv_aa)[::-1][:top_k]
        n_random = max(5, (2 * n - 1 - top_k) // 20)
        remaining = np.setdiff1d(np.arange(len(conv_aa)), top_idx)
        rng = np.random.default_rng(42)
        rand_idx = rng.choice(remaining, size=min(n_random, len(remaining)), replace=False)
        idx = np.concatenate([top_idx, rand_idx])
    else:
        idx = np.arange(2 * n - 1)

    A = A_full[idx, :]
    m = len(idx)

    # Try cvxpy
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(max(M.value, 0))
                    return g_val, M_val
        except Exception:
            continue

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((m, 1))])
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                g_val = g_val / s
                return g_val, float(res.x[n])
    except Exception:
        pass

    return None, None

        if remaining_time < 60:

                if 0 <= j < n:
                    direction[i] = -a[j]
            direction -= np.mean(direction)

            g_lp = g_lp / g_lp.sum()

def subgradient_polish(a, time_budget=60.0, topK=5, verbose=False):
    """
    Subgradient method with momentum for final polishing.
    Uses top-K peak subgradients.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    momentum_v = np.zeros(n)
    momentum = 0.0
    stall = 0
    patience = 100
    it = 0

    N = 1
    while N < 3 * n:
        N <<= 1

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Get top-K peaks
        top_idx = np.argsort(conv)[::-1][:topK]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]

        if max_val < 1e-15:
            break

        # Weighted combination of top peaks
        beta_w = 5.0  # Sharpness of peak weighting
        w = np.exp((peak_vals - max_val) * beta_w / (max_val + 1e-15))
        w /= w.sum()

        # Compute gradient as weighted sum of subgradients
        grad = np.zeros(n)
        for wk, k in zip(w, top_idx):
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += wk * a[j]
        grad *= 2.0

        # Momentum update
        momentum_v = momentum * momentum_v + grad
        a_new = a - lr * momentum_v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            momentum = min(0.9, momentum + 0.001)
            lr *= 0.999
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(it * 31 + n)
                a = best_a * (1.0 + 0.02 * rng.normal(0, 1, n))
                a = np.maximum(a, 0)
                if a.sum() > 1e-10:
                    a = a / a.sum()
                else:
                    a = best_a.copy()
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                momentum_v = np.zeros(n)
                momentum = 0.0
                lr = 1.0 / np.sqrt(n)
                stall = 0
                patience = max(50, patience // 2)
            else:
                lr *= 0.98

        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"      SG it{it}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations with more structure."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        a = np.exp(-((x - 0.25) ** 2) / (2 * 0.1 ** 2)) + np.exp(-((x - 0.75) ** 2) / (2 * 0.1 ** 2))
        a = a / a.sum()
    elif mode == 3:
        # U-shaped (edges heavy)
        a = x**2 + (1 - x)**2 + 0.1
        a = a / a.sum()
    elif mode == 4:
        # Arcsine-like
        a = 1.0 / (np.sqrt(np.abs(x - 0.5) + 0.01) + 0.1)
        a = a / a.sum()
    elif mode == 5:
        # Multi-frequency
        freqs = rng.integers(2, 8, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Chirp-like (quadratic phase, made real non-negative)
        a = np.abs(np.cos(np.pi * 3 * x * x)) + 0.1
        a = a / a.sum()
    elif mode == 7:
        # Three bumps
        a = (np.exp(-((x - 0.2) ** 2) / (2 * 0.08 ** 2)) +
             np.exp(-((x - 0.5) ** 2) / (2 * 0.12 ** 2)) +
             np.exp(-((x - 0.8) ** 2) / (2 * 0.08 ** 2)))
        a = a / a.sum()
    elif mode == 8:
        # Inverse U (dip in middle)
        a = 1.0 - np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2)) + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def coordinate_descent_polish(a, time_budget=30.0):
    """
    Coordinate descent: find the peak lag and redistribute mass to reduce it.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for rnd in range(10):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Find contributors to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        # Non-contributor targets
        contrib_set = set()
        for _, i, j in contributors[:20]:
            contrib_set.add(i)
            if i != j:
                contrib_set.add(j)
        targets = [i for i in range(n) if i not in contrib_set]

        if not targets:
            break

        improved = False
        for contrib, i, j in contributors[:15]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break

            src = i if a[i] >= (a[j] if i != j else 0) else j
            for target in targets[:5]:
                for frac in [0.02, 0.05, 0.08, 0.12]:
                    delta = frac * a[src]
                    if delta < 1e-12 or delta >= a[src] * 0.9:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break

        if not improved:
            # Try small random perturbations
            rng = np.random.default_rng(rnd * 123 + n)
            for _ in range(20):
                idx = rng.choice(n, size=2, replace=False)
                delta = rng.uniform(0.001, 0.02) * a[idx[0]]
                if delta < a[idx[0]] * 0.5:
                    a[idx[0]] -= delta
                    a[idx[1]] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                    else:
                        a[idx[0]] += delta
                        a[idx[1]] -= delta

        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: FFT-accelerated Nesterov + active-set LP alternating +
    multi-dimension exploration + subgradient polish + coordinate descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial best: n={n_current}, obj={best_score:.8f}")

    # --- Phase 1: Nesterov smooth optimization ---
    print(f"\n=== Phase 1: Nesterov Smooth Optimization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 350.0)
    if p1_budget > 60:
        a_nest = nesterov_optimize(best_seq, time_budget=p1_budget,
                                   beta_start=30.0, beta_end=10000.0, verbose=True)
        obj_nest = compute_obj(a_nest)
        if obj_nest < best_score:
            best_score = obj_nest
            best_seq = a_nest.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_nest:.8f} (no improvement)")

    # --- Phase 2: LP Alternating (main workhorse) ---
    print(f"\n=== Phase 2: LP Alternating ===")
    # Multiple rounds
    for rnd in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        alloc = min(remaining * 0.3, 300.0)
        if alloc < 30:
            break

        if rnd > 0:
            rng_r = np.random.default_rng(rnd * 99991 + n_current)
            a_start = best_seq * (1.0 + 0.03 * rng_r.normal(0, 1, n_current))
            a_start = np.maximum(a_start, 0)
            if a_start.sum() > 1e-10:
                a_start = a_start / a_start.sum()
            else:
                a_start = best_seq.copy()
        else:
            a_start = best_seq.copy()

        a_lp = lp_alternating(a_start, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # --- Phase 3: Multi-dimension exploration ---
    print(f"\n=== Phase 3: Multi-Dimension Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best, n_best - 100, n_best + 100, n_best + 200,
        600, 700, 900, 1000, 1100, 1200, 1400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1600]

    explored = set()
    for n_try in n_options[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        if n_try == n_best and n_try not in explored:
            explored.add(n_try)
            continue

        a_try = resample_sequence(best_seq, n_try)
        explored.add(n_try)

        # Quick Nesterov
        nest_budget = min(remaining * 0.12, 80.0)
        if nest_budget > 20:
            a_try = nesterov_optimize(a_try, time_budget=nest_budget,
                                      beta_start=20, beta_end=3000)

        # Quick LP
        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.08, 60.0)
        if lp_alloc > 15:
            a_try = lp_alternating(a_try, time_budget=lp_alloc)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # --- Phase 4: Diverse restarts ---
    print(f"\n=== Phase 4: Diverse Restarts ===")
    n_use = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = r % 9
        a_r = diverse_init(n_use, rng, mode=mode)
        # Blend with best
        blend = 0.3 + 0.08 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        if a_r.sum() > 1e-10:
            a_r = a_r / a_r.sum()

        # Quick optimization
        nest_b = min(remaining * 0.12, 60.0)
        if nest_b > 20:
            a_r = nesterov_optimize(a_r, time_budget=nest_b, beta_start=20, beta_end=2000)

        remaining = deadline - time.monotonic()
        lp_b = min(remaining * 0.08, 40.0)
        if lp_b > 15:
            a_r = lp_alternating(a_r, time_budget=lp_b)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # --- Phase 5: Final intensive polish ---
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Nesterov polish
    if remaining > 40:
        nest_b = min(remaining * 0.2, 100.0)
        a_nest2 = nesterov_optimize(best_seq, time_budget=nest_b,
                                    beta_start=200, beta_end=20000, verbose=False)
        obj_nest2 = compute_obj(a_nest2)
        if obj_nest2 < best_score:
            best_score = obj_nest2
            best_seq = a_nest2.copy()
            print(f"  Nesterov polish: obj={best_score:.8f} **")

    # LP polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        lp_b = min(remaining * 0.25, 100.0)
        a_lp2 = lp_alternating(best_seq, time_budget=lp_b, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP polish: obj={best_score:.8f} **")

    # Subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        sg_b = min(remaining * 0.15, 50.0)
        a_sg = subgradient_polish(best_seq, time_budget=sg_b, topK=5)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SG polish: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        cd_b = min(remaining * 0.2, 40.0)
        a_cd = coordinate_descent_polish(best_seq, time_budget=cd_b)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD polish: obj={best_score:.8f} **")

    # One final LP round
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = lp_alternating(best_seq, time_budget=min(remaining * 0.4, 40.0))
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
