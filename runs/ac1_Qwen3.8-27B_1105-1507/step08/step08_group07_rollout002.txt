    n = len(a)
    M = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        M[j:j + r, j] = a[:r]
    return M


def solve_lp_full(a):
    """Solve LP: min M s.t. M_a @ z <= M, sum(z)=1, z>=0.
    Returns (z, M_val, dual) or (None, None, None)."""
    n = len(a)
    M_a = build_conv_matrix(a)

    try:
        import cvxpy as cp
        z = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        c1 = M_a @ z <= M_var
        prob = cp.Problem(cp.Minimize(M_var), [c1, cp.sum(z) == 1])
        for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    z_val = np.maximum(z.value, 0)
                    zs = float(np.sum(z_val))
                    if zs > 1e-10:
                        z_val = z_val / zs
                        dual_y = None
                        try:
                            dual_y = np.maximum(np.array(c1.dual_value), 0)
                        except Exception:
                            pass
                        return z_val, float(M_var.value), dual_y
            except Exception:
                continue
    except Exception:
        pass

    # Fallback to scipy linprog
    try:
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = M_a
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n]), None
    except Exception:
        pass

    return None, None, None


def soft_gradient(x, T):
    """Compute gradient of soft-max objective in log-space.
    x: log-amplitude vector
    T: temperature for soft-max approximation
    Returns (objective_value, gradient)
    """
    n = len(x)
    # Stable softmax
    x_shift = x - np.max(x)
    exp_x = np.exp(x_shift)
    Z = np.sum(exp_x)
    a = exp_x / Z

    # Compute convolution
    conv = compute_conv(a)

    # Soft-max of convolution
    conv_shift = conv / T - np.max(conv / T)
    exp_conv = np.exp(conv_shift)
    S = np.sum(exp_conv)
    w = exp_conv / S

    soft_max_val = T * (np.log(S) + np.max(conv / T))
    obj_val = 2.0 * n * soft_max_val

    # Gradient of soft_max w.r.t. a: grad_a_i = 2 * sum_k w_k * a_{k-i}
    # This is 2 * cross_correlate(w, a)
    # cross_correlate(w, a)[i] = sum_k w[k] * a[k-i]
    # = conv(w_reversed, a)[i + len(w) - 1 - n + 1]... 
    # Easier: use np.correlate or FFT
    # correlate(w, a, 'full')[i] = sum_k w[k] * a[k - (i - (len(a)-1))]
    # We want: for each i in [0,n-1], sum over k of w[k] * a[k-i] (a[j]=0 outside [0,n-1])
    # This equals correlate(w, a, 'full')[(len(a)-1) : (len(a)-1)+n]
    # Actually np.correlate(w, a, 'full')[i] = sum_j w[j] * conj(a[j - i + (len(a)-1)])
    # Let me just use FFT for clarity

    size = 1
    while size < 2 * max(n, len(conv)) - 1:
        size <<= 1
    # cross-correlation: sum_k w[k] * a[k-i]
    # = ifft(conj(fft(w)) * fft(a)) ... no
    # = ifft(fft(w) * conj(fft(a))) ... let me think
    # corr[i] = sum_k w[k] * a[k-i]
    # FFT approach: ifft(fft(w, N) * conj(fft(a, N))) gives circular cross-correlation
    # For linear: zero-pad to size >= n + len(conv) - 1
    N = size
    W = rfft(w, N)
    A = rfft(a, N)
    corr_full = irfft(W * np.conj(A), N)
    # corr_full[k] = sum_j w[j] * a[j-k] (circular, but with zero padding it's linear)
    # We want sum_k w[k] * a[k-i] for i=0..n-1
    # = corr_full shifted... 
    # Actually: sum_k w[k] * a[k-i] = (w * a_flipped)[i + n - 1] where a_flipped = a[::-1]
    # Or: it's the cross-correlation of w and a at lag i
    # Using FFT: ifft(conj(FFT(w)) * FFT(a)) gives sum_k w[k] * a[k+i]
    # So sum_k w[k] * a[k-i] = ifft(conj(FFT(w)) * FFT(a)) evaluated at ... 
    # Let me just use a simpler approach:
    # grad_a[i] = 2 * sum_k w[k] * a[k-i]
    # = 2 * (w * a)[-n:] reversed... 
    
    # Simplest: use np.correlate
    # np.correlate(w, a, 'full')[m] = sum_n w[n] * conj(a[n - m + (len(a)-1)])
    # We want for each i: sum_k w[k] * a[k-i]
    # Let m = len(a) - 1 - i, then n - m + len(a)-1 = n - len(a)+1+i + len(a)-1 = n+i
    # Hmm this is getting confusing. Let me just do it with FFT directly.
    
    # Method: pad w and a to length N, compute ifft(fft(w) * conj(fft(a)))
    # This gives circular cross-correlation: result[k] = sum_j w[(k-j) % N] * a[j]
    # = sum_j w[k-j] * a[j] (with zero padding making it linear)
    # = sum_j w[k-j] * a[j]
    # We want sum_k w[k] * a[k-i] = sum_j w[j+i] * a[j] (substituting k=j+i)
    # = cross_correlation at lag i = result[(n-1-i) % N]... 
    
    # OK let me just be very explicit:
    # I want g[i] = sum_k w[k] * a[k-i] for i = 0, ..., n-1
    # where a[j] = 0 for j < 0 or j >= n
    # This is exactly (w * a_flipped)[n-1-i + len(w)-1] ... no
    # 
    # Let's use: g[i] = sum_k w[k] * a[k-i]
    # = sum_k w[k] * a_flipped[n-1-(k-i)] where a_flipped = a[::-1]
    # = sum_k w[k] * a_flipped[n-1-k+i]
    # = (w * a_flipped)[n-1+i] (convolution)
    # So g = conv(w, a_flipped)[n-1 : n-1+n]
    
    a_flipped = a[::-1]
    conv_wa = compute_conv(np.concatenate([w, np.zeros(n - 1)]) if len(w) > n else w)
    # Actually w has length 2n-1, a has length n
    # conv(w, a_flipped) has length (2n-1) + n - 1 = 3n-2
    # g[i] = conv(w, a_flipped)[n-1+i]
    
    # Let me just compute conv(w, a_flipped) directly
    size2 = 1
    while size2 < len(w) + n - 1:
        size2 <<= 1
    Wf = rfft(w, size2)
    Af = rfft(a_flipped, size2)
    conv_wf = irfft(Wf * Af, size2)
    
    # g[i] = conv_wf[n-1+i] for i=0..n-1
    grad_a = 2.0 * conv_wf[n - 1:n - 1 + n]
    
    # Transform from da gradient to dx gradient
    # a = softmax(x), da_i/dx_j = a_i*(delta_ij - a_j)
    # df/dx_j = grad_a[j]*a_j - a_j*sum_i(grad_a[i]*a_i)
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    
    return obj_val, grad_x


def gradient_refine(a_init, time_budget=200.0, T_start=0.01, T_end=0.001):
    """Refine using soft-max gradient with L-BFGS in log-space."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).astype(np.float64).copy()
    a = a / np.sum(a)
    
    x0 = np.log(a)
    best_x = x0.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Anneal temperature
    n_anneal = 3
    T_values = np.geomspace(T_start, T_end, n_anneal)
    
    for T in T_values:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        t_alloc = min(remaining * 0.9, 60.0)
        
        def func_and_grad(x):
            elapsed = time.monotonic() - start
            if elapsed > time_budget:
                return 1e10, np.zeros_like(x)
            obj, grad = soft_gradient(x, T)
            return obj, grad
        
        try:
            res = minimize(func_and_grad, x0, method='L-BFGS-B',
                          jac=True, options={'maxiter': 50, 'maxfun': 200,
                                             'gtol': 1e-8, 'max_ls': 20,
                                             'max_time': t_alloc})
            if res.fun < best_obj:
                # Convert back to a
                x_res = res.x
                x_shift = x_res - np.max(x_res)
                a_res = np.exp(x_shift) / np.sum(np.exp(x_shift))
                a_res = np.maximum(a_res, 0)
                obj_res = compute_obj(a_res)
                if obj_res < best_obj:
                    best_obj = obj_res
                    best_x = x_res.copy()
                    a = a_res
        except Exception as e:
            print(f"  Gradient step failed: {e}")
        
        # Update x0 for next temperature
        x0 = best_x.copy()
        elapsed = time.monotonic() - start
        print(f"    Gradient T={T:.4f}: obj={best_obj:.8f}, t={elapsed:.0f}s")
    
    # Convert best back
    x_shift = best_x - np.max(best_x)
    a_out = np.exp(x_shift) / np.sum(np.exp(x_shift))
    a_out = np.maximum(a_out, 0)
    obj_out = compute_obj(a_out)
    if obj_out < best_obj:
        best_obj = obj_out
        a_out = a_out
    else:
        a_out = a_init / np.sum(a_init)
    
    return a_out, best_obj


def lp_alternate(a_init, time_budget=200.0, verbose=False):
    """LP alternating with momentum, cycle detection, and adaptive mixing."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    prev_a = None
    prev_prev_a = None
    momentum = a.copy()
    stall = 0

    while time.monotonic() - start < time_budget:
        result = solve_lp_full(a)
        if result[0] is None:
            break
        g_lp, M_lp, dual_y = result

        # 2-cycle detection
        if prev_prev_a is not None:
            diff = np.max(np.abs(a - prev_prev_a))
            if diff < 5e-8:
                a_break = 0.5 * (a + prev_a)
                a_break = np.maximum(a_break, 0)
                sb = float(np.sum(a_break))
                if sb > 1e-10:
                    a_break = a_break / sb
                    obj_break = compute_obj(a_break)
                    if obj_break < best_obj - 1e-12:
                        best_obj = obj_break
                        best_a = a_break.copy()
                    a = a_break
                    prev_a = None
                    prev_prev_a = None
                    momentum = a.copy()
                    continue

        # Update momentum
        momentum = 0.5 * momentum + 0.5 * g_lp
        momentum = np.maximum(momentum, 0)
        ms = float(np.sum(momentum))
        if ms > 1e-10:
            momentum = momentum / ms

        # Try multiple mixing ratios
        targets = [g_lp, momentum]
        best_cand_obj = np.inf
        best_cand = a.copy()

        for target in targets:
            for mix in [0.3, 0.5, 0.7, 1.0, 0.8, 0.9]:
                a_try = (1.0 - mix) * a + mix * target
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_cand_obj:
                    best_cand_obj = obj_try
                    best_cand = a_try.copy()

        if best_cand_obj < best_obj - 1e-13:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            prev_prev_a = prev_a
            prev_a = a.copy()
            a = best_a
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                if int(elapsed) % 20 < 2:
                    print(f"    LP: obj={best_obj:.8f}, t={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 20:
                # Perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                if dual_y is not None and np.max(dual_y) > 0:
                    active_idx = np.where(dual_y > 0.05 * np.max(dual_y))[0]
                    if len(active_idx) > 0:
                        M_a = build_conv_matrix(a)
                        col_imp = np.sum(M_a[active_idx, :], axis=0)
                        pert = -np.sign(col_imp - np.mean(col_imp)) * 0.03
                    else:
                        pert = rng.normal(0, 0.03, n)
                else:
                    pert = rng.normal(0, 0.04, n)
                
                a = best_a * np.exp(pert)
                a = np.maximum(a, 1e-10)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                prev_prev_a = None
                momentum = a.copy()
                stall = 0
                if verbose:
                    print(f"    [Perturbed] obj={best_obj:.8f}")
            else:
                # Small step toward LP solution
                a_try = a + 0.05 * (g_lp - a)
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st > 1e-10:
                    a_try = a_try / st
                    if compute_obj(a_try) < best_obj:
                        a = a_try
                        best_a = a.copy()
                        best_obj = compute_obj(a)

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def pairwise_exchange(a, time_budget=30.0):
    """Pairwise exchange polish for final refinement."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99999)

    no_improve = 0
    while time.monotonic() - start < time_budget:
        improved = False
        for _ in range(30):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for frac in [0.02, 0.05, 0.1, 0.15]:
                delta = frac * min(a[i], a[j])
                if delta <= 0:
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0:
                    continue
                a_try = a_try / np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

    return best_a, best_obj


def hybrid_alternate(a_init, time_budget=200.0):
    """Alternate between LP steps and gradient steps."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    use_lp = True

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        if use_lp:
            step_budget = min(remaining * 0.4, 40.0)
            a_new, obj_new = lp_alternate(a, time_budget=step_budget)
            use_lp = False
        else:
            step_budget = min(remaining * 0.4, 40.0)
            a_new, obj_new = gradient_refine(a, time_budget=step_budget)
            use_lp = True

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Small perturbation to escape
            rng = np.random.default_rng(int(time.monotonic() * 1e5) % (2**31))
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 1e-10)
            a = a / np.sum(a)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: hybrid LP + gradient optimization with n-sweep and restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    best_seq = None
    best_score = np.inf
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        # Default initialization
        best_seq = np.ones(800)
        best_seq = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, 800))
        best_seq = np.maximum(best_seq, 0.01)
        best_seq = best_seq / np.sum(best_seq)
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)
    a = best_seq.copy()

    # === Phase 1: LP Alternating (250s) ===
    print(f"\n=== Phase 1: LP Alternating (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(250.0, remaining * 0.25)
        a_new, obj_new = lp_alternate(a, time_budget=alloc, verbose=True)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_new:.8f}")

    # === Phase 2: Gradient Refinement (300s) ===
    print(f"\n=== Phase 2: Gradient Refinement (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(300.0, remaining * 0.3)
        a_new, obj_new = gradient_refine(a, time_budget=alloc)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_new:.8f}")

    # === Phase 3: Hybrid LP+Gradient (250s) ===
    print(f"\n=== Phase 3: Hybrid (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(250.0, remaining * 0.25)
        a_new, obj_new = hybrid_alternate(a, time_budget=alloc)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj_new:.8f}")

    # === Phase 4: n-sweep ===
    print(f"\n=== Phase 4: n-sweep ===")
    remaining = deadline - time.monotonic()
    n_base = n_parent
    n_candidates = sorted(set([
        n_base - 200, n_base - 100, n_base - 50, n_base - 20,
        n_base + 20, n_base + 50, n_base + 100, n_base + 200,
        n_base + 300, n_base + 400,
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 2500]
    n_candidates.sort(key=lambda x: abs(x - n_base))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(45.0, remaining * 0.18)
        
        # Quick LP + gradient
        a_opt, obj_lp = lp_alternate(a_try, time_budget=alloc * 0.5)
        obj_total = obj_lp
        if remaining > 30:
            a_opt2, obj_grad = gradient_refine(a_opt, time_budget=min(alloc * 0.4, remaining * 0.15))
            if obj_grad < obj_total:
                obj_total = obj_grad
                a_opt = a_opt2

        if obj_total < best_score:
            best_score = obj_total
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_total:.8f}")

    # === Phase 5: LP on absolute best (200s) ===
    print(f"\n=== Phase 5: Final LP (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(200.0, remaining * 0.3)
        a_new, obj_new = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Phase 5: obj={best_score:.8f} **")

    # === Phase 6: Gradient on absolute best (100s) ===
    print(f"\n=== Phase 6: Final Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(100.0, remaining * 0.25)
        a_new, obj_new = gradient_refine(best_seq, time_budget=alloc)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Phase 6: obj={best_score:.8f} **")

    # === Phase 7: Random Restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 7: Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(5, int(remaining // 50))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            if r % 3 == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.05 + 0.01 * r, n_best))
            elif r % 3 == 1:
                a_r = best_seq.copy()
                n_blocks = max(5, n_best // 80)
                block_size = n_best // n_blocks
                blocks = rng.choice(n_blocks, min(4, n_blocks), replace=False)
                for b in blocks:
                    si = int(b) * block_size
                    ei = si + block_size
                    a_r[si:ei] *= rng.uniform(0.4, 1.6)
            else:
                a_r = best_seq.copy()
                n_sparse = max(10, n_best // 20)
                idx = rng.choice(n_best, n_sparse, replace=False)
                a_r[idx] *= rng.uniform(0.3, 2.5, n_sparse)

            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / np.sum(a_r)

            alloc = min(remaining * 0.35, 40.0)
            a_opt, obj_r = lp_alternate(a_r, time_budget=alloc)
            
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 8: Pairwise Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 8: Pairwise Polish ===")
        alloc = min(60.0, remaining * 0.3)
        a_new, obj_new = pairwise_exchange(best_seq, time_budget=alloc)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Polish: obj={best_score:.8f} **")

    # === Final LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 20.0)
        a_final, obj_final = lp_alternate(best_seq, time_budget=alloc)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
