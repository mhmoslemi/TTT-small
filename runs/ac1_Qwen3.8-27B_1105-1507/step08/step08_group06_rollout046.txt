            stall = 0
            if verbose and (ip % 20 == 0):
                elapsed = time.monotonic() - start
                print(f"    LPv2: obj={best_obj:.8f}, t={best_t_val:.3f}, iter={ip}, time={elapsed:.0f}s")
        else:
            stall += 1
            momentum *= 0.5
            if stall % 15 == 0:
                # Structured perturbation: swap high/low pairs
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                scale = 0.02 + 0.005 * min(stall // 15, 8)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
                last_t = 0.5

        ip += 1

    return best_a


def subgradient_nesterov(a, time_budget=100.0, verbose=False):
    """
    Nesterov-accelerated subgradient method for the max-conv objective.
    Uses active peak tracking and adaptive step sizes.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()  # current point
    x = a.copy()  # Nesterov lookahead
    beta = 0.7
    lr = 0.03
    g_cache = np.zeros(n)

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        # Compute convolution at lookahead
        conv_x = compute_conv(x)
        max_val = float(np.max(conv_x))
        threshold = max_val * 0.85
        active = np.where(conv_x >= threshold)[0]
        if len(active) == 0:
            active = [int(np.argmax(conv_x))]
        if len(active) > 40:
            active = np.argsort(conv_x)[::-1][:40]

        # Compute subgradient at y (not x, for proper subgradient method)
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_max = min(n - 1, k)
            i_min = max(0, k - n + 1)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += y[k - indices]
        grad /= len(active)

        # Project gradient to sum-0 hyperplane
        grad_centered = grad - np.mean(grad)

        # Momentum
        g_cache = beta * g_cache - lr * grad_centered

        # Update y
        y_new = y + g_cache
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
        else:
            y_new = y_new / s_new

        # Nesterov lookahead
        t_k = 2.0 / (np.sqrt(it + 2) + 2.0)
        x = y_new + (1 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        y = y_new
        obj = compute_obj(y)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = y.copy()

        # Adaptive LR
        if it % 500 == 499:
            lr *= 0.85
            if lr < 0.001:
                lr = 0.001

    return best_a


def spectral_init(n, rng):
    """
    Generate a sequence with approximately flat power spectrum.
    This gives a good starting point for reducing autocorrelation peaks.
    """
    # Random phase spectrum
    m = 2 * n  # oversample for circular -> linear
    freqs = rng.random(m // 2 + 1)
    phases = rng.uniform(0, 2 * np.pi, m // 2 + 1)
    # Target magnitude: slightly structured to avoid pure flat (which gives delta autocorr)
    # Use a "water-filled" magnitude
    target_mag = np.ones(m // 2 + 1)
    # Add some structure to encourage non-trivial sequences
    target_mag += 0.3 * np.sin(rng.uniform(1, 5, 1) * np.linspace(0, np.pi, m // 2 + 1))
    target_mag = np.maximum(target_mag, 0.1)

    spectrum = target_mag * np.exp(1j * phases)
    # Make Hermitian
    if m % 2 == 0:
        full_spec = np.zeros(m, dtype=complex)
        full_spec[:m//2+1] = spectrum
        full_spec[m//2+1:] = np.conj(spectrum[1:m//2][::-1])
    else:
        full_spec = np.zeros(m, dtype=complex)
        full_spec[:m//2+1] = spectrum
        full_spec[m//2+1:] = np.conj(spectrum[m//2:: -1])

    sig = np.real(ifft(full_spec))
    sig = np.maximum(sig, 0)
    sig = sig[:n]
    s = np.sum(sig)
    if s < 1e-10:
        return np.ones(n) / n
    return sig / s


def generate_init(kind, n, rng):
    """Generate diverse initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'spectral':
        return spectral_init(n, rng)
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.1
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'broad_gauss':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 3.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        a += 0.05
        return a / np.sum(a)
    elif kind == 'two_hump':
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - n*0.3) / (n*0.15)) ** 2) + np.exp(-0.5 * ((x - n*0.7) / (n*0.15)) ** 2)
        a += 0.05
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via spline interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, kind='quadratic')
    return np.maximum(a_new, 0)


def pair_swap_polish(a, time_budget=60.0):
    """
    Fast pairwise swap polish: move mass from positions contributing
    most to the peak to positions that reduce it.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    for iteration in range(1000):
        if time.monotonic() - start > time_budget:
            break

        k_star = int(np.argmax(conv))

        # Compute contributions to peak
        contributions = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                contributions[j] = a[j] * a[kj]

        # Find high contributors and low contributors
        sorted_desc = np.argsort(contributions)[::-1]
        sorted_asc = np.argsort(contributions)

        improved = False
        for frac in [0.01, 0.02, 0.05]:
            if improved:
                break
            for i in sorted_desc[:15]:
                if improved or time.monotonic() - start > time_budget:
                    break
                for j in sorted_asc[:15]:
                    if i == j:
                        continue
                    delta = frac * a[i]
                    if delta <= 0 or delta > a[i] * 0.5:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-15:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combining reduced-constraint LP, spectral initialization,
    Nesterov subgradient, and coordinate polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Diverse Initializations (100s) ===
    print(f"\n=== Phase 1: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    init_budget = min(100.0, remaining * 0.08)
    init_start = time.monotonic()

    init_kinds = ['spectral', 'cosine', 'sine_taper', 'gaussian', 'broad_gauss',
                  'two_hump', 'random']

    for kind in init_kinds:
        if time.monotonic() - init_start > init_budget:
            break
        a_init = generate_init(kind, n_best, rng)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init '{kind}': obj={obj_init:.8f} **")
        else:
            print(f"  Init '{kind}': obj={obj_init:.8f}")

    # Try a few more spectral inits with different random seeds
    for _ in range(5):
        if time.monotonic() - init_start > init_budget:
            break
        a_init = generate_init('spectral', n_best, rng)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Spectral init: obj={obj_init:.8f} **")

    a = best_seq.copy()
    n_cur = len(a)
    print(f"  Best after Phase 1: obj={best_score:.8f}, n={n_cur}")

    # === Phase 2: LP Alternating with Reduced Constraints (500s) ===
    print(f"\n=== Phase 2: LP Alternating (reduced constraints) ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(500.0, remaining * 0.45)

    # Multiple rounds with different top_k
    top_k_values = [40, 60, 30, 50]
    for round_idx in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        top_k = top_k_values[round_idx % len(top_k_values)]
        alloc = min(lp_budget / max(1, (4 - round_idx)), remaining * 0.7)
        print(f"  LP Round {round_idx} (top_k={top_k}):")
        a_new = lp_alternating_v2(a, time_budget=alloc, top_k=top_k, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  ** LP R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP R{round_idx}: obj={obj:.8f}")
            if round_idx > 0:
                # Perturb from best
                scale = 0.025
                a = best_seq * np.exp(rng.normal(0, scale, n_cur))
                a = np.maximum(a, 0)
                a = a / np.sum(a)

    # === Phase 3: Nesterov Subgradient (150s) ===
    print(f"\n=== Phase 3: Nesterov Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        sg_budget = min(150.0, remaining * 0.15)
        a_sg = subgradient_nesterov(a, time_budget=sg_budget)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a = a_sg.copy()
            print(f"  Nesterov SG: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov SG: obj={obj_sg:.8f}")

    # === Phase 4: Pair Swap Polish (80s) ===
    print(f"\n=== Phase 4: Pair Swap Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        ps_budget = min(80.0, remaining * 0.1)
        a_ps = pair_swap_polish(a, time_budget=ps_budget)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            a = a_ps.copy()
            print(f"  Pair Swap: obj={best_score:.8f} **")
        else:
            print(f"  Pair Swap: obj={obj_ps:.8f}")

    # === Phase 5: Length Sweep (200s) ===
    print(f"\n=== Phase 5: Length Sweep ===")
    remaining = deadline - time.monotonic()
    n_best_cur = len(best_seq)

    # Try nearby lengths - focus on a few promising ones
    n_options = sorted(set([
        max(200, n_best_cur - 100),
        max(200, n_best_cur - 50),
        n_best_cur,
        n_best_cur + 50,
        n_best_cur + 100,
        n_best_cur + 200,
    ]))
    n_options = [n for n in n_options if 200 <= n <= 3000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_best_cur:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick optimization
        alloc_lp = min(25.0, remaining * 0.4)
        alloc_sg = min(15.0, remaining * 0.15)

        a_opt = lp_alternating_v2(a_try, time_budget=alloc_lp, top_k=30)
        a_opt = subgradient_nesterov(a_opt, time_budget=alloc_sg)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final LP rounds from best (remaining time) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    a_final = best_seq.copy()

    polish_count = 0
    while remaining > 25 and polish_count < 10:
        alloc = min(remaining * 0.3, 40.0)
        top_k_final = 35 + (polish_count % 3) * 10
        a_new = lp_alternating_v2(a_final, time_budget=alloc, top_k=top_k_final)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_final = a_new.copy()
            print(f"  Final LP {polish_count}: obj={best_score:.8f} **")
        else:
            # Small perturbation
            scale = 0.01
            a_final = best_seq * np.exp(rng.normal(0, scale, n_final))
            a_final = np.maximum(a_final, 0)
            a_final = a_final / np.sum(a_final)
            print(f"  Final LP {polish_count}: obj={obj:.8f} (perturbing)")

        polish_count += 1
        remaining = deadline - time.monotonic()

    # === Phase 7: Very final pair swap ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd = pair_swap_polish(best_seq, time_budget=min(remaining * 0.5, 20.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final PS: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        # Fallback
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
