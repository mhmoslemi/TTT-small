"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a):
    """Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    from scipy.sparse import csr_matrix
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        length = end_k - start_k
        if length > 0:
            rows.extend(range(start_k, end_k))
            cols.extend([j] * length)
            vals.extend(a[:length])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def build_conv_matrix_dense(a):
    """Build dense matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp(a, active_indices=None, use_sparse=True):
    """
    Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    If active_indices given, only enforce those constraints.
    Returns (g, M, all_violated) where all_violated lists violated constraint indices.
    """
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)

    n_constraints = 2 * n - 1
    if active_indices is not None:
        n_used = len(active_indices)
        A_full = build_conv_matrix_dense(a)
        A = A_full[active_indices, :]
    else:
        n_used = n_constraints
        A = build_conv_matrix_dense(a)

    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((n_used, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_used)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status != 0:
        return None, None, None

    g_val = np.maximum(res.x[:n], 0)
    s = float(np.sum(g_val))
    if s < 1e-10:
        return None, None, None
    g_norm = g_val / s
    M_val = float(res.x[n])

    # Check all constraints
    conv_check = compute_conv_cross(a, g_norm)
    all_violated = np.where(conv_check > M_val + 1e-10)[0]

    return g_norm, float(np.max(conv_check)), all_violated

    s = np.sum(a)

                rng = np.random.default_rng(it * 7919 + n)
                a_pert = best_a * np.exp(rng.normal(0, 0.003, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a = a_pert / s_p
                active_set = None
                stall = 0
                if stall > 100:
                    break

    return best_a


def smooth_max_gd(a, time_budget=60.0, T_init=0.0005, T_min=0.00002):
    """Smooth max gradient descent with Nesterov momentum and adaptive temperature."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.002
    momentum = 0.9
    v = np.zeros(n)
    stall = 0

    max_iters = 10000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        if total < 1e-300:
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
        else:
            weights = exp_vals / total

        # Gradient of sum_k w_k * conv(k) w.r.t. a[j] = sum_k w_k * a[k-j] (valid range)
        # conv(k) = sum_j a[j]*a[k-j], d(conv(k))/d(a[j]) = a[k-j] + a[j-k] (accounting for both terms)
        # Actually d(conv(k))/d(a[j]) = a[k-j] if 0<=k-j<n, plus a[j-k] if 0<=j-k<n
        # For the smooth max: grad[j] = sum_k weights[k] * (a[k-j] + a[j-k]) (with bounds)
        # This is: grad = weights * a (convolution) + weights * a[::-1] (cross-correlation)
        # Actually simpler: grad[j] = sum_k weights[k] * d(conv[k])/d(a[j])
        # d(conv[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n, j!=k-j case)
        # For autocorrelation of real sequence: d(conv[k])/d(a[j]) = a[k-j] + a[j-k]
        # But in linear convolution conv[k] = sum_i a[i]*a[k-i], d/d(a[j]) = a[k-j] (from i=j term) + a[j-k]... 
        # No wait: conv[k] = sum_i a[i] * a[k-i]. d(conv[k])/d(a[j]) = a[k-j] (when i=j) only, since a[k-i] depends on a[j] only when k-i=j, i.e., i=k-j.
        # So d(conv[k])/d(a[j]) = a[k-j] if 0<=k-j<n, else 0.
        # Hmm, but that's only if we consider each a[j] independently. Let me be more careful.
        # conv[k] = sum_{i=0}^{n-1} a[i] * a[k-i] (where a[m]=0 for m<0 or m>=n)
        # d(conv[k])/d(a[j]) = a[k-j] (from the i=j term: d(a[j]*a[k-j])/d(a[j]) = a[k-j])
        #                + a[j-k] (from the i=k-j term if k-j != j: d(a[k-j]*a[j])/d(a[j]) = a[k-j])
        # If k-j == j (i.e., k=2j), then it's just 2*a[j]... 
        # Actually no. Let's be precise:
        # conv[k] = sum_i a[i]*a[k-i]
        # The term where a[j] appears: 
   when i=j: a[j]*a[k-j] -> derivative w.r.t. a[j] is a[k-j]
        #   when k-i=j, i.e., i=k-j: a[k-j]*a[j] -> derivative w.r.t. a[j] is a[k-j]
        #   if j == k-j (k=2j): the same term a[j]*a[j] -> derivative is 2*a[j]
        # So: d(conv[k])/d(a[j]) = a[k-j] * (1 if 0<=k-j<n) + a[j-k] * (1 if 0<=j-k<n)
        # But a[k-j] = a[j-k] only if the sequence is symmetric... no they're different indices.
        # Wait, a[k-j] and a[j-k]: these are different elements unless k-j = j-k.
        # Let me just compute it as: grad[j] = sum_k weights[k] * (a[k-j] if 0<=k-j<n else 0)
        # This is a cross-correlation: grad = correlate(weights, a, 'full')[n-1:n-1+n]
        # Hmm, let me just use the simple formula:
        # grad[j] = sum_k weights[k] * a[k-j] where 0<=k-j<n
        # This is what np.correlate(weights, a, 'valid') gives if len(weights) >= len(a)
        # Actually conv has length 2n-1, a has length n
        # grad[j] = sum_{k=j}^{j+n-1} weights[k] * a[k-j] = sum_{m=0}^{n-1} weights[j+m] * a[m]
        # This is a sliding dot product... 

        # Simpler approach: use FFT
        # grad[j] = sum_k weights[k] * a[k-j] (0<=k-j<n)
        # This is the cross-correlation of weights and a, evaluated at lags -n+1 to n-1... 
        # Actually: (weights * a)[j+m] for m=0..n-1, sum over m.
        # No: grad[j] = sum_m a[m] * weights[j+m] for m=0..n-1 (where j+m < 2n-1)
        # This is like a "valid" cross-correlation.
        
        # Let me just compute it directly using the convolution structure
        # grad = np.correlate(weights, a, mode='same') won't be quite right
        # Let me use: for each j, grad[j] = dot(a, weights[j:j+n]) (with zero padding)
        # This is O(n^2) which is too slow for n=800 per iteration.
        
        # Use FFT: create a circular convolution
        # Pad weights to size >= 2n-1 (it already is, size 2n-1)
        # grad[j] = sum_{m=0}^{n-1} a[m] * weights[j+m]
        # Let W be weights (length 2n-1), A be a (length n)
        # Define B[m] = A[m] for m=0..n-1, 0 otherwise (length 2n-1)
        # Then grad[j] = sum_m B[m] * W[j+m] = (B * W_rev)[j] where W_rev is reversed
        # Actually: sum_m B[m] W[j+m] = cross-correlation of B and W at lag j
        # = sum_m B[m] W[m+j]
        # In FFT: IFFT(FFT(B) * conj(FFT(W))) gives cross-correlation
        
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        FB = rfft(np.concatenate([a, np.zeros(size - n)]), size)
        FW = rfft(np.concatenate([weights, np.zeros(size - (2*n-1))]), size)
        grad = irfft(FB * np.conj(FW), size)[:n]
        
        # Center the gradient (remove mean to handle simplex constraint)
        grad = grad - np.mean(grad)

        # Nesterov momentum
        v = momentum * v - lr * grad
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.8
                if lr < 1e-7:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0

        if it > 50:
            T = max(T_min, T_init * (0.995 ** it))

    return best_a


def coordinate_descent_pairwise(a, time_budget=100.0, delta=5e-4):
    """
    Pair-wise coordinate descent: try adjusting pairs of coordinates.
    More effective than single-coordinate for this problem.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    pass_num = 0
    current_delta = delta
    no_improve_passes = 0

    while time.monotonic() - start < time_budget:
        pass_num += 1
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break
            if no_improve_passes > 10:
                break

            # Try pair (i, j) for a few random j
            for trial in range(3):
                j = rng.integers(0, n)
                if j == i:
                    j = (i + 1) % n

                # Try moving mass from i to j and vice versa
                for sign in [1, -1]:
                    for d in [current_delta, current_delta * 0.3]:
                        a_try = a.copy()
                        a_try[i] += sign * d
                        a_try[j] -= sign * d
                        a_try = np.maximum(a_try, 0)
                        s_try = float(np.sum(a_try))
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

            # Also try single coordinate
            if not improved:
                for d in [current_delta, current_delta * 0.3]:
                    for sign in [1, -1]:
                        a_try = a.copy()
                        a_try[i] += sign * d
                        if a_try[i] < 0:
                            continue
                        s_try = float(np.sum(a_try))
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
                    if improved:
                        break

        if not improved:
            no_improve_passes += 1
            current_delta *= 0.7
            if current_delta < 1e-8:
                break
        else:
            no_improve_passes = 0

        if verbose_getter() and pass_num % 10 == 0:
            print(f"            CD pass {pass_num}: obj={best_obj:.8f}, delta={current_delta:.2e}, time={time.monotonic()-start:.1f}s")

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def optimize_single(a, time_budget=100.0, verbose=False):
    """Full optimization pipeline for a single sequence."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: LP alternating
    if time_budget > 30:
        alloc = min(time_budget * 0.5, time_budget - 30)
        a_lp = lp_alternating_fast(a, time_budget=alloc, verbose=verbose)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            a = a_lp.copy()
            if verbose:
                print(f"      LP: obj={best_obj:.8f}")

    # Phase 2: Smooth max GD
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        alloc = min(remaining * 0.3, 40.0)
        a_sm = smooth_max_gd(a, time_budget=alloc, T_init=0.0003, T_min=0.00002)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_obj:
            best_obj = obj_sm
            best_a = a_sm.copy()
            a = a_sm.copy()
            if verbose:
                print(f"      SM-GD: obj={best_obj:.8f}")

    # Phase 3: Coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.5, 50.0)
        a_cd = coordinate_descent_pairwise(a, time_budget=alloc, delta=1e-4)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()
            a = a_cd.copy()
            if verbose:
                print(f"      CD: obj={best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-initialization + deep optimization + n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={len(parent)}): obj={best_score:.8f}")
    except Exception:
        pass

    n_parent = len(best_seq) if best_seq is not None else 800

    # === Phase 1: Multiple initializations with quick optimization ===
    print(f"\n=== Phase 1: Multi-init exploration ===")
    candidates = []

    # Candidate 1: Parent
    if best_seq is not None:
        candidates.append(('parent', best_seq.copy()))

    # Candidate 2: Uniform
    n_u = n_parent
    candidates.append(('uniform', np.ones(n_u) / n_u))

    # Candidate 3: Raised cosine
    candidates.append(('cosine', raised_cosine_init(n_parent)))

    # Candidate 4: Gaussian
    candidates.append(('gaussian', gaussian_init(n_parent)))

    # Candidate 5: Chirp
    candidates.append(('chirp', chirp_init(n_parent)))

    # Candidate 6: Votes-Quake
    candidates.append(('vq1', votes_quake_init(n_parent, num_iters=30, seed=42)))
    candidates.append(('vq2', votes_quake_init(n_parent, num_iters=30, seed=123)))

    # Candidate 7: Random smoothed
    a_rand = rng.standard_normal(n_parent) + 2.0
    a_rand = np.maximum(a_rand, 0)
    # Smooth with moving average
    kernel = np.ones(10) / 10
    a_rand = np.convolve(a_rand, kernel, mode='same')
    a_rand = np.maximum(a_rand, 0)
    s_r = np.sum(a_rand)
    if s_r > 0:
        candidates.append(('random', a_rand / s_r))

    # Quick evaluate all
    for name, a_cand in candidates:
        obj_cand = compute_obj(a_cand)
        print(f"  {name}: obj={obj_cand:.8f}")
        if obj_cand < best_score:
            best_score = obj_cand
            best_seq = a_cand.copy()

    # Quick optimize top candidates
    remaining = deadline - time.monotonic()
    top_k = min(5, len(candidates))
    candidates_sorted = sorted(candidates, key=lambda x: compute_obj(x[1]))[:top_k]

    for name, a_cand in candidates_sorted:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.2, 40.0)
        print(f"\n  Quick optimizing {name} (budget={alloc:.0f}s)...")
        a_opt = optimize_single(a_cand, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        print(f"    {name} optimized: obj={obj_opt:.8f}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()

    # === Phase 2: Deep optimization on best ===
    print(f"\n=== Phase 2: Deep optimization (n={len(best_seq)}) ===")
    print(f"  Current best: obj={best_score:.8f}")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.4, 400.0)
        print(f"  Budget: {alloc:.0f}s")
        a_deep = lp_alternating_fast(best_seq, time_budget=alloc, verbose=True)
        obj_deep = compute_obj(a_deep)
        if obj_deep < best_score:
            best_score = obj_deep
            best_seq = a_deep.copy()
        print(f"  After LP: obj={best_score:.8f}")

    # Smooth max polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 80.0)
        a_sm = smooth_max_gd(best_seq, time_budget=alloc, T_init=0.0002, T_min=0.00001)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
        print(f"  After SM-GD: obj={best_score:.8f}")

    # === Phase 3: n exploration ===
    print(f"\n=== Phase 3: n exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 200), max(200, n_best - 100), max(200, n_best - 50),
        max(200, n_best - 25), n_best, 
        n_best + 25, n_best + 50, n_best + 100, n_best + 200,
        300, 400, 500, 600, 700, 800, 900, 1000, 1100, 1200, 1400
    ]))
    # Filter to reasonable range
    n_options = [n for n in n_options if 200 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick LP + CD
        alloc_lp = min(remaining * 0.08, 20.0)
        a_opt = lp_alternating_fast(a_try, time_budget=alloc_lp)
        obj_try = compute_obj(a_opt)

        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_try < best_score + 0.003:
            alloc_cd = min(remaining * 0.05, 10.0)
            a_opt = coordinate_descent_pairwise(a_opt, time_budget=alloc_cd, delta=1e-4)
            obj_try = compute_obj(a_opt)

        status = " **" if obj_try < best_score else ""
        print(f"  n={n_try}: obj={obj_try:.8f}{status}")
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final polish ===")
    print(f"  Before: obj={best_score:.8f}, n={len(best_seq)}")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 30:
        alloc = min(remaining * 0.3, 100.0)
        a_final = lp_alternating_fast(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
        print(f"  After final LP: obj={best_score:.8f}")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 60.0)
        a_final = coordinate_descent_pairwise(best_seq, time_budget=alloc, delta=5e-5)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
        print(f"  After final CD: obj={best_score:.8f}")

    # One more round of LP with tiny budget
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 40.0)
        a_final = lp_alternating_fast(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
        print(f"  After 2nd LP: obj={best_score:.8f}")

    # Final CD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.5, 30.0)
        a_final = coordinate_descent_pairwise(best_seq, time_budget=alloc, delta=1e-5)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
        print(f"  After final-final CD: obj={best_score:.8f}")

    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
