    """

    # --- Security Checks ---

    # Verify that the input is a list

        return np.inf

    # Reject empty lists
    if not sequence:

    # Check each element in the list for validity
    for x in sequence:

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

        return np.inf

def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)

def conv_grad_at_k(a, k):

def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_norm, M_opt) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)

            continue
    return None, None

def peak_equalize(a0, time_budget=60.0, n_peaks=5, verbose=False):

    Iterative peak equalization: for each top peak, try to reduce it by

            break

        improved = False
        for k in top_k:
            k = int(k)
            # Find which a[m] contribute most to conv[k]
            contrib = np.zeros(n)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                m = k - i
                contrib[m] += a[i]
                if m != i:
                    contrib[i] += a[m]
            # Move mass from high contributors to low contributors
            top_contrib = np.argsort(contrib)[::-1][:3]
            low_contrib = np.argsort(contrib)[:3]
            for src in top_contrib:
                for dst in low_contrib:
                    if src == dst or a[src] < 1e-10:
                        continue
                    for frac in [0.05, 0.1, 0.2]:
                        delta = frac * a[src]
                        a[src] -= delta
                        a[dst] += delta
                        a = np.maximum(a, 0)
                        st = float(np.sum(a))
                        if st < 1e-10:
                            a[src] += delta
                            a[dst] -= delta
                            continue
                        a = a / st
                        obj_try = compute_obj(a)
                        if obj_try < best_obj - 1e-13:
                            best_obj = obj_try
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[dst] -= delta
                            a = a / st
                            break
        if not improved and outer > 10:
            break
        a = best_a.copy()
        if verbose:
            print(f"      PEQ iter={outer}: obj={best_obj:.8f}")

    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw
