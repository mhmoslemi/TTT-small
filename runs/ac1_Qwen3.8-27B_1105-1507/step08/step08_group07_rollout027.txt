```python
import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def mirror_descent_step(a, tau, eta):
    """
    One mirror descent step.
    a: current sequence (sum = 1, a >= 0)
    tau: temperature for softmax
    eta: step size
    Returns: updated a
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Compute softmax probabilities
    tau_conv = tau * conv
    max_val = np.max(tau_conv)
    exp_vals = np.exp(tau_conv - max_val)
    p = exp_vals / np.sum(exp_vals)
    
    # Compute t[i] = sum_k p[k] * a[k-i] for valid k-i
    # = (reverse(a) * p)[n-1 : n-1+n]
    size = 1
    while size < 3*n - 1:  # need at least n + (2n-1) - 1 = 3n-2
        size <<= 1
    
    b = a[::-1]  # reversed a
    fb = rfft(b, size)
    fp = rfft(p, size)
    conv_bp = irfft(fb * fp, size)[:3*n-2]  # linear conv of b and p
    
    # t[i] = conv_bp[i + n - 1] for i = 0, ..., n-1
    t = conv_bp[n-1:n-1+n]
    
    # Mirror descent update: a_i *= exp(-eta * (t_i - mean(t)))
    t_centered = t - np.mean(t)
    a_new = a * np.exp(-eta * t_centered)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    
    return a_new, conv, p
```

```python
def mirror_descent_optimize(a, time_budget=100, verbose=False):
    """
    Optimize using mirror descent with tau annealing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Tau schedule: start smooth, get sharper
    # tau = n / gamma, where gamma goes from 1 to n (or more)
    gamma_start = 5.0  # smooth
    gamma_end = n * 5.0  # very sharp
    n_iters = 200
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal gamma
        frac = it / max(n_iters - 1, 1)
        gamma = gamma_start * (gamma_end / gamma_start) ** frac
        tau = n / gamma
        
        # Step size: decrease over time
        eta = 0.5 / (1 + it * 0.01)
        
        a, conv, p = mirror_descent_step(a, tau, eta)
        obj = compute_obj(a)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if verbose and it % 20 == 0:
            print(f"  MD iter {it}: obj={best_obj:.8f}, gamma={gamma:.1f}")
    
    return best_a
```

"""
Optimization combining mirror descent (exponential weights) with LP alternating,
multiple diverse initializations, and coordinate polish.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_grad_smoothed(a, tau):
    """
    Compute the smoothed objective and its gradient.
    f(a) = (1/tau) * log(sum_k exp(tau * conv_k(a)))
    grad_i = 2 * sum_k p_k * a_{k-i} (with p_k = softmax(tau*conv_k))
    """
    n = len(a)
    conv = compute_conv(a)
    tau_conv = tau * conv
    max_val = np.max(tau_conv)
    exp_vals = np.exp(tau_conv - max_val)
    p = exp_vals / np.sum(exp_vals)
    f = (max_val + np.log(np.sum(exp_vals))) / tau
    
    # grad[i] = 2 * sum_j p[j+i] * a[j]
    # = 2 * (reverse(a) * p)[n-1 + i]
    size = 1
    while size < 3*n - 1:
        size <<= 1
    b = a[::-1]
    fb = rfft(b, size)
    fp = rfft(p, size)
    conv_bp = irfft(fb * fp, size)
    t = conv_bp[n-1:n-1+n]
    grad = 2.0 * t
    
    return f, grad, p, conv

def mirror_descent_optimize(a, time_budget=100.0, verbose=False):
    """
    Mirror descent with tau annealing and adaptive step size.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()

    n_inner = 30  # iterations per tau level
    
    for outer in range(n_outer):
        if time.monotonic() - start > time_budget:
            break
        
        # Tau schedule: geometric from smooth to sharp
        frac = outer / max(n_outer - 1, 1)
        # Start with tau = n/5 (smooth), end with tau = n*10 (sharp)
        tau = (n / 5.0) * (50.0) ** frac

            if time.monotonic() - start > time_budget:

                # Center the gradient for the mirror step

                a_try = a * np.exp(-eta * grad_centered / max(np.max(np.abs(grad_centered)), 1e-10))
                a_try = np.maximum(a_try, 0)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp(a):

    A = build_conv_matrix(a)
    
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(c1.dual_value), 0)
                    except:
                        pass
                    return z_val, float(M.value), dual_y
        except:
            continue
    
    # Fallback
    try:
        from scipy.optimize import linprog
        c = np.zeros(n+1)
        c[-1] = 1.0
        A_ub = np.zeros((2*n-1, n+1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2*n-1)
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)]*(n+1)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs

    except:
        pass

def coordinate_descent(a, time_budget=30.0):

    n = len(a)

    a = a / np.sum(a)

    best_obj = compute_obj(a)

            i = rng.integers(0, n)

                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
