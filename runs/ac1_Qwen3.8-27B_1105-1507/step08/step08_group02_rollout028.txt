```python
def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv) / (s * s))

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_topK(a, K=5, temperature=0.5):
    """Compute subgradient of max(conv) using top-K peaks with softmax weighting."""
    conv = compute_conv(a)
    max_val = np.max(conv)
    if max_val < 1e-15:
        return np.zeros(len(a))
    
    # Softmax weights over top peaks
    top_idx = np.argsort(conv)[::-1][:K]
    weights = np.exp((conv[top_idx] - max_val) / (temperature * max_val))
    weights /= weights.sum()
    
    g = np.zeros(len(a))
    n = len(a)

        k = int(k)
        # conv[k] = sum_i a[i] * a[k-i] (where 0<=i<n, 0<=k-i<n)
        # d(conv[k])/d(a[j]) = a[k-j] + a[j-k] (if in bounds)
```

```python
    # Build A matrix: A[k,j] = a[j] if 0 <= k-j < n else 0
    # Using the structure: for column j, A[j:j+n, j] = a[0:n] (shifted)
    # Actually A[k,j] = a[k-j] if 0 <= k-j < n
    # So column j has a[0], a[1], ..., a[n-1] starting at row j
    
    # For efficiency, build A using stride tricks or explicit loop
    A = np.zeros((2*n-1, n))
    for j in range(n):
        end = min(2*n-1, j+n)
        A[j:end, j] = a[:end-j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve(solver=cp.CLARABEL)
    
    if g.value is not None:
        g_val = np.maximum(g.value, 0)
        s = g_val.sum()
        if s > 1e-12:
            return g_val / s
    return None
```

```python
def peak_perturb(a, k_star, scale=0.1, rng=None):
    n = len(a)
    if rng is None:
        rng = np.random.default_rng()
    
    # Compute contribution of each coordinate to the peak
    contrib = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            contrib[i] = a[i] * a[j]
            if i != j:
                contrib[j] += a[j] * a[i]
    
    # Perturb: add noise proportional to contribution
    noise = scale * rng.normal(0, 1, n) * (contrib / (np.max(contrib) + 1e-15))
    a_new = a + noise
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    return a_new
```

```python
"""
Hybrid LP-alternating + subgradient optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
Focuses on efficient LP iterations with bisection line search,
peak-aware perturbations, and subgradient polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A such that A @ g = conv(a, g). Shape (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp_for_a(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
        if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(g.value, 0)
            s = float(np.sum(g_val))
            if s > 1e-12:
                return g_val / s
    except Exception:
        pass

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-12:
                return g_val / s
    except Exception:
        pass

    return None


def compute_subgradient(a, K=5, temperature=0.5):
    """
    Compute subgradient of max_k conv(a,a)[k] using top-K peaks.
    d(conv[k])/d(a[j]) = 2*a[k-j] if 0<=k-j<n, else 0.
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = np.max(conv)
    if max_val < 1e-15:
        return np.zeros(n)

    top_idx = np.argsort(conv)[::-1][:K]
    weights = np.exp((conv[top_idx] - max_val) / (temperature * max_val))
    weights /= weights.sum()

    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        # g[j] += w * 2 * a[k-j] for j in valid range
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            g[j_start:j_end + 1] += 2.0 * w * a[k - j_end:k - j_start + 1][::-1]
    return g


def mirror_descent_step(a, g, lr):
    """Mirror descent (exponential family) step on the simplex."""
    log_a = np.log(np.maximum(a, 1e-30))
    log_a_new = log_a - lr * g
    log_a_new -= np.max(log_a_new)
    a_new = np.exp(log_a_new)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def peak_perturb(a, scale=0.1, rng=None):
    """Perturb a based on peak contributions to break symmetry."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(n - 1, k_star)
    for j in range(j_start, j_end + 1):
        other = k_star - j
        if 0 <= other < n:
            contrib[j] = a[j] * a[other]
            if other != j:
                contrib[other] = max(contrib[other], a[other] * a[j])

    max_c = np.max(contrib)
    if max_c < 1e-15:
        return a.copy()

    noise = scale * rng.normal(0, 1, n) * (contrib / max_c)
    a_new = a + noise
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def lp_alternate(a, time_budget=600.0, verbose=False):
    """
    LP alternating with bisection line search and peak-aware restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall = 0
    patience = 40
    it = 0
    rng = np.random.default_rng(12345 + n)

    while time.monotonic() - start < time_budget:
        # Solve LP
        g = solve_lp_for_a(a)
        if g is None:
            g = a.copy()

        # Bisection line search on f(t) = obj((1-t)*a + t*g)
        lo, hi = 0.0, 1.0
        best_t = 0.0
        best_t_obj = best_obj

        # First try a few key points
        for t in [0.5, 0.7, 0.3, 1.0, 0.8, 0.2, 0.6, 0.4]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            st = a_try.sum()
            if st < 1e-15:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        # Refine with bisection around best_t
        if best_t > 0 and best_t < 1:
            for _ in range(5):
                mid1 = best_t * 0.7
                mid2 = best_t * 1.3
                for t in [mid1, mid2, best_t * 0.9, best_t * 1.1]:
                    t = max(0.01, min(1.0, t))
                    a_try = (1.0 - t) * a + t * g
                    a_try = np.maximum(a_try, 0)
                    st = a_try.sum()
                    if st < 1e-15:
                        continue
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_t_obj:
                        best_t_obj = obj_try
                        best_t = t

        it += 1
        if best_t_obj < best_obj - 1e-13:
            a_try = (1.0 - best_t) * a + best_t * g
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            a = a_try
            best_obj = best_t_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall >= patience:
                # Perturb and restart
                scale = 0.03 + 0.03 * min(stall // patience, 5)
                a = peak_perturb(best_a, scale=scale, rng=rng)
                best_obj = compute_obj(a)
                if best_obj < compute_obj(best_a):
                    best_a = a.copy()
                else:
                    a = best_a.copy()
                stall = 0
                patience = min(100, patience + 10)

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={elapsed:.1f}s, stall={stall}")

    return best_a


def subgradient_polish(a, time_budget=100.0, verbose=False):
    """
    Mirror descent with Nesterov momentum for polishing.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()  # Nesterov point
    v = np.zeros(n)
    momentum = 0.0
    it = 0
    stall = 0
    patience = 50
    rng = np.random.default_rng(999 + n)

    while time.monotonic() - start < time_budget:
        lr = 0.5 / (1.0 + 0.05 * it)
        K = 3 if it < 200 else 5

        g = compute_subgradient(y, K=K, temperature=0.3 + 0.1 * (it % 100) / 100)
        momentum = min(0.8, momentum + 0.005)
        v = momentum * v + (1.0 - momentum) * g

        y_new = mirror_descent_step(y, v, lr)

        obj_new = compute_obj(y_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = y_new.copy()
            a = y_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Small perturbation
                pert_scale = 0.02 + 0.01 * (stall // patience)
                y = best_a * np.exp(pert_scale * rng.normal(0, 1, n))
                y = mirror_descent_step(y, np.zeros(n), 0)
                y = y / y.sum()
                v = np.zeros(n)
                momentum = 0.0
                stall = 0
                if compute_obj(y) > best_obj:
                    y = best_a.copy()

        y = y_new
        it += 1

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    SubGrad it{it}: obj={best_obj:.8f}, lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 5:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.05
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a
```
