import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = np.sum(a)
    if s < 1e-15:
        return None, None
    a = a / s
    A = build_conv_matrix(a)
    
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_g = float(np.sum(g_val))
            if s_g > 1e-10:
                return g_val / s_g, float(res.x[n])
    except Exception:
        pass
    
    try:

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated smooth-max

descent, and multiple initializations including spectral (flat-spectrum) starts.

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def solve_lp(a):
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = np.sum(a)
    if s < 1e-15:
        return None, None
    a = a / s
    A = build_conv_matrix(a)
    
    # Try scipy linprog first (HiGHS is usually fastest for this size)
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_g = float(np.sum(g_val))
            if s_g > 1e-10:
                return g_val / s_g, float(res.x[n])
    except Exception:
        pass
    
    # Fall back to cvxpy
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in ['CLARABEL', 'ECOS', 'SCS']:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        return g_val / s_g, float(M.value)
            except Exception:
                continue
    except ImportError:
        pass
    
    return None, None


def smooth_max_nesterov(a, time_budget=200.0, T_init=0.001, T_min=0.0001, verbose=False):
    """
    Nesterov-accelerated gradient descent with smooth max approximation.
    Optimizes on the simplex using projection.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.01
    momentum = 0.9
    y = a.copy()  # Nesterov lookahead point
    v = np.zeros(n)
    stall = 0
    best_iter = 0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        if it % 100 == 0 and time.monotonic() - start > time_budget:
            break

        # Compute smooth max gradient at y
        conv_y = compute_conv(y)
        max_c = np.max(conv_y)
        shifted = np.clip((conv_y - max_c) / T, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        w = exp_vals / total  # softmax weights on convolution, length 2n-1

        # Gradient: grad[i] = 2n * sum_k w[k] * 2 * a[k-i] (valid indices)
        # = 4n * cross_corr(w, a)[i]
        # cross_corr(w, a)[i] = sum_m w[i+m] * a[m]
        # This is a valid-mode correlation
        grad = np.zeros(n)
        for i in range(n):
            # sum_m w[i+m] * a[m] for m=0..n-1 where i+m < 2n-1
            m_max = min(n, 2 * n - 1 - i)
            if m_max > 0:
                grad[i] = np.dot(w[i:i+m_max], a[:m_max])
        grad *= 4.0 * n

        # Center the gradient (remove mean to stay on simplex)
        grad = grad - np.mean(grad)

        # Nesterov update
        v = momentum * v - lr * grad
        y_new = y + v
        a_new = simplex_project(y_new)

        # Update y for next iteration (Nesterov lookahead)
        y = a_new + momentum * (a_new - a)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            best_iter = it
            if it % 500 == 0 and verbose:
                print(f"    SN it{it}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.5f}")
        else:
            stall += 1
            if stall > 50:
                # Adaptive: reduce learning rate and temperature
                lr *= 0.8
                T = max(T_min, T * 0.95)
                a = best_a.copy()
                y = a.copy()
                v = np.zeros(n)
                stall = 0
                if lr < 1e-7:
                    break
            elif stall > 20:
                T = max(T_min, T * 0.98)

        # Adaptive temperature decay
        if it > 100:
            T = max(T_min, T_init * (0.9995 ** it))

    return best_a

def lp_alternating(a, time_budget=200.0, verbose=False):
    """LP alternating with convex line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Convex line search using golden section
        a_norm = a / np.sum(a)
        g_norm = g_lp / np.sum(g_lp)

        conv_aa = compute_conv(a_norm)
        conv_ag = compute_conv_cross(a_norm, g_norm)
        conv_gg = compute_conv(g_norm)

        def obj_at_t(t):
            val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
            return 2.0 * n * float(np.max(val))

        # Golden section search
        gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
        lo, hi = 0.0, 1.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = obj_at_t(c)
        fd = obj_at_t(d)
        for _ in range(60):
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - gr * (hi - lo)
                fc = obj_at_t(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = obj_at_t(d)

        t_opt = (lo + hi) / 2.0
        a_new = (1 - t_opt) * a_norm + t_opt * g_norm
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 10 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}")
        else:
            # Try smaller step
            improved = False
            for t_try in [0.1, 0.05, 0.02]:
                a_try = (1 - t_try) * a + t_try * g_lp
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            if not improved:
                # Perturb and restart
                rng = np.random.default_rng(it + n)
                a_pert = best_a * np.exp(rng.normal(0, 0.01, n))
                a_pert = np.maximum(a_pert, 0)
                sp = float(np.sum(a_pert))
                if sp > 1e-10:
                    a = a_pert / sp
                else:
                    break

    return best_a


def pair_coordinate_descent(a, time_budget=100.0, verbose=False):
    """
    Pair coordinate descent: adjust pairs of coordinates.
    For each pair (i, j), try moving mass from j to i (or vice versa).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    # Precompute which pairs to try: favor pairs that are far apart
    # (to affect different parts of the convolution)
    pair_list = []
    for _ in range(min(50 * n, 5000)):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i != j:
            pair_list.append((int(i), int(j)))

    delta_base = 0.01
    stall = 0

    while time.monotonic() - start < time_budget:
        improved = False
        for (i, j) in pair_list:
            if time.monotonic() - start > time_budget:
                break

            # Try moving mass from j to i
            for delta in [delta_base, delta_base * 0.5, delta_base * 2]:
                if a[j] < delta:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if not improved:
                # Try moving mass from i to j
                for delta in [delta_base, delta_base * 0.5]:
                    if a[i] < delta:
                        continue
                    a_try = a.copy()
                    a_try[j] += delta
                    a_try[i] -= delta
                    a_try = np.maximum(a_try, 0)
                    st = float(np.sum(a_try))
                    if st < 1e-10:
                        continue
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if not improved:
            delta_base *= 0.7
            stall += 1
            if stall > 5 or delta_base < 1e-6:
                break
        else:
            stall = 0
            if verbose:
                print(f"    PCD: obj={best_obj:.8f}, delta={delta_base:.5f}")

    return best_a


def single_coordinate_descent(a, time_budget=100.0, verbose=False):
    """Single coordinate descent with adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    delta = 0.005
    stall = 0

    while time.monotonic() - start < time_budget:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            # Try increasing (decrease others proportionally)
            for d in [delta, delta * 0.3]:
                a_try = a.copy()
                a_try[i] += d
                # Redistribute: decrease all others proportionally
                others = np.ones(n)
                others[i] = 0
                current_sum = float(np.sum(a_try[others]))
                if current_sum < d:
                    continue
                a_try[others] -= d * a_try[others] / current_sum
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if improved:
                continue

            # Try decreasing
            for d in [delta, delta * 0.3]:
                if a[i] < d:
                    continue
                a_try = a.copy()
                a_try[i] -= d
                # Redistribute to others
                others = np.ones(n)
                others[i] = 0
                current_sum = float(np.sum(a_try[others]))
                if current_sum < 1e-10:
                    continue
                a_try[others] += d * a_try[others] / current_sum
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try
