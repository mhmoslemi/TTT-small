"""
Hybrid optimizer for minimum peak autocorrelation sequences.
Uses projected subgradient descent, smoothed L-BFGS, LP alternating,
and multi-n exploration to minimize 2*n*max(autocorr)/(sum)^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def compute_conv(a):
    """Compute linear autocorrelation a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_obj_grad(a, t):
    """
    Compute smoothed max autocorrelation and its gradient.
    f(a) = (1/t) * log(sum_k exp(t * (a*a)[k]))
    grad[m] = 2 * sum_j a[j] * w[j+m] where w = softmax(t * conv)
    """
    n = len(a)
    conv = compute_conv(a)
    tc = t * conv
    tc_shifted = tc - tc.max()
    w = np.exp(tc_shifted)
    w_sum = w.sum()
    w = w / w_sum
    obj = conv.max() + np.log(w_sum) / t

    # Gradient via cross-correlation: grad[m] = 2*sum_j a[j]*w[j+m]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w, size)
    corr = irfft(np.conj(fa) * fw, size)
    gradient = 2.0 * corr[:n]

    return obj, gradient


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_for_direction(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    A = build_conv_matrix(a)

    # Use scipy linprog (HiGHS)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g))
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def projected_subgradient(a, time_budget=120.0, verbose=True):
    """
    Projected subgradient descent on the simplex with adaptive step size.
    Uses the exact subgradient of max autocorrelation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adaptive parameters
    eta = 1.0 / np.sqrt(n)
    best_eta = eta
    patience = 0
    max_patience = 50
    momentum = 0.0
    mom_decay = 0.95

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        f_val = conv[k_star]

        # Subgradient: grad[m] = 2*a[k*-m] if 0<=k*-m<n else 0
        grad = np.zeros(n)
        valid_m = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
        if len(valid_m) > 0:
            grad[valid_m] = 2.0 * a[k_star - valid_m]

        # Momentum
        momentum = mom_decay * momentum + (1 - mom_decay) * grad
        step_dir = momentum

        # Try step
        a_try = project_simplex(a - eta * step_dir)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-15:
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                patience = 0
            a = a_try
            # Slightly increase step
            eta = min(eta * 1.01, 2.0 / np.sqrt(n))
        else:
            patience += 1
            if patience > max_patience:
                # Decrease step, try smaller
                eta = eta * 0.7
                patience = 0
                if eta < 1e-6 / n:
                    # Reset with perturbation
                    rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    eta = 1.0 / np.sqrt(n)
                    momentum = np.zeros(n)
                    mom_decay = 0.9
                    if verbose:
                        print(f"    [Reset] obj={best_obj:.8f}")
            else:
                eta = eta * 0.95

        it += 1
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    Subgrad: it={it}, obj={best_obj:.8f}, eta={eta:.6f}, t={elapsed:.0f}s")

    return best_a


def smooth_lbfgs(a, time_budget=120.0, verbose=True):
    """
    Optimize using L-BFGS-B on the smoothed max autocorrelation.
    Uses reparameterization a = softmax(x) to handle constraints.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    # Start with moderate smoothing, increase over time
    t_values = [50, 100, 200, 500]

    for t in t_values:
        if time.monotonic() - start > time_budget:
            break

        # Reparameterize: a = softmax(x)
        a_safe = np.maximum(a, 1e-15)
        x0 = np.log(a_safe)

        def fun_and_grad(x):
            aa = np.exp(x - x.max())
            aa = aa / aa.sum()
            obj, grad = smooth_obj_grad(aa, t)
            # Chain rule: d(a_i)/dx_j = a_i * (delta_ij - a_j)
            # d(obj)/dx_j = sum_i grad_i * a_i * (delta_ij - a_j)
            # = grad_j * a_j - a_j * sum_i grad_i * a_i
            ga = np.sum(grad * aa)
            xgrad = aa * (grad - ga)
            return obj, xgrad

        try:
            res = minimize(fun_and_grad, x0, method='L-BFGS-B',
                           jac=True, options={'maxiter': 200, 'ftol': 1e-12,
                                              'gtol': 1e-8, 'maxls': 50})
            if res.x is not None:
                aa = np.exp(res.x - res.x.max())
                aa = aa / aa.sum()
                obj = compute_obj(aa)
                if obj < best_obj:
                    best_obj = obj
                    best_a = aa.copy()
                    if verbose:
                        print(f"    L-BFGS t={t}: obj={best_obj:.8f} **")
                else:
                    if verbose:
                        print(f"    L-BFGS t={t}: obj={obj:.8f}")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS t={t} error: {e}")

        a = best_a.copy()

    return best_a


def multiplicative_update(a, time_budget=100.0, verbose=True):
    """
    Multiplicative (exponential) subgradient update.
    a_new[m] = a[m] * exp(-eta * 2*a[k*-m]) normalized.
    This is mirror descent with negative entropy.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 0.5
    patience = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Multiplicative update
        factor = np.zeros(n)
        valid_m = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
        if len(valid_m) > 0:
            factor[valid_m] = 2.0 * a[k_star - valid_m]

        a_try = a * np.exp(-eta * factor)
        a_try = np.maximum(a_try, 1e-15)
        a_try = a_try / np.sum(a_try)

        # Also try averaging with second peak
        conv_sorted = np.argsort(conv)[::-1]
        k_star2 = int(conv_sorted[1]) if len(conv_sorted) > 1 else k_star
        factor2 = np.zeros(n)
        valid_m2 = np.arange(max(0, k_star2 - n + 1), min(n, k_star2 + 1))
        if len(valid_m2) > 0:
            factor2[valid_m2] = 2.0 * a[k_star2 - valid_m2]
        a_try2 = a * np.exp(-0.5 * eta * (factor + factor2))
        a_try2 = np.maximum(a_try2, 1e-15)
        a_try2 = a_try2 / np.sum(a_try2)

        obj_try = compute_obj(a_try)
        obj_try2 = compute_obj(a_try2)

        if obj_try < best_obj - 1e-15 or obj_try2 < best_obj - 1e-15:
            if obj_try <= obj_try2:
                a = a_try
                obj_new = obj_try
            else:
                a = a_try2
                obj_new = obj_try2
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
            patience = 0
            eta = min(eta * 1.02, 2.0)
        else:
            patience += 1
            if patience > 30:
                eta = eta * 0.7
                patience = 0
                if eta < 0.01:
                    rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                    a = best_a * np.exp(rng.normal(0, 0.03, n))
                    a = np.maximum(a, 1e-15)
                    a = a / np.sum(a)
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    eta = 0.3
            else:
                eta = eta * 0.97

        if verbose and patience % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    Mult: obj={best_obj:.8f}, eta={eta:.4f}, t={elapsed:.0f}s")

    return best_a

def coordinate_polish(a, time_budget=40.0):
    """
    Coordinate descent polish: try adjusting pairs of coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()

    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(500):
        if time.monotonic() - start > time_budget:

        for _ in range(30):
            if time.monotonic() - start > time_budget:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            for frac in [0.02, 0.05, 0.1, 0.2, 0.03, 0.08]:
                delta = frac * min(a[i], a[j])
                if delta <= 0:
                    continue
                if delta > 0.8 * min(a[i], a[j]):
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0 or a_try[j] < 0:
                    continue
                a_try = a_try / np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved and it > 50:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng):
    """Generate a diverse initial sequence."""
    choice = rng.integers(0, 6)
    if choice == 0:
        # Raised cosine
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    elif choice == 1:
        # Sine squared
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2
    elif choice == 2:
        # Modulated flat
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4 * np.pi * t)
    elif choice == 3:
        # Gaussian bump
        t = np.linspace(0, 1, n)
        a = np.exp(-50 * (t - 0.5) ** 2)
    elif choice == 4:
        # Random smooth
        a = np.ones(n)
        a += rng.normal(0, 0.3, n)
        # Smooth
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
    else:
        # Triangular
        t = np.linspace(0, 1, n)
        a = 1 - np.abs(2 * t - 1)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)

                best_score = obj
                best_seq = a_new.copy()
                print(f"  Subgrad2: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(60.0, remaining * 0.2)
            if alloc > 15:
                a_new = multiplicative_update(best_seq, time_budget=alloc, verbose=False)
                obj = compute_obj(a_new)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_new.copy()
                    print(f"  Mult2: obj={best_score:.8f} **")

    # === Phase 7: Coordinate Polish (60s) ===
    print(f"\n=== Phase 7: Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(60.0, remaining * 0.2)
        a_new = coordinate_polish(best_seq, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Polish: obj={best_score:.8f} **")
        else:
            print(f"  Polish: obj={obj:.8f}")

    # === Phase 8: Random restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 8: Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(5, int(remaining // 40))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            if r % 2 == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.04 + 0.015 * r, n_best))
            else:
                # Block perturbation
                a_r = best_seq.copy()
                n_blocks = max(5, n_best // 100)
                block_size = max(1, n_best // n_blocks)
                blocks = rng.choice(n_blocks, min(4, n_blocks), replace=False)
                for b in blocks:
                    si = int(b) * block_size
                    ei = min(si + block_size, n_best)
                    a_r[si:ei] *= rng.uniform(0.5, 1.5)

            a_r = np.maximum(a_r, 1e-10)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr

            alloc = min(remaining * 0.3, 30.0)
            a_opt = projected_subgradient(a_r, time_budget=alloc, verbose=False)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 9: Final short LP + polish ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(25.0, remaining * 0.3)
        a_new = coordinate_polish(best_seq, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
