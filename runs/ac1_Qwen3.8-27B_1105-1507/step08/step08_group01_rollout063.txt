"""
Optimized step function coefficient search using:
1. Symmetry exploitation (halves search space)

4. Multi-start with structured initializations
5. FFT-based fast convolutions

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A

def make_u_shaped(n, alpha):
    """Create a U-shaped initial sequence."""
    x = np.linspace(0, 1, n)
    t = 2 * x - 1  # in [-1, 1]
    a = 1.0 + alpha * t**2
    a = np.maximum(a, 0.01)
    return a / np.sum(a)

def smooth_max_grad_descent(a, time_budget, T_init=0.001, T_min=0.00005, verbose=False):
    """
    Gradient descent using smooth max approximation with adaptive temperature.
    Works on the simplex (sum(a)=1, a>=0).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.005
    momentum = 0.9
    v = np.zeros(n)
    stall = 0
    max_stall = 30

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break

        # Compute convolution and smooth max
        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total

        # Gradient: grad[i] = sum_k weights[k] * a[k-i] (where valid)
        # This is: conv(weights, a) but we need to be careful with bounds
        # grad[i] = sum_{k} weights[k] * a[k-i] where 0 <= k-i < n
        # = sum_{j=0}^{n-1} a[j] * weights[i+j] (where i+j < 2n-1)
        # This is the "valid" cross-correlation of weights and a
        grad = np.correlate(weights, a, mode='valid')
        
        # Center the gradient (removes the component along the all-ones direction)
        grad = grad - np.sum(grad) / n

        # Nesterov momentum update
        v = momentum * v - lr * grad
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Adaptively increase lr slightly if making progress
            if lr < 0.02:
                lr *= 1.01
        else:
            stall += 1
            if stall > max_stall:
                lr *= 0.6
                if lr < 1e-7:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
                max_stall = max(15, max_stall // 2)

        # Decrease temperature
        if it > 20:
            T = max(T_min, T_init * (0.9995 ** it))

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"      GD it{it}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.6f}, time={elapsed:.1f}s")

    return best_a

def solve_lp_for_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    try:
        from scipy.optimize import linprog
    except ImportError:
        return None, None

    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

def solve_lp_active(a, active_idx):
    """Solve LP with only active constraints."""
    try:
        from scipy.optimize import linprog
    except ImportError:
        return None, None

    n = len(a)
    a = a / np.sum(a)
    A_full = build_conv_matrix(a)
    A = A_full[active_idx, :]

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(active_idx))

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_norm = g_val / s
            # Verify against all constraints
            conv_check = A_full @ g_norm
            true_M = float(np.max(conv_check))
            return g_norm, true_M
    return None, None

def lp_alternating(a, time_budget, verbose=False):
    """LP alternating with convex line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        # Get active set
        conv_cur = compute_conv(a)
        max_conv = np.max(conv_cur)
        threshold = max_conv * 0.95
        active_set = np.where(conv_cur >= threshold)[0]

        # Solve LP (use active set if small enough, else full)
        if len(active_set) <= 20:
            g_lp, M_lp = solve_lp_active(a, active_set)
        else:
            g_lp, M_lp = solve_lp_for_direction(a)

        if g_lp is None:
            break

        # Convex line search
        a_norm = a
        g_norm = g_lp
        conv_aa = compute_conv(a_norm)
        conv_ag = compute_conv_cross(a_norm, g_norm)
        conv_gg = compute_conv(g_norm)

        lo, hi = 0.0, 1.0
        for _ in range(12):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            v1 = (1-m1)**2 * conv_aa + 2*m1*(1-m1) * conv_ag + m1**2 * conv_gg
            v2 = (1-m2)**2 * conv_aa + 2*m2*(1-m2) * conv_ag + m2**2 * conv_gg
            if np.max(v1) < np.max(v2):
                hi = m2
            else:
                lo = m1

        t_opt = (lo + hi) / 2.0
        a_new = (1 - t_opt) * a_norm + t_opt * g_norm
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 10 == 0:
                elapsed = time.monotonic() - start
                print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, active={len(active_set)}, time={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 15:
                # Perturb
                rng = np.random.default_rng(it * 7919 + n)
                a_pert = a * np.exp(rng.normal(0, 0.01, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a = a_pert / s_p
                stall = 0
            if stall > 50:
                break

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_gradient(a, weights):
    """
    Compute gradient of smooth max of conv(a,a) w.r.t. a.
    grad[m] = 2 * sum_j a[j] * weights[m+j] (where m+j < 2n-1)
    Uses FFT for efficiency.
    """
    n = len(a)
    w_rev = weights[::-1]  # reverse weights, length 2n-1
    # Linear convolution of a (length n) and w_rev (length 2n-1), result length 3n-2
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w_rev, size)
    conv_result = irfft(fa * fw, size)[:3 * n - 2]
    # grad[m] = 2 * conv_result[2n-2-m] for m=0..n-1
    grad = 2.0 * conv_result[n - 2 : 2 * n - 2 : -1]
    # Actually: grad[m] = 2 * conv_result[2*(n-1)-m] = 2 * conv_result[2n-2-m]
    # For m=0: index 2n-2; for m=n-1: index n-1
    # So grad = 2 * conv_result[n-1 : 2n-1][::-1]
    grad = 2.0 * conv_result[n - 1:2 * n - 1][::-1]
    return grad


def smooth_max_grad_descent(a, time_budget, T_init=0.001, T_min=0.00003, lr_init=0.005, verbose=False):
    """
    Gradient descent using smooth max approximation with adaptive temperature.
    Works on the simplex (sum(a)=1, a>=0).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = lr_init
    momentum = 0.9
    v = np.zeros(n)
    stall = 0
    max_stall = 40

    for it in range(500000):
        if time.monotonic() - start > time_budget:
            break

        # Compute convolution and smooth max weights
        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        if total < 1e-300:
            weights = np.full(len(conv), 1.0 / len(conv))
        else:
            weights = exp_vals / total

        # Compute gradient via FFT
        grad = smooth_max_gradient(a, weights)

        # Center the gradient (remove component along all-ones direction)
        grad = grad - np.sum(grad) / n

        # Nesterov momentum update
        v = momentum * v - lr * grad
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if lr < 0.03:
                lr *= 1.005
        else:
            stall += 1
            if stall > max_stall:
                lr *= 0.5
                if lr < 1e-7:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
                max_stall = max(10, max_stall // 2)

        # Decrease temperature
        if it > 10:
            T = max(T_min, T_init * (0.9997 ** it))

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"      GD it{it}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.6f}, stall={stall}, time={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end_k = min(2 * n - 1, j + n)
        A[j:end_k, j] = a[:end_k - j]
    return A


def solve_lp_full(a):
    """Solve LP: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0. Returns (g, M)."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def solve_lp_active(a, active_idx):
    """Solve LP with only constraints at active_idx. Returns (g, true_M)."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A_full = build_conv_matrix(a)
    A = A_full[active_idx, :]

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(active_idx))

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_norm = g_val / s
            conv_check = A_full @ g_norm
            true_M = float(np.max(conv_check))
            return g_norm, true_M
    return None, None


def lp_alternating(a, time_budget, verbose=False):
    """LP alternating with convex line search and active set management."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        # Get active set (peaks near maximum)
        conv_cur = compute_conv(a)
        max_conv = np.max(conv_cur)
        threshold = max_conv * 0.97
        active_set = np.where(conv_cur >= threshold)[0]

        # Solve LP
        if len(active_set) <= 30:
            g_lp, M_lp = solve_lp_active(a, active_set)
            if g_lp is None:
                g_lp, M_lp = solve_lp_full(a)
        else:
            g_lp, M_lp = solve_lp_full(a)

        if g_lp is None:
            break

        # Convex line search
        conv_aa = compute_conv(a)
        conv_ag = compute_conv_cross(a, g_lp)
        conv_gg = compute_conv(g_lp)

        lo, hi = 0.0, 1.0
        for _ in range(15):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            v1 = (1 - m1) ** 2 * conv_aa + 2 * m1 * (1 - m1) * conv_ag + m1 ** 2 * conv_gg
            v2 = (1 - m2) ** 2 * conv_aa + 2 * m2 * (1 - m2) * conv_ag + m2 ** 2 * conv_gg
            if np.max(v1) < np.max(v2):
                hi = m2
            else:
                lo = m1

        t_opt = (lo + hi) / 2.0
        a_new = (1 - t_opt) * a + t_opt * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 10 == 0:
                elapsed = time.monotonic() - start
                print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, active={len(active_set)}, time={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 20:
                rng = np.random.default_rng(it * 7919 + n + 1)
                a_pert = a * np.exp(rng.normal(0, 0.005, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a = a_pert / s_p
                stall = 0
            if stall > 60:
                break

    return best_a


def make_u_shaped(n, alpha):
    """Create a U-shaped initial sequence (large at ends, small in middle)."""
    x = np.linspace(0, 1, n)
    t = 2 * x - 1
    a = 1.0 + alpha * t ** 2
    a = np.maximum(a, 0.001)
    return a / np.sum(a)


def make_bimodal(n, width):
    """Create a bimodal initial sequence."""
    x = np.linspace(0, 1, n)
    center1 = 0.1
    center2 = 0.9
    a = np.exp(-((x - center1) / width) ** 2) + np.exp(-((x - center2) / width) ** 2)
    a = np.maximum(a, 0.001)
    return a / np.sum(a)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
