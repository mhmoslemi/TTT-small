Uses fast LP alternating (HiGHS), FFT-based smooth gradient descent with Adam,

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_lp_fast(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy HiGHS for speed. Returns (g_normalized, M_opt).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build A matrix: A[k, j] = a_norm[k-j] if 0 <= k-j < n, else 0
    # A has shape (2n-1, n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    
    # LP: min [0, ..., 0, 1] @ [g; M]
    # s.t. [A, -1] @ [g; M] <= 0
    #      [1, ..., 1, 0] @ [g; M] = 1
    #      g >= 0, M free
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.hstack([np.ones((1, n)), np.array([[0.0]])])
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0 and res.x is not None:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-12:
            return g_val / s, float(res.x[n])
    
    return None, None

                    a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                last_good_t = 0.3 + 0.3 * rng.random()
                temp = min(0.25, temp * 2.5)
            else:
                temp = max(0.03, temp * 0.99)
        
        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP it{iteration}: obj={best_obj:.8f}, t={last_good_t:.3f}, "
                  f"stall={stall}, temp={temp:.3f}, elapsed={elapsed:.1f}s")
    
    return best_a


def subgradient_descent(a, time_budget=80.0, verbose=False):
    """
    Direct subgradient descent on the max-conv objective.
    Uses the active set of peak constraints.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0 / n
    momentum = 0.0
    v = np.zeros(n)
    stall = 0
    patience = 100
    
    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()
        
        # Find all k where conv[k] is close to max (within 1%)
        threshold = max_val * 0.95
        active = np.where(conv >= threshold)[0]
        
        # Subgradient: average over active constraints
        # grad_i = (1/|active|) * sum_{k in active} a[k-i] (if 0 <= k-i < n)
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad /= len(active)
        grad *= 2.0 * n  # scale for the objective
        
        v = momentum * v + grad
        log_a = np.log(a)
        log_a_new = log_a - lr * v
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            momentum = min(0.8, momentum + 0.005)
            lr = max(1e-5, lr * 0.998)
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(it * 7 + n)
                a = best_a * np.exp(rng.normal(0, 0.05 + 0.01 * (stall // patience), n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                v = np.zeros(n)
                momentum = 0.0
                lr = max(1e-5, lr * 0.7)
                stall = 0
                a = a.copy()
                best_a = a.copy()
                best_obj = compute_obj(a)
            else:
                lr = max(1e-5, lr * 0.97)
        
        it += 1
        if verbose and it % 50 == 0:
            print(f"      SubGrad it{it}: obj={best_obj:.8f}, lr={lr:.5f}, "
                  f"|active|={len(active)}, t={time.monotonic()-start:.1f}s")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps
        c1, c2 = n // 4, 3 * n // 4
        w = n // 10
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Triangular
        x = np.arange(n) / (n - 1)
        a = 1.0 - np.abs(2 * x - 1)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 0.01))
        a = a / a.sum()
    elif mode == 5:
        # Random smooth
        freqs = rng.integers(1, 4, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Power-law
        x = np.arange(1, n + 1)
        a = x ** 0.3
        a = a / a.sum()
    elif mode == 7:
        # Chirp
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(2 * np.pi * (2 + 5 * x) * x)) + 0.05
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Hybrid approach combining:
    1. FFT-based smooth gradient descent (Adam)
    2. Fast LP alternating (HiGHS) with adaptive line search
    3. Subgradient descent on active peak constraints
    4. Multi-n exploration with resampling
    5. Diverse restarts
    Target: minimize 2*n*max(conv(a,a))/sum(a)^2 below 1.5027.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0  # safety margin
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, 1000)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Parent not available: {e}")
    
    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.8f}")
    
    n_current = len(best_seq)
    print(f"Starting: n={n_current}, obj={best_score:.8f}, budget={budget_s}s")
    
    # === Phase 1: Quick smooth optimization (Adam + FFT gradient) ===
    print(f"\n=== Phase 1: Smooth Adam Optimization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.10, 120.0)
    if p1_budget > 20:
        t0 = time.monotonic()
        a_smooth = smooth_optimize_adam(best_seq, time_budget=p1_budget,
                                        beta_start=30.0, beta_end=1500.0)
        obj_s = compute_obj(a_smooth)
        if obj_s < best_score:
            best_score = obj_s
            best_seq = a_smooth.copy()
            print(f"  Smooth Adam: obj={best_score:.8f} (improved in {time.monotonic()-t0:.1f}s)")
        else:
            print(f"  Smooth Adam: obj={obj_s:.8f} (no improvement)")
    
    # === Phase 2: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 400.0)
    
    for rnd in range(3):
        remaining = deadline - time.monotonic()
        alloc = min(p2_budget / (3 - rnd), remaining * 0.35)
        if alloc < 30:
            break
        
        if rnd > 0:
            # Perturb
            rng_r = np.random.default_rng(rnd * 999 + n_current)
            a_start = best_seq * (1.0 + 0.03 * rng_r.normal(0, 1, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        else:
            a_start = best_seq.copy()
        
        t0 = time.monotonic()
        a_lp = lp_alternating(a_start, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        elapsed = time.monotonic() - t0
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} ** (improved, {elapsed:.1f}s)")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} ({elapsed:.1f}s)")
    
    # === Phase 3: Subgradient descent ===
    print(f"\n=== Phase 3: Subgradient Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.10, 100.0)
    if p3_budget > 20:
        a_sg = subgradient_descent(best_seq, time_budget=p3_budget, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SubGrad: obj={best_score:.8f} **")
        else:
            print(f"  SubGrad: obj={obj_sg:.8f}")
    
    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best + 50, n_best + 100,
        n_best + 200, n_best + 300, 700, 900, 1000, 1100, 1200, 1400
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1500 and n != n_best]
    
    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        a_try = resample_sequence(best_seq, n_try)
        alloc_total = min(remaining * 0.12, 80.0)
        
        # Quick smooth
        half = alloc_total * 0.4
        if half > 10:
            a_S = smooth_optimize_adam(a_try, time_budget=half, beta_start=30, beta_end=800)
            if compute_obj(a_S) < compute_obj(a_try):
                a_try = a_S
        
        # LP
        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.08, 50.0)
        if lp_alloc > 15:
            a_opt = lp_alternating(a_try, time_budget=lp_alloc)
            obj_opt = compute_obj(a_opt)
        else:
            a_opt = a_try
            obj_opt = compute_obj(a_opt)
        
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")
    
    # === Phase 5: Diverse restarts blended with best ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_use = len(best_seq)
    
    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        
        mode = (r + 1) % 8
        a_r = diverse_init(n_use, rng, mode=mode)
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        
        restart_budget = min(remaining * 0.12, 60.0)
        if restart_budget > 30:
            a_r = smooth_optimize_adam(a_r, time_budget=restart_budget * 0.4,
                                       beta_start=20, beta_end=500)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_r = lp_alternating(a_r, time_budget=min(remaining * 0.08, 35.0))
        
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")
    
    # === Phase 6: Final intensive LP polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    if remaining > 30:
        alloc = min(remaining * 0.4, 150.0)
        a_final = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")
    
    # One more subgradient pass
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_sg2 = subgradient_descent(best_seq, time_budget=min(remaining * 0.3, 40.0))
        obj_sg2 = compute_obj(a_sg2)
        if obj_sg2 < best_score:
            best_score = obj_sg2
            best_seq = a_sg2.copy()
            print(f"  Final SubGrad: obj={best_score:.8f} **")
    
    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.4, 30.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")
    
    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
