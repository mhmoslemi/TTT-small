```python
"""
Optimized step function search using smooth-max (log-sum-exp) subgradient with
tau annealing, mirror descent, LP alternating with active set reduction,
and multi-restart with diverse initializations.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key innovation: smooth-max with annealed temperature provides a continuous
path from a flat (easy) landscape to the sharp (hard) peak, helping escape
local minima that trap direct max-based methods.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv / (sum)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_smooth_max_grad(a, p):
    """
    Compute grad[m] = 2 * sum_j a[j] * p[j+m] for m = 0,...,n-1.
    p has length 2n-1 (the lag weights).
    Uses FFT for efficiency.
    """
    n = len(a)
    size = 1
    while size < 3 * n:
        size <<= 1
    flip_a = a[::-1]
    fa = rfft(flip_a, size)
    fp = rfft(p, size)
    conv = irfft(fa * fp, size)
    return 2.0 * conv[n - 1:2 * n - 1]


def smooth_max_optimize(a, time_budget=200.0, tau_start=2.0, tau_end=80.0,
                        verbose=False, lr_init=0.3, mirror=True):
    """
    Smooth max optimization with tau annealing.
    Uses log-sum-exp approximation of max, with annealed temperature.
    Supports both Euclidean (projected gradient) and mirror descent updates.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.85
    lr = lr_init
    patience = 0
    max_patience = 200
    n_iter = 0

    # Adaptive: track best objective per tau level for restart
    last_improve_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Anneal tau: start smooth, end sharp
        # Use exponential schedule: tau = tau_start * (tau_end/tau_start)^progress
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.7)

        # Compute convolution
        c = compute_conv(a)

        # Compute softmax weights (lag importance)
        c_shifted = c - np.max(c)
        exp_c = np.exp(tau * c_shifted)
        p = exp_c / np.sum(exp_c)

        # Compute gradient
        grad = compute_smooth_max_grad(a, p)

        # Center gradient (project to simplex tangent)
        grad_centered = grad - np.mean(grad)

        # Adaptive learning rate
        gnorm = np.linalg.norm(grad_centered)
        if gnorm < 1e-15:
            break
        # Scale lr by progress (decay)
        effective_lr = lr * (1.0 - 0.5 * progress)

        if mirror:
            # Mirror descent (exponential weights)
            log_step = -effective_lr * grad_centered / (a + 1e-15)
            log_step = log_step - np.mean(log_step)
            # Clip for numerical stability
            log_step = np.clip(log_step, -10, 10)
            a_new = a * np.exp(log_step)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
        else:
            # Euclidean projected gradient with momentum
            momentum = beta * momentum - effective_lr * grad_centered / (gnorm + 1e-10)
            a_new = a + momentum
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s_new

        a = a_new
        n_iter += 1

        # Evaluate
        if n_iter % 5 == 0 or n_iter < 20:
            obj = compute_obj(a)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                patience = 0
                if verbose and (n_iter < 50 or n_iter % 200 == 0):
                    elapsed = time.monotonic() - start
                    print(f"      smooth-max: obj={best_obj:.8f}, tau={tau:.2f}, "
                          f"iter={n_iter}, t={elapsed:.0f}s")
            else:
                patience += 1

        # Restart if stuck
        if patience > max_patience:
            patience = 0
            momentum *= 0.1
            # Perturb
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            scale = 0.02 * (1 + 0.1 * min(patience // max_patience, 5))
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            a = a / s
            lr = min(lr * 1.2, 0.5)

        # Decay learning rate slowly
        if n_iter % 100 == 99:
            lr *= 0.97

    return best_a


def build_active_conv_matrix(a, active_indices):
    """Build submatrix A[active_indices, :] for the LP."""
    n = len(a)
    m = len(active_indices)
    A = np.zeros((m, n))
    a = np.asarray(a, dtype=np.float64)
    for idx, k in enumerate(active_indices):
        k = int(k)
        # A[idx, m] = a[k-m] if 0 <= k-m < n
        valid_from = max(0, k - n + 1)
        valid_to = min(n - 1, k)
        if valid_to >= valid_from:
            cols = np.arange(valid_from, valid_to + 1)
            A[idx, cols] = a[k - cols]
    return A


def solve_lp_active(a, active_indices):
    """
    Solve LP: min M s.t. A_active @ g <= M, sum(g)=1, g>=0.
    Returns (g, M, new_active) or None.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None
    a = a / s

    A = build_active_conv_matrix(a, active_indices)
    m = len(active_indices)

    if m == 0:
        g = np.ones(n) / n
        conv = compute_conv(g)
        return g, float(np.max(conv)), np.where(conv >= np.max(conv) - 1e-8)[0]

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.HIGHS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M_var.value) if M_var.value is not None else float(np.max(A @ g_val))
                    # Find newly active constraints (full check)
                    conv_full = compute_conv(g_val)
                    new_active = np.where(conv_full >= M_val - 1e-5)[0]
                    return g_val, M_val, new_active
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(res.x[n])
                conv_full = compute_conv(g_val)
                new_active = np.where(conv_full >= M_val - 1e-5)[0]
                return g_val, M_val, new_active
    except Exception:
        pass

    return None


def lp_alternating_improved(a, time_budget=300.0, verbose=False, max_active=50):
    """
    LP alternating with active set management and adaptive step.
    Key improvement: only use a subset of active constraints in the LP.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Initial active set
    conv = compute_conv(a)
    M_current = float(np.max(conv))
    active = np.where(conv >= M_current * 0.8)[0]
    if len(active) > max_active:
        active = np.argsort(conv)[::-1][:max_active]

    momentum = np.zeros(n)
    beta_mom = 0.5
    last_t = 0.5
    stall = 0

    while time.monotonic() - start < time_budget:
        result = solve_lp_active(a, active)
        if result is None:
            break
        g_lp, M_lp, new_active = result

        # Update active set: keep union of old and new, trim to max_active
        active_set = np.unique(np.concatenate([active, new_active]))
        if len(active_set) > max_active:
            # Keep the ones with highest (a*g) values
            ag = np.zeros(2 * n - 1)
            # Compute a*g for all lags
            ag = np.convolve(a, g_lp)[:2 * n - 1]
            active_set = active_set[np.argsort(ag[active_set])[::-1][:max_active]]
        active = active_set

        # Adaptive step size
        direction = g_lp - a
        momentum = beta_mom * momentum + (1 - beta_mom) * direction

        best_t_obj = np.inf
        best_t_val = last_t
        best_a_local = best_a.copy()

        for t in [0.1, 0.3, 0.5, 0.7, 1.0, last_t]:
            if t <= 0 or t > 2:
                continue
            a_try = a + t * momentum
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"      LP: obj={best_obj:.8f}, t={best_t_val:.2f}, "
                      f"active={len(active)}, time={elapsed:.0f}s")
        else:
            stall += 1
            momentum *= 0.5
            if stall > 50:
                # Perturb to escape
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                # Reset active set
                conv = compute_conv(a)
                M_current = float(np.max(conv))
                active = np.where(conv >= M_current * 0.7)[0]
                if len(active) > max_active:
                    active = np.argsort(conv)[::-1][:max_active]
                stall = 0
                last_t = 0.5

    return best_a


def gradient_polish(a, time_budget=60.0, verbose=False):
    """
    Direct subgradient polish targeting top-K active peaks.
    Uses Nesterov acceleration.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()  # main iterate
    x = a.copy()  # Nesterov lookahead
    momentum = np.zeros(n)
    beta = 0.8
    lr = 0.05
    top_k = 15

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        # Compute conv at lookahead
        conv_x = compute_conv(x)
        max_val = float(np.max(conv_x))
        # Get top-K active lags
        active = np.argsort(conv_x)[::-1][:top_k]

        # Compute subgradient (average of active peak gradients)
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            # grad += 2 * a[k::-1] (padded)
            # A[k-m] valid when 0 <= k-m < n
            valid_from = max(0, k - n + 1)
            valid_to = min(n - 1, k)
            if valid_to >= valid_from:
                grad[valid_from:valid_to + 1] += 2.0 * a[k - np.arange(valid_from, valid_to + 1)]
        grad /= len(active)

        # Center
        grad_c = grad - np.mean(grad)

        # Nesterov + momentum
        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        momentum = beta * momentum - lr * grad_c
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
            s_new = 1.0
        y_new = y_new / s_new

        x = y_new + (1 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        y = y_new

        # Check
        if it % 10 == 0:
            obj = compute_obj(y)
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = y.copy()
            if verbose and it % 500 == 0:
                print(f"      grad-polish: obj={best_obj:.8f}, iter={it}")

        # Decay lr
        if it % 200 == 199:
            lr *= 0.92

    return best_a


def generate_init(kind, n, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)

    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'ramp_up':
        a = np.linspace(0.1, 1.0, n)
        # Make it more asymmetric
        a = a ** 2
        return a / np.sum(a)
    elif kind == 'ramp_down':
        a = np.linspace(1.0, 0.1, n)
        a = a ** 2
        return a / np.sum(a)
    elif kind == 'bathtub':
        a = np.ones(n)
        edge = n // 5
        a[:edge] = 0.2
        a[-edge:] = 0.2
        a = a + 0.1 * np.sin(np.pi * np.arange(n) / n) ** 2
        return a / np.sum(a)
    elif kind == 'two_peaks':
        a = np.ones(n) * 0.3
        a[n // 4] = 2.0
        a[3 * n // 4] = 2.0
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'geometric':
        r = 0.95
        a = r ** np.arange(n)
        return a / np.sum(a)
    elif kind == 'geometric_rev':
        r = 0.95
        a = r ** np.arange(n - 1, -1, -1)
        return a / np.sum(a)
    elif kind == 'triangle_asym':
        # Asymmetric triangle: peak at 1/3
        peak_pos = n // 3
        a = np.zeros(n)
        for i in range(n):
            a[i] = 1.0 - abs(i - peak_pos) / max(peak_pos, n - peak_pos - 1, 1)
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif kind == 'sparse_peak':
        a = np.full(n, 0.1)
        center = n // 2
        a[center - n // 8:center + n // 8 + 1] = 1.0
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combined smooth-max + LP + gradient polish with
    multi-restart, tau annealing, and length sweep.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)
    a = best_seq.copy()

    # === Phase 1: Diverse Initializations (60s) ===
    print(f"\n=== Phase 1: Diverse Initializations (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    init_budget = min(60.0, remaining * 0.05)

    init_kinds = ['cosine', 'sine_taper', 'gaussian', 'ramp_up', 'ramp_down',
                  'bathtub', 'geometric', 'geometric_rev', 'triangle_asym', 'random']
    per_init = max(3.0, init_budget / len(init_kinds))

    candidates = [a.copy()]
    for kind in init_kinds:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        a_init = generate_init(kind, n_parent, rng)
        obj_init = compute_obj(a_init)
        candidates.append(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init '{kind}': obj={obj_init:.8f} **")
        else:
            print(f"  Init '{kind}': obj={obj_init:.8f}")

    # === Phase 2: Smooth-Max on Best Init (300s) ===
    print(f"\n=== Phase 2: Smooth-Max Optimization ===")
    remaining = deadline - time.monotonic()
    sm_budget = min(300.0, remaining * 0.25)

    # Try smooth-max from best and a few diverse inits
    sm_starts = candidates[:4]  # top 4 diverse starts
    per_sm = sm_budget / len(sm_starts)

    for idx, a_start in enumerate(sm_starts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_sm, remaining * 0.4)
        a_sm = smooth_max_optimize(a_start, time_budget=alloc,
                                    tau_start=2.0, tau_end=60.0,
                                    verbose=True, lr_init=0.3, mirror=True)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            a = a_sm.copy()
            print(f"  ** Smooth-max {idx}: obj={best_score:.8f} **")
        else:
            print(f"  Smooth-max {idx}: obj={obj_sm:.8f}")

    # Also try non-mirror variant
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_sm2 = smooth_max_optimize(best_seq, time_budget=min(100.0, remaining * 0.15),
                                     tau_start=5.0, tau_end=100.0,
                                     verbose=True, lr_init=0.2, mirror=False)
        obj_sm2 = compute_obj(a_sm2)
        if obj_sm2 < best_score:
            best_score = obj_sm2
            best_seq = a_sm2.copy()
            a = a_sm2.copy()
            print(f"  ** Smooth-max (euc): obj={best_score:.8f} **")

    # === Phase 3: LP Alternating (400s) ===
    print(f"\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(400.0, remaining * 0.3)

    # Multiple rounds
    for round_idx in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(lp_budget / (3 - round_idx), remaining * 0.5)
        a_lp = lp_alternating_improved(a, time_budget=alloc, verbose=True,
                                         max_active=40)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp.copy()
            print(f"  ** LP R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP R{round_idx}: obj={obj_lp:.8f}")
            # Perturb for next round
            if round_idx > 0:
                a = best_seq * np.exp(rng.normal(0, 0.025, n_parent))
                a = np.maximum(a, 0)
                a = a / np.sum(a)

    # === Phase 4: Smooth-Max Polish with High Tau (150s) ===
    print(f"\n=== Phase 4: High-Tau Smooth-Max Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        sm2_budget = min(150.0, remaining * 0.15)
        a_polish = smooth_max_optimize(best_seq, time_budget=sm2_budget,
                                        tau_start=20.0, tau_end=200.0,
                                        verbose=True, lr_init=0.1, mirror=True)
        obj_polish = compute_obj(a_polish)
        if obj_polish < best_score:
            best_score = obj_polish
            best_seq = a_polish.copy()
            a = a_polish.copy()
            print(f"  ** High-tau polish: obj={best_score:.8f} **")
        else:
            print(f"  High-tau polish: obj={obj_polish:.8f}")

    # === Phase 5: Gradient Polish (100s) ===
    print(f"\n=== Phase 5: Gradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        gp_budget = min(100.0, remaining * 0.1)
        a_gp = gradient_polish(a, time_budget=gp_budget, verbose=True)
        obj_gp = compute_obj(a_gp)
        if obj_gp < best_score:
            best_score = obj_gp
            best_seq = a_gp.copy()
            a = a_gp.copy()
            print(f"  ** Grad polish: obj={best_score:.8f} **")
        else:
            print(f"  Grad polish: obj={obj_gp:.8f}")

    # === Phase 6: Multi-Restart Exploration (200s) ===
    print(f"\n=== Phase 6: Multi-Restart ===")
    remaining = deadline - time.monotonic()
    restart_budget = min(200.0, remaining * 0.2)
    n_restarts = max(3, int(restart_budget / 40))

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Create perturbed version of best
        scale = 0.03 + 0.01 * r
        a_restart = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_restart = np.maximum(a_restart, 0)
        a_restart = a_restart / np.sum(a_restart)

        # Quick smooth-max
        alloc = min(30.0, remaining * 0.3)
        a_r = smooth_max_optimize(a_restart, time_budget=alloc,
                                   tau_start=5.0, tau_end=80.0,
                                   mirror=True, lr_init=0.2)
        # Quick gradient polish
        a_r = gradient_polish(a_r, time_budget=min(10.0, remaining * 0.1))
        obj_r = compute_obj(a_r)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            a = a_r.copy()
            print(f"  ** Restart {r} (scale={scale:.3f}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (scale={scale:.3f}): obj={obj_r:.8f}")

    # === Phase 7: Length Sweep (150s) ===
    print(f"\n=== Phase 7: Length Sweep ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(100, n_best - 200),
        max(100, n_best - 100),
        max(100, n_best - 50),
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        n_best + 400,
    ]))
    n_options = [n for n in n_options if 100 <= n <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick optimize: smooth-max + gradient
        alloc_sm = min(25.0, remaining * 0.4)
        alloc_gp = min(10.0, remaining * 0.15)

        a_opt = smooth_max_optimize(a_try, time_budget=alloc_sm,
                                     tau_start=5.0, tau_end=80.0,
                                     mirror=True, lr_init=0.2)
        a_opt = gradient_polish(a_opt, time_budget=alloc_gp)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final Polish with Remaining Time ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Alternate between smooth-max and LP until time runs out
        while remaining > 15:
            method = remaining % 2  # alternate
            if method == 0:
                alloc = min(remaining * 0.3, 40.0)
                a_new = smooth_max_optimize(best_seq, time_budget=alloc,
                                             tau_start=30.0, tau_end=200.0,
                                             mirror=True, lr_init=0.05)
            else:
                alloc = min(remaining * 0.3, 40.0)
                a_new = lp_alternating_improved(best_seq, time_budget=alloc,
                                                  max_active=30)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Final polish: obj={best_score:.8f} **")
            else:
                # Small perturbation
                best_seq_p = best_seq * np.exp(rng.normal(0, 0.015, len(best_seq)))
                best_seq_p = np.maximum(best_seq_p, 0)
                best_seq_p = best_seq_p / np.sum(best_seq_p)
                a_new2 = smooth_max_optimize(best_seq_p, time_budget=min(alloc, 20.0),
                                              tau_start=20.0, tau_end=100.0,
                                              mirror=True, lr_init=0.1)
                obj_new2 = compute_obj(a_new2)
                if obj_new2 < best_score:
                    best_score = obj_new2
                    best_seq = a_new2.copy()
                    print(f"  Final polish (pert): obj={best_score:.8f} **")

            remaining = deadline - time.monotonic()

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
