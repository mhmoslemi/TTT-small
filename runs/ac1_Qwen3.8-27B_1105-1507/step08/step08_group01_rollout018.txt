def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp_direction(a, solver_list=None):
    """
    Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    if solver_list is None:
        solver_list = ['CLARABEL', 'MOSEK', 'ECOS', 'SCS', 'HIHGS']

    # Try cvxpy first
    try:
        import cvxpy as cp
        for solver in solver_list:
            try:
                g = cp.Variable(n, nonneg=True)
                M = cp.Variable()
                constraints = [A @ g <= M, cp.sum(g) == 1]
                prob = cp.Problem(cp.Minimize(M), constraints)
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0).astype(np.float64)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    except ImportError:
        pass

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def solve_qp_flatness(a, weight_power=2.0, solver_list=None):
    """
    Solve QP: min sum_k w_k * (A@g)[k]^2 s.t. sum(g)=1, g>=0.
    Where w_k are adaptive weights emphasizing current peaks.
    Returns g or None.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)
    max_conv = np.max(conv)
    if max_conv < 1e-15:
        return a_norm.copy()

    # Adaptive weights: emphasize peaks
    w = (conv / max_conv) ** weight_power
    w = np.maximum(w, 1e-6)

    A = build_conv_matrix(a_norm)
    # Q = A^T diag(w) A
    Aw = A * np.sqrt(w)[:, None]
    Q = Aw.T @ Aw

    if solver_list is None:
        solver_list = ['CLARABEL', 'MOSEK', 'ECOS', 'SCS']

    # Try cvxpy
    try:
        import cvxpy as cp
        for solver in solver_list:
            try:
                g = cp.Variable(n, nonneg=True)
                constraints = [cp.sum(g) == 1]
                objective = cp.Minimize(cp.quad_form(g, Q, assume_PSD=True))
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0).astype(np.float64)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue
    except ImportError:
        pass

    # Fallback: scipy minimize with Hessian
    try:
        from scipy.optimize import minimize
        def obj_fun(g):
            Ag = A @ g
            return np.dot(w * Ag, Ag)
        def grad_fun(g):
            return 2.0 * A.T @ (w * (A @ g))
        def hess_fun(g):
            return 2.0 * Q
        constraints = [{'type': 'eq', 'fun': lambda g: np.sum(g) - 1}]
        bounds = [(0, None)] * n
        g0 = np.ones(n) / n
        res = minimize(obj_fun, g0, jac=grad_fun, hess=hess_fun,
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': 200, 'ftol': 1e-12})
        if res.success:
            g_val = np.maximum(res.x, 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass

    return None

def convex_line_search(a, direction, n_evals=12):
    """
    Find optimal t in [0,1] for a_new = (1-t)*a + t*direction.
    Uses the quadratic structure of the convolution.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    d_norm = direction / float(np.sum(direction))

    conv_aa = compute_conv(a_norm)
    conv_ad = compute_conv_cross(a_norm, d_norm)
    conv_dd = compute_conv(d_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ad + t ** 2 * conv_dd
        return float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * d_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    return t_opt, a_new / s

def subgradient_descent(a, time_budget=60.0, verbose=False):
    """
    Projected subgradient method on the simplex for min max_k (a*a)[k].
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    stall = 0

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        # Active set: peaks within 0.1% of max
        active = conv >= max_c * 0.999
        active_sum = active.sum()
        if active_sum == 0:
            active = conv >= max_c * 0.99

        # Subgradient: g[i] = (1/|S|) * sum_{k in S} a[k-i] (valid indices)
        # This is the average of shifted a over active peaks
        grad = np.zeros(n)
        for k in np.where(active)[0]:
            # (a*a)[k] = sum_i a[i]*a[k-i], grad wrt a[j] = a[k-j] (if valid)
            lo_i = max(0, k - (n - 1))
            hi_i = min(k, n - 1)
            if lo_i <= hi_i:
                indices = np.arange(hi_i - lo_i, -1, -1)
                # grad[j] += a[k-j] for j in [lo_i, hi_i]
                # a[k-j] for j in [lo_i, hi_i] means a[k-hi_i] to a[k-lo_i]
                grad[lo_i:hi_i+1] += a[k - hi_i:k - lo_i + 1][::-1]
        grad /= max(active_sum, 1)

        # Center the gradient (removing the sum constraint component)
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it < 500:
                lr = min(0.05, lr * 1.01)
        else:
            stall += 1
            if stall > 50:
                lr *= 0.8
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
                if lr < 1e-6:
                    break

        # Adaptive learning rate decay
        if it > 100:
            lr = max(1e-5, 0.01 / (1.001 ** it))

    return best_a


def smooth_max_descent(a, time_budget=60.0):
    """Smooth max approximation with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = 0.0005
    lr = 0.005
    momentum = 0.85
    v = np.zeros(n)
    stall = 0

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = np.clip((conv - max_c) / T, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        if total < 1e-30:
            break
        weights = exp_vals / total

        # Gradient of smooth max: sum_k weights[k] * a[k-i]
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if weights[k] < 1e-8:
                continue
            lo_i = max(0, k - (n - 1))
            hi_i = min(k, n - 1)
            if lo_i <= hi_i:
                grad[lo_i:hi_i + 1] += weights[k] * a[k - hi_i:k - lo_i + 1][::-1]

        grad_centered = grad - np.mean(grad)
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 40:
                lr *= 0.7
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
                if lr < 1e-7:
                    break

        T = max(0.00005, T * 0.999)

    return best_a


def coordinate_descent_polish(a, time_budget=100.0, delta=1e-4):
    """Coordinate descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    pass_count = 0
    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count + n * 13)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            a_try = a.copy()
            a_try[i] += delta
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st > 1e-10:
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    continue

            if a[i] > delta * 0.5:
                a_try = a.copy()
                a_try[i] -= delta
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st > 1e-10:
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True

        if not improved and pass_count > 2:
            delta *= 0.6
            if delta < 1e-7:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_asymmetric_init(n, style='ramp', rng=None):
    """Create asymmetric initializations."""
    if rng is None:
        rng = np.random.default_rng(42)

    if style == 'ramp':
        a = np.linspace(0.1, 3.0, n)
    elif style == 'geometric':
        a = np.exp(np.linspace(0, 2, n))
    elif style == 'bimodal':
        a = np.exp(-((np.arange(n) - n * 0.3) ** 2) / (2 * (n * 0.1) ** 2)) + \
            np.exp(-((np.arange(n) - n * 0.8) ** 2) / (2 * (n * 0.05) ** 2))
    elif style == 'chirp_abs':
        t = np.arange(n) / n
        a = np.abs(np.cos(2 * np.pi * t ** 2 * 3)) + 0.1
    elif style == 'random_asym':
        a = rng.exponential(scale=1.0, size=n)
        a = a * np.linspace(0.5, 2.0, n)
    elif style == 'lowpass_mod':
        t = np.arange(n) / n
        a = (0.5 + 0.5 * np.cos(2 * np.pi * t)) ** 2 + 0.05
        a = a * np.linspace(0.3, 2.0, n)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: diverse inits + LP/QP alternating + subgradient + multi-n.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={len(parent)}): obj={best_score:.8f}")
    except Exception:
        pass

    n_default = 800
    if best_seq is None:
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    # === Phase 1: LP+QP Alternating on parent ===
    print(f"\n=== Phase 1: LP+QP Alternating (n={len(best_seq)}) ===")
    p1_budget = min(300.0, (deadline - time.monotonic()) * 0.3)
    print(f"  Budget: {p1_budget:.0f}s")
    a1 = lp_qp_alternating(best_seq, time_budget=p1_budget, verbose=True)
    obj1 = compute_obj(a1)
    if obj1 < best_score:
        best_score = obj1
        best_seq = a1.copy()
        print(f"  Phase 1 result: {best_score:.8f}")
    else:
        print(f"  Phase 1 result: {obj1:.8f} (no improvement)")

    # === Phase 2: Subgradient + Smooth max on best ===
    print(f"\n=== Phase 2: Subgradient + Smooth Max ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        sg_budget = min(80.0, remaining * 0.15)
        a2 = subgradient_descent(best_seq, time_budget=sg_budget)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Subgradient result: {best_score:.8f}")
        else:
            print(f"  Subgradient result: {obj2:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            sm_budget = min(60.0, remaining * 0.12)
            a3 = smooth_max_descent(best_seq, time_budget=sm_budget)
            obj3 = compute_obj(a3)
            if obj3 < best_score:
                best_score = obj3
                best_seq = a3.copy()
                print(f"  Smooth max result: {best_score:.8f}")
            else:
                print(f"  Smooth max result: {obj3:.8f}")

    # === Phase 3: Asymmetric restarts + LP+QP ===
    print(f"\n=== Phase 3: Asymmetric Restarts ===")
    n_base = len(best_seq)
    init_styles = ['ramp', 'geometric', 'bimodal', 'chirp_abs', 'random_asym', 'lowpass_mod']
    for style in init_styles:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_init = make_asymmetric_init(n_base, style=style, rng=rng)
        # Run brief LP+QP
        alloc = min(80.0, remaining * 0.12)
        a_opt = lp_qp_alternating(a_init, time_budget=alloc)
        obj_opt = compute_obj(a_opt)

        # If good, polish
        if obj_opt < best_score + 0.01:
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_opt = coordinate_descent_polish(a_opt, time_budget=min(30.0, remaining * 0.2))
                obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** Asymmetric '{style}': {best_score:.8f} **")
        else:
            print(f"  Asymmetric '{style}': {obj_opt:.8f}")

    # === Phase 4: Structured perturbations of best ===
    print(f"\n=== Phase 4: Structured Perturbations ===")
    perturbations = [
        ('low_freq_bump', 0.3),
        ('high_freq_noise', 0.05),
        ('spectral_shift', 0.2),
        ('mirror_blend', 0.1),
    ]
    for pname, psize in perturbations:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_pert = best_seq.copy()
        if pname == 'low_freq_bump':
            t = np.arange(len(a_pert)) / len(a_pert)
            a_pert = a_pert * (1 + psize * np.cos(2 * np.pi * t))
        elif pname == 'high_freq_noise':
            a_pert = a_pert + psize * rng.normal(0, 1, len(a_pert))
        elif pname == 'spectral_shift':
            # Shift in frequency domain
            n = len(a_pert)
            size = 1
            while size < 2 * n:
                size <<= 1
            A = rfft(a_pert, size)
            # Apply a small phase shift
            phase = psize * np.sin(np.arange(size // 2 + 1) * 0.1)
            A_mag = np.abs(A)
            A_phase = np.angle(A) + phase
            A_new = A_mag * np.exp(1j * A_phase)
            a_pert = irfft(A_new, size)[:n]
        elif pname == 'mirror_blend':
            a_mirror = a_pert[::-1]
            a_pert = (1 - psize) * a_pert + psize * a_mirror

        a_pert = np.maximum(a_pert, 0)
        sp = float(np.sum(a_pert))
        if sp > 1e-10:
            a_pert = a_pert / sp
            alloc = min(60.0, remaining * 0.1)
            a_opt = lp_qp_alternating(a_pert, time_budget=alloc)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** Perturb '{pname}': {best_score:.8f} **")
            else:
                print(f"  Perturb '{pname}': {obj_opt:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(400, n_best - 100), max(400, n_best - 50),
        n_best + 50, n_best + 100, n_best + 150,
        600, 700, 750, 800, 850, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.1, 30.0)
        a_opt = lp_qp_alternating(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score + 0.005:
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_opt = coordinate_descent_polish(a_opt, time_budget=min(20.0, remaining * 0.3))
                obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # LP+QP polish
        alloc = min(80.0, remaining * 0.35)
        a_fp = lp_qp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_fp = compute_obj(a_fp)
        if obj_fp < best_score:
            best_score = obj_fp
            best_seq = a_fp.copy()
            print(f"  Final LP+QP: {best_score:.8f}")

        # Coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(50.0, remaining * 0.3)
            a_cd = coordinate_descent_polish(best_seq, time_budget=alloc, delta=5e-5)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final CD: {best_score:.8f}")

        # Smooth max final
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(30.0, remaining * 0.3)
            a_sm = smooth_max_descent(best_seq, time_budget=alloc)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Final SmoothMax: {best_score:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
            print(f"  evaluate_sequence confirmed: {best_score:.8f}")
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
