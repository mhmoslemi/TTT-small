            else:

def subgrad_at_peak(a, k):
    n = len(a)
    g = np.zeros(n)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    if i_min > i_max:
        return g
    j_start = k - i_max
    j_end = k - i_min
    g[i_min:i_max + 1] = 2.0 * a[j_start:j_end + 1][::-1]
    if k % 2 == 0:
        mid = k // 2
        if 0 <= mid < n:
            g[mid] = a[mid]
    return g

def smooth_subgrad(a, conv, topK=3, beta=5.0):
    n = len(a)
    top_idx = np.argsort(conv)[::-1][:topK]
    max_val = conv[top_idx[0]]
    if max_val < 1e-15:
        return np.zeros(n)
    w = np.exp(beta * (conv[top_idx] - max_val) / max_val)
    w /= w.sum()
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        g += wi * subgrad_at_peak(a, int(k))
    return g

def mirror_descent_step(a, g, lr):

Actually, let me think about this more carefully. The objective is:
f(a) = 2n * max_k (sum_i a[i]*a[k-i]) / (sum a)^2

If we normalize sum(a) = 1, it's:
f(a) = 2n * max_k conv(a,a)[k]

The key insight is that this is a **quadratically constrained** problem. The conv values are quadratic in a. The max makes it non-smooth.

For the smooth approximation (log-sum-exp), we have:
f_β(a) = 2n * (1/β) * log(sum_k exp(β * conv[k]))

The gradient of this is:
∇f_β(a) = 2n * sum_k p_k * ∇conv[k]
where p_k = exp(β*conv[k]) / sum_j exp(β*conv[j])

This is a weighted sum of the subgradients, with weights proportional to the softmax of the conv values.

For large β, this concentrates on the maximum. For small β, it's more "spread out" and gives a smoother landscape.

**Key idea: Use β-adaptive optimization**
- Start with small β (smooth, explore)
- Gradually increase β (sharpen, exploit)
- This is like simulated annealing

Let me also think about a **coordinate descent** approach that's more efficient:
- For each coordinate i, find the optimal value a[i] that minimizes the objective, holding all other coordinates fixed.
- This is a 1D optimization problem (since the objective is a max of quadratics in a[i]).
- The max of quadratics is a convex function, so we can find the minimum by ternary search or by finding the breakpoints.

Actually, for a fixed set of other coordinates, the objective as a function of a[i] is:
max_k (c_k + b_k * a[i] + a[i]^2 * d_k)
where c_k, b_k, d_k depend on the other coordinates.

Wait, let me be more careful. conv[k] = sum_j a[j]*a[k-j]. If we fix all coordinates except a[i], then:
conv[k] = (sum of terms not involving a[i]) + a[i]*a[k-i] + a[i]*a[i-k] (with appropriate boundary conditions)
= c_k + a[i] * (a[k-i] + a[i-k]) [with 0 for out-of-bounds]

Wait, that's only linear in a[i], not quadratic! Because each conv[k] involves a[i] at most once (as a factor), since the other factor is a[k-i] which is fixed.

Actually wait: conv[k] = sum_j a[j] * a[k-j]. The term involving a[i] is when j=i (giving a[i]*a[k-i]) or when k-j=i (giving a[k-i]*a[i]). If i = k-i (i.e., 2i=k), then both terms are a[i]^2, so it's quadratic. Otherwise, it's linear in a[i].

So for a given k:
- If 2i ≠ k: conv[k] = c_k + a[i] * a[k-i] (linear in a[i])
- If 2i = k: conv[k] = c_k + a[i]^2 (quadratic in a[i])

The max of linear and one quadratic function is still a convex function of a[i]. We can minimize it by:
1. Finding the point where the quadratic crosses each linear function
2. The minimum is either at the vertex of the quadratic or at one of the crossing points

This gives an O(n^2) per-coordinate update (since there are O(n) conv values and each is either linear or quadratic in a[i]). But with n=800, that's 640K operations per coordinate, and 800 coordinates, so 512M per full sweep. That's too slow.

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """FFT-based linear convolution a*a. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def subgrad_peak(a, k):
    """Exact subgradient of conv(a,a)[k] w.r.t. a. O(n) via slicing."""
    n = len(a)
    g = np.zeros(n)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    if i_min > i_max:
        return g
    j_start = k - i_max
    j_end = k - i_min
    g[i_min:i_max + 1] = 2.0 * a[j_start:j_end + 1][::-1]
    if k % 2 == 0:
        mid = k // 2
        if 0 <= mid < n:
            g[mid] = a[mid]
    return g

    w /= w.sum()
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        g += wi * subgrad_peak(a, int(k))
    return g
