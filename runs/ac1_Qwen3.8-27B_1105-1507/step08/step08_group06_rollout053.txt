"""
Optimized step function search using LP alternating with warm starts,
Barzilai-Borwein subgradient, structured perturbations, and simulated annealing.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import linprog


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective for normalized a (sum=1)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, warm_g=None):
    """
    Solve LP: min M s.t. conv(a, g) <= M, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)

    # Try cvxpy solvers
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    objective = cp.Minimize(M)
    prob = cp.Problem(objective, constraints)

    if warm_g is not None:
        g.value = warm_g

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                    return g_val, M_val
        except Exception:
            continue

    # Fallback: linprog
    try:
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(m), A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(res.x[n])
                return g_val, M_val
    except Exception:
        pass

    return None, None


def lp_alternate(a, time_budget=300.0, warm_g=None, sa_temperature=0.0):
    """
    LP alternating with line search and optional simulated annealing.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a, None
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    streak = 0
    last_g = warm_g

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp(a, warm_g=last_g)
        if g_lp is None:
            break
        last_g = g_lp

        # Line search: try multiple step sizes
        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = best_a.copy()

        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = a + t * (g_lp - a)
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        # Simulated annealing: sometimes accept worse
        accept = best_t_obj < best_obj - 1e-12
        if not accept and sa_temperature > 0:
            delta = best_t_obj - best_obj
            if delta < sa_temperature and np.random.random() < np.exp(-delta / sa_temperature):
                accept = True
                sa_temperature *= 0.95

        if accept:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            streak = 0
        else:
            streak += 1
            if streak >= 15:
                # Structured perturbation
                rng = np.random.default_rng()
                scale = 0.02 + 0.01 * min(streak // 15, 5)
                # Try different perturbation types
                ptype = rng.integers(0, 4)
                if ptype == 0:
                    # Gaussian noise
                    a = best_a * np.exp(rng.normal(0, scale, n))
                elif ptype == 1:
                    # Move mass from peak contributors to others
                    conv = compute_conv(best_a)
                    k_star = int(np.argmax(conv))
                    contrib = np.zeros(n)
                    for j in range(n):
                        kj = k_star - j
                        if 0 <= kj < n:
                            contrib[j] = best_a[j] * best_a[kj]
                    top = np.argsort(contrib)[::-1][:n//4]
                    bot = np.argsort(contrib)[:n//4]
                    delta = scale * best_a[top[0]]
                    a = best_a.copy()
                    a[top[0]] -= delta
                    a[bot[0]] += delta
                elif ptype == 2:
                    # Palindromic smoothing
                    a = 0.5 * (best_a + best_a[::-1])
                    a = a * np.exp(rng.normal(0, scale/2, n))
                else:
                    # Block perturbation
                    block = max(1, n // 20)
                    idx = rng.integers(0, n // block) * block
                    a = best_a.copy()
                    a[idx:idx+block] *= (1 + scale)
                    a = np.maximum(a, 0)

                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                streak = 0

    return best_a, last_g


def subgradient_bb(a, time_budget=150.0):
    """
    Subgradient method with Barzilai-Borwein step size.
    Focuses on active peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    x = a.copy()  # Nesterov point
    beta = 0.8
    momentum = np.zeros(n)
    prev_y = y.copy()
    prev_grad = None
    bb_lr = 0.05

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        # Compute conv at Nesterov point
        conv = compute_conv(x)
        max_val = float(np.max(conv))

        # Active peaks (top k or within threshold)
        threshold = max_val * 0.85
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 20:
            active = np.argsort(conv)[::-1][:20]

        # Subgradient
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            indices = np.arange(i_min, i_max + 1)
            grad[indices] += a[k - indices]
        grad /= len(active)

        # Center (project to sum=0)
        grad = grad - np.mean(grad)

        # Barzilai-Borwein step size
        if prev_grad is not None:
            sy = y - prev_y
            yg = grad - prev_grad
            sy_norm = np.dot(sy, sy)
            yg_norm = np.dot(yg, yg)
            if sy_norm > 1e-14:
                if np.random.random() < 0.5:
                    bb_lr = np.dot(sy, yg) / sy_norm  # BB_s
                else:
                    if yg_norm > 1e-14:
                        bb_lr = np.dot(sy, yg) / yg_norm  # BB_g
                bb_lr = np.clip(bb_lr, 0.001, 0.5)

        prev_y = y.copy()
        prev_grad = grad.copy()

        # Momentum step
        momentum = beta * momentum - bb_lr * grad
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
            s_new = 1.0
        y_new = y_new / s_new

        # Nesterov update
        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        x = y_new + (1 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        y = y_new
        obj = compute_obj(y)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = y.copy()

    return best_a


def coordinate_descent(a, time_budget=60.0):
    """
    Local coordinate search: try small perturbations around current point.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    rng = np.random.default_rng(12345)

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        # Pick a random subset of coordinates to perturb
        k = rng.integers(1, max(2, n // 100))
        idx = rng.integers(0, n, size=k)

        improved = False
        for _ in range(10):
            a_try = best_a.copy()
            for i in idx:
                i = int(i)
                # Try moving mass
                delta = rng.normal(0, 0.01) * best_a[i]
                if delta > 0:
                    # Move mass from i to another position
                    j = rng.integers(0, n)
                    while j == i:
                        j = rng.integers(0, n)
                    delta = min(delta, best_a[i] * 0.5)
                    j = int(j)
                    a_try[i] -= delta
                    a_try[j] += delta
                else:
                    # Move mass to i from another position
                    j = rng.integers(0, n)
                    while j == i:
                        j = rng.integers(0, n)
                    delta = min(-delta, best_a[j] * 0.5)
                    j = int(j)
                    a_try[j] -= delta
                    a_try[i] += delta

            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                improved = True
                break

        if not improved and it % 100 == 99:
            # Larger perturbation
            scale = 0.03
            a_try = best_a * np.exp(rng.normal(0, scale, n))
            a_try = np.maximum(a_try, 0)
            a_try = a_try / np.sum(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use scipy for cubic
    from scipy.interpolate import PchipInterpolator
    try:
        f = PchipInterpolator(x_old, a, extrapolate=False)
        a_new = f(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_initialization(kind, n, rng):
    """Generate diverse initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 5.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'bilateral_exp':
        # Exponential decay from center
        center = n / 2.0
        x = np.arange(n)
        a = np.exp(-0.3 * np.abs(x - center) / (n / 4.0))
        return a / np.sum(a)
    elif kind == 'two_peak':
        a = np.ones(n) * 0.3
        a[n//4] = 2.0
        a[3*n//4] = 2.0
        # Smooth
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=3)
        return a / np.sum(a)
    elif kind == 'palindromic_random':
        half = n // 2
        a = rng.random(half + 1) + 0.01
        if n % 2 == 0:
            a = np.concatenate([a, a[-2::-1]])
        else:
            a = np.concatenate([a, a[-1::-1]])
        return a / np.sum(a)
    else:
        return np.ones(n) / n

"""
Optimized step function search: LP alternating with warm-starting, simulated annealing,
Barzilai-Borwein subgradient, dual-guided perturbations, and aggressive length exploration.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import linprog
from scipy.interpolate import PchipInterpolator
import math


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective assuming a is already normalized (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. conv(a,g) <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if warm_start is not None:
        g_var.value = np.maximum(warm_start, 0)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M_var.value) if M_var.value is not None else float(np.max(A @ g_val))
                    return g_val, M_val
        except Exception:
            continue

    # Fallback: linprog
    try:
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(m), A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(res.x[n])
                return g_val, M_val
    except Exception:
        pass

    return None, None


def lp_alternate(a, time_budget=300.0, warm_start=None, sa_temp=0.0, rng=None):
    """
    LP alternating with adaptive line search and optional simulated annealing.
    Returns (best_a, last_g_warm).
    """
    if rng is None:
        rng = np.random.default_rng()

    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a.copy(), None
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    streak = 0
    last_g = warm_start
    cur_temp = sa_temp

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp(a, warm_start=last_g)
        if g_lp is None:
            break
        last_g = g_lp.copy()

        # Adaptive line search
        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = best_a.copy()

        # Generate step candidates
        steps = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]

                        j = int(bot_idx[rng.integers(0, len(bot_idx))])

def subgradient_bb(a, time_budget=150.0, rng=None):
    """
    Subgradient method with Barzilai-Borwein step size and Nesterov acceleration.
    """
    if rng is None:
        rng = np.random.default_rng()

    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    x = a.copy()
    momentum = np.zeros(n)
    beta = 0.75
    prev_y = y.copy()
    prev_grad = None
    bb_lr = 0.05
    bb_count = 0

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        # Subgradient at Nesterov point, but evaluate at y
        conv = compute_conv(x)
        max_val = float(np.max(conv))
        threshold = max_val * 0.80
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 25:
            active = np.argsort(conv)[::-1][:25]

        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            indices = np.arange(i_min, i_max + 1)
            grad[indices] += y[k - indices]
        grad /= len(active)
        grad = grad - np.mean(grad)  # Project to sum=0

        # BB step size update
        if prev_grad is not None and it > 0:
            sy = y - prev_y
            yg = grad - prev_grad
            denom_s = float(np.dot(sy, sy))
            denom_g = float(np.dot(yg, yg))
            num = float(np.dot(sy, yg))
            if denom_s > 1e-14 and denom_g > 1e-14:
                lr_s = num / denom_s
                lr_g = num / denom_g
                if bb_count % 2 == 0:
                    bb_lr = np.clip(lr_s, 0.001, 0.3) if lr_s > 0 else 0.05
                else:
                    bb_lr = np.clip(lr_g, 0.001, 0.3) if lr_g > 0 else 0.05
                bb_count += 1

        prev_y = y.copy()
        prev_grad = grad.copy()

        # Momentum update
        momentum = beta * momentum - bb_lr * grad
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
            s_new = 1.0
        y_new = y_new / s_new

        # Nesterov
        t_k = 2.0 / (math.sqrt(it + 1.0) + 2.0)
        x = y_new + (1.0 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        y = y_new
        obj = compute_obj(y)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = y.copy()

    return best_a


def coordinate_descent(a, time_budget=60.0, rng=None):
    """
    Local coordinate search with mass redistribution.
    """
    if rng is None:
        rng = np.random.default_rng(99)

    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        # Try a few random mass moves
        improved = False
        for _ in range(8):
            a_try = best_a.copy()
            n_moves = int(rng.integers(1, 4))
            for _ in range(n_moves):
                i = int(rng.integers(0, n))
                if a_try[i] < 1e-8:
                    continue
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                frac = rng.uniform(0.005, 0.08)
                delta = frac * a_try[i]
                a_try[i] -= delta
                a_try[j] += delta

            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                improved = True
                break

        if not improved and it % 50 == 49:
            # Occasional larger perturbation
            scale = 0.02 + 0.01 * min(it // 500, 5)
            a_try = best_a * np.exp(rng.normal(0, scale, n))
            a_try = np.maximum(a_try, 0)
            a_try = a_try / np.sum(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via PCHIP interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    try:
        f = PchipInterpolator(x_old, a, extrapolate=False)
        a_new = f(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_initialization(kind, n, rng):
    """Generate diverse initializations of length n."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1.0 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n, dtype=np.float64)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'bilateral':
        center = n / 2.0
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.4 * np.abs(x - center) / (n / 4.0))
        return a / np.sum(a)
    elif kind == 'palm':
        # Palindromic random
        half = (n + 1) // 2
        a_half = rng.random(half) + 0.01
        if n % 2 == 0:
            a = np.concatenate([a_half, a_half[-2::-1]])
        else:
            a = np.concatenate([a_half, a_half[-2::-1]])
            if len(a) > n:
                a = a[:n]
            elif len(a) < n:
                a = np.concatenate([a, [a[-1]]])
        return a / np.sum(a)
    elif kind == 'hann':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(2 * x))
        a = a + 0.01  # Avoid exact zeros
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating with warm starts + SA, BB subgradient,
    coordinate descent, and aggressive length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    n_base = 800
    if best_seq is None:
        best_seq = np.ones(n_base) / n_base
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_base}): obj={best_score:.8f}")
    else:
        n_base = len(best_seq)

    a = best_seq.copy()

    # === Phase 1: Quick diverse init (max 40s) ===
    print(f"\n=== Phase 1: Diverse Initializations (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    t_p1_start = time.monotonic()
    p1_budget = min(40.0, remaining * 0.03)

    for kind in ['hann', 'palm', 'bilateral', 'random']:
        if time.monotonic() - t_p1_start > p1_budget:
            break
        a_init = generate_initialization(kind, n_base, rng)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            a = a_init.copy()
            print(f"  Init '{kind}': obj={obj_init:.8f} **")
        else:
            print(f"  Init '{kind}': obj={obj_init:.8f}")

    print(f"  After P1: obj={best_score:.8f}")

    # === Phase 2: Main LP Alternating (target ~45% of budget) ===
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(650.0, remaining * 0.45)
    t_p2_start = time.monotonic()

    # Multiple rounds with decreasing SA temperature
    rounds = [
        {'sa': 0.008, 'frac': 0.30},
        {'sa': 0.004, 'frac': 0.25},
        {'sa': 0.002, 'frac': 0.18},
        {'sa': 0.0, 'frac': 0.15},
        {'sa': 0.0, 'frac': 0.12},
    ]

    for ri, cfg in enumerate(rounds):
        if time.monotonic() - t_p2_start > p2_budget:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p2_budget * cfg['frac'], remaining * 0.6)
        print(f"  Round {ri} (SA={cfg['sa']}): {alloc:.0f}s budget")

        a_new, warm = lp_alternate(a, time_budget=alloc,
                                    warm_start=warm if 'warm' in dir() else None,
                                    sa_temp=cfg['sa'], rng=rng)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  ** R{ri}: obj={best_score:.8f} **")
        else:
            print(f"  R{ri}: obj={obj:.8f}")
            if ri > 0:
                scale = 0.015 + 0.005 * ri
                a = best_seq * np.exp(rng.normal(0, scale, n_base))
                a = np.maximum(a, 0)
                a = a / np.sum(a)

    # === Phase 3: BB Subgradient (target ~12%) ===
    print(f"\n=== Phase 3: BB Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_alloc = min(180.0, remaining * 0.12)
        a_sg = subgradient_bb(a, time_budget=p3_alloc, rng=rng)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a = a_sg.copy()
            print(f"  BB: obj={best_score:.8f} **")
        else:
            print(f"  BB: obj={obj_sg:.8f}")

    # === Phase 4: Coordinate Descent (target ~5%) ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p4_alloc = min(80.0, remaining * 0.05)
        a_cd = coordinate_descent(a, time_budget=p4_alloc, rng=rng)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 5: Length Exploration (target ~15%) ===
    print(f"\n=== Phase 5: Length Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p5_budget = min(220.0, remaining * 0.15)
        t_p5_start = time.monotonic()
        n_best = len(best_seq)

        n_candidates = sorted(set([
            max(200, n_best - 200),
            max(200, n_best - 100),
            max(200, n_best - 50),
            n_best + 50,
            n_best + 100,
            n_best + 200,
            n_best + 400,
        ]))
        n_candidates = [n for n in n_candidates if 200 <= n <= 3000]

        for n_try in n_candidates:
            if time.monotonic() - t_p5_start > p5_budget:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            if n_try == n_best:
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try

            alloc_lp = min(30.0, p5_budget * 0.5)
            alloc_sg = min(15.0, p5_budget * 0.2)

            a_opt, _ = lp_alternate(a_try, time_budget=alloc_lp, rng=rng)
            a_opt = subgradient_bb(a_opt, time_budget=alloc_sg, rng=rng)
            obj_try = compute_obj(a_opt)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                a = a_opt.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    a_final = best_seq.copy()
    n_final = len(a_final)

    while remaining > 30:
        alloc = min(remaining * 0.35, 80.0)
        a_new, warm = lp_alternate(a_final, time_budget=alloc,
                                    warm_start=warm if 'warm' in dir() else None,
                                    rng=rng)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_final = a_new.copy()
            n_final = len(a_final)
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            scale = 0.008 + 0.004 * (1.0 - remaining / max(1.0, deadline - start_time))
            a_final = best_seq * np.exp(rng.normal(0, scale, n_final))
            a_final = np.maximum(a_final, 0)
            a_final = a_final / np.sum(a_final)
            print(f"  Final LP: obj={obj:.8f} (perturbing)")

        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

    # Final quick coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.3, 25.0), rng=rng)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # === Final validation and return ===
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
