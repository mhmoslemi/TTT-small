"""
Smooth L-BFGS + LP Alternating optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: uses soft-max smoothed objective with efficient FFT-based gradients,
combined with LP alternating for final polish and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. cross_conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(c1.dual_value), 0)
                    except:
                        pass
                    return z_val, float(M.value), dual_y
        except:
            continue
    
    # Fallback
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n]), None
    except:
        pass
    
    return None, None, None


def smooth_obj_grad(a, T):
    """
    Compute the smooth approximation of the objective and its gradient.
    F_T(a) = 2n/S^2 * (1/T) * log(sum_k exp(T*c_k))
    where c = conv(a,a), S = sum(a).
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf, np.zeros(n)
    
    c = compute_conv(a)  # length 2n-1
    
    # Stable log-sum-exp
    c_max = float(np.max(c))
    exp_shifted = np.exp(T * (c - c_max))
    sum_exp = float(np.sum(exp_shifted))
    log_sum_exp = c_max + np.log(sum_exp) / T
    
    # Softmax weights
    p = exp_shifted / sum_exp
    
    # Cross-correlation: q[i] = sum_j a[j] * p[i+j]
    # Use FFT: q = ifft(conj(FFT(a)) * FFT(p))[:n]
    size = 1
    while size < 3 * n:
        size <<= 1
    
    A_f = rfft(a, size)
    P_f = rfft(p, size)
    corr = irfft(np.conj(A_f) * P_f, size)[:n]
    
    # Objective
    obj = 2.0 * n * log_sum_exp / (S * S)
    
    # Gradient
    # dF/da_i = (2n/S^2) * [2*corr[i] - 2*log_sum_exp * a_i / S]
    #         = (4n/S^2) * corr[i] - (4n*log_sum_exp/S^3) * a_i
    grad = (4.0 * n / (S * S)) * corr - (4.0 * n * log_sum_exp / (S * S * S)) * a
    
    return obj, grad

            val, _ = smooth_obj_grad(x, T)

def coordinate_polish(a, time_budget=60.0):
    """
    Coordinate descent polish: try adjusting each coordinate individually.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    improved = True
    while improved and time.monotonic() - start < time_budget:
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            # Try multiplicative updates
            for factor in [0.7, 0.85, 1.15, 1.3]:
                a_try = a.copy()
                a_try[i] *= factor
                if a_try[i] < 0:
                    continue
                st = float(np.sum(a_try))
                if st < 1e-10:

                if obj_try < best_obj - 1e-13:

                    best_a = a_try.copy()
                    a = a_try
                    improved = True

def pairwise_exchange(a, time_budget=40.0):
    """Pairwise exchange polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for it in range(100):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for _ in range(30):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

The second term goes to 0 as α → ∞, so F is approximately scale-invariant for large T and large α. But it's not exactly scale-invariant.

The true objective F(a) = 2n*max(c)/S^2 IS scale-invariant. But my smooth approximation is not exactly.

1. Normalize a to have sum = 1 at the start
2. At each gradient evaluation, normalize a to have sum = 1 (project onto the hyperplane)
3. Compute the gradient and project it onto the tangent space (orthogonal to the all-ones vector)

The projection of the gradient onto the tangent space is:
grad_proj = grad - (sum(grad)/n) * ones

This ensures that the update stays on the hyperplane sum(a) = 1 (to first order).

```python
def smooth_obj_grad_normalized(a, T):
    """
    Compute smooth objective and projected gradient, assuming sum(a) = 1.
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf, np.zeros(n)
    
    # Normalize
    a = a / S
    
    c = compute_conv(a)
    c_max = float(np.max(c))
    exp_shifted = np.exp(T * (c - c_max))
    sum_exp = float(np.sum(exp_shifted))
    log_sum_exp = c_max + np.log(sum_exp) / T
    p = exp_shifted / sum_exp
    
    # Cross-correlation
    size = 1
    while size < 3 * n:
        size <<= 1
    A_f = rfft(a, size)
    P_f = rfft(p, size)
    corr = irfft(np.conj(A_f) * P_f, size)[:n]
    
    # Objective (with S = 1): F = 2n * log_sum_exp
    obj = 2.0 * n * log_sum_exp
    
    # Gradient (with S = 1): dF/da_i = 4n * corr[i] - 4n * log_sum_exp * a_i
    grad = 4.0 * n * corr - 4.0 * n * log_sum_exp * a
    
    # Project onto tangent space (orthogonal to all-ones)
    grad_mean = float(np.sum(grad)) / n
    grad -= grad_mean
    
    return obj, grad
```
