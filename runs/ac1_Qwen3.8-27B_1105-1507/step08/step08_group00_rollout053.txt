"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses power-sum smoothing, adaptive lag-targeting subgradients, aggressive length exploration,
Fourier-mode plateau breaking, and two-level optimization with LP repair.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def powersum_smoothing_grad(a, p):
    """
    Compute gradient of (sum_k Q_k^p)^(1/p) at a using FFT.
    Returns gradient (length n) and the convolution.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    if max_conv < 1e-15:
        return np.zeros(n), conv
    # Power-sum: F = (sum conv^p)^(1/p), grad F(i) = (1/p) * F^(1-p) * p * sum(conv^p - 1) * dconv/dai
    # Simplified: use softmax-like weights
    scaled = p * (conv / max_conv)
    exp_vals = np.exp(np.minimum(scaled, 50.0))
    w = exp_vals / float(np.sum(exp_vals))
    # Cross-correlation for gradient
    N = 1
    while N < 3 * n:
        N <<= 1
    W = fft(w, N)
    A = fft(a, N)
    g_full = ifft(W * np.conj(A)).real
    g = 2.0 * max_conv * g_full[:n]
    return g, conv


def adaptive_smoothed_descent(a, time_budget=300.0, p_init=8.0, p_end=200.0, verbose=False):
    """
    Smoothed gradient descent with adaptive power-sum parameter and Nesterov momentum.
    Uses projection onto simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 4.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        y = np.maximum(y, 0)

        prev_g = g.copy()

            v = np.zeros(n)

def lag_targeted_subgradient(a, time_budget=150.0, K=8, verbose=False):
    """
    Subgradient method that targets the top-K peak lags with adaptive weights.
    Uses momentum and backtracking.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 3.0 / np.sqrt(n)
    momentum = 0.75
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        top_vals = conv[top_idx]
        max_v = top_vals[0]
        if max_v < 1e-15:
            break

        # Exponential weighting by peak value
        weights = np.exp(3.0 * (top_vals - max_v) / max_v)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 500 == 0:
                lr *= 0.9
                v *= 0.5
            if lr < 1e-12 or stall > 5000:
                break

        it += 1

    return best_a


def build_conv_matrix_row_a(a):
    """
    For LP: given fixed a, build the constraint matrix such that A @ g = conv(a, g).
    Returns A of shape (2n-1, n).
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # a[j] contributes to positions j, j+1, ..., j+n-1 in the convolution
        end = min(2 * n - 1, j + n)
        A[j:end, j] = a[:end - j]
    return A

        best_t_val = last_t
        best_a_local = best_a.copy()

        for t in candidates:
            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        iter_count += 1
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            if verbose and iter_count % 3 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t_val:.2f}")
        else:
            # Perturb
            scale = 0.03 * (1 + iter_count * 0.1)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / max(a.sum(), 1e-10)
            if iter_count > 20:
                break

    return best_a


def fourier_perturb(a, rng, strength=0.1):
    """Perturb a using random Fourier modes to escape local minima."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Random Fourier perturbation
    num_modes = max(1, n // 20)
    freqs = rng.integers(1, n, size=num_modes)
    phases = rng.uniform(0, 2 * np.pi, size=num_modes)
    amps = rng.uniform(0.5, 1.0, size=num_modes) * strength

    x = np.arange(n)
    perturbation = np.zeros(n)
    for f, p, amp in zip(freqs, phases, amps):
        perturbation += amp * np.sin(2 * np.pi * f * x / n + p)

    a_new = a + perturbation
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a
    return a_new / s_new


def coordinate_descent(a, time_budget=100.0, verbose=False):
    """
    Coordinate descent: greedily update pairs of coordinates to reduce the objective.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    it = 0
    while time.monotonic() - start < time_budget:
        # Random pair of coordinates
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Try moving mass from i to j (and vice versa)
        for direction in [1, -1]:
            for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                amount = frac * min(a[i], a[j])
                if direction == 1:
                    a_try = a.copy()
                    a_try[i] -= amount
                    a_try[j] += amount
                else:
                    a_try = a.copy()
                    a_try[i] += amount
                    a_try[j] -= amount

                if a_try[i] < -1e-10 or a_try[j] < -1e-10:
                    continue
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try

                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    break

        it += 1
        if it % 1000 == 0 and verbose:
            print(f"    CD it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation with zero-padding."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    # Map positions
    x_old = np.linspace(0, n - 1, n)
    x_new = np.linspace(0, new_n - 1, new_n)
    a_new = np.interp(x_new, x_old, a, left=0, right=0)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / max(a.sum(), 1e-10)
    elif mode == 2:
        # Two Gaussian bumps
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 10)
        x = np.arange(n)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        a = a / max(a.sum(), 1e-10)
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-0.999, 0.999, n)
        a = 1.0 / np.sqrt(1 - x ** 2)
        a = a / a.sum()
    elif mode == 4:
        # Random low-frequency signal
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        alpha = rng.uniform(-3, 3)
        a = x ** alpha
        a = a / a.sum()
    elif mode == 6:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.3 + 0.7 * np.cos(x) ** 2
        a = a / a.sum()
    elif mode == 7:
        # Sparse: random subset
        k = max(2, n // 5)
        idx = rng.choice(n, size=k, replace=False)
        a = np.zeros(n)
        a[idx] = 1.0
        a = a / a.sum()
    elif mode == 8:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def two_level_optimize(a, time_budget=200.0, verbose=False):
    """
    Two-level optimization: inner loop does tight smoothed descent,
    outer loop does Fourier perturbation + re-optimization.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(555)

    outer_it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 20:
            break

        # Inner loop: smoothed descent
        inner_budget = min(remaining * 0.5, 80.0)
        if inner_budget > 10:
            a_inner = adaptive_smoothed_descent(best_a, time_budget=inner_budget,
                                                p_init=6.0, p_end=150.0,
                                                verbose=verbose and outer_it % 3 == 0)
            obj_inner = compute_obj(a_inner)
            if obj_inner < best_obj - 1e-14:
                best_obj = obj_inner
                best_a = a_inner.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Outer: Fourier perturbation
        if no_improve >= 2 or outer_it % 3 == 0:
            strength = 0.05 + 0.03 * min(no_improve, 5)
            a_pert = fourier_perturb(best_a, rng, strength=strength)
            # Quick optimize the perturbed version
            quick_budget = min(remaining * 0.2, 30.0)
            if quick_budget > 10:
                a_quick = adaptive_smoothed_descent(a_pert, time_budget=quick_budget,
                                                    p_init=4.0, p_end=80.0)
                obj_quick = compute_obj(a_quick)
                if obj_quick < best_obj - 1e-14:
                    best_obj = obj_quick
                    best_a = a_quick.copy()
                    no_improve = 0
                    if verbose:
                        print(f"    TL outer{outer_it}: perturbed improvement obj={best_obj:.8f}")

        outer_it += 1

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization with adaptive strategy switching.
    Phases:
      1. Initial optimization from parent
      2. Length exploration
      3. Two-level optimization with Fourier perturbations
      4. LP repair
      5. Coordinate descent refinement
      6. Final polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    def remaining_time():
        return deadline - time.monotonic()

    # === Phase 1: Initial smoothed descent ===
    print(f"\n=== Phase 1: Initial Smoothed Descent ===")
    if remaining_time() > 60:
        alloc = min(remaining_time() * 0.2, 200.0)
        a1 = adaptive_smoothed_descent(best_seq, time_budget=alloc, p_init=5.0, p_end=200.0, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj1:.8f}")

    # === Phase 2: Subgradient refinement ===
    print(f"\n=== Phase 2: Lag-Targeted Subgradient ===")
    if remaining_time() > 40:
        alloc = min(remaining_time() * 0.1, 100.0)
        a2 = lag_targeted_subgradient(best_seq, time_budget=alloc, K=10, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj2:.8f}")

    # === Phase 3: Length exploration ===
    print(f"\n=== Phase 3: Length Exploration ===")
    n_current = len(best_seq)
    n_options = sorted(set([
        max(200, n_current - 300), max(200, n_current - 150),
        max(200, n_current - 50),
        n_current + 50, n_current + 150, n_current + 300,
        300, 400, 500, 600, 700, 800, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if abs(n - n_current) > 80]
    n_options = sorted(n_options, key=lambda n: abs(n - n_current))[:6]

    for n_try in n_options:
        if remaining_time() < 80:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / max(a_try.sum(), 1e-10)

        # Quick optimize
        alloc1 = min(remaining_time() * 0.15, 60.0)
        if alloc1 > 15:
            a_try = adaptive_smoothed_descent(a_try, time_budget=alloc1, p_init=5.0, p_end=100.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Two-level optimization with Fourier perturbations ===
    print(f"\n=== Phase 4: Two-Level Optimization ===")
    if remaining_time() > 60:
        alloc = min(remaining_time() * 0.25, 300.0)
        a4 = two_level_optimize(best_seq, time_budget=alloc, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4: obj={best_score:.8f} **")
        else:
            print(f"  Phase4: obj={obj4:.8f}")

    # === Phase 5: LP Repair ===
    print(f"\n=== Phase 5: LP Repair ===")
    if remaining_time() > 40:
        alloc = min(remaining_time() * 0.2, 150.0)
        a5 = lp_repair(best_seq, time_budget=alloc, verbose=True)
        obj5 = compute_obj(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Phase5: obj={best_score:.8f} **")
        else:
            print(f"  Phase5: obj={obj5:.8f}")

    # === Phase 6: Coordinate descent (if n is small enough) ===
    print(f"\n=== Phase 6: Coordinate Descent ===")
    n_now = len(best_seq)
    if n_now <= 1000 and remaining_time() > 30:
        alloc = min(remaining_time() * 0.1, 80.0)
        a6 = coordinate_descent(best_seq, time_budget=alloc, verbose=True)
        obj6 = compute_obj(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Phase6: obj={best_score:.8f} **")
        else:
            print(f"  Phase6: obj={obj6:.8f}")

    # === Phase 7: Diverse restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    n_best = len(best_seq)
    for r in range(5):
        if remaining_time() < 60:
            break

        mode = r % 9
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * best_seq + blend * a_r
        a_r = np.maximum(a_r, 0)
        a_r = a_r / max(a_r.sum(), 1e-10)

        alloc = min(remaining_time() * 0.2, 100.0)
        if alloc > 20:
            half = alloc / 2
            a_r = adaptive_smoothed_descent(a_r, time_budget=half, p_init=4.0, p_end=80.0)
            if remaining_time() > 20:
                a_r = lag_targeted_subgradient(a_r, time_budget=min(half, remaining_time() - 25), K=6)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    if remaining_time() > 20:
        alloc = min(remaining_time() * 0.3, 100.0)
        a_final = adaptive_smoothed_descent(best_seq, time_budget=alloc, p_init=20.0, p_end=500.0, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final SGD: obj={best_score:.8f} **")

    if remaining_time() > 20:
        alloc = min(remaining_time() * 0.25, 60.0)
        a_final = lag_targeted_subgradient(best_seq, time_budget=alloc, K=12)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    if remaining_time() > 30:
        alloc = min(remaining_time() * 0.3, 80.0)
        a_final = lp_repair(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation and cleanup
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    try:
        val = evaluate_sequence(result)
        if val < np.inf and val < best_score:
            best_score = val
        elif val < np.inf:
            best_score = val
        else:
            result = [1.0] * 100
            best_score = evaluate_sequence(result)
    except Exception:
        result = [1.0] * 100
        try:
            best_score = evaluate_sequence(result)
        except Exception:
            best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
