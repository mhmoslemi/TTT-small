    for it in range(50):
        active_list = sorted(active)
        g, M = solve_lp(a, active_list)
        if g is None:
            break
        
        # Check all constraints
        conv_ag = compute_conv_cross(a, g)
        violated = np.where(conv_ag > M + tol * max_conv)[0]
        
        if len(violated) == 0:
            return g, M
        
        # Add violated constraints
        active.update(violated.tolist())
        
        # If active set gets too large, just solve full
        if len(active) > 100:
            g, M = solve_lp(a, None)
            if g is not None:
                return g, M
            break
    
    # Fallback: full LP
    return solve_lp(a, None)

def convex_line_search(a, g, n_evals=15):
    """Find optimal t for (1-t)*a + t*g using the quadratic structure."""
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)
    
    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)
    
    def obj_at_t(t):
        val = (1-t)**2 * conv_aa + 2*t*(1-t) * conv_ag + t**2 * conv_gg
        return 2.0 * n * float(np.max(val))
    
    # Since f(t) is convex (max of quadratics), use ternary search
    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if obj_at_t(m1) < obj_at_t(m2):
            hi = m2
        else:
            lo = m1
    
    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    return t_opt, a_new / s

def fw_step(a, n_steps=50, decay=0.999):
    """
    Frank-Wolfe steps: a_{k+1} = (1-t_k) * a_k + t_k * e_{s_k}
    where s_k = argmin_m a_k[peak_k - m] (the coordinate that least contributes to the peak)
    """
    n = len(a)
    a = a / np.sum(a)
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    for step in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient: grad[m] = 2*a[k_star - m] if 0 <= k_star - m < n
        # We want to minimize grad^T s over simplex → s = e_{argmin grad}
        # argmin_m grad[m] = argmin_m a[k_star - m] (for valid m)
        
        # Compute a_shifted[m] = a[k_star - m] for valid m
        # Valid m: 0 <= k_star - m < n, i.e., k_star - (n-1) <= m <= k_star
        m_lo = max(0, k_star - (n - 1))
        m_hi = min(n - 1, k_star)
        
        if m_lo > m_hi:
            break
        
        a_shifted = a[m_lo:m_hi + 1][::-1]  # a[k_star - m] for m in [m_lo, m_hi]
        # Actually: for m = m_lo, k_star - m = k_star - m_lo
        # for m = m_hi, k_star - m = k_star - m_hi
        # We want a[k_star - m] as a function of m
        # m goes from m_lo to m_hi
        # k_star - m goes from k_star - m_lo to k_star - m_hi
        # Since m_lo = max(0, k_star-(n-1)) and k_star - m_lo = min(k_star, n-1)
        # and m_hi = min(n-1, k_star) and k_star - m_hi = max(0, k_star-(n-1))
        
        # So a_shifted for m in [m_lo, m_hi] is a[k_star-m_hi : k_star-m_lo+1]
        # which is a[max(0,k_star-(n-1)) : min(k_star,n-1)+1]
        vals = a[max(0, k_star-(n-1)) : min(k_star, n-1)+1][::-1]
        
        # The argmin over all m (not just valid ones) is the min of vals
        min_idx_local = int(np.argmin(vals))
        s_idx = m_lo + min_idx_local  # Wait, this isn't right
        # Let me recompute: m ranges from m_lo to m_hi
        # For m = m_lo: a[k_star - m_lo]
        # For m = m_lo+1: a[k_star - m_lo - 1]
        # ...
        # For m = m_hi: a[k_star - m_hi]
        # So vals[i] = a[k_star - m_lo - i] for i = 0, ..., m_hi-m_lo
        # We want argmin over i, then s_idx = m_lo + i_min
        
        # Actually let me just compute it directly:
        # For each m in 0..n-1, compute a[k_star-m] if 0<=k_star-m<n, else inf
        grad_vals = np.full(n, np.inf)
        valid_m = np.arange(max(0, k_star-(n-1)), min(n-1, k_star)+1)
        grad_vals[valid_m] = 2.0 * a[k_star - valid_m]
        
        s_idx = int(np.argmin(grad_vals))
        
        # Step size
        t = 2.0 / (step + 2)
        
        a_new = (1 - t) * a + t * np.zeros(n)
        a_new[s_idx] += t
        a_new = a_new / np.sum(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def fw_away_step(a, n_steps=100):
    """
    Modified FW with away steps: can move away from a coordinate
    a_{k+1} = (1 - t - s) * a_k + t * e_s - s * e_j (away from j)
    """
    n = len(a)
    a = a / np.sum(a)
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    for step in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Compute subgradient
        grad_vals = np.full(n, np.inf)
        valid_m = np.arange(max(0, k_star-(n-1)), min(n-1, k_star)+1)
        grad_vals[valid_m] = 2.0 * a[k_star - valid_m]
        
        # FW direction: towards argmin grad
        s_idx = int(np.argmin(grad_vals))
        # Away direction: from argmax grad (among support)
        support = np.where(a > 1e-10)[0]
        if len(support) > 1:
            grad_on_support = grad_vals[support]
            j_idx = support[int(np.argmax(grad_on_support))]
        else:
            j_idx = None
        
        t = 2.0 / (step + 2)
        
        if j_idx is not None and j_idx != s_idx and a[j_idx] > 1e-10:
            # Try away step
            t_away = min(t, a[j_idx] / (a[j_idx] + t) * 0.5)  # Don't go negative
            a_try = a.copy()
            a_try[s_idx] += t
            a_try[j_idx] -= t_away
            if a_try[j_idx] < 0:
                a_try[j_idx] = 0
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    continue
        
        # Regular FW step
        a_new = (1 - t) * a
        a_new[s_idx] += t
        a_new = a_new / np.sum(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def pairwise_cd(a, time_budget=30.0):
    """
    Coordinate descent updating pairs of coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        # Pick a random pair
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        
        # Try to redistribute between i and j
        # a[i] + a[j] = const, optimize the split
        total_ij = a[i] + a[j]
        if total_ij < 1e-10:
            continue
        
        # Try a few splits
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
            a_try = a.copy()
            a_try[i] = total_ij * frac
            a_try[j] = total_ij * (1 - frac)
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break
    
    return best_a

def coordinate_descent(a, time_budget=50.0, delta=1e-4):
    """Standard coordinate descent."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)
    
    for pass_num in range(100):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            for d in [delta, delta*0.1, delta*10, -delta, -delta*0.1]:
                a_try = a.copy()
                a_try[i] += d
                if a_try[i] < 0:
                    continue
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            
            if not improved:
                continue
        
        if not improved and delta > 1e-7:
            delta *= 0.5
    
    return best_a

        # Gradient at k_star: grad[m] = 2*a[k_star-m] if valid
        grad = np.zeros(n)
        valid_m = np.arange(max(0, k_star-(n-1)), min(n-1, k_star)+1)
        grad[valid_m] = 2.0 * a[k_star - valid_m]
        
        improved_any = False
        for b in range(n_blocks):
            i_start = b * block_size
            i_end = min(n, i_start + block_size)
            block_idx = slice(i_start, i_end)
            
            # Current block values
            a_block = a[block_idx].copy()
            block_sum = float(np.sum(a_block))
            
            # LP: min M s.t. grad[k, m] * a_block_new[m] <= M - (conv[k] - grad[k, block] . a_block) for all k

            # Simplification: only use the active constraint (k_star) and sum constraint
            # This gives: min grad_star . a_block_new s.t. sum = block_sum, a_block_new >= 0
            # Solution: put all mass on the coordinate with minimum grad
            
            # But to be more effective, use a few top constraints
            top_k = 5
            conv_sorted = np.argsort(conv)[::-1][:top_k]
            
            # Build small LP
            B_size = i_end - i_start
            A_block = np.zeros((len(conv_sorted), B_size))
            for ii, k in enumerate(conv_sorted):
                valid = np.arange(max(0, k-(n-1)), min(n-1, k)+1)
                # For m in block: k-m should be in [i_start, i_end)
                for local_m in range(B_size):
                    global_m = i_start + local_m
                    if 0 <= k - global_m < n and global_m >= i_start and global_m < i_end:
                        A_block[ii, local_m] = 2.0 * a[k - global_m]
            
            # b_ub = M - (conv[k] - A_block @ a_block) ... but M is variable
            # Actually: A_block @ a_block_new + (conv[k] - A_block @ a_block) <= M
            # => A_block @ a_block_new - M <= -(conv[k] - A_block @ a_block) = A_block @ a_block - conv[k]
            
            # Variables: [a_block_new (B_size), M]
            c = np.zeros(B_size + 1)
            c[-1] = 1.0
            
            A_ub = np.zeros((len(conv_sorted), B_size + 1))
            A_ub[:, :B_size] = A_block
            A_ub[:, -1] = -1.0
            b_ub = A_block @ a_block - conv[conv_sorted]
            
            A_eq = np.zeros((1, B_size + 1))
            A_eq[0, :B_size] = 1.0
            b_eq = np.array([block_sum])
            
            bounds = [(0, None)] * B_size + [(0, None)]
            
            from scipy.optimize import linprog
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            
            if res.status == 0:
                a_block_new = np.maximum(res.x[:B_size], 0)
                s_block = float(np.sum(a_block_new))
                if s_block > 1e-10 and abs(s_block - block_sum) < 1e-6:
                    a_try = a.copy()
                    a_try[block_idx] = a_block_new
                    # Renormalize
                    s_try = float(np.sum(a_try))
                    if s_try > 1e-10:
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved_any = True
        
        if verbose and it % 10 == 0:
            print(f"    Block it{it}: obj={best_obj:.8f}")
        
        if not improved_any:
            # Try smaller blocks or different strategy
            if block_size > 10:
                block_size //= 2
            else:
                break
    
    return best_a
