    Compute gradient of smooth max(conv(a,a)) using FFT-based cross-correlation.
    Returns gradient vector of same length as a.

    shifted = beta * (conv - m)
    weights = np.exp(shifted)

    # Using FFT: g = 2 * ifft(conj(fft(weights, N)) * fft(a, N))[:n]
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    
    W = np.fft.fft(weights, N)
    A = np.fft.fft(a, N)
    g = 2.0 * np.real(np.fft.ifft(np.conj(W) * A)[:n])

Optimized step function search using FFT-based gradient acceleration,
reduced-constraint LP alternating, adaptive mirror descent,
and multi-length exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def fft_cross_correlation(weights, a, N=None):
    """
    Compute cross-correlation: (w ⊛ a)[i] = sum_k w[k] * a[k-i]
    using FFT. Returns array of length len(a).
    """
    n = len(a)
    m = len(weights)
    if N is None:
        N = 1
        while N < m + n - 1:
            N <<= 1
    W = np.fft.fft(weights, N)
    A = np.fft.fft(a, N)
    result = np.real(np.fft.ifft(np.conj(W) * A))[:n]
    return result


def smooth_gradient_fft(a, beta):
    """
    Compute gradient of smooth max(conv(a,a)) using FFT.
    grad[i] = 2 * sum_k w[k] * a[k-i] (approximately)
    where w are softmax weights over convolution values.
    """
    n = len(a)
    conv = compute_conv(a)
    m = conv.max()
    if m < 1e-15:
        return np.zeros(n)
    
    shifted = beta * (conv - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -100, 0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= wsum
    
    # g[i] ≈ 2 * sum_k weights[k] * a[k-i]
    g = 2.0 * fft_cross_correlation(weights, a)
    return g


def solve_reduced_lp(a, top_k=20):
    """
    Solve reduced LP: min M s.t. conv(a,g)[k] <= M for top_k largest k values,
    sum(g) = 1, g >= 0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_a = compute_conv(a_norm)
    
    # Get top_k indices
    top_indices = np.argsort(conv_a)[::-1][:top_k]
    M_target = conv_a[top_indices[0]]  # current max
    
    # Build constraint matrix for only top_k constraints
    # conv(a_norm, g)[k] = sum_j a_norm[k-j] * g[j] for valid j
    A_ub = np.zeros((top_k, n + 1))
    for idx, k in enumerate(top_indices):
        for j in range(n):
            ki = k - j
            if 0 <= ki < n:
                A_ub[idx, j] = a_norm[ki]
        A_ub[idx, n] = -1.0
    
    b_ub = np.zeros(top_k)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 5000})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

def coordinate_descent_v2(a, time_budget=60.0):
    """
    Enhanced coordinate descent: move mass from peak contributors to
    positions that minimize impact on all convolution values.
    """

        M = conv.max()
        k_star = int(np.argmax(conv))

        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib = a[i] * a[j]
                contributors.append((contrib, i, j))
        contributors.sort(reverse=True)
        
        # Find positions with low "exposure" (small sum of a[k-i] for all k near peak)
        exposure = np.zeros(n)
        for i in range(n):

        # Try moving mass from top contributors to low-exposure positions

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
