   import numpy as np
   from scipy.fft import rfft, irfft

   def compute_conv(a):
       """Full linear convolution a*a via FFT."""
       n = len(a)
       if n == 0:
           return np.array([])
       size = 1
       while size < 2 * n - 1:
           size <<= 1
       fa = rfft(a, size)
       return irfft(fa * fa, size)[:2 * n - 1]


   def compute_obj(a):
       """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
       n = len(a)
       if n == 0:
           return np.inf
       s = float(np.sum(a))
       if s < 1e-10:
           return np.inf
       a_n = a / s
       conv = compute_conv(a_n)
       return 2.0 * n * float(np.max(conv))


   def conv_sqrt_init(n):

           src_idx = idx[np.argmax(resp)]
           move = step * a[src_idx]
           
           if move < 1e-15:
               break
           
           # Try moving to several targets
           improved = False
           for tgt in targets[:10]:
               a[src_idx] -= move
               a[tgt] += move
               conv_new = compute_conv(a)
               obj_new = 2.0 * n * conv_new.max()
               if obj_new < best_obj - 1e-12:
                   best_obj = obj_new
                   best_a = a.copy()
                   improved = True
                   break
               else:
                   a[src_idx] += move
                   a[tgt] -= move
           
           if not improved:
               step *= 0.7
               if step < 1e-5:
                   break
           else:
               step = min(0.1, step * 1.05)
           
           it += 1
           if verbose and it % 50 == 0:
               print(f"    PeakShave it{it}: obj={best_obj:.8f}, step={step:.5f}, "
                     f"elapsed={time.monotonic()-start:.1f}s")
       
       return best_a
