        if np.isnan(x) or np.isinf(x):

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of Q_k(a) using FFT cross-correlation.
    F(a) = (1/tau) * log(sum_k exp(tau * Q_k(a)))
    grad F(i) = 2 * sum_k p[k] * a[k-i] where p = softmax(tau * Q)
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    conv = compute_conv(a)

    # Compute softmax weights (with numerical stability)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    # Cross-correlation: g[i] = sum_j p[i+j] * a[j]
    # Use FFT: ifft(fft(p, N) * conj(fft(a, N)))[:n]
    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]

    return g, conv

def subgradient_refine(a, time_budget=100.0, K=5, verbose=False):
    """
    Subgradient method with top-K peaks, momentum, and adaptive step on simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]

        # Weighted subgradient (weight by peak value)
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 5.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.995

        it += 1

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        c1 = rng.integers(0, n // 3)
        c2 = rng.integers(2 * n // 3, n)
        w = max(1, n // 12)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a
