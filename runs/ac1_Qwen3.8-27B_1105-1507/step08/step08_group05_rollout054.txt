def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_cross_conv_matrix(a, K):
    """Build matrix A where A @ g = (a*g)[k] for k in K."""
    n = len(a)
    R = len(K)
    A = np.zeros((R, n))
    for idx, k in enumerate(K):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            A[idx, j] += a[i]
    return A

def solve_active_lp(a, K):
    """Solve min M s.t. (a*g)[k] <= M for k in K, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_cross_conv_matrix(a_n, K)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

                if gs > 1e-10:

        for i in range(i_start, i_end + 1):
            j = k - i
            A[idx, j] += a_n[i]
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_step(a, g, current_obj):
    """Find optimal t in [0,1] for a_new = (1-t)*a + t*g."""
    def obj_at_t(t):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            return np.inf
        a_try = a_try / st
        return compute_obj(a_try)
    try:
        res = minimize_scalar(obj_at_t, bounds=(0.0, 1.0), method='bounded',
                              options={'maxiter': 60, 'xatol': 0.003})
        return res.x if res.success else 0.5
    except Exception:
        return 0.5


def lp_alternate(a0, time_budget=60.0, use_active=False, R=5, verbose=False):
    """LP alternating with optional active-set mode and momentum."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    prev_a = a.copy()

    while time.monotonic() - start < time_budget:
        if use_active:
            conv = compute_conv(a)
            K = np.argsort(conv)[::-1][:R].tolist()
            g_lp, M_lp = solve_active_lp(a, K)
        else:
            g_lp, M_lp = solve_full_lp(a)
        if g_lp is None:
            break

        t = line_search_step(a, g_lp, best_obj)
        a_new = (1.0 - t) * a + t * g_lp
        a_new = np.maximum(a_new, 0)
        st = float(np.sum(a_new))
        if st < 1e-10:
            break
        a_new = a_new / st
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            prev_a = a.copy()
            a = a_new
            stall = 0
            if verbose and stall < 5:
                elapsed = time.monotonic() - start
                print(f"      LP{'A' if use_active else ''} obj={best_obj:.8f} t={t:.3f} M={M_lp:.6f} {elapsed:.0f}s")
        else:
            stall += 1
            if stall > 25:
                break

    return best_a


def remez_exchange(a0, time_budget=60.0, R=5, verbose=False):
    """Remez exchange: alternate active-set LP with active set updates."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        K = np.argsort(conv)[::-1][:R].tolist()

        g_lp, M_lp = solve_active_lp(a, K)
        if g_lp is None:
            break

        t = line_search_step(a, g_lp, best_obj)
        a_new = (1.0 - t) * a + t * g_lp
        a_new = np.maximum(a_new, 0)
        st = float(np.sum(a_new))
        if st < 1e-10:
            break
        a_new = a_new / st
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:

            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"      Remez obj={best_obj:.8f} t={t:.3f} M={M_lp:.6f} {elapsed:.0f}s")
        else:

            if stall > 40:
                break

    return best_a


def nesterov_subgrad(a0, time_budget=60.0, top_k=5, verbose=False):
    """Nesterov-accelerated subgradient on the simplex."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.1
    patience = 0

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                m_start = k - i_end
                m_end = k - i_start
                grad[m_start:m_end + 1] += 2.0 * y[i_start:i_end + 1][::-1]
        grad /= len(top_idx)
        grad = grad - np.mean(grad)

        y_new = y - step_size * grad
        a_new = project_simplex(y_new)
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1.0 - t_k) * (a_new - a)
        y = project_simplex(y)
        a = a_new

        if it % 15 == 0 or it < 10:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 150:
                step_size = max(step_size * 0.5, 0.003)
                patience = 0
        if it % 300 == 299:
            step_size = max(step_size * 0.93, 0.004)

    return best_a


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def poly_parameterize(a, degree):
    """Fit a polynomial to a, return coefficients and basis."""
    n = len(a)
    x = np.linspace(0, 1, n)
    basis = np.vander(x, N=degree + 1, increasing=True)
    coeffs, _, _, _ = np.linalg.lstsq(basis, a, rcond=None)
    return coeffs, x, basis


def poly_eval(coeffs, x, n):
    """Evaluate polynomial at x given n points."""
    basis = np.vander(x, N=len(coeffs), increasing=True)
    return np.maximum(basis @ coeffs, 0)


def poly_optimize(a0, time_budget=60.0, degree=5, verbose=False):
    """Optimize polynomial coefficients to minimize the objective."""
    n = len(a0)
    x = np.linspace(0, 1, n)
    coeffs, _, basis = poly_parameterize(a0, degree)
    best_c = coeffs.copy()
    best_obj = compute_obj(a0)
    start = time.monotonic()

    def obj_fn(c):
        a_try = np.maximum(basis @ c, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return 1e10
        a_try = a_try / s
        return compute_obj(a_try)

    def grad_fn(c):
        a_try = np.maximum(basis @ c, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return np.zeros(len(c))
        a_try = a_try / s
        conv = compute_conv(a_try)
        k_max = int(np.argmax(conv))
        # Subgradient of max conv[k_max]
        grad_conv = np.zeros(n)
        i_start = max(0, k_max - n + 1)
        i_end = min(n - 1, k_max)
        if i_start <= i_end:
            m_start = k_max - i_end
            m_end = k_max - i_start
            grad_conv[m_start:m_end + 1] = 2.0 * a_try[i_start:i_end + 1][::-1]
        # Chain rule: d a_try / d c = basis (approximately, ignoring the max(0,.) and normalization)
        grad_c = basis.T @ grad_conv
        return grad_c / s

    # Use Nelder-Mead for the low-dimensional optimization
    best_obj_local = obj_fn(coeffs)
    for restart in range(3):
        if time.monotonic() - start > time_budget:
            break
        c_start = coeffs + np.random.randn(degree + 1) * 0.01
        remaining = max(5, time_budget - (time.monotonic() - start))
        try:
            res = minimize(obj_fn, c_start, method='Nelder-Mead',
                           options={'maxiter': 2000, 'fatol': 1e-10, 'xatol': 1e-8,
                                    'adaptive': True},
                           timeout=remaining)
            if res.fun < best_obj_local:
                best_obj_local = res.fun
                best_c = res.x.copy()
                a_new = np.maximum(basis @ best_c, 0)
                s = float(np.sum(a_new))
                if s > 1e-10:
                    a_new = a_new / s
                    o = compute_obj(a_new)
                    if o < best_obj:
                        best_obj = o
                        if verbose:
                            print(f"      Poly: obj={best_obj:.8f} (degree={degree})")
        except Exception:
            pass
        coeffs = best_c

    a_final = np.maximum(basis @ best_c, 0)
    s = float(np.sum(a_final))
    if s < 1e-10:
        return a0.copy()
    return a_final / s


def smooth_max_descent(a0, time_budget=60.0, beta=50.0, verbose=False):
    """Gradient descent on smooth max approximation of the objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.02

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        # Softmax weights
        shifted = conv - np.max(conv)
        w = np.exp(beta * shifted)
        w /= np.sum(w)

        # Gradient: grad[m] = 4n * sum_k w[k] * a[k-m] * [valid]
        # Use FFT for efficiency
        # w is length 2n-1, a is length n
        # grad[m] = 4n * sum_{k} w[k] * a[k-m] for valid k-m
        # This is a cross-correlation
        size = 1
        while size < 3 * n:
            size <<= 1
        fw = rfft(w, size)
        fa = rfft(a, size)
        # Cross-correlation: sum_k w[k] a[k-m] = (w * reverse(a))[m + n - 1]
        # Actually let's just compute it directly for simplicity with n <= 3000
        grad = np.zeros(n)
        for m in range(n):
            k_min = max(0, m)
            k_max = min(2 * n - 2, m + n - 1)
            if k_min <= k_max:
                grad[m] = 4.0 * n * np.dot(w[k_min:k_max + 1], a[k_min - m:k_max - m + 1])

        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        a = a_new

        if it % 20 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            else:
                lr = max(lr * 0.9, 0.001)
        if it % 200 == 199:
            lr = max(lr * 0.95, 0.001)

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Remez-exchange LP alternating + polynomial parameterization
    + n-sweep + smooth-max descent. Targets < 1.5027.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = hann_window(800)
        best_score = compute_obj(best_seq)
        print(f"Hann init (n=800): obj={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: Quick diverse initialization (60s) ===
    print(f"\n=== Phase 1: Diverse Initializations ===")
    inits = [best_seq.copy()]
    inits.append(hann_window(n_base))
    # Gaussian
    gauss = np.exp(-0.5 * ((np.arange(n_base) - n_base / 2.0) / (n_base / 4.5)) ** 2)
    gauss = np.maximum(gauss, 0)
    inits.append(gauss / np.sum(gauss))
    # Asymmetric
    asym = np.exp(-((np.arange(n_base)) / (n_base * 0.25)) ** 2)
    asym = np.maximum(asym, 0)
    inits.append(asym / np.sum(asym))
    # Triangular
    tri = np.minimum(np.arange(n_base) + 1.0, n_base - np.arange(n_base))
    inits.append(tri / np.sum(tri))
    # Raised cosine
    rc = 0.5 * (1.0 + np.cos(np.pi * np.arange(n_base) / n_base))
    rc = np.maximum(rc, 1e-10)
    inits.append(rc / np.sum(rc))
    # Power law edge decay
    center = n_base / 2.0
    pw = np.power(1.0 - np.abs(np.arange(n_base) - center) / center, 1.5)
    pw = np.maximum(pw, 0)
    inits.append(pw / np.sum(pw))

    for i, init in enumerate(inits):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        obj = compute_obj(init)
        if obj < best_score:
            best_score = obj
            best_seq = init.copy()
            print(f"  Init {i}: obj={best_score:.8f} **")
        else:
            print(f"  Init {i}: obj={obj:.8f}")
        if remaining > 30:
            alloc = min(10.0, remaining * 0.2)
            a_opt = nesterov_subgrad(init, time_budget=alloc, top_k=5)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Init {i}+Nest: obj={best_score:.8f} **")

    # === Phase 2: Main LP alternating (500s) ===
    print(f"\n=== Phase 2: LP Alternating (Full) ===")
    a = best_seq.copy()
    n_cur = len(a)
    lp_rounds = 10
    for r in range(lp_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(50.0, remaining * 0.25)
        a_new = lp_alternate(a, time_budget=alloc, use_active=False, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new
            print(f"  LP-full R{r}: obj={best_score:.8f} **")
        else:
            print(f"  LP-full R{r}: obj={obj:.8f}")
            # Perturb
            a = best_seq * np.exp(rng.normal(0, 0.02, n_cur))
            a = np.maximum(a, 0)
            a /= np.sum(a)

    # === Phase 3: Remez exchange (200s) ===
    print(f"\n=== Phase 3: Remez Exchange ===")
    for R_val in [3, 5, 8]:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(60.0, remaining * 0.3)
        a_rem = remez_exchange(best_seq, time_budget=alloc, R=R_val, verbose=True)
        obj_rem = compute_obj(a_rem)
        if obj_rem < best_score - 1e-12:
            best_score = obj_rem
            best_seq = a_rem.copy()
            print(f"  Remez R={R_val}: obj={best_score:.8f} **")
        else:

    # === Phase 4: Polynomial parameterization (120s) ===
    print(f"\n=== Phase 4: Polynomial Parameterization ===")

        alloc = min(30.0, remaining * 0.3)
        a_poly = poly_optimize(best_seq, time_budget=alloc, degree=deg, verbose=True)
        obj_poly = compute_obj(a_poly)
        if obj_poly < best_score - 1e-12:
            best_score = obj_poly
            best_seq = a_poly.copy()
            print(f"  Poly deg={deg}: obj={best_score:.8f} **")
        else:
            print(f"  Poly deg={deg}: obj={obj_poly:.8f}")

    # === Phase 5: n-sweep (150s) ===
    print(f"\n=== Phase 5: n-Sweep ===")
    n_options = sorted(set([400, 500, 600, 700, 800, 900, 1000, 1200, 1500, 2000,
                            n_base, max(200, n_base - 150), min(3000, n_base + 150)]))
    n_options = [n for n in n_options if 200 <= n <= 2500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)
        alloc = min(12.0, remaining * 0.15)
        a_opt = lp_alternate(a_try, time_budget=alloc, use_active=True, R=5)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score - 1e-12:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Smooth-max descent (80s) ===
    print(f"\n=== Phase 6: Smooth-Max Descent ===")
    for beta in [30, 80, 200]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(25.0, remaining * 0.3)
        a_sm = smooth_max_descent(best_seq, time_budget=alloc, beta=float(beta), verbose=True)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score - 1e-12:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  SmoothMax beta={beta}: obj={best_score:.8f} **")
        else:
            print(f"  SmoothMax beta={beta}: obj={obj_sm:.8f}")

    # === Phase 7: Final LP + Remez polish (200s) ===
    print(f"\n=== Phase 7: Final Polish ===")
    n_cur = len(best_seq)
    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        use_act = (r % 2 == 0)
        R_val = 5 if use_act else 5
        alloc = min(35.0, remaining * 0.3)
        if use_act:
            a_new = remez_exchange(best_seq, time_budget=alloc, R=R_val, verbose=True)
        else:
            a_new = lp_alternate(best_seq, time_budget=alloc, use_active=False, verbose=True)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Polish R{r}{'A' if use_act else ''}: obj={best_score:.8f} **")
        else:
            print(f"  Polish R{r}{'A' if use_act else ''}: obj={obj_new:.8f}")
            best_seq = best_seq * np.exp(rng.normal(0, 0.01, n_cur))
            best_seq = np.maximum(best_seq, 0)
            best_seq /= np.sum(best_seq)

    # === Phase 8: Random restarts with LP (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 8: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(30):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            scale = 0.01 + 0.003 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr < 1e-10:
                continue
            a_r = a_r / sr
            alloc = min(remaining * 0.25, 12.0)
            a_opt = lp_alternate(a_r, time_budget=alloc * 0.5, use_active=True, R=5)
            a_opt = remez_exchange(a_opt, time_budget=alloc * 0.5, R=5)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score - 1e-12:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                if r % 5 == 0:
                    print(f"  Restart {r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
