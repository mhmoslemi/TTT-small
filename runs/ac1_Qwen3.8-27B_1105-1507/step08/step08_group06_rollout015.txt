import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_smoothed_grad(b, p):
    """
    Compute gradient of smoothed objective.
    b: sequence (length n), normalized
    p: weights over lags (length 2n-1), from softmax
    Returns gradient (length n) of g_epsilon(b) = sum_k p_k * f_k(b)
    """
    n = len(b)
    # grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * b[j]
    # This is cross-correlation: (p_rev * b)[i + n - 1]
    p_rev = p[::-1]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F_p = rfft(p_rev, size)
    F_b = rfft(b, size)
    conv = irfft(F_p * F_b, size)[:2 * n - 1]
    # Take indices n-1 to 2n-3 (inclusive) => n elements
    return 2.0 * conv[n-1:n-1+n]

def softmax_weights(conv_vals, epsilon):
    """Compute softmax weights over convolution values with temperature epsilon."""
    # Use numerically stable softmax
    shifted = conv_vals / epsilon - np.max(conv_vals) / epsilon
    # Actually: p_k = exp(f_k / epsilon) / sum exp(f_j / epsilon)
    # = exp(f_k/epsilon - max/epsilon) / sum exp(f_j/epsilon - max/epsilon)
    m = np.max(conv_vals)
    shifted = (conv_vals - m) / epsilon
    exp_vals = np.exp(shifted)
    return exp_vals / np.sum(exp_vals)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Binary search for the threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def resample_sequence(a, new_n):

    if n == 1:

    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        grad_centered = grad_y - np.mean(grad_y)

        t_old = t
        y_new = project_simplex(y - lr * grad_centered)
        t = (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old)) / 2.0
        x_new = x + (t_old - 1.0) / t * (y_new - x)
        x_new = project_simplex(x_new)
        x = x_new
        y = y_new
        iter_count += 1

        if iter_count % 10 == 0:
            obj = compute_obj(y)
            if obj < best_obj:
                best_obj = obj
                best_b = y.copy()

        # Decay epsilon
        if iter_count % 100 == 0:
            new_eps = max(epsilon * 0.9, eps_end)
            if abs(new_eps - epsilon) > 1e-8:
                epsilon = new_eps
                t = 1.0
                x = y.copy()

        # Slowly increase lr
        if iter_count % 500 == 0:
            lr = min(lr * 1.02, 2.0 / np.sqrt(n))

        if verbose and iter_count % 1000 == 0:
            elapsed = time.monotonic() - start
            obj = compute_obj(y)
            print(f"      FISTA it={iter_count}, obj={obj:.8f}, best={best_obj:.8f}, "
                  f"eps={epsilon:.5f}, lr={lr:.4f}, time={elapsed:.0f}s")

    obj = compute_obj(y)
    if obj < best_obj:
        best_obj = obj
        best_b = y.copy()
    return best_b


def active_peak_descent(b_init, time_budget=100.0, verbose=False):
    """
    Gradient descent focused on reducing the top autocorrelation peaks.
    """
    n = len(b_init)
    b = np.maximum(b_init.astype(np.float64), 0).copy()
    s = float(np.sum(b))
    if s < 1e-12:
        return b
    b = b / s

    best_b = b.copy()
    best_obj = compute_obj(b)
    start = time.monotonic()
    lr = 0.5 / np.sqrt(n)
    momentum = np.zeros(n)
    beta = 0.85
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(b)
        max_val = float(np.max(conv))
        threshold = max_val * 0.92
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 40:
            active = np.argsort(conv)[::-1][:40]

        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            idx = np.arange(i_min, i_max + 1)
            grad[idx] += 2.0 * b[k - idx]
        grad /= len(active)
        grad -= np.mean(grad)

        momentum = beta * momentum - lr * grad
        b_new = project_simplex(b + momentum)

        obj_new = compute_obj(b_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_b = b_new.copy()
        b = b_new
        it += 1

        if it % 200 == 199:
            lr = max(lr * 0.97, 0.002)
            beta = min(beta * 1.005, 0.95)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"      APD it={it}, obj={best_obj:.8f}, lr={lr:.5f}, time={elapsed:.0f}s")

    return best_b


def make_init(kind, n, rng):
    """Generate initialization of specified kind and length."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = (n - 1) / 2.0
        x = np.arange(n, dtype=float)
        sigma = n / 5.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'parabolic':
        x = np.linspace(-1, 1, n)
        a = 1.0 - x ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'alternating':
        a = np.ones(n)
        a[1::2] = 0.4
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'random_smooth':
        a = rng.random(n)
        ks = max(3, n // 40)
        if ks % 2 == 0:
            ks += 1
        kernel = np.exp(-0.5 * ((np.arange(ks) - ks // 2) / (ks // 6.0 + 0.1)) ** 2)
        kernel /= kernel.sum()
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 10)
        a[:edge] = 0.3
        a[-edge:] = 0.3
        return a / np.sum(a)
    elif kind == 'double_hump':
        x = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (x - 0.3) ** 2) + np.exp(-4.0 * (x - 0.7) ** 2) + 0.05
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search: diverse inits -> FISTA-LSE (main) -> active-peak descent
    -> FISTA polish -> length sweep -> final active-set refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Start: n={n_parent}, score={best_score:.8f}, time_budget={budget_s}s")

    # === Phase 1: Diverse Initializations (30s) ===
    print("\n=== Phase 1: Initializations ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        inits = {}
        for kind in ['cosine', 'sine', 'gaussian', 'parabolic', 'alternating',
                     'random', 'random_smooth', 'plateau', 'double_hump']:
            inits[kind] = make_init(kind, n_parent, rng)
        for name, a_init in inits.items():
            obj_init = compute_obj(a_init)
            flag = " **" if obj_init < best_score else ""
            print(f"  {name:15s}: {obj_init:.8f}{flag}")
            if obj_init < best_score:
                best_score = obj_init
                best_seq = a_init.copy()
        print(f"  Best after init: {best_score:.8f}")

    # === Phase 2: FISTA-LSE Main Optimization (600s) ===
    print("\n=== Phase 2: FISTA-LSE ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Multiple rounds with different epsilon schedules
        schedules = [
            (0.15, 0.01, 0.35),
            (0.05, 0.003, 0.30),
            (0.02, 0.001, 0.20),
            (0.008, 0.0005, 0.15),
        ]
        a_work = best_seq.copy()
        for idx, (eps_s, eps_e, frac) in enumerate(schedules):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            budget = min(remaining * frac, 300.0)
            print(f"  Round {idx}: eps {eps_s}->{eps_e}, budget={budget:.0f}s")
            a_new = fista_lse(a_work, time_budget=budget,
                              eps_start=eps_s, eps_end=eps_e, verbose=(idx < 2))
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                a_work = a_new.copy()
                print(f"  ** FISTA R{idx}: {best_score:.8f}")
            else:
                print(f"  FISTA R{idx}: {obj:.8f}")
                a_work = best_seq * np.exp(rng.normal(0, 0.015, n_parent))
                a_work = np.maximum(a_work, 0)
                a_work = a_work / np.sum(a_work)

    # === Phase 3: Active Peak Descent (200s) ===
    print("\n=== Phase 3: Active Peak Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        budget = min(remaining * 0.2, 150.0)
        a_act = active_peak_descent(best_seq, time_budget=budget, verbose=True)
        obj_act = compute_obj(a_act)
        if obj_act < best_score:
            best_score = obj_act
            best_seq = a_act.copy()
            print(f"  ** Active: {best_score:.8f}")
        else:
            print(f"  Active: {obj_act:.8f}")

    # === Phase 4: FISTA Polish from Best (150s) ===
    print("\n=== Phase 4: FISTA Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for attempt in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            budget = min(remaining * 0.25, 80.0)
            scale = 0.008 * (attempt + 1)
            a_start = best_seq * np.exp(rng.normal(0, scale, n_parent))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / np.sum(a_start)
            a_new = fista_lse(a_start, time_budget=budget,
                              eps_start=0.01, eps_end=0.0005, verbose=False)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  ** Polish {attempt}: {best_score:.8f}")
            else:
                print(f"  Polish {attempt}: {obj:.8f}")

    # === Phase 5: Length Sweep (200s) ===
    print("\n=== Phase 5: Length Sweep ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        n_best = len(best_seq)
        n_options = sorted(set([
            max(200, n_best - 100), max(200, n_best - 50),
            n_best,
            n_best + 50, n_best + 100, n_best + 200, n_best + 400,
        ]))
        n_options = [n for n in n_options if 200 <= n <= 5000]

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            if n_try == n_best:
                a_try = best_seq.copy()
            else:
                a_try = resample(best_seq, n_try)
            budget_lp = min(remaining * 0.3, 60.0)
            budget_apd = min(remaining * 0.15, 30.0)

            a_opt = fista_lse(a_try, time_budget=budget_lp,
                              eps_start=0.03, eps_end=0.002, verbose=False)
            if budget_apd > 5:
                a_opt = active_peak_descent(a_opt, time_budget=budget_apd, verbose=False)
            obj_try = compute_obj(a_opt)
            flag = " **" if obj_try < best_score else ""
            print(f"  n={n_try}: {obj_try:.8f}{flag}")
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()

    # === Phase 6: Final Active Peak + FISTA (remaining) ===
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_final = len(best_seq)
        a_f = best_seq.copy()
        while remaining > 15:
            budget = min(remaining * 0.4, 50.0)
            a_new = active_peak_descent(a_f, time_budget=budget, verbose=False)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                a_f = a_new.copy()
                print(f"  Final APD: {best_score:.8f} **")
            else:
                a_f = best_seq * np.exp(rng.normal(0, 0.01, n_final))
                a_f = np.maximum(a_f, 0)
                a_f = a_f / np.sum(a_f)
                print(f"  Final APD: {obj:.8f} (perturbing)")
            remaining = deadline - time.monotonic()

            # Also try a quick FISTA
            if remaining > 15:
                budget2 = min(remaining * 0.3, 30.0)
                a_new2 = fista_lse(a_f, time_budget=budget2,
                                   eps_start=0.005, eps_end=0.0005, verbose=False)
                obj2 = compute_obj(a_new2)
                if obj2 < best_score:
                    best_score = obj2
                    best_seq = a_new2.copy()
                    a_f = a_new2.copy()
                    print(f"  Final FISTA: {best_score:.8f} **")
                remaining = deadline - time.monotonic()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
