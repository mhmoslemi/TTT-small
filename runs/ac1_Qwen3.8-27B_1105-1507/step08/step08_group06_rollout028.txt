    w: length 2n-1 (softmax weights of convolution)
    Returns grad: length n

            tau = max(tau * 0.6, 3.0)
            eta = min(eta * 1.5, 0.4)
            stall_count = 0

    return best_a


def lse_multi_scale(a, time_budget=400.0, verbose=True):
    """Run LSE mirror descent at multiple temperature scales."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    scales = [
        (3.0, 20.0, 0.2, 0.35),
        (5.0, 50.0, 0.15, 0.30),
        (10.0, 80.0, 0.10, 0.25),
        (20.0, 120.0, 0.06, 0.20),
    ]

    per_scale = time_budget / len(scales)

    for si, (t_init, t_max, e_init, e_max) in enumerate(scales):
        remaining = float(deadline_time() - 0) - (time.monotonic() - start)
        if time.monotonic() - start > time_budget - 10:
            break
        alloc = min(per_scale, time_budget - (time.monotonic() - start) - 10)
        if alloc < 15:
            break

        if verbose:
            print(f"  LSE scale {si}: tau=[{t_init},{t_max}], eta=[{e_init},{e_max}], budget={alloc:.0f}s")

        a_scaled = lse_mirror_descent(a, time_budget=alloc, verbose=False,
                                      tau_init=t_init, tau_max=t_max, eta_init=e_init)
        obj = compute_obj(a_scaled)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_scaled.copy()
            a = a_scaled.copy()
            if verbose:
                print(f"    ** LSE scale {si}: obj={best_obj:.8f} **")
        else:
            if verbose:
                print(f"    LSE scale {si}: obj={obj:.8f}")

    return best_a


def spectral_flatten(a, time_budget=100.0, verbose=True):
    """Iteratively flatten the power spectrum to reduce autocorrelation peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n:
        N <<= 1
    N *= 2

    eta = 0.4
    rng = np.random.default_rng(999)

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        P = fft(a, N)
        power = np.abs(P) ** 2

        power_clipped = np.maximum(power, 1e-18)
        log_power = np.log(power_clipped)
        target_log = float(np.mean(log_power))

        ratio = np.exp((target_log - log_power) * eta)
        ratio = np.clip(ratio, 0.3, 3.0)

        P_new = P * ratio
        a_new = ifft(P_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        if it % 15 == 0:
            obj = compute_obj(a)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                if verbose and it % 150 == 0:
                    elapsed = time.monotonic() - start
                    print(f"    Spec: it={it}, obj={best_obj:.8f}, eta={eta:.3f}, t={elapsed:.0f}s")

        if it % 100 == 99:
            eta = max(eta * 0.97, 0.05)

        if it % 500 == 499:
            obj_now = compute_obj(a)
            if obj_now > best_obj * 1.005:
                a = best_a * np.exp(rng.normal(0, 0.04, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                eta = min(eta * 1.5, 0.6)

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_response(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                    return g_val, M_val
        except Exception:
            continue

    try:
        from scipy.optimize import linprog
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(res.x[n])
                return g_val, M_val
    except Exception:
        pass

    return None, None


def lp_alternating(a, time_budget=300.0, verbose=True, beta=0.5):
    """LP alternating with momentum and line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = np.zeros(n)
    last_t = 0.4
    stall = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_response(a)
        if g_lp is None:
            break

        direction = g_lp - a
        momentum = beta * momentum + (1 - beta) * direction
        momentum = momentum - float(np.mean(momentum))

        step_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if 0.05 < last_t < 1.0:
            step_candidates = sorted(set(step_candidates + [last_t * 0.7, last_t * 1.3]))
            step_candidates = [t for t in step_candidates if 0.03 <= t <= 1.5]

        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()

        for t in step_candidates:
            a_try = a + t * momentum
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose and (int(time.monotonic() - start) % 10 < 1):
                elapsed = time.monotonic() - start
                print(f"    LP: obj={best_obj:.8f}, t={best_t_val:.3f}, b={beta}, time={elapsed:.0f}s")
        else:
            stall += 1
            momentum *= 0.5
            if stall % 8 == 0:
                scale = 0.02 + 0.01 * min(stall // 8, 8)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
                last_t = 0.5

    return best_a


def active_set_qp(a, time_budget=150.0, verbose=True, K=30):
    """
    Active-set QP: minimize sum of top-K conv values.
    H[i,j] = count of active lags k with i+j=k.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(55555)

    for outer in range(50):
        if time.monotonic() - start > time_budget:
            break

        c = compute_conv(a)
        active = np.argsort(c)[::-1][:K]

        lag_count = np.zeros(2 * n - 1)
        for k in active:
            lag_count[int(k)] += 1.0

        H = cp.Variable((n, n), symmetric=True)
        p = cp.Variable(n, nonneg=True)

        H_expr = 0
        for i in range(n):
            for j in range(n):
                if lag_count[i + j] > 0:
                    H_expr += lag_count[i + j] * p[i] * p[j]

        constraints = [cp.sum(p) == 1]
        prob = cp.Problem(cp.Minimize(H_expr), constraints)

        solved = False
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if p.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    p_val = np.maximum(p.value, 0)
                    ps = float(np.sum(p_val))
                    if ps > 1e-10:
                        p_val = p_val / ps
                        obj_try = compute_obj(p_val)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = p_val.copy()
                            a = p_val.copy()
                            if verbose:
                                elapsed = time.monotonic() - start
                                print(f"    QP: outer={outer}, obj={best_obj:.8f}, K={K}, t={elapsed:.0f}s")
                        solved = True
                        break
            except Exception:
                continue

        if not solved:
            break

        c_new = compute_conv(a)
        new_active = np.argsort(c_new)[::-1][:K]
        if set(new_active.tolist()) == set(active.tolist()):
            break

    return best_a


def subgradient_polish(a, time_budget=80.0, verbose=True):
    """Nesterov subgradient on active peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    x = a.copy()
    beta_mom = 0.7
    lr = 0.04

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(x)
        max_val = float(np.max(conv))
        threshold = max_val * 0.85
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 25:
            active = np.argsort(conv)[::-1][:25]

        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += y[k - indices]
        grad /= len(active)

        grad = grad - float(np.mean(grad))

        momentum_vec = beta_mom * momentum_vec - lr * grad if it > 0 else -lr * grad
        y_new = y + momentum_vec
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
        else:
            y_new = y_new / s_new

        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        x = y_new + (1 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx
        y = y_new

        if it % 10 == 0:
            obj = compute_obj(y)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = y.copy()

        if it % 200 == 199:
            lr = max(lr * 0.9, 0.002)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = float(np.sum(a_new))
    if s > 1e-12:
        a_new = a_new / s
    return np.maximum(a_new, 0)

    return 0.0


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LSE mirror descent, spectral flattening,
    LP alternating, and active-set QP.
    """
    global _deadline
    start_time = time.monotonic()
    _deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        x = np.linspace(0, np.pi, n_init)
        best_seq = 0.5 * (1 + np.cos(x))
        best_seq = best_seq / np.sum(best_seq)
        best_score = compute_obj(best_seq)
        print(f"Default cosine init (n={n_init}): obj={best_score:.8f}")

    a = best_seq.copy()
    n_cur = len(a)
    print(f"Starting: n={n_cur}, obj={best_score:.8f}")

    # === Phase 1: LSE Multi-Scale Mirror Descent (450s) ===
    remaining = _deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 1: LSE Multi-Scale Mirror Descent ===")
        alloc = min(450.0, remaining * 0.35)
        a_lse = lse_multi_scale(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lse)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_lse.copy()
            a = a_lse.copy()
            print(f"  ** After LSE: obj={best_score:.8f} **")
        else:
            print(f"  After LSE: obj={obj:.8f}")

    # === Phase 2: Spectral Flattening (120s) ===
    remaining = _deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 2: Spectral Flattening ===")
        alloc = min(120.0, remaining * 0.12)
        a_spec = spectral_flatten(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_spec)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_spec.copy()
            a = a_spec.copy()
            print(f"  ** After Spectral: obj={best_score:.8f} **")
        else:
            print(f"  After Spectral: obj={obj:.8f}")

    # === Phase 3: LP Alternating (450s) ===
    remaining = _deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: LP Alternating ===")
        alloc_total = min(450.0, remaining * 0.38)
        betas = [0.4, 0.6, 0.7, 0.85]
        per_beta = alloc_total / len(betas)

        for bi, beta in enumerate(betas):
            remaining = _deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(per_beta, remaining - 20)
            if alloc < 15:
                break
            print(f"  LP beta={beta}: budget={alloc:.0f}s")
            a_lp = lp_alternating(a, time_budget=alloc, verbose=True, beta=beta)
            obj = compute_obj(a_lp)
            if obj < best_score - 1e-12:
                best_score = obj
                best_seq = a_lp.copy()
                a = a_lp.copy()
                print(f"  ** LP beta={beta}: obj={best_score:.8f} **")
            else:
                print(f"  LP beta={beta}: obj={obj:.8f}")
                a = best_seq * np.exp(rng.normal(0, 0.03, n_cur))
                a = np.maximum(a, 0)
                a = a / np.sum(a)

    # === Phase 4: Active-Set QP (150s) ===
    remaining = _deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: Active-Set QP ===")
        alloc = min(150.0, remaining * 0.15)
        a_qp = active_set_qp(a, time_budget=alloc, verbose=True, K=20)

        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_qp.copy()
            a = a_qp.copy()
            print(f"  ** QP: obj={best_score:.8f} **")
        else:
            print(f"  QP: obj={obj:.8f}")

    # === Phase 5: Subgradient + LSE polish (remaining/2) ===
    remaining = _deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 5: Final Polish ===")
        alloc_sg = min(100.0, remaining * 0.25)
        a_sg = subgradient_polish(a, time_budget=alloc_sg, verbose=True)
        obj = compute_obj(a_sg)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_sg.copy()
            a = a_sg.copy()
            print(f"  ** Subgrad: obj={best_score:.8f} **")

        # Another LSE pass at high temperature
        remaining = _deadline - time.monotonic()
        if remaining > 30:
            alloc_lse2 = min(100.0, remaining * 0.3)
            a_lse2 = lse_mirror_descent(a, time_budget=alloc_lse2, verbose=False,
                                        tau_init=15.0, tau_max=100.0, eta_init=0.05)
            obj = compute_obj(a_lse2)
            if obj < best_score - 1e-12:
                best_score = obj
                best_seq = a_lse2.copy()
                a = a_lse2.copy()
                print(f"  ** LSE2: obj={best_score:.8f} **")

    # === Phase 6: Length sweep (remaining) ===
    remaining = _deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 6: Length Sweep ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            max(200, n_best - 100),
            max(200, n_best - 50),
            n_best,
            n_best + 50,
            n_best + 100,
            n_best + 200,
        ]))
        n_options = [n for n in n_options if 200 <= n <= 3000]

        for n_try in n_options:
            remaining = _deadline - time.monotonic()
            if remaining < 40:
                break
            if n_try == n_best:
                continue

            a_try = resample_sequence(best_seq, n_try)
            alloc = min(remaining * 0.3, 60.0)

            # Quick LSE + LP
            a_opt = lse_mirror_descent(a_try, time_budget=alloc * 0.5, verbose=False,
                                       tau_init=5.0, tau_max=40.0, eta_init=0.12)
            a_opt = lp_alternating(a_opt, time_budget=alloc * 0.5, verbose=False, beta=0.6)
            obj_try = compute_obj(a_opt)

            if obj_try < best_score - 1e-12:
                best_score = obj_try
                best_seq = a_opt.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Final: one more LP polish from best ===
    remaining = _deadline - time.monotonic()
    if remaining > 20:
        a_final = lp_alternating(best_seq, time_budget=min(remaining * 0.6, 60.0),
                                 verbose=False, beta=0.7)
        obj = compute_obj(a_final)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_final.copy()
            print(f"  ** Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
