Actually, one more important consideration: **the LP objective**. When we solve the LP, we get g* that minimizes M_g = max_k (a*g*)_k. But we also want to know what M_a = max_k (g* * a)_k would be (i.e., the "dual" LP value). By symmetry, (a*g*)_k = (g* * a)_k, so M_a = M_g. So the LP value IS the bidirectional peak.

At a fixed point, a* = g* and M* = max_k (a* * a*)_k. The objective is 2n * M*.

So the LP alternation is:
- Given a, compute M(a) = LP value = min_g max_k (a*g)_k.
- If M(a) < max_k (a*a)_k, the LP found a better g, so update a = g.
- If M(a) = max_k (a*a)_k, we're at a fixed point.

The LP value M(a) is always <= max_k (a*a)_k (since a is a feasible g). So the LP alternation is monotonically non-increasing in the LP value.

Wait, is that true? Let's see:
- a_0 is the initial.
- g_1 = argmin_g max_k (a_0 * g)_k. So max_k (a_0 * g_1)_k = M(a_0) <= max_k (a_0 * a_0)_k.
- a_1 = g_1.
- g_2 = argmin_g max_k (a_1 * g)_k = argmin_g max_k (g_1 * g)_k. So M(a_1) = max_k (a_1 * g_2)_k.
- But is M(a_1) <= M(a_0)? Not necessarily! Because a_1 = g_1 is optimized for a_0, not for itself.

"""
Optimized step function search using aggressive LP alternation (best-response dynamics)
with adaptive oscillation handling, multi-dimension exploration, and coordinate descent.
Targets objective below 1.5027.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_best_response(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix A where A[k,j] = a_norm[k-j] if 0<=k-j<n
    # (a*g)[k] = sum_j a_norm[k-j] * g[j] for valid indices
    # We need: (a*g)[k] - M <= 0 for k=0,...,2n-2
    # Variables: g[0],...,g[n-1], M
    # A_ub shape: (2n-1) x (n+1)
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    for j in range(n):
        # j contributes to positions k = j, j+1, ..., j+n-1
        A_ub[j:j+n, j] = a_norm
    A_ub[:, n] = -1.0  # -M
    
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_opt = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_opt))
        if s > 1e-15:
            g_opt = g_opt / s
        return g_opt, float(res.x[n])
    return None, None


def lp_alternation(a_init, time_budget, verbose=False):
    """
    Aggressive LP alternation with full steps and oscillation handling.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    history = []  # (a, obj) pairs
    stall = 0
    iteration = 0
    
    while time.monotonic() - start < time_budget:
        g, M = solve_best_response(a)
        if g is None:
            break
        
        # Check if g is close to a (convergence)
        diff = np.linalg.norm(a - g)
        if diff < 1e-8:
            stall += 1
        else:
            stall = 0
        
        iteration += 1
        
        # Full step: a = g
        a_new = g.copy()
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            history.append((a.copy(), obj_new))
        elif obj_new < best_obj + 1e-4:
            # Slightly worse but not by much - accept to explore
            a = a_new.copy()
            history.append((a.copy(), obj_new))
        else:
            # Reject, try a blend
            for t in [0.5, 0.3, 0.1]:
                a_blend = (1-t) * best_a + t * g
                a_blend = np.maximum(a_blend, 0)
                a_blend = a_blend / a_blend.sum()
                obj_blend = compute_obj(a_blend)
                if obj_blend < best_obj - 1e-12:
                    best_obj = obj_blend
                    best_a = a_blend.copy()
                    a = a_blend.copy()
                    history.append((a.copy(), obj_blend))
                    break
            else:
                # No improvement, try perturbation
                stall += 1
                if stall > 5:
                    # Random restart
                    rng = np.random.default_rng(iteration * 31 + n)
                    noise = rng.normal(0, 0.1, n)
                    a = best_a * (1 + noise)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0
                    history = []
        
        if verbose and iteration % 10 == 0:

            print(f"    LP-Alt it{iteration}: obj={best_obj:.8f}, LP_M={M:.6f}, "
                  f"diff={diff:.2e}, t={elapsed:.1f}s")
        
        # Convergence check
        if stall > 20:
            break
    
    return best_a


def coord_descent(a_init, time_budget):
    """Coordinate descent: adjust pairs of coordinates to reduce peak convolution."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find top contributors to the peak
        pairs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                pairs.append((a[i] * a[j], i, j))
        pairs.sort(reverse=True)
        
        improved = False
        for contrib, i, j in pairs[:20]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break
            
            # Try moving mass from i (or j) to a random other position
            for _ in range(5):
                target = np.random.randint(n)
                if target == i or target == j:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * a[i] if a[i] >= a[j] else frac * a[j]
                    src = i if a[i] >= a[j] else j
                    if delta >= a[src]:
                        continue
                    a_old = a.copy()
                    a[src] -= delta
                    a[target] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a = a_old.copy()
        
        if not improved:
            # Try small random perturbations
            for _ in range(10):
                idx1 = np.random.randint(n)
                idx2 = np.random.randint(n)
                if idx1 == idx2:
                    continue
                for frac in [0.01, 0.05]:
                    delta = frac * min(a[idx1], a[idx2])
                    a[idx1] -= delta
                    a[idx2] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    a[idx1] += delta
                    a[idx2] -= delta
            if not improved:
                break
    
    return best_a


def resample(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

"""
Optimized step function search using aggressive full-step LP alternation (best-response dynamics),
multi-dimension systematic exploration, structure-aware restarts, and coordinate descent polish.
Key differences from previous: full LP steps, faster HiGHS solver, broader n sweep,
and oscillation-aware averaging.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_best_response(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt).
    """
    n = len(a)
    a_norm = a / np.sum(a)

    A_ub = np.zeros((2 * n - 1, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = a_norm
    A_ub[:, n] = -1.0

    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c = np.zeros(n + 1)
    c[n] = 1.0

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tol': 1e-8})
    if res.status == 0:
        g_opt = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_opt))
        if s > 1e-15:
            g_opt = g_opt / s
        return g_opt, float(res.x[n])
    return None, None


def lp_alternation_aggressive(a_init, time_budget, verbose=False):
    """
    Aggressive LP alternation with full steps, oscillation detection,
    and periodic averaging to converge to local optima.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # History for oscillation detection
    recent_objs = []
    recent_asis = []
    stall = 0
    iteration = 0
    consecutive_worse = 0

    while time.monotonic() - start < time_budget:
        g, M_lp = solve_best_response(a)
        if g is None:
            break

        iteration += 1
        obj_current = compute_obj(a)

        # Full step
        a_new = g.copy()
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            consecutive_worse = 0
            recent_objs.append(obj_new)
            recent_asis.append(a.copy())
        elif obj_new < best_obj + 0.002:
            # Small degradation: accept to explore (might lead to better basin)
            a = a_new.copy()
            consecutive_worse += 1
            recent_objs.append(obj_new)
            recent_asis.append(a.copy())
        else:
            # Reject: try partial steps
            improved = False
            for t in [0.5, 0.3, 0.7, 0.2]:
                a_blend = (1 - t) * a + t * g
                a_blend = np.maximum(a_blend, 0)
                sb = float(np.sum(a_blend))
                if sb < 1e-10:
                    continue
                a_blend = a_blend / sb
                obj_blend = compute_obj(a_blend)
                if obj_blend < best_obj - 1e-12:
                    best_obj = obj_blend
                    best_a = a_blend.copy()
                    a = a_blend.copy()
                    consecutive_worse = 0
                    improved = True
                    break
            if not improved:
                consecutive_worse += 1

        # Keep history bounded
        if len(recent_objs) > 10:
            recent_objs.pop(0)
            recent_asis.pop(0)

        # Oscillation detection: if last 4+ iterations alternate up/down
        if len(recent_objs) >= 4:
            diffs = np.diff(recent_objs[-4:])
            if len(diffs) >= 3 and (diffs[0] * diffs[1] < 0) and (diffs[1] * diffs[2] < 0):
                # Oscillating: use average of last 4
                a_avg = np.mean(recent_asis[-4:], axis=0)
                a_avg = np.maximum(a_avg, 0)
                a_avg = a_avg / a_avg.sum()
                obj_avg = compute_obj(a_avg)
                if obj_avg < best_obj:
                    best_obj = obj_avg
                    best_a = a_avg.copy()
                a = a_avg.copy()
                recent_objs = [best_obj]
                recent_asis = [a.copy()]
                consecutive_worse = 0

        # Stall detection and restart
        if consecutive_worse > 8:
            stall += 1
            consecutive_worse = 0
            if stall > 3:
                # Structure-aware restart
                rng = np.random.default_rng(iteration * 7919 + n)
                mode = rng.integers(0, 4)
                if mode == 0:
                    # Multiplicative noise
                    a = best_a * np.exp(rng.normal(0, 0.15, n))
                elif mode == 1:
                    # Blend with uniform
                    a = 0.7 * best_a + 0.3 * np.ones(n) / n
                elif mode == 2:
                    # Blend with a shifted version
                    shift = rng.integers(1, max(2, n // 10))
                    a_shift = np.roll(best_a, shift)
                    a = 0.6 * best_a + 0.4 * a_shift
                else:
                    # Blend with random Dirichlet
                    a_rand = rng.dirichlet(np.ones(n) * 0.5)
                    a = 0.6 * best_a + 0.4 * a_rand
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                recent_objs = [best_obj]
                recent_asis = [a.copy()]

        if verbose and iteration % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"      it{iteration}: obj={best_obj:.8f}, LP={M_lp:.6f}, "
                  f"consec_worse={consecutive_worse}, stall={stall}, t={elapsed:.1f}s")

        # Early convergence
        if len(recent_objs) >= 5 and max(recent_objs[-5:]) - min(recent_objs[-5:]) < 1e-9:
            break

    return best_a


def smooth_optimize_fast(a_init, time_budget, beta=2000):
    """
    Fast smooth optimization using log-sum-exp with mirror descent.
    Good for quick polish.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.0 / np.sqrt(n)

    it = 0
    while time.monotonic() - start < time_budget and it < 500:
        conv = compute_conv(a)
        m = conv.max()
        if m < 1e-15:
            break
        shifted = beta * (conv - m)
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            break
        weights /= w_sum

        # Gradient: g_i = 2 * sum_{j: i+j < 2n-1} a_j * w_{i+j}
        g = np.zeros(n)
        for i in range(n):
            j_start = 0
            j_end = min(n - 1, 2 * n - 2 - i)
            if j_start <= j_end:
                g[i] = 2.0 * np.dot(a[j_start:j_end + 1], weights[i + j_start:i + j_end + 1])

        # Mirror descent
        log_a = np.log(a)
        log_a_new = log_a - lr * g
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = max(1e-6, lr * 0.999)
        else:
            lr = max(1e-6, lr * 0.95)

        it += 1

    return best_a


def coord_descent_refine(a_init, time_budget):
    """Targeted coordinate descent on peak contributors."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Find top contributors
        contribs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(reverse=True)

        improved = False
        for contrib_val, i, j in contribs[:15]:
            if time.monotonic() - start > time_budget:
                break
            if contrib_val < peak_val * 0.01:
                break

            src = i if a[i] >= a[j] else j
            for _ in range(3):
                target = rng.integers(n)
                if target == i or target == j:
                    continue
                for frac in [0.03, 0.07, 0.12]:
                    delta = frac * a[src]
                    if delta < 1e-15 or delta > a[src]:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
            if improved:
                break

        if not improved:
            # Try random pair exchanges
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                for frac in [0.02, 0.05]:
                    delta = frac * min(a[i], a[j])
                    a[i] -= delta
                    a[j] += delta
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        break
                    a[i] += delta
                    a[j] -= delta
            if not improved:
                # Try reducing the peak by scaling down contributors
                for contrib_val, i, j in contribs[:5]:
                    for scale in [0.95, 0.9, 0.85]:
                        a_old_i = a[i]
                        a_old_j = a[j] if i != j else a[i]
                        a[i] *= scale
                        if i != j:
                            a[j] *= scale
                        diff = a_old_i - a[i]
                        if i != j:
                            diff += a_old_j - a[j]
                        # Distribute the freed mass
                        target = rng.integers(n)
                        if target != i and target != j:
                            a[target] += diff
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        obj_try = compute_obj(a)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if not improved:
                    break

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.dirichlet(np.ones(n) * 0.3)
    elif mode == 2:
        # Bell shape
        x = np.linspace(-1, 1, n)
        a = np.exp(-x**2 * 2)
        a = a / a.sum()
    elif mode == 3:
        # U-shape
        x = np.linspace(-1, 1, n)
        a = 1 + x**2
        a = a / a.sum()
    elif mode == 4:
        # W-shape
        x = np.linspace(-1, 1, n)
        a = 1 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Exponential decay from both ends
        x = np.arange(n) / n
        a = np.exp(-4 * np.minimum(x, 1 - x))
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n) * 0.1)
    return a
