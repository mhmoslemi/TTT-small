import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of Q_k(a) using FFT cross-correlation.
    """
    n = len(a)
    conv = compute_conv(a)

    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]

    return g, conv


def cosine_anneal(t, t_max, init, final, warmup_frac=0.1):
    """Cosine annealing schedule with linear warmup."""
    if t_max <= 0:
        return final
    frac = min(1.0, t / t_max)
    if frac < warmup_frac:
        # Linear warmup
        w = frac / warmup_frac
        return init + (final - init) * w * 0.1
    # Cosine decay
    cos_val = 0.5 * (1 + np.cos(np.pi * (frac - warmup_frac) / (1 - warmup_frac)))
    return init + (final - init) * (1 - cos_val)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, optimal_M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None


def fourier_perturbation(a, rng, strength=0.05):
    """Add low-frequency Fourier modes as structured perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 3
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

def block_coordinate_refine(a, time_budget=60.0, n_blocks=4, verbose=False):
    """

    """

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    block_size = max(1, n // n_blocks)
    blocks = [a[i * block_size:min(n, (i + 1) * block_size)] for i in range(n_blocks)]

    while time.monotonic() - start < time_budget:
        improved = False
        for b_idx in range(n_blocks):
            if time.monotonic() - start >= time_budget:
                    break
            start_idx = b_idx * block_size
            end_idx = min(n, (b_idx + 1) * block_size)
            n_block = end_idx - start_idx

            # Solve LP for this block only (fix other blocks)
            # conv(a, g) = conv(a_fixed, g_block) + conv(a_block_old, g_block) + ...
            # This is complex, so use a simpler approach:
            # Just do smoothed GD on the block
            a_sub = a[start_idx:end_idx].copy()
            a_sub = np.maximum(a_sub, 0)
            s_sub = float(np.sum(a_sub))

            # Compute sub-gradient for this block
            conv_full = compute_conv(a)
            # Gradient contribution from this block
            for _ in range(50):
                tau = 100.0
                g_full, _ = smoothed_gradient_fft(a, tau)
                g_sub = g_full[start_idx:end_idx]
                a_sub_new = a_sub - 0.1 / np.sqrt(n_block) * g_sub
                a_sub_new = np.maximum(a_sub_new, 0)
                s_sub_new = float(np.sum(a_sub_new))
                if s_sub_new > 1e-10:
                    a_sub = a_sub_new / s_sub_new

                    a[start_idx:end_idx] = a_sub

def mirror_symmetrize(a):
    """Make sequence mirror-symmetric: a[i] = (a[i] + a[n-1-i]) / 2."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    return a_sym


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Triangular
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 4 * np.abs(x - 0.5), 0.01)
        a = a / a.sum()
    elif mode == 9:
        # Two-decay exponential
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 4))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def spectral_init(n, rng, seed):
    """Initialize in frequency domain: random phases, specific amplitude profile."""
    size = 1
    while size < 2 * n:
        size <<= 1
    # Create frequency-domain magnitude
    freqs = np.fft.rfftfreq(size)
    # Low-pass + some structure
    mag = np.exp(-freqs * n * 0.5)
    phases = rng.uniform(0, 2 * np.pi, size=size // 2 + 1)
    F = mag * np.exp(1j * phases)
    F[0] = abs(F[0])  # Make DC real positive
    a = np.fft.irfft(F, size)[:n]
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining adaptive smoothed GD, enhanced LP polishing,
    frequency-domain diversification, multi-n exploration, and mirror symmetry.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        n_current = 800
        best_seq = np.ones(n_current) / n_current
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # === Phase 1: Quick adaptive smoothed GD (warm start) ===
    print(f"\n=== Phase 1: Adaptive Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 280.0)
    if p1_budget > 30:
        a_sgd = adaptive_smoothed_descent(best_seq, time_budget=p1_budget,
                                          tau_init=5.0, tau_end=600.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_sgd:.8f} (no improvement)")

    # === Phase 2: Enhanced LP Polishing (main workhorse) ===
    print(f"\n=== Phase 2: Enhanced LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.3, 400.0)
    if p2_budget > 30:
        # Multiple LP rounds with perturbation
        for rnd in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p2_budget / (6 - rnd), remaining * 0.5)
            if alloc < 15:
                break
            a_lp = lp_polish_enhanced(best_seq, time_budget=alloc, verbose=True,
                                      seed=1000 + rnd)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                # Fourier perturbation to escape
                if rnd < 5:
                    strength = 0.015 * (rnd + 1)
                    best_seq = fourier_perturbation(best_seq, rng, strength=strength)

    # === Phase 3: Block coordinate refinement ===
    print(f"\n=== Phase 3: Block Coordinate Refinement ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 100.0)
    if p3_budget > 20:
        a_block = block_coordinate_refine(best_seq, time_budget=p3_budget, n_blocks=4)
        obj_block = compute_obj(a_block)
        if obj_block < best_score:
            best_score = obj_block
            best_seq = a_block.copy()
            print(f"  Block: obj={best_score:.8f} **")
        else:
            print(f"  Block: obj={obj_block:.8f}")

    # === Phase 4: Mirror symmetry exploration ===
    print(f"\n=== Phase 4: Mirror Symmetry ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_mirror = mirror_symmetrize(best_seq)
        a_mirror = a_mirror / a_mirror.sum()
        # Quick GD on mirrored version
        alloc = min(remaining * 0.1, 80.0)
        a_mirror = adaptive_smoothed_descent(a_mirror, time_budget=alloc,
                                             tau_init=20.0, tau_end=400.0)
        obj_mirror = compute_obj(a_mirror)
        if obj_mirror < best_score:
            best_score = obj_mirror
            best_seq = a_mirror.copy()
            print(f"  Mirror: obj={best_score:.8f} **")
        else:
            print(f"  Mirror: obj={obj_mirror:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best + 50, n_best + 100,
        n_best + 200, n_best + 400,
        600, 700, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1600 and abs(n - n_best) > 30]
    n_options = n_options[:6]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick adaptive GD
        alloc1 = min(remaining * 0.2, 60.0)
        if alloc1 > 20:
            a_try = adaptive_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=400.0)

        # Short LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 50.0)
        if alloc2 > 15:
            a_try = lp_polish_enhanced(a_try, time_budget=alloc2, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts with spectral init ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 70:
            break

        if r == 0:
            a_r = spectral_init(n_current, rng, seed=42 + r * 100)
        else:
            mode = 1 + r % 9
            a_r = diverse_init(n_current, rng, mode=mode)
            blend = 0.2 + 0.1 * r
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 90.0)
        if alloc > 25:
            half = alloc * 0.6
            a_r = adaptive_smoothed_descent(a_r, time_budget=half,
                                            tau_init=15.0, tau_end=300.0)
            remaining = deadline - time.monotonic()
            lp_alloc = min(remaining * 0.1, 35.0)
            if lp_alloc > 12:
                a_r = lp_polish_enhanced(a_r, time_budget=lp_alloc, seed=500 + r)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau adaptive GD
    if remaining > 40:
        alloc = min(remaining * 0.25, 100.0)
        a_sgd = adaptive_smoothed_descent(best_seq, time_budget=alloc,
                                          tau_init=50.0, tau_end=1200.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Multiple LP passes
    remaining = deadline - time.monotonic()
    for lp_rnd in range(3):
        if remaining <= 20:
            break
        alloc = min(remaining * 0.3, 80.0)
        a_lp = lp_polish_enhanced(best_seq, time_budget=alloc, verbose=False,
                                  seed=9000 + lp_rnd * 100)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP {lp_rnd}: obj={best_score:.8f} **")
        else:
            # Small perturbation and retry
            if lp_rnd < 2:
                best_seq = fourier_perturbation(best_seq, rng, strength=0.01)
            break

    # One more GD if time
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 60.0)
        a_sgd = adaptive_smoothed_descent(best_seq, time_budget=alloc,
                                          tau_init=100.0, tau_end=1500.0)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Ultra-final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
