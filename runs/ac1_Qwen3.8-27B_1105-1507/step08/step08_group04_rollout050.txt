"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-scale Lp-norm annealing,
soft-max L-BFGS optimization, peak-directed coordinate descent, and LP alternating.
Key innovation: uses a hierarchy of surrogate objectives (Lp norms for increasing p)
to navigate the non-convex landscape, combined with adaptive peak-shaving heuristics.
"""

import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time


def conv_full(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = fft(a, size)
    return ifft(A * A, size).real[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = conv_full(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_params(x):
    """Convert unbounded params x to simplex vector a via softmax."""
    x_max = np.max(x)
    ex = np.exp(x - x_max)
    return ex / np.sum(ex)


def lp_norm_obj_grad(x, p, n):
    """
    Objective and gradient for (sum conv[k]^p)^(1/p) with softmax parameterization.
    Minimizes 2n * Lp(conv(a,a)) where a = softmax(x).
    """
    a = softmax_params(x)
    conv = conv_full(a)

    if p == 1:
        # L1 norm = sum(conv) = (sum a)^2 = 1 (constant on simplex)
        obj_val = 2.0 * n * float(np.sum(conv))
        return obj_val, np.zeros(n)

    # Lp norm: (sum conv[k]^p)^(1/p)
    # Use log-sum-exp trick for numerical stability
    log_conv = np.log(np.maximum(conv, 1e-30))
    log_sum = p * np.max(log_conv) + np.log(np.sum(np.exp(p * log_conv - p * np.max(log_conv))))
    lp_val = np.exp(log_sum / p)

    obj_val = 2.0 * n * lp_val

    # Gradient of Lp wrt conv: p * conv[k]^(p-1) / Lp
    if p == 2:
        grad_conv = 2.0 * conv / lp_val
    elif p == 4:
        grad_conv = 4.0 * conv**3 / lp_val
    else:
        grad_conv = p * np.power(conv, p - 1) / lp_val

    # d(conv[k])/da[j] = a[k-j] + a[j-k] (for valid indices)
    # Use FFT for efficiency
    size = 1
    while size < 3 * n:
        size <<= 1

    a_pad = np.zeros(size)
    a_pad[:n] = a
    gc_pad = np.zeros(size)
    gc_pad[:2 * n - 1] = grad_conv

    A_fft = fft(a_pad)
    GC_fft = fft(gc_pad)

    # C[j] = sum_m a[m] * gc[m-j] (cross-correlation via FFT)
    C = ifft(np.conj(A_fft) * GC_fft, size).real

    # grad_a[j] = sum_k grad_conv[k] * (a[k-j] + a[j-k])
    # F1[j] = sum_k gc[k] * a[k-j] = C[j] (from cross-correlation definition)
    # F2[j] = sum_k gc[k] * a[j-k] = C[N-j] (reversed)
    F1 = C[:n]
    F2 = C[np.array([(size - j) % size for j in range(n)])]

    grad_a = F1 + F2

    # Chain rule through softmax: damped/dx[l] = damped/da[l] - a[l]*sum_j damped/da[j]*a[j]
    sum_term = float(np.dot(grad_a, a))
    grad_x = grad_a - a * sum_term
    grad_x *= 2.0 * n

    return obj_val, grad_x


def soft_max_obj_grad(x, tau, n):
    """
    Objective and gradient for soft_max(conv) with softmax parameterization.
    """
    a = softmax_params(x)
    conv = conv_full(a)

    # Soft max with numerical stability
    v = conv / tau
    v_max = np.max(v)
    log_sum = v_max + np.log(np.sum(np.exp(v - v_max)))
    sm_val = tau * log_sum

    obj_val = 2.0 * n * sm_val

    # Weights for soft max
    w = np.exp(v - v_max)
    w /= np.sum(w)

    # Gradient of soft_max wrt conv = w
    # d(conv[k])/da[j] = a[k-j] + a[j-k]
    size = 1
    while size < 3 * n:
        size <<= 1

    a_pad = np.zeros(size)
    a_pad[:n] = a
    w_pad = np.zeros(size)
    w_pad[:2 * n - 1] = w

    A_fft = fft(a_pad)
    W_fft = fft(w_pad)

    C = ifft(np.conj(A_fft) * W_fft, size).real

    F1 = C[:n]
    F2 = C[np.array([(size - j) % size for j in range(n)])]

    grad_a = F1 + F2

    sum_term = float(np.dot(grad_a, a))
    grad_x = grad_a - a * sum_term
    grad_x *= 2.0 * n

    return obj_val, grad_x

    """Optimize Lp norm objective using L-BFGS."""
    n = len(a_init)
    a_work = a_init.copy()
    a_work = np.maximum(a_work, 1e-15)
    a_work = a_work / np.sum(a_work)

    # Initialize x from a (inverse of softmax, approximately)
    x0 = np.log(a_work + 1e-30)
    x0 -= np.mean(x0)  # center for numerical stability

    start = time.monotonic()
    best_x = x0.copy()
    best_obj = 2.0 * n * float(np.max(conv_full(a_work)))

    try:
        res = minimize(
            lp_norm_obj_grad, x0, args=(p, n),
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': 5000, 'iprint': 0}
        )
        if res.fun < best_obj:
            best_obj = res.fun
            best_x = res.x
    except Exception:
        pass

    # Check time
    if time.monotonic() - start > time_budget:
        return best_x

    # Verify and return
    a_result = softmax_params(best_x)
    obj_check = compute_obj(a_result)
    return best_x


def optimize_soft_max(a_init, tau, time_budget, maxiter=3000):
    """Optimize soft-max objective using L-BFGS."""
    n = len(a_init)
    a_work = np.maximum(a_init, 1e-15)
    a_work = a_work / np.sum(a_work)

    x0 = np.log(a_work + 1e-30)
    x0 -= np.mean(x0)

    start = time.monotonic()
    best_x = x0.copy()
    best_obj = compute_obj(a_work)

    try:
        res = minimize(
            soft_max_obj_grad, x0, args=(tau, n),
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': 8000, 'iprint': 0}
        )
        if res.x is not None:
            a_try = softmax_params(res.x)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_x = res.x
    except Exception:
        pass

    return best_x


def peak_shaving(a, n_shave=5, time_budget=30.0):
    """
    Identify peak lags in conv(a,a) and try to reduce them by
    transferring mass from high-product positions to low-product positions.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for iteration in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = conv_full(best_a)
        # Find peak lags
        peak_indices = np.argsort(conv)[::-1][:n_shave]
        peak_val = conv[peak_indices[0]]

        # For each peak lag k, the term sum_j a[j]*a[k-j] is large.
        # Try to reduce it by moving mass from positions j where a[j]*a[k-j] is large
        # to positions where the product would be smaller.
        improved = False
        for k in peak_indices:
            k = int(k)
            # Compute contribution of each j to conv[k]
            # conv[k] = sum_j a[j]*a[k-j] (valid j)
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            if j_min >= j_max:
                continue

            js = np.arange(j_min, j_max + 1)
            contributions = best_a[js] * best_a[k - js]

            # Find the j with largest contribution
            j_large = js[np.argmax(contributions)]
            a_large = best_a[j_large]
            a_partner = best_a[k - j_large]

            if a_large < 1e-10:
                continue

            # Try transferring a small amount from j_large to a position
            # that doesn't contribute to conv[k]
            # Find a position j2 that minimizes impact on all peak lags
            # Simple heuristic: try positions with small a[j]*a[k-j] for all peak k
            for j2 in range(n):
                if j2 == j_large:
                    continue
                # Check if transferring would help
                eps = min(0.05 * a_large, 0.01 / n)
                a_try = best_a.copy()
                a_try[j_large] -= eps
                a_try[j2] += eps
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try
                    improved = True
                    break

        if not improved:
            # Try random perturbation
            rng = np.random.default_rng(hash((n, iteration)) % (2**31))
            scale = 0.02
            a_try = best_a * np.exp(rng.normal(0, scale, n))
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try
                else:
                    break

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_start = j
        k_end = min(m - 1, j + n - 1)
        length = k_end - k_start + 1
        A[k_start:k_end + 1, j] = a[:length]
    return A


def solve_lp(a):
    """Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    m = 2 * n - 1

    # Use scipy linprog (HiGHS)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.ones((1, n + 1))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except Exception:
        pass

    return None


def lp_alternate(a, time_budget=100.0):
    """LP alternating method."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp = solve_lp(a)
        if g_lp is None:
            break

        best_local_obj = np.inf
        best_local_a = best_a.copy()

        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                scale = min(0.03 + 0.01 * (stall // 5), 0.15)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    if compute_obj(a) < best_obj:
                        best_obj = compute_obj(a)
                        best_a = a.copy()
                stall = 0
                if stall > 15:
                    break

    return best_a


def diverse_initialization(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        # Asymmetric bump (favored since we want anti-symmetric structure)
        a = np.exp(-30 * (x - 0.3) ** 2) + 0.1 * np.exp(-30 * (x - 0.8) ** 2)
    elif choice == 1:
        # Ramp
        a = x ** rng.uniform(0.5, 3.0)
    elif choice == 2:
        # Two separated bumps
        c1, c2 = sorted(rng.uniform(0.1, 0.9, 2))
        a = np.exp(-60 * (x - c1) ** 2) + np.exp(-60 * (x - c2) ** 2)
    elif choice == 3:
        # Oscillatory envelope
        a = (0.3 + 0.7 * x) * (0.5 + 0.5 * np.cos(2 * np.pi * rng.integers(2, 8) * x))
        a = np.maximum(a, 0)
    elif choice == 4:
        # Random smooth (low-pass filtered noise)
        a = np.ones(n)
        for _ in range(4):
            center = rng.uniform(0, 1)
            width = rng.uniform(0.05, 0.3)
            amp = rng.uniform(-0.5, 1.5)
            a *= 1.0 + amp * np.exp(-((x - center) / width) ** 2)
        a = np.maximum(a, 0.01)
    else:
        # Inverse ramp (large on right)
        a = (1 - x) ** rng.uniform(0.5, 3.0)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_seq(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining Lp-norm annealing, soft-max L-BFGS,
    peak shaving, LP alternating, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a_parent = parent / s
        n_base = len(a_parent)
        print(f"Parent: n={n_base}, obj={compute_obj(a_parent):.8f}")
    except Exception:
        n_base = 800
        a_parent = np.ones(n_base) / n_base
        print(f"No parent, using uniform: n={n_base}, obj={compute_obj(a_parent):.8f}")

    best_a = a_parent.copy()
    best_obj = compute_obj(a_parent)

    # === Phase 1: Lp-norm annealing (key innovation) ===
    print("\n=== Phase 1: Lp-Norm Annealing ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Try p = 2, 4, 8, 16, 32, 64
        p_values = [2, 4, 8, 16, 32, 64]
        time_per_p = min(remaining / len(p_values), 120.0)

        x_current = best_a.copy()
        x_current = np.maximum(x_current, 1e-15)
        x_current = x_current / np.sum(x_current)

        for p in p_values:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            alloc = min(time_per_p, remaining * 0.6)
            print(f"  Lp p={p} (budget={alloc:.0f}s)...")

            x_opt = optimize_lp_norm(x_current, p, time_budget=alloc)
            a_opt = softmax_params(x_opt)
            obj_opt = compute_obj(a_opt)

            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                print(f"    ** Improved: obj={best_obj:.8f} **")
            else:
                print(f"    obj={obj_opt:.8f} (no improvement)")

            # Use as warm start for next p
            x_current = a_opt.copy()

    # === Phase 2: Soft-max annealing ===
    print("\n=== Phase 2: Soft-Max Annealing ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        conv_best = conv_full(best_a / np.sum(best_a))
        tau_max = float(np.max(conv_best))
        tau_values = [tau_max * 0.5, tau_max * 0.25, tau_max * 0.125,
                      tau_max * 0.0625, tau_max * 0.03125]

        a_sm = best_a.copy()
        for tau in tau_values:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            alloc = min(80.0, remaining * 0.4)
            print(f"  SoftMax tau={tau:.6f} (budget={alloc:.0f}s)...")

            x_opt = optimize_soft_max(a_sm, tau, time_budget=alloc)
            a_opt = softmax_params(x_opt)
            obj_opt = compute_obj(a_opt)

            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                a_sm = a_opt.copy()
                print(f"    ** Improved: obj={best_obj:.8f} **")
            else:
                print(f"    obj={obj_opt:.8f}")
                a_sm = a_opt.copy()

    # === Phase 3: Peak shaving ===
    print("\n=== Phase 3: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(60.0, remaining * 0.3)
        a_shaved = peak_shaving(best_a, n_shave=5, time_budget=alloc)
        obj_shaved = compute_obj(a_shaved)
        if obj_shaved < best_obj:
            best_obj = obj_shaved
            best_a = a_shaved.copy()
            print(f"  ** Peak shaving: obj={best_obj:.8f} **")
        else:
            print(f"  Peak shaving: obj={obj_shaved:.8f}")

    # === Phase 4: LP Alternating ===
    print("\n=== Phase 4: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(150.0, remaining * 0.5)
        a_lp = lp_alternate(best_a, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            print(f"  ** LP alternating: obj={best_obj:.8f} **")
        else:
            print(f"  LP alternating: obj={obj_lp:.8f}")

    # === Phase 5: Multi-Lp from LP result (if improved) ===
    remaining = deadline - time.monotonic()

            if remaining < 15:

            alloc = min(40.0, remaining * 0.3)
            x_opt = optimize_lp_norm(best_a, p, time_budget=alloc)
            a_opt = softmax_params(x_opt)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                print(f"  ** Lp p={p} round 2: obj={best_obj:.8f} **")

    # === Phase 6: Multi-n exploration ===
    print("\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([
        max(100, n_best - 100),
        n_best,
        n_best + 100,
        n_best + 200,
        n_best + 400,
        n_best + 800,
    ]))
    n_options = [x for x in n_options if 100 <= x <= 3000]
    # Remove duplicates and n_best (already tried)
    n_options = [x for x in n_options if x != n_best]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 25:

        a_try = resample_seq(best_a, n_try)
        a_try = a_try / np.sum(a_try)

        # Quick Lp optimization
        alloc = min(30.0, remaining * 0.3)
        x_opt = optimize_lp_norm(a_try, 8, time_budget=alloc)
        a_opt = softmax_params(x_opt)
        obj_opt = compute_obj(a_opt)

        # Quick peak shaving
        if remaining > 10:
            a_opt = peak_shaving(a_opt, n_shave=3, time_budget=min(10, remaining * 0.2))
            obj_opt = compute_obj(a_opt)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 7: Diverse restarts ===
    print("\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_a)

    for d in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_div = diverse_initialization(n_final, rng)
        alloc = min(30.0, remaining * 0.3)

        # Quick Lp optimization from diverse start
        x_opt = optimize_lp_norm(a_div, 4, time_budget=alloc)
        a_opt = softmax_params(x_opt)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            print(f"  ** Diverse {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Diverse {d}: obj={obj_opt:.8f}")

    # === Phase 8: Final polish ===
    print("\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Soft-max with very small tau
    if remaining > 15:
        conv_best = conv_full(best_a / np.sum(best_a))
        tau_small = float(np.max(conv_best)) * 0.01
        alloc = min(25.0, remaining * 0.4)
        x_opt = optimize_soft_max(best_a, tau_small, time_budget=alloc)
        a_opt = softmax_params(x_opt)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            print(f"  ** Final soft-max: obj={best_obj:.8f} **")

    # Peak shaving
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_shaved = peak_shaving(best_a, n_shave=3, time_budget=min(15, remaining * 0.5))
        obj_shaved = compute_obj(a_shaved)
        if obj_shaved < best_obj:
            best_obj = obj_shaved
            best_a = a_shaved.copy()
            print(f"  ** Final peak shaving: obj={best_obj:.8f} **")

    # LP one more time
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_lp = lp_alternate(best_a, time_budget=min(20, remaining * 0.5))
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            print(f"  ** Final LP: obj={best_obj:.8f} **")

    # === Final validation ===
    best_a = np.maximum(best_a, 0)
    result = [float(min(1000.0, x)) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:

    else:

        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")
    return result
