Optimized step function search using active-set LP with Nesterov acceleration,
adaptive constraint management, and multi-dimension exploration.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build A matrix where A @ g = conv(a, g). A has shape (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_active(a, active_indices, M_bound):

    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Use only active constraints tightly, others with loose bound
    # Actually, let's just use all constraints but with the ones we care about
    A = build_conv_matrix(a_norm)
    
    # LP: min M
    # s.t. A @ g <= M  (2n-1 constraints)
    #      sum(g) = 1
    #      g >= 0
    
    # Variables: g[0..n-1], M  -> n+1 variables
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))

    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

        s = float(np.sum(g_val))
        if s > 1e-10:

    return None, None


def solve_lp_cvxpy(a):
    """Solve the LP using cvxpy."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CBC, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None

            print(f"      LP-Acc it{iteration}: obj={best_obj:.8f}, t={last_good_t:.3f}, "
                  f"stall={stall}, temp={temperature:.3f}, elapsed={elapsed:.1f}s")
    
    return best_a


def logsumexp_optimize(a, time_budget=200.0, beta_start=100.0, beta_end=5000.0, verbose=False):
    """
    Smooth optimization using log-sum-exp with adaptive beta and 
    backtracking line search.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0 / np.sqrt(n)
    momentum = 0.0
    v = np.zeros(n)
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / 200.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Compute smooth gradient
        _, g = smooth_obj_grad(a, beta)
        
        # Project gradient onto tangent space of simplex
        g = g - np.mean(g)
        
        # Momentum update
        v = momentum * v + g
        momentum = min(0.9, momentum + 0.001)
        
        # Backtracking line search
        step_size = lr
        a_new = None
        for _ in range(10):
            candidate = a - step_size * v
            candidate = np.maximum(candidate, 1e-15)
            candidate = candidate / candidate.sum()
            obj_cand = compute_obj(candidate)
            if obj_cand < best_obj - 1e-12 or obj_cand < compute_obj(a) - 1e-12:
                a_new = candidate
                break
            step_size *= 0.5
        
        it += 1
        if a_new is not None:
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                stall = 0
            else:
                a = a_new.copy()
                stall += 1
        else:
            stall += 1
            lr = max(1e-6, lr * 0.9)
        
        if stall > 50:
            # Restart
            rng = np.random.default_rng(it * 17 + n)
            a = best_a * np.exp(0.05 * rng.normal(0, 1, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            v = np.zeros(n)
            momentum = 0.0
            lr = 1.0 / np.sqrt(n)
            stall = 0
        
        if verbose and it % 50 == 0:
            print(f"      LSE it{it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.4f}, t={time.monotonic()-start:.1f}s")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        # Two gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Triangle in frequency domain (sinc-like in time)
        x = np.linspace(-1, 1, n)
        a = (1.0 - np.abs(x)) ** 2 + 0.01
        a = a / a.sum()
    elif mode == 4:
        # Arcsine distribution
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-8) + 0.5)
        a = a / a.sum()
    elif mode == 5:
        # Periodic
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # W-shaped
        x = np.linspace(0, 1, n)
        a = (np.abs(4*x - 2) - 0.5)**2 + 0.05
        a = a / a.sum()
    elif mode == 7:
        # Random low-frequency
        freqs = rng.integers(1, 4, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def dual_game_approx(a, num_iters=50):
    """
    Approximate the dual of the best-response LP using multiplicative weights.
    The dual is: max_y min_i (A^T*y)[i] s.t. y in simplex.
    
    Returns approximate best response g.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # A has shape (2n-1, n), A[k,i] = a_norm[k-i] if 0<=k-i<n
    # A^T has shape (n, 2n-1), A^T[i,k] = a_norm[k-i] if 0<=k-i<n
    # (A^T*y)[i] = sum_k A^T[i,k]*y[k] = sum_k a_norm[k-i]*y[k] (valid k)
    #            = convolution of a_norm and y, evaluated at position i
    #            = (a_norm * y)[i] where * is linear convolution
    
    # Initialize y uniformly
    m = 2 * n - 1
    y = np.ones(m) / m
    
    eta = 1.0 / np.sqrt(np.log(m))
    
    for it in range(num_iters):
        # Compute (A^T*y)[i] for all i
        # This is the cross-correlation: (a_norm * y)[i]
        # = sum_k a_norm[k-i] * y[k] for valid k
        # Actually (a_norm * y)[i] = sum_j a_norm[j] * y[i-j]
        # But we want sum_k a_norm[k-i] * y[k] = sum_j a_norm[j] * y[j+i] (sub j=k-i)
        # = correlation of a_norm and y at lag i
        
        # Let's compute it via FFT
        size = 1
        while size < m + n:
            size <<= 1
        fa = rfft(a_norm, size)
        fy = rfft(y, size)
        # Cross-correlation: sum_k a[k] * y[k+i] = IRFFT(FA * conj(FY)) at position i
        # Actually, let's just use the direct formula
        # (A^T*y)[i] = sum_{k=i}^{i+n-1} a_norm[k-i] * y[k] (for k in [0, m-1])
        #            = sum_{j=0}^{n-1} a_norm[j] * y[i+j] (for i+j < m)
        
        vals = np.zeros(n)
        for i in range(n):
            k_start = i
            k_end = min(n - 1, m - 1 - i)
            if k_start <= k_end:
                vals[i] = np.dot(a_norm[:k_end+1], y[i:i+k_end+1])
        
        # Find the minimum (adversary's best response)
        i_min = np.argmin(vals)
        min_val = vals[i_min]
        
        # Update y using MW: increase weight on rows that give high value to i_min
        # (A^T)[i_min, k] = a_norm[k - i_min] if 0 <= k - i_min < n
        # We want to increase y[k] where A^T[i_min, k] is large
        row = np.zeros(m)
        for k in range(i_min, min(i_min + n, m)):
            row[k] = a_norm[k - i_min]
        
        # MW update: y[k] proportional to y[k] * exp(eta * row[k])
        log_y = np.log(y + 1e-30)
        log_y += eta * row
        log_y -= np.max(log_y)
        y = np.exp(log_y)
        y = y / y.sum()
    
    # Extract best response from y: g[i] > 0 only where (A^T*y)[i] is near minimum
    vals = np.zeros(n)
    for i in range(n):
        k_start = i
        k_end = min(n - 1, m - 1 - i)
        if k_start <= k_end:
            vals[i] = np.dot(a_norm[:k_end+1], y[i:i+k_end+1])
    
    # g is supported on the minimizers
    threshold = vals.min() + 0.1 * (vals.max() - vals.min())
    support = vals <= threshold
    if not np.any(support):
        support = vals <= vals.min() + 1e-6
    
    g = np.zeros(n)
    g[support] = 1.0 / np.sum(support)
    
    return g

"""
Optimized step function search using hybrid LP/MW (multiplicative weights) 
alternating method with Nesterov-style momentum, adaptive step sizes,
multi-dimension exploration, and diverse restarts.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def cross_correlate(a, y):
    """Compute (A^T @ y)[i] for all i, where A[k,i]=a[k-i] if valid.
    This equals sum_j a[j]*y[i+j] for valid indices."""
    n = len(a)
    m = len(y)
    vals = np.zeros(n)
    for i in range(n):
        k_end = min(n - 1, m - 1 - i)
        if k_end >= 0:
            vals[i] = np.dot(a[:k_end + 1], y[i:i + k_end + 1])
    return vals


def build_conv_matrix(a):
    """Build A matrix where A @ g = conv(a, g). Shape (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_highs(a):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0 using HiGHS."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def solve_lp_cvxpy(a):
    """Solve LP using cvxpy with multiple solvers."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.CBC]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def mw_best_response(a, num_iters=80, eta=None):
    """
    Approximate the best response g using multiplicative weights on the dual game.
    Dual: max_y min_i (A^T*y)[i] s.t. y in simplex over {0,...,2n-2}.
    Returns (g, approximate_value).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    m = 2 * n - 1
    if eta is None:
        eta = 1.0 / np.sqrt(np.log(max(m, 2)))

    y = np.ones(m) / m

    for it in range(num_iters):
        # Compute (A^T*y)[i] for all i
        vals = cross_correlate(a_norm, y)
        i_min = int(np.argmin(vals))

        # Update y: increase weight on k where a_norm[k - i_min] is large
        row = np.zeros(m)
        k_start = i_min
        k_end = min(i_min + n, m)
        row[k_start:k_end] = a_norm[:k_end - i_min]

        log_y = np.log(y + 1e-30) + eta * row
        log_y -= np.max(log_y)
        y = np.exp(log_y)
        y = y / y.sum()

    # Extract best response
    vals = cross_correlate(a_norm, y)
    min_val = vals.min()
    spread = vals.max() - min_val
    if spread < 1e-12:
        g = np.ones(n) / n
    else:
        threshold = min_val + 0.15 * spread
        support = vals <= threshold
        if not np.any(support):
            support = vals <= min_val + 1e-10
        g = np.zeros(n)
        g[support] = 1.0 / np.sum(support)

    return g, float(min_val)


def smooth_obj_and_grad(a, beta):
    """
    Compute smooth objective value and gradient using log-sum-exp on conv(a,a).
    Returns (smooth_value, gradient).
    """
    n = len(a)
    a = a / np.sum(a)
    conv = compute_conv(a)
    m = conv.max()
    shifted = beta * (conv - m)
    weights = np.exp(shifted)
    weights /= weights.sum()

    smooth_val = m + np.log(weights.sum()) / beta

    # Gradient: g[i] = 2 * sum_{k'} a[k'] * w[i+k']
    g = np.zeros(n)
    for i in range(n):
        k_end = min(n - 1, 2 * n - 2 - i)
        if k_end >= 0:
            g[i] = 2.0 * np.dot(a[:k_end + 1], weights[i:i + k_end + 1])

    return 2.0 * n * smooth_val, 2.0 * n * g


def logsumexp_optimize(a, time_budget=200.0, beta_start=100.0, beta_end=5000.0, verbose=False):
    """Smooth optimization via log-sum-exp with momentum and backtracking."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    momentum = 0.0
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / 300.0)
        beta = beta_start * (beta_end / beta_start) ** frac

        _, g = smooth_obj_and_grad(a, beta)
        g = g - np.mean(g)  # Project to tangent space

        v = momentum * v + g
        momentum = min(0.85, momentum + 0.0005)

        # Backtracking line search
        step_size = lr
        a_new = None
        obj_curr = compute_obj(a)
        for _ in range(8):
            candidate = a - step_size * v
            candidate = np.maximum(candidate, 1e-15)
            candidate = candidate / candidate.sum()
            obj_cand = compute_obj(candidate)
            if obj_cand < obj_curr - 1e-12:
                a_new = candidate
                break
            step_size *= 0.5

        it += 1
        if a_new is not None:
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            stall += 1
            lr = max(1e-6, lr * 0.85)

        if stall > 40:

            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            v = np.zeros(n)
            momentum = 0.0
            lr = 2.0 / np.sqrt(n)
            stall = 0

        if verbose and it % 50 == 0:
            print(f"      LSE it{it}: obj={best_obj:.8f}, beta={beta:.0f}, t={time.monotonic()-start:.1f}s")

    return best_a

def coordinate_refine(a, time_budget=40.0):
    """
    Coordinate descent: move mass from high-contributing positions to reduce peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    for rnd in range(3):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_val = conv[k_star]

        # Find top contributors to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        # Find low positions (not contributing to peak)
        peak_positions = set()
        for _, i, j in contributors[:20]:
            peak_positions.add(i)
            if i != j:
                peak_positions.add(j)

        low_positions = [i for i in range(n) if i not in peak_positions and a[i] < np.median(a)]

        improved = False
        for idx, (contrib, i, j) in enumerate(contributors[:15]):
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break

            for target in low_positions[:6]:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    src = i if a[i] >= a[j] else j
                    delta = frac * a[src]
                    if delta <= 0 or delta >= a[src]:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initial sequences."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.2, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps
        a = np.zeros(n)
        for c, w in [(n // 4, n // 7), (3 * n // 4, n // 7)]:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Parabolic
        x = np.linspace(-1, 1, n)
        a = 1.0 - 0.5 * x**2 + 0.05
        a = a / a.sum()
    elif mode == 4:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - 0.99 * x**2) + 0.3)
        a = a / a.sum()
    elif mode == 5:
        # Periodic
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x + 1.0)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode == 6:
        # W-shape
        x = np.linspace(0, 1, n)
        a = (np.abs(4 * x - 2) - 0.5) ** 2 + 0.08
        a = a / a.sum()
    elif mode == 7:
        # Plateau
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        mid_start, mid_end = n // 4, 3 * n // 4
        a[mid_start:mid_end] = 0.5
        a = a / a.sum()
    elif mode == 8:
        # Exponential taper
        x = np.linspace(0, 1, n)
        a = np.exp(-2.0 * np.abs(x - 0.5)) + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.10, 55.0)
        if restart_budget > 25:
            a_r = logsumexp_optimize(a_r, time_budget=restart_budget * 0.45,
                                     beta_start=80, beta_end=2000)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_r = hybrid_lp_search(a_r, time_budget=min(remaining * 0.08, 25.0))

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} ** IMPROVED")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # --- Phase 5: Final intensive polish ---
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Full LP polish
    if remaining > 30:
        lp_alloc = min(remaining * 0.35, 100.0)
        a_lp = hybrid_lp_search(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} ** IMPROVED")
        else:
            print(f"  Final LP: obj={obj_lp:.8f}")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        cd_alloc = min(remaining * 0.25, 40.0)
        a_cd = coordinate_refine(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CoordDesc: obj={best_score:.8f} ** IMPROVED")
        else:
            print(f"  CoordDesc: obj={obj_cd:.8f}")

    # One more LSE with very high beta for final precision
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_final = logsumexp_optimize(best_seq, time_budget=min(remaining * 0.4, 30.0),
                                     beta_start=2000, beta_end=20000)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LSE: obj={best_score:.8f} ** IMPROVED")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
