    return 2.0 * n * float(np.max(conv))

    """Build row k of the conv matrix: (A[g] @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    row = np.zeros(n)
    for j in range(n):
        if 0 <= k - j < n:
            row[j] = a[k - j]

    """Solve LP with only active constraints.
    min M s.t. conv(a,g)[k] <= M for k in active_indices, sum(g)=1, g>=0
    """

    a = a / np.sum(a)

    A = np.zeros((m, n))
    for i, k in enumerate(active_indices):
        A[i] = build_conv_matrix_row(a, k)

    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    A_eq = np.ones((1, n + 1))
    A_eq[0, :n] = 1.0
    A_eq[0, n] = 0.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

        gs = float(np.sum(g_val))

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast LP with active set, coordinate descent, multi-n exploration,
and diverse perturbations.
"""

from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize_scalar
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    an = a / s
    conv = compute_conv(an)
    return 2.0 * n * float(np.max(conv))

def subgradient_step_full(a):
    """Compute full subgradient of max_k conv_k at a (normalized)."""
    n = len(a)
    conv = compute_conv(a)
    max_val = np.max(conv)
    # Find all k where conv_k is within epsilon of max
    tol = max_val * 1e-4 + 1e-12
    active = np.where(conv >= max_val - tol)[0]

    # Subgradient: grad[j] = (2/n_active) * sum_{k in active} a[k-j] (for valid indices)
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            grad[j_min:j_max + 1] += a[k - idx]
    grad /= len(active)
    grad *= 2.0  # Factor of 2 from max of sum of products

    # Center for simplex constraint
    grad -= np.mean(grad)
    return grad, max_val

        grad, max_val = subgradient_step_full(x)

        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            # Use a diminishing step with restarts
            eff_step = step / (1.0 + 0.001 * it)
        else:
            eff_step = 0.0

        momentum = beta * momentum - eff_step * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Nesterov update
        t_k = 2.0 / (np.sqrt(it + 1) + 1)
        x = a_new + (1 - t_k) * (a_new - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            best_step = eff_step
        else:
            step *= 0.995

        if it % 500 == 499:
            step *= 0.95

        if verbose and it % 2000 == 0:
            print(f"    SubGrad it={it}: obj={best_obj:.8f}, step={eff_step:.6f}")

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, seed_type=None):
    """Generate diverse structured initialization."""
    if seed_type is None:
        seed_type = rng.integers(0, 8)

    x = np.linspace(0, 1, n)

    if seed_type == 0:
        # Asymmetric Gaussian (shifted center)
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        a = np.exp(-0.5 * ((x - center) / width) ** 2)
    elif seed_type == 1:
        # Linear ramp (asymmetric)
        p = rng.uniform(1.0, 5.0)
        a = x ** p
    elif seed_type == 2:
        # Two asymmetric bumps
        c1, c2 = sorted(rng.uniform(0.1, 0.9, 2))
        w = rng.uniform(0.05, 0.2)
        a = np.exp(-0.5 * ((x - c1) / w) ** 2) + 0.7 * np.exp(-0.5 * ((x - c2) / w) ** 2)
    elif seed_type == 3:
        # Cosine window (asymmetric)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x + phase)
        a = np.maximum(a, 0.01)
    elif seed_type == 4:
        # Spectral: flat-ish spectrum with random phases
        nfft = 1
        while nfft < 2 * n:
            nfft <<= 1
        phases = rng.uniform(0, 2 * np.pi, nfft // 2 + 1)
        magnitudes = rng.uniform(0.5, 1.0, nfft // 2 + 1)
        F = np.zeros(nfft, dtype=complex)
        F[:nfft // 2 + 1] = magnitudes * np.exp(1j * phases)
        # Make it conjugate symmetric for real output
        for k in range(1, nfft // 2):
            F[nfft - k] = np.conj(F[k])
        a = irfft(F, nfft)[:n]
        a = np.abs(a) + 0.01
    elif seed_type == 5:
        # Geometric decay
        r = rng.uniform(0.7, 0.99)
        a = r ** np.arange(n)
        if rng.random() > 0.5:
            a = a[::-1]
    elif seed_type == 6:
        # Random smooth (low-pass filtered noise)
        a = rng.standard_normal(n)
        # Simple moving average smoothing
        k = max(3, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = a - np.min(a) + 0.1
    else:
        # Asymmetric Lorentzian
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.05, 0.3)
        skew = rng.uniform(0.5, 3.0)
        a = 1.0 / (1.0 + (skew * (x - center) / width) ** 2)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def perturbation(a, rng, scale=0.1):
    """Apply multiplicative log-normal perturbation."""
    n = len(a)
    noise = rng.normal(0, scale, n)
    a_new = a * np.exp(noise)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating, coordinate descent,
    subgradient polishing, multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        print(f"Parent loaded: n={len(a)}, obj={compute_obj(a):.8f}")
    except Exception:
        a = np.ones(800) / 800
        print(f"No parent, using uniform: n=800, obj={compute_obj(a):.8f}")

    best_a = a.copy()
    best_obj = compute_obj(a)
    n_base = len(best_a)
    print(f"Initial: n={n_base}, obj={best_obj:.8f}")

    # === Phase 1: LP Alternating (primary) ===
    print(f"\n=== Phase 1: LP Alternating (n={n_base}, budget=300s) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    a_p1 = lp_alternate_fast(best_a, time_budget=p1_budget, n_active=50, verbose=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1
        print(f"  Phase 1 result: obj={best_obj:.8f} **")
    else:
        print(f"  Phase 1: no improvement (obj={obj_p1:.8f})")

    # === Phase 2: Coordinate Descent Polish ===
    print(f"\n=== Phase 2: Coordinate Descent (budget=200s) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = min(remaining * 0.15, 200.0)
        a_cd = coord_descent(best_a, time_budget=p2_budget, verbose=True, max_sweeps=150)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd
            print(f"  Phase 2 result: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 2: no improvement (obj={obj_cd:.8f})")

    # === Phase 3: LP Again from improved point ===
    print(f"\n=== Phase 3: LP Round 2 (budget=200s) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.15, 200.0)
        a_p3 = lp_alternate_fast(best_a, time_budget=p3_budget, n_active=40, verbose=True)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3
            print(f"  Phase 3 result: obj={best_obj:.8f} **")

    # === Phase 4: Coordinate Descent Again ===
    print(f"\n=== Phase 4: Coordinate Descent Round 2 (budget=100s) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = min(remaining * 0.08, 100.0)
        a_cd2 = coord_descent(best_a, time_budget=p4_budget, verbose=True, max_sweeps=100)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_obj:
            best_obj = obj_cd2
            best_a = a_cd2
            print(f"  Phase 4 result: obj={best_obj:.8f} **")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best,
        n_best + 50, n_best + 100, n_best + 200,
        n_best - 200, n_best + 300,
    ]))
    n_options = [x for x in n_options if 100 <= x <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if n_try == len(best_a):
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try

        # Quick LP + coord descent
        alloc = min(40.0, remaining * 0.12)
        a_opt = lp_alternate_fast(a_try, time_budget=alloc * 0.6, n_active=30)
        remaining = deadline - time.monotonic()
        if remaining > 10 and alloc > 20:
            a_opt = coord_descent(a_opt, time_budget=min(alloc * 0.4, remaining * 0.3), max_sweeps=30)

        obj_try = compute_obj(a_opt)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Initializations ===
    print(f"\n=== Phase 6: Diverse Initializations (n={n_best}) ===")
    remaining = deadline - time.monotonic()

    for d in range(12):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_div = diverse_init(n_best, rng, seed_type=d)
        alloc = min(35.0, remaining * 0.08)

        # LP then coord descent
        a_opt = lp_alternate_fast(a_div, time_budget=alloc * 0.6, n_active=30)
        remaining = deadline - time.monotonic()
        if remaining > 10 and alloc > 20:
            a_opt = coord_descent(a_opt, time_budget=min(alloc * 0.4, remaining * 0.3), max_sweeps=20)

        obj_div = compute_obj(a_opt)
        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Diverse {d}: obj={obj_div:.8f}")

    # === Phase 7: Perturbation + Re-optimize ===
    print(f"\n=== Phase 7: Perturbation Search ===")
    remaining = deadline - time.monotonic()

    for p in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        scale = 0.03 + 0.02 * p
        a_pert = perturbation(best_a, rng, scale=scale)
        alloc = min(30.0, remaining * 0.1)

        a_opt = lp_alternate_fast(a_pert, time_budget=alloc * 0.7, n_active=30)
        remaining = deadline - time.monotonic()
        if remaining > 10 and alloc > 15:
            a_opt = coord_descent(a_opt, time_budget=min(alloc * 0.3, remaining * 0.3), max_sweeps=15)

        obj_pert = compute_obj(a_opt)
        if obj_pert < best_obj:
            best_obj = obj_pert
            best_a = a_opt.copy()
            print(f"  ** Perturb {p} (scale={scale:.3f}): obj={best_obj:.8f} **")
        else:
            print(f"  Perturb {p} (scale={scale:.3f}): obj={obj_pert:.8f}")

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_lp = min(remaining * 0.3, 60.0)
        a_final = lp_alternate_fast(best_a, time_budget=alloc_lp, n_active=50, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final LP: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_cd = min(remaining * 0.3, 50.0)
        a_cd_final = coord_descent(best_a, time_budget=alloc_cd, verbose=True, max_sweeps=50)
        obj_cd_final = compute_obj(a_cd_final)
        if obj_cd_final < best_obj:
            best_obj = obj_cd_final
            best_a = a_cd_final
            print(f"  Final CoordDesc: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_sg = min(remaining * 0.3, 30.0)
        a_sg = subgradient_optimize(best_a, time_budget=alloc_sg)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"  Final SubGrad: obj={best_obj:.8f} **")

    # One last LP
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc_last = min(remaining * 0.5, 20.0)
        a_last = lp_alternate_fast(best_a, time_budget=alloc_last, n_active=40)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Last LP: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
