"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with warm-starting, multiplicative subgradient,
peak shaving, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2 (scale invariant, so normalize first)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A


def solve_lp_cvxpy(a, g_init=None, time_limit=30):
    """Solve LP: min M s.t. conv(a,g) <= M (all lags), sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # conv(a, g)[k] = sum_j a[max(0, k-n+1):min(n, k+1) shifted] * g[...]
    # Using the matrix form
    A = build_conv_matrix(a)
    
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    if g_init is not None:
        g.value = g_init
        M.value = float(np.max(A @ g_init))
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > time_limit:
                prob.value = None
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    
    return None, None


def solve_lp_scipy(a, g_init=None, time_limit=30):
    """Fallback LP using scipy highs."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A = build_conv_matrix(a)
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.ones((1, n + 1))
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    x0 = None
    if g_init is not None:
        g_init = np.maximum(g_init, 0)
        gs = float(np.sum(g_init))
        if gs > 1e-10:
            g_init = g_init / gs
            x0 = np.concatenate([g_init, [float(np.max(A @ g_init))]])
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', x0=x0)
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g_val))
        if gs > 1e-10:
            return g_val / gs, float(res.x[n])
    
    return None, None


def lp_step(a, g_init=None):
    """One LP step: solve for optimal g given a."""
    g, M = solve_lp_cvxpy(a, g_init=g_init)
    if g is None:
        g, M = solve_lp_scipy(a, g_init=g_init)
    return g, M

def subgradient_step_v2(a, top_k=20):
    """Compute subgradient of max(conv(a,a)) at a (sum(a)=1)."""
    n = len(a)
    conv = compute_conv(a)
    
    if top_k >= len(conv):
        k_indices = np.arange(len(conv))
    else:
        k_indices = np.argsort(conv)[::-1][:top_k]

    # grad[j] = (2/K) * sum_{k in K} 2*a[(k-j) mod n] (circular approx)
    # Actually for linear conv: grad[j] = (2/K) * sum_{k in K} a[k-j] * 1[0 <= k-j < n]
    # Using FFT: this is cross-correlation
    w = np.zeros(2 * n - 1)
    for k in k_indices:
        w[int(k)] = 1.0 / len(k_indices)

    n_fft = 1
    while n_fft < 4 * n:
        n_fft <<= 1

    A_fft = rfft(a, n_fft)
    W_fft = rfft(w, n_fft)
    R = irfft(np.conj(A_fft) * W_fft, n_fft)
    
    grad = np.zeros(n)
    for j in range(n):
        idx = (n_fft - j) % n_fft
        grad[j] = R[idx]
    
    # The full gradient of 2n*max(conv)/sum^2 at sum=1 is 2n * grad
    grad *= 2.0 * n
    return grad


def multiplicative_optimize(a, time_budget=60.0, top_k=20, verbose=False):
    """Multiplicative (exponential) subgradient optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0 / n
    decay = 0.999

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        grad = subgradient_step_v2(a, top_k=top_k)
        
        # Multiplicative update: a_new = a * exp(-lr * (grad - mean(grad)))
        grad_centered = grad - np.mean(grad)
        a_new = a * np.exp(-lr * grad_centered)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        if it % 100 == 99:
            lr *= decay

    return best_a


def peak_shaving(a, time_budget=60.0, verbose=False):
    """Directly reduce elements that contribute most to the peak of conv."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        
        # Find which elements contribute most to the peak
        # conv[peak_idx] = sum_j a[j] * a[peak_idx - j] for valid j
        contrib = np.zeros(n)
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n:
                contrib[j] = a[j] * a[k]
        
        # Reduce the top contributors
        top_m = max(5, n // 10)
        top_indices = np.argsort(contrib)[::-1][:top_m]
        
        eta = 0.01 + 0.04 * (it / 10000.0)  # Increasing aggressiveness
        a_new = a.copy()
        a_new[top_indices] *= (1.0 - eta)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # If it didn't help, try reducing from the peak position
            a = best_a.copy()
            # Small random perturbation
            rng = np.random.default_rng(it + 12345)
            a = a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def coordinate_descent(a, time_budget=30.0):
    """Simple coordinate descent on individual elements."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget:
        improved = False
        for j in range(n):
            if time.monotonic() - start > time_budget:
                    break
            
            if a[j] < 1e-8:
                continue
            
            # Try different values for a[j]
            for delta in [0.1, -0.1, 0.3, -0.3, 0.5]:
                a_try = a.copy()
                a_try[j] = max(0, a[j] * (1 + delta))
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        
        if not improved:
            # Random perturbation
            j = rng.integers(0, n)
            a[j] *= np.exp(rng.normal(0, 0.1))
            a = np.maximum(a, 0)
            a = a / np.sum(a)

    return best_a

import numpy as np

def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2 (scale invariant)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def subgradient_step(a, top_k=20):
    """Compute subgradient of max(conv(a,a)) at a (sum(a)=1) using FFT."""
    n = len(a)
    conv = compute_conv(a)

    if top_k >= len(conv):
        k_indices = np.arange(len(conv))
    else:
        k_indices = np.argsort(conv)[::-1][:top_k]

    w = np.zeros(2 * n - 1)
    for k in k_indices:
        w[int(k)] = 1.0 / len(k_indices)

    n_fft = 1
    while n_fft < 4 * n:
        n_fft <<= 1

    A_fft = rfft(a, n_fft)
    W_fft = rfft(w, n_fft)
    R = irfft(np.conj(A_fft) * W_fft, n_fft)

    grad = np.zeros(n)
    for j in range(n):
        idx = (n_fft - j) % n_fft
        grad[j] = R[idx]

    grad *= 2.0 * n
    return grad

        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Get top peaks
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        # Find contributing elements
        contrib = np.zeros(n)
        for pk in peak_indices:
            pk = int(pk)
            for j in range(n):
                k = pk - j
                if 0 <= k < n:
                    contrib[j] += a[j] * a[k]

        # Reduce top contributors
        top_m = max(10, n // 20)
        top_indices = np.argsort(contrib)[::-1][:top_m]

        eta = 0.005 + 0.02 * (it / 20000.0)
        a_new = a.copy()
        a_new[top_indices] *= (1.0 - eta)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Perturbation to escape
            if it % 100 == 99:
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)

    return best_a


def coordinate_descent(a, time_budget=30.0):
    """Coordinate descent on individual elements with multiplicative updates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    while time.monotonic() - start < time_budget:
        improved = False
        # Random order for diversity
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            if a[j] < 1e-8:
                continue

            # Try multiplicative changes
            for factor in [1.1, 0.9, 1.2, 0.8, 1.05, 0.95]:
                a_try = a.copy()
                a_try[j] *= factor
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            # Broader perturbation
            n_pert = max(5, n // 100)
            idx = rng.integers(0, n, n_pert)
            a[idx] *= np.exp(rng.normal(0, 0.1, n_pert))
            a = np.maximum(a, 0)
            a = a / np.sum(a)

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_initialization(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)

    if choice == 0:
        a = np.exp(-50 * (x - 0.5) ** 2)
    elif choice == 1:
        freq = rng.uniform(1, 10)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    elif choice == 2:
        c1, c2 = sorted(rng.uniform(0.1, 0.9, 2))
        a = np.exp(-60 * (x - c1) ** 2) + np.exp(-60 * (x - c2) ** 2)
    elif choice == 3:
        a = np.ones(n)
        for _ in range(5):
            center = rng.uniform(0, 1)
            width = rng.uniform(0.05, 0.3)
            amp = rng.uniform(-0.5, 0.5)
            a *= 1.0 + amp * np.exp(-((x - center) / width) ** 2)
        a = np.maximum(a, 0.01)
    elif choice == 4:
        alpha = rng.uniform(0.5, 3.0)
        a = x ** alpha
    elif choice == 5:
        beta = rng.uniform(0.5, 2.0)
        a = 0.5 * (1 + np.cos(2 * np.pi * beta * (x - 0.5)))
        a = np.maximum(a, 0.01)
    elif choice == 6:
        a = np.ones(n) * 0.3
        for _ in range(4):
            freq = rng.uniform(1, 15)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0.01)
    elif choice == 7:
        # Spectral constructor: pick random spectrum, take IFFT, take abs
        n_fft = n
        phases = rng.uniform(0, 2 * np.pi, n_fft)
        mags = np.ones(n_fft) * (1.0 + 0.3 * rng.standard_normal(n_fft))
        mags = np.maximum(mags, 0.1)
        F = mags * np.exp(1j * phases)
        # Make it Hermitian for real signal
        F_conj = F
        n_p = n_fft // 2 + 1
        F[1:n_fft - 1] = F_conj[1:n_fft - 1]
        F[1:n_p] = mags[1:n_p] * np.exp(1j * phases[1:n_p])
        F[n_fft - (n_p - 1):] = np.conj(F[1:n_p][::-1])
        a = np.abs(np.real(ifft(F)))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        # sinc-like (flat spectrum in a band)
        bandwidth = rng.uniform(0.3, 0.8)
        a = np.sinc(bandwidth * (x - 0.5))
        a = np.abs(a)
        a = np.maximum(a, 0.01)
    else:
        # Random log-normal
        a = np.exp(rng.normal(0, rng.uniform(0.3, 1.0), n))
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses improved LP alternating, multiplicative subgradient,
    spectral flattening, peak shaving, coordinate descent, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        n_base = len(a)
        print(f"Parent loaded: n={n_base}, obj={compute_obj(a):.8f}")
    except Exception:
        n_base = 800
        a = np.ones(n_base) / n_base
        print(f"No parent, using uniform: n={n_base}, obj={compute_obj(a):.8f}")

    best_a = a.copy()
    best_obj = compute_obj(a)

    # === Phase 1: Primary LP Alternating ===
    print(f"\n=== Phase 1: LP Alternating (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.28, 280.0)
    a_p1 = lp_alternate(best_a, time_budget=p1_budget, verbose=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1
        print(f"  Phase 1 result: obj={best_obj:.8f} **")

    # === Phase 2: Multiplicative Subgradient ===
    print(f"\n=== Phase 2: Multiplicative Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p2_budget = min(remaining * 0.12, 120.0)
        for top_k in [10, 30]:
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            alloc = min(p2_budget / 2.0, remaining * 0.4)
            a_sg = multiplicative_optimize(best_a, time_budget=alloc, top_k=top_k)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg
                print(f"  MultiSG top_k={top_k}: obj={best_obj:.8f} **")
            else:
                print(f"  MultiSG top_k={top_k}: obj={obj_sg:.8f}")

    # === Phase 3: Spectral Flattening ===
    print(f"\n=== Phase 3: Spectral Flattening ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.08, 60.0)
        a_spec = spectral_flatten(best_a, time_budget=p3_budget)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_obj:
            best_obj = obj_spec
            best_a = a_spec
            print(f"  Spectral: obj={best_obj:.8f} **")
        else:
            print(f"  Spectral: obj={obj_spec:.8f}")

    # === Phase 4: Peak Shaving ===
    print(f"\n=== Phase 4: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        p4_budget = min(remaining * 0.08, 60.0)
        a_ps = peak_shaving(best_a, time_budget=p4_budget)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_obj:
            best_obj = obj_ps
            best_a = a_ps
            print(f"  PeakShave: obj={best_obj:.8f} **")
        else:
            print(f"  PeakShave: obj={obj_ps:.8f}")

    # === Phase 5: LP Alternating Round 2 ===
    print(f"\n=== Phase 5: LP Alternating Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p5_budget = min(remaining * 0.22, 220.0)
        a_p5 = lp_alternate(best_a, time_budget=p5_budget, verbose=True)
        obj_p5 = compute_obj(a_p5)
        if obj_p5 < best_obj:
            best_obj = obj_p5
            best_a = a_p5
            print(f"  Phase 5 result: obj={best_obj:.8f} **")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    n_options = sorted(set([
        max(400, n_best - 200),
        n_best,
        n_best + 100, n_best + 200, n_best + 400,
    ]))
    n_options = [x for x in n_options if 400 <= x <= 4000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        if n_try == len(best_a):
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))

            else:

        alloc = min(35.0, remaining * 0.2)
        a_opt = lp_alternate(a_try, time_budget=alloc)

            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Initializations ===
    print(f"\n=== Phase 7: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    for d in range(12):

        if remaining < 30:

        a_div = diverse_initialization(n_best, rng)
        alloc = min(30.0, remaining * 0.15)
        a_opt = lp_alternate(a_div, time_budget=alloc)
        obj_div = compute_obj(a_opt)

        if obj_div < best_obj:

            best_a = a_opt.copy()
            print(f"  ** Diverse {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Diverse {d}: obj={obj_div:.8f}")

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 25:
        alloc = min(remaining * 0.30, 70.0)
        a_final = lp_alternate(best_a, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final LP: obj={best_obj:.8f} **")

    # Multiplicative subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.20, 40.0)
        a_sg = multiplicative_optimize(best_a, time_budget=alloc, top_k=15)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"  Final MultiSG: obj={best_obj:.8f} **")

    # Peak shaving
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.15, 25.0)
        a_ps = peak_shaving(best_a, time_budget=alloc)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_obj:
            best_obj = obj_ps
            best_a = a_ps
            print(f"  Final PeakShave: obj={best_obj:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.15, 25.0)
        a_cd = coordinate_descent(best_a, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd
            print(f"  Final CoordDescent: obj={best_obj:.8f} **")

    # One last LP
    remaining = deadline - time.monotonic()
    if remaining > 12:
        alloc = min(remaining * 0.4, 25.0)
        a_last = lp_alternate(best_a, time_budget=alloc)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Last LP: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
