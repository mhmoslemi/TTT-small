        improved = False
        # Focus on coordinates affecting top peaks
        c = compute_conv(a)
        S_total = float(np.sum(a))
        n_peaks = min(focus_peaks, 2 * n - 1)
        top_peak_indices = np.argsort(c)[::-1][:n_peaks]

        # Determine which coordinates to update (those affecting top peaks)
        coords_to_try = set()
        for kp in top_peak_indices:
            kp = int(kp)
            for i in range(n):
                if 0 <= kp - i < n:
                    coords_to_try.add(i)
        # Limit to at most n coordinates (all of them if needed)
        coord_list = list(coords_to_try)
        np.random.shuffle(coord_list)

        for i in coord_list:
            if time.monotonic() - start > time_budget:
                break
            v_old = a[i]
            S = float(np.sum(a)) - v_old  # sum excluding a[i]
            if S < 1e-12:
                continue

            # Compute c_base: remove contribution of a[i]
            c_base = c.copy()
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            if k_lo <= k_hi:
                a_segment = a[k_lo - i:k_hi - i + 1]  # a[k-i] for k in [k_lo, k_hi]
                c_base[k_lo:k_hi + 1] -= 2.0 * v_old * a_segment
            # Correct for k=2i (we subtracted 2*v_old*a[i] but should subtract v_old^2)
            k_2i = 2 * i
            if 0 <= k_2i < 2 * n - 1:
                c_base[k_2i] += v_old ** 2  # add back the extra v_old^2 we subtracted

            # Grid of candidate values
            v_max = max(3.0 * float(np.max(a)), S * 2.0)
            v_grid = np.linspace(0.0, v_max, v_grid_size)
            # Add current value
            v_grid = np.unique(np.concatenate([v_grid, [v_old]]))
            v_grid = v_grid[v_grid >= 0]

            # Vectorized evaluation: c_new[k, j] for all k and all v_j
            M = len(v_grid)
            c_new = np.tile(c_base[:, None], (1, M))
            if k_lo <= k_hi:
                a_seg = a[k_lo - i:k_hi - i + 1][:, None]  # shape (num_k, 1)
                c_new[k_lo:k_hi + 1, :] = c_base[k_lo:k_hi + 1][:, None] + 2.0 * a_seg * v_grid[None, :]
            if 0 <= k_2i < 2 * n - 1:
                c_new[k_2i, :] = c_base[k_2i] + v_grid ** 2

            peaks = np.max(c_new, axis=0)  # shape (M,)
            F_vals = 2.0 * n * peaks / (S + v_grid) ** 2
            F_vals = np.where((S + v_grid) > 1e-10, F_vals, np.inf)
            best_idx = int(np.argmin(F_vals))
            v_new = v_grid[best_idx]
            F_new = F_vals[best_idx]

            if F_new < best_obj - 1e-12:
                # Accept and update
                a[i] = v_new
                c = compute_conv(a)
                S_total = float(np.sum(a))
                best_obj = float(2.0 * n * np.max(c) / S_total ** 2)
                improved = True

        # Full sweep every 10 partial sweeps
        if (sweep + 1) % 10 == 0 and time.monotonic() - start < time_budget * 0.8:
            for i in range(n):
                if time.monotonic() - start > time_budget:
                    break
                v_old = a[i]
                S = float(np.sum(a)) - v_old
                if S < 1e-12:
                    continue
                c_base = c.copy()
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                if k_lo <= k_hi:
                    a_segment = a[k_lo - i:k_hi - i + 1]
                    c_base[k_lo:k_hi + 1] -= 2.0 * v_old * a_segment
                k_2i = 2 * i
                if 0 <= k_2i < 2 * n - 1:
                    c_base[k_2i] += v_old ** 2

                v_max = max(3.0 * float(np.max(a)), S * 2.0)
                v_grid = np.linspace(0.0, v_max, v_grid_size)
                v_grid = np.unique(np.concatenate([v_grid, [v_old]]))
                v_grid = v_grid[v_grid >= 0]
                M = len(v_grid)
                c_new = np.tile(c_base[:, None], (1, M))
                if k_lo <= k_hi:
                    a_seg = a[k_lo - i:k_hi - i + 1][:, None]
                    c_new[k_lo:k_hi + 1, :] = c_base[k_lo:k_hi + 1][:, None] + 2.0 * a_seg * v_grid[None, :]
                if 0 <= k_2i < 2 * n - 1:
                    c_new[k_2i, :] = c_base[k_2i] + v_grid ** 2
                peaks = np.max(c_new, axis=0)
                F_vals = 2.0 * n * peaks / (S + v_grid) ** 2
                F_vals = np.where((S + v_grid) > 1e-10, F_vals, np.inf)
                best_idx = int(np.argmin(F_vals))
                v_new = v_grid[best_idx]
                F_new = F_vals[best_idx]
                if F_new < best_obj - 1e-12:
                    a[i] = v_new
                    c = compute_conv(a)
                    S_total = float(np.sum(a))
                    best_obj = float(2.0 * n * np.max(c) / S_total ** 2)
                    improved = True

        if verbose and sweep % 20 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, elapsed={time.monotonic()-start:.1f}s")
        if not improved and sweep > 5:
            break

    a = a / float(np.sum(a))
    return a


def gradient_refine(a, time_budget=50.0, top_k=5, verbose=False):
    """Nesterov-accelerated subgradient on simplex for fast local refinement."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / float(np.sum(a))
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.05

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            k_lo = max(0, k - n + 1)
            k_hi = min(n - 1, k)
            if k_lo <= k_hi:
                m_start = k - k_hi
                m_end = k - k_lo
                grad[m_start:m_end + 1] += 2.0 * y[k_lo:k_hi + 1][::-1]
        grad /= len(top_idx)
        grad = grad - np.mean(grad)

        y_new = y - step_size * grad
        # Project to simplex
        y_new = y_new - np.min(y_new)
        s = y_new.sum()
        if s > 1e-15:
            y_new = y_new / s
        else:
            y_new = np.ones(n) / n
        t_k = 2.0 / (it + 2.0)
        y = y_new + (1.0 - t_k) * (y_new - a)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        a = y_new

        if it % 5 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        if it % 100 == 99:
            step_size = max(step_size * 0.95, 0.003)

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if new_n == 1:
        return np.array([a.argmax() and a.max() or 1.0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    result = np.interp(x_new, x_old, a)
    return np.maximum(result, 0)


def hann_window(n):
    """Hann window normalized to sum 1."""
    if n == 1:
        return np.array([1.0])
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
    return w / np.sum(w)


def modulated_window(n, A=0.3, freq=1.0, phase=0.0, alpha=1.0):
    """
    Modulated window: a[i] = hann[i] * (1 + A*cos(2*pi*freq*i/n + phase)).
    Normalized to sum 1.
    """
    base = hann_window(n)
    mod = 1.0 + A * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
    a = base * mod
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return base
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: coordinate descent + gradient refinement + n-sweep + restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # --- Load parent ---
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = hann_window(800)
        best_score = compute_obj(best_seq)
        print(f"Hann init (n=800): obj={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: Structured initializations (60s) ===
    print("\n=== Phase 1: Structured Initializations ===")
    phase1_deadline = start_time + 60.0
    inits = [best_seq.copy()]

    # Modulated windows with various parameters
    for A in [0.2, 0.4, 0.6, 0.8]:
        for freq in [1.0, 2.0, 3.0, 5.0]:
            for phase in [0.0, np.pi / 3, np.pi / 2]:
                if time.monotonic() > phase1_deadline:
                    break
                a_mod = modulated_window(n_base, A=A, freq=freq, phase=phase)
                inits.append(a_mod)

    # Gaussian with different widths
    for sigma_frac in [0.1, 0.2, 0.3, 0.4]:
        x = np.arange(n_base)
        gauss = np.exp(-0.5 * ((x - n_base / 2.0) / (n_base * sigma_frac)) ** 2)
        gauss = np.maximum(gauss, 0)
        inits.append(gauss / gauss.sum())

    # Linear ramp (asymmetric)
    ramp = np.linspace(0.1, 1.0, n_base)
    inits.append(ramp / ramp.sum())
    ramp2 = np.linspace(1.0, 0.1, n_base)
    inits.append(ramp2 / ramp2.sum())

    # Random
    for _ in range(3):
        a_r = rng.random(n_base)
        inits.append(a_r / a_r.sum())

    for i, init in enumerate(inits):
        if time.monotonic() > phase1_deadline:
            break
        obj = compute_obj(init)
        if obj < best_score:
            best_score = obj
            best_seq = init.copy()
            print(f"  Init {i}: obj={best_score:.8f} **")
        if i < 8 and time.monotonic() < phase1_deadline - 5:
            a_cd = coordinate_descent(init, max_sweeps=5, time_budget=3.0, v_grid_size=100)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Init {i}+CD: obj={best_score:.8f} **")

    # === Phase 2: Main coordinate descent (500s) ===
    print(f"\n=== Phase 2: Main Coordinate Descent ===")
    a = best_seq.copy()
    phase2_budget = 500.0
    a_cd = coordinate_descent(a, max_sweeps=1000, time_budget=phase2_budget,
                              v_grid_size=200, focus_peaks=5, verbose=True)
    obj_cd = compute_obj(a_cd)
    if obj_cd < best_score:
        best_score = obj_cd
        best_seq = a_cd.copy()
        print(f"  CD main: obj={best_score:.8f} **")
    else:
        print(f"  CD main: obj={obj_cd:.8f}")

    # === Phase 3: Gradient refinement (200s) ===
    print(f"\n=== Phase 3: Gradient Refinement ===")
    for k_val in [3, 5, 10]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(50.0, remaining * 0.3)
        a_grad = gradient_refine(best_seq, time_budget=alloc, top_k=k_val, verbose=False)
        obj_g = compute_obj(a_grad)
        if obj_g < best_score:
            best_score = obj_g
            best_seq = a_grad.copy()
            print(f"  Grad k={k_val}: obj={best_score:.8f} **")
        else:
            print(f"  Grad k={k_val}: obj={obj_g:.8f}")
        # Follow up with CD
        remaining = deadline - time.monotonic()
        if remaining > 30 and obj_g < best_score + 1e-4:
            a_cd2 = coordinate_descent(a_grad, max_sweeps=50, time_budget=min(30, remaining * 0.3),
                                       v_grid_size=150)
            obj_cd2 = compute_obj(a_cd2)
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  CD after Grad: obj={best_score:.8f} **")

    # === Phase 4: n-sweep (300s) ===
    print(f"\n=== Phase 4: n-Sweep ===")
    n_candidates = [100, 150, 200, 300, 400, 500, 600, 700, 800, 900, 1000, 1200, 1500, 2000]
    n_candidates = [n for n in n_candidates if n >= 50 and n <= 2500]
    # Add neighbors of current best n
    n_candidates.append(len(best_seq))
    n_candidates.append(max(50, len(best_seq) - 50))
    n_candidates.append(len(best_seq) + 50)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)
        alloc_cd = min(15.0, remaining * 0.15)
        a_opt = coordinate_descent(a_try, max_sweeps=50, time_budget=alloc_cd, v_grid_size=150)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Random restarts (200s) ===
    print(f"\n=== Phase 5: Random Restarts ===")
    n_best = len(best_seq)
    for r in range(30):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        # Various perturbation strategies
        strategy = r % 4
        if strategy == 0:
            # Multiplicative noise
            scale = 0.02 + 0.01 * min(r, 10)
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
        elif strategy == 1:
            # Add small uniform noise
            a_r = best_seq + rng.uniform(0, 0.02, n_best)
        elif strategy == 2:
            # Reshape: mix with a different window
            w = hann_window(n_best) * 0.5 + modulated_window(n_best, A=0.5, freq=rng.integers(1, 5)) * 0.5
            a_r = (1 - 0.1) * best_seq + 0.1 * w
        else:
            # Shift and blend
            shift = rng.integers(1, max(2, n_best // 10))
            a_shifted = np.roll(best_seq, shift)
            a_r = 0.7 * best_seq + 0.3 * a_shifted

        a_r = np.maximum(a_r, 0)
        sr = float(np.sum(a_r))
        if sr < 1e-10:
            continue
        a_r = a_r / sr

        alloc = min(remaining * 0.25, 15.0)
        a_opt = coordinate_descent(a_r, max_sweeps=30, time_budget=alloc, v_grid_size=150)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  R{r} (strat={strategy}, n={n_best}): obj={best_score:.8f} **")
        else:
            print(f"  R{r} (strat={strategy}): obj={obj_r:.8f}")

    # === Phase 6: Final polish (remaining) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Long coordinate descent
        alloc = min(60.0, remaining * 0.5)
        a_final = coordinate_descent(best_seq, max_sweeps=200, time_budget=alloc,
                                     v_grid_size=300, focus_peaks=8, verbose=True)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_grad = gradient_refine(best_seq, time_budget=min(30.0, remaining * 0.4), top_k=3)
            obj_g = compute_obj(a_grad)
            if obj_g < best_score:
                best_score = obj_g
                best_seq = a_grad.copy()
                print(f"  Final Grad: obj={best_score:.8f} **")
            # One more CD
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_cd3 = coordinate_descent(best_seq, max_sweeps=50, time_budget=min(20, remaining * 0.6),
                                           v_grid_size=300)
                obj_cd3 = compute_obj(a_cd3)
                if obj_cd3 < best_score:
                    best_score = obj_cd3
                    best_seq = a_cd3.copy()
                    print(f"  Final CD2: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure valid output
    best_seq = np.clip(best_seq, 0, 1000.0)
    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
