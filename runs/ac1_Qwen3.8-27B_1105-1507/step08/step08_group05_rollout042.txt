```python
"""
Soft-max gradient descent with multiplicative updates + LP polishing.
Uses entropic smoothing of the max-convolution objective for efficient
gradient-based optimization on the simplex. Combines with LP alternating
for exact min-max improvements.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_grad(a, t=0.001):
    """
    Compute the gradient of the soft-max of conv(a,a).
    Uses FFT for efficient cross-correlation.
    Returns (soft_max_value, gradient).
    """
    n = len(a)
    conv = compute_conv(a)
    # Softmax for numerical stability
    conv_shifted = conv - np.max(conv)
    exp_vals = np.exp(conv_shifted / t)
    w = exp_vals / np.sum(exp_vals)
    soft_max_val = t * np.log(np.sum(exp_vals)) + np.max(conv)
    
    # grad[m] = 2 * sum_{j=0}^{n-1} a[j] * w[m+j]
    # = 2 * conv(a, w_rev)[2n-2-m]
    w_rev = w[::-1]
    size = 1
    while size < n + (2 * n - 1):
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w_rev, size)
    conv_aw = irfft(fa * fw, size)[:n + 2 * n - 1]
    grad = 2.0 * conv_aw[n - 1: 2 * n - 1][::-1]
    
    return soft_max_val, grad

        return a0.copy()
    a = a / s
    best_a = a.copy()

    start = time.monotonic()
    eta = eta_init
    t = t_init

    total_iters = 5000

        t = t_init * (t_final / t_init) ** frac
        eta = eta_init * (eta_min / eta_init) ** frac

def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_norm, M_opt) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None

def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
```

```python
"""
Soft-max mirror descent + targeted peak reduction + LP polishing.
Uses entropic smoothing of max(conv(a,a)) with FFT-based gradient computation.
Combines multiplicative weight updates, top-k targeted gradient, and LP alternating
for exact min-max improvements. Sweeps sequence length and uses random restarts.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_grad_fft(a, t=0.001):
    """
    Compute gradient of soft-max of conv(a,a) using FFT cross-correlation.
    Returns (soft_max_value, gradient_vector).
    """
    n = len(a)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    conv_shifted = conv - conv_max
    exp_vals = np.exp(conv_shifted / t)
    sum_exp = np.sum(exp_vals)
    if sum_exp < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_vals / sum_exp
    soft_max_val = conv_max + t * np.log(sum_exp)

    # grad[m] = 2 * sum_{j=0}^{n-1} a[j] * w[m+j]
    # = 2 * conv(a, w_rev)[2n-2-m]
    w_rev = w[::-1]
    total_len = n + (2 * n - 1) - 1
    size = 1
    while size < total_len:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w_rev, size)
    conv_aw = irfft(fa * fw, size)[:total_len]
    # conv_aw[n-1 : 2n-1] reversed gives indices 2n-2, 2n-3, ..., n-1
    grad = 2.0 * conv_aw[n - 1: 2 * n - 1][::-1]
    # Clean up small numerical errors
    grad = np.where(np.abs(grad) < 1e-15, 0.0, grad)

    return soft_max_val, grad


def topk_grad(a, k=10):
    """Compute gradient using only top-k peaks (sparse, targeted)."""
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:k]
    top_vals = conv[top_idx]
    w = top_vals / np.sum(top_vals)

    grad = np.zeros(n)
    for idx, weight in zip(top_idx, w):
        k_val = int(idx)
        j_start = max(0, k_val - n + 1)
        j_end = min(n - 1, k_val)
        if j_start <= j_end:
            m_start = k_val - j_end
            m_end = k_val - j_start
            grad[m_start:m_end + 1] += 2.0 * weight * a[j_start:j_end + 1][::-1]
    return grad


def mirror_descent(a0, t_init=0.01, t_final=0.0001, eta_init=0.3, eta_min=0.005,

    """
    Multiplicative weights (mirror descent) on simplex.
    a_new = a * exp(-eta * (grad - mean(grad))), then normalize.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init
    t = t_init
    patience = 0
    max_iters = 200000

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        frac = min(1.0, it / (max_iters * 0.7))
        t = t_init * (t_final / t_init) ** frac
        eta = eta_init * (eta_min / eta_init) ** (frac * 0.5)

        if use_topk:
            grad = topk_grad(a, topk_k)
        else:
            _, grad = softmax_grad_fft(a, t)

        grad_c = grad - np.mean(grad)
        log_update = -eta * grad_c
        log_update = np.clip(log_update, -6.0, 6.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-16)
        sn = float(np.sum(a_new))
        if sn < 1e-10:

                    eta = max(eta * 0.6, eta_min * 0.1)

            a = a_new

def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    if n > 3000:
        return None, None
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=60.0, verbose=False):
    """LP alternating with 1D line search."""
    n = len(a0)
    if n > 3000:
        return a0.copy()
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_good_t = 0.5

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)

        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 40, 'xatol': 0.005})
            best_t = res.x if res.success else last_good_t
        except Exception:
            best_t = last_good_t

        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            last_good_t = best_t
            stall_count = 0
            if verbose:
                print(f"      LP: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall_count += 1
            if stall_count > 8:
                break

    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def coordinate_descent_peak(a0, time_budget=30.0, n_peaks=5, verbose=False):
    """Directly reduce top peaks by moving mass from contributors."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for outer in range(100):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        top_k = np.argsort(conv)[::-1][:n_peaks]
        improved = False
        for k in top_k:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue
            # Contributors to conv[k]: a[j] * a[k-j] for j in [j_start, j_end]
            contrib = np.zeros(n)
            for j in range(j_start, j_end + 1):
                m = k - j
                contrib[j] += a[m]
                contrib[m] += a[j]
            # Try moving mass from high contributors to low
            sorted_idx = np.argsort(contrib)
            high = sorted_idx[-3:]
            low = sorted_idx[:3]
            for src in high:
                for dst in low:
                    if src == dst or a[src] < 1e-12:
                        continue
                    for frac in [0.03, 0.08, 0.15]:
                        delta = frac * a[src]
                        a[src] -= delta
                        a[dst] += delta
                        a = np.maximum(a, 0)
                        st = float(np.sum(a))
                        if st < 1e-10:
                            a[src] += delta
                            a[dst] -= delta
                            continue
                        a = a / st
                        obj_try = compute_obj(a)
                        if obj_try < best_obj - 1e-13:
                            best_obj = obj_try
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[dst] -= delta
                            a = a / st
                            break
        if not improved and outer > 20:
            break
        a = best_a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Soft-max mirror descent + top-k gradient + LP + coordinate descent.
    Multi-phase with n-sweep and random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n0) / max(1, n0 - 1)))
        best_seq = w / np.sum(w)
        best_score = compute_obj(best_seq)
        print(f"Hann init (n={n0}): obj={best_score:.8f}")

    # === Phase 1: Mirror Descent (main workhorse) ===
    print(f"\n=== Phase 1: Soft-max Mirror Descent ===")
    a_cur = best_seq.copy()

    # Round 1: Wide temperature, large steps
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(100.0, remaining * 0.25)
        print(f"  MD-R1: t=[0.02->0.002], eta=[0.4->0.01], budget={alloc:.0f}s")
        a_new = mirror_descent(a_cur, t_init=0.02, t_final=0.002,
                               eta_init=0.4, eta_min=0.01,
                               time_budget=alloc, verbose=False)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-13:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new
            print(f"  ** MD-R1: obj={best_score:.8f} **")
        else:
            a_cur = best_seq.copy()
            print(f"  MD-R1: obj={obj_new:.8f}")

    # Round 2: Medium temperature
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(100.0, remaining * 0.25)
        print(f"  MD-R2: t=[0.005->0.0003], eta=[0.2->0.005], budget={alloc:.0f}s")
        a_new = mirror_descent(a_cur, t_init=0.005, t_final=0.0003,
                               eta_init=0.2, eta_min=0.005,
                               time_budget=alloc, verbose=False)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-13:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new
            print(f"  ** MD-R2: obj={best_score:.8f} **")
        else:
            a_cur = best_seq.copy()
            print(f"  MD-R2: obj={obj_new:.8f}")

    # Round 3: Fine temperature
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(80.0, remaining * 0.25)
        print(f"  MD-R3: t=[0.001->0.00005], eta=[0.08->0.002], budget={alloc:.0f}s")
        a_new = mirror_descent(a_cur, t_init=0.001, t_final=0.00005,
                               eta_init=0.08, eta_min=0.002,
                               time_budget=alloc, verbose=False)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-13:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new
            print(f"  ** MD-R3: obj={best_score:.8f} **")
        else:
            a_cur = best_seq.copy()
            print(f"  MD-R3: obj={obj_new:.8f}")

    # === Phase 2: Top-k targeted gradient ===
    print(f"\n=== Phase 2: Top-k Targeted Gradient ===")
    for k_val in [5, 10, 20]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(50.0, remaining * 0.2)
        a_new = mirror_descent(best_seq, t_init=0.003, t_final=0.0001,
                               eta_init=0.15, eta_min=0.003,
                               time_budget=alloc, use_topk=True, topk_k=k_val,
                               verbose=False)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-13:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  ** TopK-{k_val}: obj={best_score:.8f} **")
        else:
            print(f"  TopK-{k_val}: obj={obj_new:.8f}")

    # === Phase 3: Coordinate Descent Peak Reduction ===
    print(f"\n=== Phase 3: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(60.0, remaining * 0.15)
        a_new = coordinate_descent_peak(best_seq, time_budget=alloc, n_peaks=5)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-13:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  ** CoordDesc: obj={best_score:.8f} **")
        else:
            print(f"  CoordDesc: obj={obj_new:.8f}")

    # === Phase 4: LP Alternating ===
    print(f"\n=== Phase 4: LP Alternating ===")
    n_cur = len(best_seq)
    if n_cur <= 2500:
        a = best_seq.copy()
        for round_idx in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(60.0, remaining * 0.2)
            a_new = lp_alternate(a, time_budget=alloc, verbose=False)
            obj_new = compute_obj(a_new)
            if obj_new < best_score - 1e-13:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new
                print(f"  ** LP-R{round_idx}: obj={best_score:.8f} **")
            else:
                a = best_seq * np.exp(rng.normal(0, 0.02, n_cur))
                a = np.maximum(a, 0)
                a /= np.sum(a)
                print(f"  LP-R{round_idx}: obj={obj_new:.8f}")
    else:
        print("  Skipping LP (n too large)")

    # === Phase 5: n-sweep ===
    print(f"\n=== Phase 5: n-Sweep ===")
    n_base = len(best_seq)
    n_options = set()
    for delta in [-300, -200, -100, -50, 0, 50, 100, 200, 300, 500]:
        n_try = n_base + delta
        if 150 <= n_try <= 4000:
            n_options.add(n_try)

    for n_try in sorted(n_options):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        if n_try == n_base:
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)
        alloc = min(25.0, remaining * 0.12)
        a_opt = mirror_descent(a_try, t_init=0.005, t_final=0.0002,
                               eta_init=0.15, eta_min=0.005,
                               time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score - 1e-13:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 6: MD + LP interleaved ===
    print(f"\n=== Phase 6: MD + LP Interleaved ===")
    a = best_seq.copy()
    n_cur = len(a)
    for round_idx in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        # MD
        alloc_md = min(40.0, remaining * 0.3)
        a_md = mirror_descent(a, t_init=0.002, t_final=0.00005,
                              eta_init=0.1, eta_min=0.002,
                              time_budget=alloc_md, use_topk=(round_idx % 2 == 1),
                              topk_k=10, verbose=False)
        obj_md = compute_obj(a_md)
        if obj_md < best_score - 1e-13:
            best_score = obj_md
            best_seq = a_md.copy()
            a = a_md
            print(f"  ** MD+LP R{round_idx} MD: obj={best_score:.8f} **")

        # LP
        remaining = deadline - time.monotonic()
        if remaining < 20 or n_cur > 2500:
            break
        alloc_lp = min(30.0, remaining * 0.3)
        a_lp = lp_alternate(a, time_budget=alloc_lp, verbose=False)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-13:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp
            n_cur = len(a)
            print(f"  ** MD+LP R{round_idx} LP: obj={best_score:.8f} **")
        else:
            a = best_seq * np.exp(rng.normal(0, 0.015, len(best_seq)))
            a = np.maximum(a, 0)
            a /= np.sum(a)

    # === Phase 7: Random restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(12):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            scale = 0.02 + 0.008 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr < 1e-10:
                continue
            a_r = a_r / sr
            alloc = min(remaining * 0.25, 35.0)
            a_opt = mirror_descent(a_r, t_init=0.005, t_final=0.0001,
                                   eta_init=0.2, eta_min=0.003,
                                   time_budget=alloc * 0.5, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 15 and len(a_opt) <= 2500:
                a_opt = lp_alternate(a_opt, time_budget=min(alloc * 0.5, remaining * 0.5))
            obj_r = compute_obj(a_opt)
            if obj_r < best_score - 1e-13:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  ** R{r}: obj={best_score:.8f} **")
            else:
                print(f"  R{r}: obj={obj_r:.8f}")

    # === Phase 8: Final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 8: Final Polish ===")
        # Quick MD
        alloc = min(30.0, remaining * 0.4)
        a_f = mirror_descent(best_seq, t_init=0.001, t_final=0.00003,
                             eta_init=0.05, eta_min=0.001,
                             time_budget=alloc, use_topk=True, topk_k=5)
        obj_f = compute_obj(a_f)
        if obj_f < best_score - 1e-13:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  ** Final MD: obj={best_score:.8f} **")

        # Final LP
        remaining = deadline - time.monotonic()
        if remaining > 15 and len(best_seq) <= 2500:
            alloc = min(25.0, remaining * 0.5)
            a_f2 = lp_alternate(best_seq, time_budget=alloc, verbose=True)
            obj_f2 = compute_obj(a_f2)
            if obj_f2 < best_score - 1e-13:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  ** Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
```
