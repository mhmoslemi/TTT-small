O(n) update structure of the convolution when one element changes.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_obj_from_conv(a, conv):
    """Compute objective given a and its precomputed convolution."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A


def solve_lp(a, max_time=60.0):
    """Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_value) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > max_time:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None


def coordinate_descent(a, conv, time_budget=60.0, n_sweeps=None, verbose=False):
    """
    Efficient coordinate descent for minimizing the objective.
    Exploits the fact that changing a[j] only affects n positions in the convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a, conv
    a = a / s
    conv = compute_conv(a).copy()

    best_obj = compute_obj_from_conv(a, conv)
    best_a = a.copy()
    start = time.monotonic()

    if n_sweeps is None:
        n_sweeps = 10000  # will be limited by time_budget

    sweep_order = np.argsort(a)[::-1]  # process large elements first initially

    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break

        # Use random order after first few sweeps
        if sweep < 3:
            order = sweep_order
        else:
            order = np.random.permutation(n)

        improved = False
        for j in order:
            old_val = a[j]
            if old_val < 1e-12:
                # Try small positive values
                candidates = [0.0, 1e-4, 1e-3, s / n * 0.1]
            else:
                # Geometric + linear candidates around current value
                candidates = [0.0]
                for alpha in [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                    candidates.append(old_val * alpha)
                # Also try some absolute offsets
                scale = max(old_val, s / n)
                for delta in [-0.5 * scale, -0.2 * scale, 0.2 * scale, 0.5 * scale]:
                    candidates.append(max(0.0, old_val + delta))

            best_cand = old_val
            best_cand_obj = best_obj

            for w in candidates:
                if w == old_val:
                    continue
                if w < 0:
                    w = 0.0
                diff = w - old_val
                if abs(diff) < 1e-15:
                    continue

                # Update convolution: only positions k where 0 <= k-j < n are affected
                # delta_conv[k] = 2*diff*a[k-j] for k != 2j
                # delta_conv[2j] = diff*(2*a[k-j] + diff) = diff*(w + old_val) if 2j in range
                # More precisely:
                # The contribution of a[j] to (a*a)[k]:
                #   if k = 2j: a[j]^2
                #   if 0 <= k-j < n and k != 2j: 2*a[j]*a[k-j]
                #   else: 0
                # Change from old_val to w:
                #   if k = 2j and 2j < 2n-1: w^2 - old_val^2 = diff*(w+old_val)
                #   if 0 <= k-j < n and k != 2j: 2*diff*a[k-j]
                #   else: 0

                old_max = float(np.max(conv))

                # Check which k values are affected
                # k ranges from j to j+n-1 (where a[k-j] is valid)
                k_min = j
                k_max = min(2 * n - 2, j + n - 1)

                # For affected positions, compute new convolution values
                # We need to find the new max efficiently
                # Positions NOT affected keep their old values
                old_max_affected = float(np.max(conv[k_min:k_max + 1])) if k_max >= k_min else 0.0
                # We need max of: (old max of unaffected) and (new max of affected)

                # Compute new values for affected positions
                affected = np.zeros(k_max - k_min + 1)
                for idx, k in enumerate(range(k_min, k_max + 1)):
                    if k == 2 * j:
                        affected[idx] = conv[k] + diff * (w + old_val)
                    else:
                        affected[idx] = conv[k] + 2.0 * diff * a[k - j]

                new_max_affected = float(np.max(affected))
                # Unaffected max
                if k_min > 0:
                    new_max = max(float(np.max(conv[:k_min])), new_max_affected)
                else:
                    new_max = new_max_affected
                if k_max < 2 * n - 2:
                    new_max = max(new_max, float(np.max(conv[k_max + 1:])))

                new_sum = s + diff
                if new_sum < 1e-10:
                    continue
                new_obj = 2.0 * n * new_max / (new_sum * new_sum)

                if new_obj < best_cand_obj - 1e-14:
                    best_cand_obj = new_obj
                    best_cand = w

            if best_cand != old_val:
                # Apply the update
                diff = best_cand - old_val
                k_min = j
                k_max = min(2 * n - 2, j + n - 1)
                for k in range(k_min, k_max + 1):
                    if k == 2 * j:
                        conv[k] += diff * (best_cand + old_val)
                    else:
                        conv[k] += 2.0 * diff * a[k - j]
                a[j] = best_cand
                s += diff
                improved = True
                best_obj = best_cand_obj
                best_a = a.copy()

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"      CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

        if not improved and sweep > 5:
            # No improvement in this sweep, try broader search
            # Add some noise to escape stagnation
            break

    return best_a, conv


def coordinate_descent_fine(a, conv, time_budget=30.0, verbose=False):
    """Fine coordinate descent with more candidate values and ternary search."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a, conv

    best_obj = compute_obj_from_conv(a, conv)
    best_a = a.copy()
    start = time.monotonic()

    # Fine grid around each element
    for j in range(n):
        if time.monotonic() - start > time_budget:
            break
        old_val = a[j]
        if old_val < 1e-10:
            continue

        # Fine search: try values in a narrow band
        scale = old_val
        candidates = []
        for alpha in np.arange(0.7, 1.35, 0.05):
            candidates.append(old_val * alpha)
        # Also try ternary-like search
        lo, hi = max(0.0, old_val * 0.5), old_val * 2.0
        for _ in range(5):
            m1 = lo + (hi - lo) / 3
            m2 = hi - (hi - lo) / 3

            def eval_at(w):
                diff = w - old_val
                if abs(diff) < 1e-15:
                    return best_obj
                k_min = j
                k_max = min(2 * n - 2, j + n - 1)
                # Compute new max
                affected_old = conv[k_min:k_max + 1].copy()
                for idx, k in enumerate(range(k_min, k_max + 1)):
                    if k == 2 * j:
                        affected_old[idx] += diff * (w + old_val)
                    else:
                        affected_old[idx] += 2.0 * diff * a[k - j]
                new_max_aff = float(np.max(affected_old))
                if k_min > 0:
                    new_max = max(float(np.max(conv[:k_min])), new_max_aff)
                else:
                    new_max = new_max_aff
                if k_max < 2 * n - 2:
                    new_max = max(new_max, float(np.max(conv[k_max + 1:])))
                new_sum = s + diff
                if new_sum < 1e-10:
                    return np.inf
                return 2.0 * n * new_max / (new_sum * new_sum)

            obj_m1 = eval_at(m1)
            obj_m2 = eval_at(m2)
            if obj_m1 < obj_m2:
                hi = m2
            else:
                lo = m1

        candidates.extend([lo, (lo + hi) / 2, hi])

        best_cand = old_val
        best_cand_obj = best_obj

        for w in candidates:
            if w < 0:
                w = 0.0
            diff = w - old_val
            if abs(diff) < 1e-15:
                continue
            k_min = j
            k_max = min(2 * n - 2, j + n - 1)
            affected_old = conv[k_min:k_max + 1].copy()
            for idx, k in enumerate(range(k_min, k_max + 1)):
                if k == 2 * j:
                    affected_old[idx] += diff * (w + old_val)
                else:
                    affected_old[idx] += 2.0 * diff * a[k - j]
            new_max_aff = float(np.max(affected_old))
            if k_min > 0:
                new_max = max(float(np.max(conv[:k_min])), new_max_aff)
            else:
                new_max = new_max_aff
            if k_max < 2 * n - 2:
                new_max = max(new_max, float(np.max(conv[k_max + 1:])))
            new_sum = s + diff
            if new_sum < 1e-10:
                continue
            new_obj = 2.0 * n * new_max / (new_sum * new_sum)
            if new_obj < best_cand_obj - 1e-15:
                best_cand_obj = new_obj
                best_cand = w

        if best_cand != old_val:
            diff = best_cand - old_val
            k_min = j
            k_max = min(2 * n - 2, j + n - 1)
            for k in range(k_min, k_max + 1):
                if k == 2 * j:
                    conv[k] += diff * (best_cand + old_val)
                else:
                    conv[k] += 2.0 * diff * a[k - j]
            a[j] = best_cand
            s += diff
            best_obj = best_cand_obj
            best_a = a.copy()

    return best_a, conv


def lp_alternate(a, time_budget=200.0, verbose=False):
    """LP alternating with adaptive blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    lp_count = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp(a, max_time=min(60.0, time_budget - (time.monotonic() - start)))
        if g_lp is None:
            break
        lp_count += 1

        # Adaptive blending
        best_local_obj = np.inf
        best_local_a = best_a.copy()

        t_values = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-13:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a.copy()
            stall = 0
            if verbose and lp_count % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP iter {lp_count}: obj={best_obj:.10f}, time={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 5:
                # Perturbation
                rng = np.random.default_rng((n * 104729 + lp_count * 31) % (2**31))
                scale = min(0.03 + 0.01 * (stall // 5), 0.2)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_initialization(n, rng):
    """Generate diverse structured initialization."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-50 * (x - 0.5) ** 2)
    elif choice == 1:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 3 * x)
    elif choice == 2:
        a = np.exp(-80 * (x - 0.3) ** 2) + np.exp(-80 * (x - 0.7) ** 2)
    elif choice == 3:
        # U-shape (high at edges, low in middle)
        a = 1.0 + (2 * x - 1) ** 2
    elif choice == 4:
        # W-shape
        a = 1.0 + (4 * x - 2) ** 4
    else:
        a = np.ones(n)
        for _ in range(3):
            center = rng.integers(0, n)
            width = rng.integers(n // 10, n // 3)
            for i in range(n):
                d = abs(i - center)
                a[i] *= 1.0 + rng.standard_normal() * np.exp(-d / (width * 0.5))
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines LP alternating with efficient coordinate descent
    polishing, multi-n exploration, and diverse initializations.
    Key innovation: O(n^2) coordinate descent that exploits the sparse update
    structure of the convolution when a single element changes.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        print(f"Parent loaded: n={len(a)}, obj={compute_obj(a):.10f}")
    except Exception:
        a = np.ones(800) / 800.0
        print(f"No parent, using uniform: n=800, obj={compute_obj(a):.10f}")

    best_a = a.copy()
    best_obj = compute_obj(a)
    n_base = len(best_a)

    # === Phase 1: LP Alternating ===
    print(f"\n=== Phase 1: LP Alternating (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    a_p1 = lp_alternate(best_a, time_budget=p1_budget, verbose=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1
        print(f"  Phase 1 result: obj={best_obj:.10f} **")
    else:
        print(f"  Phase 1 result: obj={obj_p1:.10f}")

    # === Phase 2: Coordinate Descent Polish ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = min(remaining * 0.15, 120.0)
        conv = compute_conv(best_a)
        a_cd, conv = coordinate_descent(best_a, conv, time_budget=p2_budget, verbose=True)
        obj_cd = compute_obj_from_conv(a_cd, conv)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd
            print(f"  CD result: obj={best_obj:.10f} **")
        else:
            print(f"  CD result: obj={obj_cd:.10f}")

    # === Phase 3: LP Alternating Round 2 ===
    print(f"\n=== Phase 3: LP Alternating Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.20, 200.0)
        a_p3 = lp_alternate(best_a, time_budget=p3_budget, verbose=True)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3
            print(f"  Phase 3 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 3 result: obj={obj_p3:.10f}")

    # === Phase 4: Fine Coordinate Descent ===
    print(f"\n=== Phase 4: Fine Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p4_budget = min(remaining * 0.10, 80.0)
        conv = compute_conv(best_a)
        a_fine, conv = coordinate_descent_fine(best_a, conv, time_budget=p4_budget)
        obj_fine = compute_obj_from_conv(a_fine, conv)
        if obj_fine < best_obj:
            best_obj = obj_fine
            best_a = a_fine
            print(f"  Fine CD result: obj={best_obj:.10f} **")
        else:
            print(f"  Fine CD result: obj={obj_fine:.10f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        max(200, n_base - 200), max(200, n_base - 100), max(200, n_base - 50),
        n_base,
        n_base + 50, n_base + 100, n_base + 200, n_base + 400, n_base + 600,
    ]))
    n_options = [x for x in n_options if 200 <= x <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == len(best_a):
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try

        # Quick LP + CD optimization
        alloc_lp = min(40.0, remaining * 0.2)
        alloc_cd = min(30.0, remaining * 0.1)

        a_opt = lp_alternate(a_try, time_budget=alloc_lp)
        obj_try = compute_obj(a_opt)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.10f} ** (LP)")

        if remaining > 20:
            conv_try = compute_conv(best_a if len(best_a) == n_try else a_opt)
            a_cd2, _ = coordinate_descent(a_opt if obj_try < best_obj else best_a,
                                          compute_conv(a_opt if obj_try < best_obj else best_a),
                                          time_budget=alloc_cd)
            obj_cd2 = compute_obj(a_cd2)
            if obj_cd2 < best_obj:
                best_obj = obj_cd2
                best_a = a_cd2.copy()
                print(f"  ** n={n_try}: obj={best_obj:.10f} ** (CD)")
            elif obj_try >= best_obj:
                print(f"  n={n_try}: obj={max(obj_try, obj_cd2):.10f}")

    # === Phase 6: Diverse Initializations ===
    print(f"\n=== Phase 6: Diverse Initializations ===")

            a_cd3, _ = coordinate_descent(a_opt, conv_div, time_budget=20.0)
            obj_cd3 = compute_obj(a_cd3)
            if obj_cd3 < obj_div:
                a_opt = a_cd3
                obj_div = obj_cd3

        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse init {d}: obj={best_obj:.10f} **")
        else:
            print(f"  Diverse init {d}: obj={obj_div:.10f}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # LP polish
        alloc_lp = min(remaining * 0.3, 60.0)
        a_final = lp_alternate(best_a, time_budget=alloc_lp, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final LP: obj={best_obj:.10f} **")

    # CD polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_cd = min(remaining * 0.3, 50.0)
        conv_final = compute_conv(best_a)
        a_cd_final, _ = coordinate_descent(best_a, conv_final, time_budget=alloc_cd)
        obj_cd_final = compute_obj(a_cd_final)
        if obj_cd_final < best_obj:
            best_obj = obj_cd_final
            best_a = a_cd_final
            print(f"  Final CD: obj={best_obj:.10f} **")

    # Fine CD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_fine = min(remaining * 0.3, 30.0)
        conv_fine = compute_conv(best_a)
        a_fine_final, _ = coordinate_descent_fine(best_a, conv_fine, time_budget=alloc_fine)
        obj_fine_final = compute_obj(a_fine_final)
        if obj_fine_final < best_obj:
            best_obj = obj_fine_final
            best_a = a_fine_final
            print(f"  Final Fine CD: obj={best_obj:.10f} **")

    # One last LP if time allows
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_last = min(remaining * 0.4, 30.0)
        a_last = lp_alternate(best_a, time_budget=alloc_last)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Last LP: obj={best_obj:.10f} **")

    # Final CD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        conv_last = compute_conv(best_a)
        a_last, _ = coordinate_descent(best_a, conv_last, time_budget=min(remaining * 0.5, 15.0))
        obj_last2 = compute_obj(a_last)
        if obj_last2 < best_obj:
            best_obj = obj_last2
            best_a = a_last
            print(f"  Last CD: obj={obj_best:.10f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
