def solve_lp_fast(a):
    """Solve: min M s.t. A@a_conv(z) <= M, sum(z)=1, z>=0 using HiGHS."""
    n = len(a)
    A = build_conv_matrix_fast(a)  # (2n-1) x n
    
    # Variables: z_0, ..., z_{n-1}, M
    # c = [0, ..., 0, 1] (minimize M)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A @ z - M <= 0  =>  [A, -1] @ [z; M] <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(z) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: z_i >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s

    return 2.0 * n * float(np.max(conv))

    """Build (2n-1) x n matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve: min M s.t. (a*z)[k] <= M for all k, sum(z)=1, z>=0.
    Returns (z_normalized, M_opt) or (None, None)."""

    A = build_A(a)
    
    # Variables: [z_0, ..., z_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A @ z - M <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

        zs = float(np.sum(z_val))

            a = a_try
            lr = min(lr * 1.03, 0.3)
            T = min(T * 1.01, 300.0)
            it += 1
        else:
            lr *= 0.85
            if lr < 1e-7:
                break

        if verbose and it % 30 == 0:
            print(f"    Grad: it={it}, obj={best_obj:.8f}, T={T:.0f}, lr={lr:.6f}")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_polish(a, time_budget=30.0):
    """Coordinate-wise descent polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        improved = False
        perm = rng.permutation(n)
        for idx in perm:
            if time.monotonic() - start > time_budget:
                break
            for frac in [0.02, 0.05, 0.1, 0.01]:
                # Try increasing a[idx] by taking from others
                delta = frac * a[idx] if a[idx] > 0 else frac / n
                a_try = a.copy()
                a_try[idx] += delta
                # Distribute reduction among other coordinates proportionally
                others = np.ones(n)
                others[idx] = 0
                total_others = a[idx]  # approximately
                reduction = delta * others / (n - 1)
                a_try -= reduction
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try /= st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Fast LP alternating (HiGHS) + gradient polish +
    multi-n exploration + diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0

    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                print(f"Parent loaded (n={parent.size}): obj={compute_obj(parent):.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(800) / 800
        print(f"Default init (n=800): obj={compute_obj(parent):.8f}")

    best_seq = parent.copy()
    best_score = compute_obj(best_seq)

    # === Phase 1: Intensive LP Alternating ===
    print(f"\n=== Phase 1: LP Alternating (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(450.0, remaining * 0.35)
    a_new = lp_alternating(best_seq, time_budget=alloc, verbose=True)
    obj = compute_obj(a_new)
    if obj < best_score:
        best_score = obj
        best_seq = a_new.copy()
        print(f"  Phase 1 done: obj={best_score:.8f} **")
    else:
        print(f"  Phase 1 done: obj={obj:.8f}")

    # === Phase 2: Gradient Polish ===
    print(f"\n=== Phase 2: Gradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(60.0, remaining * 0.05)
        a_new = gradient_polish(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 2 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 done: obj={obj:.8f}")

    # === Phase 3: Second LP Round ===
    print(f"\n=== Phase 3: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(300.0, remaining * 0.25)
        a_new = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 3 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 done: obj={obj:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_candidates = sorted(set([
        max(400, n_best - 100), max(400, n_best - 50), max(400, n_best - 20),
        n_best + 20, n_best + 50, n_best + 100, n_best + 200,
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 2500]
    n_candidates.sort(key=lambda x: abs(x - n_best))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        if n_try == n_best:
            continue
        a_try = resample(best_seq, n_try)
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            continue
        a_try /= st
        alloc = min(35.0, remaining * 0.08)
        a_opt = lp_alternating(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts ===
    print(f"\n=== Phase 5: Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    max_restarts = min(10, int(remaining // 25))

    for r in range(max_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if r % 4 == 0:
            sigma = 0.02 + 0.008 * r
            a_r = best_seq * np.exp(rng.normal(0, sigma, n_best))
        elif r % 4 == 1:
            # Block perturbation
            a_r = best_seq.copy()
            n_blocks = max(3, n_best // 100)
            block_size = n_best // n_blocks
            for b in rng.choice(n_blocks, min(3, n_blocks), replace=False):
                si = int(b * block_size)
                ei = si + block_size
                a_r[si:ei] *= rng.uniform(0.6, 1.4)
        elif r % 4 == 2:
            # Sparse perturbation
            a_r = best_seq.copy()
            n_sparse = max(5, n_best // 20)
            idx = rng.choice(n_best, n_sparse, replace=False)
            a_r[idx] *= rng.uniform(0.4, 2.0, n_sparse)
        else:
            # Cosine modulation
            freq = rng.uniform(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.15 * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best + phase)
            a_r = best_seq * mod

        a_r = np.maximum(a_r, 0)
        st = float(np.sum(a_r))
        if st < 1e-10:
            continue
        a_r /= st

        alloc = min(remaining * 0.3, 30.0)
        a_opt = lp_alternating(a_r, time_budget=alloc)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 6: Final Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 40.0)
        a_final = lp_alternating(best_seq, time_budget=alloc)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(remaining * 0.4, 20.0)
            a_grad = gradient_polish(best_seq, time_budget=alloc)
            obj_g = compute_obj(a_grad)
            if obj_g < best_score:
                best_score = obj_g
                best_seq = a_grad.copy()
                print(f"  Final Grad: obj={best_score:.8f} **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
