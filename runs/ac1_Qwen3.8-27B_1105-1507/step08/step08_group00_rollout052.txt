"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses spectral initialization (Gerchberg-Saxton), FFT-accelerated smoothed gradient
descent, LP polishing with line search, peak-order LP, and adaptive n exploration.
Key improvements: spectral init for diversity, line search for LP blending,
peak-order LP for flatter peaks, and adaptive tau scheduling.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

try:
    import cvxpy as cp
except ImportError:
    cp = None


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv coefficients using FFT.
    Returns (gradient, convolution).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p_sum = float(np.sum(exp_vals))
    if p_sum < 1e-300:
        p = np.zeros_like(conv)
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / p_sum

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=5.0, tau_end=800.0, verbose=False):
    """
    Nesterov accelerated gradient descent on smoothed objective over simplex.
    Adaptive tau annealing with periodic reset.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 4.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    stalled = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))

        # Multi-stage tau: smooth -> medium -> sharp
        if progress < 0.3:
            tau = tau_init * (10.0 / tau_init) ** (progress / 0.3)
        elif progress < 0.7:
            tau = 10.0 * (100.0 / 10.0) ** ((progress - 0.3) / 0.4)
        else:
            tau = 100.0 * (tau_end / 100.0) ** ((progress - 0.7) / 0.3)

        if it % 5000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient at y

        # Barzilai-Borwein step size

            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / max(abs(sy), 1e-12)
                lr = 0.6 * lr + 0.4 * np.clip(bb_lr, 1e-8, 20.0 / np.sqrt(n))

        # Update
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stalled = 0
        else:
            stalled += 1

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        # Decay learning rate
        lr = max(1e-8, lr * 0.9997)

        # Reset if stuck
        if stalled > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 4.0 / np.sqrt(n) * (0.5 + 0.5 * np.random.random())
            stalled = 0
            best_it = it

    return best_a


def subgradient_refine(a, time_budget=80.0, K=8, verbose=False):
    """
    Subgradient method with top-K peaks, adaptive weights, and momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 3.0 / np.sqrt(n)
    momentum = 0.75
    v = np.zeros(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]

        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break

        # Soft weights based on peak values
        temp = max_pv * 0.3
        weights = np.exp((peak_vals - max_pv) / temp)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + lr * g
        momentum = max(0.5, momentum * 0.999)
        a_new = a - v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 500:
                lr *= 0.5
                v *= 0.2
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.998

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, extra_constraints=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Optionally add extra constraints for peak-order approach.
    Returns (g, M, dual_vars) or (None, None, None).
    """
    if cp is None:
        return None, None, None

    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    lam = cp.Variable(2 * n - 1, nonneg=True)

    constraints = [A @ g <= M, cp.sum(g) == 1]

    if extra_constraints is not None:
        for (k_idx, max_val) in extra_constraints:
            constraints.append(A[k_idx] @ g <= max_val)

    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Recompute actual M
                    conv_check = A @ g_val
                    actual_M = float(np.max(conv_check))
                    dual_vals = None
                    if lam.value is not None:
                        dual_vals = lam.value
                    return g_val, actual_M, dual_vals
        except Exception:
            continue

    return None, None, None


def line_search_blend(a, g, n_evals=15):
    """
    Golden section line search along direction a -> g.
    Finds t in [0, 1] that minimizes obj((1-t)*a + t*g).
    """
    n = len(a)
    a = np.maximum(a, 0)
    g = np.maximum(g, 0)
    s_a = float(np.sum(a))
    s_g = float(np.sum(g))
    if s_a < 1e-10 or s_g < 1e-10:
        return a.copy(), 0.0

    a = a / s_a
    g = g / s_g

    def objective_t(t):
        a_t = (1.0 - t) * a + t * g
        s_t = float(np.sum(a_t))
        if s_t < 1e-10:
            return np.inf
        a_t = a_t / s_t
        conv = compute_conv(a_t)
        return 2.0 * n * float(np.max(conv))

    # Golden section search on [0, 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_lo = 0.0
    a_hi = 1.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = objective_t(c)
    fd = objective_t(d)

    for _ in range(n_evals):
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = objective_t(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = objective_t(d)

    t_opt = (a_lo + a_hi) / 2.0
    a_opt = (1.0 - t_opt) * a + t_opt * g
    s_opt = float(np.sum(a_opt))
    if s_opt > 1e-10:
        a_opt = a_opt / s_opt
    return a_opt, t_opt

def peak_order_lp(a, time_budget=100.0, max_rounds=20, verbose=False):

                break

    return best_a


def spectral_init(n, rng, iterations=50, target_flatness=0.0):
    """
    Gerchberg-Saxton style spectral initialization.
    Tries to find a non-negative sequence with low autocorrelation peaks.
    """
    x = np.zeros(n)

    # Random phase initialization
    phase = rng.uniform(0, 2 * np.pi, n)
    magnitude = np.ones(n) / np.sqrt(n)
    # Make DC component consistent
    magnitude[0] = 1.0  # DC = sum(a) = 1

    for it in range(iterations):
        # IFFT to get time domain
        spectrum = magnitude * np.exp(1j * phase)
        x = np.real(ifft(spectrum))

        # Project to non-negative and normalize
        x = np.maximum(x, 0)
        s = float(np.sum(x))
        if s < 1e-10:
            x = np.ones(n) / n
            break
        x = x / s

        # Compute new spectrum
        X = fft(x)
        new_phase = np.angle(X)
        new_mag = np.abs(X)

        # Target: flatten the spectrum (reduce peaks)
        # Blend current magnitude with uniform magnitude
        target_mag = np.ones(n) / np.sqrt(n)
        target_mag[0] = 1.0  # Keep DC fixed

        # Adaptive blending: if peaks are too high, flatten more aggressively
        max_mag = float(np.max(new_mag))
        mean_mag = float(np.mean(new_mag))
        if max_mag > mean_mag * (1.0 + target_flatness):
            blend = 0.3
        else:
            blend = 0.1

        magnitude = (1.0 - blend) * new_mag + blend * target_mag
        phase = new_phase

    return x


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n) * 0.5
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        # Bell-shaped
        c = n // 2
        w = n // 4
        a = np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Two peaks
        c1 = n // 5
        c2 = 4 * n // 5
        w = n // 15
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent_refine(a, time_budget=60.0, verbose=False):

        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Spectral initialization + quick GD ===
    print(f"\n=== Phase 1: Spectral Init + GD ===")
    remaining = deadline - time.monotonic()

    spectral_candidates = []
    n_spectral = min(4, int(remaining / 40))
    for i in range(n_spectral):
        if time.monotonic() > deadline - 60:
            break
        flatness = 0.1 + 0.1 * i
        a_spec = spectral_init(n_best, rng, iterations=40 + 10 * i, target_flatness=flatness)
        obj_spec = compute_obj(a_spec)
        spectral_candidates.append((a_spec, obj_spec))
        print(f"  Spectral {i}: obj={obj_spec:.8f}, flatness={flatness:.1f}")

    # Pick best spectral candidate and blend with parent
    if spectral_candidates:
        spectral_candidates.sort(key=lambda x: x[1])
        best_spec = spectral_candidates[0][0]
        # Try different blends
        for t in [0.1, 0.2, 0.3, 0.5]:
            a_blend = (1 - t) * best_seq + t * best_spec
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            obj_blend = compute_obj(a_blend)
            if obj_blend < best_score:
                best_score = obj_blend
                best_seq = a_blend.copy()
                print(f"  Blend t={t}: obj={best_score:.8f} **")

    # Quick smoothed GD on current best
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 200.0)
    if p1_budget > 30:
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                          tau_init=5.0, tau_end=500.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 GD: obj={obj_sgd:.8f}")

    # === Phase 2: LP polishing with line search ===
    print(f"\n=== Phase 2: LP + Line Search ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 450.0)
    if p2_budget > 30:
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(p2_budget / (4 - rnd), remaining * 0.7)
            if alloc < 20:
                break
            a_lp = lp_polish_with_line_search(best_seq, time_budget=alloc,
                                              max_iters=50, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                # Perturb to escape
                scale = 0.015 * (rnd + 1)
                best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                best_seq = np.maximum(best_seq, 0)
                best_seq = best_seq / best_seq.sum()
                if rnd >= 3:
                    break

    # === Phase 3: Peak-order LP ===
    print(f"\n=== Phase 3: Peak-Order LP ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 20:
        a_po = peak_order_lp(best_seq, time_budget=p3_budget,
                             max_rounds=15, verbose=True)
        obj_po = compute_obj(a_po)
        if obj_po < best_score:
            best_score = obj_po
            best_seq = a_po.copy()
            print(f"  Peak-order: obj={best_score:.8f} **")
        else:
            print(f"  Peak-order: obj={obj_po:.8f}")

    # === Phase 4: Subgradient + Coordinate Descent ===
    print(f"\n=== Phase 4: Subgradient + Coord Descent ===")
    remaining = deadline - time.monotonic()

    p4a_budget = min(remaining * 0.15, 120.0)
    if p4a_budget > 20:
        a_sg = subgradient_refine(best_seq, time_budget=p4a_budget, K=8, verbose=False)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    remaining = deadline - time.monotonic()
    p4b_budget = min(remaining * 0.1, 60.0)
    if p4b_budget > 15:
        a_cd = coordinate_descent_refine(best_seq, time_budget=p4b_budget, verbose=False)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CoordDesc: obj={best_score:.8f} **")
        else:
            print(f"  CoordDesc: obj={obj_cd:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 200), max(500, n_current - 100),
        n_current + 100, n_current + 200, n_current + 400,
        1000, 1200
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1500 and abs(n - n_current) >= 100]
    n_options = n_options[:4]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick GD
        alloc1 = min(remaining * 0.2, 50.0)
        if alloc1 > 15:
            a_sgd = nesterov_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=300.0)
            if compute_obj(a_sgd) < compute_obj(a_try):
                a_try = a_sgd

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 40.0)
        if alloc2 > 15:
            a_try = lp_polish_with_line_search(a_try, time_budget=alloc2, max_iters=20)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r * 3) % 9
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.3, 80.0)
        if alloc > 20:
            half = alloc * 0.5
            a_r = nesterov_smoothed_descent(a_r, time_budget=half,
                                            tau_init=10.0, tau_end=200.0)
            remaining = deadline - time.monotonic()
            lp_alloc = min(remaining * 0.2, 40.0)
            if lp_alloc > 10:
                a_r = lp_polish_with_line_search(a_r, time_budget=lp_alloc, max_iters=15)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau smoothed GD
    if remaining > 20:
        alloc = min(remaining * 0.25, 80.0)
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                          tau_init=50.0, tau_end=1000.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.35, 100.0)
        a_lp = lp_polish_with_line_search(best_seq, time_budget=alloc, max_iters=30, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Subgradient
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.25, 40.0)
        a_sg = subgradient_refine(best_seq, time_budget=alloc, K=10)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Peak-order
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 50.0)
        a_po = peak_order_lp(best_seq, time_budget=alloc, max_rounds=10)
        obj_po = compute_obj(a_po)
        if obj_po < best_score:
            best_score = obj_po
            best_seq = a_po.copy()
            print(f"  Final PO: obj={best_score:.8f} **")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 50.0)
        a_lp = lp_polish_with_line_search(best_seq, time_budget=alloc, max_iters=20)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
