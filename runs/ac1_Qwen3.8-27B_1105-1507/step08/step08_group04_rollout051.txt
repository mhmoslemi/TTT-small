  Let me just use the product rule:
  (a*a)_k = sum_l a[l]*a[k-l]
  ∂(a*a)_k/∂a[j] = sum_l (δ(l,j)*a[k-l] + a[l]*δ(k-l,j))

  But if k-j = j-k, i.e., 2j=k, then we're counting a[j]*a[j] twice, which is correct because:
  (a*a)_k contains the term a[j]*a[j] when l=j, and ∂/∂a[j] gives a[k-j] + a[j-k] = a[j] + a[j] = 2a[j]. ✓

OK so the KKT condition for j with a[j] > 0 is:
sum_{i=1}^{m} μ_i * (a[k_i - j] + a[j - k_i]) = λ

where a[x] = 0 for x notin [0, n-1].

And for j with a[j] = 0:
sum_{i=1}^{m} μ_i * (a[k_i - j] + a[j - k_i]) <= λ

Plus sum(a) = 1, and (a*a)_{k_i} = M for all i.

The issue is step 2. If we could exactly optimize a given g, the method would be a proper block coordinate descent and would converge (to a local optimum) monotonically. But optimizing a given g is the same problem (min max_k (g*a)_k s.t. sum(a)=1, a>=0), which is non-convex.

However, we can approximate it: the "gradient flow" direction for optimizing a given g is:
a_new[j] = a[j] - η * sum_k y_k * (g[k-j] + g[j-k])

where y_k are the "dual weights" (proportional to how much each peak exceeds the target).

This is essentially what the subgradient method does.

So the full algorithm is:
1. Solve LP to get g and dual y
2. Use y to compute subgradient of the objective at a
3. Take a step in the negative subgradient direction (with projection)
4. Repeat

This is actually the same as just doing the subgradient step, using the LP dual as the subgradient weights!

Let me formalize:
- The objective (after normalization) is f(a) = 2n * max_k h_k(a) where h_k(a) = (a*a)_k
- The subgradient of f at a is: ∂f(a) = {2n * sum_k α_k * ∇h_k(a) : α_k >= 0, sum α_k = 1, α_k > 0 only if h_k(a) = max}
- The LP dual gives us exactly such an α! (The dual variables of the active constraints)

So the algorithm is:
1. Compute conv(a,a), find max and active set S
2. Solve LP to get dual weights α (or just set α uniformly over S if LP is too slow)
3. Compute subgradient: grad[j] = 2n * sum_{k in S} α_k * (a[k-j]*1[0<=k-j<n] + a[j-k]*1[0<=j-k<n])
4. Take a step: a_new = a - η * grad
5. Project onto simplex: a_new = max(a_new, 0), normalize
6. Repeat

The key advantage of using the LP dual is that it gives the "best" subgradient direction (the one that most reduces the objective).

def compute_subgradient_fast(a, active_indices, alpha):
    """Fast subgradient computation using vectorization."""
    n = len(a)
    grad = np.zeros(n)
    for idx, k in enumerate(active_indices):
        w = alpha[idx]
        if w < 1e-15:
            continue
        # First part: a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # We need a[k-j] for j = j_start, ..., j_end
            # = a[k-j_end], a[k-j_end+1], ..., a[k-j_start]
            # In numpy indexing: a[k-j_end : k-j_start+1]
            vals = a[k-j_end : k-j_start+1]  # This gives a[k-j_end], a[k-j_end+1], ..., a[k-j_start]
            # But we want to add vals[j - j_start] to grad[j] for j in [j_start, j_end]
            # vals[0] = a[k-j_end], which should go to grad[j_end]
            # vals[j_end-j_start] = a[k-j_start], which should go to grad[j_start]
            # So: grad[j_start:j_end+1] += w * vals[::-1]
            grad[j_start:j_end+1] += w * vals[::-1]
        
        # Second part: a[j-k] for j in [max(k,0), n-1]
        if k < n:
            # a[j-k] for j = k, k+1, ..., n-1
            # = a[0], a[1], ..., a[n-1-k]
            vals2 = a[:n-k]
            grad[k:n] += w * vals2
    
    return grad * 2.0 * n

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with reduced constraints, dual-based subgradient,
backtracking line search, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
from scipy.optimize import linprog


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_subgradient(a, active_idx, weights):
    """Compute subgradient of max(conv(a,a)) at a using given active set and weights."""
    n = len(a)
    grad = np.zeros(n)
    for idx, k in enumerate(active_idx):
        w = weights[idx]
        if w < 1e-15:
            continue
        k = int(k)
        # Part 1: a[k-j] for valid j
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # grad[j_start:j_end+1] += w * [a[k-j_end], a[k-j_end+1], ..., a[k-j_start]]
            # = w * a[k-j_end : k-j_start+1][::-1]
            grad[j_start:j_end+1] += w * a[k - j_end : k - j_start + 1][::-1]
        # Part 2: a[j-k] for valid j (j >= k, j < n)
        if k < n:
            grad[k:n] += w * a[:n - k]
    return grad * 2.0 * n


def solve_lp_reduced(a, active_idx, time_limit=10.0):
    """
    Solve reduced LP: min M s.t. conv(a,g)_k <= M for k in active_idx,
    sum(g) = 1, g >= 0.
    Returns (g, M, dual_weights_on_active) or (None, None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    full_n = 2 * n - 1
    active_set = set(int(k) for k in active_idx)
    
    # Build matrix A: rows are active constraints, columns are g variables
    # A[k_active, j] = a[k_active - j] if 0 <= k_active - j < n else 0
    # ... (this is the same as the full matrix but with fewer rows)
    
    A_rows = []
    A_cols = []
    A_vals = []
    row_map = {}
    for i, k in enumerate(sorted(active_set)):
        row_map[k] = i
        for j in range(n):
            val = a[k - j] if 0 <= k - j < n else 0.0
            if val > 0:
                A_rows.append(i)
                A_cols.append(j)
                A_vals.append(val)
    
    from scipy.sparse import csr_matrix
    m_active = len(active_set)
    A = csr_matrix((A_vals, (A_cols, A_rows)), shape=(n, m_active)).T  # (m_active, n)
    
    # LP: min M s.t. A@g - M <= 0, sum(g) = 1, g >= 0, M >= 0
    # Variables: [g (n vars), M (1 var)]
    # Objective: [0, ..., 0, 1]
    # Inequality: A@g - M <= 0 => A@g + (-1)*M <= 0
    # Equality: sum(g) = 1
    
    # Use scipy linprog (more reliable for this formulation)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack((A.toarray(), -np.ones((m_active, 1))))
    b_ub = np.zeros(m_active)
    A_eq = np.hstack((np.ones((1, n)), np.zeros((1, 1))))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g_val))
        if gs > 1e-10:
            g_val = g_val / gs
            M_val = float(res.x[n])
            # Get dual variables for inequality constraints
            dual = np.abs(getattr(res, 'ineqlin', None)) if hasattr(res, 'ineqlin') and res.ineqlin is not None else None
            if dual is not None and np.sum(dual) > 1e-10:
                dual = dual / np.sum(dual)
            else:
                dual = np.ones(m_active) / m_active
            return g_val, M_val, dual, sorted(active_set)
    
    return None, None, None, None


def solve_lp_full(a, time_limit=10.0):
    """Solve full LP using cvxpy."""
    n = len(a)
    a = a / np.sum(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # Build constraint matrix
    A = np.zeros((2*n-1, n))
    for j in range(n):
        k_start = j
        k_end = min(2*n-2, j+n-1)
        A[k_start:k_end+1, j] = a[:k_end-k_start+1]
    
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > time_limit:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value)
                    dual_y = constraints[0].dual_value
                    if dual_y is not None:
                        dual_y = np.maximum(dual_y, 0)
                        if np.sum(dual_y) > 1e-10:
                            dual_y = dual_y / np.sum(dual_y)
                        else:
                            dual_y = np.ones(2*n-1) / (2*n-1)
                    else:
                        dual_y = np.ones(2*n-1) / (2*n-1)
                    return g_val, M_val, dual_y
        except Exception:
            continue
    return None, None, None
