def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))²."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_A(a):
    """Build convolution matrix: A[k,j] = a[k-j] if valid, else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a, g_init=None):
    """
    Solve: min M s.t. A@g <= M, sum(g)=1, g>=0.
    Returns (g_opt, M_opt).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_A(a_norm)
    
    # Decision variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True})
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    
    return None, None

        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_A(a):
    """Build convolution matrix A where (A@g) = conv(a, g). Size (2n-1) x n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, use_lp=True):
    """
    Solve: min M s.t. A(a)@g <= M, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_A(a_norm)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})

    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])

    return None, None


def line_search(a, g, n_points=16):
    """Find best t in [0,1] minimizing obj((1-t)*a + t*g). Returns (best_t, best_a, best_obj)."""
    best_obj = np.inf
    best_t = 0.0
    best_a = a.copy()

    ts = np.linspace(0, 1, n_points)
    for t in ts:
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
            best_a = a_try.copy()

    # Refine around best
    if 0 < best_t < 1:
        lo = max(0, best_t - 1.0 / n_points)
        hi = min(1, best_t + 1.0 / n_points)
        for t in np.linspace(lo, hi, 11):
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_t = t
                best_a = a_try.copy()

    return best_t, best_a, best_obj


def lp_alternate(a, time_budget, verbose=False):
    """
    Main LP alternating optimization loop.
    Alternates between solving LP for g (fixed a) and updating a via line search.
    Includes dual LP (solve for a' given g) and annealing for escape.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall = 0
    patience = 25
    temperature = 0.1
    iteration = 0
    best_t_recent = 0.3
    rng = np.random.default_rng(42 + n)

    while time.monotonic() - start < time_budget:
        # Solve LP for g given current a
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Also solve dual LP: find a' minimizing max conv(g, a')
        a_dual, M_dual = solve_lp(g_lp)

        # Build candidate update directions
        candidates = [g_lp]
        if a_dual is not None:
            candidates.append(a_dual)
            # Also blend of g_lp and a_dual
            candidates.append(0.5 * g_lp + 0.5 * a_dual)

        best_t = 0.0
        best_a_new = best_a.copy()
        best_obj_new = best_obj

        for cand in candidates:
            cand = np.maximum(cand, 0)
            sc = float(np.sum(cand))
            if sc < 1e-10:
                continue
            cand = cand / sc
            t, a_new, obj_new = line_search(a, cand, n_points=12)
            if obj_new < best_obj_new:
                best_obj_new = obj_new
                best_a_new = a_new
                best_t = t

        iteration += 1

        if best_obj_new < best_obj - 1e-12:
            # Accept
            improvement = best_obj - best_obj_new
            best_obj = best_obj_new
            best_a = best_a_new.copy()
            a = best_a_new.copy()
            best_t_recent = best_t
            stall = 0
            temperature = max(0.005, temperature * 0.998)
        else:
            stall += 1
            if stall >= patience:
                # Anneal: perturb and restart
                noise = temperature * rng.normal(0, 1, n)
                a_pert = best_a * (1.0 + noise)
                a_pert = np.maximum(a_pert, 0)
                sp = float(np.sum(a_pert))
                if sp > 1e-10:
                    a_pert = a_pert / sp
                else:
                    a_pert = best_a.copy()

                # Also try random direction blend
                if rng.random() < 0.4:
                    rand_dir = rng.dirichlet(np.ones(n) * 0.5)
                    blend = 0.1 + 0.2 * rng.random()
                    a_pert = (1 - blend) * a_pert + blend * rand_dir
                    a_pert = np.maximum(a_pert, 0)
                    sp = float(np.sum(a_pert))
                    if sp > 1e-10:
                        a_pert = a_pert / sp

                a = a_pert
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                temperature = min(0.25, temperature * 1.8)
                best_t_recent = 0.5

            else:
                temperature = max(0.01, temperature * 0.995)

        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP it{iteration}: obj={best_obj:.8f}, stall={stall}, "
                  f"temp={temperature:.4f}, t={best_t_recent:.3f}, elapsed={elapsed:.1f}s")

    return best_a


def subgradient_polish(a, time_budget, verbose=False):
    """
    Subgradient / mirror descent polish using top-K active constraints.
    Faster than LP but less precise - good for final polish.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 5.0 / np.sqrt(n)
    momentum = 0.0
    v = np.zeros(n)
    stall = 0

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Top-K active constraints
        K = min(5, 2 * n - 1)
        top_idx = np.argsort(conv)[::-1][:K]
        max_val = conv[top_idx[0]]
        if max_val < 1e-15:
            break

        # Softmax weights over top-K
        shifted = (conv[top_idx] - max_val) / (max_val * 0.1 + 1e-15)
        weights = np.exp(shifted * 5.0)
        weights /= weights.sum()

        # Gradient: average of gradients of active conv components
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            # grad of conv[k] w.r.t. a[j] = a[k-j] (with bounds, doubled if k-j != j)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for j in range(i_start, i_end + 1):
                partner = k - j
                if 0 <= partner < n:
                    if j == partner:
                        g[j] += w * 0.5 * a[j]
                    else:
                        g[j] += w * a[partner]

        v = momentum * v + g
        log_a = np.log(a)
        log_a_new = log_a - lr * v
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            momentum = min(0.8, momentum + 0.001)
            lr *= 0.9995
        else:
            stall += 1
            if stall > 100:
                rng = np.random.default_rng(it + n)
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                momentum = 0.0
                lr = 5.0 / np.sqrt(n)
                stall = 0
            else:
                lr = max(0.01, lr * 0.98)

        it += 1
        if verbose and it % 200 == 0:
            print(f"      SG it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a


def coordinate_descent_polish(a, time_budget):
    """
    Coordinate descent: redistribute mass from high-convolution contributors
    to low-convolution positions.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    max_rounds = 20
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to conv[k_star]
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        # Find low-conv positions (good targets)
        low_positions = np.argsort(conv)[:20]

        improved = False
        for contrib, i, j in contributors[:20]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break

            for target in low_positions[:6]:
                target = int(target)
                if target == i or target == j:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    src = i if a[i] >= a[j] else j
                    if src == target:
                        continue
                    delta = frac * a[src]
                    if delta < 1e-15 or delta >= a[src]:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations for exploration."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        x = np.arange(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Triangle
        x = np.linspace(-1, 1, n)
        a = np.maximum(0.01, 1.0 - np.abs(x))
        a = a / a.sum()
    elif mode == 4:
        # Arcsine-like (U-shaped)
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x ** 2 + 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Random Fourier
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Skewed geometric
        r = rng.uniform(0.9, 0.99)
        a = r ** np.arange(n)
        a = a / a.sum()
    elif mode == 7:
        # Middle-heavy
        x = np.arange(n) / (n - 1)
        a = np.exp(-2.0 * (x - 0.5) ** 2)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines LP alternating optimization, subgradient
    polish, coordinate descent, multi-length exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0  # 20s safety buffer

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_seq = parent.copy()
                best_score = obj
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"Initial: n={n_current}, obj={best_score:.8f}")

    # --- Phase 1: Main LP optimization on current length ---
    print(f"\n=== Phase 1: LP Optimization (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 450.0)
    if p1_budget > 60:
        a_opt = lp_alternate(best_seq, time_budget=p1_budget, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_opt:.8f} (no improvement)")

    # --- Phase 2: Subgradient polish ---
    print(f"\n=== Phase 2: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.1, 100.0)
    if p2_budget > 20:
        a_sg = subgradient_polish(best_seq, time_budget=p2_budget, verbose=False)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_sg:.8f} (no improvement)")

    # --- Phase 3: Second LP round with perturbation ---
    print(f"\n=== Phase 3: LP Round 2 (perturbed) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 250.0)
    if p3_budget > 30:
        # Several perturbed restarts
        for restart in range(3):
            remaining = deadline - time.monotonic()
            alloc = p3_budget / (3 - restart)
            if alloc < 30 or remaining < 30:
                break
            alloc = min(alloc, remaining)

            if restart == 0:
                a_start = best_seq.copy()
            else:
                noise_scale = 0.03 + 0.02 * restart
                a_start = best_seq * (1.0 + noise_scale * rng.normal(0, 1, n_current))
                a_start = np.maximum(a_start, 0)
                sa = float(np.sum(a_start))
                if sa > 1e-10:
                    a_start = a_start / sa

            a_opt = lp_alternate(a_start, time_budget=alloc, verbose=(restart == 0))
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Round {restart}: obj={best_score:.8f} **")
            else:
                print(f"  Round {restart}: obj={obj_opt:.8f}")

    # --- Phase 4: Length exploration ---
    print(f"\n=== Phase 4: Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = []
    for dn in [-100, -50, 50, 100, 150, 200, -200]:
        n_try = n_best + dn
        if 300 <= n_try <= 2000 and n_try != n_best:
            n_options.append(n_try)
    # Add some fixed options
    for n_fix in [500, 700, 900, 1000, 1200]:
        if 300 <= n_fix <= 2000 and n_fix != n_best and n_fix not in n_options:
            n_options.append(n_fix)

    n_options = sorted(set(n_options))[:8]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 100.0)

        a_opt = lp_alternate(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # --- Phase 5: Diverse restarts ---
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_use = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r + 1) % 8
        a_r = diverse_init(n_use, rng, mode=mode)
        # Blend with best (more weight on diverse as r increases)
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * best_seq + blend * a_r
        a_r = np.maximum(a_r, 0)
        sr = float(np.sum(a_r))
        if sr > 1e-10:
            a_r = a_r / sr

        alloc = min(remaining * 0.15, 120.0)
        a_opt = lp_alternate(a_r, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_opt:.8f}")

    # --- Phase 6: Final intensive polish ---
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    # LP polish
    if remaining > 50:
        alloc = min(remaining * 0.3, 150.0)
        a_opt = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 80.0)
        a_sg = subgradient_polish(best_seq, time_budget=alloc)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 60.0)
        a_cd = coordinate_descent_polish(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more short LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 60.0)
        a_opt = lp_alternate(best_seq, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
