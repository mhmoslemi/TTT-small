```python
"""
Optimized step function search using L-BFGS with log-sum-exp smoothing,
softmax parameterization, multi-start, and LP-based restarts.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fft_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-15:
        return np.inf
    a_norm = a / S
    conv = fft_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    return exp_x / np.sum(exp_x)


def lbs_obj_and_grad(x, T):
    """
    Objective and gradient for L-BFGS in softmax parameter space.
    a = softmax(x), objective = 2n * (1/T) * logsumexp(T * conv(a,a))
    """
    n = len(x)
    a = softmax(x)
    
    # Convolution
    conv = fft_conv(a)
    
    # Log-sum-exp: (1/T)*log(sum(exp(T*conv)))
    max_c = float(np.max(conv))
    shifted = T * (conv - max_c)
    # For numerical stability, clip shifted
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(np.sum(exp_shifted))
    f_T = max_c + log_sum / T
    
    # Objective
    obj = 2.0 * n * f_T
    
    # Probabilities p_k = exp(T*c_k) / sum(exp(T*c_j))
    p = exp_shifted / np.sum(exp_shifted)
    
    # Compute g[i] = sum_{j=0}^{n-1} a[j] * p[i+j]
    g = np.zeros(n)
    # Direct computation
    for i in range(n):
        # sum_{j=0}^{n-1} a[j] * p[i+j] where i+j < 2n-1
        max_j = min(n, 2*n-1-i)
        g[i] = np.dot(a[:max_j], p[i:i+max_j])
    
    # dJ/da_i = 4n * g[i] (since dJ/da_i = 2n * 2 * sum_k p[k]*a[k-i])
    dJ_da = 4.0 * n * g
    
    # Gradient w.r.t. x:
    # dJ/dx_k = a_k * (dJ/da_k - sum_i a_i * dJ/da_i)
    mean_term = float(np.dot(a, dJ_da))
    grad_x = a * (dJ_da - mean_term)
    
    return obj, grad_x


def lbs_optimize(a_init, T_schedule, max_iter_per_T=200, time_budget=300.0):
    """
    Run L-BFGS with annealing schedule for T.
    a_init: initial normalized sequence (sum=1)
    T_schedule: list of T values to anneal through
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Convert to softmax parameters
    # x = log(a) works if a > 0; for a=0, use a large negative
    a_safe = np.maximum(a, 1e-15)
    x = np.log(a_safe)
    
    start = time.monotonic()
    
    for T in T_schedule:
        if time.monotonic() - start > time_budget:
            break
        
        result = minimize(
            lbs_obj_and_grad,
            x,
            args=(T,),
            method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': max_iter_per_T,
                'ftol': 1e-14,
                'gtol': 1e-10,
                'maxls': 50,
            }
        )
        
        x = result.x
        a_new = softmax(x)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        print(f"    T={T}: L-BFGS obj_smooth={result.fun:.8f}, obj_true={obj_new:.8f}, "
              f"best={best_obj:.8f}, iters={result.nit}, status={result.status}")
    
    return best_a


def lp_response_and_update(a, time_budget=60.0):
    """
    Use LP to find optimal response, then blend.
    """
    from scipy.optimize import linprog
    
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    a = a / S
    
    # Build convolution matrix: A[i,j] such that (A@g)[k] = conv(a,g)[k]
    # conv(a,g)[k] = sum_j a[j] * g[k-j]
    # For a fixed a, this is linear in g.
    # (A@g)[k] = sum_j A[k,j] * g[j]
    # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    
    m = 2 * n - 1
    # LP: min M s.t. conv(a,g) <= M, sum(g) = 1, g >= 0
    # Variables: g[0..n-1], M
    # Total variables: n+1
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    # Inequality constraints: conv(a,g) - M <= 0 for each k
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for k in range(m):
        for j in range(n):
            diff = k - j
            if 0 <= diff < n:
                A_ub[k, j] = a[diff]
        A_ub[k, n] = -1.0
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': time_budget})
    
    if res.status != 0 or res.x is None:
        return a
    
    g = np.maximum(res.x[:n], 0)
    gs = np.sum(g)
    if gs < 1e-10:
        return a
    g = g / gs
    
    # Try blending
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        a_try = (1-t)*a + t*g
        a_try = np.maximum(a_try, 0)
        st = np.sum(a_try)
        if st < 1e-10:
            continue
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
    
    return best_a


def generate_diverse_init(n, kind, rng):
    """Generate diverse initialization."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.01
        return a / np.sum(a)
    elif kind == 'gaussian':
        x = np.arange(n, dtype=float)
        center = n / 2.0
        a = np.exp(-0.5 * ((x - center) / (n/5.0)) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'two_peaks':
        a = np.ones(n) * 0.3
        a[n//4] = 1.0
        a[3*n//4] = 1.0
        # Smooth
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=n//20)
        return a / np.sum(a)
    elif kind == 'triangle':
        x = np.arange(n, dtype=float)
        center = n / 2.0
        a = 1.0 - np.abs(x - center) / center
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif kind == 'ramanujan':
        # Based on Ramanujan's idea of flat autocorrelation
        x = np.arange(n, dtype=float)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x / n * 3) + 0.1 * np.cos(2 * np.pi * x / n * 7)
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_polish(a, time_budget=30.0):
    """
    Direct coordinate descent on the original (non-smooth) objective.
    Tries to reduce the peak by adjusting individual coordinates.
    """
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv = fft_conv(a)
    
    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))
        
        # Find which indices contribute most to the peak
        # conv[k_star] = sum_j a[j] * a[k_star - j]
        contrib = np.zeros(n)
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib[j] = a[j] * a[k2]
        
        # Try reducing high-contributing elements
        sorted_idx = np.argsort(contrib)[::-1][:30]
        improved = False
        
        for i in sorted_idx:
            if time.monotonic() - start > time_budget:
                break
            if a[i] < 1e-10:
                continue
            
            # Try moving mass from i to a "far" position
            # Far means: position j where a[j]*a[k_star-j] is small
            best_delta = None
            best_new_obj = best_obj
            
            # Try a few target positions
            targets = [(i + n//2) % n, (i + n//3) % n, (i + 2*n//3) % n, (i + 1) % n, (i - 1 + n) % n]
            targets = [t for t in set(targets) if t != i]
            
            for delta_frac in [0.001, 0.005, 0.01, 0.02]:
                delta = delta_frac * a[i]
                if delta <= 0 or delta >= a[i]:
                    continue
                for target in targets[:3]:
                    a[i] -= delta
                    a[target] += delta
                    conv_new = fft_conv(a)
                    obj_new = 2.0 * n * np.max(conv_new)
                    if obj_new < best_new_obj:
                        best_new_obj = obj_new
                        best_delta = (i, target, delta)
                    a[i] += delta
                    a[target] -= delta
            
            if best_delta is not None:
                i_d, target_d, delta_d = best_delta
                a[i_d] -= delta_d
                a[target_d] += delta_d
                conv = fft_conv(a)
                S = np.sum(a)
                a = a / S
                best_obj = compute_obj(a)
                best_a = a.copy()
                improved = True
        
        if not improved:
            # Try random perturbations
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            for _ in range(3):
                i = rng.integers(0, n)
                if a[i] < 1e-10:
                    continue
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.005 * a[i]
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                S = np.sum(a)
                a = a / S
                obj_new = compute_obj(a)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
                else:
                    # Revert
                    a = best_a.copy()
            break
    
    return best_a
```

```python
"""
Optimized step function search using L-BFGS with log-sum-exp smoothing,
softmax parameterization, LP-based kicks, diverse multi-start, and coordinate polish.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def fft_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-15:
        return np.inf
    a_norm = a / S
    conv = fft_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_fn(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    s = np.sum(exp_x)
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s

    """
    n = len(x)
    a = softmax_fn(x)

    # Convolution
    conv = fft_conv(a)

    # Log-sum-exp: f_T = max_c + (1/T)*log(sum(exp(T*(conv - max_c))))
    max_c = float(np.max(conv))
    shifted = T * (conv - max_c)
    shifted = np.clip(shifted, -600, 600)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    if sum_exp < 1e-300:
        sum_exp = 1e-300
    log_sum = np.log(sum_exp)
    f_T = max_c + log_sum / T

    # Objective
    obj = 2.0 * n * f_T

    # Probabilities p_k
    p = exp_shifted / sum_exp

    # Compute g[i] = sum_{j=0}^{n-1} a[j] * p[i+j]
    g = np.zeros(n)
    for i in range(n):
        # p[i:i+n] has length n, a has length n
        g[i] = np.dot(a, p[i:i + n])

    # dJ/da_i = 4n * g[i]
    dJ_da = 4.0 * n * g

    # Gradient w.r.t. x:
    mean_term = float(np.dot(a, dJ_da))
    grad_x = a * (dJ_da - mean_term)

    return obj, grad_x


def lbs_optimize(a_init, T_schedule, max_iter_per_T=200, time_budget=300.0, verbose=True):
    """
    Run L-BFGS with annealing schedule for T.
    a_init: initial normalized sequence (sum=1)
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)

    # Convert to softmax parameters
    a_safe = np.maximum(a, 1e-15)
    x = np.log(a_safe)

    start = time.monotonic()

    for T in T_schedule:
        if time.monotonic() - start > time_budget:
            break

        result = minimize(
            lbs_obj_and_grad,
            x,
            args=(T,),
            method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': max_iter_per_T,
                'ftol': 1e-15,
                'gtol': 1e-12,
                'maxls': 100,
            }
        )

        x = np.asarray(result.x)
        a_new = softmax_fn(x)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        if verbose:
            print(f"    T={T}: smooth={result.fun:.8f}, true={obj_new:.8f}, "
                  f"best={best_obj:.8f}, iters={result.nit}, status={result.status}")

    return best_a


def lp_response(a, time_budget=30.0):
    """
    Solve LP: min M s.t. conv(a,g) <= M for all k, sum(g)=1, g>=0.
    Returns optimal g or None.
    """
    n = len(a)
    a = a.copy()
    S = float(np.sum(a))
    if S < 1e-12:
        return None
    a = a / S

    m = 2 * n - 1

    # Build constraint matrix efficiently
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((m, n + 1))
    for k in range(m):
        for j in range(n):
            d = k - j
            if 0 <= d < n:
                A_ub[k, j] = a[d]
        A_ub[k, n] = -1.0

    b_ub = np.zeros(m)
    c = np.zeros(n + 1)
    c[n] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': max(5, time_budget)})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g))
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass

    return None


def lp_kick(a, time_budget=60.0):
    """LP kick: solve LP, blend, return improved sequence."""
    g = lp_response(a, time_budget=time_budget)
    if g is None:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)

    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            continue
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a


def generate_init(n, kind, rng):
    """Generate diverse initialization of length n."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.01
        return a / np.sum(a)
    elif kind == 'sine':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.01
        return a / np.sum(a)
    elif kind == 'gaussian':
        x = np.arange(n, dtype=float)
        center = n / 2.0
        a = np.exp(-0.5 * ((x - center) / (n / 5.0)) ** 2)
        return a / np.sum(a)
    elif kind == 'two_peaks':
        a = np.ones(n) * 0.2 + 0.01
        a[int(n * 0.25)] = 1.0
        a[int(n * 0.75)] = 1.0
        # Smooth with moving average
        kernel_size = max(1, n // 30)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'triangle':
        x = np.arange(n, dtype=float)
        center = n / 2.0
        a = 1.0 - np.abs(x - center) / center + 0.01
        return a / np.sum(a)
    elif kind == 'modulated':
        x = np.arange(n, dtype=float)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 3 * x / n) + 0.15 * np.cos(2 * np.pi * 7 * x / n)
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif kind == 'step':
        a = np.ones(n)
        a[:n // 2] = 1.5
        a[n // 2:] = 0.5
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'random_smooth':
        a = rng.random(n // 10 + 1) + 0.5
        x_old = np.linspace(0, 1, len(a))
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, a)
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def coordinate_polish(a, time_budget=40.0):
    """
    Direct coordinate descent on the non-smooth objective.
    Moves mass from high-contributing positions to reduce the peak.
    """
    n = len(a)
    a = a.copy()
    S = float(np.sum(a))
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = fft_conv(a)
    rng = np.random.default_rng(12345)

    stall = 0
    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))

        # Compute contributions to the peak
        contrib = np.zeros(n)
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib[j] = a[j] * a[k2]

        improved = False

        # Try reducing top contributors
        sorted_idx = np.argsort(contrib)[::-1][:20]

        for i in sorted_idx:
            if time.monotonic() - start > time_budget:
                break
            if a[i] < 1e-12:
                continue

            # Try several targets and deltas
            best_delta_obj = best_obj
            best_move = None

            targets = [(i + n // 2) % n, (i + n // 3) % n,
                       (i + 2 * n // 3) % n, (i + 1) % n, (i - 1 + n) % n]
            targets = list(set(t for t in targets if t != i))[:4]

            for target in targets:
                for frac in [0.001, 0.003, 0.01, 0.03]:
                    delta = frac * a[i]
                    if delta <= 0 or delta > 0.9 * a[i]:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    conv_new = fft_conv(a)
                    obj_new = 2.0 * n * float(np.max(conv_new))
                    if obj_new < best_delta_obj:
                        best_delta_obj = obj_new
                        best_move = (i, target, delta)
                    a[i] += delta
                    a[target] -= delta

            if best_move is not None:
                i_m, t_m, d_m = best_move
                a[i_m] -= d_m
                a[t_m] += d_m
                S = float(np.sum(a))
                a = a / S
                conv = fft_conv(a)
                best_obj = compute_obj(a)
                best_a = a.copy()
                improved = True

        if not improved:
            # Random exploration
            for _ in range(5):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j or a[i] < 1e-12:
                    continue
                delta = 0.005 * a[i]
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                S = float(np.sum(a))
                if S < 1e-12:
                    a = best_a.copy()
                    break
                a = a / S
                conv_new = fft_conv(a)
                obj_new = 2.0 * n * float(np.max(conv_new))
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
                    conv = conv_new
                    improved = True
                else:
                    a = best_a.copy()
            if not improved:
                stall += 1
                if stall > 30:
                    break

    return best_a


def subgradient_fista(a_init, time_budget=100.0):
    """
    FISTA (Accelerated Proximal Gradient) on the simplex.
    Uses the log-sum-exp smooth approximation for the gradient.
    """
    n = len(a_init)
    a = a_init.copy()
    S = float(np.sum(a))
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = 200.0
    t_k = 1.0
    y = a.copy()
    L = 8.0 * n * T  # Lipschitz constant estimate

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        # Gradient at y
        conv_y = fft_conv(y)
        max_c = float(np.max(conv_y))
        shifted = T * (conv_y - max_c)
        shifted = np.clip(shifted, -600, 600)
        exp_s = np.exp(shifted)
        sum_exp = float(np.sum(exp_s))
        if sum_exp < 1e-300:
            sum_exp = 1e-300
        p = exp_s / sum_exp

        g = np.zeros(n)
        for i in range(n):
            g[i] = np.dot(y, p[i:i + n])

        grad = 4.0 * n * g

        # FISTA step: x = y - (1/L) * grad, then project to simplex
        step = 1.0 / L
        x = y - step * grad

        # Project x onto simplex {a >= 0, sum(a) = 1}
        x_sorted = np.sort(x)[::-1]
        css = np.cumsum(x_sorted)
        rho = np.arange(1, n + 1)
        cond = x_sorted - (css - 1) / rho > 0
        if np.any(cond):
            r = int(np.max(np.where(cond)[0]))
            theta = (css[r - 1] - 1) / r
        else:
            theta = 0.0
        x_proj = np.maximum(x - theta, 0)
        x_proj = x_proj / np.sum(x_proj)

        # Update y for next iteration
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_proj + ((t_k - 1) / t_new) * (x_proj - a)
        y = np.maximum(y, 0)
        Sy = float(np.sum(y))
        if Sy > 1e-15:
            y = y / Sy
        t_k = t_new
        a = x_proj

        # Evaluate
        if it % 50 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        # Adaptive T increase
        if it % 2000 == 1999 and it < 20000:
            T *= 2.0
            L = 8.0 * n * T
            t_k = 1.0
            y = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: L-BFGS with log-sum-exp annealing, LP kicks,
    FISTA subgradient, diverse multi-start, and coordinate polish.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30  # 30s safety margin

    rng = np.random.default_rng(seed)

    # Load parent
    best_a = None
    best_score = np.inf
    n_best = 800

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        S = float(parent.sum())
        if S > 0.01:
            parent = parent / S
            best_a = parent.copy()
            best_score = compute_obj(parent)
            n_best = len(parent)
            print(f"Parent (n={n_best}): obj={best_score:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_a is None:
        n_best = 800
        best_a = np.ones(n_best) / n_best
        best_score = compute_obj(best_a)
        print(f"Uniform init (n={n_best}): obj={best_score:.8f}")

    # === Phase 1: L-BFGS Main Optimization (450s) ===
    print("\n=== Phase 1: L-BFGS Main ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(450.0, remaining * 0.35)

    T_schedule = [10, 30, 100, 300, 1000, 3000, 10000, 50000]
    a_opt = lbs_optimize(best_a, T_schedule, max_iter_per_T=300,
                         time_budget=p1_budget, verbose=True)
    obj_opt = compute_obj(a_opt)
    if obj_opt < best_score:
        best_score = obj_opt
        best_a = a_opt.copy()
        print(f"  ** Phase 1: obj={best_score:.8f} **")
    else:
        # Keep better of original and optimized
        orig_obj = compute_obj(best_a)
        if orig_obj < best_score:
            best_score = orig_obj
        print(f"  Phase 1: obj={obj_opt:.8f} (best={best_score:.8f})")

    # === Phase 2: Diverse Multi-Start (400s) ===
    print("\n=== Phase 2: Multi-Start ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(400.0, remaining * 0.3)

    init_kinds = ['cosine', 'sine', 'gaussian', 'two_peaks', 'triangle',
                  'modulated', 'step', 'random_smooth']

    per_init = p2_budget / (len(init_kinds) + 2)

    for kind in init_kinds:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_init = generate_init(n_best, kind, rng)
        T_quick = [50, 500, 5000]
        a_try = lbs_optimize(a_init, T_quick, max_iter_per_T=100,
                             time_budget=per_init, verbose=False)
        obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_a = a_try.copy()
            print(f"  ** Init '{kind}': obj={best_score:.8f} **")
        else:
            print(f"  Init '{kind}': obj={obj_try:.8f}")

    # Random starts
    for ri in range(2):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        a_init = generate_init(n_best, 'random', rng)
        a_try = lbs_optimize(a_init, [100, 2000], max_iter_per_T=80,
                             time_budget=per_init * 0.7, verbose=False)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_a = a_try.copy()
            print(f"  ** Random {ri}: obj={best_score:.8f} **")
        else:
            print(f"  Random {ri}: obj={obj_try:.8f}")

    # === Phase 3: LP Kicks + L-BFGS Polish (250s) ===
    print("\n=== Phase 3: LP Kicks ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(250.0, remaining * 0.22)

    for kick in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # LP to get new direction
        lp_time = min(25.0, remaining * 0.15)
        a_lp = lp_kick(best_a, time_budget=lp_time)

        # Perturb
        scale = 0.005 + 0.005 * kick
        a_pert = a_lp * np.exp(rng.normal(0, scale, n_best))
        a_pert = np.maximum(a_pert, 0)
        Sp = float(np.sum(a_pert))
        if Sp < 1e-10:
            a_pert = best_a.copy()
        else:
            a_pert = a_pert / Sp

        # L-BFGS polish
        lbs_time = min(25.0, remaining * 0.2)
        a_opt = lbs_optimize(a_pert, [500, 5000, 20000], max_iter_per_T=150,
                             time_budget=lbs_time, verbose=False)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_a = a_opt.copy()
            print(f"  ** Kick {kick}: obj={best_score:.8f} **")
        else:
            print(f"  Kick {kick}: obj={obj_opt:.8f}")

    # === Phase 4: FISTA Subgradient (150s) ===
    print("\n=== Phase 4: FISTA ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        fista_time = min(150.0, remaining * 0.15)
        a_fista = subgradient_fista(best_a, time_budget=fista_time)
        obj_f = compute_obj(a_fista)
        if obj_f < best_score:
            best_score = obj_f
            best_a = a_fista.copy()
            print(f"  ** FISTA: obj={best_score:.8f} **")
        else:
            print(f"  FISTA: obj={obj_f:.8f}")

    # === Phase 5: Length Sweep (100s) ===
    print("\n=== Phase 5: Length Sweep ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n_options = sorted(set([
            max(300, n_best - 150),
            max(300, n_best - 50),
            n_best + 50,
            n_best + 150,
            n_best + 300,
        ]))
        n_options = [n for n in n_options if 300 <= n <= 3000 and n != n_best]

        per_n = min(25.0, (remaining * 0.1) / max(1, len(n_options)))

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_res = resample_seq(best_a, n_try)
            a_try = lbs_optimize(a_res, [200, 2000], max_iter_per_T=80,
                                 time_budget=per_n, verbose=False)
            obj_try = compute_obj(a_try)

            if obj_try < best_score:
                best_score = obj_try
                best_a = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final L-BFGS Polish (remaining - 60s) ===
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        final_time = min(120.0, remaining * 0.5)
        T_final = [1000, 10000, 50000, 200000]
        a_final = lbs_optimize(best_a, T_final, max_iter_per_T=200,
                               time_budget=final_time, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_a = a_final.copy()
            print(f"  ** Final L-BFGS: obj={best_score:.8f} **")
        else:
            print(f"  Final L-BFGS: obj={obj_final:.8f}")

    # === Phase 7: Final Coordinate Descent ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        cd_time = min(40.0, remaining * 0.6)
        a_cd = coordinate_polish(best_a, time_budget=cd_time)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_a = a_cd.copy()
            print(f"  ** Coord: obj={best_score:.8f} **")
        else:
            print(f"  Coord: obj={obj_cd:.8f}")

    # Final L-BFGS squeeze with very high T
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_squeeze = lbs_optimize(best_a, [100000, 500000], max_iter_per_T=100,
                                 time_budget=min(20.0, remaining * 0.7), verbose=False)
        obj_sq = compute_obj(a_squeeze)
        if obj_sq < best_score:
            best_score = obj_sq
            best_a = a_squeeze.copy()
            print(f"  ** Squeeze: obj={best_score:.8f} **")

    # === Final Validation ===
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, "
          f"time={elapsed:.1f}s ===")

    return result
```
