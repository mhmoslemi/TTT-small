def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a = a / s
    return 2.0 * n * float(np.max(compute_conv(a)))

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    rho = np.where(u*idx > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n)/n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp(a):
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    c = np.zeros(n+1)

    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)]*n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            g = np.maximum(res.x[:n], 0)
            s = g.sum()
            if s > 1e-10:

    return np.ones(n)/n, None

    g = np.zeros(n)

from scipy.optimize import linprog, minimize as sp_minimize


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build A matrix (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_full(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M, dual_vars) where dual_vars are marginals for inequality constraints.
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = g.sum()
            if s > 1e-10:
                g = g / s
                dual = None
                try:
                    dual = res.ineqlin['marginals']
                except (AttributeError, KeyError):
                    pass
                return g, float(res.x[n]), dual
    except Exception:
        pass
    # Fallback
    return np.ones(n) / n, None, None


def solve_lp_reduced(a, n_constraints=300):
    """
    Solve LP with only the most important constraints (constraint generation).
    Uses the top n_constraints lags by self-convolution value.
    """
    n = len(a)
    a = a / float(np.sum(a))
    A_full = build_conv_matrix(a)
    self_conv = A_full @ a
    max_val = self_conv.max()
    if max_val < 1e-15:
        return np.ones(n) / n, 0.0, None

    # Select important constraints
    n_sel = min(n_constraints, 2 * n - 1)
    top_idx = np.argsort(self_conv)[::-1][:n_sel]

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A = A_full[top_idx]
    A_ub = np.zeros((len(top_idx), n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(top_idx))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15.0})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = g.sum()
            if s > 1e-10:
                g = g / s
                # Check full constraint satisfaction
                full_conv = A_full @ g
                M = float(np.max(full_conv))
                return g, M, None
    except Exception:
        pass
    return np.ones(n) / n, None, None


def grad_conv_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a. Returns 2*a[k-i] for valid i."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
    return g


def weighted_subgradient(a, weights, peak_indices):
    """Compute weighted subgradient from multiple peaks."""
    n = len(a)
    g = np.zeros(n)
    for w, k in zip(weights, peak_indices):
        if w > 1e-12:
            g += w * grad_conv_k(a, int(k))
    return g


def lp_alternating_sa(a, time_budget, rng, use_full_lp=True, verbose=True):
    """
    LP Alternating with Simulated Annealing acceptance and dual-guided perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    start = time.monotonic()
    stall = 0
    T_sa = 0.002
    T_min = 1e-5
    T_decay = 0.9995
    it = 0
    last_dual = None

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        # Solve LP (use reduced for speed when time is tight)
        if remaining > 30 and use_full_lp:
            g, M, dual = solve_lp_full(a)
        else:
            g, M, dual = solve_lp_reduced(a, n_constraints=400)

        if g is None:
            break

        if dual is not None:
            last_dual = dual

        # Adaptive step sizes: cluster around previously good values
        base_ts = [0.15, 0.25, 0.35, 0.5, 0.65, 0.8, 1.0]
        best_t_obj = np.inf
        best_a_try = a.copy()
        best_t = 0.4

        for t in base_ts:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_a_try = a_try.copy()
                    best_t = t

        it += 1

        # SA acceptance
        delta = best_t_obj - current_obj
        if delta < -1e-14:
            accept = True
        elif delta < 1e-10:
            accept = True
        else:
            accept = (rng.random() < np.exp(-delta / max(T_sa, T_min)))

        if accept:
            a = best_a_try
            current_obj = best_t_obj
            if current_obj < best_obj - 1e-13:
                best_obj = current_obj
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        T_sa = max(T_min, T_sa * T_decay)

        # Escape mechanism with dual guidance
        if stall > 25:
            noise_scale = min(0.25, 0.04 + 0.03 * (stall // 25))
            if last_dual is not None and np.any(np.abs(last_dual) > 1e-6):
                # Dual-guided perturbation
                active = np.where(np.abs(last_dual) > 1e-4)[0]
                if len(active) > 0:
                    g_dual = np.zeros(n)
                    for k in active[:50]:
                        g_dual -= last_dual[k] * grad_conv_k(a, int(k))
                    norm_g = np.linalg.norm(g_dual)
                    if norm_g > 1e-10:
                        a = best_a + noise_scale * 0.5 * g_dual / norm_g * a.sum()
                        a = np.maximum(a, 0)
                        a = a / a.sum()
            # Add random component
            a = a * np.exp(rng.normal(0, noise_scale * 0.3, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            current_obj = compute_obj(a)
            if current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()
            stall = 0
            T_sa = max(T_sa, 0.001)

        if verbose and it % 30 == 0:
            m_str = f"{M:.6f}" if M is not None else "N/A"
            print(f"      SA-LP it={it}, best={best_obj:.8f}, cur={current_obj:.8f}, M={m_str}, T={T_sa:.5f}, el={time.monotonic()-start:.1f}s")

    return best_a


def smooth_logsumexp_optimize(a, time_budget, T=50.0, verbose=True):
    """Smooth optimization using log-sum-exp approximation with L-BFGS-B."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def obj_grad(x):
        x = np.maximum(x, 1e-15)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x = x / s
        conv = compute_conv(x)
        max_c = conv.max()
        z = T * (conv - max_c)
        e = np.exp(np.minimum(z, 50))  # clip for numerical stability
        sum_e = e.sum()
        if sum_e < 1e-300:
            sum_e = 1e-300
        obj_val = 2.0 * n * (max_c + np.log(sum_e) / T)
        w = e / sum_e
        g = np.zeros(n)
        for k in range(len(conv)):
            if w[k] > 1e-6:
                g += w[k] * grad_conv_k(x, k)
        g *= 2.0 * n
        return obj_val, g

    try:
        res = sp_minimize(obj_grad, a, method='L-BFGS-B',
                          bounds=[(1e-12, None)] * n,
                          options={'maxiter': 300, 'maxfun': 600, 'disp': False,
                                   'ftol': 1e-14, 'gtol': 1e-8})
        a_opt = np.maximum(res.x, 0)
        s_opt = float(np.sum(a_opt))
        if s_opt > 1e-10:
            a_opt = a_opt / s_opt
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
    except Exception:
        pass

    if verbose:
        print(f"    Smooth(T={T}): obj={best_obj:.8f}, elapsed={time.monotonic()-start:.1f}s")

    return best_a


def nesterov_subgrad(a, time_budget, K=5, verbose=True):
    """Nesterov-accelerated subgradient on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        # Temperature-weighted peaks
        spread = (peak_vals[0] - peak_vals[-1]) / max_val
        temp = 3.0 / (1.0 + spread * 10.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / max_val)
        weights /= weights.sum()

        g = weighted_subgradient(y, weights, top_idx)
        lr = lr0 / t_k
        a_new = project_simplex(y - lr * g)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                y = best_a.copy()
                a_prev = best_a.copy()
                a_new = best_a.copy()
                t_k = 1.0
                stall = 0

        a_prev = a_new.copy()
        t_k = t_new
        it += 1

    if verbose:
        print(f"    Nesterov: obj={best_obj:.8f}, iters={it}, elapsed={time.monotonic()-start:.1f}s")
    return best_a


def parameterized_init(n, mode, rng):
    """Generate parameterized initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        # Double Gaussian
        c1, c2 = n // 4, 3 * n // 4
        w = n // 8
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 3:
        # Inverted triangle (concentrated at edges)
        x = np.linspace(-1, 1, n)
        a = np.abs(x) + 0.1
        return a / a.sum()
    elif mode == 4:
        # Lorentzian
        x = np.linspace(-1, 1, n)
        a = 1.0 / (1.0 + 5.0 * x ** 2)
        return a / a.sum()
    elif mode == 5:
        # Periodic modulation
        f = rng.integers(2, 8)
        alpha = rng.uniform(0.1, 0.4)
        x = np.linspace(0, 1, n)
        a = 1.0 + alpha * np.cos(2 * np.pi * f * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 6:
        # Quadratic
        x = np.linspace(-1, 1, n)
        a = 1.0 + 0.3 * x ** 2
        return a / a.sum()
    elif mode == 7:
        # Sinc-like
        x = np.linspace(-1, 1, n)
        a = np.sinc(3 * x) ** 2
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 8:
        # Fourier series
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for k in range(1, 5):
            ak = rng.uniform(-0.2, 0.2)
            a += ak * np.cos(2 * np.pi * k * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_refine(a, time_budget):
    """Coordinate descent: move mass to reduce dominant peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find contributing positions
        contrib = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib.append((a[i] * a[j], i, j))
        contrib.sort(reverse=True)

        # Find non-contributing positions
        non_contrib = []
        for i in range(n):
            j = k_star - i
            if j < 0 or j >= n:
                non_contrib.append(i)

        if not non_contrib or len(contrib) == 0:
            break

        improved = False
        for ci in range(min(20, len(contrib))):
            val, i, j = contrib[ci]
            if val < 1e-12:
                break
            for target in non_contrib[:10]:
                for frac in [0.02, 0.05, 0.1, 0.15]:
                    src = i if a[i] >= a[j] else j
                    delta = frac * a[src]
                    if delta < 1e-15:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    if a[src] < 0:
                        a[src] += delta
                        a[target] -= delta
                        continue
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break

        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating with SA, multi-n exploration, and smooth polish.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 40.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent loaded: n={parent.size}, obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init: n={n0}, obj={best_score:.8f}")

    n_primary = len(best_seq)

    # === Phase 1: LP Alternating with SA on primary n (50% of time) ===
    print(f"\n=== Phase 1: LP-Alt+SA (n={n_primary}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.45, 600.0)
    if p1_budget > 30:
        a1 = lp_alternating_sa(best_seq, time_budget=p1_budget, rng=rng,
                               use_full_lp=True, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase 1 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj1:.8f} (no improvement)")

    # === Phase 2: Multi-n exploration (20% of time) ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    n_current = len(best_seq)

    n_options = sorted(set([
        max(300, n_current - 100),
        max(300, n_current - 50),
        n_current + 50,
        n_current + 100,
        n_current + 200,
        600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500 and n != n_current]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 50 or n_tried >= 4:
            break
        n_tried += 1

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(p2_budget / max(1, 5 - n_tried), remaining * 0.3)
        if alloc < 20:
            continue

        print(f"  Trying n={n_try}, budget={alloc:.0f}s")
        a_opt = lp_alternating_sa(a_try, time_budget=alloc, rng=rng,
                                  use_full_lp=(alloc > 60), verbose=False)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 3: Diverse restarts on best n (15% of time) ===
    print(f"\n=== Phase 3: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        mode = r  # Use different initialization modes
        a_r = parameterized_init(n_best, mode, rng)
        # Blend with current best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(p3_budget / max(1, 5 - r), remaining * 0.3)
        if alloc < 15:
            continue

        a_opt = lp_alternating_sa(a_r, time_budget=alloc, rng=rng,
                                  use_full_lp=(alloc > 40), verbose=False)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 4: Final polish on best (20% of time) ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()

    # 4a: Extended LP alternating
    if remaining > 40:
        alloc = min(remaining * 0.35, 120.0)
        a_lp = lp_alternating_sa(best_seq, time_budget=alloc, rng=rng,
                                 use_full_lp=True, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_lp:.8f}")

    # 4b: Smooth log-sum-exp optimization
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for T_val in [30.0, 80.0]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.2, 40.0)
            a_sm = smooth_logsumexp_optimize(best_seq, time_budget=alloc, T=T_val, verbose=True)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Smooth T={T_val}: obj={best_score:.8f} **")

    # 4c: Nesterov subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 40.0)
        a_ns = nesterov_subgrad(best_seq, time_budget=alloc, K=7, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")

    # 4d: Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.2, 30.0)
        a_cd = coordinate_refine(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CoordDesc: obj={best_score:.8f} **")

    # 4e: One more LP round
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 40.0)
        a_last = lp_alternating_sa(best_seq, time_budget=alloc, rng=rng,
                                   use_full_lp=True, verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isinf(val):
        result = [1.0] * 100
        best_score = evaluate_sequence(result)
    elif val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
