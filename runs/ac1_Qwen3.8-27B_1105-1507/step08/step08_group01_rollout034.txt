3. Project `a - eta * sg` onto the simplex
4. Repeat

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(j, min(j + n, 2 * n - 1)):
            A[k, j] = a[k - j]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    u = np.sort(v)[::-1]
    n = len(u)
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n+1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

def line_search(a, g):
    """Find optimal blend parameter t for a_new = (1-t)*a + t*g."""
    n = len(a)
    a = a / np.sum(a)

    conv_ag = compute_conv_cross(a, g)
    conv_gg = compute_conv(g)
    
    def obj_at_t(t):
        val = (1-t)**2 * conv_aa + 2*t*(1-t) * conv_ag + t**2 * conv_gg
        return 2.0 * n * float(np.max(val))
    
    # Grid search first
    best_t = 0.0
    best_val = np.inf
    for t in np.linspace(0, 1, 21):

            best_t = t

    s = float(np.sum(a_new))

"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    u = np.sort(v)[::-1]
    n = len(u)
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp_full(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def solve_lp_active(a, active_indices):
    """Solve LP with only active set constraints."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    A_reduced = A[active_indices, :]

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A_reduced, -np.ones((len(active_indices), 1))])
    b_ub = np.zeros(len(active_indices))

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_norm = g_val / s
            # Verify against ALL constraints
            full_conv = build_conv_matrix(a) @ g_norm
            true_M = float(np.max(full_conv))
            return g_norm, true_M
    return None, None


def line_search_blend(a, g):
    """Find optimal t for a_new = (1-t)*a + t*g using grid + golden section."""
    n = len(a)
    a = a / np.sum(a)
    g = g / np.sum(g)

    conv_aa = compute_conv(a)
    conv_ag = compute_conv_cross(a, g)
    conv_gg = compute_conv(g)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    # Grid search
    best_t = 0.0
    best_val = np.inf
    for t in np.linspace(0, 1, 11):
        v = obj_at_t(t)
        if v < best_val:
            best_val = v
            best_t = t

    # Golden section refinement
    if 0.02 < best_t < 0.98:
        lo = max(0, best_t - 0.1)
        hi = min(1, best_t + 0.1)
        gr = (np.sqrt(5) - 1) / 2
        for _ in range(25):
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            f1 = obj_at_t(x1)
            f2 = obj_at_t(x2)
            if f1 < f2:
                hi = x2
            else:
                lo = x1
        t_mid = (lo + hi) / 2
        v_mid = obj_at_t(t_mid)
        if v_mid < best_val:
            best_t = t_mid
            best_val = v_mid

    a_new = (1 - best_t) * a + best_t * g
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a
    return best_t, a_new / s

def compute_smooth_gradient(a, weights):
    """
    Compute gradient of smooth_max_k conv(a,a)(k) w.r.t. a.
    weights has length 2n-1, sums to 1.
    Uses FFT for O(n log n) computation.
    grad[i] = 2 * sum_{j=0}^{n-1} a[j] * weights[i+j]
    """
    n = len(a)
    w_rev = weights[::-1].astype(np.float64)
    N = 3 * n - 2
    size = 1
    while size < N:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w_rev, size)
    conv_aw = irfft(fa * fw, size)[:N]
    # grad[i] = 2 * conv_aw[2n-2-i] for i = 0, ..., n-1
    grad = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
    return grad


def smooth_max_gd(a, time_budget=100.0, verbose=False):
    """Gradient descent with smooth max and Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)

        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = 0.0005
    lr = 0.003
    momentum = 0.85
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -600, 600)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        if total <= 0:
            break
        weights = exp_vals / total

        grad = compute_smooth_gradient(a, weights)
        # Center the gradient (remove mean to maintain sum=1 approximately)
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        else:

                lr *= 0.8
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
                if lr < 1e-8:
                    break
            # Anneal T
            if it > 50:
                T = max(0.00005, T * 0.999)

    return best_a


def subgradient_descent(a, time_budget=60.0, verbose=False):
    """Pure subgradient descent with decreasing step size."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n, else 0
        lo_idx = max(0, k_star - n + 1)
        hi_idx = min(n - 1, k_star)
        if lo_idx <= hi_idx:
            grad = np.zeros(n)
            # grad[i] = 2 * a[k_star - i] for i in [lo_idx, hi_idx]
            # a[k_star - i] for i = lo_idx..hi_idx gives a[k_star-lo_idx]..a[k_star-hi_idx] reversed
            grad[lo_idx:hi_idx + 1] = 2.0 * a[lo_idx:hi_idx + 1][::-1]
        else:
            grad = np.zeros(n)

        # Step size: O(1/sqrt(t))
        eta = 0.5 / np.sqrt(float(it))
        grad_centered = grad - np.mean(grad)
        a_new = simplex_project(a - eta * grad_centered)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 200 == 0:
                print(f"    SG it{it}: obj={best_obj:.8f}")

    return best_a


def coordinate_descent(a, time_budget=80.0, verbose=False):
    """Coordinate descent with FFT-based evaluation and smart ordering."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    delta = 1e-4
    pass_count = 0
    no_improve_passes = 0

    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False

        # Order by contribution to max convolution
        conv = compute_conv(a)
        max_conv = np.max(conv)
        # Score each coordinate by how much it contributes to active peaks
        scores = np.zeros(n)
        active = np.where(conv >= max_conv * 0.95)[0]
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    scores[i] += a[j]
        order = np.argsort(-scores)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            # Try increase
            for d in [delta, delta * 0.1]:
                a_try = a.copy()
                a_try[i] += d
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if improved:
                continue

            # Try decrease
            if a[i] > delta * 0.1:
                for d in [delta, delta * 0.1]:
                    if a[i] < d * 0.5:
                        continue
                    a_try = a.copy()
                    a_try[i] -= d
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if verbose and pass_count % 5 == 0:
            print(f"    CD pass {pass_count}: obj={best_obj:.8f}")

        if not improved:
            no_improve_passes += 1
            delta *= 0.7
            if delta < 1e-8 or no_improve_passes > 5:
                break
        else:
            no_improve_passes = 0

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_hann(n):
    """Hann window."""
    return np.hanning(n) + 1e-8


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian window."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-x ** 2 / (2 * sigma ** 2)) + 1e-8


def make_triangular(n):
    """Triangular window."""
    mid = (n - 1) / 2.0
    return np.maximum(0, 1 - np.abs(np.arange(n) - mid) / mid) + 1e-8


def make_power_taper(n, p=1.0):
    """Power-law taper: a[i] = ((i+1)*(n-i))^p normalized."""
    i = np.arange(n).astype(float)
    vals = ((i + 1) * (n - i)) ** p
    return vals + 1e-8

        'gauss': make_gaussian(n_init, 0.3),
        'gauss_wide': make_gaussian(n_init, 0.4),
        'tri': make_triangular(n_init),
        'power1': make_power_taper(n_init, 1.0),
        'power2': make_power_taper(n_init, 2.0),
        'uniform': np.ones(n_init),
    }

    # Quick evaluation of each init
    for name, init_arr in inits.items():
        init_arr = init_arr / np.sum(init_arr)
        obj = compute_obj(init_arr)
        print(f"  {name}: obj={obj:.8f}")
        if obj < best_score:
            best_score = obj
            best_seq = init_arr.copy()

    # Quick LP on the best 3 inits
    init_objects = [(compute_obj(arr / np.sum(arr)), name, arr) for name, arr in inits.items()]
    init_objects.sort(key=lambda x: x[0])

    time_per_init = min(30.0, (deadline - time.monotonic()) * 0.15)
    for obj_i, name, arr in init_objects[:3]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        a_try = arr / np.sum(arr)
        a_opt = lp_alternating(a_try, time_budget=min(time_per_init, remaining * 0.3), verbose=False)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  {name} + LP: obj={best_score:.8f}")

    a = best_seq.copy()
    print(f"  After Phase 1: best={best_score:.8f} (n={len(a)})")

    # === Phase 2: Intensive LP Alternating ===
    print("\n=== Phase 2: Intensive LP Alternating ===")
    remaining = deadline - time.monotonic()
    phase2_budget = min(400.0, remaining * 0.4)
    a_opt = lp_alternating(a, time_budget=phase2_budget, verbose=True)
    obj_opt = compute_obj(a_opt)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        a = a_opt.copy()
    print(f"  After Phase 2: best={best_score:.8f}")

    # === Phase 3: Smooth Max Gradient Descent ===
    print("\n=== Phase 3: Smooth Max GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        phase3_budget = min(100.0, remaining * 0.15)
        a_sm = smooth_max_gd(a, time_budget=phase3_budget, verbose=True)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            a = a_sm.copy()
            print(f"  After Phase 3: best={best_score:.8f}")
        else:
            print(f"  Phase 3: no improvement ({obj_sm:.8f})")

    # === Phase 4: More LP (different active set thresholds) ===
    print("\n=== Phase 4: Additional LP rounds ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        phase4_budget = min(100.0, remaining * 0.15)
        a_opt = lp_alternating(a, time_budget=phase4_budget, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  After Phase 4: best={best_score:.8f}")
        else:
            print(f"  Phase 4: no improvement ({obj_opt:.8f})")

    # === Phase 5: Coordinate Descent ===
    print("\n=== Phase 5: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        phase5_budget = min(80.0, remaining * 0.15)
        a_cd = coordinate_descent(a, time_budget=phase5_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a = a_cd.copy()
            print(f"  After Phase 5: best={best_score:.8f}")
        else:
            print(f"  Phase 5: no improvement ({obj_cd:.8f})")

    # === Phase 6: Multi-n Exploration ===
    print("\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 200), max(200, n_best - 100), max(200, n_best - 50),
        max(200, n_best + 50), max(200, n_best + 100), max(200, n_best + 200),
        400, 500, 600, 700, 750, 800, 850, 900, 1000, 1100, 1200, 1300
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick LP
        alloc = min(remaining * 0.1, 20.0)
        a_opt = lp_alternating(a_try, time_budget=alloc, verbose=False)
        obj_try = compute_obj(a_opt)

        # Quick CD if close
        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_try < best_score + 0.003:
            a_opt = coordinate_descent(a_opt, time_budget=min(15.0, remaining * 0.3), verbose=False)
            obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final Polish ===
    print("\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = best_seq.copy()

        # LP polish
        if remaining > 30:
            alloc = min(60.0, remaining * 0.25)
            a_lp = lp_alternating(a_final, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_final = a_lp.copy()

        # Smooth max final
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(40.0, remaining * 0.25)
            a_sm = smooth_max_gd(a_final, time_budget=alloc, verbose=False)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                a_final = a_sm.copy()

        # CD final
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(30.0, remaining * 0.3)
            a_cd = coordinate_descent(a_final, time_budget=alloc, verbose=False)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with actual evaluator
    try:
        val = evaluate_sequence(result)
        if np.isfinite(val) and val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
