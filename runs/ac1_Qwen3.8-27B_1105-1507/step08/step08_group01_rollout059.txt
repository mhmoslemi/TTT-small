```python
def peak_shave(a, n_peaks=10, eta=0.01):
    """Reduce the top peaks of conv(a,a) by multiplicatively adjusting a."""
    n = len(a)
    a = a.copy()
    c = conv(a, a)
    # Find top peaks
    top_k = np.argsort(c)[::-1][:n_peaks]
    # For each peak, compute sensitivity
    for k in top_k:
        # c[k] = sum_j a[j] * a[k-j]
        # d(c[k])/da[i] = a[k-i] + a[i-k] (when indices valid)
        # We want to reduce c[k], so decrease a[i] where sensitivity is high
        sens = np.zeros(n)
        for i in range(n):
            val = 0
            if 0 <= k - i < n:
                val += a[k - i]
            if 0 <= i - k < n:
                val += a[i - k]
            sens[i] = val
        # Multiplicative update: a[i] *= (1 - eta * sens[i] / max(sens))
        max_sens = np.max(sens)
        if max_sens > 0:
            a *= (1 - eta * sens / max_sens)
        a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0:
        a /= s
    return a
```

def compute_conv(a):

        size <<= 1
    A = fft(a, size)
    return ifft(A * A, size).real[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_max_weights(conv, T):
    """Compute soft-max weights over convolution values."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, max_c
    return exp_vals / total, max_c


def gradient_smooth_max(a, T):
    """Compute gradient of smooth-max(conv(a,a)) w.r.t. a. O(n^2) via correlate."""
    n = len(a)
    conv = compute_conv(a)
    weights, max_c = smooth_max_weights(conv, T)
    # grad[i] = 2 * sum_k weights[k] * a[k-i] = 2 * correlate(weights, a, 'valid')[i]
    g = np.correlate(weights, a, mode='valid')
    grad = 2.0 * g
    smooth_val = max_c + T * np.log(np.sum(np.exp((conv - max_c) / T)) / len(conv))
    return grad, smooth_val


def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_grad_descent(a, time_budget, T_init=0.0005, T_min=0.00002,
                        lr_init=0.005, momentum=0.9, verbose=False):
    """Smooth max gradient descent with adaptive lr and temperature decay."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = lr_init
    v = np.zeros(n)
    stall = 0
    max_iters = 20000

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        grad, _ = gradient_smooth_max(a, T)
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                lr *= 0.7
                if lr < 1e-8:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0

        if it > 10:
            T = max(T_min, T_init * (0.9995 ** it))

        if verbose and it % 200 == 0:
            print(f"    SGrad it{it}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.6f}")

    return best_a


def peak_shave_step(a, n_peaks=8, eta=0.015):
    """Single peak shaving step: multiplicatively reduce elements blamed for top peaks."""
    n = len(a)
    a = a.copy().astype(np.float64)
    conv = compute_conv(a)
    top_k = np.argsort(conv)[::-1][:n_peaks]

    total_blame = np.zeros(n)
    for k in top_k:
        # Blame for a[i] in conv[k] is proportional to a[k-i]
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            idx = k - i
            if 0 <= idx < n:
                total_blame[i] += a[idx]

    max_blame = np.max(total_blame)
    if max_blame > 1e-15:
        a *= (1.0 - eta * total_blame / max_blame)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a /= s
    return a


def peak_shave_loop(a, time_budget, n_peaks=8, eta_init=0.015, verbose=False):
    """Iterative peak shaving with adaptive eta."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        a_new = peak_shave_step(a, n_peaks=n_peaks, eta=eta)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            eta *= 0.85
            if stall > 30:
                break

        if verbose and it % 50 == 0:
            print(f"    PeakShave it{it}: obj={best_obj:.8f}, eta={eta:.6f}")

    return best_a


def build_conv_matrix(a):
    """Build matrix C (2n-1 x n) such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        C[j:end, j] = a[:end - j]
    return C


def solve_lp_active(a, active_indices):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for k in active_indices, sum(g)=1, g>=0."""
    try:
        import cvxpy as cp
    except ImportError:
        return _solve_lp_scipy(a, active_indices)

    n = len(a)
    a = a / np.sum(a)
    C = build_conv_matrix(a)
    C_active = C[active_indices, :]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [C_active @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in ['CLARABEL', 'ECOS', 'SCS', 'MOSEK', 'PDLP']:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_norm = g_val / s
                    # Verify against all constraints
                    conv_check = C @ g_norm
                    true_M = float(np.max(conv_check))
                    return g_norm, true_M
        except Exception:
            continue
    return _solve_lp_scipy(a, active_indices)


def _solve_lp_scipy(a, active_indices=None):
    """Fallback LP via scipy linprog (HiGHS)."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    C = build_conv_matrix(a)

    if active_indices is not None:
        C_use = C[active_indices, :]
        n_constr = len(active_indices)
    else:
        C_use = C
        n_constr = 2 * n - 1

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((n_constr, n + 1))
    A_ub[:, :n] = C_use
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_constr)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'maxiter': 5000})
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_norm = g_val / s
            conv_check = C @ g_norm
            true_M = float(np.max(conv_check))
            return g_norm, true_M
    return None, None


def lp_alternating(a, time_budget, verbose=False):
    """LP alternating with active set and line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    max_iters = 300

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv_cur = compute_conv(a)
        max_conv = np.max(conv_cur)
        threshold = max_conv * 0.97
        active_set = np.where(conv_cur >= threshold)[0]

        g_lp, M_lp = solve_lp_active(a, active_set)
        if g_lp is None:
            break

        # Line search: try several blend values
        best_t = None
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t

        if best_t is not None:
            best_obj = best_obj_ls
            a_new = (1 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / float(np.sum(a_new))
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 15:
                # Perturb
                rng = np.random.default_rng(it * 7919 + n)
                a_pert = best_a * np.exp(rng.normal(0, 0.003, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a = a_pert / s_p
                stall = 0

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, time={elapsed:.1f}s")

    return best_a


def coordinate_descent_polish(a, time_budget, delta=1e-4, verbose=False):
    """Coordinate descent with multiple step sizes."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    delta_cur = delta
    pass_count = 0

    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count * 104729 + n)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            for d in [delta_cur, delta_cur * 0.1, delta_cur * 5]:
                a_try = a.copy()
                a_try[i] += d
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if not improved:
                for d in [delta_cur, delta_cur * 0.1]:
                    if a[i] < d * 0.5:
                        continue
                    a_try = a.copy()
                    a_try[i] -= d
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if verbose and pass_count % 3 == 0:
            print(f"    CD pass{pass_count}: obj={best_obj:.8f}, delta={delta_cur:.2e}")

        if not improved and pass_count > 2:
            delta_cur *= 0.5
            if delta_cur < 1e-7:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_structured_init(n, seed=0):
    """Generate a structured initialization: V-shaped (high at edges)."""
    rng = np.random.default_rng(seed)
    t = np.linspace(0, 1, n)
    # V-shaped: high at edges, low in middle
    base = np.abs(t - 0.5) * 2.0  # 0 at center, 1 at edges
    a = 0.3 + 0.7 * base + 0.05 * rng.normal(0, 1, n)
    a = np.maximum(a, 0.01)
    a /= np.sum(a)
    return a


def make_dual_bump_init(n, seed=0):
    """Two-bump initialization."""
    rng = np.random.default_rng(seed)
    t = np.linspace(0, 1, n)
    sigma = 0.15
    a = np.exp(-((t - 0.25) ** 2) / (2 * sigma ** 2)) + np.exp(-((t - 0.75) ** 2) / (2 * sigma ** 2))
    a = a + 0.02 * np.abs(rng.normal(0, 1, n))
    a = np.maximum(a, 0.001)
    a /= np.sum(a)
    return a

        alloc = min(40.0, remaining * 0.3)
        a_gd = smooth_grad_descent(a, time_budget=alloc,
                                   T_init=0.00002, T_min=0.000002,
                                   lr_init=0.001, momentum=0.95, verbose=False)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            a = a_gd.copy()
            print(f"  Final SGrad: obj={best_score:.8f}")

    # === Phase 8: Structured Multi-Start (if time remains) ===
    print(f"\n=== Phase 8: Structured Multi-Start ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        inits = [
            ("V-shape", make_structured_init(n_best, seed=1)),
            ("Dual-bump", make_dual_bump_init(n_best, seed=2)),
            ("Uniform+noise", None),
        ]
        for name, a_init in inits:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            if a_init is None:
                rng2 = np.random.default_rng(99)
                a_init = np.abs(rng2.normal(0, 1, n_best)) + 0.1
                a_init = a_init / np.sum(a_init)

            obj_init = compute_obj(a_init)
            if obj_init < best_score + 0.05:
                alloc_gd = min(40.0, remaining * 0.3)
                a_opt = smooth_grad_descent(a_init, time_budget=alloc_gd,
                                            T_init=0.001, T_min=0.0001,
                                            lr_init=0.005, verbose=False)
                remaining = deadline - time.monotonic()
                if remaining > 20:
                    alloc_ps = min(20.0, remaining * 0.2)
                    a_opt = peak_shave_loop(a_opt, time_budget=alloc_ps,
                                            n_peaks=5, eta_init=0.01)
                remaining = deadline - time.monotonic()
                if remaining > 20:
                    alloc_lp = min(30.0, remaining * 0.4)
                    a_opt = lp_alternating(a_opt, time_budget=alloc_lp, verbose=False)

                obj_opt = compute_obj(a_opt)
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    print(f"  {name}: obj={best_score:.8f} **")
                else:
                    print(f"  {name}: obj={obj_opt:.8f}")

    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
