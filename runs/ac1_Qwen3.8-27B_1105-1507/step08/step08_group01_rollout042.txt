        size <<= 1
    fa = rfft(a, size)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

    return np.maximum(v - theta, 0)

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_max_gradient(a, p):
    """
    Compute grad of sum_k p[k] * conv(a,a)[k] w.r.t. a.
    grad[i] = 2 * sum_j a[j] * p[i+j] (with p out of bounds = 0)
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    N = 1
    while N < 2 * (2*n - 1):
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    corr = ifft(np.conj(A) * P).real
    # grad[i] = 2 * corr[N-i] for i=1,...,n-1; grad[0] = 2 * corr[0]
    grad = np.empty(n)
    grad[0] = 2.0 * corr[0]
    if n > 1:
        grad[1:] = 2.0 * corr[N-n+1:N][::-1]
    return grad

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based smooth-max gradient descent with Nesterov momentum,
incremental coordinate descent, LP alternating, and multi-n exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    c = compute_conv(a_n)
    return 2.0 * n * float(np.max(c))


def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_gradient(a, p):
    """
    Compute gradient of sum_k p[k]*conv(a,a)[k] w.r.t. a.
    grad[i] = 2 * sum_j a[j] * p[i+j] (with p out of bounds = 0).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    corr = ifft(np.conj(A) * P).real
    grad = np.empty(n)
    grad[0] = 2.0 * corr[0]
    if n > 1:
        grad[1:] = 2.0 * corr[N - n + 1:N][::-1]
    return grad

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end_k = min(2 * n - 1, j + n)
        A[j:end_k, j] = a[:end_k - j]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2 * n - 1),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None

def _convex_line_search(a, g):
    """Find optimal t in [0,1] for a_new = (1-t)*a + t*g."""
    n = len(a)
    a_n = a / np.sum(a)
    g_n = g / np.sum(g)

    conv_aa = compute_conv(a_n)
    # Cross convolution
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a_n, size)
    fg = fft(g_n, size)

    conv_gg = compute_conv(g_n)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(15):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if obj_at_t(m1) < obj_at_t(m2):
            hi = m2
        else:
            lo = m1
    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_n + t_opt * g_n
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_n
    return t_opt, a_new / s

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, method='uniform'):
    """Create structured initializations."""
    x = np.linspace(0, 1, n)
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        a = 0.5 * (1 + np.cos(np.pi * x))
    elif method == 'gaussian':
        sigma = 0.3
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    elif method == 'bimodal':
        a = np.exp(-0.5 * ((x - 0.25) / 0.15) ** 2) + np.exp(-0.5 * ((x - 0.75) / 0.15) ** 2)
    elif method == 'raised_cosine':
        a = np.sin(np.pi * x) ** 2
    elif method == 'edge_conc':
        a = 1.0 / (0.1 + x * (1 - x))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s
