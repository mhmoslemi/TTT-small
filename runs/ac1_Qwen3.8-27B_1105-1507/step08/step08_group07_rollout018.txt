def compute_conv(a):

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def smooth_obj_and_grad(a, beta):

    n = len(a)

from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_valid_corr(w, a):
    n = len(a)
    a_rev = a[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    f_a = np.fft.fft(a_rev, size)
    f_w = np.fft.fft(w, size)
    conv = np.fft.ifft(f_a * f_w).real
    return conv[n-1 : 2*n-1]

def softmax_smoothed(x, beta, conv_cache=None):
    """
    Compute smoothed objective and gradient.
    x: parameter vector (length n), a = softmax(x)
    beta: smoothing parameter
    Returns (objective, gradient).
    """
    n = len(x)
    # Softmax
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / np.sum(exp_x)
    
    # Convolution
    c = compute_conv(a)
    
    # Log-sum-exp
    c_max = np.max(c)
    logits = beta * (c - c_max)
    # Clip for numerical stability
    logits = np.clip(logits, -50, 50)
    exp_logits = np.exp(logits)
    sum_exp = np.sum(exp_logits)
    obj = c_max + np.log(sum_exp) / beta
    
    # Softmax weights (substochastic)
    w = exp_logits / sum_exp
    
    # Gradient w.r.t. a
    g = compute_valid_corr(w, a)
    grad_a = 2.0 * g / beta
    
    # Chain rule through softmax
    dot = np.sum(grad_a * a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x

def lsm_optimizer(a_init, time_budget=300.0, verbose=True):
    """
    Optimize using L-SMm (log-sum-exp continuation + L-BFGS).
    """
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / np.sum(a_init)
    
    # Initial x: log(a_init) (since softmax(log(a)) = a / sum(a) = a when sum=1)
    x = np.log(a_init)
    
    best_x = x.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()
    
    # Continuation schedule
    # Start with beta such that beta * (max(c) - min(c)) ~ 2
    c_init = compute_conv(a_init)
    c_range = np.max(c_init) - np.min(c_init)
    if c_range < 1e-10:
        c_range = np.max(c_init)
    beta = max(2.0 / c_range, 1.0)
    
    max_beta = max(200.0 / c_range, 100.0)
    n_stages = 0
    
    while time.monotonic() - start < time_budget:
        n_stages += 1
        t_stage_start = time.monotonic()
        stage_time = min(time_budget - (t_stage_start - start), max(5.0, (time_budget - (t_stage_start - start)) * 0.3))
        
        # Run L-BFGS at current beta
        result = minimize(
            softmax_smoothed,
            x,
            args=(beta, None),
            method='L-BFGS-B',
            options={'maxiter': 200, 'gtol': 1e-8, 'ftol': 1e-12,
                     'maxfun': 500, 'disp': False}
        )
        
        x = result.x
        # Get the actual objective
        x_shifted = x - np.max(x)
        a = np.exp(x_shifted) / np.sum(np.exp(x_shifted))
        obj = compute_obj(a)
        
        if obj < best_obj:
            best_obj = obj
            best_x = x.copy()
        
        if verbose:
            elapsed = time.monotonic() - start
            print(f"    β={beta:.1f}, obj={obj:.8f}, best={best_obj:.8f}, iter={result.nit}, t={elapsed:.0f}s")
        
        # Increase beta
        beta *= 2.0
        if beta > max_beta:
            # A few more iterations at max_beta
            beta = max_beta
            break
        
        if time.monotonic() - start > time_budget - 30:
            break
    
    # Final: get the best sequence
    x_shifted = best_x - np.max(best_x)
    a_best = np.exp(x_shifted) / np.sum(np.exp(x_shifted))
    
    return a_best

"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_valid_corr(w, a):
    """Compute g_i = sum_{m=0}^{n-1} a[m] * w[i+m] for i=0..n-1 via FFT."""
    n = len(a)
    a_rev = a[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    f_a = np.fft.fft(a_rev, size)
    f_w = np.fft.fft(w, size)
    conv = np.fft.ifft(f_a * f_w).real
    return conv[n - 1:2 * n - 1]


def softmax_smoothed(x, beta):
    """
    Compute smoothed objective and gradient via softmax parametrization.
    a = softmax(x), so sum(a)=1, a>0 automatically.
    """
    n = len(x)
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / np.sum(exp_x)

    c = compute_conv(a)

    c_max = np.max(c)
    logits = beta * (c - c_max)
    logits = np.clip(logits, -50.0, 50.0)
    exp_logits = np.exp(logits)
    sum_exp = np.sum(exp_logits)
    obj = c_max + np.log(sum_exp) / beta

    w = exp_logits / sum_exp
    g = compute_valid_corr(w, a)
    grad_a = 2.0 * g / beta

    dot = np.sum(grad_a * a)
    grad_x = a * (grad_a - dot)

    return obj, grad_x


def pnorm_smoothed(x, p):
    """
    Compute p-norm smoothed objective and gradient via softmax parametrization.
    f_p = (sum_k c_k^p)^(1/p), approximates max for large p.
    """
    n = len(x)
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / np.sum(exp_x)

    c = compute_conv(a)
    c = np.maximum(c, 0) # ensure non-negative for power
    c_max = np.max(c)
    c_norm = c / c_max  # in [0, 1]
    c_p = c_norm ** p
    sum_cp = np.sum(c_p)
    obj = c_max * sum_cp ** (1.0 / p)

    # Gradient: d(f_p)/da_i = (sum c_k^p)^(1/p - 1) * sum_k c_k^(p-1) * 2*a_{k-i}*1[valid]
    # = c_max^(p-1) * (sum c_norm^p)^(1/p - 1) * sum_k (c_norm_k)^(p-1) * 2*a_{k-i}
    # Simplify: let S = sum c_norm^p, then f = c_max * S^(1/p)
    # df/dc_k = c_max * (1/p) * S^(1/p - 1) * p * c_norm_k^(p-1) = c_max * S^(1/p-1) * c_norm_k^(p-1)
    # = c_max * S^(1/p-1) * (c_k/c_max)^(p-1) = S^(1/p-1) * c_k^(p-1) / c_max^(p-2)
    
    # Actually let's just do it numerically-friendly:
    # weights_k = c_k^(p-1) / (sum c_j^p)^(1/p - 1) ... this gets messy
    # Let me just use: grad_f/dc_k = c_k^(p-1) / (sum c_j^p)^(1-1/p)
    # Then grad_f/da_i = sum_k (grad_f/dc_k) * 2*a_{k-i} * 1[valid]
    
    cp_1 = c_norm ** (p - 1)  # c_k^(p-1) / c_max^(p-1)
    # Sum of c_k^p = c_max^p * sum c_norm^p, so (sum c_k^p)^(1/p - 1) = c_max^(1-p) * (sum c_norm^p)^(1/p-1)
    # grad_f/d c_k = c_k^(p-1) * (sum c_j^p)^(1/p - 1) = c_max^(p-1) * c_norm_k^(p-1) * c_max^(1-p) * sum_cp^(1/p-1)
    # = c_norm_k^(p-1) * sum_cp^(1/p-1)
    
    weights = cp_1 * (sum_cp ** (1.0 / p - 1.0))
    g = compute_valid_corr(weights, a)
    grad_a = 2.0 * g

    dot = np.sum(grad_a * a)
    grad_x = a * (grad_a - dot)

    return obj, grad_x


def lsmm_optimize(a_init, time_budget=300.0, verbose=True):
    """

    a_init = np.maximum(a_init, 1e-10).astype(np.float64)
    a_init = a_init / np.sum(a_init)

    best_x = x.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()

    c_init = compute_conv(a_init)
    c_range = float(np.max(c_init) - np.min(c_init))
    if c_range < 1e-10:
        c_range = float(np.max(c_init))
    if c_range < 1e-10:
        c_range = 1.0

    beta = max(3.0 / c_range, 1.0)
    beta_max = max(500.0 / c_range, 200.0)

    use_pnorm = False
    p_current = 4.0

    n_stages = 0
    stall_count = 0

    while time.monotonic() - start < time_budget - 10:
        n_stages += 1
        remaining = time_budget - (time.monotonic() - start)
        stage_time = min(remaining * 0.25, 30.0)
        if stage_time < 3.0:
            stage_time = remaining

        # Choose between log-sum-exp and p-norm based on beta level
        if use_pnorm:
            func = pnorm_smoothed
            args = (p_current,)
        else:
            func = softmax_smoothed
            args = (beta,)

        try:
            result = minimize(
                func, x, args=args,
                method='L-BFGS-B',
                options={'maxiter': 300, 'gtol': 1e-10, 'ftol': 1e-14,
                         'maxfun': 1000, 'disp': False}
            )
            x = result.x
        except Exception:
            if use_pnorm:
                p_current = max(p_current * 0.5, 1.0)
            else:
                beta = max(beta * 0.5, 1.0)
            continue

        x_shifted = x - np.max(x)
        a_curr = np.exp(x_shifted) / np.sum(np.exp(x_shifted))
        obj = compute_obj(a_curr)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_x = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if verbose and n_stages % 3 == 1:
            elapsed = time.monotonic() - start
            tag = f"p={p_current:.1f}" if use_pnorm else f"beta={beta:.1f}"

        # Advance continuation parameter
        if not use_pnorm:
            beta *= 2.0
            if beta > beta_max * 0.5:
                # Switch to p-norm for the final stages
                use_pnorm = True
                p_current = 16.0
        else:
            p_current *= 2.0
            if p_current > 2048.0:
                p_current = 2048.0
                if stall_count > 3:
                    break

        if stall_count > 8:
            # Try a small perturbation to escape
            rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
            x = best_x + rng.normal(0, 0.05, n)
            stall_count = 0

    x_shifted = best_x - np.max(best_x)
    a_best = np.exp(x_shifted) / np.sum(np.exp(x_shifted))
    return a_best


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. A@g <= M*1, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt, dual_y) or (None, None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    try:
        import cvxpy as cp
        z = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        c1 = A @ z <= M
        c2 = cp.sum(z) == 1
        prob = cp.Problem(cp.Minimize(M), [c1, c2])

        for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    z_val = np.maximum(z.value, 0)
                    zs = float(np.sum(z_val))
                    if zs > 1e-10:
                        z_val = z_val / zs
                        dual_y = None
                        try:
                            dual_y = np.maximum(np.array(c1.dual_value), 0)
                        except:
                            pass
                        return z_val, float(M.value), dual_y
            except:
                continue
    except:
        pass

    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n]), None
    except:
        pass

    return None, None, None
