def compute_conv(a):
    """Compute linear convolution a * a via FFT."""
    n = len(a)
    # Determine the size for FFT
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    return conv

def compute_obj(a):

    return 2.0 * n * max_conv_val / (s ** 2)
