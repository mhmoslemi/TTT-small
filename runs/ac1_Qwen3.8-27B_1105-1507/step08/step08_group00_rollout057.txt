"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Novel approaches: bisection+penalty feasibility search, multiplicative deflation,
bilinear LP alternation, boundary-tapered initialization, multi-scale refinement,
and adaptive smoothed gradient descent with peak-targeted updates.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Returns (gradient, conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_smooth_descent(a, time_budget=250.0, tau_init=5.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        if it % 3000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_grad(y, tau)

        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9997)

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            best_it = it

    return best_a


def deflation_iteration(a, K=5, n_lambdas=15, verbose=False):
    """
    One iteration of multiplicative deflation.
    Reduces amplitude at peak-contributing indices multiplicatively.
    Returns updated a.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    if max_conv < 1e-15:
        return a

    top_idx = np.argsort(conv)[::-1][:K]
    top_vals = conv[top_idx]

    tau_w = 8.0 / max(top_vals[0], 1e-10)
    weights = np.exp(tau_w * (top_vals - top_vals[0]))
    weights /= weights.sum()

    d = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        j_start = k - i_end
        j_end = k - i_start
        d[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

    d_max = float(np.max(d))
    if d_max < 1e-15:
        return a
    d = d / d_max

    best_a = a.copy()
    best_obj = compute_obj(a)

    lambdas = np.linspace(0.05, 2.5, n_lambdas)
    for lam in lambdas:
        a_try = a * np.exp(-lam * d)
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a

def penalty_feasibility(a_init, M, max_iters=500, verbose=False):
    """
    Check if there exists a with sum(a)=1, a>=0, conv(a,a)[k] <= M for all k.
    Uses penalty method with gradient descent.
    Returns (a_feasible, min_violation) where min_violation ~ 0 means feasible.
    """
    n = len(a_init)
    a = a_init.copy()
    s = float(np.sum(a))
    a = a / s

    best_violation = float('inf')
    best_a = a.copy()
    lr = 5.0 / n
    momentum = 0.8
    v = np.zeros(n)

    for it in range(max_iters):
        conv = compute_conv(a)
        violations = np.maximum(conv - M, 0)
        total_violation = float(np.sum(violations ** 2))

        if total_violation < best_violation:
            best_violation = total_violation
            best_a = a.copy()

        if total_violation < 1e-14:
            break

        # Gradient of penalty + constraint
        # d(sum max(conv[k]-M,0)^2)/da[i] = 2*sum_k max(conv[k]-M,0) * d(conv[k])/da[i]
        # d(conv[k])/da[i] = a[k-i] + a[i-k] (with bounds)
        active_mask = violations > 0
        if not np.any(active_mask):
            break

        w = violations  # weights = max(conv[k]-M, 0)

        # Gradient via FFT cross-correlation (similar to smoothed grad)
        N = 1
        while N < 3 * n:
            N <<= 1
        W = fft(w, N)
        A = fft(a, N)
        g_full = ifft(W * np.conj(A)).real
        g = 2.0 * g_full[:n]

        # Add constraint gradient: 2*mu*(sum(a)-1)
        mu = 10.0
        g_sum = 2.0 * mu * (float(np.sum(a)) - 1.0)

        v = momentum * v + g + g_sum
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = best_a.copy()
            v = np.zeros(n)

        # Adaptive lr
        obj_new_guess = float(np.max(compute_conv(a_new)))
        if obj_new_guess > M * 2:
            lr *= 0.5
            a = a_new  # still update but with smaller step next
        else:
            lr = min(lr * 1.01, 10.0 / n)

        a = a_new

    return best_a, best_violation


def bisection_search(a_init, M_high, time_budget=250.0, verbose=False):
    """
    Binary search on M to find the minimum feasible peak level.
    Uses penalty method for feasibility check.
    Returns (best_a, best_M).
    """
    n = len(a_init)
    M_low = 0.4 / n  # optimistic lower bound
    M_high = M_high

    best_a = a_init.copy()
    best_M = M_high

    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget and (M_high - M_low) > 1e-8:
        M_mid = (M_low + M_high) / 2.0

        # Use fewer iterations for coarse search
        remaining = time_budget - (time.monotonic() - start)
        iters = min(300, max(50, int(remaining * 2)))

        a_try, violation = penalty_feasibility(a_init, M_mid, max_iters=iters)

        if violation < 1e-8:
            # Feasible! Try lower M
            M_high = M_mid
            obj_try = compute_obj(a_try)
            if obj_try < 2.0 * n * best_M:
                best_M = M_mid
                best_a = a_try.copy()
            if verbose:
                print(f"    Bisect it{it}: M_mid={M_mid:.8f} FEASIBLE, viol={violation:.2e}, obj={obj_try:.8f}")
        else:
            # Not feasible, try higher M
            M_low = M_mid
            if verbose and it % 5 == 0:
                print(f"    Bisect it{it}: M_mid={M_mid:.8f} infeasible, viol={violation:.2e}")

        it += 1

    # Final try at best_M
    a_final, _ = penalty_feasibility(best_a, best_M, max_iters=500)
    obj_final = compute_obj(a_final)
    if obj_final < 2.0 * n * best_M:

    return best_a, best_M


def build_conv_matrix_for_g(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_for_g(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_for_g(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_for_a(g):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(a)=1, a>=0. Returns (a, M)."""
    n = len(g)
    g_norm = g / np.sum(g)

    # conv(a, g)[k] = sum_i a[i] * g[k-i]
    # Build matrix: row k has entries g[k-i] in column i
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = g_norm[j]

    a = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ a <= M, cp.sum(a) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if a.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.maximum(a.value, 0)
                s = float(np.sum(a_val))
                if s > 1e-10:
                    return a_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def bilateral_lp_alternation(a, time_budget=300.0, verbose=False):

                elif it - last_improve > 20:

            else:

            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new

        it += 1

        # Staleness check

    return best_a


def tapered_init(n, alpha=1.5, rng=None, mode=0):
    """Generate tapered initializations with small boundary values."""
    if rng is None:
        rng = np.random.default_rng()

    x = np.linspace(0, 1, n)

    if mode == 0:
        # Power-law taper (symmetric)
        a = (x + 0.01) ** alpha * (1 - x + 0.01) ** alpha
    elif mode == 1:
        # Asymmetric power-law
        a = (x + 0.01) ** (alpha * 1.5) * (1 - x + 0.01) ** alpha
    elif mode == 2:
        # Raised cosine with taper
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a ** alpha
    elif mode == 3:
        # Product of two power laws (peaked in middle)
        center = 0.5 + rng.uniform(-0.1, 0.1)
        width = 0.3 + rng.uniform(0, 0.2)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif mode == 4:
        # Power law with random exponent
        a1 = (x + 0.01) ** rng.uniform(0.5, 3.0)
        a2 = (1 - x + 0.01) ** rng.uniform(0.5, 3.0)
        a = a1 * a2
    elif mode == 5:
        # Bimodal
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.6, 0.8)
        w = rng.uniform(0.05, 0.15)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    else:
        a = (x + 0.01) ** alpha * (1 - x + 0.01) ** alpha

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def multi_scale_optimize(a_best, time_budget=300.0, verbose=False):
    """
    Multi-scale optimization: optimize at smaller n, upscale, refine.
    """
    n_best = len(a_best)
    start = time.monotonic()
    rng = np.random.default_rng(99)

    # Try smaller scales
    small_ns = [max(100, n_best // 4), max(200, n_best // 2)]
    small_ns = sorted(set([n for n in small_ns if n < n_best]))

    for n_small in small_ns:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 60:
            break

        a_s = resample_sequence(a_best, n_small)

        # Optimize at small scale
        alloc = min(remaining * 0.3, 80.0)
        a_s = nesterov_smooth_descent(a_s, time_budget=alloc, tau_init=5.0, tau_end=200.0)
        a_s = deflation_optimize(a_s, time_budget=min(remaining * 0.2, 40.0))

        # Upscale back
        a_up = resample_sequence(a_s, n_best)

        # Refine at full scale
        remaining = time_budget - (time.monotonic() - start)
        alloc2 = min(remaining * 0.2, 40.0)
        a_up = nesterov_smooth_descent(a_up, time_budget=alloc2, tau_init=20.0, tau_end=200.0)

        obj_up = compute_obj(a_up)
        obj_best = compute_obj(a_best)
        if verbose:
            print(f"  Multi-scale n={n_small}: obj_up={obj_up:.8f} vs best={obj_best:.8f}")

        if obj_up < obj_best:
            a_best = a_up

    return a_best
