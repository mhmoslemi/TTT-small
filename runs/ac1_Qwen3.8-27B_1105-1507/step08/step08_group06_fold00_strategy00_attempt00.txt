def evaluate_sequence_normalized(seq: list[float]) -> float:
    """
    Evaluates a sequence to get an accurate score, respecting the important objective.
    """
    if isinstance(seq, list):
        seq_np = np.array(seq)
    else:
        seq_np = np.array(seq)
    s = np.sum(seq_np)
    if s < 0.01:
        # If normalization fails, create a normalized version
        max_val = np.max(seq_np)
        if max_val < 0.01:
            return np.inf
        scaled = seq_np / max_val
        s = np.sum(scaled)
    # NORM
    a = seq_np / s
    conv = np.convolve(a, a, mode="full")
    return 2.0 * len(a) * np.max(conv) / (sum(a) ** 2)

def evaluate_sequence_normal_scaled(sequence: list[float]) -> float:
    """
    Evaluates a sequence with a scaled normalization to compare across different lengths.
    """
    n = len(sequence)
    # Ensure all numbers are within reasonable bounds
    for i, x in enumerate(sequence):
        if x < 0:
            sequence[i] = 0
        if x > 1000:
            sequence[i] = max(sequence[i], 100)  # Bounded by 100

    # Normalization factor in case of non-zero sum
    s = np.sum(sequence)
    sequence = np.array(sequence) / s

    # Compute the real objective considering the multi-term formula
    conv = np.convolve(sequence, sequence, mode="full")
    max_val = np.max(conv)
    obj = 2.0 * n * max_val / (len(sequence) ** 2)
    return obj
