"""
Multi-method optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating, mirror descent with smooth max, coordinate descent with
exact line search, and multi-n exploration. Key improvements over previous:
- Correct FFT-based gradient for smooth max
- Mirror descent (multiplicative weights) for better simplex handling
- Coordinate descent with golden-section line search per coordinate
- Systematic multi-n exploration with diverse initializations
- Adaptive parameter tuning
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_max_and_grad(a, T):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Uses FFT for efficient gradient computation.
    Returns (smooth_value, gradient).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    smooth_val = max_c + T * np.log(total)

    # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] (a[m]=0 for m notin [0,n-1])
    # = 2 * IFFT(FFT(weights) * conj(FFT(a)))[:n] (with zero-padding to size >= 2n-1)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    W = rfft(weights, size)
    A = rfft(a, size)
    cc = irfft(W * np.conj(A), size)
    grad = 2.0 * cc[:n]

    return smooth_val, grad


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp_for_a(a):
    """
    Solve LP: min M s.t. conv(a,g) <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    from scipy.optimize import linprog
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_norm = g_val / s
            conv_check = A @ g_norm
            true_M = float(np.max(conv_check))
            return g_norm, true_M
    return None, None


def solve_lp_reduced(a, active_indices):
    """Solve LP with only active constraints, then verify against all."""
    from scipy.optimize import linprog
    n = len(a)
    a_norm = a / np.sum(a)
    A_full = build_conv_matrix(a_norm)
    A = A_full[active_indices, :]

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((len(active_indices), n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(active_indices))

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_norm = g_val / s
            conv_check = A_full @ g_norm
            true_M = float(np.max(conv_check))
            return g_norm, true_M
    return None, None


def convex_line_search(a, g, n_evals=12):
    """
    Find optimal t in [0,1] for a_new = (1-t)*a + t*g.
    Uses the fact that the objective is convex in t.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    a_new = a_new / s
    return t_opt, a_new

def mirror_descent_smooth(a, time_budget=100.0, T_init=0.002, T_min=0.00003,
                          lr_init=0.005, verbose=False):
    """
    Mirror descent (multiplicative weights) with smooth max objective.
    Work in log-space for numerical stability.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = lr_init
    stall = 0
    log_a = np.log(a)
    log_a = log_a - np.max(log_a)

    max_iters = 3000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Reconstruct from log-space
        a_cur = np.exp(log_a)
        a_cur = a_cur / np.sum(a_cur)

        # Compute gradient of smooth max
        _, grad = smooth_max_and_grad(a_cur, T)

        # Center gradient (invariance to adding constant)
        grad_centered = grad - np.mean(grad)

        # Mirror descent step in log-space
        log_a = log_a - lr * grad_centered

        # Numerical stability: rescale
        log_max = np.max(log_a)
        if log_max > 100 or log_max < -100:
            log_a = log_a - log_max

        # Update temperature (exponential cooling)
        T = max(T_min, T_init * (0.9992 ** it))

        # Adaptive learning rate
        if it % 100 == 0 and it > 0:
            lr = lr * 1.005

        obj_new = compute_obj(a_cur)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_cur.copy()
            stall = 0
            if verbose and it % 200 == 0:
                elapsed = time.monotonic() - start
                print(f"          MD it{it}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.6f}, time={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 150:
                lr *= 0.5
                if lr < 1e-8:
                    break
                log_a = np.log(np.maximum(best_a, 1e-15))
                log_a = log_a - np.max(log_a)
                stall = 0

    return best_a


def projected_smooth_grad(a, time_budget=100.0, T_init=0.001, T_min=0.00005,

    """

    lr = lr_init

    v = np.zeros(n)

    stall = 0

    for it in range(max_iters):

            break

        a_y = np.maximum(a_y, 0)

        else:
            a_y = a

        _, grad = smooth_max_and_grad(a_y, T)

        a_new = a + v
        a_new = simplex_project(a_new)

        a_prev = a.copy()
        a = a_new

        # Temperature cooling
        T = max(T_min, T_init * (0.999 ** it))

        obj_new = compute_obj(a)

            best_obj = obj_new
            best_a = a.copy()
            stall = 0
        else:

                momentum = 0.9
                if lr < 1e-7:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                a_prev = a.copy()
                stall = 0

        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - start
            print(f"          PG it{it}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.6f}, time={elapsed:.1f}s")

def coordinate_descent_ls(a, time_budget=100.0, verbose=False):

    For each coordinate i, find optimal value by 1D search.
    """

            # Search range: [0, 2 * a_old + 0.1/n]

                    return np.inf

                return 2.0 * n * np.max(conv)

            fc = obj_1d(c_pt)
            fd = obj_1d(d_pt)

                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)

                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = obj_1d(d_pt)

            x_opt = (lo + hi) / 2.0
            obj_opt = obj_1d(x_opt)

            if obj_opt < best_obj - 1e-14:
                a[i] = x_opt
                best_obj = obj_opt
                best_a = a.copy()
                improved = True

        if verbose and pass_count % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"          CD pass {pass_count}: obj={best_obj:.8f}, time={elapsed:.1f}s")

        if not improved and pass_count > 3:
            break

    return best_a


def coordinate_descent_fast(a, time_budget=100.0, delta=1e-4, verbose=False):
    """
    Fast coordinate descent with fixed step sizes (faster than golden section).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv_cache = compute_conv(a)
    max_conv = np.max(conv_cache)
    base_obj = 2.0 * n * max_conv

    pass_count = 0
    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count + n * 131)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            a_i_old = a[i]

            # Try perturbations
            for delta_try in [delta, delta * 0.1, delta * 5]:
                a[i] = a_i_old + delta_try
                if a[i] < 0:
                    a[i] = a_i_old
                    continue
                s_try = float(np.sum(a))
                if s_try < 0.01:
                    a[i] = a_i_old
                    continue
                a_norm = a / s_try
                conv_try = compute_conv(a_norm)
                obj_try = 2.0 * n * np.max(conv_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a.copy()
                    a[i] /= (s_try)  # normalize
                    a = best_a.copy()
                    improved = True
                    break
                else:
                    a[i] = a_i_old

            if improved:
                continue

            # Try decreasing
            for delta_try in [delta, delta * 0.1]:
                if a[i] < delta_try * 0.5:
                    continue
                a[i] = a_i_old - delta_try
                s_try = float(np.sum(a))
                if s_try < 0.01:
                    a[i] = a_i_old
                    continue
                a_norm = a / s_try
                conv_try = compute_conv(a_norm)
                obj_try = 2.0 * n * np.max(conv_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a.copy()
                    a = a_norm
                    improved = True
                    break
                else:
                    a[i] = a_i_old

        if verbose and pass_count % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"          CDf pass {pass_count}: obj={best_obj:.8f}, delta={delta:.2e}, time={elapsed:.1f}s")

        if not improved:
            delta *= 0.5
            if delta < 1e-8:
                break

    return best_a


def simulated_annealing_perturb(a, time_budget=60.0, verbose=False):
    """
    Simulated annealing: randomly perturb and accept based on temperature.
    Helps escape local minima.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = 0.01
    T_final = 0.0001
    rng = np.random.default_rng(42)
    cur_obj = best_obj

    max_iters = 2000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Geometric cooling
        T = T_final + (T - T_final) * (0.9995 ** it)

        # Random perturbation
        scale = 0.01 * (1 + T / T_final * 0.5)
        a_try = a * np.exp(rng.normal(0, scale, n))
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 0.01:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)

        # Acceptance criterion
        if obj_try < cur_obj:
            a = a_try
            cur_obj = obj_try
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                if verbose and it % 200 == 0:
                    print(f"          SA it{it}: obj={best_obj:.8f}, T={T:.6f}")
        else:
            # Metropolis acceptance
            if np.log(rng.random()) < (cur_obj - obj_try) / max(T * 0.01, 1e-10):
                a = a_try
                cur_obj = obj_try

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng):
    """Generate diverse initial sequences of length n."""
    methods = []

    # Uniform
    methods.append(np.ones(n) / n)

    # Sinusoidal
    x = np.linspace(0, 2 * np.pi, n)
    s = 1 + 0.3 * np.cos(x)
    methods.append(s / np.sum(s))

    # Two-peak
    a = np.zeros(n)
    a[:n//4] = 1.0
    a[3*n//4:] = 1.0
    methods.append(a / np.sum(a))

    # Random
    a = rng.random(n) + 0.1
    methods.append(a / np.sum(a))

    # Power law
    x = np.linspace(1, n, n)
    a = x ** -0.5
    methods.append(a / np.sum(a))

    # Block pattern
    a = np.zeros(n)
    block_size = max(1, n // 10)
    for i in range(0, n, block_size):
        a[i:i+block_size] = 1.0 + 0.5 * np.cos(2 * np.pi * i / n)
    a = np.maximum(a, 0)
    if np.sum(a) > 0:
        methods.append(a / np.sum(a))

    return methods


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple optimization methods.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_best = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                n_best = len(parent)
                print(f"Parent (n={n_best}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_best = 800
        best_seq = np.ones(n_best) / n_best
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_best}): obj={best_score:.8f}")

    a = best_seq.copy()

    # === Phase 1: LP Alternating from parent ===
    print(f"\n=== Phase 1: LP Alternating (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    phase1_budget = min(250.0, remaining * 0.30)
    a_opt = lp_alternating(a, time_budget=phase1_budget, verbose=True)
    obj_opt = compute_obj(a_opt)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        a = a_opt.copy()
    print(f"  Phase 1 done: obj={best_score:.8f}, time={time.monotonic()-start_time:.1f}s")

    # === Phase 2: Mirror Descent with Smooth Max ===
    print(f"\n=== Phase 2: Mirror Descent (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        phase2_budget = min(150.0, remaining * 0.15)
        a_md = mirror_descent_smooth(a, time_budget=phase2_budget,
                                     T_init=0.003, T_min=0.00005, lr_init=0.003, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            a = a_md.copy()
            print(f"  Phase 2 improved: obj={best_score:.8f}")
        else:
            print(f"  Phase 2: obj={obj_md:.8f} (no improvement)")

    # === Phase 3: Projected Gradient with Nesterov ===
    print(f"\n=== Phase 3: Projected Gradient (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        phase3_budget = min(100.0, remaining * 0.12)
        a_pg = projected_smooth_grad(a, time_budget=phase3_budget,
                                     T_init=0.002, T_min=0.00005, lr_init=0.008, verbose=True)
        obj_pg = compute_obj(a_pg)
        if obj_pg < best_score:
            best_score = obj_pg
            best_seq = a_pg.copy()
            a = a_pg.copy()
            print(f"  Phase 3 improved: obj={best_score:.8f}")
        else:
            print(f"  Phase 3: obj={obj_pg:.8f} (no improvement)")

    # === Phase 4: LP Alternating again (from improved point) ===
    print(f"\n=== Phase 4: LP Alternating Round 2 (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        phase4_budget = min(200.0, remaining * 0.22)
        a_opt = lp_alternating(a, time_budget=phase4_budget, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  Phase 4 improved: obj={best_score:.8f}")
        else:
            print(f"  Phase 4: obj={obj_opt:.8f} (no improvement)")

    # === Phase 5: Coordinate Descent with Line Search ===
    print(f"\n=== Phase 5: Coordinate Descent LS (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        phase5_budget = min(120.0, remaining * 0.15)
        a_cd = coordinate_descent_ls(a, time_budget=phase5_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a = a_cd.copy()
            print(f"  Phase 5 improved: obj={best_score:.8f}")
        else:
            print(f"  Phase 5: obj={obj_cd:.8f} (no improvement)")

    # === Phase 6: Simulated Annealing for escape ===
    print(f"\n=== Phase 6: Simulated Annealing (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        phase6_budget = min(60.0, remaining * 0.10)
        a_sa = simulated_annealing_perturb(a, time_budget=phase6_budget, verbose=True)
        obj_sa = compute_obj(a_sa)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()

            print(f"  Phase 6 improved: obj={best_score:.8f}")

            print(f"  Phase 6: obj={obj_sa:.8f} (no improvement)")

    # === Phase 7: Multi-n Exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best - 50, n_best - 25, n_best, n_best + 25, n_best + 50, n_best + 100,
        600, 650, 700, 750, 800, 850, 900, 950, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc_lp = min(remaining * 0.08, 30.0)
        a_opt = lp_alternating(a_try, time_budget=alloc_lp)
        obj_try = compute_obj(a_opt)

        remaining = deadline - time.monotonic()
        if remaining > 20 and obj_try < best_score + 0.01:
            a_opt = coordinate_descent_fast(a_opt, time_budget=min(20.0, remaining * 0.2), delta=5e-5)
            obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Diverse Initializations ===
    print(f"\n=== Phase 8: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for n_try in [n_best, n_best - 50, n_best + 50]:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            n_try = max(400, min(1500, n_try))
            inits = diverse_init(n_try, rng)
            for init in inits[:3]:
                remaining = deadline - time.monotonic()
                if remaining < 20:
                    break
                alloc = min(remaining * 0.06, 20.0)
                a_opt = lp_alternating(init, time_budget=alloc)
                obj_try = compute_obj(a_opt)
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_opt.copy()
                    print(f"  ** Diverse n={n_try}: obj={best_score:.8f} **")

    # === Phase 9: Final Polish ===
    print(f"\n=== Phase 9: Final Polish (n={len(best_seq)}) ===")
    a_final = best_seq.copy()
    remaining = deadline - time.monotonic()

    if remaining > 40:
        alloc = min(80.0, remaining * 0.35)
        a_lp = lp_alternating(a_final, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_final = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(60.0, remaining * 0.35)
        a_cd = coordinate_descent_ls(a_final, time_budget=alloc, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a_final = a_cd.copy()
            print(f"  Final CD-LS: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(40.0, remaining * 0.4)
        a_cd2 = coordinate_descent_fast(a_final, time_budget=alloc, delta=1e-5)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Final CD-fast: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(30.0, remaining * 0.4)
        a_md = mirror_descent_smooth(a_final, time_budget=alloc,
                                     T_init=0.0005, T_min=0.00002, lr_init=0.002)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Final MD: obj={best_score:.8f}")

    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if np.isfinite(val) and val < best_score:
            best_score = val
            print(f"  evaluate_sequence confirms: {val:.8f}")
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
