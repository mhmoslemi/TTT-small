```python
"""
Smoothed-max mirror descent with LP polishing for step function optimization.
Uses exponential weights (mirror descent) on the smoothed max-convolution objective,
combined with LP-based polishing and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n]

def compute_conv(a):
    """Compute a*a using FFT."""
    return fft_convolve(a, a)

def compute_obj(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))

def mirror_descent_smoothmax(f, beta, eta0, n_iter, verbose=False):
    """
    Mirror descent (exponential weights) on smoothed max-convolution.
    f: initial sequence (non-negative, sum=1)
    beta: temperature for softmax
    eta0: initial step size
    n_iter: number of iterations
    """
    n = len(f)
    f = np.maximum(f, 0).astype(np.float64).copy()
    s = float(np.sum(f))
    if s < 1e-10:
        return f
    f = f / s
    
    best_f = f.copy()
    best_obj = compute_obj(f)
    
    momentum = np.zeros(n)
    
    for t in range(n_iter):
        conv = compute_conv(f)
        
        # Softmax weights (numerically stable)
        conv_max = np.max(conv)
        exp_conv = np.exp(beta * (conv - conv_max))
        w = exp_conv / np.sum(exp_conv)
        
        # Compute gradient: g = correlate(w, f, 'valid')
        g = np.correlate(w, f, mode='valid')
        g = np.maximum(g, 1e-15)  # avoid log of zero
        
        # Nesterov momentum
        alpha = 2.0 / (t + 2)
        momentum = alpha * g + (1 - alpha) * momentum
        
        # Step size (diminishing)
        eta = eta0 / (1.0 + t * eta0 / n)
        
        # Mirror descent update
        log_f = np.log(f + 1e-30) - eta * momentum
        log_f = log_f - np.max(log_f)  # for numerical stability
        f_new = np.exp(log_f)
        f_new = np.maximum(f_new, 0)
        s = float(np.sum(f_new))
        if s < 1e-10:
            break
        f_new = f_new / s
        
        obj = compute_obj(f_new)
        if obj < best_obj:
            best_obj = obj
            best_f = f_new.copy()
        
        f = f_new
    
    return best_f
```

```python
"""
Smoothed-max mirror descent with LP hybrid polishing for step function optimization.
Uses exponential weights (mirror descent) on the softmax-smoothed max-convolution,
combined with periodic LP best-response updates, multi-n exploration, and coordinate polish.
Key innovation over LP-only approach: many cheap O(n log n) gradient steps between
expensive LP solves, with adaptive active-set annealing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n]


def compute_conv(a):
    """Compute a*a (linear auto-convolution) via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)_k <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ z <= M, cp.sum(z) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    return z_val, float(M.value)
        except Exception:
            continue

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n])
    except Exception:
        pass

    return None, None


def mirror_descent_smoothmax(f, start_time, time_budget, beta_init=10.0,
                             beta_final=1.0, eta_init=0.5, verbose=False):
    """
    Mirror descent (exponential weights) on smoothed max-convolution with annealing.
    f: non-negative sequence, will be normalized internally.
    Returns best sequence found.
    """
    n = len(f)
    f = np.maximum(f, 0).astype(np.float64).copy()
    s = float(np.sum(f))
    if s < 1e-10:
        return f
    f = f / s

    best_f = f.copy()
    best_obj = compute_obj(f)
    deadline = start_time + time_budget
    t = 0
    no_improve = 0

    # Exponential annealing schedule
    total_est = 5000

    while time.monotonic() < deadline:
        # Adaptive beta
        progress = min(1.0, t / total_est)
        beta = beta_init * (beta_final / beta_init) ** progress
        # Increase beta back towards the end for sharper focus
        if progress > 0.8:
            beta = beta_final + (beta_init - beta_final) * (progress - 0.8) / 0.2

        # Step size: diminishing
        eta = eta_init / (1.0 + 0.01 * t)

        conv = compute_conv(f)

        # Softmax weights (numerically stable)
        conv_shifted = conv - np.max(conv)
        exp_conv = np.exp(beta * conv_shifted)
        w = exp_conv / np.sum(exp_conv)

        # Compute gradient: g_i = sum_j f_j * w_{i+j}
        # This is np.correlate(w, f, 'valid')
        if len(w) >= n:
            g = np.correlate(w, f, mode='valid')
        else:
            # Pad w if needed
            w_pad = np.zeros(2 * n - 1)
            w_pad[:len(w)] = w
            g = np.correlate(w_pad, f, mode='valid')

        g = np.maximum(g, 1e-15)

        # Mirror descent update: f_i *= exp(-eta * g_i), then normalize
        log_factor = -eta * g
        log_factor -= np.max(log_factor)  # numerical stability
        f_new = f * np.exp(log_factor)
        f_new = np.maximum(f_new, 1e-30)
        s_new = float(np.sum(f_new))
        if s_new < 1e-10:
            break
        f_new = f_new / s_new

        obj = compute_obj(f_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        t += 1

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - start_time
            # Compute flatness diagnostic
            active = f > 1e-10
            if np.sum(active) > 0:
                g_active = g[active]
                flatness = np.std(g_active) / (np.mean(g_active) + 1e-15)
            else:
                flatness = 0
            print(f"    MD: t={t}, obj={best_obj:.8f}, beta={beta:.1f}, "
                  f"eta={eta:.4f}, flatness={flatness:.4f}, time={elapsed:.0f}s")

        # If stuck for too long, perturb
        if no_improve > 500:
            rng = np.random.default_rng(12345 + t)
            f = best_f * np.exp(rng.normal(0, 0.03, n))
            f = np.maximum(f, 0)
            f = f / np.sum(f)
            no_improve = 0

    return best_f


def mirror_descent_harmonic(f, start_time, time_budget, top_k=20, eta_init=0.3, verbose=False):
    """
    Harmonic (deflation) mirror descent: f_i <- f_i / (1 + eta * g_i).
    Uses hard top-k active set.
    """
    n = len(f)
    f = np.maximum(f, 0).astype(np.float64).copy()
    s = float(np.sum(f))
    if s < 1e-10:
        return f
    f = f / s

    best_f = f.copy()
    best_obj = compute_obj(f)
    deadline = start_time + time_budget
    t = 0
    no_improve = 0

    while time.monotonic() < deadline:
        conv = compute_conv(f)

        # Hard top-k active set
        k = min(top_k, len(conv))
        top_indices = np.argpartition(conv, -k)[-k:]
        w = np.zeros(len(conv))
        w[top_indices] = 1.0 / k

        # Gradient
        g = np.correlate(w, f, mode='valid')
        g = np.maximum(g, 1e-10)

        # Harmonic update
        eta = eta_init / (1.0 + 0.005 * t)
        f_new = f / (1.0 + eta * g)
        f_new = np.maximum(f_new, 0)
        s_new = float(np.sum(f_new))
        if s_new < 1e-10:
            break
        f_new = f_new / s_new

        obj = compute_obj(f_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        t += 1

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    Harmonic: t={t}, obj={best_obj:.8f}, eta={eta:.4f}, time={elapsed:.0f}s")

        if no_improve > 800:
            rng = np.random.default_rng(99999 + t)
            f = best_f * np.exp(rng.normal(0, 0.02, n))
            f = np.maximum(f, 0)
            f = f / np.sum(f)
            no_improve = 0

    return best_f


def lp_hybrid_alternating(f, start_time, time_budget, lp_every=100, verbose=False):
    """
    Hybrid: mirror descent steps with periodic LP best-response mixing.
    """
    n = len(f)
    f = np.maximum(f, 0).astype(np.float64).copy()
    f = f / np.sum(f)

    best_f = f.copy()
    best_obj = compute_obj(f)
    deadline = start_time + time_budget
    t = 0
    beta = 5.0
    eta = 0.3

    while time.monotonic() < deadline:
        # Mirror descent step
        conv = compute_conv(f)
        conv_shifted = conv - np.max(conv)
        exp_conv = np.exp(beta * conv_shifted)
        w = exp_conv / np.sum(exp_conv)
        g = np.correlate(w, f, mode='valid')
        g = np.maximum(g, 1e-15)

        log_factor = -eta * g
        log_factor -= np.max(log_factor)
        f_new = f * np.exp(log_factor)
        f_new = np.maximum(f_new, 1e-30)
        f_new = f_new / np.sum(f_new)

        obj = compute_obj(f_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()

        f = f_new
        t += 1

        # Periodic LP update
        if t % lp_every == 0 and time.monotonic() < deadline - 5:
            g_lp, M_lp = solve_lp(f)
            if g_lp is not None:
                # Mix with LP solution
                for mix in [0.3, 0.5, 0.7]:
                    f_try = (1 - mix) * f + mix * g_lp
                    f_try = np.maximum(f_try, 0)
                    f_try = f_try / np.sum(f_try)
                    obj_try = compute_obj(f_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_f = f_try.copy()
                        f = f_try
                        break
                # Anneal beta
                beta = max(1.0, beta * 0.95)
                eta = max(0.05, eta * 0.98)

    return best_f


def coordinate_polish(f, start_time, time_budget, n_trials=2000):
    """
    Random coordinate descent: perturb pairs of coordinates.
    """
    n = len(f)
    f = np.maximum(f, 0).astype(np.float64).copy()
    f = f / np.sum(f)

    best_f = f.copy()
    best_obj = compute_obj(f)
    deadline = start_time + time_budget
    rng = np.random.default_rng(777)

    count = 0
    while time.monotonic() < deadline and count < n_trials:
        count += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        for frac in [0.02, 0.05, 0.1, 0.01, 0.15]:
            delta = frac * min(f[i], f[j])
            if delta < 1e-12:
                continue
            a_try = f.copy()
            a_try[i] -= delta
            a_try[j] += delta
            if a_try[i] < 0 or a_try[j] < 0:
                continue
            a_try = a_try / np.sum(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_f = a_try.copy()
                f = a_try
                break

    return best_f


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        for n_init in [500, 800, 1000, 1200]:
            t_arr = np.linspace(0, 1, n_init)
            a_init = 0.5 + 0.5 * np.cos(2 * np.pi * t_arr)
            a_init = np.maximum(a_init, 0.01)
            a_init = a_init / np.sum(a_init)
            obj_init = compute_obj(a_init)
            if obj_init < best_score:
                best_score = obj_init
                best_seq = a_init.copy()
        if best_seq is None:
            best_seq = np.ones(800) / 800
            best_score = compute_obj(best_seq)
        print(f"Initial (n={len(best_seq)}): obj={best_score:.8f}")

    a = best_seq.copy()
    n_parent = len(a)

    # === Phase 1: Annealed Mirror Descent (main optimizer) ===
    print(f"\n=== Phase 1: Annealed MD (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(350.0, remaining * 0.3)
    a_new = mirror_descent_smoothmax(a, start_time, alloc,
                                     beta_init=20.0, beta_final=2.0,
                                     eta_init=0.4, verbose=True)
    obj = compute_obj(a_new)
    if obj < best_score:
        best_score = obj
        best_seq = a_new.copy()
        a = a_new.copy()
        print(f"  Phase 1a done: obj={best_score:.8f} **")
    else:
        print(f"  Phase 1a done: obj={obj:.8f}")

    # Harmonic MD variant
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(150.0, remaining * 0.15)
        a_new = mirror_descent_harmonic(a, start_time, alloc,
                                        top_k=30, eta_init=0.2, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 1b (harmonic) done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1b (harmonic) done: obj={obj:.8f}")

    # === Phase 2: LP Hybrid Alternating ===
    print(f"\n=== Phase 2: LP Hybrid (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        alloc = min(200.0, remaining * 0.2)
        a_new = lp_hybrid_alternating(a, start_time, alloc,
                                      lp_every=200, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 2 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 done: obj={obj:.8f}")

    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_candidates = sorted(set([
        n_parent, n_parent - 100, n_parent + 100,
        n_parent - 200, n_parent + 200, n_parent + 400,
        500, 600, 700, 900, 1000, 1200, 1400, 1600,
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 2500]
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample_sequence(best_seq, n_try)
        # Quick MD on new n
        alloc = min(40.0, remaining * 0.12)
        a_opt = mirror_descent_smoothmax(a_try, start_time, alloc,
                                         beta_init=15.0, beta_final=3.0,
                                         eta_init=0.3)
        # Quick LP polish
        remaining = deadline - time.monotonic()
        if remaining > 30 and compute_obj(a_opt) < best_score + 0.01:
            alloc_lp = min(20.0, remaining * 0.08)
            a_opt = lp_polish(a_opt, start_time, alloc_lp)

        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Targeted LP Polish on Best ===
    print(f"\n=== Phase 4: LP Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(200.0, remaining * 0.25)
        a_new = lp_polish(best_seq, start_time, alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 4 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4 done: obj={obj:.8f}")

    # === Phase 5: Second MD pass with different params ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 5: Second MD (n={len(best_seq)}) ===")
        alloc = min(120.0, remaining * 0.15)
        a_new = mirror_descent_smoothmax(best_seq, start_time, alloc,
                                         beta_init=50.0, beta_final=1.0,
                                         eta_init=0.2, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 5 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 5 done: obj={obj:.8f}")

    # === Phase 6: Coordinate Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Coordinate Polish ===")
        alloc = min(60.0, remaining * 0.15)
        a_new = coordinate_polish(best_seq, start_time, alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 6 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 6 done: obj={obj:.8f}")

    # === Phase 7: Random Restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 7: Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(5, int(remaining // 40))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            if r % 3 == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.03 + 0.01 * r, n_best))
            elif r % 3 == 1:
                a_r = best_seq.copy()
                n_blocks = max(5, n_best // 100)
                block_size = n_best // n_blocks
                blocks = rng.choice(n_blocks, min(3, n_blocks), replace=False)
                for b in blocks:
                    si = int(b) * block_size
                    ei = si + block_size
                    a_r[si:ei] *= rng.uniform(0.5, 1.5)
            else:
                a_r = best_seq.copy()
                n_sparse = max(5, n_best // 50)
                idx = rng.choice(n_best, n_sparse, replace=False)
                a_r[idx] *= rng.uniform(0.4, 2.0, n_sparse)

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr

            alloc_r = min(remaining * 0.3, 30.0)
            a_opt = mirror_descent_smoothmax(a_r, start_time, alloc_r,
                                             beta_init=10.0, beta_final=2.0,
                                             eta_init=0.3)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 8: Final LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 30.0)
        a_final = lp_polish(best_seq, start_time, alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
