Nesterov subgradient, and multi-n exploration.

Key insight: At the optimum (full support), KKT gives H_w @ a = c*1 where
H_w[i,j] = w_{i+j}. This reduces each iteration to a linear system solve,

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except Exception:
    HAVE_CVXPY = False


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return np.maximum(irfft(fa * fa, size)[:2 * n - 1], 0.0)


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_hankel(w, n):
    """Build n x n Hankel matrix H[i,j] = w[i+j]."""
    idx = np.add.outer(np.arange(n), np.arange(n))
    return w[idx]


def hankel_fixed_point(a, time_budget=300.0, verbose=True):
    """
    Dual-primal fixed-point iteration based on KKT conditions.
    Given w (dual weights), solve H_w @ x = 1, set a = x/sum(x).
    Update w via softmax on conv(a,a) with adaptive temperature.
    """

    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()

    T = 5.0  # temperature for softmax
    damping = 0.7
    stall = 0
    patience = 100
    it = 0

        M = float(np.max(conv_a))

        if M < 1e-15:
            break

        # Update w: softmax on conv with temperature T
        z = (conv_a - M) / (T * M + 1e-30)
        w = np.exp(z)
        w = w / w.sum()

        # Build Hankel and solve
        H = build_hankel(w, n)
        # Add small regularization for numerical stability
        H_reg = H + 1e-10 * np.eye(n)

        try:
            x = np.linalg.solve(H_reg, np.ones(n))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(H_reg, np.ones(n), rcond=None)[0]
            except Exception:
                T = max(0.1, T * 0.5)
                continue

            x = np.maximum(x, 0)
            s_x = float(np.sum(x))

                T = max(0.1, T * 0.7)
                continue

            x_pos = np.maximum(x, 0)
            s_x = float(np.sum(x_pos))
            if s_x < 1e-10:
                T = max(0.1, T * 0.7)
                continue
            a_new = x_pos / s_x

        a_candidate = (1.0 - damping) * a + damping * a_new
        a_candidate = np.maximum(a_candidate, 0)
        s_c = float(np.sum(a_candidate))
        if s_c > 1e-10:

            a = a_candidate
            stall = 0

        else:
            stall += 1
            a = a_candidate  # still move forward
            if stall > patience:
                # Reset: increase temperature, reduce damping
                T = min(50.0, T * 2.0)
                damping = max(0.3, damping * 0.7)
                stall = 0
                patience = int(patience * 1.3)
                # Small perturbation
                rng = np.random.default_rng(it * 31 + n)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)

        it += 1
        if verbose and it % 100 == 0:
            print(f"    HFP it{it}: obj={best_obj:.8f}, T={T:.3f}, stall={stall}, el={time.monotonic()-start:.1f}s")

    return best_a


def hankel_fixed_point_v2(a, time_budget=300.0, verbose=True):
    """
    Improved version: uses iterative active set for the a>=0 constraint
    and adaptive top-K peak selection instead of full softmax.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    K = max(10, n // 10)  # number of active peaks
    damping = 0.5
    stall = 0
    patience = 150
    it = 0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 1.0:
            break
        conv_a = compute_conv(a)
        M = float(np.max(conv_a))
        if M < 1e-15:
            break

        # Select top-K peaks with smoothing
        top_idx = np.argsort(conv_a)[::-1][:K]
        peak_vals = conv_a[top_idx]
        # Soft weights over top-K
        z = (peak_vals - peak_vals[0]) / (0.1 * peak_vals[0] + 1e-30)
        w_top = np.exp(z)
        w_top = w_top / w_top.sum()

        # Distribute remaining weight uniformly over other positions
        n_other = len(conv_a) - K
        w_rest = (1.0 - 0.95) / max(n_other, 1)  # 5% spread out

        w = np.full(len(conv_a), w_rest)
        for i, idx in enumerate(top_idx):
            w[idx] += w_top[i] * 0.95

        # Build Hankel and solve
        H = build_hankel(w, n)
        H_reg = H + 1e-8 * np.eye(n)

        try:
            x = np.linalg.solve(H_reg, np.ones(n))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.pinv(H_reg) @ np.ones(n)
            except Exception:
                continue

        # Handle non-negativity with active set
        if np.min(x) >= -0.01 * np.max(np.abs(x)):
            x = np.maximum(x, 0)
            s_x = float(np.sum(x))
            if s_x > 1e-10:
                a_new = x / s_x
            else:
                continue
        else:
            # Active set: find support
            support = x > 0
            if support.sum() < 10:
                x = np.maximum(x, 0)
                s_x = float(np.sum(x))
                if s_x > 1e-10:
                    a_new = x / s_x
                else:
                    continue
            else:
                # Solve on support
                H_sub = H[np.ix_(support, support)]
                try:
                    x_sub = np.linalg.solve(H_sub + 1e-8 * np.eye(support.sum()), np.ones(support.sum()))
                    a_new = np.zeros(n)
                    a_new[support] = np.maximum(x_sub, 0)
                    s_x = float(np.sum(a_new))
                    if s_x > 1e-10:
                        a_new = a_new / s_x
                    else:
                        continue
                except np.linalg.LinAlgError:
                    x = np.maximum(x, 0)
                    s_x = float(np.sum(x))
                    if s_x > 1e-10:
                        a_new = x / s_x
                    else:
                        continue

        # Damped update with adaptive damping
        a_candidate = (1.0 - damping) * a + damping * a_new
        a_candidate = np.maximum(a_candidate, 0)
        s_c = float(np.sum(a_candidate))
        if s_c > 1e-10:
            a_candidate = a_candidate / s_c

        obj_candidate = compute_obj(a_candidate)
        if obj_candidate < best_obj - 1e-12:
            best_obj = obj_candidate
            best_a = a_candidate.copy()
            a = a_candidate
            stall = 0
            damping = min(0.9, damping + 0.005)
            # Gradually increase K (more peaks active)
            K = min(2 * n // 3, K + max(1, n // 200))
        else:
            stall += 1
            a = a_candidate
            if stall > patience:
                T_perturb = 0.03 + 0.02 * (stall // patience)
                rng = np.random.default_rng(it * 7 + n)
                a = a * np.exp(rng.normal(0, T_perturb, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                patience = int(patience * 1.2)
                K = max(10, n // 20)
                damping = max(0.3, damping * 0.8)

        it += 1
        if verbose and it % 100 == 0:
            print(f"    HFP2 it{it}: obj={best_obj:.8f}, K={K}, stall={stall}, el={time.monotonic()-start:.1f}s")

    return best_a


def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def nesterov_subgradient(a, time_budget=60.0, K=5, verbose=False):
    """Nesterov-accelerated subgradient method on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 80

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        spread = (peak_vals[0] - peak_vals[-1]) / max_val
        temp = 3.0 / (1.0 + spread * 10.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / max_val)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                g += w * grad_conv_at_peak(y, int(k))

        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                t_k = 1.0
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.3)

        a_prev = a_new.copy()
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        y = y_new.copy()
        t_k = t_new
        it += 1

        if verbose and it % 500 == 0:
            print(f"    NS it{it}: obj={best_obj:.8f}, el={time.monotonic()-start:.1f}s")

    return best_a


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=10.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def lp_alternating(a, time_budget=200.0, verbose=False):
    """LP alternating with adaptive line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    best_t = 0.4
    rng = np.random.default_rng(777)
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        ts = [best_t, best_t * 0.5, best_t * 1.5, 0.2, 0.5, 0.8, 1.0]
        ts = [t for t in ts if 0.01 < t <= 1.0]
        ts = list(set(ts))

        best_t_obj = np.inf
        best_t_val = best_t
        best_a_local = best_a.copy()

        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            best_t = best_t_val
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 10:
                noise_scale = min(0.3, 0.05 + 0.03 * (stall // 10))
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                best_t = 0.4

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.01))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Triangle-like with sharp center
        x = np.arange(n) / n
        a = 1.0 - np.abs(2 * x - 1)
        a = a ** 0.5
        a = a / a.sum()
    elif mode == 6:
        # Semicircle-like
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x**2, 0.001))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def cross_entropy_method(a, n_samples=20, time_budget=60.0):
    """
    Cross-entropy method: sample perturbations, evaluate, update distribution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    mean = np.log(np.maximum(a, 1e-10))
    std = np.full(n, 0.15)
    elite_frac = 0.25

    it = 0
    while time.monotonic() - start < time_budget:
        # Sample
        samples = []
        objs = []
        for _ in range(n_samples):
            perturbation = np.exp(np.random.default_rng(it * n + _).normal(0, 1, n) * std)
            a_s = a * perturbation
            a_s = np.maximum(a_s, 1e-10)
            a_s = a_s / a_s.sum()
            obj_s = compute_obj(a_s)
            samples.append(a_s)
            objs.append(obj_s)

        objs = np.array(objs)
        samples = np.array(samples)

        # Select elites
        n_elite = max(3, int(n_samples * elite_frac))
        elite_idx = np.argsort(objs)[:n_elite]

        # Update mean and std
        elite_samples = samples[elite_idx]
        elite_mean = np.mean(np.log(np.maximum(elite_samples, 1e-15)), axis=0)
        elite_std = np.std(np.log(np.maximum(elite_samples, 1e-15)), axis=0)

        # Smooth update
        mean = 0.7 * mean + 0.3 * elite_mean
        std = 0.8 * std + 0.2 * np.maximum(elite_std, 0.02)
        std = np.clip(std, 0.01, 0.5)

        # Check best
        best_idx = np.argmin(objs)
        if objs[best_idx] < best_obj - 1e-14:
            best_obj = objs[best_idx]
            best_a = samples[best_idx].copy()
            a = best_a

        it += 1
        if it % 10 == 0:
            print(f"    CEM it{it}: best={best_obj:.8f}, el={time.monotonic()-start:.1f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses Hankel-based KKT fixed-point as the primary
    optimizer, with LP alternating and Nesterov as complementary methods.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Hankel Fixed-Point (main new method, ~400s) ===
    print(f"\n=== Phase 1: Hankel Fixed-Point v1 (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 400.0)
    if p1_budget > 30:
        a_hfp = hankel_fixed_point(best_seq, time_budget=p1_budget * 0.6, verbose=True)
        obj_hfp = compute_obj(a_hfp)
        if obj_hfp < best_score:
            best_score = obj_hfp
            best_seq = a_hfp.copy()
            print(f"  HFP1: obj={best_score:.8f} **")
        else:
            print(f"  HFP1: obj={obj_hfp:.8f} (no improvement)")

    remaining = deadline - time.monotonic()
    p1b_budget = min(remaining * 0.15, 200.0)
    if p1b_budget > 20:
        a_hfp2 = hankel_fixed_point_v2(best_seq, time_budget=p1b_budget, verbose=True)
        obj_hfp2 = compute_obj(a_hfp2)
        if obj_hfp2 < best_score:
            best_score = obj_hfp2
            best_seq = a_hfp2.copy()
            print(f"  HFP2: obj={best_score:.8f} **")
        else:
            print(f"  HFP2: obj={obj_hfp2:.8f} (no improvement)")

    # === Phase 2: LP Alternating (~250s) ===
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 20:
        # Run 2 rounds with perturbation between
        for rnd in range(2):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(p2_budget / 2.0, remaining * 0.5)
            if alloc < 15:
                break
            if rnd > 0:
                a_pert = best_seq * np.exp(rng.normal(0, 0.03, len(best_seq)))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
            else:
                a_pert = best_seq.copy()
            a_lp = lp_alternating(a_pert, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP r{rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP r{rnd}: obj={obj_lp:.8f}")

    # === Phase 3: Nesterov Subgradient (~100s) ===
    print(f"\n=== Phase 3: Nesterov Subgradient ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 100.0)
    if p3_budget > 15:
        a_ns = nesterov_subgradient(best_seq, time_budget=p3_budget, K=7, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  NS: obj={best_score:.8f} **")
        else:
            print(f"  NS: obj={obj_ns:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(100, n_current - 200),
        max(100, n_current - 100),
        n_current + 50,
        n_current + 100,
        n_current + 200,
        400, 500, 600, 700, 800, 900, 1000, 1100, 1200, 1500
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50 or n_tried >= 8:
            break
        if abs(n_try - n_current) < 50:
            continue

        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1

        # Hankel + LP for this n
        alloc_h = min(remaining * 0.08, 40.0)
        if alloc_h > 10:
            a_try = hankel_fixed_point_v2(a_try, time_budget=alloc_h, verbose=False)

        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.05, 30.0)
        if alloc_lp > 10:
            a_try = lp_alternating(a_try, time_budget=alloc_lp, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Cross-Entropy / Diverse Restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = r % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # CEM for this restart
        cem_budget = min(remaining * 0.1, 50.0)
        if cem_budget > 10:
            a_opt = cross_entropy_method(a_r, n_samples=15, time_budget=cem_budget)
            obj_cem = compute_obj(a_opt)
            if obj_cem < best_score:
                best_score = obj_cem
                best_seq = a_opt.copy()
                print(f"  Restart {r} CEM (mode={mode}): obj={best_score:.8f} **")

        # Then polish with HFP
        remaining = deadline - time.monotonic()
        hfp_budget = min(remaining * 0.08, 40.0)
        if hfp_budget > 10:
            a_opt2 = hankel_fixed_point_v2(a_r, time_budget=hfp_budget, verbose=False)
            obj_hfp = compute_obj(a_opt2)
            if obj_hfp < best_score:
                best_score = obj_hfp
                best_seq = a_opt2.copy()
                print(f"  Restart {r} HFP (mode={mode}): obj={best_score:.8f} **")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        lp_alloc = min(remaining * 0.25, 60.0)
        a_lp = lp_alternating(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        hfp_alloc = min(remaining * 0.25, 50.0)
        a_hfp = hankel_fixed_point_v2(best_seq, time_budget=hfp_alloc, verbose=True)
        obj_hfp = compute_obj(a_hfp)
        if obj_hfp < best_score:
            best_score = obj_hfp
            best_seq = a_hfp.copy()
            print(f"  Final HFP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        ns_alloc = min(remaining * 0.2, 30.0)
        a_ns = nesterov_subgradient(best_seq, time_budget=ns_alloc, K=10)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Final NS: obj={best_score:.8f} **")

    # One more LP if time allows
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.5, 30.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
