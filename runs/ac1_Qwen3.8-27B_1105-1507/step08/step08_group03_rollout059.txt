"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided Nesterov acceleration.
Key innovations: dual-informed subgradients, adaptive LP with backtracking line search,

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g

def fast_lp_maxsum(a, M_target):
    """
    Solve: max sum(g) s.t. conv(a,g)[k] <= M_target for all k, g >= 0.
    Returns (g_unnormalized, max_sum) or (None, None).
    """
    n = len(a)
    # Build constraint matrix A where (A @ g)_k = conv(a, g)_k
    # A(k, j) = a_{k-j} if 0 <= k-j < n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        length = min(n, 2*n-1-j)
        A[j:j+length, j] = a[:length]
    
    # LP: max -sum(g) s.t. A@g <= M_target, g >= 0
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M_target),
                  bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        g_val = np.maximum(res.x, 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val, s
    return None, None

def fast_lp_minpeak(a):
    """

    Returns (g_normalized, M_opt) or (None, None).
    """

    A = np.zeros((2*n-1, n))
    for j in range(n):
        length = min(n, 2*n-1-j)
        A[j:j+length, j] = a[:length]
    
    # Decision variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize M
    
    # A_ub: conv(a,g) - M <= 0 => A@g - M*1 <= 0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)]*n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    if res.status == 0:

        s = float(np.sum(g_val))
        if s > 1e-10:

def subgradient_method(a0, time_budget=60.0, K=5, verbose=False):
    """Weighted subgradient with adaptive temperature and Polyak step size."""
    n = len(a0)
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Polyak step size needs a lower bound on the objective
    # Theoretical lower bound is 2n/(2n-1) ≈ 1, use 1.0
    obj_lb = 1.0
    lr = 2.0 / np.sqrt(n)
    
    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()
        
        # Adaptive weights from top K peaks
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        spread = (peak_vals[0] - peak_vals[-1]) / (max_val + 1e-15)
        temp = 3.0 / (1.0 + spread * 10.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / (max_val + 1e-15))
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                g += w * grad_conv_at_peak(a, int(k))
        
        # Polyak step size
        obj_curr = compute_obj(a)
        polyak_lr = (obj_curr - obj_lb) / (np.sum(g**2) + 1e-15)
        step = min(lr, polyak_lr)
        
        a_new = a - step * g
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            lr = max(1e-6, lr * 0.999)
        else:
            lr = max(1e-6, lr * 0.95)
        
        a = a_new
        it += 1
        
        if verbose and it % 200 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}, el={time.monotonic()-start:.1f}s")
    
    return best_a

def mirror_descent_fast(a0, time_budget=60.0, verbose=False):
    """Fast mirror descent with multiplicative updates."""
    n = len(a0)
    a = np.maximum(a0, 1e-12).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.0 / np.sqrt(n)
    v = np.zeros(n)
    momentum = 0.5
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()
        
        # Softmax weights over all peaks
        T = 5.0 * max_val
        w = np.exp((conv - max_val) / T)
        w /= w.sum()
        
        # Compute subgradient using top 10% of weights
        threshold = np.percentile(w, 90)
        mask = w > threshold
        if mask.sum() == 0:
            mask[np.argmax(conv)] = True
        w_masked = w[mask]
        w_masked /= w_masked.sum()
        idx_masked = np.where(mask)[0]
        
        g = np.zeros(n)
        for wi, ki in zip(w_masked, idx_masked):
            if wi > 1e-10:
                g += wi * grad_conv_at_peak(a, int(ki))
        
        v = momentum * v + (1 - momentum) * g
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        else:
            lr *= 0.95
        
        a = a_new
        it += 1
    
    return best_a

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided Nesterov acceleration.
Key improvements: aggressive backtracking line search, dual-informed subgradients,
systematic multi-length exploration with structured initializations (chirps, 
quadratic-residue-like patterns), and adaptive momentum scheduling.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        A[j:j + length, j] = a[:length]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g >= 0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10.0})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_maxsum(a, M_target):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M_target, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M_target),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 10.0})
        if res.status == 0:
            g_val = np.maximum(res.x, 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def get_lp_direction(a):
    """Get LP direction, trying minpeak first then maxsum."""
    g_lp, M_lp = solve_lp_minpeak(a)
    if g_lp is not None:
        return g_lp, M_lp, 'minpeak'
    conv_a = compute_conv(a)
    M_cur = float(np.max(conv_a))
    g_lp = solve_lp_maxsum(a, M_cur)
    if g_lp is not None:
        return g_lp, M_cur, 'maxsum'
    return None, None, None


def lp_nesterov(a0, time_budget=300.0, verbose=True, seed=777):
    """
    LP alternating with Nesterov momentum and aggressive backtracking line search.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    a_prev = a.copy()
    t_k = 1.0
    stall = 0
    patience = 25
    rng = np.random.default_rng(seed)
    it = 0
    best_alpha = 0.5

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        g_lp, M_lp, method = get_lp_direction(y)

        if g_lp is None:
            # Subgradient fallback
            conv = compute_conv(y)
            top_idx = np.argsort(conv)[::-1][:5]
            g = np.zeros(n)
            for k in top_idx:
                g += grad_conv_at_peak(y, int(k))
            g /= max(np.linalg.norm(g), 1e-15)
            a_try = project_simplex(y - 0.01 * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
            a_prev = a.copy()
            a = a_try
            y = a.copy()
            t_k = 1.0
            stall += 1
            it += 1
            continue

        # Aggressive backtracking line search
        found = False
        alphas = [best_alpha, best_alpha * 0.6, best_alpha * 1.4, 0.8, 0.4, 0.2, 0.1, 0.05]
        alphas = [max(0.01, min(1.0, al)) for al in alphas]
        alphas = sorted(set(alphas), reverse=True)

        for alpha_try in alphas:
            a_try = (1.0 - alpha_try) * y + alpha_try * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                best_alpha = alpha_try
                found = True
                break

        if found:
            stall = 0
        else:
            stall += 1
            if stall >= patience:
                # Structured restart
                scale = min(0.3, 0.05 + 0.04 * (stall // patience))
                mode = rng.integers(0, 3)
                if mode == 0:
                    a = best_a * np.exp(rng.normal(0, scale, n))
                elif mode == 1:
                    # Chirp perturbation
                    x = np.arange(n) / n
                    chirp = 1.0 + scale * np.sin(2 * np.pi * rng.integers(2, 8) * x)
                    a = best_a * chirp
                else:
                    # Block perturbation
                    blocks = max(1, n // 10)
                    block_scale = np.exp(rng.normal(0, scale, blocks))
                    a = best_a.copy()
                    for b in range(blocks):
                        sl = slice(b * (n // blocks), (b + 1) * (n // blocks))
                        a[sl] *= block_scale[b]
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                best_a = a.copy()
                best_obj = compute_obj(a)
                y = a.copy()
                a_prev = a.copy()
                t_k = 1.0
                best_alpha = 0.4
                stall = 0
                patience = int(patience * 1.3)
                if verbose:
                    print(f"    LP-N restart it{it}: obj={best_obj:.8f}, mode={mode}")

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = t_k / t_new - 1.0
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y > 1e-10:
            y = y / s_y
        a_prev = a.copy()
        t_k = t_new
        it += 1

        if verbose and it % 15 == 0:
            print(f"    LP-N it{it}: obj={best_obj:.8f}, M={M_lp:.6f}, "
                  f"a={best_alpha:.3f}, t_k={t_k:.1f}, el={time.monotonic()-start:.1f}s")

    return best_a


def subgradient_opt(a0, time_budget=60.0, K=7, verbose=False):
    """Weighted subgradient with adaptive temperature."""
    n = len(a0)
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 2.0 / np.sqrt(n)
    min_lr = 1e-7
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()

        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        spread = (peak_vals[0] - peak_vals[-1]) / (max_val + 1e-15)
        temp = 3.0 / (1.0 + spread * 10.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / (max_val + 1e-15))
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                g += w * grad_conv_at_peak(a, int(k))

        a_new = a - lr * g
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            lr = max(min_lr, lr * 0.998)
        else:
            lr = max(min_lr, lr * 0.95)

        a = a_new
        it += 1
        if verbose and it % 300 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}, lr={lr:.2e}, el={time.monotonic()-start:.1f}s")

    return best_a


def mirror_descent_opt(a0, time_budget=60.0, verbose=False):
    """Mirror descent with multiplicative updates and momentum."""
    n = len(a0)
    a = np.maximum(a0, 1e-12).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.5 / np.sqrt(n)
    v = np.zeros(n)
    momentum = 0.5
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()
        T = 5.0 * max_val
        w = np.exp((conv - max_val) / T)
        w /= w.sum()

        threshold = np.percentile(w, 92)
        mask = w > threshold
        if mask.sum() == 0:
            mask[int(np.argmax(conv))] = True
        w_m = w[mask]
        w_m /= w_m.sum()
        idx_m = np.where(mask)[0]

        g = np.zeros(n)
        for wi, ki in zip(w_m, idx_m):
            if wi > 1e-10:
                g += wi * grad_conv_at_peak(a, int(ki))

        v = momentum * v + (1 - momentum) * g
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            momentum = min(0.8, momentum + 0.001)
        else:
            lr = max(1e-6, lr * 0.96)

        a = a_new
        it += 1

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def structured_init(n, mode, rng):
    """Generate structured initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        # Chirp
        freq = rng.uniform(5, 50)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x)
    elif mode == 2:
        # Quadratic residue pattern (approximate)
        p = int(np.sqrt(n)) * 2 + 1
        while p * p < n:
            p += 2
        a = np.zeros(n)
        for i in range(n):
            if (i * i) % p == 1 or (i * i) % p == p - 1:
                a[i] = 1.0
        a += 0.1
    elif mode == 3:
        # Two-peak
        c1, c2 = n // 3, 2 * n // 3
        w = n // 8
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2)) + 0.05
    elif mode == 4:
        # Arctan-shaped
        a = 1.0 / (1.0 + (np.arange(n) - n / 2) ** 2 / (n ** 2 / 20))
    elif mode == 5:
        # Random with structure
        a = rng.uniform(0.5, 1.5, n)
        a = a * np.exp(2 * np.pi * 1j * rng.uniform(0.5, 3) * x * n).real
        a = np.abs(a) + 0.1
    else:
        a = rng.uniform(0.3, 1.0, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a

        a_pert = best_seq * np.exp(rng.normal(0, 0.04, n_best))

        a_pert = a_pert / a_pert.sum()
        lp_budget2 = min(remaining * 0.25, 300.0)
        a_lp2 = lp_nesterov(a_pert, time_budget=lp_budget2, verbose=True, seed=222)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2
            n_best = len(best_seq)
            print(f"  LP-N r2: {best_score:.8f} **")
        else:
            print(f"  LP-N r2: {obj_lp2:.8f}")

    # === Phase 4: Multi-n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: Multi-n exploration ===")
        n_options = []
        for delta in [-200, -100, -50, 50, 100, 200, 300, 400, -300]:
            n_try = n_best + delta
            if 300 <= n_try <= 2000:
                n_options.append(n_try)
        for n_round in [400, 500, 600, 700, 900, 1000, 1100, 1200, 1400, 1500]:
            if n_round != n_best and 300 <= n_round <= 2000:
                n_options.append(n_round)
        n_options = sorted(set(n_options), key=lambda x: abs(x - n_best))

        tried = 0
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 30 or tried >= 5:
                break
            tried += 1

            a_try = resample_seq(best_seq, n_try)
            alloc = min(remaining * 0.12, 50.0)
            a_try = lp_nesterov(a_try, time_budget=alloc, verbose=False,
                                seed=300 + tried)
            obj_try = compute_obj(a_try)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try
                n_best = n_try
                print(f"  ** n={n_try}: {best_score:.8f} **")
            else:
                print(f"  n={n_try}: {obj_try:.8f}")

    # === Phase 5: Structured restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 5: Structured restarts ===")
        n_cur = len(best_seq)
        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            mode = r % 6
            a_r = structured_init(n_cur, mode, rng)
            blend = 0.3 + 0.1 * r
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            alloc = min(remaining * 0.15, 60.0)
            a_opt = lp_nesterov(a_r, time_budget=alloc, verbose=False, seed=500 + r)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt
                n_best = len(best_seq)
                print(f"  Restart {r} (mode={mode}): {best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): {obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 6: Final polish ===")

        if remaining > 40:
            a_f1 = lp_nesterov(best_seq, time_budget=min(remaining * 0.4, 120.0),
                               verbose=True, seed=999)
            obj_f1 = compute_obj(a_f1)
            if obj_f1 < best_score:
                best_score = obj_f1
                best_seq = a_f1
                print(f"  Final LP: {best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_f2 = subgradient_opt(best_seq, time_budget=min(remaining * 0.3, 30.0), K=10)
            obj_f2 = compute_obj(a_f2)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2
                print(f"  Final SG: {best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_f3 = mirror_descent_opt(best_seq, time_budget=min(remaining * 0.3, 25.0))
            obj_f3 = compute_obj(a_f3)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3
                print(f"  Final MD: {best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_f4 = lp_nesterov(best_seq, time_budget=min(remaining * 0.5, 30.0),
                               verbose=False, seed=1234)
            obj_f4 = compute_obj(a_f4)
            if obj_f4 < best_score:
                best_score = obj_f4
                best_seq = a_f4
                print(f"  Final LP2: {best_score:.8f} **")

    # Validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
