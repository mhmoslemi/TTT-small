def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

"""
Coordinate descent + LP alternating optimizer.
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using:
1. Cyclic coordinate descent with analytic 1D minimax
2. LP alternating for global improvements
3. n-sweeping and random restarts
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    result = np.maximum(v - theta, 0)
    s = float(np.sum(result))
    if s > 1e-15:
        result /= s
    else:
        result = np.ones(n) / n
    return result


def build_conv_matrix_rows(a):
    """
    For LP: given a (normalized), build the matrix structure for conv(a, g).
    conv(a, g)[k] = sum_j a[j] * g[k-j] (over valid j, k-j)
    For fixed a, this is linear in g.
    We need: for each k, the vector of coefficients of g[0], ..., g[n-1].
    coeff[k, m] = a[k-m] if 0 <= k-m < n, else 0
    """
    n = len(a)
    m = 2 * n - 1
    # For g[m] to appear in conv[k], we need k-m to be a valid index of a: 0 <= k-m < n
    # So m ranges from max(0, k-n+1) to min(n-1, k)
    # coeff[k, m] = a[k-m]
    return n, m


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M for all k, sum(g) = 1, g >= 0.
    Returns optimal g (normalized) or None.
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    m = 2 * n - 1
    
    # Build A matrix: A[k, j] = a_n[k-j] if 0 <= k-j < n, else 0
    # For efficiency, build it as a banded matrix or just compute on the fly
    # For cvxpy, we need the actual matrix
    
    A = np.zeros((m, n))
    for j in range(n):
        # A[k, j] = a_n[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        k_start = j
        k_end = min(m - 1, j + n - 1)
        A[k_start:k_end+1, j] = a_n[:k_end+1-j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs
        except Exception:
            continue
    return None

    conv is the current convolution (will be updated in place).
    Returns updated b and conv.

    m = 2 * n - 1
    if start_time is None:
        start_time = time.monotonic()

            break

                break
            
            x_old = b[i]
            if x_old < 1e-15:
                continue

            k_min = i

            j_range = np.arange(k_min, k_max + 1) - i  # 0, 1, ..., k_max-i
            
            # base = conv - contribution
            base = conv.copy()
            base[k_min:k_max+1] -= 2.0 * x_old * b[j_range]
            # Fix k=2i
            if 2 * i < m:
                base[2*i] += x_old * x_old  # Add back one b[i]^2 (we subtracted 2*b[i]^2, should be 1*b[i]^2)

            has_quad = (2 * i < m)
            quad_base = base[2*i] if has_quad else 0.0
            quad_idx = (2 * i - k_min) if has_quad else -1
            
            # Max of base outside the affected range
            if k_min > 0:
                base_max_before = float(np.max(base[:k_min]))
            else:

            else:
                base_max_after = 0.0
            base_max_outside = max(base_max_before, base_max_after)

            def g_func(x):
                if x < 0:
                    x = 0.0
                vals = base[k_min:k_max+1].copy() + coeff * x
                if has_quad:
                    vals[quad_idx] = quad_base + x * x
                local_max = float(np.max(vals))
                return max(local_max, base_max_outside)
            
            # Find minimum of g on [0, 10*x_old + 0.01]
            x_upper = min(10.0 * x_old + 0.01, 1.0)
            # Also try x = 0

                    res = minimize_scalar(g_func, bounds=(0.0, x_upper), method='bounded',
                                         options={'maxiter': 40, 'xatol': 1e-10})
                    x_new = max(0.0, res.x)
                except Exception:
                    x_new = x_old
            
            # Also try x=0 explicitly
            g_zero = g_func(0.0)
            g_new = g_func(x_new)
            if g_zero < g_new:
                x_new = 0.0
                g_new = g_zero
            
            # Update
            if abs(x_new - x_old) > 1e-15:
                delta = x_new - x_old
                # Update conv
                for k in range(k_min, k_max + 1):
                    j = k - i
                    if j == i:
                        conv[k] += x_new * x_new - x_old * x_old
                    else:
                        conv[k] += 2.0 * delta * b[j]
                b[i] = x_new
                improved = True
        
        if not improved:
            break
        if verbose and cycle % 20 == 0:
            obj = 2.0 * n * float(np.max(conv))
            print(f"      CD cycle {cycle}: obj = {obj:.8f}")
    
    return b, conv


def lp_alternate(b, time_budget=30.0, verbose=False):
    """LP alternating with line search."""
    start = time.monotonic()
    n = len(b)
    prev_obj = 2.0 * n * float(np.max(compute_conv(b)))
    stall = 0
    
    while time.monotonic() - start < time_budget:
        g_dir = solve_lp_direction(b)
        if g_dir is None:
            break
        
        # Line search: minimize obj((1-t)*b + t*g_dir) over t in [0,1]
        def line_obj(t):
            a_try = (1.0 - t) * b + t * g_dir
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                return np.inf
            a_try = a_try / s
            conv = compute_conv(a_try)
            return 2.0 * n * float(np.max(conv))
        
        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 30, 'xatol': 0.01})
            best_t = res.x if res.success else 0.0
        except Exception:
            best_t = 0.0
        
        a_new = (1.0 - best_t) * b + best_t * g_dir
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s
        conv_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(conv_new))
        
        if obj_new < prev_obj - 1e-10:
            b = a_new
            prev_obj = obj_new
            stall = 0
            if verbose:
                print(f"      LP: obj={prev_obj:.8f}, t={best_t:.3f}, t_elapsed={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 10:
                break
    
    return b


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    result = np.interp(x_new, x_old, a)
    return np.maximum(result, 0)
