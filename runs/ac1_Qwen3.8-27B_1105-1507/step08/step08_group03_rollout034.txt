            best_f = f_try
            best_t = t

    # Refine with golden section around best_t
    if best_f < best_obj:
        lo = max(0.01, best_t - 0.15)
        hi = min(1.0, best_t + 0.15)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a1 = hi - gr * (hi - lo)
        b1 = lo + gr * (hi - lo)
        f_a1 = obj_val(np.maximum((1 - a1) * a + a1 * g, 0))
        f_b1 = obj_val(np.maximum((1 - b1) * a + b1 * g, 0))
        for _ in range(20):
            if f_a1 < f_b1:
                hi = b1
                b1 = a1
                f_b1 = f_a1
                a1 = hi - gr * (hi - lo)
                f_a1 = obj_val(np.maximum((1 - a1) * a + a1 * g, 0))
            else:
                lo = a1
                a1 = b1
                f_a1 = f_b1
                b1 = lo + gr * (hi - lo)
                f_b1 = obj_val(np.maximum((1 - b1) * a + b1 * g, 0))
        t_opt = (lo + hi) / 2.0
        if f_a1 < best_f:
            best_t = t_opt
            best_f = f_a1
        elif f_b1 < best_f:
            best_t = t_opt
            best_f = f_b1

    a_opt = (1.0 - best_t) * a + best_t * g
    a_opt = np.maximum(a_opt, 0)
    s_opt = float(np.sum(a_opt))
    if s_opt > 1e-10:
        a_opt = a_opt / s_opt
    return a_opt, best_f, best_t


def lp_alternating(a, time_budget=300.0, verbose=False):
    """
    LP alternating optimization with adaptive line search and dual-guided escape.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()
    stall = 0
    best_t = 0.3
    rng = np.random.default_rng(12345)
    it = 0
    patience = 20

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        # Solve LP for direction
        g_lp, M_lp, active_set = solve_lp_minpeak(a)
        if g_lp is None:
            conv_a = full_conv(a)
            M_cur = float(np.max(conv_a))
            g_lp = solve_lp_maxsum(a, M_cur)
            if g_lp is None:
                break
            M_lp = M_cur
            active_set = np.where(conv_a > 0.9 * M_cur)[0]

        # Line search
        a_try, obj_try, t_try = line_search(a, g_lp, best_obj)

        it += 1
        if obj_try < best_obj - 1e-10:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            best_t = t_try
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t_try:.3f}, M_lp={M_lp:.6f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > patience:
                # Dual-guided escape: perturb using active constraint info
                if active_set is not None and len(active_set) > 0:
                    # Compute a pseudo-gradient from active constraints
                    g_escape = np.zeros(n)
                    for k in active_set[:20]:
                        g_escape += grad_conv_at(a, int(k))
                    # Move in the direction that reduces active peaks
                    a_pert = a - 0.05 * g_escape / max(1.0, np.max(np.abs(g_escape)))
                    a_pert = np.maximum(a_pert, 0)
                    sp = float(np.sum(a_pert))
                    if sp > 1e-10:
                        a_pert = a_pert / sp
                    if obj_val(a_pert) < best_obj:
                        a = a_pert
                        stall = 0
                        patience = int(patience * 1.2)
                        continue

                # Random structured escape
                scale = min(0.25, 0.04 + 0.03 * (stall // patience))
                a_pert = a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                sp = float(np.sum(a_pert))
                if sp > 1e-10:
                    a_pert = a_pert / sp
                a = a_pert
                best_a = a.copy()
                best_obj = obj_val(a)
                stall = 0
                best_t = 0.3
                patience = int(patience * 1.3)

    return best_a


def smooth_lse_objective_and_grad(x, T, n):
    """
    Compute log-sum-exp smoothed objective and gradient.
    x is on the simplex (sum=1, x>=0). T is temperature.
    Returns (obj, grad).
    """
    x = np.maximum(x, 1e-15)
    s = x.sum()
    x = x / s
    conv = full_conv(x)
    max_c = conv.max()
    # Shift for numerical stability
    z = T * (conv - max_c)
    e = np.exp(z)
    sum_e = e.sum()
    obj = 2.0 * n * (max_c + np.log(sum_e) / T)
    w = e / sum_e  # softmax weights

    # Gradient: 2n/T * sum_k w[k] * grad_c[k]
    # Use efficient computation
    grad = np.zeros(n)
    # For each i, grad[i] = 2n/T * sum_k w[k] * (a[k-i] if valid and k!=2i, else 2a[i])
    # Simpler: just sum over k
    for k in range(len(conv)):
        if w[k] > 1e-10:
            grad += w[k] * grad_conv_at(x, k)
    grad *= 2.0 * n / T
    return obj, grad


def smooth_optimize(a, time_budget=120.0, T_start=100.0, T_end=500.0, verbose=False):
    """
    Smooth optimization using log-sum-exp with annealed temperature and L-BFGS-B.
    Uses reparameterization x = exp(y) / sum(exp(y)) to handle non-negativity.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()

    # Anneal temperature from T_start to T_end
    T = T_start
    n_anneal = 4
    step = 0

    for phase in range(n_anneal):
        T = T_start * (T_end / T_start) ** (phase / max(1, n_anneal - 1))
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break
        phase_budget = min(remaining * 0.4, 60.0)

        def obj_grad_log(y):
            """Objective and gradient in log-space parameterization."""
            y_shifted = y - y.max()
            x = np.exp(y_shifted)
            s = x.sum()
            if s < 1e-15:
                return 1e10, np.zeros(len(y))
            x = x / s
            obj, grad_x = smooth_lse_objective_and_grad(x, T, n)
            # Transform gradient: d/dy log-sum-exp param
            # x_i = exp(y_i) / S, dx_i/dy_j = x_i * delta_ij - x_i * x_j
            # grad_y_j = sum_i grad_x_i * (delta_ij - x_j) * x_i = grad_x_j * x_j - x_j * sum_i grad_x_i * x_i
            inner = np.dot(grad_x, x)
            grad_y = grad_x * x - x * inner
            return obj, grad_y

        y0 = np.log(np.maximum(a, 1e-15))
        try:
            res = minimize(obj_grad_log, y0, method='L-BFGS-B',
                           options={'maxiter': 300, 'maxfun': 600, 'disp': False,
                                    'gtol': 1e-8})
            y_opt = res.x
            x_opt = np.exp(y_opt - y_opt.max())
            x_opt = x_opt / x_opt.sum()
            obj_opt = obj_val(x_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = x_opt.copy()
                a = x_opt.copy()
                if verbose:
                    print(f"    Smooth phase {phase} (T={T:.1f}): obj={best_obj:.8f}")
        except Exception:
            pass
        step += 1

    return best_a


def nesterov_subgrad(a, time_budget=60.0, K=7, verbose=False):
    """Nesterov-accelerated subgradient on the simplex with top-K peak weighting."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 1.5 / np.sqrt(n)
    stall = 0
    patience = 100
    it = 0

    while time.monotonic() - start < time_budget:
        conv = full_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        # Adaptive temperature weighting
        spread = (peak_vals[0] - peak_vals[-1]) / (max_val + 1e-15)
        temp = 2.0 / (1.0 + spread * 8.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / (max_val + 1e-15))
        weights /= weights.sum()

        # Weighted subgradient
        g = np.zeros(n)
        for w_k, k in zip(weights, top_idx):
            if w_k > 1e-12:
                g += w_k * grad_conv_at(y, int(k))

        lr = lr0 / np.sqrt(t_k)
        a_new = y - lr * g
        a_new = proj_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = obj_val(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Restart with perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1000) + n)
                a_new = best_a * np.exp(rng.normal(0, 0.1, n))
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                t_k = 1.0
                y = a_new.copy()
                a_prev = a_new.copy()
                y_new = a_new.copy()
                stall = 0
                patience = int(patience * 1.2)

        a_prev = a_new.copy()
        y = y_new.copy()
        t_k = t_new
        it += 1

        if verbose and it % 200 == 0:
            print(f"    NS it{it}: obj={best_obj:.8f}, t_k={t_k:.1f}, el={time.monotonic()-start:.1f}s")

    return best_a


def coordinate_refine(a, time_budget=30.0):
    """
    Coordinate descent: try small transfers between coordinates to reduce max convolution.
    Uses the smoothed objective for gradient guidance.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()
    T = 100.0

    it = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        conv = full_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Compute smoothed gradient for direction finding
        _, grad = smooth_lse_objective_and_grad(a, T, n)

        # Try moving mass from high-gradient positions to low-gradient positions
        # (in the direction that reduces the peak)
        sorted_idx = np.argsort(grad)
        sources = sorted_idx[-10:]  # highest gradient (most "costly")
        targets = sorted_idx[:10]   # lowest gradient (least costly)

        improved = False
        for src in sources:
            if a[src] < 1e-6:
                continue
            for tgt in targets:
                if tgt == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    delta = frac * a[src]
                    if delta < 1e-8:
                        continue
                    a[src] -= delta
                    a[tgt] += delta
                    a[tgt] = max(a[tgt], 0)
                    obj_try = obj_val(a)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[tgt] -= delta
                        break

        if not improved:
            # Try random pairs
            rng = np.random.default_rng(it + n)
            for _ in range(50):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-6:
                    continue
                delta = 0.01 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                obj_try = obj_val(a)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] += delta
                    a[j] -= delta
            if not improved:
                break
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation with non-negativity."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations for restarts."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Quadratic (U-shaped)
        x = np.linspace(0, 1, n)
        a = x**2 + (1 - x)**2 + 0.05
        a = a / a.sum()
    elif mode == 3:
        # Inverse cosine (inverted U)
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * (2 * x - 1)))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Exponential ramp
        a = np.exp(np.linspace(0, 2, n))
        a = a / a.sum()
    elif mode == 5:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Gaussian
        x = np.arange(n)
        a = np.exp(-((x - n // 2) ** 2) / (2 * (n // 4) ** 2))
        a = a / a.sum()
    else:
        x = np.linspace(0, 1, n)
        a = 1.0 / (0.1 + np.abs(x - 0.5))
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating, smooth optimization,
    Nesterov subgradient, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = obj_val(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = obj_val(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Nesterov subgradient (quick warm-up) ===
    print(f"\n=== Phase 1: Nesterov Subgradient ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 90.0)
    if p1_budget > 15:
        a_ns = nesterov_subgrad(best_seq, time_budget=p1_budget, K=7, verbose=True)
        obj_ns = obj_val(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_ns:.8f} (no improvement)")

    # === Phase 2: LP Alternating (main workhorse, ~500s) ===
    print(f"\n=== Phase 2: LP Alternating (Round 1) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.30, 450.0)
    if p2_budget > 30:
        a_lp = lp_alternating(best_seq, time_budget=p2_budget, verbose=True)
        obj_lp = obj_val(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP Round 1: obj={best_score:.8f} **")
        else:
            print(f"  LP Round 1: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: LP Alternating with perturbation (escape) ===
    print(f"\n=== Phase 3: LP Alternating (Perturbed) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)
    if p3_budget > 20:
        # Multiple perturbation levels
        for pert_scale in [0.05, 0.1, 0.02]:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            a_pert = best_seq * np.exp(rng.normal(0, pert_scale, len(best_seq)))
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 1e-10:
                a_pert = a_pert / s_p
            alloc = min(p3_budget / 3.0 + 30, remaining * 0.5)
            if alloc < 20:
                break
            a_lp = lp_alternating(a_pert, time_budget=alloc, verbose=False)
            obj_lp = obj_val(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP pert={pert_scale}: obj={best_score:.8f} **")
            else:
                print(f"  LP pert={pert_scale}: obj={obj_lp:.8f}")

    # === Phase 4: Smooth optimization (log-sum-exp + L-BFGS-B) ===
    print(f"\n=== Phase 4: Smooth Optimization ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 180.0)
    if p4_budget > 30:
        a_sm = smooth_optimize(best_seq, time_budget=p4_budget,
                               T_start=50.0, T_end=500.0, verbose=True)
        obj_sm = obj_val(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Smooth: obj={best_score:.8f} **")
        else:
            print(f"  Smooth: obj={obj_sm:.8f} (no improvement)")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Try nearby n values
    n_options = sorted(set([
        max(200, n_current - 100),
        max(200, n_current - 50),
        max(200, n_current - 20),
        n_current + 20,
        n_current + 50,
        n_current + 100,
        n_current + 200,
        500, 600, 700, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and n != n_current]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50 or n_tried >= 5:
            break
        n_tried += 1

        a_try = resample_sequence(best_seq, n_try)

        # Quick optimization: Nesterov + short LP
        alloc_ns = min(remaining * 0.06, 30.0)
        if alloc_ns > 10:
            a_try = nesterov_subgrad(a_try, time_budget=alloc_ns, K=5)

        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.06, 40.0)
        if alloc_lp > 15:
            a_try = lp_alternating(a_try, time_budget=alloc_lp)

        obj_try = obj_val(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = r % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with best to stay in good region
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        restart_budget = min(remaining * 0.1, 50.0)
        if restart_budget < 15:
            break
        a_opt = lp_alternating(a_r, time_budget=restart_budget)
        obj_r = obj_val(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 40:
        lp_alloc = min(remaining * 0.25, 100.0)
        a_lp = lp_alternating(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = obj_val(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Nesterov polish with more peaks
    remaining = deadline - time.monotonic()
    if remaining > 25:
        ns_alloc = min(remaining * 0.15, 40.0)
        a_ns = nesterov_subgrad(best_seq, time_budget=ns_alloc, K=15)
        obj_ns = obj_val(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Final NS: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        cd_alloc = min(remaining * 0.15, 30.0)
        a_cd = coordinate_refine(best_seq, time_budget=cd_alloc)
        obj_cd = obj_val(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more LP round
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.5, 20.0))
        obj_last = obj_val(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = obj_val(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
