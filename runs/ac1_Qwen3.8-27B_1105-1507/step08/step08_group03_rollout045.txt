"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses accelerated primal-dual hybrid gradient with LP-based direction finding,
adaptive log-sum-exp smoothing, dual-informed subgradients, symmetry enforcement,

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M, lambda_dual) or (None, None, None).
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            # Extract dual variables for inequality constraints
            lam = np.zeros(2 * n - 1)
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                # Duals are in res.ineqlin.marginals for HiGHS
                duals = -res.ineqlin.marginals  # negate because of sign convention
                duals = np.maximum(duals, 0)
                if duals.sum() > 0:
                    lam = duals / duals.sum()
            return g_val / s, float(res.x[n]), lam
    return None, None, None

def solve_lp_maxsum(a, M_target):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M_target, g >= 0."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x, 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None

        conv_a = compute_conv(a)

        lam_dual = np.zeros(len(conv_a))
        lam_dual[top_idx] = 1.0 / K
    
    lam = lam_dual.copy()
    
    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 3.0 / np.sqrt(n)
    stall = 0
    patience = 60
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv_y = compute_conv(y)
        max_val = float(np.max(conv_y))
        if max_val < 1e-15:
            break
        
        # Update dual weights (exponential tilt)
        # Use the LP dual as a "prior" and blend with current peaks
        temp = 5.0 / (1.0 + it * 0.01)  # decreasing temperature
        soft_w = np.exp((conv_y - max_val) / (temp * max_val + 1e-15))
        soft_w /= soft_w.sum()
        
        # Blend LP dual with soft peaks
        blend = 0.5  # 50% LP dual, 50% soft
        lam_new = blend * lam + (1 - blend) * soft_w
        lam_new /= lam_new.sum()
        lam = lam_new
        
        # Compute subgradient using current dual
        g = np.zeros(n)
        # Use top contributing peaks for efficiency
        active_idx = np.argsort(lam)[::-1]
        active_count = 0
        active_sum = 0.0
        for k in active_idx:
            if lam[k] < 1e-10:
                break
            g += lam[k] * grad_conv_at_peak(y, int(k))
            active_count += 1
            active_sum += lam[k]
            if active_count > 200:  # cap for efficiency
                break
        
        # Nesterov step
        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                t_k = 1.0
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.4)
                # Re-solve LP for fresh dual
                g_lp2, M_lp2, lam2 = solve_lp_minpeak(best_a)
                if lam2 is not None and lam2.sum() > 1e-10:
                    lam = lam2
        
        a_prev = a_new.copy()
        y = y_new.copy()
        t_k = t_new
        it += 1
        
        if verbose and it % 200 == 0:
            print(f"    DSG it{it}: obj={best_obj:.8f}, t_k={t_k:.1f}, el={time.monotonic()-start:.1f}s")
    
    return best_a

def smooth_optimize_adaptive(a, time_budget=80.0, verbose=False):
    """
    Smooth optimization with adaptive temperature schedule.
    Uses log-sum-exp with geometrically decreasing temperature.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adaptive temperature schedule
    T_values = [50.0, 30.0, 20.0, 15.0, 10.0, 8.0, 5.0]
    
    for T in T_values:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break
        
        def obj_grad_logsumexp(x, T_val=T):
            x = np.maximum(x, 1e-15)
            s = x.sum()
            if s < 1e-10:
                return 1e10, np.zeros(len(x))
            x = x / s
            conv = compute_conv(x)
            max_c = conv.max()
            z = T_val * (conv - max_c)
            # Numerical stability
            z = z - z.max()
            e = np.exp(z)
            sum_e = e.sum()
            if sum_e < 1e-300:
                sum_e = 1e-300
            obj_val = 2.0 * n * (max_c + np.log(sum_e) / T_val)
            
            w = e / sum_e
            g = np.zeros(n)
            # Only use significant weights
            threshold = np.sum(w) * 1e-6
            for k in range(len(conv)):
                if w[k] > threshold:
                    g += w[k] * grad_conv_at_peak(x, int(k))
            g *= 2.0 * n
            return obj_val, g
        
        # Use scipy minimize
        from scipy.optimize import minimize as sp_minimize
        try:
            res = sp_minimize(obj_grad_logsumexp, a, method='L-BFGS-B',
                           bounds=[(1e-12, None)] * n,
                           options={'maxiter': 150, 'maxfun': 300, 'disp': False})
            a_opt = np.maximum(res.x, 0)
            s_opt = float(np.sum(a_opt))
            if s_opt > 1e-10:
                a_opt = a_opt / s_opt
                obj_opt = compute_obj(a_opt)
                if obj_opt < best_obj:
                    best_obj = obj_opt
                    best_a = a_opt.copy()
                    a = a_opt.copy()
                    if verbose:
                        print(f"    Smooth T={T}: obj={best_obj:.8f}")
        except Exception:
            pass
    
    return best_a

def chamboelle_pock(a, time_budget=120.0, verbose=False):

def enforce_symmetry(a):
    """Make sequence symmetric: a[i] = a[n-1-i]."""
    n = len(a)
    half = (n + 1) // 2
    sym = np.zeros(n)
    for i in range(half):
        sym[i] = (a[i] + a[n - 1 - i]) / 2.0
        sym[n - 1 - i] = sym[i]
    s = sym.sum()
    if s > 1e-10:
        sym /= s
    return sym

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian peaks
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Triangle
        x = np.linspace(-1, 1, n)
        a = np.maximum(1.0 - np.abs(x), 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Arcsine
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.01))
        a = a / a.sum()
    elif mode == 5:
        # Multi-frequency
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in range(1, 6):
            a += 0.2 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # W-shaped
        x = np.linspace(0, 1, n)
        a = (x - 0.25)**2 + (x - 0.75)**2 + 0.1
        a = a / a.sum()
    elif mode == 7:
        # Random smooth
        a = rng.uniform(0.5, 1.5, n)
        # Smooth with moving average
        k = max(1, n // 20)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a
