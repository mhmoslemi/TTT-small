def fft_convolve(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = fft_convolve(a_norm)
    return 2.0 * n * float(np.max(conv))


def grad_conv_at_lag(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout_s=10.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        return g_val / s_g, float(M.value)
            except Exception:
                continue

    # Fallback to scipy
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

                  bounds=bounds, method='highs')

        s_g = float(np.sum(g_val))
        if s_g > 1e-10:
            return g_val / s_g, float(res.x[n])
    return None, None

    # Create a perturbation direction in time domain
    # For each high-power bin m, the direction that reduces |A[m]| is:
    # -Re(exp(i*angle(A[m])) * exp(2pi*i*m*k/n)) for real part
    # We approximate using the top few bins
    
    top_bins = np.argsort(P)[::-1][:min(5, size // 2)]
    direction = np.zeros(n)
    for m in top_bins:
        if m >= size // 2:
            continue
        weight = (P[m] - mean_P) / (P.max() + 1e-15)
        # Direction to reduce power at bin m
        # For real signal, bin m and bin (size-m) are conjugates
        phase = np.angle(A[m])
        # The direction is -cos(2*pi*m*k/n + phase) for the real DFT
        k = np.arange(n)
        direction -= weight * np.cos(2 * np.pi * m * k / size + phase)
    
    # Project direction onto tangent space of {sum=1, a>=0}
    # Remove mean component
    direction -= direction.mean()
    # For non-negativity, only apply where a > 0
    mask = a > 1e-6
    direction[~mask] = 0
    
    # Apply perturbation
    a_pert = a + strength * direction
    a_pert = np.maximum(a_pert, 0)
    s = float(np.sum(a_pert))
    if s > 1e-10:
        a_pert /= s
    else:
        a_pert = a.copy()
    
    return a_pert


def structured_init(n, rng, mode):
    """Generate structured initializations based on signal processing theory."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Uniform
        a = np.ones(n) / n
    elif mode == 1:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 0.01)
        a /= a.sum()
    elif mode == 2:
        # Squared raised cosine (more concentrated)
        a = (0.5 * (1 + np.cos(2 * np.pi * x))) ** 2
        a = np.maximum(a, 0.01)
        a /= a.sum()
    elif mode == 3:
        # Two-hump (reduces middle auto-corr)
        a = 0.5 + 0.4 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        a /= a.sum()
    elif mode == 4:
        # Three-hump
        a = 0.5 + 0.3 * np.cos(6 * np.pi * x)
        a = np.maximum(a, 0.01)
        a /= a.sum()
    elif mode == 5:
        # Inverse sinc-squared (concentrated in middle)
        t = (x - 0.5) * n * 0.3
        a = np.sinc(t) ** 2
        a = np.maximum(a, 0.01)
        a /= a.sum()
    elif mode == 6:
        # Chirp-squared (quadratic phase)
        phase = 2 * np.pi * (0.5 * x ** 2)
        a = (1 + 0.5 * np.cos(phase)) ** 2
        a = np.maximum(a, 0.01)
        a /= a.sum()
    elif mode == 7:
        # Random structured
        freq = rng.integers(2, 8)
        amp = rng.uniform(0.2, 0.5)
        a = 1 + amp * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
        a /= a.sum()
    elif mode == 8:
        # Power-law decay from center
        center = n // 2
        dist = np.abs(np.arange(n) - center) / (n / 2)
        a = (1 + dist) ** (-rng.uniform(0.5, 2.0))
        a /= a.sum()
    elif mode == 9:
        # Bimodal
        c1, c2 = 0.25, 0.75
        a = np.exp(-((x - c1) ** 2) / 0.02) + np.exp(-((x - c2) ** 2) / 0.02)
        a /= a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a /= a.sum()
    return a


def nesterov_subgrad(a, time_budget=60.0, K_init=3, K_max=15):
    """
    Nesterov-accelerated subgradient with adaptive K.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 60
    K = K_init

    while time.monotonic() - start < time_budget:
        conv = fft_convolve(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        # Adaptive temperature
        spread = (peak_vals[0] - peak_vals[-1]) / max_val if K > 1 else 1.0
        temp = 5.0 / (1.0 + spread * 5.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / max_val)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                g += w * grad_conv_at_lag(y, int(k))

        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            # Increase K slowly
            if stall == 0 and K < K_max:
                K = min(K_max, K + 1)
        else:
            stall += 1
            if stall > patience:
                # Reset with spectral perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                a = spectral_perturbation(best_a, rng, strength=0.05)
                best_a = a.copy()
                best_obj = compute_obj(a)
                t_k = 1.0
                y = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.2)
                K = K_init

        a_prev = a_new.copy()
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        y = y_new.copy()
        t_k = t_new

    return best_a


def lp_alternating_improved(a, time_budget=200.0, verbose=False):
    """
    LP alternating with dual-based line search and spectral escape.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    best_t = 0.3
    rng = np.random.default_rng(12345)
    it = 0
    dual_M = None  # Track the LP optimal M

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        # Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        if dual_M is not None and M_lp < dual_M * (1 - 1e-6):
            # M decreased, good sign
            pass
        dual_M = M_lp

        # Adaptive line search
        # Use the ratio M_lp / current_max to guide step size
        conv_a = fft_convolve(a)
        max_conv = float(np.max(conv_a))
        
        if max_conv > 1e-15:
            suggested_t = min(1.0, max(0.05, (max_conv - M_lp) / max_conv * 2.0))
        else:
            suggested_t = 0.3
        
        ts = [best_t, suggested_t, best_t * 0.5, best_t * 2.0, 0.1, 0.3, 0.5, 0.7, 1.0]
        ts = sorted(set([t for t in ts if 0.02 < t <= 1.0]))[:8]

        best_t_obj = np.inf
        best_t_val = best_t
        best_a_local = best_a.copy()

        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            best_t = best_t_val
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, M={M_lp:.6f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 20:
                # Spectral perturbation escape
                strength = min(0.2, 0.03 + 0.02 * (stall // 20))
                a = spectral_perturbation(best_a, rng, strength=strength)
                # Also try random perturbation
                if rng.random() < 0.3:
                    a2 = best_a * np.exp(rng.normal(0, 0.05, n))
                    a2 = np.maximum(a2, 0)
                    a2 /= a2.sum()
                    if compute_obj(a2) < compute_obj(a):
                        a = a2
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                best_t = 0.3
                dual_M = None

    return best_a


def logsumexp_optimize(a, time_budget=60.0, T=30.0, verbose=False):
    """
    L-BFGS optimization of log-sum-exp smoothed objective.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def obj_func(x):
        x = np.maximum(x, 1e-15)
        s = x.sum()
        if s < 1e-10:
            return 1e10
        x = x / s
        conv = fft_convolve(x)
        max_c = conv.max()
        z = T * (conv - max_c)
        e = np.exp(z)
        sum_e = e.sum()
        return 2.0 * n * (max_c + np.log(sum_e) / T)

    def obj_grad_func(x):
        x = np.maximum(x, 1e-15)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x = x / s
        conv = fft_convolve(x)
        max_c = conv.max()
        z = T * (conv - max_c)
        e = np.exp(z)
        sum_e = e.sum()
        obj_val = 2.0 * n * (max_c + np.log(sum_e) / T)
        w = e / sum_e
        g = np.zeros(n)
        for k in range(len(conv)):
            if w[k] > 1e-8:
                g += w[k] * grad_conv_at_lag(x, k)
        g *= 2.0 * n
        return obj_val, g

    max_iter = 300
    try:
        res = sp_minimize(obj_grad_func, a, method='L-BFGS-B',
                          bounds=[(1e-12, None)] * n,
                          options={'maxiter': max_iter, 'maxfun': max_iter * 3, 'disp': False})
        a_opt = np.maximum(res.x, 0)
        s_opt = float(np.sum(a_opt))
        if s_opt > 1e-10:
            a_opt = a_opt / s_opt
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                if verbose:
                    print(f"    LSE T={T}: obj={best_obj:.8f}")
    except Exception:
        pass

    return best_a


def two_scale_optimize(a, time_budget=120.0):
    """
    Two-scale optimization: downsample, optimize, upsample, refine.
    This can escape local minima by finding a better coarse structure.
    """
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    # Downsample to n/4
    n_coarse = max(50, n // 4)
    # Use strided sampling
    stride = n // n_coarse
    a_coarse = a[::stride][:n_coarse]
    s_c = float(np.sum(a_coarse))
    if s_c > 1e-10:
        a_coarse /= s_c

    # Optimize coarse version
    alloc_coarse = time_budget * 0.3
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a_coarse = lp_alternating_improved(a_coarse, time_budget=min(alloc_coarse, remaining))
    
    # Upsample back
    x_coarse = np.linspace(0, 1, n_coarse)
    x_fine = np.linspace(0, 1, n)
    a_up = np.interp(x_fine, x_coarse, a_coarse)
    a_up = np.maximum(a_up, 0)
    s_up = float(np.sum(a_up))
    if s_up > 1e-10:
        a_up /= s_up

    # If upsampled is better, use it as new starting point
    obj_up = compute_obj(a_up)
    if obj_up < best_obj:
        best_obj = obj_up
        best_a = a_up.copy()

    # Refine at full scale
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        a_refined = lp_alternating_improved(best_a, time_budget=remaining)
        obj_ref = compute_obj(a_refined)
        if obj_ref < best_obj:
            best_obj = obj_ref
            best_a = a_refined.copy()

    return best_a


def coordinate_descent_final(a, time_budget=30.0):
    """
    Final coordinate descent polish: try individual coordinate moves.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    while time.monotonic() - start < time_budget:
        conv = fft_convolve(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]
        
        # Find contributing indices
        contrib_pairs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib_pairs.append((a[i] * a[j], i, j))
        
        # Find non-contributing indices
        non_contrib = []
        for i in range(n):
            j = k_star - i
            if j < 0 or j >= n:
                non_contrib.append(i)
        
        if not non_contrib:
            break
        
        # Try moving mass from high contributors to non-contributors
        contrib_pairs.sort(reverse=True)
        improved = False
        
        for val, i, j in contrib_pairs[:20]:
            if val < 1e-10:
                break
            for target in non_contrib[:5]:
                if target == i and target == j:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    src = i if a[i] >= a[j] else j
                    delta = frac * a[src]
                    if delta < 1e-12:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_conv = fft_convolve(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    a[src] += delta
                    a[target] -= delta
                    break
        
        if not improved:
            # Try random coordinate swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = 0.01 * a[i]
                a[i] -= delta
                a[j] += delta
                new_conv = fft_convolve(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] += delta
                    a[j] -= delta
            
            if not improved:
                break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: hybrid multi-phase optimization combining LP alternating,
    Nesterov subgradient, L-BFGS log-sum-exp, two-scale optimization,
    spectral perturbations, and coordinate descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    n_target = 800 if best_seq is None else len(best_seq)

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Best initial structure via diverse starts ===
    print("\n=== Phase 1: Diverse Initialization ===")
    n = len(best_seq)
    remaining = deadline - time.monotonic()
    
    for mode in range(10):
        if remaining < 5:
            break
        a_try = structured_init(n, rng, mode)
        # Quick Nesterov to evaluate
        alloc = min(remaining * 0.03, 20.0)
        if alloc > 5:
            a_opt = nesterov_subgrad(a_try, time_budget=alloc, K_init=3, K_max=8)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Init mode {mode} (n={n}): obj={best_score:.8f} **")
            remaining = deadline - time.monotonic()

    # === Phase 2: Nesterov subgradient ===
    print(f"\n=== Phase 2: Nesterov Subgradient (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.1, 90.0)
    if p2_budget > 15:
        a_ns = nesterov_subgrad(best_seq, time_budget=p2_budget, K_init=3, K_max=15)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_ns:.8f}")

    # === Phase 3: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p3_total = min(remaining * 0.4, 500.0)
    
    n_rounds = 4
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p3_total / (n_rounds - rnd), remaining * 0.5)
        if alloc < 20:
            break

        if rnd > 0:
            # Perturb for later rounds
            if rnd % 2 == 1:
                a_pert = spectral_perturbation(best_seq, rng, strength=0.03 * rnd)
            else:
                a_pert = best_seq * np.exp(rng.normal(0, 0.02 * rnd, len(best_seq)))
                a_pert = np.maximum(a_pert, 0)
                a_pert /= a_pert.sum()
        else:
            a_pert = best_seq.copy()

        a_lp = lp_alternating_improved(a_pert, time_budget=alloc, verbose=(rnd == 0))
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f}")

    # === Phase 4: L-BFGS log-sum-exp ===
    print(f"\n=== Phase 4: L-BFGS Log-Sum-Exp ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 150.0)
    for T_val in [15.0, 40.0, 80.0]:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        t_budget = min(p4_budget / 3.0, remaining * 0.4)
        if t_budget < 15:
            continue
        a_sm = logsumexp_optimize(best_seq, time_budget=t_budget, T=T_val, verbose=True)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  LSE T={T_val}: obj={best_score:.8f} **")

    # === Phase 5: Two-scale optimization ===
    print(f"\n=== Phase 5: Two-Scale Optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_2s = two_scale_optimize(best_seq, time_budget=min(remaining * 0.2, 120.0))
        obj_2s = compute_obj(a_2s)
        if obj_2s < best_score:
            best_score = obj_2s
            best_seq = a_2s.copy()
            print(f"  Two-scale: obj={best_score:.8f} **")
        else:
            print(f"  Two-scale: obj={obj_2s:.8f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(200, n_current - 100),
        max(200, n_current - 50),
        n_current + 50,
        n_current + 100,
        n_current + 200,
        400, 500, 600, 700, 800, 900, 1000, 1200, 1500
    ]))
    n_options = [n_opt for n_opt in n_options if 200 <= n_opt <= 2000]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40 or n_tried >= 5:
            break
        if n_try == n_current:
            continue

        # Resample
        x_old = np.linspace(0, 1, n_current)
        x_new = np.linspace(0, 1, n_try)
        a_try = np.interp(x_new, x_old, best_seq)
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try > 1e-10:
            a_try /= s_try
        n_tried += 1

        # Quick optimize
        alloc_ns = min(remaining * 0.08, 30.0)
        if alloc_ns > 10:
            a_try = nesterov_subgrad(a_try, time_budget=alloc_ns, K_init=3)

        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.08, 40.0)
        if alloc_lp > 15:
            a_try = lp_alternating_improved(a_try, time_budget=alloc_lp)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")

"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternating with dual-based line search, frequency-domain
perturbations for escape, multi-temperature log-sum-exp L-BFGS,
adaptive Nesterov subgradient with increasing K, two-scale (coarse-to-fine)
optimization, structured signal-processing initializations, and coordinate
descent polish. Key new ideas: spectral perturbation escaping, dual-variable
guided step sizing, and diverse structured starts from window/chirp theory.
"""

import numpy as np
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize as sp_minimize

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except Exception:
    HAVE_CVXPY = False


def fft_convolve(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = fft_convolve(a_norm)
    return 2.0 * n * float(np.max(conv))


def grad_conv_at_lag(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        return g_val / s_g, float(M.value)
            except Exception:
                continue

    # Fallback to scipy
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s_g = float(np.sum(g_val))
        if s_g > 1e-10:
            return g_val / s_g, float(res.x[n])
    return None, None


def spectral_perturbation(a, rng, strength=0.1):
    """
    Perturb a in a direction that flattens its power spectrum.
    Targets high-power frequency bins for reduction.
    """
    n = len(a)
    a = a / float(np.sum(a))

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = rfft(a, size)
    P = np.abs(A) ** 2

    mean_P = P.mean()
    top_bins = np.argsort(P)[::-1][:min(8, size // 2)]

    direction = np.zeros(n)
    for m in top_bins:
        if m >= size // 2:
            continue
        weight = (P[m] - mean_P) / (P.max() + 1e-15)
        if weight <= 0:
            continue
        phase = np.angle(A[m])
        k = np.arange(n)
        direction -= weight * np.cos(2.0 * np.pi * m * k / size + phase)

    direction -= direction.mean()
    mask = a > 1e-6
    direction[~mask] = 0.0

    a_pert = a + strength * direction
    a_pert = np.maximum(a_pert, 0)
    s = float(np.sum(a_pert))
    if s > 1e-10:
        a_pert /= s
    else:
        a_pert = a.copy()

    return a_pert


def structured_init(n, rng, mode):
    """Generate structured initializations from signal processing theory."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) / n

    elif mode == 4:

        a = np.maximum(a, 0.01)
        a /= a.sum()
    elif mode == 5:
        t = (x - 0.5) * n * 0.3
        a = np.sinc(t) ** 2
        a = np.maximum(a, 0.01)
        a /= a.sum()
    elif mode == 6:
        phase = 2.0 * np.pi * (0.5 * x ** 2)

        a /= a.sum()
    elif mode == 7:
        freq = rng.integers(2, 8)

    elif mode == 8:

    elif mode == 9:

    else:

        a /= a.sum()
    return a


def nesterov_subgrad(a, time_budget=60.0, K_init=3, K_max=15):
    """Nesterov-accelerated subgradient with adaptive K and spectral escape."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 60
    K = K_init
    rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)

    while time.monotonic() - start < time_budget:
        conv = fft_convolve(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        spread = (peak_vals[0] - peak_vals[-1]) / max_val if K > 1 else 1.0
        temp = 5.0 / (1.0 + spread * 5.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / max_val)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                g += w * grad_conv_at_lag(y, int(k))

        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            if K < K_max:
                K = min(K_max, K + 1)
        else:
            stall += 1
            if stall > patience:
                a = spectral_perturbation(best_a, rng, strength=0.05)
                best_a = a.copy()
                best_obj = compute_obj(a)
                t_k = 1.0
                y = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.2)
                K = K_init

        a_prev = a_new.copy()
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        y = y_new.copy()
        t_k = t_new

    return best_a


def lp_alternating_improved(a, time_budget=200.0, verbose=False):
    """LP alternating with dual-based line search and spectral escape."""
    n = len(a)

    stall = 0

def logsumexp_optimize(a, time_budget=60.0, T=30.0, verbose=False):
    """L-BFGS on log-sum-exp smoothed objective."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def obj_func(x):
        x = np.maximum(x, 1e-15)
        s = x.sum()
        if s < 1e-10:
            return 1e10
        x = x / s
        conv = fft_convolve(x)
        max_c = conv.max()
        z = T * (conv - max_c)
        e = np.exp(z)
        return 2.0 * n * (max_c + np.log(e.sum()) / T)

    def obj_grad_func(x):
        x = np.maximum(x, 1e-15)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x = x / s
        conv = fft_convolve(x)
        max_c = conv.max()
        z =
