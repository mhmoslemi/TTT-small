```python
import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

def soft_max_gradient(a, t):
    """
    Compute gradient of (1/t) * log(sum_k exp(t * b[k] / (2n)))
    where b = conv(a,a).
    
    Returns (smooth_obj, gradient) where gradient is w.r.t. a.
    """
    n = len(a)
    b = compute_conv(a)  # length 2n-1
    # Softmax weights
    scaled = t * b / (2.0 * n)

```python
"""
Soft-max annealing + LP alternating optimizer.
Key innovation: smooth approximation of max via log-sum-exp enables
fast FISTA convergence, combined with LP alternating for global structure.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2. Scale-invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def soft_max_obj_grad(a, t):
    """
    Compute smooth approximation of 2n*max(conv(a,a)) and gradient.
    f(a) = (2n/t) * log(sum_k exp(t * b[k] / (2n))) where b = conv(a,a)
    As t -> inf, f(a) -> 2n * max(b) = 2n * max(conv(a,a)).
    
    Returns (smooth_obj, grad) where grad = df/da.
    """
    n = len(a)
    b = compute_conv(a)  # length 2n-1
    scaled = (t / (2.0 * n)) * b
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    Z = np.sum(exp_scaled)
    w = exp_scaled / Z  # softmax weights, length 2n-1
    
    # grad[m] = 2 * sum_j a[j] * w[j+m] for m=0..n-1
    # = 2 * cross_corr(w, a)[m]
    # Computed via: conv(a_flipped, w)[m + n - 1]
    a_flip = a[::-1]
    size = 1
    while size < n + (2 * n - 1) - 1:
        size <<= 1
    fa = rfft(a_flip, size)
    fw = rfft(w, size)
    conv_fw = irfft(fa * fw, size)[:n + (2 * n - 1) - 1]
    c = conv_fw[np.arange(n - 1, 2 * n - 1)]
    
    grad = 2.0 * c
    smooth_obj = (2.0 * n / t) * (max_scaled + np.log(Z))
    return smooth_obj, grad


def fista_soft_max(a0, t_schedule, max_iters=5000, time_budget=100.0, verbose=False):
    """
    FISTA (accelerated proximal gradient) on the soft-max smoothed objective.
    a0: initial point on simplex (sum=1, a>=0)
    t_schedule: function or list of t values to anneal through
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    if abs(float(np.sum(a)) - 1.0) > 1e-6:
        a = a / float(np.sum(a))
    a = np.maximum(a, 0)
    a = a / float(np.sum(a))
    
    y = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # If t_schedule is a list, iterate through it
    if isinstance(t_schedule, (list, np.ndarray)):
        phases = [(t_val, max_iters // len(t_schedule)) for t_val in t_schedule]
    else:
        phases = [(t_schedule, max_iters)]
    
    for phase_idx, (t_val, n_iters) in enumerate(phases):
        a = best_a.copy()
        y = a.copy()
        t_mom = 1.0
        
        for it in range(n_iters):
            if time.monotonic() - start > time_budget:
                return best_a
            
            _, g = soft_max_obj_grad(y, t_val)
            
            # Adaptive step size based on gradient norm
            grad_norm = np.linalg.norm(g)
            if grad_norm < 1e-15:
                break
            step = 1.0 / (grad_norm + 0.1)
            # Clip step to reasonable range
            step = min(step, 1.0)
            
            a_new = project_simplex(y - step * g)
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            y = a_new + ((t_mom - 1.0) / t_k) * (a_new - a)
            y = np.maximum(y, 0)
            y = y / float(np.sum(y))
            a = a_new
            t_mom = t_k
            
            if it % 100 == 0 or it < 10:
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    if verbose:
                        print(f"    FISTA p{phase_idx} t={t_val:.1f} it={it}: obj={best_obj:.8f}")
        
        # Update best after each phase
        obj_end = compute_obj(a)
        if obj_end < best_obj:
            best_obj = obj_end
            best_a = a.copy()
    
    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, top_k=3):

    Also includes all positions for safety.
    Returns (g_norm, M_opt).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    # Compute conv(a_n, a_n) to find top positions
    b = compute_conv(a_n)
    top_positions = np.argsort(b)[::-1][:top_k]
    
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=100.0, verbose=False):
    """LP alternating with 1D line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_good_t = 0.5

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)
        
        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 50, 'xatol': 0.005})
            best_t = res.x if res.success else last_good_t
        except Exception:
            best_t = last_good_t
        
        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            last_good_t = best_t
            stall_count = 0
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall_count += 1
            if stall_count > 20:
                break

    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def nesterov_subgrad(a0, time_budget=60.0, top_k=5, verbose=False):
    """Nesterov-accelerated subgradient on the simplex using exact max."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.05
    patience = 0

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            for m in range(n):
                j = k - m
                if 0 <= j < n:
                    grad[m] += 2.0 * y[j]
        grad /= len(top_idx)
        grad = grad - np.mean(grad)
        
        step = step_size / (np.linalg.norm(grad) + 0.01)
        step = min(step, 0.5)
        
        y_new = y - step * grad
        a_new = project_simplex(y_new)
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1.0 - t_k) * (a_new - a)
        y = project_simplex(y)
        a = a_new
        
        if it % 50 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 100:
                step_size = max(step_size * 0.7, 0.001)
                patience = 0
    
    return best_a

            s = float(np.sum(parent_arr))

                parent = parent_arr / s

        w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n_try) / max(1, n_try - 1)))
