"""
Novel peak-surgery + spectral initialization + simulated annealing approach
for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective for normalized sequence."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def spectral_init(n, rng):
    """Generate a non-negative sequence with approximately flat power spectrum."""
    # Random phases with flat magnitude
    freqs = rng.random(n)
    phases = rng.uniform(0, 2 * np.pi, n)
    # Create a signal with flat spectrum
    # Use inverse FFT of complex exponentials
    signal = irfft(freqs * np.exp(1j * phases), n)
    # Make non-negative
    signal = signal - np.min(signal) + 0.01
    signal = np.maximum(signal, 0)
    signal = signal / np.sum(signal)
    return signal

def spectral_init_2(n, rng):
    """Generate using random Fourier features."""
    # Random Gaussian in frequency domain
    spec = rng.standard_normal(n) + 1j * rng.standard_normal(n)
    # Ensure Hermitian symmetry for real signal
    signal = irfft(spec, n * 2)[:n]
    signal = np.abs(signal) + 0.01

def cosine_init(n, alpha):
    """Cosine-tapered window with parameter alpha controlling the taper."""
    x = np.linspace(0, np.pi, n)
    a = np.cos(alpha * (x - np.pi/2)) ** 2 + 0.01

def peak_surgery_sa(a, time_budget, rng, verbose=True, T0=0.01, T_end=1e-6):
    """
    Simulated annealing with peak surgery moves.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    cur_obj = best_obj
    cur_a = a.copy()
    
    start = time.monotonic()
    total_iters = 0
    accepted = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = elapsed / time_budget
        T = T0 * (T_end / T0) ** progress
        
        # Compute convolution
        conv = compute_conv(cur_a)
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        
        # Find top contributors to the peak
        k = int(max_idx)
        # (a*a)[k] = sum_j a[j] * a[k-j] for valid j
        valid_j = np.arange(max(0, k - n + 1), min(n, k + 1))
        if len(valid_j) == 0:
            break
        partners = k - valid_j
        products = cur_a[valid_j] * cur_a[partners]
        
        # Pick the largest contributor
        if rng.random() < 0.3:
            # Pick randomly among top contributors
            top_k = min(10, len(products))
            top_idx = np.argsort(products)[::-1][:top_k]
            chosen = top_idx[rng.integers(0, top_k)]
        else:
            chosen = np.argmax(products)
        
        j = int(valid_j[chosen])
        partner = k - j
        
        # Move mass from position j to a "safe" position
        # Safe position: far from both j and partner in convolution sense
        # For peak at k, positions i and k-i contribute.
        # Moving from j to m: reduces contribution to peak k by a[j]*a[k-j] -> (a[j]-d)*a[k-j]
        # But adds a[m]*... to other peaks.
        
        # Choose target position to minimize impact on other peaks
        # Simple heuristic: choose a position that minimizes the sum of products with neighbors
        candidates = np.random.choice(n, size=min(20, n), replace=False)
        best_target = -1
        best_impact = np.inf
        
        for m in candidates:
            if m == j:
                continue
            # Estimate impact: how much would (a*a) increase at other positions if we add mass to m?
            # The new contribution at peak k: (a[j]-d)*a[k-j] - a[j]*a[k-j] = -d*a[k-j] (decrease)
            # The new contribution at position k' where m participates: increases by d*a[k'-m] or d*a[m-(k'-n+1)] etc.
            # Simplified: estimate the max additional contribution
            # At position m + j (if valid): a[m]*a[j] increases
            # At position k - m (if valid): a[k-m]*a[m] increases
            impact = 0
            # Check position m + (k - j) = m + partner
            pos1 = m + partner
            if 0 <= pos1 < 2*n - 1:
                # (a*a)[pos1] would increase by d * a[pos1 - m] if pos1 - m is valid
                idx = pos1 - m
                if 0 <= idx < n:
                    impact = max(impact, cur_a[idx])
            pos2 = m + j
            if 0 <= pos2 < 2*n - 1:
                idx = pos2 - m
                if 0 <= idx < n:
                    impact = max(impact, cur_a[idx])
            
            if impact < best_impact:
                best_impact = impact
                best_target = int(m)
        
        if best_target < 0:
            best_target = int(rng.integers(0, n))
        
        if best_target == j:
            continue
        
        # Determine move size
        d = rng.uniform(0.001, 0.05) * cur_a[j]
        if d > cur_a[j]:
            d = cur_a[j] * 0.5
        
        # Apply move
        cur_a[j] -= d
        cur_a[best_target] += d
        cur_a = np.maximum(cur_a, 0)
        cur_a = cur_a / np.sum(cur_a)
        
        new_obj = compute_obj(cur_a)
        total_iters += 1
        
        # Accept/reject
        if new_obj < cur_obj:
            cur_obj = new_obj
            accepted += 1
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = cur_a.copy()
                if verbose and total_iters % 100 == 0:
                    print(f"    SA iter {total_iters}: best={best_obj:.8f}, cur={cur_obj:.8f}, T={T:.6f}")
        else:
            # Simulated annealing acceptance
            delta = new_obj - cur_obj
            if delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_obj = new_obj
                accepted += 1
            else:
                # Reject: undo
                cur_a[best_target] -= d
                cur_a[j] += d
                cur_a = np.maximum(cur_a, 0)
                cur_a = cur_a / np.sum(cur_a)
    
    return best_a

def population_search(a, time_budget, rng, verbose=True, pop_size=5):
    """
    Population-based local search with crossover.
    """
    n = len(a)
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            a_i = a.copy()
        else:
            # Perturb
            scale = 0.02 * (i + 1)
            a_i = a * np.exp(rng.normal(0, scale, n))
            a_i = np.maximum(a_i, 0)
            a_i = a_i / np.sum(a_i)
        population.append(a_i)
    
    objs = [compute_obj(p) for p in population]
    best_idx = int(np.argmin(objs))
    best_obj = objs[best_idx]
    best_a = population[best_idx].copy()
    
    start = time.monotonic()
    gen = 0
    
    while time.monotonic() - start < time_budget:
        gen += 1
        # Select parents (tournament of 2)
        i1 = int(rng.integers(0, pop_size))
        i2 = int(rng.integers(0, pop_size))
        if objs[i1] < objs[i2]:
            p1 = population[i1]
        else:
            p1 = population[i2]
        
        i3 = int(rng.integers(0, pop_size))
        i4 = int(rng.integers(0, pop_size))
        if objs[i3] < objs[i4]:
            p2 = population[i3]
        else:
            p2 = population[i4]
        
        # Crossover: mixture
        alpha = rng.uniform(0.3, 0.7)
        child = alpha * p1 + (1 - alpha) * p2
        child = np.maximum(child, 0)
        child = child / np.sum(child)
        
        # Mutation
        if rng.random() < 0.5:
            child *= np.exp(rng.normal(0, 0.01, n))
            child = np.maximum(child, 0)
            child = child / np.sum(child)
        
        obj_child = compute_obj(child)
        
        # Replace worst member
        worst_idx = int(np.argmax(objs))
        if obj_child < objs[worst_idx]:
            population[worst_idx] = child
            objs[worst_idx] = obj_child
        
        if obj_child < best_obj - 1e-14:
            best_obj = obj_child
            best_a = child.copy()
            if verbose and gen % 50 == 0:
                print(f"    Pop gen {gen}: best={best_obj:.8f}")
    
    return best_a

        # (a*g)[k] includes a[j]*g[k-j] when k-j is a valid index of g
        # So a[j] appears in row k = j + m' for m' = 0, 1, ..., n-1
        # i.e., rows j to j+n-1
        start = j
        end = min(j + n, m)
        length = end - start
        A[start:end, j] = a[:length]
    return A

Now let me think about the **peak-balancing local search**. The idea is:

1. Compute the convolution.
2. Identify the top K peaks (K=5).
3. For each peak at position k, find the dominant contributors (j, k-j) with the largest product a[j]*a[k-j].
4. For each dominant contributor, try moving a small amount of mass from position j to a "safe" position m.
5. A position m is "safe" if moving mass there doesn't significantly increase any peak.
6. Accept the move if it reduces the objective.

The key is step 5: how to efficiently find safe positions.

**Efficient safe position finding**:
For a given position m, the "exposure" is the maximum over all peaks k' of the coefficient of a[m] in (a*a)[k']. The coefficient of a[m] in (a*a)[k'] is a[k'-m] (if valid). So the exposure of position m is:
exposure(m) = max over k' of a[k'-m] (where k'-m is in [0, n-1]).

This is just max(a) if m is such that k'-m can range over all of [0, n-1]. For m in the "interior" (m >= 1 and m <= n-2), the range of k'-m is [1-m, 2n-2-m] ∩ [0, n-1] = [0, n-1] (for most m). So exposure(m) ≈ max(a) for most positions.

This means the exposure approach doesn't differentiate well. A better approach:

**Contribution-based safety**: A position m is "safe" if the current convolution values at positions involving m are low. Specifically, if we add mass δ to position m, the increase in (a*a)[k'] is approximately δ * a[k'-m] + a[m]*δ (for the cross term). So the increase at position k' is δ * (a[k'-m] + a[m]) (approximately, ignoring the δ² term).

The maximum increase is δ * max_k' (a[k'-m] + a[m]). Since a[m] is fixed, this is δ * (max(a) + a[m]). Again, not very discriminating.
