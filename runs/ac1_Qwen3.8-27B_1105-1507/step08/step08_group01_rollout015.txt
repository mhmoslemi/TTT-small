Combines LP alternating, multiplicative gradient descent with Nesterov momentum,
pairwise coordinate descent, and multi-n exploration.
Key innovations over previous approach:

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(n):
            idx = j + k
            if idx < 2 * n - 1:
                A[idx, j] += a[k]
    return A


def build_conv_matrix_vec(a):
    """Build conv matrix using vectorized approach for speed."""
    n = len(a)
    rows = np.arange(2 * n - 1)
    cols_from = np.arange(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        valid_k = np.arange(n)
        indices = j + valid_k
        mask = indices < (2 * n - 1)
        A[indices[mask], j] = a[valid_k[mask]]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp_full(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M, y_dual)."""
    try:
        import cvxpy as cp
    except ImportError:
        return _solve_lp_scipy(a)

    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None, None
    a = a / s

    A = build_conv_matrix_vec(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = ['CLARABEL', 'ECOS', 'SCS', 'MOSEK']
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    # Get dual variables
                    y_dual = None
                    for con in constraints:
                        if con.dual_value is not None and len(np.atleast_1d(con.dual_value)) == 2 * n - 1:
                            y_dual = np.abs(np.atleast_1d(con.dual_value)) / max(1e-10, float(np.sum(np.abs(np.atleast_1d(con.dual_value)))))
                    return g_val, float(M.value), y_dual
        except Exception:

    return _solve_lp_scipy(a)


def _solve_lp_scipy(a):
    """Fallback LP solver using scipy linprog (HiGHS)."""
    from scipy.optimize import linprog
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None, None
    a = a / s
    A = build_conv_matrix_vec(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g_val))
        if gs > 1e-10:
            return g_val / gs, float(res.x[n]), None
    return None, None, None

def convex_line_search(a, g, n_evals=10):
    """Find optimal t in [0,1] for a_new = (1-t)*a + t*g using convexity."""
    n = len(a)
    sa = float(np.sum(a))
    sg = float(np.sum(g))
    if sa < 1e-10 or sg < 1e-10:
        return 0.0, a.copy()
    a_norm = a / sa
    g_norm = g / sg

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return 0.0, a_norm
    return t_opt, a_new / s_new

def multiplicative_gradient_descent(a, time_budget=100.0, verbose=False):
    """
    Multiplicative update rule with Nesterov momentum.
    a_new[i] = a[i] * exp(-lr * grad_i) where grad_i = sum_k y_k * a_{k-i}
    This naturally preserves non-negativity.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.01
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    T = 0.001  # Temperature for softmax peak weighting

    for it in range(3000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)

        # Softmax weights on peaks
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -400, 400)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        if total < 1e-30:
            break
        y = exp_vals / total

        # Gradient: grad[i] = sum_k y_k * a_{k-i} (valid indices only)
        # (a*a)_k = sum_j a_j * a_{k-j}, so d(a*a)_k / da_i = a_{k-i} (for valid k-i)
        # grad[i] = sum_k y_k * a_{k-i}
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if y[k] < 1e-10:
                continue
            # valid j: 0 <= j < n and 0 <= k-j < n
            j_lo = max(0, k - (n - 1))
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                a_part = a[j_lo:j_hi + 1]
                indices = k - np.arange(j_lo, j_hi + 1)
                grad[indices] += y[k] * a_part

        # Nesterov momentum
        v = momentum * v - lr * grad
        v = np.maximum(v, -2.0)  # Clip to prevent explosive updates

        # Multiplicative update
        a_new = a * np.exp(v)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 200 == 0 and lr < 0.05:
                lr *= 1.02
        else:
            stall += 1
            if stall > 30:
                lr *= 0.7
                momentum = max(0.5, momentum * 0.95)
                if lr < 1e-6:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
                T = max(0.0001, T * 0.9)

        # Decay temperature
        if it > 50:
            T = max(0.00005, T * 0.999)

    return best_a

def dual_guided_update(a, y_dual, time_budget=60.0, verbose=False):
    """

    The dual variables y_k indicate which convolution peaks are "hottest".
    We use them to compute a gradient-like direction and apply multiplicative updates.
    """

        return a

        max_c = np.max(conv)
        T = max_c * 0.1
        shifted = (conv - max_c) / T
        y_dual = np.exp(np.clip(shifted, -300, 300))
        y_dual = y_dual / np.sum(y_dual)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.005
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        # Compute pressure: p[i] = sum_k y_k * a_{k-i}
        pressure = np.zeros(n)
        for k in range(2 * n - 1):
            if y_dual[k] < 1e-12:
                continue
            j_lo = max(0, k - (n - 1))
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                indices = k - np.arange(j_lo, j_hi + 1)
                pressure[indices] += y_dual[k] * a[j_lo:j_hi + 1]

        # Multiplicative update
        update = np.exp(-lr * pressure / max(pressure.max(), 1e-10))
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.8
            if lr < 1e-6:
                break

        # Update dual variables based on current peaks
        conv = compute_conv(a)
        max_c = np.max(conv)
        T = max_c * 0.15
        shifted = (conv - max_c) / T
        y_dual = np.exp(np.clip(shifted, -300, 300))
        y_sum = np.sum(y_dual)
        if y_sum > 1e-30:

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    if n >= 4:
        from scipy.interpolate import PchipInterpolator
        try:
            interp = PchipInterpolator(x_old, a)
            a_new = interp(x_new)
        except Exception:
            a_new = np.interp(x_new, x_old, a)
    else:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, rng):
    """Generate structured initial sequences for diversity."""
    strategy = rng.integers(0, 5)
    if strategy == 0:
        # Uniform
        a = np.ones(n) / n
    elif strategy == 1:
        # Cosine taper
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
    elif strategy == 2:
        # Gaussian bump
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
        a = a / np.sum(a)
    elif strategy == 3:
        # Random with smoothing
        a = rng.random(n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    else:
        # Power law
        x = np.arange(1, n + 1, dtype=np.float64)
        a = x ** (-0.5)
        a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-strategy optimization combining LP alternating,
    multiplicative gradient descent, dual-guided updates, and pairwise CD.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 50.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={len(parent)}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = structured_init(n_init, rng)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    a = best_seq.copy()
    n = len(a)

    # === Phase 1: LP Alternating (primary method) ===
    print(f"\n=== Phase 1: LP Alternating (n={n}) ===")
    remaining = deadline - time.monotonic()
    phase1_budget = min(300.0, remaining * 0.3)
    print(f"  Budget: {phase1_budget:.0f}s")

    a_opt = lp_alternating(a, time_budget=phase1_budget, use_reduced_lp=True, verbose=True)
    obj_opt = compute_obj(a_opt)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        a = a_opt.copy()
    print(f"  Phase 1 done: obj={best_score:.8f}")

    # === Phase 2: Multiplicative Gradient Descent (new method) ===
    print(f"\n=== Phase 2: Multiplicative GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        phase2_budget = min(150.0, remaining * 0.18)
        a_mgd = multiplicative_gradient_descent(a, time_budget=phase2_budget, verbose=True)
        obj_mgd = compute_obj(a_mgd)
        if obj_mgd < best_score:
            best_score = obj_mgd
            best_seq = a_mgd.copy()
            a = a_mgd.copy()
            print(f"  Phase 2 improved: obj={best_score:.8f}")
        else:
            print(f"  Phase 2: obj={obj_mgd:.8f} (no improvement)")

    # === Phase 3: LP Alternating again (from new point) ===
    print(f"\n=== Phase 3: LP Alternating (round 2) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        phase3_budget = min(200.0, remaining * 0.22)
        a_opt = lp_alternating(a, time_budget=phase3_budget, use_reduced_lp=False, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  Phase 3 improved: obj={best_score:.8f}")
        else:
            print(f"  Phase 3: obj={obj_opt:.8f} (no improvement)")

    # === Phase 4: Dual-guided update (new) ===
    print(f"\n=== Phase 4: Dual-guided Update ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        # Get dual from LP
        g_lp, M_lp, y_dual = solve_lp_full(a)
        phase4_budget = min(60.0, remaining * 0.08)
        a_dl = dual_guided_update(a, y_dual, time_budget=phase4_budget, verbose=True)
        obj_dl = compute_obj(a_dl)
        if obj_dl < best_score:
            best_score = obj_dl
            best_seq = a_dl.copy()
            a = a_dl.copy()
            print(f"  Phase 4 improved: obj={best_score:.8f}")
        else:
            print(f"  Phase 4: obj={obj_dl:.8f} (no improvement)")

    # === Phase 5: Pairwise Coordinate Descent (new) ===
    print(f"\n=== Phase 5: Pairwise CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        phase5_budget = min(120.0, remaining * 0.15)
        a_cd = pairwise_coordinate_descent(a, time_budget=phase5_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a = a_cd.copy()
            print(f"  Phase 5 improved: obj={best_score:.8f}")
        else:
            print(f"  Phase 5: obj={obj_cd:.8f} (no improvement)")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best - 50, n_best - 25, n_best + 25, n_best + 50,
        n_best + 75, n_best + 100,
        700, 750, 800, 850, 900, 950, 1000, 1050, 1100, 1200
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1400]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick multi-method pipeline
        alloc_lp = min(remaining * 0.08, 30.0)
        a_t1 = lp_alternating(a_try, time_budget=alloc_lp, use_reduced_lp=True)
        obj_t1 = compute_obj(a_t1)

        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc_mgd = min(remaining * 0.06, 25.0)
            a_t2 = multiplicative_gradient_descent(a_t1, time_budget=alloc_mgd)
            obj_t2 = compute_obj(a_t2)
            if obj_t2 < obj_t1:
                a_t1 = a_t2
                obj_t1 = obj_t2

        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_t1 < best_score + 0.003:
            alloc_cd = min(remaining * 0.08, 25.0)
            a_t1 = pairwise_coordinate_descent(a_t1, time_budget=alloc_cd)
            obj_t1 = compute_obj(a_t1)

        if obj_t1 < best_score:
            best_score = obj_t1
            best_seq = a_t1.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_t1:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    a_final = best_seq.copy()
    n_final = len(a_final)

    # 7a: LP
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(80.0, remaining * 0.3)
        a_lp = lp_alternating(a_final, time_budget=alloc, use_reduced_lp=True, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_final = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f}")

    # 7b: Multiplicative GD
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(50.0, remaining * 0.2)
        a_m = multiplicative_gradient_descent(a_final, time_budget=alloc)
        obj_m = compute_obj(a_m)
        if obj_m < best_score:
            best_score = obj_m
            best_seq = a_m.copy()
            a_final = a_m.copy()
            print(f"  Final MGD: obj={best_score:.8f}")

    # 7c: Pairwise CD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(60.0, remaining * 0.25)
        a_pd = pairwise_coordinate_descent(a_final, time_budget=alloc, verbose=True)
        obj_pd = compute_obj(a_pd)
        if obj_pd < best_score:
            best_score = obj_pd
            best_seq = a_pd.copy()
            a_final = a_pd.copy()
            print(f"  Final PCD: obj={best_score:.8f}")

    # 7d: One more LP from polished point
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(50.0, remaining * 0.3)
        a_lp2 = lp_alternating(a_final, time_budget=alloc, use_reduced_lp=False)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Final LP2: obj={best_score:.8f}")

    # === Final CD squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(30.0, remaining * 0.5)
        a_cd_final = pairwise_coordinate_descent(best_seq, time_budget=alloc)
        obj_cd_final = compute_obj(a_cd_final)
        if obj_cd_final < best_score:
            best_score = obj_cd_final
            best_seq = a_cd_final.copy()
            print(f"  Final CD squeeze: obj={best_score:.8f}")

    # === Validation ===
    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
            print(f"  evaluate_sequence confirms: {best_score:.8f}")
        else:
            print(f"  evaluate_sequence: {val:.8f} vs computed {best_score:.8f}")
    except Exception as e:
        print(f"  Evaluation warning: {e}")

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
