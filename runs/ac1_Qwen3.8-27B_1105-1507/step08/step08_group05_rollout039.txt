```python
    return irfft(fa * fa, size)[:2 * n - 1]
```

```python
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < 2*n - 1
    grad[valid] -= w[diag_idx[valid]] * a[valid]
    
    return grad
```

"""
Optimizer using smooth log-sum-exp subgradient with Nesterov acceleration,
FFT-based O(n log n) gradient, LP alternating, and multi-resolution search.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def smooth_gradient(a, tau=5.0):
    """
    Compute the smooth gradient of max(conv(a,a)) using log-sum-exp.
    O(n log n) via FFT.
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    if max_c < 1e-15:
        return np.zeros(n), conv
    # Normalize: w[k] = exp(tau * (conv[k]/max_c - 1))
    # This makes the peak have weight 1 and others decay
    scaled = tau * (conv / max_c - 1.0)
    scaled -= np.max(scaled)  # numerical stability (should be 0 at peak)
    w = np.exp(scaled)
    w_sum = float(np.sum(w))
    if w_sum < 1e-30:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum
    
    # cross[m] = sum_j a[j] * w[m+j] = conv(a_rev, w)[m + n - 1]
    a_rev = a[::-1]
    size = 1
    while size < n + 2 * n - 1:
        size <<= 1
    fa = rfft(a_rev, size)
    fw = rfft(w, size)
    conv_cw = irfft(fa * fw, size)[:n + 2 * n - 2]
    cross = conv_cw[n - 1:2 * n - 1]
    
    # grad[m] = 2*cross[m] - w[2m]*a[m] (if 2m < 2n-1)
    grad = 2.0 * cross
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < 2 * n - 1
    grad[valid] -= w[diag_idx[valid]] * a[valid]
    
    return grad, conv


def nesterov_optimize(a0, time_budget=100.0, tau=5.0, step_size=0.1, verbose=False):
    """Nesterov-accelerated subgradient on the simplex with smooth gradient."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    patience = 0
    momentum = 0.9
    
    for it in range(500000):
        if time.monotonic() - start > time_budget:
            break
        
        grad, conv = smooth_gradient(y, tau=tau)
        # Center gradient (remove component in "1" direction)
        grad = grad - np.mean(grad)
        
        y_new = y - step_size * grad
        a_new = project_simplex(y_new)
        
        # Nesterov update
        t_k = 2.0 / (it + 2.0)
        # y = a_new + (1 - t_k) * (a_new - a_prev) ... actually standard is:
        # y_{k+1} = x_{k+1} + (t_k - 1)/t_{k+1} * (x_{k+1} - x_k)
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum_factor = (t_k - 1.0) / t_next
        y = a_new + momentum_factor * (a_new - a_prev)
        y = project_simplex(y)
        
        a_prev = a
        a = a_new
        t_k = t_next
        
        # Check objective periodically
        if it % 5 == 0:
            obj = compute_obj(a)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 100:
                step_size = max(step_size * 0.7, 0.005)
                patience = 0
                # Reset momentum
                y = a.copy()
                a_prev = a.copy()
        
        if it % 500 == 499:
            step_size = max(step_size * 0.97, 0.005)
        
        if verbose and it % 5000 == 0:
            print(f"    Nesterov it={it}: best={best_obj:.8f}, step={step_size:.4f}")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, time_budget=10.0):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    start = time.monotonic()
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        if time.monotonic() - start > time_budget:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None
