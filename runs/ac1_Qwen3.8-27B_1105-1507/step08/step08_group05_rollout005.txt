"""
Hybrid optimizer: Fast LP alternating + Nesterov subgradient + multiplicative updates.
Uses exact subgradient formula, adaptive step sizes, and systematic exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize_scalar
import time

def compute_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def conv_subgrad(a, k):
    """Subgradient of conv(a,a)[k] w.r.t. a. grad[j] = 2*a[k-j] if valid."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        grad[lo:hi+1] = 2.0 * a[k - hi : k - lo + 1][::-1]
    return grad

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

def nesterov_subgrad(a0, time_budget=60.0, top_k=5, verbose=False):
    """Nesterov-accelerated subgradient on the simplex."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.05
    patience = 0
    eval_count = 0

    for it in range(500000):
        if time.monotonic() - start > time_budget:
            break
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            grad += conv_subgrad(y, int(k))
        grad /= len(top_idx)
        grad = grad - np.mean(grad)
        
        y_new = y - step_size * grad
        a_new = project_simplex(y_new)
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1.0 - t_k) * (a_new - a)
        y = np.maximum(y, 0)
        ys = float(np.sum(y))
        if ys < 1e-10:
            y = a_new
        else:
            y = y / ys
        a = a_new
        
        if (it + 1) % 5 == 0:
            obj = compute_obj(a)
            eval_count += 1
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 100:
                step_size = max(step_size * 0.7, 0.001)
                patience = 0
        
        if (it + 1) % 500 == 0:
            step_size = max(step_size * 0.98, 0.002)

    return best_a

def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    m = 2 * n - 1
    
    # Build constraint matrix: conv(a,g)[k] - M <= 0 for each k
    # Variables: [g_0, ..., g_{n-1}, M]
    # A_ub: (2n-1) x (n+1)
    # A_ub[k, j] = a_n[k-j] if valid, A_ub[k, n] = -1
    A_ub = np.zeros((m, n + 1))
    for k in range(m):
        A_ub[k, n] = -1.0
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_ub[k, j] = a_n[k - j]
    
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize M
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g))
            if gs > 1e-10:
                return g / gs, float(res.x[n])
    except Exception:
        pass
    return None, None

def lp_alternate(a0, time_budget=100.0, verbose=False):
    """LP alternating with 1D line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        
        # Try several blending factors
        best_t = 0.0
        best_t_obj = best_obj
        for t_candidate in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t_candidate) * a + t_candidate * g_lp
            a_try = np.maximum(a_try, 0)

            a_try = a_try / st

def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

    return w / sw
