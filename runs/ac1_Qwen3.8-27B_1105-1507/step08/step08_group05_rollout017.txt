def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

        frac = min(1.0, it / (total_iters * 0.8))
        p_cur = p_start + (p_end - p_start) * frac
        
        # Compute gradient of p-norm
        grad, conv = compute_pnorm_grad(a, p_cur)
        
        # Center gradient for simplex (remove mean component)
        grad = grad - np.mean(grad)
        
        # Adam update
        t = it + 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t)
        v_hat = v_state / (1 - beta2 ** t)
        
        # Adaptive step size
        lr = 0.05 / (np.sqrt(np.mean(v_hat)) + eps)
        y = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to simplex
        a_new = project_simplex(y)
        
        # Evaluate
        if it % 5 == 0:
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """
    LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_norm, M_opt) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None

def lp_alternate(a0, time_budget=80.0, verbose=True):
    """LP alternating with 1D line search."""
    n = len(a0)

def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw

"""
Adam p-norm optimization + softmax mirror descent + LP polish for minimizing
2*n*max(conv(a,a))/(sum(a))^2.

Key innovations over previous approach:
1. Full gradient via FFT cross-correlation (not just top-k peaks)
2. p-norm annealing for smooth -> sharp focus transition
3. Mirror descent (multiplicative weights) variant for better positivity handling
4. Softmax-weighted gradient for adaptive peak focus
5. Aggressive n-sweep with multiple resampling strategies
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_pnorm_grad(a, p):
    """
    Compute gradient of sum(conv(a,a)[k]^p) w.r.t. a using FFT.
    Returns (gradient, conv) where gradient has same length as a.
    """
    n = len(a)
    conv = compute_conv(a)
    conv = np.maximum(conv, 0)
    w = conv ** (p - 1)

    # grad[m] = 2p * sum_j a[j] * w[m+j]
    # = 2p * conv(a_rev, w)[m + n - 1]
    a_rev = a[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    cross = irfft(rfft(a_rev, size) * rfft(w, size), size)
    grad = 2.0 * p * cross[n - 1:2 * n - 1]
    return grad, conv


def compute_softmax_grad(a, temperature):
    """
    Compute gradient of the smooth max approximation:
    f(a) = temperature * log(sum(exp(conv[k]/temperature)))
    grad[m] = 2 * sum_k softmax(conv[k]/T)[k] * a[k-m]
    """
    n = len(a)
    conv = compute_conv(a)
    # Softmax weights
    conv_shifted = conv - np.max(conv)
    weights = np.exp(conv_shifted / temperature)
    weights /= np.sum(weights)

    # grad[m] = 2 * sum_j a[j] * weights[m+j]
    a_rev = a[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    cross = irfft(rfft(a_rev, size) * rfft(weights, size), size)
    grad = 2.0 * cross[n - 1:2 * n - 1]
    return grad, conv


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

    """
    Adam optimizer on the simplex with p-norm annealing.

    s = float(np.sum(a))
    if s < 1e-10:

    best_obj = compute_obj(a)

    m_state = np.zeros(n)
    v_state = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8

    start = time.monotonic()
    total_iters = 100000
    patience = 0
    max_patience = 500

        p_cur = p_start + (p_end - p_start) * frac

        grad, conv = compute_pnorm_grad(a, p_cur)
        grad = grad - np.mean(grad)

        t = it + 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t)
        v_hat = v_state / (1 - beta2 ** t)

        norm_v = np.sqrt(np.mean(v_hat)) + eps
        lr = lr_base / norm_v
        lr = min(lr, 0.5)

        y = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(y)

        if it % 10 == 0:
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()
                patience = 0
            else:
                patience += 1
            if patience > max_patience:
                lr_base = max(lr_base * 0.5, 0.001)
                patience = 0

        a = a_new

    if verbose:
        print(f"    Adam p-norm done: obj={best_obj:.8f}, iters={it+1}, lr_base={lr_base:.4f}")
    return best_a


def mirror_descent_sm(a0, time_budget=100.0, temp_start=0.1, temp_end=0.01,
                      lr_base=0.01, verbose=True):
    """
    Mirror descent (multiplicative weights) with softmax-weighted gradient.
    a_new = a * exp(-lr * grad) / Z (log-domain for stability)
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    log_a = np.log(np.maximum(a, 1e-300))

    start = time.monotonic()
    total_iters = 50000
    patience = 0
    max_patience = 300

    for it in range(total_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = min(1.0, it / (total_iters * 0.8))
        temp = temp_start + (temp_end - temp_start) * frac
        lr = lr_base

        grad, conv = compute_softmax_grad(a, temp)
        grad = grad - np.mean(grad)

        # Multiplicative update in log domain
        log_a_new = log_a - lr * grad
        # Normalize (project to simplex in multiplicative form)
        log_a_new = log_a_new - np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        log_a = np.log(np.maximum(a_new, 1e-300))

        if it % 5 == 0:
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()
                patience = 0
            else:
                patience += 1
            if patience > max_patience:
                lr_base = max(lr_base * 0.7, 0.001)
                patience = 0

        a = a_new

    if verbose:
        print(f"    Mirror descent done: obj={best_obj:.8f}, iters={it+1}")
    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_norm, M_opt) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=80.0, verbose=True):
    """LP alternating with 1D line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)

        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 30, 'xatol': 0.01})
            best_t = res.x if res.success else 0.5
        except Exception:
            best_t = 0.5

        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            stall_count = 0
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall_count += 1
            if stall_count > 8:
                break

    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def truncate_pad(a, new_n):
    """Truncate or zero-pad to new_n."""
    n = len(a)
    if new_n >= n:
        out = np.zeros(new_n)
        out[:n] = a
        return out
    else:
        return a[:new_n].copy()


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def combined_optimize(a0, time_budget=100.0, verbose=True):
    """Run Adam p-norm then mirror descent then LP in sequence."""
    n = len(a0)
    start = time.monotonic()

    # Adam p-norm
    alloc1 = min(time_budget * 0.45, 40.0)
    a1 = adam_pnorm(a0, time_budget=alloc1, p_start=4, p_end=80, lr_base=0.05, verbose=verbose)

    # Mirror descent
    remaining = time_budget - (time.monotonic() - start)
    alloc2 = min(remaining * 0.3, 20.0)
    a2 = mirror_descent_sm(a1, time_budget=alloc2, temp_start=0.05, temp_end=0.005,
                           lr_base=0.01, verbose=verbose)

    # LP polish
    remaining = time_budget - (time.monotonic() - start)
    alloc3 = max(remaining * 0.3, 5.0)
    a3 = lp_alternate(a2, time_budget=alloc3, verbose=verbose)

    # Pick best
    objs = [compute_obj(a1), compute_obj(a2), compute_obj(a3)]
    best_idx = int(np.argmin(objs))
    result = [a1, a2, a3][best_idx]
    if verbose:
        print(f"    Combined: best={objs[best_idx]:.8f} (Adam={objs[0]:.6f}, MD={objs[1]:.6f}, LP={objs[2]:.6f})")
    return result


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Adam p-norm + mirror descent + LP polish + n-sweep.
    Uses full gradient via FFT for efficient optimization.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = hann_window(800)
        best_score = compute_obj(best_seq)
        print(f"Hann init (n=800): obj={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: Adam p-norm on parent (250s) ===
    print(f"\n=== Phase 1: Adam p-norm (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(130.0, remaining * 0.35)
    a_opt = adam_pnorm(best_seq, time_budget=alloc, p_start=4, p_end=100, lr_base=0.05, verbose=True)
    obj_opt = compute_obj(a_opt)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        print(f"  ** Adam phase 1: obj={best_score:.8f}")

    # Higher p continuation
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc2 = min(100.0, remaining * 0.3)
        a_opt2 = adam_pnorm(best_seq, time_budget=alloc2, p_start=20, p_end=300, lr_base=0.03, verbose=True)
        obj_opt2 = compute_obj(a_opt2)
        if obj_opt2 < best_score:
            best_score = obj_opt2
            best_seq = a_opt2.copy()
            print(f"  ** Adam phase 1b (high p): obj={best_score:.8f}")
        else:
            print(f"  Adam phase 1b: obj={obj_opt2:.8f}")

    # === Phase 2: Mirror descent (120s) ===
    print(f"\n=== Phase 2: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(100.0, remaining * 0.25)
        a_md = mirror_descent_sm(best_seq, time_budget=alloc, temp_start=0.08, temp_end=0.003,
                                 lr_base=0.015, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  ** Mirror descent: obj={best_score:.8f}")
        else:
            print(f"  Mirror descent: obj={obj_md:.8f}")

    # === Phase 3: LP Alternating (200s) ===
    print(f"\n=== Phase 3: LP Alternating ===")
    a = best_seq.copy()
    for round_idx in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = min(40.0, remaining * 0.3)
        a_new = lp_alternate(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new
            print(f"  ** LP R{round_idx}: obj={best_score:.8f}")
        else:
            print(f"  LP R{round_idx}: obj={obj:.8f}")
            # Perturb to escape
            a = best_seq * np.exp(rng.normal(0, 0.02, len(best_seq)))
            a = np.maximum(a, 0)
            a /= np.sum(a)

    # === Phase 4: n-sweep (200s) ===
    print(f"\n=== Phase 4: n-Sweep ===")
    n_options = [400, 500, 600, 800, 1000, 1200, 1500, 2000]
    n_options = sorted(set(n_options + [n_base]))
    n_options = [n for n in n_options if 300 <= n <= 3000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            # Try both resampling and truncation
            a_res = resample_seq(best_seq, n_try)
            a_trunc = truncate_pad(best_seq, n_try)
            # Pick whichever has better initial obj
            obj_res = compute_obj(a_res)
            obj_trunc = compute_obj(a_trunc)
            if obj_res <= obj_trunc:
                a_try = a_res
            else:
                a_try = a_trunc

        s_try = float(np.sum(a_try))
        if s_try < 0.01:
            continue
        a_try = a_try / s_try

        alloc = min(18.0, remaining * 0.25)
        a_opt = adam_pnorm(a_try, time_budget=alloc, p_start=6, p_end=100, lr_base=0.04, verbose=False)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Final combined polish (200s) ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(100.0, remaining * 0.4)
        a_fin = combined_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score - 1e-12:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  ** Final combined: obj={best_score:.8f}")
        else:
            print(f"  Final combined: obj={obj_fin:.8f}")

    # Adam with very high p for final squeeze
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(50.0, remaining * 0.4)
        a_sq = adam_pnorm(best_seq, time_budget=alloc, p_start=100, p_end=1000, lr_base=0.01, verbose=True)
        obj_sq = compute_obj(a_sq)
        if obj_sq < best_score - 1e-12:
            best_score = obj_sq
            best_seq = a_sq.copy()
            print(f"  ** High-p squeeze: obj={best_score:.8f}")
        else:
            print(f"  High-p squeeze: obj={obj_sq:.8f}")

    # === Phase 6: Random restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 6: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(12):
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break
            scale = 0.015 + 0.005 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr < 1e-10:
                continue
            a_r = a_r / sr
            alloc = min(remaining * 0.3, 10.0)
            a_opt = adam_pnorm(a_r, time_budget=alloc * 0.5, p_start=6, p_end=80, lr_base=0.04, verbose=False)
            a_opt = lp_alternate(a_opt, time_budget=alloc * 0.5, verbose=False)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score - 1e-12:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  ** R{r}: obj={best_score:.8f}")
            else:
                print(f"  R{r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
