    n = len(a)

    n = len(a)

    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    j_start = k - i_end
    j_end = k - i_start
    g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]

        A[j:j + r, j] = a[:r]
    return A

                                best_t_obj = obj_try
                                best_t_val = t_par
                                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            best_t = best_t_val
            stall = 0
            # Update A and conv
            A = build_conv_matrix(a)
            conv_a = compute_conv(a)
            M_cur = float(np.max(conv_a))
            # Adapt n_active based on peak spread
            sorted_conv = np.sort(conv_a)[::-1]
            if len(sorted_conv) > 1:
                spread = (sorted_conv[0] - sorted_conv[min(10, len(sorted_conv)-1)]) / (sorted_conv[0] + 1e-15)
                n_active = max(30, int((0.05 + 0.1 * spread) * (2 * n - 1)))
            if verbose and it % 10 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 20:
                # Escape: perturb and reset
                noise_scale = min(0.25, 0.03 * stall / 10.0)
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                A = build_conv_matrix(a)
                conv_a = compute_conv(a)
                M_cur = float(np.max(conv_a))
                stall = 0
                best_t = rng.uniform(0.3, 0.7)

    return best_a


def nesterov_accelerated(a, time_budget=60.0, K=8):
    """Nesterov-accelerated subgradient with adaptive temperature."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 3.0 / np.sqrt(n)
    stall = 0
    patience = 100

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        # Adaptive temperature
        spread = (peak_vals[0] - peak_vals[-1]) / (max_val + 1e-15)
        temp = 5.0 / (1.0 + spread * 5.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / (max_val + 1e-15))
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-10:
                g += w * grad_conv_at_peak(y, int(k))

        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                t_k = 1.0
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.5)

        a_prev = a_new.copy()
        y = y_new.copy()
        t_k = t_new
        it += 1

    return best_a


def mirror_descent_fast(a, time_budget=50.0):
    """Mirror descent with momentum and adaptive LR."""
    n = len(a)
    a = np.maximum(a, 1e-12).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    momentum = 0.3
    v = np.zeros(n)
    stall = 0
    rng = np.random.default_rng(999)

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()
        if max_val < 1e-15:
            break

        # Softmax weights over all peaks
        temp = 2.0
        w = np.exp((conv - max_val) / (temp * max_val + 1e-15))
        w_sum = w.sum()
        if w_sum < 1e-15:
            break
        w = w / w_sum

        # Only use top 10% for efficiency
        threshold = np.percentile(w, 90)
        active = w > threshold
        if not active.any():
            idx_max = np.argmax(conv)
            active[idx_max] = True
        w_active = w[active]
        w_active /= w_active.sum()
        idx_active = np.where(active)[0]

        g = np.zeros(n)
        for wi, k in zip(w_active, idx_active):
            if wi > 1e-10:
                g += wi * grad_conv_at_peak(a, int(k))

        v = momentum * v + g
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = max(1e-6, lr * 0.998)
            momentum = min(0.7, momentum + 0.001)
        else:
            stall += 1
            if stall > 80:
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                momentum = 0.3
                lr = 2.0 / np.sqrt(n)
                stall = 0

        it += 1

    return best_a


def smooth_lbfgs(a, time_budget=80.0, T=50.0):
    """Smooth optimization using log-sum-exp with L-BFGS-B."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)

    call_count = [0]
    max_calls = 300

    def obj_grad(x):
        call_count[0] += 1
        if call_count[0] > max_calls:
            return 1e10, np.zeros(len(x))
        x = np.maximum(x, 1e-12)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x = x / s
        conv = compute_conv(x)
        max_c = conv.max()
        if max_c < 1e-15:
            return 0, np.zeros(len(x))
        z = T * (conv - max_c)
        e = np.exp(z)
        sum_e = e.sum()
        obj_val = 2.0 * n * (max_c + np.log(sum_e) / T)
        w = e / sum_e
        g = np.zeros(n)
        for k in range(len(conv)):
            if w[k] > 1e-6:
                g += w[k] * grad_conv_at_peak(x, int(k))
        g *= 2.0 * n
        return obj_val, g

    try:
        res = sp_minimize(obj_grad, a, method='L-BFGS-B',
                          bounds=[(1e-10, None)] * n,
                          options={'maxiter': 150, 'maxfun': 400, 'disp': False})
        a_opt = np.maximum(res.x, 0)
        s_opt = float(np.sum(a_opt))
        if s_opt > 1e-10:
            a_opt = a_opt / s_opt
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
    except Exception:
        pass

    return best_a


def coordinate_refine(a, time_budget=30.0):
    """Coordinate descent: move mass from peak-contributing positions to non-contributing ones."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find contributors to peak
        contrib = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib.append((a[i] * a[j], i, j))
        contrib.sort(reverse=True)

        # Non-contributors
        non_contrib = [i for i in range(n) if k_star - i < 0 or k_star - i >= n]
        if not non_contrib:
            break

        improved = False
        for ci in range(min(20, len(contrib))):
            val, i, j = contrib[ci]
            if val < 1e-12:
                break
            for target in non_contrib[:6]:
                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * a[min(i, j)]
                    if delta < 1e-14:
                        continue
                    src = min(i, j)
                    a[src] -= delta
                    a[target] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Gaussian bump
        center = n // 2
        sigma = n // 6
        x = np.arange(n)
        a = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Two bumps
        x = np.arange(n)
        a = np.exp(-((x - n//4) ** 2) / (2 * (n//8) ** 2)) + np.exp(-((x - 3*n//4) ** 2) / (2 * (n//8) ** 2))
        a = a / a.sum()
    elif mode == 4:
        # U-shape
        x = np.linspace(-1, 1, n)
        a = 1.0 + x**2
        a = a / a.sum()
    elif mode == 5:
        # Sine modulated
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode == 6:
        # Trapezoid
        x = np.arange(n)
        a = np.ones(n)
        ramp = n // 10
        for i in range(ramp):
            a[i] *= (i + 1) / ramp
            a[n-1-i] *= (i + 1) / ramp
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Hybrid optimization combining LP alternating,
    Nesterov subgradient, mirror descent, smooth L-BFGS, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Intensive LP Alternating (35% time) ===
    print(f"\n=== Phase 1: LP Alternating (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 450.0)
    if p1_budget > 30:
        a_lp = lp_alternating_fast(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Nesterov + Mirror Descent (15% time) ===
    print(f"\n=== Phase 2: Nesterov + Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 150.0)
    if p2_budget > 20:
        ns_budget = p2_budget * 0.5
        a_ns = nesterov_accelerated(best_seq, time_budget=ns_budget, K=10)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")

        md_budget = p2_budget * 0.5
        a_md = mirror_descent_fast(best_seq, time_budget=md_budget)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Mirror: obj={best_score:.8f} **")

    # === Phase 3: More LP with perturbation (20% time) ===
    print(f"\n=== Phase 3: LP with Perturbation ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 250.0)
    if p3_budget > 30:
        # Run LP from slightly perturbed start
        a_pert = best_seq * np.exp(rng.normal(0, 0.05, n_current))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_lp2 = lp_alternating_fast(a_pert, time_budget=p3_budget, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase 3 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 LP: obj={obj_lp2:.8f} (no improvement)")

    # === Phase 4: Multi-n Exploration (15% time) ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        max(200, n_current - 100),
        max(200, n_current - 50),
        n_current + 50,
        n_current + 100,
        n_current + 200,
        400, 500, 600, 700, 800, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000 and n != n_current]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 40 or n_tried >= 5:
            break
        n_tried += 1

        a_try = resample_sequence(best_seq, n_try)
        # Quick LP optimization
        alloc = min(remaining * 0.08, 60.0)
        if alloc > 15:
            a_try = lp_alternating_fast(a_try, time_budget=alloc)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Smooth L-BFGS (10% time) ===
    print(f"\n=== Phase 5: Smooth L-BFGS ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.1, 100.0)
    if p5_budget > 20:
        for T_val in [30.0, 80.0]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            t_budget = min(p5_budget / 2.0, remaining * 0.5)
            if t_budget < 10:
                continue
            a_sm = smooth_lbfgs(best_seq, time_budget=t_budget, T=T_val)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  L-BFGS T={T_val}: obj={best_score:.8f} **")

    # === Phase 6: Final LP Polish (5% time) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    if remaining > 30:
        lp_alloc = min(remaining * 0.5, 80.0)
        a_final = lp_alternating_fast(best_seq, time_budget=lp_alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Coordinate descent polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd = coordinate_refine(best_seq, time_budget=min(remaining * 0.4, 20.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coord descent: obj={best_score:.8f} **")

    # === Phase 7: Diverse restarts with best so far ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        mode = (r + 2) % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.15 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.12, 40.0)
        a_opt = lp_alternating_fast(a_r, time_budget=restart_budget)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
