One more thought: the **simplex projection** step. After computing the gradient step, I need to project back to {a >= 0, sum(a) = 1}. The standard algorithm for this is:
1. Sort v in decreasing order
2. Find rho = max{j : v[j] - (sum_{i<=j} v[i] - 1)/j > 0}
3. theta = (sum_{i<=rho} v[i] - 1) / (rho + 1)
4. w = max(v - theta, 0)

"""
Optimized step function search using smooth-max Adam optimizer with
simplex projection, LP-alternating refinement, and coordinate descent.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def compute_smooth_max_grad(conv, temperature, a):
    """
    Compute gradient of (1/t)*log(sum exp(t*conv(a,a))) w.r.t. a.
    Returns gradient vector of length n.
    """
    n = len(a)
    # Compute softmax weights
    shifted = conv - np.max(conv)
    log_w = temperature * shifted - np.log(np.sum(np.exp(temperature * shifted)))
    w = np.exp(log_w)  # normalized weights, sum to 1
    
    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i in [0,n-1])
    # Use FFT: g = 2 * (w * b) where b is a "negated index" version of a
    # b[x] = a[-x], so b has support on [-(n-1), 0]
    # In circular indexing of length N: b[N-j] = a[j] for j=0..n-1
    # b = [0]*(N-n) + [a[n-1], a[n-2], ..., a[0]]
    
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    
    b_pad = np.zeros(N)
    b_pad[N-n:] = a[::-1]
    
    W = fft(w, N)
    B = fft(b_pad, N)
    g_full = ifft(W * B).real
    grad = 2.0 * g_full[:n]
    
    return grad

def smooth_max_adam(a, time_budget=300.0, t_start=5.0, t_end=50.0, 
                     lr=0.02, beta1=0.9, beta2=0.999, eps=1e-8,
                     verbose=False):
    """
    Optimize smoothed max convolution using Adam with simplex projection.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    it = 0
    
    while time.monotonic() - start < time_budget:
        # Anneal temperature
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        temperature = t_start + (t_end - t_start) * progress
        # Actually, let's use a slower schedule
        temperature = t_start * (t_end / t_start) ** progress
        
        # Compute convolution and smoothed max
        conv = compute_conv(a)
        smax = smoothed_max(conv, temperature)
        
        # Compute gradient
        grad = compute_smooth_max_grad(conv, temperature, a)
        
        # Center gradient (project to tangent of simplex)
        grad = grad - np.mean(grad)
        
        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"      Adam: it={it}, obj={best_obj:.8f}, t={temperature:.1f}, "
                  f"time={elapsed:.0f}s")
        
        # Adaptive learning rate
        if it % 500 == 499:
            lr *= 0.85
            if lr < 0.001:
                lr = 0.001
    
    return best_a

def smoothed_max(conv, temperature):
    """Compute log-sum-exp approximation of max."""
    shifted = conv - np.max(conv)
    return float(np.max(conv) + np.log(np.sum(np.exp(temperature * shifted))) / temperature)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_for_response(a):
    """Solve LP: min M s.t. conv(a, g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)
    
    try:
        from scipy.optimize import linprog
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(np.max(A @ g_val))
                return g_val, M_val
    except Exception:
        pass
    
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(g.value, 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                return g_val, M_val
    except Exception:
        pass
    
    return None, None

def lp_step(a, t=0.5):
    """One LP-alternating step."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    g_lp, M_lp = solve_lp_for_response(a)
    if g_lp is None:
        return a
    
    a_new = (1 - t) * a + t * g_lp
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new

def lp_alternating(a, time_budget=100.0, t=0.3, verbose=False):
    """LP alternating with fixed step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_for_response(a)
        if g_lp is None:
            break
        
        # Try a few step sizes
        best_local = None
        best_local_obj = best_obj
        for t_try in [0.1, t, t * 2, 0.8]:
            t_try = min(t_try, 1.0)
            a_try = (1 - t_try) * a + t_try * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local = a_try.copy()
        
        if best_local is not None and best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local
            a = best_local
            it += 1
            if verbose and it % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP: it={it}, obj={best_obj:.8f}, time={elapsed:.0f}s")
        else:
            break
    
    return best_a

def coordinate_descent(a, time_budget=60.0):
    """Coordinate descent: adjust pairs of coordinates to reduce peak."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    rng = np.random.default_rng(12345)
    
    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        k_star = int(np.argmax(conv))
        improved = False
        
        # Compute contributions to the peak
        contributions = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                contributions[j] = a[j] * a[k_j]
        
        # Try to reduce the peak by moving mass
        sorted_idx = np.argsort(contributions)[::-1][:30]
        
        for i in sorted_idx:
            if time.monotonic() - start > time_budget:
                break
            if a[i] <= 0:
                continue
            
            # Try moving some mass from i to a "good" location
            # Good locations are those that don't contribute to the peak
            for frac in [0.01, 0.03, 0.05]:
                delta = frac * a[i]
                # Find target: position that minimizes contribution to peak
                # a[k_star - target] should be small
                target_scores = np.array([a[max(0, min(n-1, k_star - j))] for j in range(n)])
                # Avoid moving to i itself
                target_scores[i] = np.inf
                target = int(np.argmin(target_scores))
                
                a[i] -= delta
                a[target] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                a = a / s
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[target] -= delta
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    a = a / s
            
            if improved:
                break
        
        if not improved:
            # Random perturbation
            scale = 0.02
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            a = a / s
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            else:
                a = best_a.copy()
        
        if it > 500:
            break
    
    return best_a

def generate_initialization(kind, n, rng):
    """Generate diverse initialization."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n) * 0.5
        edge = max(1, n // 8)
        a[edge:n-edge] = 1.0
        # Smooth edges
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'bell':
        x = np.arange(n) / n
        a = np.exp(-5 * (x - 0.5) ** 2) + 0.1
        return a / np.sum(a)
    else:
        return np.ones(n) / n

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
