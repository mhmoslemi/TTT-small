def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_optimize_improved(a, time_budget=60.0, T=30.0, verbose=False):
    """
    Smooth optimization using log-sum-exp with L-BFGS-B.
    Uses a better gradient that accounts for the normalization.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def obj_grad(x):
        x = np.maximum(x, 1e-12)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x_n = x / s
        conv = compute_conv(x_n)
        max_c = conv.max()
        z = T * (conv - max_c)
        e = np.exp(z)
        sum_e = e.sum()
        obj_val = 2.0 * n * (max_c + np.log(sum_e) / T)
        
        # Gradient with respect to x (before normalization)
        # d/dx_n log-sum-exp = T * softmax(T*(conv - max_c)) @ (d conv / d x_n)
        w = e / sum_e
        
        # d conv[k] / d x_n[i] = sum_j A[k,i] where the bilinear form is x_n^T A_k x_n
        # Actually, conv(x_n, x_n)[k] = sum_i x_n[i] * x_n[k-i] (with bounds)
        # d/dx_n[i] conv[k] = x_n[k-i] + x_n[i-k] (with bounds)
        # But we need d/dx (where x_n = x/s)
        
        # For simplicity, compute gradient w.r.t. x_n directly
        g_n = np.zeros(n)
        for k in range(len(conv)):
            if w[k] > 1e-8:
                g_n += w[k] * T * grad_conv_at_peak(x_n, int(k))
        g_n *= 2.0 * n
        
        # Chain rule: d/dx = d/dx_n * dx_n/dx
        # x_n = x/s, so dx_n/dx = (I - x_n x_n^T)/s
        # For L-BFGS-B, we can just use the gradient w.r.t. x_n as an approximation
        # (since we're optimizing on the simplex approximately)
        return obj_val, g_n

    max_iter = 300
    try:
        res = sp_minimize(obj_grad, a, method='L-BFGS-B',
                          bounds=[(1e-12, None)] * n,
                          options={'maxiter': max_iter, 'maxfun': max_iter * 3, 'disp': False,
                                   'gtol': 1e-6})
        a_opt = np.maximum(res.x, 0)
        s_opt = float(np.sum(a_opt))
        if s_opt > 1e-10:

                if verbose:

    except Exception:
        pass

    return best_a

        sorted_idx = np.argsort(conv)[::-1]
        top_peaks = sorted_idx[:10]
        peak_vals = conv[top_peaks]
        max_peak = peak_vals[0]
        
        if max_peak < 1e-15:
            break
        
        improved = False
        
        # For each top peak, try to reduce it by moving mass
        for k in top_peaks[:5]:
            # Find contributing indices
            i_start = max(0, int(k) - (n - 1))
            i_end = min(n - 1, int(k))
            
            if i_start > i_end:
                continue
            
            # Get contributions
            contribs = []
            for i in range(i_start, i_end + 1):
                j = int(k) - i
                c = a[i] * a[j]
                contribs.append((c, i, j))
            contribs.sort(reverse=True)
            
            # Find non-contributing positions (good targets for mass)
            non_contrib = []
            for i in range(n):
                j = int(k) - i
                if j < 0 or j >= n:
                    non_contrib.append(i)
            
            # Also find positions that contribute to lower peaks
            low_peaks = set()
            for lk in sorted_idx[-10:]:
                for i in range(max(0, int(lk) - (n-1)), min(n-1, int(lk)) + 1):
                    low_peaks.add(i)
            
            non_contrib = [i for i in non_contrib if i in low_peaks]
            if not non_contrib:
                non_contrib = [i for i in non_contrib if i not in set(range(i_start, i_end+1))]
            
            if not non_contrib:
                continue
            
            # Try moving small amounts from top contributors to non-contributing positions
            for ci, (val, i, j) in enumerate(contribs[:5]):
                if val < 1e-10:
                    break
                for target in non_contrib[:5]:
                    if target == i or target == j:
                        continue
                    for frac in [0.005, 0.01, 0.02, 0.05]:
                        delta = frac * a[max(i, j)] if i != j else frac * a[i]
                        if delta < 1e-15 or delta > a[max(i,j)] * 0.5:
                            continue
                        src = max(i, j) if i != j else i
                        a[src] -= delta
                        a[target] += delta
                        a = np.maximum(a, 0)
                        s_try = float(np.sum(a))
                        if s_try < 1e-10:
                            a[src] += delta
                            a[target] -= delta
                            break
                        a = a / s_try
                        new_obj = compute_obj(a)
                        if new_obj < best_obj - 1e-13:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            # Revert
                            a = a * s_try
                            a[src] += delta
                            a[target] -= delta
                            a = np.maximum(a, 0)
                            break
                    if improved:
                        break
                if improved:
                    break
            
            if improved:
                it += 1
                break
        
        if not improved:
            break
    
    return best_a


def multiplicative_updates(a, time_budget=100.0, verbose=False):
    """
    Multiplicative (exponentiated gradient) updates with adaptive rates.
    Works in log-space for numerical stability.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    log_a = np.log(a)
    lr = 1.0 / np.sqrt(n)
    momentum = 0.5
    v = np.zeros(n)
    stall = 0
    rng = np.random.default_rng(999)
    
    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()
        
        if max_val < 1e-15:
            break
        
        # Compute "gradient" using softmax-weighted peaks
        T = 20.0
        z = T * (conv - max_val)
        w = np.exp(z)
        w /= w.sum()
        
        g = np.zeros(n)
        # Only use top contributing peaks for efficiency
        sorted_idx = np.argsort(w)[::-1]
        active = sorted_idx[:max(10, len(conv) // 10)]
        
        for k in active:
            if w[k] > 1e-8:
                g += w[k] * grad_conv_at_peak(a, int(k))
        
        # Exponentiated gradient update with momentum
        v = momentum * v + (1 - momentum) * g
        log_a = log_a - lr * v
        
        # Normalize
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            log_a = np.log(np.maximum(a, 1e-15))
            stall = 0
            lr = max(1e-5, lr * 0.998)
        else:
            stall += 1
            if stall > 50:
                # Reset with perturbation
                scale = 0.05 + 0.02 * (stall // 50)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                log_a = np.log(a)
                v = np.zeros(n)
                lr = max(1e-5, lr * 0.5)
                stall = 0
            else:
                lr = max(1e-5, lr * 0.98)
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"    MU it{it}: obj={best_obj:.8f}, lr={lr:.6f}, el={time.monotonic()-start:.1f}s")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.arange(n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        # Two Gaussian bumps
        c1, c2 = n // 4, 3 * n // 4
        w = n // 8
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    elif mode == 3:
        # Triangle
        a = 1.0 - 2.0 * np.abs(x / (n - 1.0) - 0.5)
    elif mode == 4:
        # Semicircle
        t = 2.0 * x / (n - 1.0) - 1.0
        a = np.sqrt(np.maximum(1 - t**2, 0.01))
    elif mode == 5:
        # Random Fourier
        coeffs = rng.normal(0, 0.5, min(10, n // 2))
        a = np.ones(n)
        for k, c in enumerate(coeffs):
            a += c * np.cos(2 * np.pi * (k + 1) * x / n)
        a = np.maximum(a, 0.01)
    elif mode == 6:
        # Bimodal
        a = np.minimum(x, n - 1 - x) + 1
    elif mode == 7:
        # Exponential decay from center
        c = n // 2
        a = np.exp(-np.abs(x - c) / (n // 4))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a
