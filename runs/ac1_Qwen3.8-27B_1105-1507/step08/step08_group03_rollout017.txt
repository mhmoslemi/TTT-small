from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a. Returns array of length n."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
    return g


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=10.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / float(np.sum(a))
    
    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        A = build_conv_matrix(a)
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue

    if HAVE_SCIPY:
        A = build_conv_matrix(a)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    return None, None

        sum_e = e.sum()
        obj_val = 2.0 * n * (max_c + np.log(sum_e) / T)

    best_obj = compute_obj(a)

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        a = a / a.sum()
    elif mode == 3:

        a = 1.0 + 0.3 * np.abs(2 * x - 1) ** 4
        a = a / a.sum()
    elif mode == 4:

        a = np.sinc(t) ** 2 + 0.01
        a = a / a.sum()
    elif mode == 6:

        a = np.exp(-((x - 0.25) ** 2) / 0.02) + np.exp(-((x - 0.75) ** 2) / 0.02) + 0.01
        a = a / a.sum()

        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a. Returns array of length n."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
    return g


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=10.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / float(np.sum(a))

    if HAVE_CVXPY and n <= 2000:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        A = build_conv_matrix(a)
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue

    if HAVE_SCIPY and n <= 3000:
        A = build_conv_matrix(a)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    return None, None

            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            best_t = best_t_val
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, M={M_lp:.6f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 15:
                noise_scale = min(0.15, 0.03 + 0.02 * (stall // 15))
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                best_t = 0.3

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations with structured patterns."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = 1.0 + 0.5 * (2 * x - 1) ** 2
        a = a / a.sum()
    elif mode == 3:
        a = 1.0 + 0.4 * np.abs(2 * x - 1) ** 4
        a = a / a.sum()
    elif mode == 4:
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.01
        a = a / a.sum()
    elif mode == 5:
        t = (x - 0.5) * 4
        a = np.sinc(t) ** 2 + 0.01
        a = a / a.sum()
    elif mode == 6:
        a = np.exp(-((x - 0.25) ** 2) / 0.02) + np.exp(-((x - 0.75) ** 2) / 0.02) + 0.01
        a = a / a.sum()
    elif mode == 7:
        a = x ** 0.3 + (1 - x) ** 0.3 + 0.1
        a = a / a.sum()
    elif mode == 8:
        a = np.where(np.abs(x - 0.5) > 0.3, 1.5, 0.7)
        a = a / a.sum()
    elif mode == 9:
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses Adam smooth optimization as primary method,
    with subgradient-smooth hybrid and LP alternating as supporting methods.
    Multi-scale n exploration and diverse structured initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 40.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Adam smooth optimization (primary, ~350s total) ===
    print(f"\n=== Phase 1: Adam Smooth Optimization (n={len(best_seq)}) ===")
    adam_configs = [
        (30.0, 0.005), (50.0, 0.004), (100.0, 0.006), (200.0, 0.003),
    ]
    for T_val, lr_val in adam_configs:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        t_budget = min(remaining * 0.1, 90.0)
        if t_budget < 20:
            continue
        a_adam = adam_smooth_optimize(best_seq, time_budget=t_budget, T=T_val, lr=lr_val, verbose=True)
        obj_adam = compute_obj(a_adam)
        if obj_adam < best_score:
            best_score = obj_adam
            best_seq = a_adam.copy()
            print(f"  Adam T={T_val} lr={lr_val}: obj={best_score:.8f} **")
        else:
            print(f"  Adam T={T_val} lr={lr_val}: obj={obj_adam:.8f} (no improvement)")

    # === Phase 2: Subgradient-smooth hybrid (~150s) ===
    print(f"\n=== Phase 2: Subgradient-Smooth Hybrid ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.1, 150.0)
    if p2_budget > 20:
        a_sg = subgradient_smooth_combined(best_seq, time_budget=p2_budget, K=15, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SubGradSmooth K=15: obj={best_score:.8f} **")
        else:
            print(f"  SubGradSmooth K=15: obj={obj_sg:.8f} (no improvement)")

    # === Phase 3: LP Alternating (~250s) ===
    print(f"\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p3_total = min(remaining * 0.2, 250.0)
    n_rounds = 3
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p3_total / (n_rounds - rnd), remaining * 0.4)
        if alloc < 20:
            break

        if rnd > 0:
            a_pert = best_seq * np.exp(rng.normal(0, 0.02 * (rnd + 1), len(best_seq)))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
        else:
            a_pert = best_seq.copy()

        a_lp = lp_alternating_v2(a_pert, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 4: Second round Adam with different params (~150s) ===
    print(f"\n=== Phase 4: Adam Round 2 ===")
    remaining = deadline - time.monotonic()
    adam_configs2 = [(40.0, 0.008), (60.0, 0.002), (150.0, 0.005)]
    for T_val, lr_val in adam_configs2:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        t_budget = min(remaining * 0.08, 60.0)
        if t_budget < 15:
            continue
        a_adam = adam_smooth_optimize(best_seq, time_budget=t_budget, T=T_val, lr=lr_val, verbose=False)
        obj_adam = compute_obj(a_adam)
        if obj_adam < best_score:
            best_score = obj_adam
            best_seq = a_adam.copy()
            print(f"  Adam2 T={T_val} lr={lr_val}: obj={best_score:.8f} **")
        else:
            print(f"  Adam2 T={T_val} lr={lr_val}: obj={obj_adam:.8f}")

    # === Phase 5: Multi-n exploration (~200s) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(200, n_current - 100),
        max(200, n_current - 50),
        n_current + 50,
        n_current + 100,
        n_current + 200,
        400, 500, 600, 700, 800, 900, 1000, 1200, 1400,
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1500]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40 or n_tried >= 8:
            break
        if n_try == n_current:
            continue

        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1

        # Quick Adam optimization
        alloc = min(remaining * 0.08, 40.0)
        if alloc > 10:
            a_try = adam_smooth_optimize(a_try, time_budget=alloc, T=50.0, lr=0.005)

        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc2 = min(remaining * 0.05, 25.0)
            if alloc2 > 10:
                a_try = subgradient_smooth_combined(a_try, time_budget=alloc2, K=10)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts (~100s) ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        mode = (r * 3 + 2) % 10
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.15 + 0.05 * r
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.1, 50.0)
        a_opt = adam_smooth_optimize(a_r, time_budget=restart_budget, T=50.0, lr=0.005)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 25:
        a_last = adam_smooth_optimize(best_seq, time_budget=min(remaining * 0.25, 60.0), T=40.0, lr=0.003)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final Adam: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_sg = subgradient_smooth_combined(best_seq, time_budget=min(remaining * 0.2, 40.0), K=20)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SubGrad: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15 and len(best_seq) <= 1200:
        a_lp = lp_alternating_v2(best_seq, time_budget=min(remaining * 0.3, 30.0))
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
