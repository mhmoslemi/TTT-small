LP-alternating optimization with asymmetric initialization, 1D line search,
convergence-aware iteration, fast LP via HiGHS, subgradient polish, and length sweep.
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 over non-negative sequences.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def conv_full(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_val(a):
    """Compute objective: 2n * max(a*a) / (sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    c = conv_full(a_norm)
    return 2.0 * n * float(np.max(c))


def build_A(a):
    """Build matrix A (2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        end = min(m, j + n)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, verbose=False):
    """
    Solve LP: min M s.t. A(a) @ g <= M, sum(g) = 1, g >= 0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    m = 2 * n - 1

    A = build_A(a)

    # Variables: x = [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g))
            if gs > 1e-10:
                g = g / gs
            M = float(res.x[n])
            return g, M
    except Exception:
        pass

    return None, None

        if s_r > 1e-10:
            a_refined = a_refined / s_r
            o_refined = obj_val(a_refined)
            if o_refined < best_obj_local:
                best_obj_local = o_refined
                best_t = t_best
                best_a = a_refined.copy()

    return best_t, best_a, best_obj_local


def lp_alternate(a_init, time_budget, verbose=True, max_iter=5000):
    """
    LP alternating with 1D line search and perturbation-based restart.
    """
    n = len(a_init)
    a = a_init.copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(42)

    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break

        g, M_lp = solve_lp(a)
        if g is None:
            break

        best_t, a_new, obj_new = line_search(a, g, best_obj)

        if best_t > 0 and obj_new < best_obj - 1e-13:
            a = a_new.copy()
            best_obj = obj_new
            best_a = a.copy()
            stall = 0
            if verbose and (it % 10 == 0 or it < 5):
                elapsed = time.monotonic() - start
                print(f"    LP-iter {it}: obj={best_obj:.8f}, t={best_t:.4f}, "
                      f"M_lp={M_lp:.8f}, time={elapsed:.1f}s")
        else:
            stall += 1
            if stall >= 30:
                # Perturb to escape local optimum
                scale = 0.03 + 0.02 * min(stall // 30, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
            elif stall >= 10:
                # Smaller perturbation
                a = best_a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)

        # Early convergence check
        if stall >= 100:
            if verbose:
                print(f"    LP converged at iter {it}: obj={best_obj:.8f}")
            break

    return best_a


def subgradient_polish(a_init, time_budget, verbose=True):
    """
    Subgradient method with Nesterov acceleration for the max-convolution objective.
    """
    n = len(a_init)
    a = a_init.copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()

    y = a.copy()  # current point
    x = a.copy()  # Nesterov lookahead
    momentum = np.zeros(n)
    beta_mom = 0.6
    lr = 0.03

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        # Compute convex at lookahead
        c_x = conv_full(x)
        max_val = float(np.max(c_x))

        # Active peaks (within 5% of max)
        threshold = max_val * 0.95
        active = np.where(c_x >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c_x))])
        if len(active) > 40:
            top_idx = np.argsort(c_x)[::-1][:40]
            active = top_idx

        # Subgradient: average gradient of active peaks
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            # d(a@a)[k]/da[i] = 2*a[k-i] if 0 <= k-i < n
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_max >= i_min:
                idx = np.arange(i_min, i_max + 1)
                grad[idx] += 2.0 * a[k - idx]
        grad /= len(active)

        # Normalize gradient magnitude
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm

        # Center gradient (project to sum=0 hyperplane)
        grad_centered = grad - np.mean(grad)

        # Momentum update
        momentum = beta_mom * momentum - lr * grad_centered
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
        else:
            y_new = y_new / s_new

        # Nesterov
        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        x = y_new + (1 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        y = y_new

        obj = obj_val(y)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y.copy()

        # Adaptive learning rate
        if it % 500 == 499:
            lr = max(lr * 0.92, 0.001)

        if verbose and it % 2000 == 0:
            print(f"    SG iter {it}: obj={best_obj:.8f}, lr={lr:.5f}")

    return best_a


def coordinate_descent(a_init, time_budget, verbose=True):
    """
    Fast coordinate/pairwise adjustment to reduce peak.
    """
    n = len(a_init)
    a = a_init.copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()
    rng = np.random.default_rng(99)

    # Identify peak lag
    c = conv_full(a)
    k_star = int(np.argmax(c))

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        c = conv_full(a)
        k_star = int(np.argmax(c))
        peak_val = c[k_star]

        # Find positions contributing most to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = 2.0 * a[i] * a[j]
            else:
                contrib[i] = 0.0

        # Try moving mass from high-contribution positions to low ones
        sorted_idx = np.argsort(contrib)[::-1]
        improved = False

        for i in sorted_idx[:15]:
            if time.monotonic() - start > time_budget:
                break
            if a[i] < 1e-8:
                continue

            # Target: position that minimizes contribution at k_star
            # Moving mass from i to j reduces contrib at k_star by ~2*a[i]*a[k_star-i]
            # and adds ~2*a[j]*a[k_star-j]
            # Want k_star - j to be where a is small
            j = (k_star - int(i) + n // 2) % n
            if j == i:
                j = (i + 1) % n

            for frac in [0.005, 0.01, 0.02, 0.05]:
                delta = frac * a[i]
                if delta <= 0 or delta >= a[i]:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s2 = float(np.sum(a))
                if s2 < 1e-10:
                    a[i] += delta
                    a[j] -= delta
                    continue
                a = a / s2
                o = obj_val(a)
                if o < best_obj - 1e-14:
                    best_obj = o
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    a = a / s2

            if improved:
                break

        if not improved:
            # Random perturb
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = 0.01 * a[i]
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                o = obj_val(a)
                if o < best_obj:
                    best_obj = o
                    best_a = a.copy()

    return best_a


def generate_asymmetric_init(n, kind, rng):
    """Generate asymmetric initializations."""
    if kind == 'ramp':
        a = np.arange(1, n + 1, dtype=np.float64)
        return a / np.sum(a)
    elif kind == 'ramp_reverse':
        a = np.arange(n, 0, -1, dtype=np.float64)
        return a / np.sum(a)
    elif kind == 'exponential':
        a = np.exp(np.linspace(0, -2, n))
        return a / np.sum(a)
    elif kind == 'exponential_rev':
        a = np.exp(np.linspace(-2, 0, n))
        return a / np.sum(a)
    elif kind == 'two_bump':
        a = np.ones(n) * 0.1
        a[n // 4] = 1.0
        a[3 * n // 4] = 0.5
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'quadratic_ramp':
        x = np.arange(n, dtype=np.float64) / n
        a = x * x + 0.01
        return a / np.sum(a)
    elif kind == 'sqrt_ramp':
        x = np.arange(n, dtype=np.float64) / n
        a = np.sqrt(x + 0.001)
        return a / np.sum(a)
    elif kind == 'sine_asym':
        x = np.arange(n, dtype=np.float64) / n
        a = 0.5 + 0.5 * np.sin(np.pi * x - 0.5) + 0.1
        return a / np.sum(a)
    elif kind == 'random_asym':
        a = np.exp(rng.normal(0, 1, n) * np.linspace(0.5, 1.5, n))
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: asymmetric inits + LP-alternating with line search +
    subgradient polish + coordinate descent + length sweep.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 40.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = obj_val(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = obj_val(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting: n={n_parent}, best_score={best_score:.8f}")

    # === Phase 1: Try asymmetric initializations with quick LP (120s) ===
    print(f"\n=== Phase 1: Asymmetric Init + Quick LP ===")
    remaining = deadline - time.monotonic()
    phase1_budget = min(120.0, remaining * 0.1)
    per_init = max(8.0, phase1_budget / 12)

    inits = [
        ('ramp', n_parent),
        ('ramp_reverse', n_parent),
        ('exponential', n_parent),
        ('exponential_rev', n_parent),
        ('two_bump', n_parent),
        ('quadratic_ramp', n_parent),
        ('sqrt_ramp', n_parent),
        ('sine_asym', n_parent),
        ('random_asym', n_parent),
        ('random_asym', n_parent),
        ('random_asym', n_parent),
    ]

    for kind, n_try in inits:
        remaining = deadline - time.monotonic()
        if remaining < phase1_budget / len(inits) + 10:
            break
        a_init = generate_asymmetric_init(n_try, kind, rng)
        obj_init = obj_val(a_init)
        tag = f"  Init '{kind}' (n={n_try}): init_obj={obj_init:.6f}"

        # Quick LP to see if it improves
        alloc = min(per_init, remaining * 0.3)
        a_opt = lp_alternate(a_init, time_budget=alloc, verbose=False)
        obj_opt = obj_val(a_opt)
        tag += f" -> lp_obj={obj_opt:.6f}"

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            tag += " ** NEW BEST **"
        print(tag)

    # Also try parent with slight asymmetry perturbation
    for scale in [0.05, 0.1, 0.2]:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        a_pert = parent * np.exp(rng.normal(0, scale, n_parent)) if parent is not None \
            else best_seq * np.exp(rng.normal(0, scale, n_parent))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / np.sum(a_pert)
        alloc = min(30.0, remaining * 0.15)
        a_opt = lp_alternate(a_pert, time_budget=alloc, verbose=False)
        obj_opt = obj_val(a_opt)
        tag = f"  Perturbed parent (scale={scale}): obj={obj_opt:.6f}"
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            tag += " ** NEW BEST **"
        print(tag)

    a = best_seq.copy()
    n_best = len(a)
    print(f"  Phase 1 done: best_score={best_score:.8f}, n={n_best}")

    # === Phase 2: Main LP alternating (long run, 700s) ===
    print(f"\n=== Phase 2: Main LP Alternating (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    phase2_budget = min(700.0, remaining * 0.55)

    # Run LP alternating for extended time
    a_opt = lp_alternate(a, time_budget=phase2_budget, verbose=True)
    obj_opt = obj_val(a_opt)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        print(f"  ** Phase 2: obj={best_score:.8f} **")
    else:
        print(f"  Phase 2: obj={obj_opt:.8f}")
    a = best_seq.copy()

    # === Phase 3: Subgradient polish (150s) ===
    print(f"\n=== Phase 3: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        sg_budget = min(150.0, remaining * 0.15)
        a_sg = subgradient_polish(a, time_budget=sg_budget, verbose=True)
        obj_sg = obj_val(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a = a_sg.copy()
            print(f"  ** Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    # === Phase 4: Coordinate descent (100s) ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        cd_budget = min(100.0, remaining * 0.1)
        a_cd = coordinate_descent(a, time_budget=cd_budget, verbose=True)
        obj_cd = obj_val(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a = a_cd.copy()
            print(f"  ** CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 5: Second LP round from best (150s) ===
    print(f"\n=== Phase 5: Second LP Round ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        lp2_budget = min(150.0, remaining * 0.2)
        a = best_seq * np.exp(rng.normal(0, 0.02, len(best_seq)))
        a = np.maximum(a, 0)
        a = a / np.sum(a)
        a_opt2 = lp_alternate(a, time_budget=lp2_budget, verbose=True)
        obj_opt2 = obj_val(a_opt2)
        if obj_opt2 < best_score:
            best_score = obj_opt2
            best_seq = a_opt2.copy()
            print(f"  ** LP2: obj={best_score:.8f} **")
        else:
            print(f"  LP2: obj={obj_opt2:.8f}")

    # === Phase 6: Length sweep (150s) ===
    print(f"\n=== Phase 6: Length Sweep ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(100, n_best - 100),
        max(100, n_best - 50),
        max(100, n_best - 200),
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        n_best + 400,
    ]))
    n_options = [n for n in n_options if 200 <= n <= 3000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(40.0, remaining * 0.3)
        a_opt = lp_alternate(a_try, time_budget=alloc, verbose=False)
        obj_try = obj_val(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final polish with remaining time ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Alternate between LP and subgradient
        a_final = best_seq.copy()
        for round_idx in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            if round_idx % 2 == 0:
                alloc = min(remaining * 0.3, 60.0)
                a_new = lp_alternate(a_final, time_budget=alloc, verbose=False)
            else:
                alloc = min(remaining * 0.2, 40.0)
                a_new = subgradient_polish(a_final, time_budget=alloc, verbose=False)

            obj_new = obj_val(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_final = a_new.copy()
                print(f"  Final R{round_idx}: obj={best_score:.8f} **")
            else:
                # Perturb
                a_final = best_seq * np.exp(rng.normal(0, 0.01, len(best_seq)))
                a_final = np.maximum(a_final, 0)
                a_final = a_final / np.sum(a_final)
                print(f"  Final R{round_idx}: obj={obj_new:.8f} (no improvement)")

        # Final coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.4, 30.0))
            obj_cd = obj_val(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(500)
        best_score = obj_val(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif np.isinf(val):
        result = [1.0] * 500
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
