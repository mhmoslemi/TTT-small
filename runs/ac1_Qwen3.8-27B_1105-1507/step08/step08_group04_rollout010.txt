import numpy as np

def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_grad_smoothed(a, beta):
    """Compute gradient of 2n * (1/beta) * logsumexp(beta * conv(a,a)).
    Returns (centered_gradient, smoothed_max_value)."""
    n = len(a)
    conv = compute_conv(a)

    # Numerically stable softmax weights
    max_conv = float(np.max(conv))
    shifted = beta * (conv - max_conv)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    weights = exp_shifted / sum_exp

    # Smoothed max value
    smoothed_max = max_conv + np.log(sum_exp) / beta

    # Compute cross-correlation: cross[j] = sum_k weights[k] * a[k-j]
    n_fft = 1
    while n_fft < 3 * n:
        n_fft <<= 1

    A_fft = rfft(a, n_fft)
    W_fft = rfft(weights, n_fft)
    cross = irfft(np.conj(A_fft) * W_fft, n_fft)[:n]

    # Correction for diagonal term: when k=2j, the derivative is a[j] not 2*a[j]
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    correction = np.zeros(n)
    correction[valid] = a[valid] * weights[idx[valid]]

    grad = 2.0 * n * (2.0 * cross - correction)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A


def solve_lp(a):
    """Solve: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_value) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    # Try HiGHS via scipy (fast for this problem size)
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 30})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])

        pass

    # Try cvxpy

        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    best_obj = compute_obj_normalized(a_parent)

    # === Phase 1: Smoothed gradient optimization (primary new method) ===
    print(f"\n=== Phase 1: Smoothed Gradient (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 300.0)
    if p1_budget > 30:
        a_p1 = smoothed_gradient_optimize(a_parent, time_budget=p1_budget,
                                           beta_start=30, beta_end=800,
                                           verbose=True, seed=seed)
        obj_p1 = compute_obj_normalized(a_p1)
        if obj_p1 < best_obj:
            best_obj = obj_p1
            best_a = a_p1.copy()
            print(f"  Phase 1 result: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 1 result: obj={obj_p1:.8f}")

    # === Phase 2: LP alternation from smoothed gradient result ===
    print(f"\n=== Phase 2: LP Alternating (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 250.0)
    if p2_budget > 20:
        a_p2 = lp_alternate(best_a, time_budget=p2_budget, verbose=True, seed=seed + 1)
        obj_p2 = compute_obj_normalized(a_p2)
        if obj_p2 < best_obj:
            best_obj = obj_p2
            best_a = a_p2.copy()
            print(f"  Phase 2 result: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 2 result: obj={obj_p2:.8f}")

    # === Phase 3: Smoothed gradient again (from LP result) ===
    print(f"\n=== Phase 3: Smoothed Gradient Round 2 ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 20:
        a_p3 = smoothed_gradient_optimize(best_a, time_budget=p3_budget,
                                           beta_start=100, beta_end=2000,
                                           verbose=True, seed=seed + 2)
        obj_p3 = compute_obj_normalized(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3.copy()
            print(f"  Phase 3 result: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 3 result: obj={obj_p3:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([
        max(200, n_best - 300), max(200, n_best - 150),
        n_best,
        n_best + 100, n_best + 200, n_best + 400, n_best + 600, n_best + 1000,
    ]))
    n_options = [x for x in n_options if 200 <= x <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == len(best_a):
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try

        # Quick optimization: smoothed gradient + brief LP
        alloc_sg = min(30.0, remaining * 0.3)
        alloc_lp = min(20.0, remaining * 0.2)

        if alloc_sg > 10:
            a_opt = smoothed_gradient_optimize(a_try, time_budget=alloc_sg,
                                                beta_start=50, beta_end=500, seed=seed + n_try)
        else:
            a_opt = a_try

        if alloc_lp > 10 and time.monotonic() < deadline - 15:
            a_opt2 = lp_alternate(a_opt, time_budget=alloc_lp, seed=seed + n_try + 100)
            obj_opt2 = compute_obj_normalized(a_opt2)
            if obj_opt2 < compute_obj_normalized(a_opt):
                a_opt = a_opt2

        obj_try = compute_obj_normalized(a_opt)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse initializations at best n ===
    print(f"\n=== Phase 5: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    for d in range(12):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_div = diverse_initialization(n_best, rng)
        alloc = min(35.0, remaining * 0.2)

        # Try both smoothed gradient and LP from this init
        a_opt_sg = smoothed_gradient_optimize(a_div, time_budget=alloc * 0.5,
                                               beta_start=20, beta_end=300, seed=seed + 1000 + d)
        obj_sg = compute_obj_normalized(a_opt_sg)

        a_opt_lp = lp_alternate(a_div, time_budget=alloc * 0.4, seed=seed + 2000 + d)
        obj_lp = compute_obj_normalized(a_opt_lp)

        if obj_sg < obj_lp:
            a_opt = a_opt_sg
            obj_opt = obj_sg
        else:
            a_opt = a_opt_lp
            obj_opt = obj_lp

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            print(f"  ** Diverse {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Diverse {d}: obj={obj_opt:.8f}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # LP polish
        alloc_lp = min(remaining * 0.35, 80.0)
        a_final = lp_alternate(best_a, time_budget=alloc_lp, verbose=True, seed=seed + 5000)
        obj_final = compute_obj_normalized(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final LP: obj={best_obj:.8f} **")

    # Smoothed gradient micro-polish with high beta
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_sg = min(remaining * 0.3, 50.0)
        a_sg = smoothed_gradient_optimize(best_a, time_budget=alloc_sg,
                                           beta_start=200, beta_end=5000,
                                           seed=seed + 6000)
        obj_sg = compute_obj_normalized(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            print(f"  Final SmoothGrad: obj={best_obj:.8f} **")

    # One more LP if time allows
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_last = min(remaining * 0.5, 30.0)
        a_last = lp_alternate(best_a, time_budget=alloc_last, seed=seed + 7000)
        obj_last = compute_obj_normalized(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last.copy()
            print(f"  Last LP: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = 2.0

    # Scale to keep values in [0, 1000]
    max_val = float(np.max(best_a))
    if max_val > 0:
        scale = min(1.0, 1000.0 / max_val)
        result = [float(max(0.0, x * scale)) for x in best_a]
    else:
        result = [1.0] * 100

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
