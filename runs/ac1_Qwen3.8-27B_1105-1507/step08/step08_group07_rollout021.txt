from scipy.fft import rfft, irfft

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:

    """Project onto {x >= 0, sum(x) = target}."""
    v = np.maximum(v, 0)
    s = float(np.sum(v))
    if s < 1e-30:
        return np.full(len(v), target / len(v))
    return v * (target / s)

def gerchberg_saxton(a, time_budget=100.0, verbose=False):
    """
    Gerchberg-Saxton spectral flattening.
    Alternates between:
    - Time domain: project onto non-negative, normalize
    - Frequency domain: flatten magnitude spectrum
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    if np.sum(a) < 1e-20:
        return a
    a = a / np.sum(a)
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    X = rfft(a, size)
    prev_obj = best_obj
    stall = 0
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        # Frequency domain: flatten magnitude
        mag = np.abs(X)
        phase = np.angle(X)
        # Target magnitude from Parseval
        target_mag = np.sqrt(np.sum(a**2))  # sqrt of total power
        
        # Blend: partially flatten (relaxation factor helps convergence)
        blend = 0.5  # 1.0 = full flattening, <1 = relaxed
        new_mag = target_mag + blend * (mag - target_mag)
        # Actually, let's do: new_mag = (1-blend)*mag + blend*target_mag
        new_mag = (1.0 - blend) * mag + blend * target_mag
        
        X_new = new_mag * np.exp(1j * phase)
        
        # Time domain: IFFT, project
        a_new = irfft(X_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-20:
            break
        a_new = a_new / s
        
        a = a_new
        X = rfft(a, size)
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Try different blend
                blend = 1.0 - blend  # Toggle
                stall = 0
        
        if verbose and it % 1000 == 0:
            print(f"    GS: it={it}, obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    
    return best_a

def lse_gradient_descent(a, time_budget=200.0, verbose=False):
    """
    Log-sum-exp smooth gradient descent.
    Minimizes eps*log(sum exp(c[k]/eps)) which approximates max c[k].
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    if np.sum(a) < 1e-20:
        return a
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    # Start with larger eps, decrease over time
    eps = 0.05 / n
    lr = 1.0 / (2.0 * n)  # Initial learning rate
    momentum_buf = np.zeros(n)
    momentum = 0.9
    start = time.monotonic()
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_conv = np.max(conv)
        cur_obj = 2.0 * n * max_conv
        
        # Softmax weights
        shifted = (conv - max_conv) / eps
        exp_vals = np.exp(shifted)
        weights = exp_vals / np.sum(exp_vals)
        
        # Gradient of LSE approximation
        g = np.zeros(n)
        # For efficiency, only process significant weights
        threshold = 1e-8
        significant = np.where(weights > threshold)[0]
        for k in significant:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = k - np.arange(j_start, j_end + 1)
                g[j_start:j_end+1] += weights[k] * 2.0 * a[idx]
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-30:
            break
        
        # Momentum (heavy ball)
        momentum_buf = momentum * momentum_buf - lr * g
        a_new = a + momentum_buf
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = a_new  # Still move to explore
            # If stuck for too long, reduce lr and reset momentum
            if it % 1000 == 999:
                lr *= 0.8
                momentum_buf = np.zeros(n)
        
        # Decrease eps over time (annealing)
        if it % 500 == 0:
            eps *= 0.9
        
        if verbose and it % 5000 == 0:
            print(f"    LSE: it={it}, obj={best_obj:.8f}, eps={eps:.6f}, t={time.monotonic()-start:.1f}s")
    
    return best_a
