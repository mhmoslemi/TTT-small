2. Smooth L-BFGS refinement (p-approximation of max)
3. Multi-n exploration
4. Targeted local search

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_cross(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):

    if s < 1e-10:

    # grad_a_j = 2 * sum_{m=0}^{n-1} w[j+m] * a[m] (where w[k]=0 for k>=2n-1)
    
    # Pad w to length 3n-1 for safe indexing
    w_padded = np.zeros(3 * n - 1)
    w_padded[:2*n-1] = w
    
    # This is like a "sliding dot product" - for each j, dot(w[j:j+n], a)
    # We can compute this efficiently
    for j in range(n):
        grad_a[j] = 2.0 * float(np.dot(w_padded[j:j+n], a))
    
    coeff = 2.0 * n * S**(1.0/p - 1.0)
    grad_a = coeff * grad_a
    
    # Now convert from grad_a to grad_x:
    # x -> a = x/S where S = sum(x)
    # d(a_j)/d(x_k) = delta_{jk}/S - x_j/(S^2)
    # d(val)/d(x_k) = sum_j d(val)/d(a_j) * (delta_{jk}/S - x_j/S^2)
    # = grad_a_k / S - (x_k/S^2) * sum_j grad_a_j
    sum_grad = float(np.sum(grad_a))
    S_x = float(np.sum(x))
    grad_x = grad_a / S_x - x * sum_grad / (S_x * S_x)
    
    return val, grad_x


def smooth_optimize(a_init, p_list=[4, 8, 16, 32], time_budget=300):
    """Run smooth L-BFGS optimization with increasing p."""
    start = time.monotonic()
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for p in p_list:
        if time.monotonic() - start > time_budget:
            break
        
        x0 = a * np.sum(a)  # = a (since sum(a)=1)... actually just use a as x0
        x0 = a.copy()  # start from current a (which sums to 1)
        
        def fun(x):
            x_pos = np.maximum(x, 1e-10)
            return smooth_objective_and_grad(x_pos, p)
        
        # Actually, let's handle the constraint differently
        # We want x >= 0, so use L-BFGS-B with bounds
        x0 = a.copy()
        
        def obj_fun(x):
            x_pos = np.maximum(x, 1e-15)
            v, _ = smooth_objective_and_grad(x_pos, p)
            return v
        
        def obj_grad(x):
            x_pos = np.maximum(x, 1e-15)
            _, g = smooth_objective_and_grad(x_pos, p)
            return g
        
        bounds = [(1e-10, None)] * len(x0)
        
        res = minimize(obj_fun, x0, jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 200, 'gtol': 1e-8,
                                               'ftol': 1e-10})
        
        x_opt = np.maximum(res.x, 1e-15)
        a_new = x_opt / np.sum(x_opt)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

            print(f"    p={p}: obj={best_obj:.8f} (L-BFGS, iters={res.nit})")
        else:
            print(f"    p={p}: obj={obj_new:.8f} (no improvement, best={best_obj:.8f})")
    
    return best_a


def lp_with_optimal_t(a, time_budget=200):
    """LP alternating with optimal step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        g, M = solve_lp(a)
        if g is None:
            break
        
        # Find optimal t
        t_opt = find_optimal_t(a, g)
        
        # Try a few t values around t_opt
        best_cand = None
        best_cand_obj = np.inf
        for t in [t_opt, max(0, t_opt - 0.1), min(1, t_opt + 0.1), 0.5 * t_opt]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_cand_obj:
                best_cand_obj = obj_try
                best_cand = a_try.copy()
        
        if best_cand is not None and best_cand_obj < best_obj - 1e-12:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            a = best_cand.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Perturb and restart
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
    
    return best_a


def coordinate_descent(a, time_budget=60):
    """Coordinate descent: try adjusting each coordinate."""
    n = len(a)
    a = a.copy()
    a = a / np.sum(a)
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    
    improved = True
    while improved and time.monotonic() - start < time_budget:
        improved = False
        # Random order
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            for factor in [0.5, 0.7, 1.3, 1.5, 2.0, 0.3]:
                a[j] *= factor
                if a[j] < 0:
                    a[j] = 0
                a_norm = a / np.sum(a)
                obj = compute_obj(a_norm)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    a = a_norm.copy()
                    improved = True
                    break
                else:
                    a[j] /= factor  # undo
    
    return a / np.sum(a)
