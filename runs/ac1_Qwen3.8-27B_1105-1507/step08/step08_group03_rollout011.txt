from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)
    
    # Formulation: min M
    # Variables: g (n) + M (1) = n+1
    # Constraints: A@g - M <= 0 (2n-1), sum(g) = 1 (1)
    # Bounds: g >= 0, M >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True})

The least-squares solution to "make the convolution as flat as possible" can be found by minimizing:

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Cross convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt, dual_y) or (None, None, None).
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        # Get dual variables
                        dual_y = None
                        try:
                            if constraints[0].dual_value is not None:
                                dual_y = np.abs(constraints[0].dual_value)
                        except Exception:
                            pass
                        return g_val / s, float(M.value), dual_y
            except Exception:
                continue

    # Fallback to scipy HiGHS
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            dual_y = None
            try:
                if res.ineqlin is not None and len(res.ineqlin) > 0:
                    dual_y = np.abs(res.ineqlin)
            except Exception:
                pass
            return g_val / s, float(res.x[n]), dual_y
    return None, None, None


def solve_lp_maxsum(a, M_target):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M_target, g >= 0."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M_target]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.status == 0:
        g_val = np.maximum(res.x, 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None


def grad_conv_at_peak(a, k):
    """Gradient of (a*g)[k] with respect to g, evaluated at g=a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1]
    return g


def fine_line_search(a, g, n_ts=30):
    """
    Fine grid line search: find best t in (0,1] for (1-t)*a + t*g.
    Returns (best_t, best_a, best_obj).
    """
    best_obj = np.inf
    best_t = 0.0
    best_a = a.copy()

    # Log-spaced and linear-spaced t values
    ts_log = np.logspace(-3, 0, 10)  # 0.001 to 1.0
    ts_lin = np.linspace(0.01, 1.0, n_ts)
    ts = np.unique(np.concatenate([ts_log, ts_lin]))
    ts = ts[ts > 0]

    for t in ts:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        conv = compute_conv(a_try)
        obj = 2.0 * len(a) * float(np.max(conv))
        if obj < best_obj:
            best_obj = obj
            best_t = t
            best_a = a_try.copy()

    return best_t, best_a, best_obj

def nesterov_subgradient(a, time_budget=60.0, K=5, verbose=False):
    """Nesterov-accelerated subgradient method on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 100
    rng = np.random.default_rng(999)

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        # Softmax weighting over top peaks
        temps = [1.0, 3.0, 10.0]
        all_w = np.zeros(2 * n - 1)
        for temp in temps:
            w = np.exp((conv - max_val) / (temp * max_val + 1e-15))
            w /= w.sum()
            all_w += w
        all_w /= len(temps)

        # Compute subgradient from softmax weights
        g = np.zeros(n)
        active_threshold = np.percentile(all_w, 90)
        for k_idx in range(2 * n - 1):
            if all_w[k_idx] > active_threshold:
                g += all_w[k_idx] * grad_conv_at_peak(y, k_idx)

        lr = lr0 / t_k
        a_new = y - lr * g
        # Project onto simplex
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                a_new = best_a * np.exp(rg.normal(0, 0.1, n)

                a_new = np.maximum(a_new, 0)
                s_new = float(np.sum(a_new))
                if s_new > 1e-10:
                    a_new = a_new / s_new
                best_a = a_new.copy()
                best_obj = compute_obj(a_new)
                y = a_new.copy()
                a_prev = a_new.copy()
                y_new = a_new.copy()
                t_k = 1.0
                stall = 0
                patience = int(patience * 1.5)

        a_prev = a_new.copy()
        y = y_new.copy()
        t_k = t_new
        it += 1

    return best_a


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def mirror_descent(a, time_budget=60.0, verbose=False):
    """Mirror descent with adaptive step size and multi-temperature softmax."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    momentum = 0.3
    v = np.zeros(n)
    stall = 0
    rng = np.random.default_rng(42)

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()
        if max_val < 1e-15:
            break

        # Multi-temperature softmax weights
        temps = [0.5, 1.0, 3.0, 10.0]
        all_w = np.zeros(2 * n - 1)
        for temp in temps:
            w = np.exp((conv - max_val) / (temp * max_val + 1e-15))
            w /= w.sum()
            all_w += w
        all_w /= len(temps)

        # Subgradient
        g = np.zeros(n)
        threshold = np.percentile(all_w, 92)
        for k_idx in range(2 * n - 1):
            if all_w[k_idx] > threshold:
                g += all_w[k_idx] * grad_conv_at_peak(a, k_idx)

        v = momentum * v + g
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = max(1e-6, lr * 0.999)
            momentum = min(0.7, momentum + 0.001)
        else:
            stall += 1
            if stall > 200:
                a = best_a * np.exp(rg.normal(0, 0.12, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                v = np.zeros(n)
                momentum = 0.3
                lr = max(1e-5, lr * 0.5)
                stall = 0
            else:
                lr = max(1e-6, lr * 0.995)

        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps
        c1, c2 = 0.25, 0.75
        a = np.exp(-40*(x-c1)**2) + np.exp(-40*(x-c2)**2) + 0.01
        a = a / a.sum()
    elif mode == 3:
        # Triangle
        a = 1.0 - 2.0 * np.abs(x - 0.5) + 0.01
        a = a / a.sum()
    elif mode == 4:
        # Arcsine-like
        a = 1.0 / (np.sqrt(0.01 + 4.0 * x * (1 - x)) + 0.1)
        a = a / a.sum()
    elif mode == 5:
        # Sine mixture
        a = 1.0 + 0.4 * np.sin(4 * np.pi * x) + 0.2 * np.sin(8 * np.pi * x + 1)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # U-shaped
        a = (x - 0.5)**2 + 0.05
        a = a / a.sum()
    elif mode == 7:
        # Plateau
        a = np.ones(n) * 0.1
        center = n // 2
        width = n // 4
        a[center - width:center + width] = 1.0
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

            print(f"  Nesterov: obj={obj_ns:.8f}")
        # Follow up with LP
        remaining = deadline - time.monotonic()
        if remaining > 50:
            a_lp2 = lp_alternating(best_seq, time_budget=min(remaining * 0.15, 150.0), verbose=True)
            obj_lp2 = compute_obj(a_lp2)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  LP after NS: obj={best_score:.8f} **")

    # === Phase 3: Mirror Descent ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 100.0)
    if p3_budget > 20:
        a_md = mirror_descent(best_seq, time_budget=p3_budget, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Mirror: obj={best_score:.8f} **")
        else:
            print(f"  Mirror: obj={obj_md:.8f}")
        # Follow up with LP
        remaining = deadline - time.monotonic()
        if remaining > 50:
            a_lp3 = lp_alternating(best_seq, time_budget=min(remaining * 0.15, 150.0), verbose=True)
            obj_lp3 = compute_obj(a_lp3)
            if obj_lp3 < best_score:
                best_score = obj_lp3
                best_seq = a_lp3.copy()
                print(f"  LP after MD: obj={best_score:.8f} **")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Focus on n values near current and some larger ones
    n_options = sorted(set([
        max(300, n_current - 100),
        n_current,
        n_current + 50,
        n_current + 100,
        n_current + 200,
        n_current + 300,
        500, 600, 700, 900, 1000, 1100, 1200, 1400
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50 or n_tried >= 8:
            break
        if n_try == n_current:
            continue

        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1

        # Quick optimization: LP
        alloc = min(remaining * 0.12, 80.0)
        if alloc > 20:
            a_try = lp_alternating(a_try, time_budget=alloc)

        obj_try = compute_obj(a_try)
        status = " $" if obj_try < best_score else ""
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
            # Try Nesterov on this n if time allows
            remaining = deadline - time.monotonic()
            if remaining > 40 and obj_try - best_score < 0.02:
                a_ns = nesterov_subgradient(a_try, time_budget=min(remaining * 0.08, 40.0), K=5)
                obj_ns = compute_obj(a_ns)
                if obj_ns < best_score:
                    best_score = obj_ns
                    best_seq = a_ns.copy()
                    print(f"  ** n={n_try} NS: obj={best_score:.8f} **")

    # === Phase 5: Diversified Restarts ===
    print(f"\n=== Phase 5: Diversified Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = r
        a_r = diverse_init(n_best, rg, mode=mode)
        # Blend with best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.12, 80.0)
        a_opt = lp_alternating(a_r, time_budget=restart_budget)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final Intensive LP Polish ===
    print(f"\n=== Phase 6: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Multiple short LP rounds with different perturbations
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            if rnd > 0:
                a_pert = best_seq * np.exp(rg.normal(0, 0.01 * rnd, len(best_seq)))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
            else:
                a_pert = best_seq.copy()

            a_lp = lp_alternating(a_pert, time_budget=min(remaining * 0.25, 100.0), verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP {rnd}: obj={best_score:.8f} **")

    # === Phase 7: Quick Nesterov + LP on best ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_ns = nesterov_subgradient(best_seq, time_budget=min(remaining * 0.2, 50.0), K=10)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Final NS: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_lp = lp_alternating(best_seq, time_budget=min(remaining * 0.4, 60.0))
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"Target: 1.5027, Previous: 1.506329, Achieved: {best_score:.8f}")

    return result
