def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x) = target_sum, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    # Scale to make the problem standard (sum = 1)
    v_scaled = v / target_sum if target_sum > 0 else v
    # Standard simplex projection (sum = 1, x >= 0)
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    proj = np.maximum(v_scaled - theta, 0)
    return proj * target_sum


def accelerated_subgradient(a, time_budget=100.0, verbose=False, top_k=20, 
                            threshold_frac=0.95, initial_lr=0.01):
    """
    Nesterov-accelerated subgradient method for min max_k (a*a)[k] s.t. sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    target_sum = s  # preserve the sum
    a = a / target_sum  # normalize to sum=1 for optimization
    
    best_a = a.copy()
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    
    y = a.copy()  # current point
    x = a.copy()  # Nesterov lookahead
    momentum = np.zeros(n)
    
    # Adaptive parameters
    lr = initial_lr
    patience = 0
    best_patience_obj = best_obj
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Evaluate at lookahead point
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-12:
            x = x / sx
            conv_x = compute_conv(x)
        else:
            conv_x = conv
        
        max_val = float(np.max(conv_x))
        threshold = max_val * threshold_frac
        active = np.where(conv_x >= threshold)[0]
        if len(active) > top_k:
            active = active[np.argsort(conv_x[active])[::-1][:top_k]]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv_x))])
        
        # Compute subgradient at x
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            indices = np.arange(i_min, i_max + 1)
            grad[indices] += 2.0 * x[k - indices]
        grad /= len(active)
        
        # Center the gradient (project to hyperplane sum=0 for the simplex constraint)
        grad = grad - np.mean(grad)
        
        # Nesterov momentum step
        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        momentum = (it / (it + 1 + t_k)) * momentum - lr * grad
        y_new = y + momentum
        y_new = project_simplex(y_new, 1.0)
        
        # Update lookahead
        x = y_new + (1.0 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-12:
            x = x / sx
        
        y = y_new
        it += 1
        
        # Evaluate and track best
        if it % 5 == 0 or it < 20:
            conv_y = compute_conv(y)
            obj = 2.0 * n * float(np.max(conv_y))
            if obj < best_obj:
                best_obj = obj
                best_a = y.copy()
                patience = 0
            else:
                patience += 1
            
            if patience > 500:
                # Decay learning rate
                lr *= 0.7
                patience = 0
                if verbose:
                    print(f"    SG it={it}: obj={best_obj:.8f}, lr={lr:.6f}")
        
        # Periodic perturbation
        if it % 2000 == 1999:
            rng = np.random.default_rng(it)
            scale = 0.01
            y = best_a * np.exp(rng.normal(0, scale, n))
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy > 1e-12:
                y = y / sy
            x = y.copy()
            momentum = np.zeros(n)
    
    return best_a * target_sum


def fw_alternation(a, time_budget=100.0, verbose=False, inner_iters=50):
    """
    Frank-Wolfe based alternating minimization.
    Alternates between a and b, solving min max_k (a*b)_k with FW at each step.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    target_sum = s
    a = a / target_sum
    b = a.copy()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    outer_it = 0
    
    while time.monotonic() - start < time_budget:
        # Solve: min max_k (b * g)_k s.t. sum(g)=1, g>=0 using FW
        g = b.copy()
        for inner in range(inner_iters):
            # Compute (b*g) via FFT
            size = 1
            while size < 2 * n - 1:
                size <<= 1
            fb = rfft(b, size)
            fg = rfft(g, size)
            conv_bg = irfft(fb * fg, size)[:2 * n - 1]
            
            # Find active peaks
            max_val = float(np.max(conv_bg))
            threshold = max_val * 0.95
            active = np.where(conv_bg >= threshold)[0]
            if len(active) > 30:
                active = active[np.argsort(conv_bg[active])[::-1][:30]]
            
            # Gradient of max_k (b*g)_k w.r.t. g
            # (b*g)[k] = sum_i b[k-i]*g[i], so d/d g[j] = b[k-j] (if valid)
            grad = np.zeros(n)
            for k in active:
                k = int(k)
                i_min = max(0, k - n + 1)
                i_max = min(n - 1, k)
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += b[k - indices]
            grad /= len(active)
            
            # FW direction: minimize grad · e_i => pick i with min grad[i]
            i_star = int(np.argmin(grad))
            
            # Step size
            step = 2.0 / (inner + 2)  # or use line search
            g_new = (1 - step) * g + step * np.zeros(n)
            g_new[i_star] += step
            g_new = np.maximum(g_new, 0)
            sg = float(np.sum(g_new))
            if sg > 1e-12:
                g_new = g_new / sg
            g = g_new
        
        # Update b
        b = g.copy()
        
        # Evaluate
        conv_ab = compute_conv(a)  # Actually we need conv(a, a)...
        # Wait, in alternating we're trying to minimize max_k (a*a)[k]
        # by alternating: fix b=a, find best g, then set a=g, fix b=a, find best g, etc.
        # So after solving for g with b fixed, we set a = g.
        a = g.copy()
        
        conv_a = compute_conv(a)
        obj = 2.0 * n * float(np.max(conv_a))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        outer_it += 1
        if verbose and outer_it % 10 == 0:
            print(f"    FW alt it={outer_it}: obj={best_obj:.8f}")
    
    return best_a * target_sum


def multiplicative_update(a, time_budget=60.0, verbose=False, eta=0.1, top_k=20):
    """
    Multiplicative weight update: reduce elements that contribute to peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    target_sum = s
    a = a / target_sum
    
    best_a = a.copy()
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    
    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        threshold = max_val * 0.9
        active = np.where(conv >= threshold)[0]
        if len(active) > top_k:
            active = active[np.argsort(conv[active])[::-1][:top_k]]
        
        # Compute how much each element contributes to the peaks
        penalty = np.zeros(n)
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            indices = np.arange(i_min, i_max + 1)
            # a[indices] * a[k-indices] contributes to peak
            # Reduce a[j] proportionally to a[k-j] (its partner)
            penalty[indices] += a[k - indices]
        
        penalty /= len(active)
        # Normalize penalty
        max_pen = float(np.max(penalty))
        if max_pen > 0:
            penalty = penalty / max_pen
        
        # Multiplicative update: a[j] *= (1 - eta * penalty[j])
        a = a * (1.0 - eta * penalty)
        a = np.maximum(a, 1e-10)  # keep positive
        sa = float(np.sum(a))
        a = a / sa
        
        if it % 10 == 0:
            conv = compute_conv(a)
            obj = 2.0 * n * float(np.max(conv))
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        
        it += 1
    
    return best_a * target_sum


def generate_initialization(kind, n, rng):
    """Generate diverse initializations."""
    if kind == 'ramp':
        a = np.linspace(0.1, 1.0, n)
        return a / np.sum(a)
    elif kind == 'reverse_ramp':
        a = np.linspace(1.0, 0.1, n)
        return a / np.sum(a)
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n * 0.3  # asymmetric center
        x = np.arange(n)
        sigma = n / 5.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2) + 0.01
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'exponential':
        a = np.exp(-np.arange(n) / (n * 0.3))
        return a / np.sum(a)
    elif kind == 'bimodal':
        a = np.ones(n) * 0.1
        a[n//5] = 2.0
        a[4*n//5] = 1.5
        kernel = np.ones(max(3, n//20)) / max(3, n//20)
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'power':
        a = np.arange(1, n+1) ** 0.5
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

1. Start with p (length n)

3. Compute F = FFT(p_pad)
4. Compute target spectrum: T = FFT(desired_autocorrelation)

6. p_pad_new = IFFT(F_new)
7. Take the first n elements: p_new = p_pad_new[:n]
8. Enforce non-negativity: p_new = max(p_new, 0)
9. Normalize: p_new = p_new / sum(p_new)
10. Repeat

"""
Peak-shaving optimization for step function coefficient sequences using
FFT-based phase retrieval (Gerchberg-Saxton / Fienup), accelerated subgradient,
Frank-Wolfe alternation, multiplicative updates, and multi-start strategy.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def fft_phase_retrieval(a, time_budget=100.0, verbose=False, clip_frac=0.85, 
                         alpha=0.5, n_inner_check=10):
    """
    Fienup/Gerchberg-Saxton style FFT phase retrieval for minimum peak autocorrelation.
    Works by iteratively: (1) compute autocorrelation, (2) clip peaks, 
    (3) update magnitude spectrum, (4) enforce non-negativity and support.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    target_sum = s
    a = a / target_sum

    fft_len = 1
    while fft_len < 2 * n - 1:
        fft_len <<= 1

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    p_pad = np.zeros(fft_len)
    p_pad[:n] = a

    # Current phase (initialized from the initial guess)
    F = rfft(p_pad, fft_len)
    phase = np.exp(1j * np.angle(F))
    magnitude = np.abs(F)

    it = 0
    stall = 0
    best_stall_obj = best_obj
    current_clip = clip_frac
    current_alpha = alpha

    while time.monotonic() - start < time_budget:
        # Forward: get autocorrelation from current spectrum
        power = magnitude ** 2
        autocorr = irfft(power, fft_len)[:2 * n - 1]

        # Find peak and compute target
        max_val = float(np.max(autocorr))
        if max_val < 1e-15:
            break
        target_level = max_val * current_clip
        target_autocorr = np.minimum(autocorr, target_level)
        target_autocorr = np.maximum(target_autocorr, 0)

        # Target magnitude spectrum
        target_F = rfft(target_autocorr, fft_len)
        target_magnitude = np.abs(target_F)

        # Update magnitude (geometric interpolation)
        magnitude = magnitude ** (1.0 - current_alpha) * target_magnitude ** current_alpha

        # Inverse FFT with current phase
        F_new = magnitude * phase
        p_pad_new = irfft(F_new, fft_len)

        # Enforce support (first n elements) and non-negativity
        p_new = np.maximum(p_pad_new[:n], 0)
        s_new = float(np.sum(p_new))
        if s_new < 1e-12:
            # Perturb and retry
            rng = np.random.default_rng(it + 1)
            p_new = a * np.exp(rng.normal(0, 0.1, n))
            p_new = np.maximum(p_new, 0)
            s_new = float(np.sum(p_new))
        p_new = p_new / s_new

        # Update phase for next iteration
        F_updated = rfft(np.concatenate([p_new, np.zeros(fft_len - n)]), fft_len)
        phase = np.exp(1j * np.angle(F_updated))
        magnitude = np.abs(F_updated)

        # Track best
        if it % n_inner_check == 0:
            obj = 2.0 * n * float(np.max(compute_conv(p_new)))
            if obj < best_obj - 1e-12:

                best_a = p_new.copy()
                stall = 0

                    print(f"    FPR it={it}: obj={best_obj:.8f}")
            else:
                stall += n_inner_check

        # Adaptive parameters

            current_clip = max(0.5, current_clip - 0.05)
            current_alpha = min(0.9, current_alpha + 0.1)
            stall = 0
            # Perturb phase
            rng = np.random.default_rng(it)
            phase_noise = np.exp(1j * rng.normal(0, 0.1, len(phase)))
            phase = phase * phase_noise

        it += 1

    return best_a * target_sum


def accelerated_subgradient(a, time_budget=100.0, verbose=False, top_k=30,
                            threshold_frac=0.95, initial_lr=0.01):
    """
    Nesterov-accelerated subgradient method for min max_k (a*a)[k] s.t. sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    target_sum = s
    a = a / target_sum

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    y = a.copy()  # current point
    x = a.copy()  # Nesterov lookahead
    momentum = np.zeros(n)
    lr = initial_lr
    patience = 0
    it = 0

    while time.monotonic() - start < time_budget:
        # Evaluate at lookahead point
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-12:
            x = x / sx
        conv_x = compute_conv(x)

        max_val = float(np.max(conv_x))
        threshold = max_val * threshold_frac
        active = np.where(conv_x >= threshold)[0]
        if len(active) > top_k:
            active = active[np.argsort(conv_x[active])[::-1][:top_k]]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv_x))])

        # Compute subgradient: average of 2*a[k-j] over active peaks
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            indices = np.arange(i_min, i_max + 1)
            grad[indices] += 2.0 * x[k - indices]
        grad /= len(active)

        # Project gradient onto feasible direction (sum=0)
        grad = grad - np.mean(grad)

        # Nesterov step
        t_k = 2.0 / (float(it + 1) + 2.0)
        momentum = (float(it) / (float(it + 1) + t_k)) * momentum - lr * grad
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        sy = float(np.sum(y_new))
        if sy > 1e-12:
            y_new = y_new / sy

        # Update lookahead
        x = y_new + (1.0 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-12:
            x = x / sx

        y = y_new
        it += 1

        # Track best every few iterations
        if it % 10 == 0 or it < 20:
            obj = 2.0 * n * float(np.max(compute_conv(y)))
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = y.copy()
                patience = 0
            else:
                patience += 1

            if patience > 300:
                lr = max(lr * 0.8, 1e-4)
                patience = 0
                if verbose and it % 500 == 0:
                    print(f"    SG it={it}: obj={best_obj:.8f}, lr={lr:.6f}")

        # Periodic perturbation to escape local minima
        if it % 3000 == 2999:
            rng = np.random.default_rng(it)
            scale = 0.015
            y = best_a * np.exp(rng.normal(0, scale, n))
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy > 1e-12:
                y = y / sy
            x = y.copy()
           

            momentum = np.zeros(n)

    return best_a * target_sum


def fw_alternation(a, time_budget=100.0, verbose=False, inner_iters=50):
    """
    Frank-Wolfe based alternating minimization for the inner convex subproblem.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    target_sum = s
    a = a / target_sum
    b = a.copy()

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    outer_it = 0

    fft_len = 1
    while fft_len < 2 * n - 1:
        fft_len <<= 1

    while time.monotonic() - start < time_budget:
        # Solve: min max_k (b * g)_k s.t. sum(g)=1, g>=0 using FW
        g = b.copy()
        fb = rfft(b, fft_len)

        for inner in range(inner_iters):
            fg = rfft(g, fft_len)
            conv_bg = irfft(fb * fg, fft_len)[:2 * n - 1]

            max_val = float(np.max(conv_bg))
            threshold = max_val * 0.95
            active = np.where(conv_bg >= threshold)[0]
            if len(active) > 30:
                active = active[np.argsort(conv_bg[active])[::-1][:30]]
            if len(active) == 0:
                active = np.array([int(np.argmax(conv_bg))])

            # Gradient of max_k (b*g)_k w.r.t. g: for each active k, grad[j] += b[k-j]
            grad = np.zeros(n)
            for k in active:
                k = int(k)
                i_min = max(0, k - n + 1)
                i_max = min(n - 1, k)
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += b[k - indices]
            grad /= len(active)

            # FW step: move towards e_{i*} where i* = argmin grad
            i_star = int(np.argmin(grad))
            step = 2.0 / (inner + 2.0)
            g[i_star] += step
            g = np.maximum(g, 0)
            sg = float(np.sum(g))
            if sg > 1e-12:
                g = g / sg

        b = g.copy()
        a = g.copy()

        if outer_it % 5 == 0:
            obj = 2.0 * n * float(np.max(compute_conv(a)))
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        outer_it += 1

    return best_a * target_sum


def multiplicative_update(a, time_budget=60.0, verbose=False, eta=0.1, top_k=30):
    """
    Multiplicative weight update: reduce elements contributing to peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    target_sum = s
    a = a / target_sum

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        threshold = max_val * 0.85
        active = np.where(conv >= threshold)[0]
        if len(active) > top_k:
            active = active[np.argsort(conv[active])[::-1][:top_k]]

        # Compute penalty for each element
        penalty = np.zeros(n)
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            indices = np.arange(i_min, i_max + 1)
            penalty[indices] += a[k - indices]
        penalty /= len(active)

        max_pen = float(np.max(penalty))
        if max_pen > 0:
            penalty = penalty / max_pen
            a = a * (1.0 - eta * penalty)
            a = np.maximum(a, 1e-10)
            sa = float(np.sum(a))
            a = a / sa

        if it % 20 == 0:
            obj = 2.0 * n * float(np.max(compute_conv(a)))
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        it += 1

    return best_a * target_sum


def generate_initialization(kind, n, rng):
    """Generate diverse initializations."""
    if kind == 'ramp':
        a = np.linspace(0.1, 1.0, n)
        return a / np.sum(a)
    elif kind == 'reverse_ramp':
        a = np.linspace(1.0, 0.1, n)
        return a / np.sum(a)
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.1
        return a / np.sum(a)
    elif kind == 'gaussian_asym':
        center = n * 0.3
        x = np.arange(n, dtype=np.float64)
        sigma = n / 5.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2) + 0.01
        return a / np.sum(a)
    elif kind == 'exponential':
        a = np.exp(-np.arange(n, dtype=np.float64) / (n * 0.3))
        return a / np.sum(a)
    elif kind == 'power':
        a = np.arange(1, n + 1, dtype=np.float64) ** 0.5
        return a / np.sum(a)
    elif kind == 'bimodal':
        a = np.ones(n) * 0.1
        a[n // 5] = 2.0
        a[4 * n // 5] = 1.5
        ksize = max(3, n // 20)
        kernel = np.ones(ksize) / ksize
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'sigmoid':
        x = np.linspace(-5, 5, n)
        a = 1.0 / (1.0 + np.exp(-x))
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'noise_smooth':
        a = rng.random(n)
        ksize = max(3, n // 30)
        kernel = np.ones(ksize) / ksize
        a = np.convolve(a, kernel, mode='same')
        a = a + 0.01
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
