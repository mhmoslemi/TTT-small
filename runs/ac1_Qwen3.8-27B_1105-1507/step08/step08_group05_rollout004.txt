import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax(theta):
    """Numerically stable softmax."""
    theta = theta - np.max(theta)
    e = np.exp(theta)
    return e / np.sum(e)


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def natural_gradient_step(theta, a, top_k=5, eta=0.1):
    """
    One step of natural gradient (in θ-space) on f(a) = max_k conv(a,a)[k].
    Uses weighted combination of top-k peaks for a more stable gradient.
    """
    n = len(a)
    conv = compute_conv(a)
    # Get top-k peak indices
    top_idx = np.argsort(conv)[::-1][:top_k]
    # Weight by peak value (higher peaks get more weight)
    peak_vals = conv[top_idx]
    weights = peak_vals ** 2  # quadratic weighting
    weights /= np.sum(weights)
    
    # Gradient in a-space: sum_k w_k * 2*a[k-j] (reversed)
    grad_a = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        for j in range(n):
            ki = k - j
            if 0 <= ki < n:
                grad_a[j] += w * 2.0 * a[ki]
    
    # Natural gradient: grad_theta[i] = a[i] * (grad_a[i] - sum_j a[j]*grad_a[j])
    mean_grad = np.sum(a * grad_a)
    grad_theta = a * (grad_a - mean_grad)
    
    theta_new = theta - eta * grad_theta
    return theta_new

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ b = conv(a, b)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """LP: min M s.t. conv(a,b)[k] <= M, sum(b)=1, b>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if b.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                bv = np.maximum(b.value, 0)
                bs = float(np.sum(bv))
                if bs > 1e-10:
                    return bv / bs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=200.0, momentum=0.7):
    """LP alternating with momentum and line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    v = np.zeros(n)  # momentum
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        b_lp, M_lp = solve_lp(a)
        if b_lp is None:
            break
        
        # Update with momentum
        v = momentum * v + (1 - momentum) * (b_lp - a)
        a_try = a + v
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            a_try = a
        else:
            a_try = a_try / st
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            stall = 0
        else:
            # Try without momentum

            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = best_a
                v = np.zeros(n)
                stall = 0
            else:

def pair_exchange_search(a0, time_budget=200.0, n_trials=100000):
    """Local search: randomly pick pairs and transfer mass."""
    n = len(a0)
    a = a0.astype(np.float64).copy()

    if s < 1e-10:
        return a0.copy()
    a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

        if time.monotonic() - start > time_budget:
            break
        
        # Pick random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] < 1e-10:
            continue
        
        # Try several transfer amounts
        for frac in [0.01, 0.05, 0.1, 0.2, 0.3]:
            delta = frac * a[i]
            a[i] -= delta
            a[j] += delta
            a = np.maximum(a, 0)
            st = float(np.sum(a))
            a = a / st
            
            obj = compute_obj(a)
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                break
            else:
                # Revert
                a[i] += delta
                a[j] -= delta
                a = a / float(np.sum(a))
    
    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw

def compute_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax(theta):
    """Numerically stable softmax."""
    theta = theta - np.max(theta)
    e = np.exp(theta)
    return e / np.sum(e)


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def ng_subgradient(a, top_k=3):
    """
    Compute weighted subgradient of max_k conv(a,a)[k].
    Returns gradient vector in a-space.
    """
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:top_k]
    peak_vals = conv[top_idx]
    # Use exponential weighting for stability
    weights = np.exp(peak_vals / (peak_vals[0] + 1e-15))
    weights /= np.sum(weights)

    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        # grad[j] += w * 2 * a[k-j] where 0 <= k-j < n
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            grad[j] += w * 2.0 * a[k - j]
    return grad

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ b = conv(a, b)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, prev_b=None):
    """LP: min M s.t. conv(a,b)[k] <= M, sum(b)=1, b>=0."""
    n = len(a)
    a_s = float(np.sum(a))
    if a_s < 1e-10:
        return None, None
    a_n = a / a_s
    A = build_conv_matrix(a_n)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    # Try with warm start if available
    if prev_b is not None:
        try:
            prob.solve(solver=cp.CLARABEL, warm_start=True,
                       b_val=prev_b, verbose=False)
            if b.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                bv = np.maximum(b.value, 0)
                bs = float(np.sum(bv))
                if bs > 1e-10:
                    return bv / bs, float(M.value)
        except Exception:
            pass

    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:

def resample_seq(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

    if sw < 1e-10:

    return w / sw

            best_seq = a_lp2.copy()
            print(f"  LP2: obj={best_score:.8f} **")
        else:
            print(f"  LP2: obj={obj_lp2:.8f}")

    # === Phase 6: Random local search - 120s ===
    print(f"\n=== Phase 6: Random Local Search ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(120.0, remaining * 0.15)
        a_rls = random_local_search(best_seq, time_budget=alloc, rng=rng)
        obj_rls = compute_obj(a_rls)
        if obj_rls < best_score:
            best_score = obj_rls
            best_seq = a_rls.copy()
            print(f"  RLS: obj={best_score:.8f} **")
        else:
            print(f"  RLS: obj={obj_rls:.8f}")

    # === Phase 7: n-sweep - remaining time ===
    print(f"\n=== Phase 7: n-Sweep ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best,
        n_best - 100, n_best + 100,
        n_best - 200, n_best + 200,
        n_best - 50, n_best + 50,
    ]))
    n_options = [n for n in n_options if 50 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 12:
            break
        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)

        alloc = min(12.0, remaining * 0.15)
        if alloc > 8:
            a_opt = lp_alternate(a_try, time_budget=alloc, blend=0.5)
        else:
            a_opt = a_try
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final NG + LP combo - remaining ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 8: Final Polish ===")
        alloc = min(60.0, remaining * 0.4)
        if alloc > 15:
            a_f1 = natural_gradient_optimize(best_seq, time_budget=alloc, eta_init=0.005)
            obj_f1 = compute_obj(a_f1)
            if obj_f1 < best_score:
                best_score = obj_f1
                best_seq = a_f1.copy()
                print(f"  FinalNG: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(60.0, remaining * 0.5)
            a_f2 = lp_alternate(best_seq, time_budget=alloc, blend=0.5)
            obj_f2 = compute_obj(a_f2)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  FinalLP: obj={best_score:.8f} **")

    # === Phase 9: Random restarts - remaining ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        print(f"\n=== Phase 9: Random Restarts ===")
        for r in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            scale = 0.01 + 0.015 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr < 1e-10:
                continue
            a_r = a_r / sr

            alloc = min(remaining * 0.2, 25.0)
            if alloc > 10:
                a_opt = lp_alternate(a_r, time_budget=alloc * 0.5, blend=0.5)
                a_opt = natural_gradient_optimize(a_opt, time_budget=alloc * 0.5, eta_init=0.01)
            else:
                a_opt = a_r
            obj_r = compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  R{r} (s={scale:.3f}): obj={best_score:.8f} **")
            else:
                print(f"  R{r} (s={scale:.3f}): obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
