def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    P = fft(p, N)
    A = fft(a_norm, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]

    return g, conv


def lp_subgradient_descent(a, time_budget=200.0, verbose=False):
    """
    Main optimizer: LP-dual-weighted subgradient with Nesterov momentum.
    Periodically uses LP primal solution for direct jumps.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov parameters
    beta = 0.85
    lr = 1.5 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    lp_interval = 50  # Solve LP every 50 iterations
    lp_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient
        if it % lp_interval == 0:
            # Use LP-based subgradient (more expensive but better weights)
            g, conv = lp_weighted_subgradient(y)
            lp_count += 1
        else:
            # Use smoothed gradient (cheaper)
            tau = 50.0 + 200.0 * progress
            g, conv = smoothed_gradient_fft(y, tau)

        # Barzilai-Borwein adaptive step (every 2 iterations)
        if it >= 2 and it % 2 == 0:
            s_vec = y - (y + beta * v + lr * g)  # approximate
            # Use simpler BB: based on recent gradient changes
            pass

        # Update
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        # Adaptive learning rate
        if it % 100 == 0:
            lr *= 0.995
        if it - best_it > 500:
            # Stuck - try LP jump
            if it % lp_interval == 0 or elapsed < time_budget:
                g_lp, M_lp, _ = solve_lp_with_dual(best_a)
                if g_lp is not None:
                    # Backtracking line search
                    for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
                        a_try = (1 - t) * best_a + t * g_lp
                        a_try = np.maximum(a_try, 0)
                        s_try = float(np.sum(a_try))
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try.copy()
                            v = np.zeros(n)
                            it = best_it
                            break
            else:
                # Reset
                a = best_a.copy()
                v = np.zeros(n)
                lr = 1.0 / np.sqrt(n)
                best_it = it

        if verbose and it % 200 == 0:
            print(f"    LP-SGD it{it}: obj={best_obj:.8f}, lr={lr:.6f}, t={elapsed:.0f}s, lp_calls={lp_count}")

    return best_a


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """
    Nesterov accelerated gradient descent on smoothed objective over simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if it % 500 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.7 * lr + 0.3 * min(max(bb_lr, 1e-8), 5.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 1.5 / np.sqrt(n) * (0.5 ** (it - best_it) // 15000)
            best_it = it

    return best_a


def peak_shaving(a, time_budget=60.0, K=10, verbose=False):
    """
    Local peak shaving: identify top peaks and directly reduce them
    by adjusting the contributing sequence elements.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    iteration = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_cv = float(np.max(conv))
        top_idx = np.argsort(conv)[::-1][:K]

        improved = False
        for k in top_idx:
            k = int(k)
            # Find which pairs (i, j) with i+j=k contribute most to conv[k]
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue

            # Contribution of each pair
            contributions = np.array([a[i] * a[k - i] for i in range(i_start, i_end + 1)])
            total = float(np.sum(contributions))
            if total < 1e-15:
                continue

            # Find the top contributing index
            top_contrib_idx = np.argmax(contributions)
            i_star = i_start + top_contrib_idx
            j_star = k - i_star

            # Try reducing the larger of a[i_star], a[j_star]
            if a[i_star] >= a[j_star]:
                # Reduce a[i_star]
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    a_try = a.copy()
                    reduction = a_try[i_star] * frac
                    a_try[i_star] -= reduction
                    # Redistribute to indices that don't contribute to this peak
                    # Spread to indices far from the peak
                    spread_indices = [idx for idx in range(n) if idx != i_star and abs(idx - k) > n // 4]
                    if len(spread_indices) > 0:
                        spread_indices = np.random.choice(spread_indices, size=min(10, len(spread_indices)), replace=False)
                        a_try[spread_indices] += reduction / len(spread_indices)
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break
            else:
                # Reduce a[j_star]
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    a_try = a.copy()
                    reduction = a_try[j_star] * frac
                    a_try[j_star] -= reduction
                    spread_indices = [idx for idx in range(n) if idx != j_star and abs(idx - k) > n // 4]
                    if len(spread_indices) > 0:
                        spread_indices = np.random.choice(spread_indices, size=min(10, len(spread_indices)), replace=False)
                        a_try[spread_indices] += reduction / len(spread_indices)
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break

        iteration += 1
        if not improved and iteration > 50:
            # Perturb to escape
            noise = np.random.normal(0, 0.01, n)
            a = best_a * np.exp(noise)
            a = np.maximum(a, 0)
            a = a / a.sum()
            iteration = 0

        if verbose and iteration % 20 == 0:
            print(f"    PeakShave it{iteration}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Cosine-tapered window
        x = np.linspace(0, np.pi, n)
        a = 0.3 + 0.7 * np.cos(x) ** 2
        a = a / a.sum()
    elif mode == 6:
        # Two-gaussian with random centers
        c1 = rng.integers(0, n // 3)
        c2 = rng.integers(2 * n // 3, n)
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 7:
        # Symmetric: a[i] = a[n-1-i]
        half = n // 2
        a_half = rng.uniform(0.5, 1.0, half)
        a = np.concatenate([a_half, a_half[::-1]]) if n % 2 == 0 else \
            np.concatenate([a_half, [a_half[0]], a_half[::-1]])
        a = a[:n]
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP-dual subgradient + smoothed GD + peak shaving + multi-n.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: LP-dual subgradient descent (main workhorse) ===
    print(f"\n=== Phase 1: LP-Dual Subgradient Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 350.0)
    if p1_budget > 50:
        a_sgd = lp_subgradient_descent(best_seq, time_budget=p1_budget, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 LP-SGD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP-SGD: obj={obj_sgd:.8f}")

    # === Phase 2: Smoothed gradient refinement ===
    print(f"\n=== Phase 2: Smoothed GD Refinement ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.20, 200.0)
    if p2_budget > 30:
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=p2_budget,
                                          tau_init=20.0, tau_end=800.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase2 NSD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 NSD: obj={obj_sgd:.8f}")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.10, 100.0)
    if p3_budget > 20:
        a_ps = peak_shaving(best_seq, time_budget=p3_budget, K=8, verbose=True)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Phase3 PeakShave: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 PeakShave: obj={obj_ps:.8f}")

    # === Phase 4: LP jump + blend (iterative) ===
    print(f"\n=== Phase 4: LP Jump & Blend ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.20, 250.0)
    if p4_budget > 30:
        start_p4 = time.monotonic()
        iter_count = 0
        while time.monotonic() - start_p4 < p4_budget:
            g_lp, M_lp, _ = solve_lp_with_dual(best_seq)
            if g_lp is None:
                break

            # Check if LP solution itself is better
            obj_lp = compute_obj(g_lp)
            if obj_lp < best_score - 1e-14:
                best_score = obj_lp
                best_seq = g_lp.copy()
                print(f"    LP jump: obj={best_score:.8f} **")
                iter_count += 1
                continue

            # Backtracking from large to small step
            found = False
            for t in [0.7, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
                a_try = (1.0 - t) * best_seq + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_score - 1e-14:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    found = True
                    break

            iter_count += 1
            if not found:
                # Perturb and try again
                scale = 0.01 + 0.005 * min(iter_count, 5)
                best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                best_seq = np.maximum(best_seq, 0)
                best_seq = best_seq / best_seq.sum()
                if iter_count > 20:
                    break

        print(f"  Phase4 LP-Jump: {iter_count} iters, obj={best_score:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = []
    for delta in [-300, -200, -100, 100, 200, 300, 400, 500]:
        n_try = n_current + delta
        if 400 <= n_try <= 2000:
            n_options.append(n_try)
    # Also try some absolute values
    for n_abs in [500, 700, 900, 1100, 1300]:
        if 400 <= n_abs <= 2000 and n_abs not in n_options:
            n_options.append(n_abs)

    n_options = sorted(set(n_options))
    n_options = [n for n in n_options if abs(n - n_current) > 80]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        alloc1 = min(remaining * 0.20, 60.0)
        if alloc1 > 20:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=300.0)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.10, 40.0)
        if alloc2 > 15:
            a_try = lp_subgradient_descent(a_try, time_budget=alloc2)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.2 + 0.08 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.20, 80.0)
        if alloc > 25:
            half = alloc * 0.6
            a_r = lp_subgradient_descent(a_r, time_budget=half)
            remaining = deadline - time.monotonic()
            alloc2 = min(remaining * 0.08, 30.0)
            if alloc2 > 10:
                a_r = nesterov_smoothed_descent(a_r, time_budget=alloc2,
                                                tau_init=30.0, tau_end=400.0)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP subgradient with high precision
    if remaining > 40:
        alloc = min(remaining * 0.3, 120.0)
        a_final = lp_subgradient_descent(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP-SGD: obj={best_score:.8f} **")

    # Smoothed GD with very high tau
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 80.0)
        a_final = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                            tau_init=100.0, tau_end=2000.0)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final High-tau NSD: obj={best_score:.8f} **")

    # Peak shaving
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.2, 50.0)
        a_final = peak_shaving(best_seq, time_budget=alloc, K=5)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final PeakShave: obj={best_score:.8f} **")

    # One last LP jump
    remaining = deadline - time.monotonic()
    if remaining > 15:
        g_lp, M_lp, _ = solve_lp_with_dual(best_seq)
        if g_lp is not None:
            for t in [1.0, 0.5, 0.3, 0.1]:
                a_try = (1 - t) * best_seq + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_score:
                        best_score = obj_try
                        best_seq = a_try.copy()
                        print(f"  Final LP jump (t={t}): obj={best_score:.8f} **")
                        break

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:

        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
