"""

Uses entropic smoothing with FFT gradients, LP polishing, spectral-domain perturbations,

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft, rfftfreq
import cvxpy as cp
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT.
    Returns (gradient, convolution).
    """
    n = len(a)
    conv = compute_conv(a)
    
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    N = 1
    while N < 2 * (2 * n - 1):
        N <<= 1
    
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    
    return g, conv

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCIP, cp.CBC, cp.GLPK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None

def spectral_perturb(a, rng, strength=0.1):
    """
    Perturb the sequence in the frequency domain to explore new local minima.
    Adds random sinusoidal components while preserving non-negativity.
    """
    n = len(a)
    a = a / float(np.sum(a))
    
    # Get FFT
    A = fft(a)
    
    # Add random perturbations at random frequencies
    num_perturb = max(3, n // 50)
    for _ in range(num_perturb):
        freq = rng.integers(1, n // 2)
        phase = rng.uniform(0, 2 * np.pi)
        amp = strength * rng.uniform(0.5, 1.0) / np.sqrt(n)
        A[freq] += amp * np.exp(1j * phase)
        A[n - freq] += amp * np.exp(-1j * phase)
    
    # Inverse FFT
    a_new = ifft(A).real
    
    # Ensure non-negativity and normalize
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    a_new = a_new / s
    
    return a_new


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.arange(n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Triangular
        mid = n // 2
        a = np.ones(n) - 0.8 * np.abs(x - mid) / mid
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        xx = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - xx**2 + 1e-8))
        a = a / a.sum()
    elif mode == 4:
        # Sum of sinusoids
        a = np.ones(n)
        for _ in range(5):
            f = rng.uniform(1, 8)
            p = rng.uniform(0, 2 * np.pi)
            a += 0.3 * np.sin(2 * np.pi * f * x / n + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Exponential decay from center
        mid = n // 2
        a = np.exp(-np.abs(x - mid) / (n * 0.3))
        a = a / a.sum()
    elif mode == 6:
        # Bimodal
        c1 = n // 4
        c2 = 3 * n // 4
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((x - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 7:
        # Power law
        xx = np.linspace(0.01, 1, n)
        a = xx ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 8:
        # Cosine window
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
        a = a / a.sum()
    elif mode == 9:
        # Random walk (smoothed)
        a = np.cumsum(rng.normal(0, 1, n))
        a = a - a.min() + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def subgradient_refine(a, time_budget=100.0, K=8, verbose=False):
    """
    Subgradient method with top-K peaks, momentum, and adaptive step on simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 3.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break
        
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        weights = np.exp(8.0 * (peak_vals - max_conv) / max_conv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 300:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.998

        it += 1

    return best_a

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a) via FFT cross-correlation.
    Returns (gradient_vector, convolution_array).
    """
    n = len(a)
    conv = compute_conv(a)

    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]

    return g, conv


def entropic_gd(a, time_budget=300.0, tau_init=10.0, tau_end=800.0, verbose=False):
    """
    Nesterov accelerated gradient descent on smoothed objective over simplex.
    Adaptive tau annealing, BB step sizes, and stagnation detection.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    lr_min = 1e-10
    v = np.zeros(n)

    prev_g = None
    it = 0
    stale = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(1.0, time_budget * 0.85))

        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if it % 5000 == 0 and verbose:
            print(f"        EGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        # Nesterov lookahead
        y = a + beta * v
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()

            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))

                lr = 0.3 * lr + 0.7 * bb_lr

        # Nesterov update
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:

            a_new = a_new / s_new

        a = a_new
        it += 1

        else:

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: minimize M subject to conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_value) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCIP, cp.CBC, cp.GLPK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None

def spectral_perturb(a, rng, strength=0.1):
    """
    Perturb sequence in frequency domain: add random sinusoidal components.
    Preserves the mean (DC component) and ensures non-negativity after projection.
    """
    n = len(a)
    a = a.copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    A = fft(a).copy()
    num_perturb = max(3, n // 40)
    for _ in range(num_perturb):
        freq = rng.integers(1, n // 2)
        phase = rng.uniform(0, 2 * np.pi)
        amp = strength * rng.uniform(0.3, 1.0) / math.sqrt(n)
        A[freq] += amp * np.exp(1j * phase)
        if freq != n - freq:
            A[n - freq] += amp * np.exp(-1j * phase)

    a_new = ifft(A).real
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a
    return a_new / s_new


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    """Generate diverse initializations with different structural properties."""
    x = np.arange(n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Triangular (peaked center)
        mid = n // 2
        a = np.ones(n) - 0.8 * np.abs(x - mid) / max(1, mid)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        # Arcsine (peaked at edges)
        xx = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - xx ** 2 + 1e-8))
        a = a / a.sum()
    elif mode == 4:
        # Multi-sinusoid
        a = np.ones(n)
        for _ in range(6):
            f = rng.uniform(1, 10)
            p = rng.uniform(0, 2 * np.pi)
            a += 0.25 * np.sin(2 * np.pi * f * x / n + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Gaussian centered
        mid = n // 2
        a = np.exp(-((x - mid) ** 2) / (2 * (n * 0.25) ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Bimodal
        c1 = n // 4
        c2 = 3 * n // 4
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((x - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 7:
        # Power law
        xx = np.linspace(0.01, 1, n)
        a = xx ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 8:
        # Cosine window
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 9:
        # Smoothed random walk
        a = np.cumsum(rng.normal(0, 1, n))
        a = a - a.min() + 0.1
        a = a / a.sum()
    elif mode == 10:
        #Inverse triangular (valley center)
        mid = n // 2
        a = 0.2 + 0.8 * np.abs(x - mid) / max(1, mid)
        a = a / a.sum()
    elif mode == 11:
        # Periodic pattern
        period = max(2, n // rng.integers(3, 8))
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x / period)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def subgradient_refine(a, time_budget=100.0, K=8, verbose=False):
    """
    Subgradient method targeting top-K peaks with momentum and adaptive step.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 3.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break

        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        weights = np.exp(10.0 * (peak_vals - max_conv) / max_conv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = project_simplex(a - lr * v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 400:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr = max(lr * 0.998, 1e-12)

        it += 1

    return best_a
