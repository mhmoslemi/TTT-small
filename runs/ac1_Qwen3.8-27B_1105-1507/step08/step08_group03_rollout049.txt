def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_conv_smooth(a, T):
    """
    Compute gradient of (2n/T)*log(sum_k exp(T*conv(a,a)_k)) w.r.t. a.
    Uses FFT for efficiency.
    Returns gradient of same shape as a.
    """
    n = len(a)
    conv = compute_conv(a)
    c_max = conv.max()
    z = T * (conv - c_max)
    e = np.exp(z)
    sum_e = e.sum()
    w = e / sum_e  # weights on each convolution tap

    # Gradient: df/da_i = 4n * sum_k w_k * a_{k-i}
    # This is 4n * (w circularly convolved with a) evaluated at specific indices
    # More precisely: sum_k w_k * a[k-i] for valid k,i
    # = cross-correlation of w and a

    # Compute using FFT: the sum over k of w[k]*a[k-i] is the cross-correlation
    # Let's compute it directly using the convolution structure
    # g[i] = sum_k w[k] * a[k-i] where 0<=k<2n-1 and 0<=k-i<n
    # = sum_{k=max(i,0)}^{min(2n-2, n-1+i)} w[k] * a[k-i]

    # Use FFT: this is the convolution of w (padded) with reversed a
    size = 1
    while size < 3 * n:
        size <<= 1
    w_pad = np.zeros(size)
    w_pad[:len(w)] = w
    a_rev = a[::-1]
    a_pad = np.zeros(size)
    a_pad[:n] = a_rev

    result = irfft(rfft(w_pad, size) * rfft(a_pad, size), size)
    # The cross-correlation at lag i is result[i + n - 1] for i in 0..n-1
    g = 4.0 * n * result[n - 1:n - 1 + n]

    return g


def smooth_lu_sum_exp_obj(a, T):
    """Compute smooth approximation value."""
    n = len(a)
    conv = compute_conv(a)
    c_max = conv.max()
    z = T * (conv - c_max)
    val = 2.0 * n * (c_max + np.log(np.exp(z).sum()) / T)
    return val


def smooth_lu_sum_exp_grad(a, T):
    """Gradient of smooth log-sum-exp objective."""
    n = len(a)
    conv = compute_conv(a)
    c_max = conv.max()
    z = T * (conv - c_max)
    e = np.exp(z)
    sum_e = e.sum()
    w = e / sum_e

    # g[i] = 4n * sum_k w[k] * a[k-i]
    size = 1
    while size < 3 * n:
        size <<= 1
    w_pad = np.zeros(size)
    w_pad[:len(w)] = w
    a_rev = np.zeros(size)
    a_rev[:n] = a[::-1]
    result = irfft(rfft(w_pad, size) * rfft(a_rev, size), size)
    g = 4.0 * n * result[n - 1:n - 1 + n]

    return g


def smooth_optimize_lbfgs(a, time_budget=120.0, T_start=50.0, T_end=5.0, verbose=False):
    """
    L-BFGS optimization on log-sum-exp smoothed objective with adaptive T annealing.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Multi-stage T annealing
    n_stages = 5
    Ts = np.geomspace(T_start, T_end, n_stages)

    for stage, T in enumerate(Ts):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        stage_budget = min(remaining / (n_stages - stage), time_budget * 0.4)

        def obj_grad(x):
            x = np.maximum(x, 1e-14)
            s = x.sum()
            if s < 1e-10:
                return 1e10, np.zeros(len(x))
            x_norm = x / s
            val = smooth_lu_sum_exp_obj(x_norm, T)
            g = smooth_lu_sum_exp_grad(x_norm, T)
            return val, g

        max_iter = max(50, int(stage_budget * 3))
        try:
            res = sp_minimize(obj_grad, a, method='L-BFGS-B',
                              bounds=[(1e-14, None)] * n,
                              options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                       'gtol': 1e-6, 'disp': False})
            a_opt = np.maximum(res.x, 1e-14)
            s_opt = float(np.sum(a_opt))
            if s_opt > 1e-10:
                a_opt = a_opt / s_opt
                obj_opt = compute_obj(a_opt)
                if obj_opt < best_obj:
                    best_obj = obj_opt
                    best_a = a_opt.copy()
                    if verbose:
                        print(f"    Smooth stage {stage} (T={T:.1f}): obj={best_obj:.8f}")
                    a = a_opt
        except Exception:
            pass

    return best_a


def build_conv_matrix(A_rows, a, n):
    """Build matrix rows for specific convolution indices. A[k] @ g = conv(a,g)[k]."""
    rows = np.zeros((len(A_rows), n))
    for idx, k in enumerate(A_rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                rows[idx, i] += a[j]
    return rows


def solve_lp_inner(a, active_indices=None, max_active=50):
    """
    Solve LP: min t s.t. conv(a,g)[k] <= t for active k, sum(g)=1, g>=0.
    Returns (g, t, dual_weights).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv_a = compute_conv(a)

    if active_indices is None:
        # Use top max_active indices
        active_indices = np.argsort(conv_a)[::-1][:max_active]

    A = build_conv_matrix(active_indices, a_norm, n)
    m = len(active_indices)

    if HAVE_CVXPY and m < 200:
        g_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [A @ g_var <= t_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        # Get dual variables
                        try:
                            duals = prob.constraints[0].dual_value
                        except Exception:
                            duals = np.zeros(m)
                        return g_val / s, float(t_var.value), duals
            except Exception:
                continue

    # Fallback to scipy linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            duals = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else np.zeros(m)
            return g_val / s, float(res.x[n]), duals
    return None, None, None


def golden_section_line_search(a, g, f, max_iter=20):
    """
    Golden section search for optimal t in [0,1] for a_new = (1-t)*a + t*g.
    f is the objective function.
    """
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_t = 0.0
    b_t = 1.0
    c_t = b_t - gr * (b_t - a_t)
    d_t = a_t + gr * (b_t - a_t)
    f_c = f((1 - c_t) * a + c_t * g)
    f_d = f((1 - d_t) * a + d_t * g)

    for _ in range(max_iter):
        if f_c < f_d:
            b_t = d_t
            d_t = c_t
            f_d = f_c
            c_t = b_t - gr * (b_t - a_t)
            f_c = f((1 - c_t) * a + c_t * g)
        else:
            a_t = c_t
            c_t = d_t
            f_c = f_d
            d_t = a_t + gr * (b_t - a_t)
            f_d = f((1 - d_t) * a + d_t * g)
        if abs(b_t - a_t) < 1e-6:
            break

    t_opt = (a_t + b_t) / 2.0
    return t_opt


def lp_alternating_v2(a, time_budget=200.0, verbose=False):
    """
    Improved LP alternating with golden-section line search and
    adaptive active set management.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(42)
    it = 0
    active_count = 30

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break

        conv = compute_conv(a)
        # Adaptive active set: start small, grow if stalling
        cur_active = min(active_count, 2 * n - 1)
        active_idx = np.argsort(conv)[::-1][:cur_active]

        g_lp, t_lp, duals = solve_lp_inner(a, active_indices=active_idx, max_active=cur_active)
        if g_lp is None:
            # Try with all indices
            g_lp, t_lp, duals = solve_lp_inner(a, active_indices=None, max_active=min(100, 2*n-1))
            if g_lp is None:
                break

        # Golden section line search
        def f_line(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return np.inf
            a_try = a_try / s_try
            return compute_obj(a_try)

        t_opt = golden_section_line_search(a, g_lp, f_line, max_iter=15)
        a_new = (1.0 - t_opt) * a + t_opt * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        it += 1
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slowly grow active set
            if stall == 0 and it % 5 == 0:
                active_count = min(active_count + 2, 100)
            if verbose and it % 10 == 0:
                print(f"    LP2 it{it}: obj={best_obj:.8f}, t={t_opt:.3f}, active={active_count}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 10:
                # Escape: perturb and restart
                noise_scale = min(0.2, 0.05 + 0.02 * (stall // 10))
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                active_count = 20

    return best_a


def dual_guided_refine(a, time_budget=60.0, verbose=False):
    """
    Use LP dual variables to guide coordinate-level adjustments.
    High-dual coordinates get mass moved away; low-dual get mass moved toward.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        conv = compute_conv(a)
        top_k = np.argsort(conv)[::-1][:40]

        g_lp, t_lp, duals = solve_lp_inner(a, active_indices=top_k, max_active=40)
        if duals is None or g_lp is None:
            break

        # Compute pressure on each coordinate from duals
        # pressure[i] = sum_k duals[k] * a[k-i]
        pressure = np.zeros(n)
        for idx, k in enumerate(top_k):
            if abs(duals[idx]) > 1e-10:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                for i in range(i_start, i_end + 1):
                    j = k - i
                    if 0 <= j < n:
                        pressure[i] += abs(duals[idx]) * a[j]

        # Normalize pressure
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        pressure /= p_max

        # Move mass from high-pressure to low-pressure coordinates
        # Try several small moves
        improved = False
        high_idx = np.argsort(pressure)[::-1][:20]
        low_idx = np.argsort(pressure)[:20]

        for hi in high_idx[:10]:
            if a[hi] < 1e-8:
                continue
            for lo in low_idx[:10]:
                if hi == lo:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    delta = frac * a[hi]
                    if delta < 1e-12:
                        continue
                    a[hi] -= delta
                    a[lo] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[hi] += delta
                        a[lo] -= delta
                        break

        it += 1
        if not improved:
            # Try random perturbations
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            new_obj = compute_obj(a)
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
            else:
                a = best_a.copy()
            if it > 50:
                break

    return best_a


def nesterov_subgradient_v2(a, time_budget=60.0, K=7, verbose=False):
    """
    Improved Nesterov-accelerated subgradient with adaptive K and temperature.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.5 / np.sqrt(n)
    stall = 0
    patience = 60
    rng = np.random.default_rng(999)

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        # Adaptive K
        max_val = conv.max()
        if max_val < 1e-15:
            break

        # Use soft top-K: include all taps above 80% of max
        threshold = 0.8 * max_val
        active = conv > threshold
        top_count = int(active.sum())
        if top_count < K:
            top_idx = np.argsort(conv)[::-1][:K]
        else:
            top_idx = np.where(active)[0]

        peak_vals = conv[top_idx]
        # Temperature-scaled weights
        temp = 5.0 / (1.0 + (peak_vals[0] - peak_vals[-1]) / max_val)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / max_val)
        weights /= weights.sum()

        # Compute subgradient
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                k_int = int(k)
                i_start = max(0, k_int - (n - 1))
                i_end = min(n - 1, k_int)
                j_start = k_int - i_end
                j_end = k_int - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Restart with perturbation
                scale = min(0.3, 0.1 + 0.05 * (stall // patience))
                a_new = best_a * np.exp(rng.normal(0, scale, n))
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                best_a = a_new.copy()
                best_obj = compute_obj(a_new)
                t_k = 1.0
                y = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.4)

        a_prev = a_new.copy()
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        y = y_new.copy()
        t_k = t_new
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Triangle
        x = np.linspace(-1, 1, n)
        a = np.maximum(1.0 - np.abs(x), 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Arcsine-like
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x**2)
        a = a / a.sum()
    elif mode == 5:
        # Random Fourier
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Parabolic
        x = np.linspace(0, 1, n)
        a = 4 * x * (1 - x) + 0.1
        a = a / a.sum()
    elif mode == 7:
        # Plateau
        x = np.linspace(0, 1, n)
        a = np.where((x > 0.25) & (x < 0.75), 1.0, 0.3)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def population_local_search(best_seq, n, budget=200.0, pop_size=5, verbose=False):
    """
    Maintain a small population and apply local search to each member.
    """
    start = time.monotonic()
    rng = np.random.default_rng(7777)

    # Initialize population
    pop = [best_seq.copy()]
    for i in range(1, pop_size):
        blend = 0.1 + 0.15 * i
        a_new = diverse_init(n, rng, mode=i % 8)
        a_new = (1 - blend) * a_new + blend * best_seq
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        pop.append(a_new)

    # Evaluate population
    pop_objs = [compute_obj(p) for p in pop]

    best_overall = pop[0].copy()
    best_overall_obj = pop_objs[0]

    # Iterative local search on best member + periodic reshuffling
    it = 0
    while time.monotonic() - start < budget:
        remaining = budget - (time.monotonic() - start)
        if remaining < 20:
            break

        # Focus on the best member
        best_idx = int(np.argmin(pop_objs))
        a_cur = pop[best_idx].copy()
        obj_cur = pop_objs[best_idx]

        # Apply a quick optimization step
        step_budget = min(30.0, remaining * 0.3)
        if step_budget > 5:
            # Alternate between smooth and LP
            if it % 3 == 0:
                a_new = smooth_optimize_lbfgs(a_cur, time_budget=step_budget, T_start=20.0, T_end=5.0)
            elif it % 3 == 1:
                a_new = lp_alternating_v2(a_cur, time_budget=step_budget)
            else:
                a_new = nesterov_subgradient_v2(a_cur, time_budget=step_budget, K=5)

            obj_new = compute_obj(a_new)
            if obj_new < obj_cur - 1e-13:
                pop[best_idx] = a_new
                pop_objs[best_idx] = obj_new
                if obj_new < best_overall_obj:
                    best_overall_obj = obj_new
                    best_overall = a_new.copy()
                    if verbose:
                        print(f"    Pop it{it}: obj={best_overall_obj:.8f} **")
            it += 1

        # Periodically add a new diverse member
        if it % 4 == 3 and len(pop) < pop_size:
            a_div = diverse_init(n, rng, mode=it % 8)
            blend = 0.3
            a_div = (1 - blend) * a_div + blend * best_overall
            a_div = np.maximum(a_div, 0)
            a_div = a_div / a_div.sum()
            if len(pop) < pop_size:
                pop.append(a_div)
                pop_objs.append(compute_obj(a_div))
            else:
                # Replace worst
                worst_idx = int(np.argmax(pop_objs))
                if compute_obj(a_div) < pop_objs[worst_idx]:
                    pop[worst_idx] = a_div
                    pop_objs[worst_idx] = compute_obj(a_div)

    return best_overall


def tensor_train_init(n, rng, rank=3):
    """Initialize using a low-rank tensor structure."""
    # Create a smooth base with some structure
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    for k in range(rank):
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        a += amp * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.05)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: hybrid of adaptive smooth optimization, LP alternating,
    dual-guided refinement, population local search, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 35.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Quick Nesterov to improve starting point (~80s) ===
    print(f"\n=== Phase 1: Nesterov (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.05, 80.0)
    if p1_budget > 15:
        a_ns = nesterov_subgradient_v2(best_seq, time_budget=p1_budget, K=7, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_ns:.8f} (no improvement)")

    # === Phase 2: Smooth L-BFGS with T annealing (~250s) ===
    print(f"\n=== Phase 2: Smooth L-BFGS ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.18, 250.0)
    if p2_budget > 30:
        # Multiple T schedules
        for (T_s, T_e) in [(100.0, 10.0), (50.0, 5.0), (30.0, 3.0)]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            t_budget = min(p2_budget / 3.0, remaining * 0.5)
            if t_budget < 10:
                continue
            a_sm = smooth_optimize_lbfgs(best_seq, time_budget=t_budget,
                                         T_start=T_s, T_end=T_e, verbose=True)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Smooth T=[{T_s},{T_e}]: obj={best_score:.8f} **")

    # === Phase 3: LP Alternating with golden section (~350s) ===
    print(f"\n=== Phase 3: LP Alternating v2 ===")
    remaining = deadline - time.monotonic()
    p3_total = min(remaining * 0.25, 350.0)

    n_rounds = 3
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p3_total / (n_rounds - rnd), remaining * 0.6)
        if alloc < 20:
            break

        if rnd > 0:
            a_pert = best_seq * np.exp(rng.normal(0, 0.02 * rnd, len(best_seq)))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
        else:
            a_pert = best_seq.copy()

        a_lp = lp_alternating_v2(a_pert, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 4: Dual-guided refinement (~100s) ===
    print(f"\n=== Phase 4: Dual-Guided Refinement ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.07, 100.0)
    if p4_budget > 10:
        a_dg = dual_guided_refine(best_seq, time_budget=p4_budget, verbose=True)
        obj_dg = compute_obj(a_dg)
        if obj_dg < best_score:
            best_score = obj_dg
            best_seq = a_dg.copy()
            print(f"  Dual-guided: obj={best_score:.8f} **")
        else:
            print(f"  Dual-guided: obj={obj_dg:.8f} (no improvement)")

    # === Phase 5: Multi-n exploration (~250s) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(200, n_current - 100),
        max(200, n_current - 50),
        max(200, n_current - 200),
        n_current + 50,
        n_current + 100,
        n_current + 200,
        400, 500, 600, 700, 800, 850, 900, 950, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60 or n_tried >= 8:
            break
        if n_try == n_current:
            continue

        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1

        # Quick optimization pipeline
        alloc1 = min(remaining * 0.08, 40.0)
        if alloc1 > 10:
            a_try = smooth_optimize_lbfgs(a_try, time_budget=alloc1, T_start=30.0, T_end=5.0)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.08, 40.0)
        if alloc2 > 10:
            a_try = lp_alternating_v2(a_try, time_budget=alloc2)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Population-based local search (~150s) ===
    print(f"\n=== Phase 6: Population Local Search ===")
    remaining = deadline - time.monotonic()
    p6_budget = min(remaining * 0.1, 150.0)
    if p6_budget > 30:
        a_pop = population_local_search(best_seq, len(best_seq), budget=p6_budget,
                                        pop_size=4, verbose=True)
        obj_pop = compute_obj(a_pop)
        if obj_pop < best_score:
            best_score = obj_pop
            best_seq = a_pop.copy()
            print(f"  Population: obj={best_score:.8f} **")
        else:
            print(f"  Population: obj={obj_pop:.8f} (no improvement)")

    # === Phase 7: Aggressive smooth optimization on best (~150s) ===
    print(f"\n=== Phase 7: Aggressive Smooth Polish ===")
    remaining = deadline - time.monotonic()
    p7_budget = min(remaining * 0.12, 150.0)
    if p7_budget > 20:
        # Very fine T for final polish
        a_final = smooth_optimize_lbfgs(best_seq, time_budget=p7_budget * 0.6,
                                         T_start=20.0, T_end=2.0, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Fine smooth: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_lp = lp_alternating_v2(best_seq, time_budget=min(remaining * 0.5, 60.0))
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

    # === Phase 8: Diverse restarts (~100s) ===
    print(f"\n=== Phase 8: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_restarts = 5
    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Different initialization strategies
        if r < 3:
            a_r = diverse_init(n_best, rng, mode=r * 2 + 1)
            blend = 0.15 + 0.1 * r
            a_r = (1 - blend) * a_r + blend * best_seq
        else:
            # Tensor-train-like init
            a_r = tensor_train_init(n_best, rng, rank=3 + r)
            blend = 0.25
            a_r = (1 - blend) * a_r + blend * best_seq

        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.15, 50.0)
        if restart_budget < 15:
            break

        # Quick optimization
        a_opt = smooth_optimize_lbfgs(a_r, time_budget=restart_budget * 0.5,
                                       T_start=50.0, T_end=10.0)
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_opt = lp_alternating_v2(a_opt, time_budget=remaining * 0.4)

        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 9: Final intensive polish (~60s) ===
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        a_fp = smooth_optimize_lbfgs(best_seq, time_budget=min(remaining * 0.4, 40.0),
                                      T_start=15.0, T_end=2.0)
        obj_fp = compute_obj(a_fp)
        if obj_fp < best_score:
            best_score = obj_fp
            best_seq = a_fp.copy()
            print(f"  Final smooth: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_fp2 = lp_alternating_v2(best_seq, time_budget=min(remaining * 0.6, 30.0))
        obj_fp2 = compute_obj(a_fp2)
        if obj_fp2 < best_score:
            best_score = obj_fp2
            best_seq = a_fp2.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_fp3 = nesterov_subgradient_v2(best_seq, time_budget=min(remaining * 0.5, 20.0), K=10)
        obj_fp3 = compute_obj(a_fp3)
        if obj_fp3 < best_score:
            best_score = obj_fp3
            best_seq = a_fp3.copy()
            print(f"  Final NS: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
