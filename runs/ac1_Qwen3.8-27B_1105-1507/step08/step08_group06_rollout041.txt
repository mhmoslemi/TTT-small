Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2n * max_conv / (sum)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a, g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    
    from scipy.optimize import linprog
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g_val))
        if gs > 1e-10:
            g_val = g_val / gs
            M_val = float(res.x[n])
            conv_val = A @ g_val
            active = np.where(conv_val >= M_val - 1e-6)[0]
            return g_val, M_val, active
    return None, None, None

                    else:
                        a[i] += delta
                        a[target] -= delta

"""
Optimized step function search using log-sum-exp gradient smoothing,
LP alternating with symmetry exploitation, adaptive line search,
and dual-guided perturbation.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv / (sum)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def build_sym_conv_matrix(a):
    """Build matrix for symmetric g: A @ g_sym = conv(a, g) where g is symmetric.
    g has n_half free vars: g[0..n_half-1] with g[n-1-i] = g[i].
    """
    n = len(a)
    m = 2 * n - 1
    n_half = (n + 1) // 2
    A = np.zeros((m, n_half))
    for i in range(n_half):
        # g[i] = g[n-1-i] (same value)
        # Contribution to conv[k] from g[i]: a[k-i] for valid k-i
        # Contribution to conv[k] from g[n-1-i]: a[k-(n-1-i)] for valid
        for k in range(m):
            val = 0.0
            if 0 <= k - i < n:
                val += a[k - i]
            j2 = n - 1 - i
            if 0 <= k - j2 < n:
                val += a[k - j2]
            A[k, i] = val
    return A


def solve_lp_full(a):
    """Solve LP: min M s.t. conv(a, g) <= M, sum(g)=1, g>=0. Returns (g, M, active)."""
    n = len(a)
    A = build_conv_matrix(a)
    from scipy.optimize import linprog
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0 and res.x is not None:
        g_val = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g_val))
        if gs > 1e-10:
            g_val = g_val / gs
            M_val = float(res.x[n])
            conv_val = A @ g_val
            active = np.where(conv_val >= M_val - 1e-5)[0]
            return g_val, M_val, active
    return None, None, None


def solve_lp_sym(a):
    """Solve LP with symmetric g constraint. Returns (g_symmetric, M, active)."""
    n = len(a)
    n_half = (n + 1) // 2
    A = build_sym_conv_matrix(a)
    m = 2 * n - 1
    from scipy.optimize import linprog
    c = np.zeros(n_half + 1)
    c[-1] = 1.0

    A_ub[:, :n_half] = A
    A_ub[:, n_half] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n_half + 1))
    A_eq[0, :n_half] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n_half + [(None, None)]

def lse_gradient(a, tau):
    """
    Compute gradient of 2n * tau * log(sum exp(conv_k/tau)) w.r.t. a.
    Uses FFT for the cross-correlation part. Returns (grad, conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    scaled = (conv - max_conv) / tau
    exp_scaled = np.exp(scaled)
    softmax = exp_scaled / exp_scaled.sum()

    # grad[i] = 2 * sum_k softmax[k] * a[k-i] (with 0<=k-i<n)
    # This is a cross-correlation: grad = 2 * cross_corr(softmax, a)
    # cross_corr[i] = sum_k softmax[k] * a[k-i]
    # Using FFT: cross_corr = ifft(fft(softmax) * conj(fft(a)))
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    f_s = rfft(softmax, size)
    f_a = rfft(a, size)
    cross_corr = irfft(f_s * np.conj(f_a), size)[:2 * n - 1]
    # cross_corr[m] = sum_j softmax[j] * a[j - m + (n-1)]... 
    # Let me verify: ifft(fft(s) * conj(fft(a)))[m] = sum_j s[j] * conj(a[j-m+N])
    # For real: = sum_j s[j] * a[(j-m) mod N]
    # This is the circular cross-correlation. For linear, we need zero-padding.
    # With size >= (2n-1) + n - 1 = 3n - 2, we have enough zero-padding.
    # cross_corr[m] = sum_j softmax[j] * a[j - m] (valid when 0 <= j-m < n)
    # We want: grad[i] = 2 * sum_k softmax[k] * a[k-i]
    # Let j=k, m=i: cross_corr[i] = sum_j softmax[j] * a[j-i] = sum_k softmax[k] * a[k-i]
    # Yes! So grad = 2 * cross_corr[:n]
    grad = 2.0 * cross_corr[:n]
    return grad, conv


def lse_smoothed_optimize(a, time_budget=300.0, verbose=False):
    """Optimize using animated log-sum-exp smoothing."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    tau_start = max_conv * 0.5
    tau_end = max_conv * 0.005

    # Animated temperature optimization
    n_outer = 60
    n_inner = 8

    for outer in range(n_outer):
        if time.monotonic() - start > time_budget:
            break
        progress = outer / max(1, n_outer - 1)
        # Non-linear annealing (faster at start)
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.7)
        tau = max(tau, 1e-12)

        for inner in range(n_inner):
            if time.monotonic() - start > time_budget:
                break
            grad, conv = lse_gradient(a, tau)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            # Project gradient to sum=0 hyperplane
            grad_c = grad - np.mean(grad)
            # Adaptive step size
            lr = 0.5 / (1.0 + np.linalg.norm(grad_c))
            # Backtracking line search
            best_local = a
            best_local_obj = compute_obj(a)
            for shrink in range(10):

                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_local = a_try
                    break
            a = best_local
            if best_local_obj < best_obj:
                best_obj = best_local_obj
                best_a = a.copy()
                if verbose and outer % 10 == 0:
                    elapsed = time.monotonic() - start
                    print(f"    LSE o={outer} tau={tau:.6f} obj={best_obj:.8f} t={elapsed:.0f}s")

                at/=st2; ot=ob(at)
                if ot<bo-1e-14:
                    bo=ot; best=at.copy(); a=at; ok=True; break
            if not ok: a=best.copy()
    return best

def lap(a,tb,sy=False):
    n=len(a); a=np.maximum(a.astype(np.float64),0); s=float(np.sum(a))
    if s<1e-12: return a
    a/=s; best=a.copy(); bo=ob(a); st=time.monotonic()
    stall=0; rng=np.random.default_rng(99)
    while time.monotonic()-st<tb:
        g,M,act=lp(a,sy)
        if g is None: break
        d=g-a; dn=np.linalg.norm(d)
        if dn<1e-12: break
        bt=None; bto=bo
        for t in [1.,.9,.8,.7,.6,.5,.4,.3,.2,.1]:
            at=a+t*d; at=np.maximum(at,0); st2=float(np.sum(at))
            if st2<1e-10: continue
            at/=st2; ot=ob(at)
            if ot<bto-1e-14:
                bto=ot; bt=t
                if t>=.7: break
        if bt is not None:
            a=a+bt*d; a=np.maximum(a,0); a/=float(np.sum(a))
            bo=bto; best=a.copy(); stall=0
        else:
            stall+=1
            if stall>=8:
                if act is not None and len(act)>0:
                    k=int(act[0]); ci=np.zeros(n)
                    for j in range(n):
                        kj=k-j
                        if 0<=kj<n: ci[j]=a[j]*a[kj]
                    tj=np.argsort(ci)[::-1][:5]
                    for j in tj:
                        d2=0.02*a[j]
                        if d2>0:
                            tgt=(k-int(j)+n//2)%n
                            a[j]-=d2; a[tgt]+=d2
                    a=np.maximum(a,0); a/=float(np.sum(a))
                    if ob(a)>=bo: a=best.copy()
                else:
                    sc=0.02+0.01*min(stall//8,5)
                    a=best*np.exp(rng.normal(0,sc,n))
                    a=np.maximum(a,0); a/=float(np.sum(a))
                stall=0
    return best

def cd(a,tb):
    n=len(a); a=np.maximum(a.astype(np.float64),0)
    s=float(np.sum(a))
    if s<1e-12: return a
    a/=s; best=a.copy(); bo=ob(a); st=time.monotonic()
    for it in range(300):
        if time.monotonic()-st>tb: break
        cv=cc(a); ks=int(np.argmax(cv)); imp=False
        ci=np.zeros(n)
        for j in range(n):
            kj=ks-j
            if 0<=kj<n: ci[j]=a[j]*a[kj]
        tj=np.argsort(ci)[::-1][:10]
        for i in tj:
            if time.monotonic()-st>tb: break
            for fo in [n//3,n//2,2*n//3]:
                tg=(i+fo)%n
                if tg==i: continue
                for fr in [0.02,0.05,0.1]:
                    d=fr*a[i]
                    if d<=0: continue
                    a[i]-=d; a[tg]+=d
                    no=ob(a)
                    if no<bo-1e-14:
                        bo=no; best=a.copy(); imp=True; break
                    else:
                        a[i]+=d; a[tg]-=d
                if imp: break
            if imp: break
        if not imp and it>80: break
    return best

def sym(a):
    return (a+a[::-1].copy())/2.

def resamp(a,nn):
    n=len(a)
    if n==nn: return a.copy()
    x0=np.linspace(0,1,n); x1=np.linspace(0,1,nn)
    return np.maximum(np.interp(x1,x0,a),0)

def propose_candidate(seed=42,budget_s=1500):
    st0=time.monotonic(); dl=st0+float(budget_s)-30.
    best=None; bs=np.inf; rng=np.random.default_rng(seed)

    # Load parent
    try:
        pa=np.asarray(height_sequence_1,dtype=np.float64)
        pa=np.clip(pa,0,None)
        if pa.size>0 and np.isfinite(pa).all():
            s=float(np.sum(pa))
            if s>0.01:
                best=pa/s; bs=ob(best)
                print(f"Parent n={best.size} obj={bs:.8f}")
    except: pass

    if best is None:
        n=800; h=n//2
        x=np.linspace(0,np.pi,h)
        hs=0.5*(1+np.cos(x))
        best=np.concatenate([hs,hs[::-1]])
        if len(best)>n: best=best[:n]
        best/=best.sum(); bs=ob(best)
        print(f"Init n={len(best)} obj={bs:.8f}")

    # Try symmetrizing
    sa=sym(best)/float(np.sum(sym(best)))
    so=ob(sa)
    if so<bs:
        bs=so; best=sa.copy()
        print(f"Sym: obj={bs:.8f} **")

    n_par=len(best)

    # Diverse inits
    inits=[np.ones(n_par)/n_par]
    x=np.linspace(0,np.pi,n_par)
    inits.append(0.5*(1+np.cos(x)))
    c=n_par/2.; sg=n_par/4.
    inits.append(np.exp(-0.5*((np.arange(n_par)-c)/sg)**2))
    h=n_par//2+1; hr=rng.random(h)+0.1
    sr=np.concatenate([hr[:n_par//2],hr[:n_par//2][::-1]])
    inits.append(sr)
    for ini in inits:
        ini=np.maximum(ini,0); s=float(np.sum(ini))
        if s>0:
            ini/=s; oi=ob(ini)
            if oi<bs:
                bs=oi; best=ini.copy()
                print(f"  Init: obj={bs:.8f} **")

    a=best.copy()
    print(f"Start: obj={bs:.8f} n={n_par}")

    # Phase 1: LSE smoothing
    rem=dl-time.monotonic()
    if rem>60:
        ab=min(300.,rem*0.4)
        print(f"\nP1: LSE ({ab:.0f}s)")
        an=lopt(a,ab)
        ao=ob(an)
        if ao<bs:
            bs=ao; best=an.copy(); a=an.copy()
            print(f"  ** obj={bs:.8f}")
        else: print(f"  obj={ao:.8f}")

    # Phase 2: LP alternating (symmetric LP)
    rem=dl-time.monotonic()
    if rem>60:
        ab=min(400.,rem*0.45)
        print(f"\nP2: LP-sym ({ab:.0f}s)")
        an=lap(a,ab,sy=True)
        ao=ob(an)
        if ao<bs:
            bs=ao; best=an.copy(); a=an.copy()
            print(f"  ** obj={bs:.8f}")
        else: print(f"  obj={ao:.8f}")

    # Phase 3: LP alternating (full)
    rem=dl-time.monotonic()
    if rem>60:
        ab=min(300.,rem*0.3)
        print(f"\nP3: LP-full ({ab:.0f}s)")
        an=lap(a,ab,sy=False)
        ao=ob(an)
        if ao<bs:
            bs=ao; best=an.copy(); a=an.copy()
            print(f"  ** obj={bs:.8f}")
        else: print(f"  obj={ao:.8f}")

    # Phase 4: LSE again (refined)
    rem=dl-time.monotonic()
    if rem>40:
        ab=min(150.,rem*0.2)
        print(f"\nP4: LSE refine ({ab:.0f}s)")
        an=lopt(a,ab)
        ao=ob(an)
        if ao<bs:
            bs=ao; best=an.copy(); a=an.copy()
            print(f"  ** obj={bs:.8f}")
        else: print(f"  obj={ao:.8f}")

    # Phase 5: Coordinate descent
    rem=dl-time.monotonic()
    if rem>30:
        ab=min(80.,rem*0.15)
        print(f"\nP5: CD ({ab:.0f}s)")
        an=cd(a,ab)
        ao=ob(an)
        if ao<bs:
            bs=ao; best=an.copy(); a=an.copy()
            print(f"  ** obj={bs:.8f}")
        else: print(f"  obj={ao:.8f}")

    # Phase 6: Length sweep
    rem=dl-time.monotonic()
    if rem>60:
        nb=len(best)
        opts=sorted(set([max(200,nb-100),max(200,nb-50),nb,nb+50,nb+100,nb+200]))
        opts=[x for x in opts if 200<=x<=3000]
        print(f"\nP6: Length sweep")
        for nt in opts:
            rem=dl-time.monotonic()
            if rem<30: break
            if nt==nb: continue
            at=resamp(best,nt)/max(1e-10,float(np.sum(resamp(best,nt))))
            at=np.maximum(at,0); at/=float(np.sum(at))
            an=lap(at,min(20.,rem*0.25),sy=True)
            ao=ob(an)
            if ao<bs:
                bs=ao; best=an.copy(); a=an.copy()
                print(f"  ** n={nt} obj={bs:.8f}")
            else: print(f"  n={nt} obj={ao:.8f}")

    # Phase 7: Final polish
    rem=dl-time.monotonic()
    if rem>20:
        print(f"\nP7: Final")
        af=best.copy(); nf=len(af)
        while rem>15:
            ab=min(rem*0.4,40.)
            an=lap(af,ab,sy=True)
            ao=ob(an)
            if ao<bs:
                bs=ao; best=an.copy(); af=an.copy()
                print(f"  obj={bs:.8f} **")
            else:
                sc=0.01; af=best*np.exp(rng.normal(0,sc,nf))
                af=np.maximum(af,0); af/=float(np.sum(af))
            rem=dl-time.monotonic()

    # Validate
    if best is None:
        best=np.ones(200); bs=ob(best)
    res=[float(max(0.,min(1000.,x))) for x in best]
    v=evaluate_sequence(res)
    if v<np.inf: bs=v
    else: res=[1.]*100; bs=2.
    print(f"\nFINAL: n={len(res)} score={bs:.8f} t={time.monotonic()-st0:.1f}s")
    return res
