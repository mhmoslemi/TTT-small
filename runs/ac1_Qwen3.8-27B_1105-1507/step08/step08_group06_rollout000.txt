"""
Improved step function search using LP alternating with Nesterov momentum,
adaptive backtracking line search, subgradient-based perturbation, and
symmetry exploitation. Focuses 75%+ of time on the core LP loop.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import linprog


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_for_response(a, prefer_solvers=None):
    """
    Solve LP: min M s.t. conv(a, g) <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt, active_indices).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None, None
    a = a / s
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = prefer_solvers if prefer_solvers else [cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                    conv_val = A @ g_val
                    active = np.where(conv_val >= M_val - 1e-7)[0]
                    return g_val, M_val, active
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(res.x[n])
                conv_val = A @ g_val
                active = np.where(conv_val >= M_val - 1e-7)[0]
                return g_val, M_val, active
    except Exception:
        pass

    return None, None, None

                    a = a / s
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                    else:
                        a[int(i)] += delta
                        a[target] -= delta
                        a = a / np.sum(a)

        if not improved:
            # Random pair swap
            for _ in range(10):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                delta = 0.005 * min(a[i], a[j])
                if delta <= 1e-10:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s < 1e-10:
                    break
                a = a / s
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break

    return best_a


def generate_initialization(kind, n, rng):
    """Generate diverse initialization."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'bimodal':
        a = np.ones(n) * 0.3
        a[n//3] = 2.0
        a[2*n//3] = 2.0
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'triangular':
        a = np.minimum(np.arange(n), n - 1 - np.arange(n)) / n
        return a / np.sum(a)
    elif kind == 'flat_top':
        a = np.ones(n) * 0.2
        edge = max(1, n // 8)
        a[edge:-edge] = 1.0
        a[:edge] = np.linspace(0.2, 1.0, edge)
        a[-edge:] = np.linspace(1.0, 0.2, edge)
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating with Nesterov momentum, subgradient polish,
    coordinate descent, and length sweep.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    n_default = 800
    if best_seq is None:
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: Quick diverse initializations (30s) ===
    remaining = deadline - time.monotonic()
    init_budget = min(30.0, remaining * 0.03)
    print(f"\n=== Phase 1: Initializations (budget={init_budget:.0f}s) ===")

    init_kinds = ['cosine', 'gaussian', 'bimodal', 'triangular', 'flat_top', 'sine_taper']
    for kind in init_kinds:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        a_init = generate_initialization(kind, n_base, rng)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init '{kind}': obj={obj_init:.8f} **")
        else:
            print(f"  Init '{kind}': obj={obj_init:.8f}")

    # Also try parent with small perturbation
    if parent is not None:
        a_pert = parent * np.exp(rng.normal(0, 0.05, n_base))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / np.sum(a_pert)
        obj_pert = compute_obj(a_pert)
        if obj_pert < best_score:
            best_score = obj_pert
            best_seq = a_pert.copy()
            print(f"  Perturbed parent: obj={obj_pert:.8f} **")

    print(f"  Best after Phase 1: obj={best_score:.8f}")

    # === Phase 2: Main LP Alternating with Nesterov (majority of time) ===
    remaining = deadline - time.monotonic()
    lp_budget = remaining * 0.65
    print(f"\n=== Phase 2: Nesterov LP Alternating (budget={lp_budget:.0f}s, n={n_base}) ===")

    a = best_seq.copy()

    # Multiple rounds with different perturbations between
    n_rounds = 5
    per_round = lp_budget / n_rounds
    for r in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_round, remaining * 0.7)
        print(f"  LP Round {r} (alloc={alloc:.0f}s):")

        a_new = nesterov_lp_alternating(a, time_budget=alloc, rng=rng, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  ** Round {r}: obj={best_score:.8f} **")
        else:
            print(f"  Round {r}: obj={obj:.8f}")
            # Perturb for next round
            if r < n_rounds - 1:
                scale = 0.03 + 0.01 * r
                a = best_seq * np.exp(rng.normal(0, scale, n_base))
                a = np.maximum(a, 0)
                a = a / np.sum(a)

    # === Phase 3: Subgradient Polish (10% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        sg_budget = min(remaining * 0.1, 80.0)
        print(f"\n=== Phase 3: Subgradient Polish (budget={sg_budget:.0f}s) ===")
        a_sg = subgradient_polish(best_seq, time_budget=sg_budget, rng=rng)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    # === Phase 4: Coordinate Polish (8% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        cd_budget = min(remaining * 0.08, 60.0)
        print(f"\n=== Phase 4: Coordinate Polish (budget={cd_budget:.0f}s) ===")
        a_cd = coordinate_polish(best_seq, time_budget=cd_budget, rng=rng)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coord: obj={best_score:.8f} **")
        else:
            print(f"  Coord: obj={obj_cd:.8f}")

    # === Phase 5: Length Sweep (10% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        ls_budget = min(remaining * 0.1, 120.0)
        n_best = len(best_seq)
        n_options = sorted(set([
            max(200, n_best - 100),
            max(200, n_best - 50),
            n_best + 50,
            n_best + 100,
            n_best + 200,
            min(3000, n_best + 400),
        ]))
        n_options = [n for n in n_options if 200 <= n <= 4000]

        print(f"\n=== Phase 5: Length Sweep (budget={ls_budget:.0f}s) ===")
        per_n = ls_budget / max(1, len(n_options))

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            if n_try == n_best:
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, n_try)

            alloc = min(per_n, remaining * 0.5)
            a_opt = nesterov_lp_alternating(a_try, time_budget=alloc * 0.7, rng=rng)
            a_opt = subgradient_polish(a_opt, time_budget=alloc * 0.2, rng=rng)
            obj_try = compute_obj(a_opt)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final LP polish with remaining time ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 6: Final Polish (remaining={remaining:.0f}s) ===")
    a_final = best_seq.copy()

    while remaining > 15:
        alloc = min(remaining * 0.4, 60.0)
        a_new = nesterov_lp_alternating(a_final, time_budget=alloc, rng=rng)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_final = a_new.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            scale = 0.01
            a_final = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
            a_final = np.maximum(a_final, 0)
            a_final = a_final / np.sum(a_final)
            print(f"  Final LP: obj={obj:.8f} (perturbing)")

        # Quick subgradient
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_sg = subgradient_polish(a_final, time_budget=min(remaining * 0.3, 20.0), rng=rng)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                a_final = a_sg.copy()
                print(f"  Final SG: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()

    # === Final coordinate polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_cd = coordinate_polish(best_seq, time_budget=min(remaining * 0.5, 20.0), rng=rng)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
