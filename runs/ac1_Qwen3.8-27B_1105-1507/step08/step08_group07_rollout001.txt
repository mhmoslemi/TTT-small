"""
Hybrid optimizer combining:
1. Smooth gradient descent (log-sum-exp approximation of max) with annealing
2. LP alternating for local refinement
3. Multi-n exploration
4. Simulated annealing restarts
5. Coordinate descent polish

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smooth_obj_and_grad(a, T=0.01):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf, np.zeros(n)
    a_norm = a / s
    conv = compute_conv(a_norm)
    
    max_c = float(np.max(conv))
    exp_c = np.exp((conv - max_c) / T)
    sum_exp = float(np.sum(exp_c))
    weights = exp_c / sum_exp
    smooth_max = T * (max_c / T + np.log(sum_exp))
    
    obj = 2.0 * n * smooth_max
    
    # Gradient computation
    corr = np.correlate(weights, a_norm, mode='full')
    grad = 4.0 * n * corr[:n]
    
    return obj, grad

def gradient_descent_smooth(a, time_budget=100.0, verbose=False):

    if s < 1e-10:

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a):
    n = len(a)
    A = build_conv_matrix(a)
    
    # Try cvxpy first
    try:
        z = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        prob = cp.Problem(cp.Minimize(M), [A @ z <= M, cp.sum(z) == 1])
        for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    z_val = np.maximum(z.value, 0)
                    zs = float(np.sum(z_val))
                    if zs > 1e-10:
                        z_val = z_val / zs
                        dual_y = None
                        try:
                            dual_y = np.maximum(np.array(prob.constraints[0].dual_value), 0)
                        except:
                            pass
                        return z_val, float(M.value), dual_y
            except:
                continue
    except:
        pass
    
    # Fallback to scipy
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        if res.status == 0:

            zs = float(np.sum(z_val))

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
