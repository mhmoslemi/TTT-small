Multi-strategy optimizer combining multiplicative KKT-based updates,
smooth-max gradient descent, LP polishing, and peak-directed coordinate descent.
Key innovation: Uses the KKT condition (weighted autocorrelation constant on support)
to drive a proportional/multiplicative update that directly targets the optimum.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def cross_corr_lambda_a(lambda_vec, a):
    """Compute h[j] = sum_k lambda[k] * a[k-j] for j=0..n-1."""
    n = len(a)
    m = len(lambda_vec)
    h = np.zeros(n)
    for j in range(n):
        k_min = max(0, j)
        k_max = min(m - 1, j + n - 1)
        if k_min <= k_max:
            length = k_max - k_min + 1
            h[j] = np.dot(lambda_vec[k_min:k_min+length], a[0:length])
    return h

                               damping=0.5, verbose=False):

    KKT-based multiplicative update.
    At optimum: sum_k lambda_k * a[k-j] = constant for j in support(a).
    Update: a_j <- a_j / h_j, normalized.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    T = T_init
    stall = 0
    
    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Soft assignment of lambda to peaks
        # Sharpen with temperature
        lambda_vec = np.exp((conv - max_conv) / T)
        lambda_sum = float(np.sum(lambda_vec))
        if lambda_sum < 1e-30:
            lambda_vec = np.zeros_like(conv)
            lambda_vec[np.argmax(conv)] = 1.0
        else:
            lambda_vec /= lambda_sum

        h = cross_corr_lambda_a(lambda_vec, a)

        s_new = float(np.sum(a_new))
        if s_new < 1e-10:

        a_new = a_new / s_new

            best_a = a_new.copy()
            stall = 0
        else:
            a = a_new  # still update to explore
            stall += 1
            if stall > 30:

                damping = max(0.1, damping * 0.8)
                T = min(T_init, T * 1.1)
                stall = 0
                if T > T_init * 0.5:
                    break
        
        # Anneal temperature
        if it > 10:
            T = max(T_min, T_init * (0.995 ** it))
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    MKKT it{it}: obj={best_obj:.8f}, T={T:.5f}, damping={damping:.3f}, time={elapsed:.1f}s")
    
    return best_a


def smooth_max_gradient_descent(a, time_budget=60.0, T_init=0.001, T_min=0.0001):
    """Gradient descent using smooth max with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    T = T_init
    lr = 0.001
    momentum = 0.9
    v = np.zeros(n)
    stall = 0
    best_lr = lr
    
    max_iters = 10000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = float(np.sum(exp_vals))
        if total < 1e-30:
            break
        weights = exp_vals / total
        
        # Gradient: grad[i] = sum_j weights[j] * a[j-i] (cross-correlation)
        # This is the same as cross_corr_lambda_a(weights, a)
        grad = cross_corr_lambda_a(weights, a)
        grad_centered = grad - float(np.sum(grad) * a)  # project out sum=1 direction
        
        # Actually, the gradient of max_k conv_k w.r.t. a_j is sum_k w_k * a[k-j]
        # But we need to account for the constraint sum(a)=1
        # The projected gradient is: grad - mean(grad) (for the equality constraint)
        # Wait, more precisely: minimize f(a) = max_k (a*a)[k] s.t. sum(a)=1, a>=0
        # Using Lagrange: f(a) - lambda*(sum(a)-1)
        # Gradient: grad_j = sum_k w_k * a[k-j] - lambda
        # At optimum, this = 0 for j in support
        # For the step: a_new_j = a_j - lr * (sum_k w_k * a[k-j] - mean(grad))
        # Because the constraint is sum(a)=1, the "mean" is the right projection
        
        grad_mean = float(np.mean(grad))
        grad_proj = grad - grad_mean
        
        v = momentum * v - lr * grad_proj
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Occasionally increase lr
            if stall == 0 and it % 50 == 0:
                lr = min(lr * 1.05, 0.01)
        else:
            a = a_new
            stall += 1
            if stall > 20:
                lr *= 0.7
                if lr < 1e-7:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
        
        if it > 20:
            T = max(T_min, T_init * (0.997 ** it))
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def solve_lp_for_direction(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    try:
        from scipy.optimize import linprog
    except ImportError:
        return None, None
    
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 30})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based polishing using alternating direction."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp_for_direction(a)
        if g_lp is None:
            break
        
        # Line search
        best_t = 0.0
        best_t_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t > 0 and best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            a = (1 - best_t) * a + best_t * g_lp
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_a = a.copy()
            if verbose and it % 10 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            # Try smaller steps
            improved = False
            for t in [0.01, 0.05]:
                a_try = (1 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    a = a_try
                    best_a = a.copy()
                    improved = True
                    break
            if not improved:
                break
    
    return best_a


def peak_directed_cd(a, time_budget=60.0, verbose=False):
    """Coordinate descent that focuses on reducing the peak convolution values."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(hash(n) % (2**32))
    
    delta = 1e-3
    pass_count = 0
    
    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        peak_idx = int(np.argmax(conv))
        
        # Sort coordinates by their contribution to the peak
        contributions = np.zeros(n)
        for j in range(n):
            if 0 <= peak_idx - j < 2*n - 1:
                contributions[j] = a[j] * a[peak_idx - j] if (peak_idx - j) < n else 0
        
        order = np.argsort(contributions)[::-1]
        
        for i in order:
            if time.monotonic() - start > time_budget:
                    break
            if a[i] < 1e-10:
                continue
            
            # Try decreasing
            for d in [delta, delta*0.1, delta*0.01]:
                a_try = a.copy()
                a_try[i] = max(0, a_try[i] - d)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            
            if improved:
                continue
            
            # Try increasing (to shift the peak)
            a_try = a.copy()
            a_try[i] += delta
            st = float(np.sum(a_try))
            if st > 1e-10:
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
        
        if not improved:
            delta *= 0.5
            if delta < 1e-8:
                if pass_count > 5:
                    break
        
        if verbose and pass_count % 3 == 0:
            print(f"    PCDP pass {pass_count}: obj={best_obj:.8f}, delta={delta:.2e}")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

Wait, I realize there might be an issue with my `cross_corr_lambda_a` function. Let me double-check the indexing.

We have:
- a of length n, indices 0..n-1
- lambda_vec of length 2n-1, indices 0..2n-2
- We want h[j] = sum_k lambda_vec[k] * a[k-j] for j=0..n-1

For h[j], a[k-j] is non-zero only when 0 <= k-j <= n-1, i.e., j <= k <= j+n-1.
Also 0 <= k <= 2n-2.

So k ranges from max(0, j) to min(2n-2, j+n-1).

For each valid k, the term is lambda_vec[k] * a[k-j].

Let me verify: for j=0, k ranges from 0 to min(2n-2, n-1) = n-1.
h[0] = sum_{k=0}^{n-1} lambda_vec[k] * a[k] ✓

For j=n-1, k ranges from n-1 to min(2n-2, 2n-2) = 2n-2.
h[n-1] = sum_{k=n-1}^{2n-2} lambda_vec[k] * a[k-(n-1)] = sum_{k=n-1}^{2n-2} lambda_vec[k] * a[k-n+1]
Let m = k-n+1, then m ranges from 0 to n-1.
h[n-1] = sum_{m=0}^{n-1} lambda_vec[m+n-1] * a[m] ✓

Good, the formula is correct.

Now let me also think about whether the multiplicative update is actually correct. The KKT condition says:
At optimum, for j in support(a): sum_k lambda_k * a[k-j] = lambda_sum (constant)

The multiplicative update a_j <- a_j / h_j where h_j = sum_k lambda_k * a[k-j] would drive h_j toward a constant (since if h_j > constant, a_j decreases; if h_j < constant, a_j increases).

"""
Multi-strategy optimizer combining KKT-based multiplicative updates,
smooth-max gradient descent, LP polishing, and peak-directed coordinate descent.
Key innovation: Uses the KKT condition (weighted autocorrelation constant on support)
to drive a proportional/multiplicative update that directly targets the optimum.
Explores diverse initializations and multiple sequence lengths.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def cross_corr_lambda_a(lambda_vec, a):
    """Compute h[j] = sum_k lambda[k] * a[k-j] for j=0..n-1."""
    n = len(a)
    m = len(lambda_vec)
    h = np.zeros(n)
    for j in range(n):
        k_min = max(0, j)
        k_max = min(m - 1, j + n - 1)
        if k_min <= k_max:
            length = k_max - k_min + 1
            h[j] = np.dot(lambda_vec[k_min:k_min + length], a[0:length])
    return h


def cross_corr_fft(lambda_vec, a):
    """Efficient cross-correlation via FFT. h[j] = sum_k lambda[k] * a[k-j]."""
    n = len(a)
    m = len(lambda_vec)
    size = 1
    while size < m + n - 1:
        size <<= 1
    F_lam = rfft(lambda_vec, size)
    F_a = rfft(a, size)
    # Cross-correlation: R[m] = sum_n x[n+m] * y[n]
    # In FFT: R = ifft(FFT(x) * conj(FFT(y)))
    # But we want h[j] = sum_k lambda[k] * a[k-j] = sum_n lambda[n+j] * a[n]
    # This is R(lambda, a)[j] = sum_n lambda[n+j] * a[n]
    # Standard cross-correlation: R[m] = sum_n x[n+m] * conj(y[n])
    # So R(lambda, a)[j] with x=lambda, y=a gives sum_n lambda[n+j] * a[n] = h[j]
    # FFT formula: R = ifft(FFT(x) * conj(FFT(y)))
    # But this gives R for m = 0, 1, ..., size-1 (circular)
    # We need the linear part: m = 0, 1, ..., m-1 (i.e., 0 to len(lambda)-1)
    # Actually for cross-correlation with x length m1 and y length m2:
    # R[m] is defined for m = -(m2-1) to (m1-1)
    # The FFT gives us the circular version; we need to extract the right part.
    # Let's use: h = ifft(FFT(lambda_reversed) * FFT(a)) ... no.
    # Let me just use the direct formula with the FFT trick:
    # h[j] = sum_k lambda[k] * a[k-j]
    # = sum_k lambda[k] * a[k-j] (with a[k-j]=0 if out of range)
    # Define b[t] = a[t] for t in [0,n-1], 0 otherwise. b has length n.
    # h[j] = sum_k lambda[k] * b[k-j]
    # This is (lambda * b)[j] in the "correlation" sense = sum_k lambda[k] * b[k-j]
    # In terms of convolution: sum_k lambda[k] * b[k-j] = conv(lambda, b_rev)[j + n - 1]
    # where b_rev[t] = b[n-1-t]
    b_rev = a[::-1]
    F_lr = rfft(lambda_vec, size)
    F_br = rfft(b_rev, size)
    conv_full = irfft(F_lr * F_br, size)
    # (lambda * b_rev)[t] = sum_k lambda[k] * b_rev[t-k]
    # = sum_k lambda[k] * a[n-1-(t-k)] = sum_k lambda[k] * a[n-1-t+k]
    # We want: h[j] = sum_k lambda[k] * a[k-j]
    # Set n-1-t+k = k-j => n-1-t = -j => t = n-1+j
    # So h[j] = (lambda * b_rev)[n-1+j]
    h = conv_full[n - 1:n - 1 + n]
    return h


def multiplicative_kkt_update(a, time_budget=100.0, T_init=0.05, T_min=0.001,
                               damping=0.5, use_fft=False, verbose=False):
    """
    KKT-based multiplicative update.
    At optimum: sum_k lambda_k * a[k-j] = constant for j in support(a).
    Update: a_j <- a_j * (mean_h / h_j), damped and normalized.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    stall = 0
    best_T = T_init

    max_iters = 10000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Soft assignment of lambda to peaks
        lambda_vec = np.exp((conv - max_conv) / T)
        lambda_sum = float(np.sum(lambda_vec))
        if lambda_sum < 1e-30:
            lambda_vec = np.zeros_like(conv)
            lambda_vec[np.argmax(conv)] = 1.0
        else:
            lambda_vec /= lambda_sum

        # Compute h = cross-correlation of lambda with a
        if use_fft:
            h = cross_corr_fft(lambda_vec, a)
        else:
            h = cross_corr_lambda_a(lambda_vec, a)

        # Normalize h by its mean for stable update
        h_pos = h[h > 1e-15]
        if len(h_pos) == 0:
            break
        h_mean = float(np.mean(h_pos))

        if h_mean < 1e-15:
            break

        # Multiplicative update with damping
        ratio = np.where(h > 1e-15, h_mean / h, 1.0)
        a_new = (1.0 - damping) * a + damping * a * ratio
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            # Record good T
            best_T = T
        else:
            a = a_new
            stall += 1
            if stall > 40:
                # Perturb to escape local minimum
                rng_local = np.random.default_rng(hash((it, n)) % (2**32))
                scale = 0.01
                a_pert = best_a * np.exp(rng_local.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                sp = float(np.sum(a_pert))
                if sp > 1e-10:
                    a = a_pert / sp
                # Widen temperature
                T = min(T_init, T * 1.5)
                damping = min(0.8, damping * 1.2)
                stall = 0
                if T > T_init * 0.9 and damping > 0.7:
                    break

        # Anneal temperature
        if it > 20:
            T = max(T_min, T_init * (0.997 ** it))

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    MKKT it{it}: obj={best_obj:.8f}, T={T:.5f}, d={damping:.3f}, time={elapsed:.1f}s")

    return best_a


def smooth_max_gradient_descent(a, time_budget=60.0, T_init=0.001, T_min=0.0001,
                                  use_fft=False, verbose=False):
    """Gradient descent using smooth max with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.002
    momentum = 0.9
    v = np.zeros(n)
    stall = 0

    max_iters = 20000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = float(np.max(conv))
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -400, 400)
        exp_vals = np.exp(shifted)
        total = float(np.sum(exp_vals))
        if total < 1e-30:
            break
        weights = exp_vals / total

        # Gradient: grad[j] = sum_k weights[k] * a[k-j]
        if use_fft:
            grad = cross_corr_fft(weights, a)
        else:
            grad = cross_corr_lambda_a(weights, a)

        # Project onto sum(a)=1 constraint
        grad_mean = float(np.mean(grad))
        grad_proj = grad - grad_mean

        v = momentum * v - lr * grad_proj
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 100 == 0:
                lr = min(lr * 1.03, 0.015)
        else:
            a = a_new
            stall += 1
            if stall > 30:
                lr *= 0.6
                if lr < 1e-7:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0

        if it > 30:
            T = max(T_min, T_init * (0.997 ** it))

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    SMGD it{it}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.5f}, time={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def solve_lp_for_direction(a, time_limit=20):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    try:
        from scipy.optimize import linprog
    except ImportError:
        return None, None

    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': time_limit})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based polishing using alternating direction with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))

        return a
    a = a / s

    best_a = a.copy()

    start = time.monotonic()
    stall = 0

    max_iters = 300

            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        tl = min(remaining - 2, 20)
        if tl < 3:
            break

        g_lp, M_lp = solve_lp_for_direction(a, time_limit=tl)
        if g_lp is None:
            break

        # Convex line search
        a_norm = a / np.sum(a)
        g_norm = g_lp / np.sum(g_lp)

        best_t = 0.0
        best_t_obj = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:

            a = (1 - best_t) * a_norm + best_t * g_norm
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_t_obj
            best_a = a.copy()
            stall = 0
            if verbose and it % 10 == 0:

def peak_directed_cd(a, time_budget=60.0, verbose=False):
    """Coordinate descent that focuses on reducing peak convolution values."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(hash(n * 7919) % (2**32))

    delta = 5e-4
    pass_count = 0
    consecutive_no_improve = 0

    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        peak_idx = int(np.argmax(conv))

        # Compute contribution of each coordinate to the peak
        contributions = np.zeros(n)
        for j in range(n):
            k_val = peak_idx - j
            if 0 <= k_val < n:
                contributions[j] = a[j] * a[k_val]

        # Process in order of contribution (highest first)
        order = np.argsort(contributions)[::-1]
        # Shuffle within groups to avoid bias
        for i in order[:min(n, 200)]:
            if time.monotonic() - start > time_budget:
                break
            if a[i] < 1e-12:
                continue

            # Try decreasing
            for d in [delta, delta * 0.5, delta * 0.1]:
                a_try = a.copy()
                a_try[i] = max(0.0, a_try[i] - d)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if improved:
                continue

            # Try increasing (to redistribute)
            if a[i] < 0.5:
                a_try = a.copy()
                a_try[i] += delta * 0.5
                st = float(np.sum(a_try))
                if st > 1e-10:
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True

        if improved:
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            delta *= 0.7
            if delta < 1e-9:
                if consecutive_no_improve > 5:
                    break

        if verbose and pass_count % 5 == 0:
            print(f"    PCDP pass {pass_count}: obj={best_obj:.8f}, delta={delta:.2e}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    # 1. Parent
    initializations.append(('parent', best_seq.copy()))

    # 2. Cosine taper (various powers)
    for pw in [0.5, 1.0, 2.0]:
        i = np.arange(n_base)
        a_cos = np.cos(np.pi * i / (2.0 * n_base)) ** pw
        a_cos = np.maximum(a_cos, 0)
        if np.sum(a_cos) > 0:
            initializations.append((f'cos_{pw}', a_cos / np.sum(a_cos)))

    # 3. Exponential decay
    for c in [0.5, 1.0, 2.0, 3.0, 5.0]:
        a_exp = np.exp(-c * np.arange(n_base) / n_base)
        initializations.append((f'exp_{c}', a_exp / np.sum(a_exp)))

    # 4. Gaussian (various widths)
    for c in [1.0, 2.0, 3.0, 5.0]:
        center = (n_base - 1) / 2.0
        a_gauss = np.exp(-c * ((np.arange(n_base) - center) / (n_base / 2.0)) ** 2)
        initializations.append((f'gauss_{c}', a_gauss / np.sum(a_gauss)))

    # 5. Symmetric exponential (V-shape inverted)
    for c in [1.0, 2.0, 3.0]:
        center = (n_base - 1) / 2.0
        a_sym = np.exp(-c * np.abs(np.arange(n_base) - center) / (n_base / 2.0))
        initializations.append((f'symexp_{c}', a_sym / np.sum(a_sym)))

    # 6. Random (various)
    a_rand1 = rng.uniform(0.1, 1.0, n_base)
    initializations.append(('rand_unif', a_rand1 / np.sum(a_rand1)))
    a_rand2 = rng.exponential(1.0, n_base)
    initializations.append(('rand_exp', a_rand2 / np.sum(a_rand2)))

    # 7. Two-pulse
    a_two = np.zeros(n_base)
    a_two[:n_base // 3] = 1.0
    a_two[n_base // 3:2 * n_base // 3] = 0.5
    a_two[2 * n_base // 3:] = 0.3
    initializations.append(('two_pulse', a_two / np.sum(a_two)))

    # 8. Power law
    for p in [0.3, 0.5, 1.0, 1.5]:
        x = np.linspace(0.01, 1, n_base)
        a_pl = x ** p
        initializations.append((f'pow_{p}', a_pl / np.sum(a_pl)))

    # 9. Symmetric random
    half_n = n_base // 2
    a_half = rng.uniform(0.1, 1.0, half_n)
    a_sym_r = np.concatenate([a_half, a_half[::-1]])[:n_base]
    initializations.append(('sym_rand', a_sym_r / np.sum(a_sym_r)))

    # 10. Trigonometric
    for freq in [1, 2, 3]:
        a_trig = 1.0 + 0.3 * np.cos(2 * np.pi * freq * np.arange(n_base) / n_base)
        a_trig = np.maximum(a_trig, 0)
        initializations.append((f'trig_{freq}', a_trig / np.sum(a_trig)))

    print(f"\nGenerated {len(initializations)} initializations, n={n_base}")

    # === Phase 1: Quick evaluation ===
    print("\n=== Phase 1: Quick Evaluation ===")
    init_results = []
    for name, a_init in initializations:
        obj_init = compute_obj(a_init)
        init_results.append((obj_init, name, a_init))
    init_results.sort(key=lambda x: x[0])
    for obj_init, name, _ in init_results[:5]:
        print(f"  {name}: obj={obj_init:.8f}")
    print(f"  ... ({len(init_results)} total)")

    # Keep top initializations
    n_keep = min(6, len(init_results))
    top_inits = init_results[:n_keep]

    # === Phase 2: MKKT on top initializations ===
    print("\n=== Phase 2: MKKT Optimization ===")
    remaining = deadline - time.monotonic()
    time_per_init = max(30.0, remaining * 0.35 / n_keep)

    for idx, (obj_init, name, a_init) in enumerate(top_inits):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        alloc = min(time_per_init, remaining * 0.6)
        print(f"\n  [{idx+1}/{n_keep}] {name} (init obj={obj_init:.8f}), budget={alloc:.0f}s")

        a_opt = multiplicative_kkt_update(a_init, time_budget=alloc, T_init=0.08, T_min=0.002,
                                           damping=0.5, verbose=False)
        obj_opt = compute_obj(a_opt)

        # Follow up with SMGD if improved
        remaining = deadline - time.monotonic()
        if remaining > 30 and obj_opt < best_score + 0.01:
            a_opt2 = smooth_max_gradient_descent(a_opt, time_budget=min(40.0, remaining * 0.4),
                                                  T_init=0.002, T_min=0.0002)
            obj_opt2 = compute_obj(a_opt2)
            if obj_opt2 < obj_opt:
                a_opt = a_opt2
                obj_opt = obj_opt2

        print(f"    Result: obj={obj_opt:.8f}")

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"    ** New best! **")

    # === Phase 3: Fine MKKT on best ===
    print(f"\n=== Phase 3: Fine MKKT ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(120.0, remaining * 0.25)
        a_mk = multiplicative_kkt_update(best_seq, time_budget=alloc, T_init=0.02, T_min=0.001,
                                          damping=0.3, verbose=True)
        obj_mk = compute_obj(a_mk)
        if obj_mk < best_score:
            best_score = obj_mk
            best_seq = a_mk.copy()
            print(f"  Fine MKKT: obj={best_score:.8f} ** improved **")
        else:
            print(f"  Fine MKKT: obj={obj_mk:.8f}")

    # === Phase 4: Smooth-max GD ===
    print(f"\n=== Phase 4: Smooth-Max GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(100.0, remaining * 0.2)
        a_sm = smooth_max_gradient_descent(best_seq, time_budget=alloc, T_init=0.001, T_min=0.0001,
                                           verbose=True)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  SMGD: obj={best_score:.8f} ** improved **")
        else:
            print(f"  SMGD: obj={obj_sm:.8f}")

    # === Phase 5: LP Polish ===
    print(f"\n=== Phase 5: LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(80.0, remaining * 0.18)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} ** improved **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 6: Peak-directed CD ===
    print(f"\n=== Phase 6: Peak-Directed CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(80.0, remaining * 0.18)
        a_cd = peak_directed_cd(best_seq, time_budget=alloc, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  PCDP: obj={best_score:.8f} ** improved **")
        else:
            print(f"  PCDP: obj={obj_cd:.8f}")

    # === Phase 7: Multi-n Exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(500, n_best - 50), max(500, n_best - 25),
        n_best + 25, n_best + 50, n_best + 100, n_best + 150,
        600, 700, 750, 800, 850, 900, 950, 1000, 1100, 1200, 1300, 1400, 1500
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1500 and n != n_best]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick MKKT
        alloc = min(remaining * 0.08, 25.0)
        a_opt = multiplicative_kkt_update(a_try, time_budget=alloc, T_init=0.04, T_min=0.001,
                                           damping=0.5)
        obj_try = compute_obj(a_opt)

        # If promising, do more
        remaining = deadline - time.monotonic()
        if remaining > 30 and obj_try < best_score + 0.005:
            a_opt = smooth_max_gradient_descent(a_opt, time_budget=min(30.0, remaining * 0.3),
                                                 T_init=0.001, T_min=0.0001)
            obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Second round MKKT with different params ===
    print(f"\n=== Phase 8: Second-round MKKT (different params) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Try with different temperature schedule
        alloc = min(60.0, remaining * 0.25)
        a_mk2 = multiplicative_kkt_update(best_seq, time_budget=alloc, T_init=0.015, T_min=0.0005,
                                           damping=0.4, verbose=False)
        obj_mk2 = compute_obj(a_mk2)
        if obj_mk2 < best_score:
            best_score = obj_mk2
            best_seq = a_mk2.copy()
            print(f"  MKKT2: obj={best_score:.8f} ** improved **")
        else:
            print(f"  MKKT2: obj={obj_mk2:.8f}")

    # === Phase 9: Final polish sequence ===
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()

    # SMGD fine
    if remaining > 30:
        alloc = min(40.0, remaining * 0.25)
        a_f = smooth_max_gradient_descent(best_seq, time_budget=alloc, T_init=0.0005, T_min=0.00005)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final SMGD: obj={best_score:.8f}")

    # MKKT very fine
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(35.0, remaining * 0.3)
        a_f = multiplicative_kkt_update(best_seq, time_budget=alloc, T_init=0.005, T_min=0.0003,
                                         damping=0.2)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final MKKT: obj={best_score:.8f}")

    # PCDP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(40.0, remaining * 0.4)
        a_f = peak_directed_cd(best_seq, time_budget=alloc)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final PCDP: obj={best_score:.8f}")

    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(800)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if np.isfinite(val) and val < best_score:
            best_score = val
        print(f"  evaluate_sequence returned: {val:.8f}")
    except Exception as e:
        print(f"  evaluate_sequence error: {e}")

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
