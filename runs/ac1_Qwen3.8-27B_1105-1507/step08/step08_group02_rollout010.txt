    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    """Compute smooth objective and gradient for log-sum-exp approximation."""
    n = len(a)
    a = a / a.sum()  # normalize
    conv = compute_conv(a)
    m = conv.max()
    shifted = beta * (conv - m)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()

    # Gradient: grad_i = 2 * sum_k w_k * a_{k-i} * 1[0 <= k-i < n]

        k_end = min(2*n-2, i + n - 1)

        j_start = k_start - i
        j_end = k_end - i
        grad[i] = 2.0 * np.dot(a[j_start:j_end+1], w[k_start:k_end+1])
    
    # Scale for objective 2n * smooth_val
    obj = 2.0 * n * smooth_val
    grad_scaled = 2.0 * n * grad

```python
"""
Optimized step function search using:
1. FFT-based smooth gradient with L-BFGS
2. LP alternating method (workhorse)
3. Aggressive restart strategy for escaping local minima
4. Multi-dimension (n) exploration
5. Coordinate descent finishing
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def fft_conv(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n]


def compute_conv(a):
    """Auto-convolution a*a."""
    return fft_conv(a, a)


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_gradient(a, beta):
    """
    Compute gradient of 2n * lse(beta * (a*a)) / beta w.r.t. a.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    conv = compute_conv(a)
    m = conv.max()
    shifted = beta * (conv - m)
    w = np.exp(shifted)
    w /= w.sum()
    
    # grad_i = 4n * sum_j a_j * w_{i+j} (for valid i+j)
    # = 4n * convolve(a, w_flipped)[2n-2-i]
    w_flip = w[::-1]
    full = fft_conv(a, w_flip)
    grad = 4.0 * n * full[n-1:2*n-1][::-1]
    return grad


def smooth_objective(a, beta):
    n = len(a)
    conv = compute_conv(a)
    m = conv.max()
    lse = m + np.log(np.sum(np.exp(beta * (conv - m))))
    return 2.0 * n * lse / beta


def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def lbfgs_optimize(a_init, time_budget=150.0, beta=200.0, verbose=False):
    """
    L-BFGS optimization using softmax parametrization.
    Optimizes f(x) where a = softmax(x).
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    # Initial x from a
    x0 = np.log(a)  # a = softmax(x0) up to a constant
    
    best_x = x0.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    def f_and_g(x):
        a = softmax(x)
        obj = smooth_objective(a, beta)
        grad_a = smooth_gradient(a, beta)
        # Chain rule: df/dx_j = a_j * (grad_a_j - sum(a * grad_a))
        grad_x = a * (grad_a - float(a @ grad_a))
        return obj, grad_x
    
    # Use L-BFGS-B
    max_iter = 500
    result = minimize(f_and_g, x0, method='L-BFGS-B', 
                      options={'maxiter': max_iter, 'maxfun': 10000,
                              'maxls': 50, 'gtol': 1e-8})
    
    a_opt = softmax(result.x)
    obj_opt = compute_obj(a_opt)
    
    elapsed = time.monotonic() - start
    if verbose:
        print(f"    L-BFGS: obj={obj_opt:.8f}, iters={result.nit}, time={elapsed:.1f}s")
    
    return a_opt


def lp_solve(a, top_k=None):
    """
    Solve LP: min M s.t. (a*g)_k <= M for all k (or top_k), sum(g)=1, g>=0.
    Returns optimal g or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Compute current convolution to identify top_k constraints
    conv_a = compute_conv(a_norm)  # This is a*a, but we need the constraint matrix
    
    # Build constraint matrix: (a*g)_k = sum_j a_k-j * g_j (for valid indices)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    if top_k is not None and top_k < 2*n-1:
        # Only use top_k constraints
        idx = np.argsort(conv_a)[::-1][:top_k]
    else:
        idx = np.arange(2*n-1)
    
    num_constraints = len(idx)
    
    # Build A_ub for linprog: A_ub @ [g; M] <= b_ub
    # (a*g)_k - M <= 0 for k in idx
    A_con = np.zeros((num_constraints, n))
    for ii, k in enumerate(idx):
        for j in range(n):
            ki = k - j
            if 0 <= ki < n:
                A_con[ii, j] = a_norm[ki]
    
    A_ub = np.hstack([A_con, -np.ones((num_constraints, 1))])
    b_ub = np.zeros(num_constraints)
    
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            g = g / s
            return g, float(res.x[n])
    
    return None, None

                noise_scale = min(noise_scale, 0.8)
                a = best_a * (1.0 + noise_scale * rng.normal(0, 1, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                
                # Sometimes try a very different direction
                if rng.random() < 0.3:
                    random_dir = rng.dirichlet(np.ones(n) * 0.5)
                    blend = 0.3 + 0.3 * rng.random()
                    a = (1 - blend) * best_a + blend * random_dir
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                last_good_t = 0.5
                temperature = min(0.5, temperature * 2.0)
            else:
                temperature = max(0.02, temperature * 0.98)
        
        if verbose and iteration % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{iteration}: obj={best_obj:.8f}, t={last_good_t:.3f}, "
                  f"stall={stall}, temp={temperature:.3f}, time={elapsed:.1f}s")
    
    return best_a


def nesterov_subgradient(a_init, time_budget=100.0, verbose=False):
    """
    Nesterov-accelerated subgradient method for the non-smooth objective.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    y = a.copy()  # Nesterov momentum variable
    t = 1.0
    lr_base = 2.0 / np.sqrt(n)
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute subgradient at y
        conv_y = compute_conv(y)
        k_star = int(np.argmax(conv_y))
        
        # Gradient of (a*a)_{k_star} w.r.t. a
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * y[j]
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        lr = lr_base * t / t_new
        
        # Gradient step
        y_new = y - lr * grad
        # Project to simplex
        y_new = project_simplex(y_new)
        
        # Update y for next iteration
        momentum = (t - 1.0) / t_new
        y = y_new + momentum * (y_new - y)
        y = project_simplex(y)
        t = t_new
        
        obj_y = compute_obj(y_new)
        if obj_y < best_obj - 1e-12:
            best_obj = obj_y
            best_a = y_new.copy()
    
    if verbose:
        elapsed = time.monotonic() - start
        print(f"    Nesterov: obj={best_obj:.8f}, time={elapsed:.1f}s")
    
    return best_a
