So L ≈ O(n * beta) for the log-sum-exp with parameter beta. For stability, I need step_size < 1/L.

For practical purposes, I'll use a backtracking line search or empirical tuning.

Minimizes 2n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. LP alternating (primary)
2. Log-sum-exp gradient descent (fast refinement)
3. Coordinate descent (polishing)
4. Multi-n exploration

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A


def solve_lp_min_M(a, M_ub=None):
    """Solve: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # LP: min M
    # Variables: g[0], ..., g[n-1], M
    # Constraints: A@g - M <= 0 (2n-1 constraints)
    #              sum(g) = 1 (1 equality)
    #              g >= 0, M >= 0 (bounds)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, M_ub) if M_ub else (0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g_val))
        if gs > 1e-10:
            return g_val / gs, float(res.x[n])
    
    return None, None


def lse_objective_and_grad(a, beta):
    """Compute log-sum-exp objective and its gradient (for normalized a).
    F(a) = (2n/beta) * log(sum_k exp(beta * conv(a,a)[k]))
    Returns (F, grad)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = beta * (conv - max_c)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    p = exp_vals / sum_exp  # softmax weights
    
    F = (2.0 * n / beta) * (beta * max_c + np.log(sum_exp))
    
    # Gradient: dF/da[m] = 4n * sum_j a[j] * p[m+j]
    # = 4n * (conv(a_flipped, p))[m+n-1]
    a_flipped = a[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    A_fft = rfft(a_flipped, size)
    P_fft = rfft(p, size)
    conv_ap = irfft(A_fft * P_fft, size)
    C = conv_ap[n-1:2*n-1]
    
    grad = 4.0 * n * C
    return F, grad


def lse_optimize(a, beta, time_budget=60.0, verbose=False):
    """Nesterov-accelerated gradient descent on log-sum-exp objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Lipschitz constant estimate
    # L ~ O(n * beta), use conservative estimate
    L = 4.0 * n * beta * 1.5
    lr = 1.0 / L
    
    # Nesterov parameters
    t_k = 1.0
    x = a.copy()  # main iterate
    y = a.copy()  # lookahead
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        F_y, grad_y = lse_objective_and_grad(y, beta)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        alpha = (t_k - 1.0) / t_new
        
        # Nesterov step
        y = x + alpha * (x - y)
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy > 1e-10:
            y = y / sy
        
        # Gradient step
        F_y, grad_y = lse_objective_and_grad(y, beta)
        x_new = y - lr / t_new * grad_y
        x_new = np.maximum(x_new, 0)
        sx = float(np.sum(x_new))
        if sx > 1e-10:
            x_new = x_new / sx
        
        x = x_new
        t_k = t_new
        
        obj = compute_obj(x)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = x.copy()
        
        # Decay learning rate
        if it % 500 == 0:
            lr *= 0.98
        
        if verbose and it % 5000 == 0:
            print(f"    LSE: it={it}, obj={best_obj:.8f}")
    
    return best_a, best_obj

def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ab(a, b):
    """FFT-based linear convolution of a and b."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        length = k_end - k_start + 1
        A[k_start:k_end + 1, j] = a[:length]
    return A
