from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def cross_corr_via_fft(a, p):
    """
    Compute cross-correlation of a (length n) and p (length 2n-1).
    Result[j] = sum_{m=0}^{n-1} a[m] * p[m+j] for j=0,...,n-1
    (with p[k]=0 for k outside [0, 2n-2])
    """
    n = len(a)
    size = 1
    while size < 3 * n:
        size <<= 1
    fa = rfft(a[::-1], size)
    fp = rfft(p, size)
    result = irfft(fa * fp, size)
    return result[n - 1: n - 1 + n]


def smooth_grad_step(a, tau, lr):
    """
    One step of smooth gradient descent using log-sum-exp.
    Returns updated a and the gradient.
    """
    n = len(a)
    C = compute_conv(a)
    
    # Softmax weights (numerically stable)
    C_max = np.max(C)
    exp_tC = np.exp(tau * (C - C_max))
    p = exp_tC / np.sum(exp_tC)
    
    # Gradient: grad[j] = 2/tau * sum_m p[m+j] * a[m]
    corr = cross_corr_via_fft(a, p)
    grad = 2.0 * corr / tau
    
    # Multiplicative weights update
    # Clip to prevent underflow
    grad_clipped = np.clip(grad, -50/lr, 50/lr)
    a_new = a * np.exp(-lr * grad_clipped)
    s = np.sum(a_new)
    if s < 1e-30:
        return a, grad
    a_new = a_new / s
    return a_new, grad


def smooth_grad_descent(a, time_budget=200.0, verbose=False):
    """
    Smooth gradient descent with continuation on tau.
    Start with small tau (minimize variance) and increase to large tau (minimize max).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Continuation schedule: tau from small to large
    tau_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0]
    
    for tau in tau_schedule:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        iters = max(5, int(remaining * 3))  # ~3 iterations per second
        lr = min(1.0 / (tau * np.max(compute_conv(a)) + 1e-10), 0.5)
        
        # Adaptive learning rate
        lr_init = 1.0 / (np.max(np.abs(compute_conv(a))) * tau + 1e-10)
        lr_init = min(lr_init, 1.0)
        
        for it in range(iters):
            if time.monotonic() - start > time_budget - 5:
                break
            
            a_new, grad = smooth_grad_step(a, tau, lr_init)
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
            
            a = a_new
            
            # Adaptive lr: reduce if objective increases
            if obj_new > best_obj * 1.01:
                lr_init *= 0.5
        
        if verbose:
            print(f"  Smooth GD tau={tau}: obj={compute_obj(a):.8f}, best={best_obj:.8f}")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    # Use scipy linprog for reliability
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                return g_val, float(res.x[n])
    except:
        pass

    # Fallback to cvxpy
    try:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        prob = cp.Problem(cp.Minimize(M), [A @ g <= M, cp.sum(g) == 1])
        for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        g_val = g_val / gs
                        return g_val, float(M.value)
            except:
                continue
    except:
        pass

    return None, None


def lp_alternating(a, time_budget=200.0, verbose=False):
    """
    LP alternating with momentum and cycle detection.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()

    prev_a = None
    momentum = a.copy()
    stall = 0
    
    while time.monotonic() - start < time_budget - 5:
        result = solve_lp(a)
        if result[0] is None:
            break
        g_lp, M_lp = result
        
        # Update momentum
        momentum = 0.5 * momentum + 0.5 * g_lp
        momentum = np.maximum(momentum, 0)
        ms = float(np.sum(momentum))
        if ms > 1e-10:
            momentum = momentum / ms
        
        # Try different mixing ratios
        best_cand = a.copy()
        best_cand_obj = np.inf
        
        for mix in [0.3, 0.5, 0.7, 0.9, 1.0]:
            for target in [g_lp, momentum]:
                a_try = (1.0 - mix) * a + mix * target
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_cand_obj:
                    best_cand_obj = obj_try
                    best_cand = a_try.copy()
        
        if best_cand_obj < best_obj - 1e-13:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            
            # Cycle detection
            if prev_a is not None:
                diff = np.max(np.abs(a - prev_a))
                if diff < 1e-6:
                    # 2-cycle: nudge
                    rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                    best_a = best_a * np.exp(rng.normal(0, 0.01, n))
                    best_a = np.maximum(best_a, 0) / np.sum(np.maximum(best_a, 0))
                    best_obj = compute_obj(best_a)
                    prev_a = None
                    momentum = best_a.copy()
            prev_a = a.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 20:
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0) / np.sum(np.maximum(a, 0))
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                momentum = a.copy()
                stall = 0

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, mode=0):
    """Generate structured initial sequences."""
    t = np.linspace(0, 1, n)
    if mode == 0:
        return np.sin(np.pi * t) ** 2
    elif mode == 1:
        return 0.5 + 0.5 * np.cos(2 * np.pi * t)
    elif mode == 2:
        return np.ones(n) / n
    elif mode == 3:
        # Fejer-like kernel
        return np.maximum(1.0 - np.abs(t - 0.5) * 2, 0)
    elif mode == 4:
        # Modulated
        return np.maximum(1.0 + 0.2 * np.cos(6 * np.pi * t), 0)
    elif mode == 5:
        # Triangle
        return 1.0 - 2.0 * np.abs(t - 0.5)
    else:
        return np.ones(n) / n


def pairwise_exchange(a, time_budget=60.0):
    """Pairwise exchange polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        for _ in range(10):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            for frac in [0.05, 0.1, 0.2]:
                delta = frac * min(a[i], a[j])
                a_try = best_a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0:
                    continue
                a_try = a_try / np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                    break
            if improved:
                break
    
    return best_a


def coordinate_descent(a, time_budget=60.0):
    """
    Coordinate-wise optimization: for each coordinate, find the optimal value
    given all others fixed (using the structure of the convolution).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget - 5:
        improved = False
        for j in range(n):
            if time.monotonic() - start > time_budget - 5:
                return best_a
            
            # When we change a[j], it affects C_k for k = j, j+1, ..., j+n-1
            # (and symmetrically k = 0, ..., j-1 via C_k = sum_i a[i]*a[k-i])
            # Actually C_k involves a[j] when k-j is in [0,n-1], i.e., k in [j, j+n-1]
            # AND when j-k is in [0,n-1], i.e., k in [j-n+1, j]
            # So affected k are in [max(0, j-n+1), min(2n-2, j+n-1)]
            
            # Current contribution of a[j] to C_k:
            # C_k = ... + a[j]*a[k-j] + a[k-j]*a[j] + ... (if k-j in range)
            # = ... + 2*a[j]*a[k-j] + ... (when j ≠ k-j)
            # or a[j]^2 (when j = k-j, i.e., k = 2j)
            
            # New C_k when a[j] changes to x:
            # The terms involving a[j] are: for each k, the contribution is
            # a[j]*a[k-j] + a[k-j]*a[j] = 2*a[j]*a[k-j] (if 0<=k-j<n and k-j != j)
            # or a[j]^2 (if k-j = j)
            
            # So C_k_new = C_k_old - 2*a[j]*a[k-j] + 2*x*a[k-j] (for k-j in range, k-j≠j)
            # or C_k_new = C_k_old - a[j]^2 + x^2 (for k=2j)
            
            # We want to choose x >= 0 to minimize max_k C_k_new
            # This is a 1D optimization!
            
            # Let's use ternary search or grid search on x
            # The range of x is [0, a[j] * 3] (reasonable bound)
            
            # Actually, let's think about this more carefully.
            # If we increase a[j] by delta, C_k increases by 2*delta*a[k-j] for affected k.
            # The max is currently M = max(C). 
            # We want to find the optimal a[j] that minimizes the new max.
            
            # This is a convex 1D problem (max of linear/quadratic functions is convex).
            # We can use bisection on the derivative (subgradient).
            
            # Simpler: just try a few values
            a_save = a[j]
            best_x = a_save
            best_val = best_obj
            
            # Try increasing and decreasing
            for scale in [0.5, 0.7, 0.8, 1.2, 1.5, 2.0, 0.3, 3.0]:
                x = a_save * scale
                a[j] = x
                # Renormalize (this changes all C_k slightly, but let's approximate)
                # Actually, let's not renormalize in the inner loop.
                # Just compute the max of C and compare.
                C = compute_conv(a / np.sum(a))
                val = 2 * n * np.max(C)
                if val < best_val:
                    best_val = val
                    best_x = x
            
            a[j] = best_x
            a = a / np.sum(a)
            obj_now = compute_obj(a)
            if obj_now < best_obj - 1e-14:
                best_obj = obj_now
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a

The objective we're minimizing (after normalization) is:
f(a) = max_k C_k(a) where C_k = conv(a,a)[k]

The smooth surrogate is:
f_τ(a) = (1/τ) * log(Σ_k exp(τ * C_k))

The gradient is:
∇f_τ(a)[j] = (2/τ) * Σ_k p_k * a[k-j] (with p_k = softmax(τ*C)[k])

The Lipschitz constant of this gradient is roughly O(1) (since a is on the simplex and p is a probability distribution).

For gradient descent with step size lr, we need lr < 2/L where L is the Lipschitz constant.
For multiplicative weights, the effective step size depends on the gradient magnitude.

For the non-LP approach, let me try a **proximal gradient method** with the max function:

The proximal operator of max(C(a)) is not straightforward, but we can use the following trick:
- min max_k C_k(a) s.t. a ∈ Δ is equivalent to min_{a, M} M s.t. C_k(a) ≤ M for all k, a ∈ Δ
- We can use a penalty method: min_{a} max_k C_k(a) + (μ/2) * ||a - a_prev||²
- Or use the dual: max_{p ∈ Δ_{2n-1}} min_{a ∈ Δ} Σ_k p_k * C_k(a)

The inner problem min_{a ∈ Δ} Σ_k p_k * C_k(a) is a quadratic program! Because C_k(a) is quadratic in a.

Σ_k p_k * C_k(a) = Σ_k p_k * Σ_i a[i]*a[k-i] = Σ_i,j a[i]*a[j] * (Σ_k p_k * δ(i+j=k))
= a^T * Q(p) * a

where Q(p) is the (n×n) matrix with Q(i,j) = p[i+j] (padded to be in [0, 2n-2]).

So the dual problem is:
max_{p ∈ Δ_{2n-1}} min_{a ∈ Δ} a^T Q(p) a

This is a max-min problem. The inner min is a QP (quadratic minimization over the simplex), and the outer max is over the simplex in p-space.

We can use iterative methods:
1. Initialize p (e.g., uniform)
2. Solve inner QP: min a^T Q(p) a s.t. a ∈ Δ
3. Update p in the direction of the solution (multiplicative weights or gradient)

This is essentially the **Dantzig-Wolfe decomposition** or **column generation** approach!

The inner QP can be solved with standard methods. For large n, we can use projected gradient on the simplex.

Let me implement this:

```python
def solve_inner_qp(Q, time_budget=10.0):
    """
    Solve min a^T Q a s.t. a >= 0, sum(a) = 1.
    Using projected gradient on the simplex.
    """
    n = Q.shape[0]
    a = np.ones(n) / n  # uniform init
    
    lr = 0.5 / (np.linalg.norm(Q, ord=2) + 1e-10)  # simple bound
    
    for it in range(100):
        # Gradient of a^T Q a is 2*Q*a (if Q is symmetric)
        # Make Q symmetric: Q_sym = (Q + Q.T) / 2
        Q_sym = (Q + Q.T) / 2
        grad = 2 * Q_sym @ a
        
        # Gradient step
        a_new = a - lr * grad
        
        # Project onto simplex
        a_new = project_simplex(a_new)
        
        # Check convergence
        if np.max(np.abs(a_new - a)) < 1e-8:
            break
        a = a_new
    
    return a


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    k = ind[cond].max()
    theta = cssv[k - 1] / k
    return np.maximum(v - theta, 0)
```

Wait, but solving the inner QP with projected gradient might be slow for large n. Let me think of a better way.

Actually, for the inner problem min a^T Q a s.t. a ∈ Δ, if Q is positive semi-definite, the minimum is 0 (achieved at a=0, but that's not on the simplex). Since a must be on the simplex, the minimum is the minimum eigenvalue of Q restricted to the affine subspace {sum(x)=1}.

But in general Q(p) = [p[i+j]] might not be PSD. Let me check: for a to minimize a^T Q a, we need Q to be PSD (otherwise the problem is unbounded below... but we're on the simplex, so it's bounded).

Actually, Q(p) with p ≥ 0 is always PSD! Because:
a^T Q(p) a = Σ_{i,j} a[i]*a[j]*p[i+j] = Σ_k p[k] * (Σ_i a[i]*a[k-i]) = Σ_k p[k] * C_k(a) ≥ 0

So Q(p) is PSD for any p ≥ 0. 

For the inner QP, since Q is PSD and we're on the simplex, the minimum is well-defined. We can use:
- Projected gradient (slow but simple)
- Active set method
- Or just use scipy's minimize with bounds

For the outer problem, we use multiplicative weights:
p_new[k] = p[k] * exp(-lr_outer * C_k(a*)) where a* is the solution of the inner problem.

This is the **multiplicative weights algorithm** for the max-min game!

Let me implement this properly:

```python
def maxmin_game(a_init, time_budget=100.0, verbose=False):
    """
    Solve max_p min_a p^T C(a) using multiplicative weights.
    This gives a lower bound on max_k C_k(a) for any a.
    The optimal a* minimizes the maximum (by minimax theorem).
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    a = a / np.sum(a)
    
    # Initialize p uniformly
    p = np.ones(2 * n - 1) / (2 * n - 1)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.1  # learning rate for p updates
    
    for t in range(1000):
        if time.monotonic() - start > time_budget - 5:
            break
        
        # Build Q(p)
        Q = np.zeros((n, n))
        for i in range(n):
            for j in range(n):
                if i + j < 2 * n - 1:
                    Q[i, j] = p[i + j]
        Q = (Q + Q.T) / 2  # ensure symmetry
        
        # Solve inner: min a^T Q a s.t. a in simplex
        # Use projected gradient
        a_inner = a.copy()  # warm start
        lr_inner = 1.0 / (2 * np.linalg.norm(Q, ord=2) + 1e-10)
        
        for it in range(50):
            grad = 2 * Q @ a_inner
            a_inner = a_inner - lr_inner * grad
            a_inner = project_simplex(a_inner)
        
        # Update p using multiplicative weights
        C = compute_conv(a_inner)
        C_shifted = C - np.min(C)
        p_new = p * np.exp(-lr * C_shifted)
        p_new = p_new / np.sum(p_new)
        p = p_new
        
        # Track best
        obj = compute_obj(a_inner)
        if obj < best_obj:
            best_obj = obj
            best_a = a_inner.copy()
            a = a_inner
    
    return best_a
```

Let me think of a way to make this more efficient.

Actually, Q(p) is a Toeplitz-like matrix (constant on anti-diagonals). We can exploit this structure!

Q[i,j] = p[i+j], so Q is determined by the first row and first column (it's a Hankel matrix).

For matrix-vector product Q @ v:
(Q @ v)[i] = Σ_j p[i+j] * v[j]

This is a cross-correlation! (Q @ v)[i] = Σ_j v[j] * p[i+j] = cross_correlation(v, p)[i]

We can compute this using FFT in O(n log n)!

So the projected gradient on the inner QP can be made fast:
1. grad = 2 * (Q @ a) = 2 * cross_correlation(a, p) → O(n log n) via FFT
2. a = a - lr * grad
3. Project onto simplex → O(n log n)

This is much better!

Let me also think about the outer update. We need C = conv(a, a) which is O(n log n) via FFT. Then p update is element-wise.

So the full iteration is O(n log n) per step! That's very fast.
