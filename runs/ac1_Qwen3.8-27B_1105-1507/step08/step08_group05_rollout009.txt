"""
Optimized step function search combining efficient LP alternating, 
adaptive subgradient, coordinate descent, and multi-scale n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar
import time
import cvxpy as cp
from scipy.sparse import csc_matrix

def compute_conv(a):
    n = len(a)
    if n == 1:
        return np.array([a[0]**2])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    result = irfft(fa * fa, size)[:2*n-1]
    return np.maximum(result, 0)  # Numerical safety

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Dense (2n-1) x n matrix for conv(a, .)."""
    n = len(a)
    m = 2*n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp(a, max_constraints=None):
    """
    LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    If max_constraints is set, only use top peaks + random sample.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    m = 2*n - 1
    
    if max_constraints is not None and m > max_constraints:
        # Use subset of constraints
        # Compute a reference convolution to identify important constraints
        ref_conv = compute_conv(a_n)
        # Top peaks
        top_k = min(m, max_constraints // 2)
        top_idx = np.argsort(ref_conv)[::-1][:top_k]
        # Random sample
        n_rand = max_constraints - top_k
        rand_idx = np.random.choice(m, size=n_rand, replace=False)
        idx = np.unique(np.concatenate([top_idx, rand_idx]))
        idx = np.sort(idx)
    else:
        idx = np.arange(m)
    
    A = np.zeros((len(idx), n))
    for j in range(n):
        for i_local, k in enumerate(idx):
            val = k - j
            if 0 <= val < n:
                A[i_local, j] = a_n[val]
    
    # LP: min M
    # s.t. A @ g - M <= 0  (len(idx) constraints)
    #      sum(g) = 1
    #      g >= 0
    
    # Using cvxpy
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g_var.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M_var.value)
        except Exception:
            continue
    return None, None

"""
Step function optimizer: LP alternating + peak shaving + multi-scale search.
Focuses on efficient LP solving, adaptive subgradient, and peak equalization
to minimize 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar
import time
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    if n == 1:
        return np.array([a[0]*a[0]])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    result = irfft(fa * fa, size)[:2*n-1]
    # Clip tiny negative values from numerical error
    result[np.abs(result) < 1e-15] = 0
    return result

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    m = 2*n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp_full(a):
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g_var.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M_var.value)
