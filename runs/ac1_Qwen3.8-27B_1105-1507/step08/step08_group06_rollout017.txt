The theoretical asymptotic optimum is 2 (achieved by uniform sequences), but for finite n, clever shapes can do better. The entropy-penalized approach finds near-optimal balance between flatness and boundary effects.

```python
"""
Fast entropy-mirror-descent optimizer for step function autocorrelation minimization.
Uses log-barrier (entropic) regularization for non-negativity, fast FFT-based
subgradient steps, and explores multiple lengths initialized from the parent.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""
import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for given a (normalization-invariant)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def mirror_descent_step(log_vals, active_indices, conv, a, lr, n, K):
    """
    One mirror descent (entropic) step.
    log_vals: current log-a values
    active_indices: indices of near-max conv positions
    conv: current convolution values
    a: current a values
    lr: learning rate
    n: sequence length
    K: number of active peaks to use
    """
    # Compute subgradient from active peaks
    grad = np.zeros(n)
    for k in active_indices[:K]:
        k = int(k)
        i_max = min(n - 1, k)
        i_min = max(0, k - n + 1)
        if i_max >= i_min:
            indices = np.arange(i_min, i_max + 1)
            grad[indices] += a[k - indices]
    grad /= min(len(active_indices), K)
    
    # Center gradient (project to sum=0 hyperplane)
    grad_centered = grad - np.mean(grad)
    
    # Mirror descent update in log space
    log_vals_new = log_vals - lr * grad_centered
    
    return log_vals_new


def entropy_optimize(a_init, time_budget, lr_base=0.02, K=120, 
                     ema_alpha=0.9, n_refine=None):
    """
    Entropy (mirror) descent optimization with entropy regularization.
    Optimizes over a = softmax(log_vals) * scale.
    
    The entropy regularization implicitly encourages uniformity while
    the subgradient pushes to flatten the autocorrelation.
    """
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-12).astype(np.float64)
    
    # Initialize log space
    log_vals = np.log(a_init / np.mean(a_init))
    a = np.exp(log_vals)
    a = a / np.sum(a)  # normalize
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv = compute_conv(a)
    ema_lr = lr_base
    stall = 0
    
    iteration = 0
    while time.monotonic() - start < time_budget:
        # Find active peaks (top K or those within 85% of max)
        max_val = float(np.max(conv))
        if max_val < 1e-15:
            break
        threshold = max_val * 0.85
        active_all = np.where(conv >= threshold)[0]
        if len(active_all) > K:
            # Take top K
            active_indices = np.argsort(conv)[::-1][:K]
        else:
            active_indices = active_all
        
        # Adaptive learning rate (decay over time)
        lr = ema_lr / (1.0 + 0.001 * iteration)
        
        # Mirror descent step
        log_vals_new = mirror_descent_step(
            log_vals, active_indices, conv, a, lr, n, K
        )
        
        # Convert back to a
        # Subtract max for numerical stability
        log_vals_new = log_vals_new - np.max(log_vals_new)
        a_new = np.exp(log_vals_new)
        s_new = float(np.sum(a_new))
        if s_new < 1e-12:
            break
        a_new = a_new / s_new
        
        # Evaluate
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            # If stalled, reduce learning rate
            if stall > 20:
                ema_lr *= 0.7
                stall = 0
        
        # Accept the new point (always move in mirror descent)
        a = a_new
        log_vals = log_vals_new
        conv = compute_conv(a)
        iteration += 1
        
        # Periodic refinement: try small perturbations
        if iteration % 500 == 499:
            rng_pert = np.random.default_rng(iteration * 12345)
            for _ in range(3):
                pert_scale = 0.01 * (1.0 + rng_pert.random())
                a_pert = a * np.exp(rng_pert.normal(0, pert_scale, n))
                a_pert = np.maximum(a_pert, 1e-12)
                a_pert = a_pert / np.sum(a_pert)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj - 1e-12:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    log_vals = np.log(a_pert) - np.max(np.log(a_pert))
                    a = a_pert.copy()
                    conv = compute_conv(a)
    
    # Refine with smaller active set count
    if n_refine is None:
        n_refine = max(20, K // 3)
    
    return best_a, best_obj


def subgradient_polish_fast(a, time_budget, K=50):
    """
    Fast subgradient polish with Nesterov acceleration.
    Works in the probability simplex (sum=1, a>=0).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = a.copy()  # Nesterov point
    y = a.copy()  # actual point
    momentum = np.zeros(n)
    beta = 0.85
    lr = 0.03
    
    conv = compute_conv(x)
    
    iteration = 0
    while time.monotonic() - start < time_budget:
        max_val = float(np.max(conv))
        if max_val < 1e-15:
            break
        
        # Active peaks
        threshold = max_val * 0.8
        active = np.where(conv >= threshold)[0]
        if len(active) > K:
            active = np.argsort(conv)[::-1][:K]
        
        # Subgradient
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_max = min(n - 1, k)
            i_min = max(0, k - n + 1)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += y[k - indices]
        grad /= len(active)
        
        # Center (project to tangent space of simplex)
        grad_centered = grad - np.mean(grad)
        
        # Momentum + step
        momentum = beta * momentum - lr * grad_centered
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-12:
            y_new = y.copy()
            s_new = 1.0
        y_new = y_new / s_new
        
        # Nesterov extrapolation
        t_k = 2.0 / (np.sqrt(iteration + 1) + 2.0)
        x = y_new + (1 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-12:
            x = x / sx
        
        y = y_new
        obj = compute_obj(y)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = y.copy()
        
        conv = compute_conv(x)
        iteration += 1
        
        # Adaptive LR
        if iteration % 300 == 299:
            lr = max(lr * 0.92, 0.002)
        
        # Periodic perturbation
        if iteration % 1000 == 999:
            rng_p = np.random.default_rng(iteration * 777)
            a_pert = best_a * np.exp(rng_p.normal(0, 0.008, n))
            a_pert = np.maximum(a_pert, 0)
            sp = float(np.sum(a_pert))
            if sp > 1e-12:
                a_pert = a_pert / sp
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    y = a_pert.copy()
                    x = a_pert.copy()
                    momentum = np.zeros(n)
    
    return best_a, best_obj


def coordinate_descent_local(a, time_budget):
    """
    Local coordinate descent: adjust pairs of elements to reduce peak.
    Focuses on the dominant peak position.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    
    max_iters = 2000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        k_star = int(np.argmax(conv))
        improved = False
        
        # Elements contributing to peak
        i_max = min(n - 1, k_star)
        i_min = max(0, k_star - n + 1)
        involved = np.arange(i_min, i_max + 1)
        
        # Try moving mass from high a[i] to low a[j] within involved set
        # or to positions that don't contribute to the peak
        sorted_inv = involved[np.argsort(a[involved])[::-1]]
        
        # Also try moving to "safe" positions (far from k_star)
        safe_positions = np.arange(n)
        safe_mask = (safe_positions < i_min) | (safe_positions > i_max)
        safe_positions = safe_positions[safe_mask]
        
        for i in sorted_inv[:10]:
            if time.monotonic() - start > time_budget:
                break
            for frac in [0.002, 0.005, 0.01, 0.02]:
                delta = frac * a[i]
                if delta <= 0 or delta >= a[i]:
                    continue
                
                # Try several target positions
                targets = []
                if len(safe_positions) > 0:
                    # Pick positions with lowest a
                    top_safe = safe_positions[np.argsort(a[safe_positions])[:5]]
                    targets.extend(top_safe.tolist())
                # Also try symmetric position
                sym = k_star - int(i)
                if 0 <= sym < n and sym != i:
                    targets.append(sym)
                # Midpoint-ish
                mid = (i + (n - 1 - i)) // 2
                if mid != i and 0 <= mid < n:
                    targets.append(mid)
                
                for j in targets[:5]:
                    if j == i:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    a = np.maximum(a, 0)
                    s_tmp = float(np.sum(a))
                    a = a / s_tmp
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[j] -= delta
                        a[i] += delta
                
                if improved:
                    break
        
        if not improved:
            # Random perturbation
            rng_r = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            for _ in range(3):
                i = int(rng_r.integers(0, n))
                j = int(rng_r.integers(0, n))
                if i == j:
                    continue
                delta = 0.005 * min(a[i], a[j])
                if delta <= 0:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s_tmp = float(np.sum(a))
                a = a / s_tmp
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = compute_conv(a)
                    improved = True
                    break
        
        if not improved and it > 500:
            break
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-15)


def generate_ramp(n, alpha=1.0):
    """Generate a power-law ramp: a[i] = (i+1)^alpha."""
    i = np.arange(n) + 1
    a = i ** alpha
    return a / np.sum(a)


def generate_poly(n, coeffs):
    """Generate polynomial-shape sequence."""
    i = np.linspace(0, 1, n)
    a = np.zeros(n)
    for c, p in coeffs:
        a += c * (i ** p)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500):

            s = float(np.sum(parent_arr))

                obj = compute_obj(parent)
                best_score = obj
```
