```
def softmax(v, n):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    vmax = float(v.max())
    if not np.isfinite(vmax): return np.ones(n)/n
    e = np.exp(v - vmax)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s): return np.ones(n)/n
    a = e / s
    if a.min() <= 0.0:
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
    return a
```

```
def lp_polish(a0, time_budget, seed=123, verbose=True):
    if time_budget < 30.0:
```

```python
"""
Optimizes nonnegative step-function sequences by minimizing
2*n*max(conv(a,a))/(sum(a))^2.

Strategy:
- Use FFT-based smooth log-sum-exp gradients in the nonnegative simplex.
- Mirror-descent / Adam-style updates in log-odds space.
- Multiple lengths, multiple diverse starts, and a low-dimensional Fourier
  profile search to escape local basins.
- Optional dense LP polishing for small enough lengths.
"""

import time
import math
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def next_pow2(m):
    size = 1
    m = int(m)
    while size < m:
        size <<= 1
    return size


def prepare_a(a0):
    a = np.asarray(a0, dtype=np.float64).copy()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a[~np.isfinite(a)] = 0.0
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        a = np.ones_like(a, dtype=np.float64)
    a += 1e-15
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones_like(a, dtype=np.float64)
    else:
        a = a / s
    return a


def softmax(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    vmax = float(v.max())
    if not np.isfinite(vmax):
        return np.ones_like(v, dtype=np.float64) / max(1, v.size)
    e = np.exp(v - vmax)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(v, dtype=np.float64) / max(1, v.size)
    a = e / s
    if float(a.min()) <= 0.0:
        a = np.maximum(a, 1e-15)
        a = a / float(a.sum())
    return a


def objective_value(a):
    a = prepare_a(a)
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    b = irfft(fa * fa, size)[:2 * n - 1]
    bmax = float(np.max(b))
    return float(2.0 * n * bmax)


def resample_sequence(seq, new_n):
    seq = np.asarray(seq, dtype=np.float64)
    seq = np.maximum(seq, 0.0)
    seq[~np.isfinite(seq)] = 0.0
    s = float(seq.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    seq = seq / s
    new_n = int(new_n)
    if new_n <= 0:
        return np.ones(1, dtype=np.float64)
    if len(seq) == new_n:
        return seq.copy()
    if len(seq) == 1:
        return np.full(new_n, float(seq[0]), dtype=np.float64)
    xold = np.linspace(0.0, 1.0, len(seq))
    xnew = np.linspace(0.0, 1.0, new_n)
    y = np.interp(xnew, xold, seq)
    return prepare_a(y)


def compute_b_p_grad(a, gamma, size):
    """
    Returns:
      bmax = max conv(a,a)
      b = conv(a,a)
      p = softmax((b-bmax)/beta), beta=bmax*gamma
      grad = n * d logsumexp(b) / d a
    """
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = float(a.sum())
    L = 2 * n - 1
    if not np.isfinite(s) or s <= 1e-12:
        bmax = 0.0
        b = np.zeros(L, dtype=np.float64)
        p = np.full(L, 1.0 / max(1, L), dtype=np.float64)
        grad = np.zeros(n, dtype=np.float64)
        return bmax, b, p, grad
    if abs(s - 1.0) > 1e-12:
        a = a / s

    fa = rfft(a, size)
    b = irfft(fa * fa, size)[:L]
    b = np.maximum(b, 0.0)
    bmax = float(np.max(b))
    if not np.isfinite(bmax) or bmax <= 1e-15:
        p = np.full(L, 1.0 / max(1, L), dtype=np.float64)
        return bmax, b, p, np.zeros(n, dtype=np.float64)

    beta = max(float(gamma) * bmax, 1e-15)
    z = (b - bmax) / beta
    z = np.clip(z, -745.0, 0.0)
    p = np.exp(z)
    psum = float(p.sum())
    if not np.isfinite(psum) or psum <= 1e-15:
        p = np.full(L, 1.0 / max(1, L), dtype=np.float64)
    else:
        p = p / psum

    fp = rfft(p, size)
    frp = rfft(p[::-1], size)

    conv_ap = irfft(fa * fp, size)[:n]
    conv_ar = irfft(fa * frp, size)
    T1 = conv_ar[n - 1:2 * n - 1][::-1]
    T2 = conv_ap
    grad = float(n) * (T1 + T2)
    return bmax, b, p, grad


def smooth_optimize(a0, time_budget, gamma_start=0.25, gamma_end=0.005,
                    initial_lr=0.03, label="", verbose=False):
    if time_budget <= 0.0:
        a = prepare_a(a0)
        return a, objective_value(a)

    a = prepare_a(a0)
    n = len(a)
    size = next_pow2(3 * n - 2)

    loga = np.log(a)
    loga -= float(loga.mean())

    best_a = a.copy()
    best_loga = loga.copy()
    best_obj = objective_value(a)

    adam_m = np.zeros(n, dtype=np.float64)
    adam_v = np.zeros(n, dtype=np.float64)

    lr = float(initial_lr)
    min_lr = 0.0015
    max_lr = 0.18
    no_improve = 0
    t = 0
    start = time.monotonic()
    limit = start + float(time_budget)

    beta1 = 0.90
    beta2 = 0.999

    while time.monotonic() < limit:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(1.0, float(time_budget)))
        gamma = gamma_start * (gamma_end / gamma_start) ** progress

        bmax, _b, _p, grad = compute_b_p_grad(a, gamma, size)
        obj = float(2.0 * n * bmax)

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            best_loga = loga.copy()
            no_improve = 0
            if lr < max_lr:
                lr *= 1.004
        else:
            no_improve += 1

        if no_improve > 12000:
            lr *= 0.88
            no_improve = 0
        if no_improve > 30000:
            if obj > best_obj * 1.01 + 1e-12:
                loga = best_loga.copy()
                a = best_a.copy()
                adam_m *= 0.1
                adam_v *= 0.2
                lr *= 0.8
            no_improve = 0

        if lr < min_lr:
            break

        grad = np.asarray(grad, dtype=np.float64)
        grad = grad - float(np.mean(grad))
        rms = float(np.sqrt(np.mean(grad * grad)))
        if not np.isfinite(rms):
            grad = np.where(np.isfinite(grad), grad, 0.0)
            rms = float(np.sqrt(np.mean(grad * grad)))
        if not np.isfinite(rms) or rms <= 1e-14:
            break
        if rms > 50.0:
            grad = grad * (50.0 / rms)

        adam_m = beta1 * adam_m + (1.0 - beta1) * grad
        adam_v = beta2 * adam_v + (1.0 - beta2) * (grad * grad)

        step = lr * adam_m / (np.sqrt(adam_v) + 1e-8)
        step_rms = float(np.sqrt(np.mean(step * step)))
        if not np.isfinite(step_rms) or step_rms <= 1e-16:
            break
        if step_rms > 0.5:
            step = step * (0.5 / step_rms)
        step = np.clip(step, -2.0, 2.0)

        loga = loga - step
        loga -= float(loga.mean())

        vmin = float(loga.min())
        vmax = float(loga.max())
        if vmax - vmin > 50.0:
            mid = float(loga.mean())
            loga = np.clip(loga, mid - 25.0, mid + 25.0)

        a = softmax(loga)

        if verbose and (t % 10000 == 0):
            print(f"{label} it={t} best={best_obj:.9f} cur={obj:.9f} "
                  f"lr={lr:.4f} gamma={gamma:.4f}")

    best_a = prepare_a(best_a)
    final_obj = objective_value(best_a)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = prepare_a(best_a)
    return best_a, best_obj


def candidate_fourier_odd(n, amp, rng):
    n = int(n)
    x = (np.arange(n) + 0.5) / float(n)
    loga = np.zeros(n, dtype=np.float64)
    n_modes = max(1, min(24, n // 16))
    for m in range(1, n_modes + 1):
        coeff = float(rng.normal(0.0, float(amp))) / math.sqrt(float(m))
        if coeff != 0.0:
            loga += coeff * np.sin(2.0 * math.pi * m * (x - 0.5))
    loga -= float(loga.mean())
    a = softmax(loga)
    return prepare_a(a)


def candidate_random_smooth(n, amp, rng):

    x = (np.arange(n) + 0.5) / float(n)
    v = np.zeros(n, dtype=np.float64)
    nfeat = max(1, min(40, n // 12))
    for k in range(1, nfeat + 1):
        coeff = float(rng.normal(0.0, float(amp))) / (float(k) ** 0.7)
        phase = float(rng.uniform(0.0, 2.0 * math.pi))
        v += coeff * np.cos(2.0 * math.pi * k * x + phase)
    v -= float(v.mean())
    a = softmax(v)
    return prepare_a(a)


def candidate_skew(n, c):
    n = int(n)
    x = (np.arange(n) + 0.5) / float(n)
    a = 1.0 + float(c) * (2.0 * x - 1.0)
    return prepare_a(a)


def fourier_profile(n, time_budget, seed, max_modes=16, verbose=False):
    n = int(n)
    if time_budget < 10.0:
        a = np.ones(n, dtype=np.float64)
        return a, objective_value(a)

    total_start = time.monotonic()
    deadline = total_start + float(time_budget)

    x = (np.arange(n) + 0.5) / float(n)
    M = 2 * max_modes
    feats = np.empty((n, M), dtype=np.float64)
    idx = 0
    for m in range(1, max_modes + 1):
        feats[:, idx] = np.cos(2.0 * math.pi * m * x)
        idx += 1
        feats[:, idx] = np.sin(2.0 * math.pi * m * x)
        idx += 1

    size = next_pow2(3 * n - 2)

    best_a = np.ones(n, dtype=np.float64)
    best_obj = objective_value(best_a)

    rng = np.random.default_rng(seed + 101)
    inits = []
    inits.append(np.zeros(M, dtype=np.float64))
    for scale in (0.5, 1.0, 1.8):
        c = rng.normal(0.0, 1.0, M)
        decay = 1.0 / np.sqrt(np.arange(1, M + 1, dtype=np.float64) + 1.0)
        c = c * float(scale) * decay
        inits.append(c)

    base_per = max(5.0, (float(time_budget) - 5.0) / max(1, len(inits)))

    for ii in range(len(inits)):
        if time.monotonic() > deadline - 2.0:
            break
        c = np.array(inits[ii], dtype=np.float64, copy=True)
        m = np.zeros(M, dtype=np.float64)
        v = np.zeros(M, dtype=np.float64)
        lr = 0.05
        t = 0

        sub_start = time.monotonic()
        remaining_sub = min(base_per, deadline - sub_start)
        if remaining_sub < 3.0:
            break

        while time.monotonic() - sub_start < remaining_sub:
            t += 1
            loga = feats.dot(c)
            loga -= float(loga.max())
            a = softmax(loga)
            a = prepare_a(a)

            elapsed = time.monotonic() - sub_start
            progress = min(1.0, elapsed / max(1.0, remaining_sub))
            gamma = 1.0 * (0.10 / 1.0) ** progress

            bmax, _b, _p, grad_scaled = compute_b_p_grad(a, gamma, size)
            obj = float(2.0 * n * bmax)
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()

            gm = grad_scaled - float(np.dot(a, grad_scaled))
            gc = feats.T.dot(a * gm)
            gc = np.where(np.isfinite(gc), gc, 0.0)
            gc_rms = float(np.sqrt(np.mean(gc * gc)))
            if not np.isfinite(gc_rms) or gc_rms <= 1e-14:
                break
            if gc_rms > 50.0:
                gc = gc * (50.0 / gc_rms)

            m = 0.90 * m + 0.10 * gc
            v = 0.999 * v + 0.001 * (gc * gc)
            step = lr * m / (np.sqrt(v) + 1e-8)
            step_rms = float(np.sqrt(np.mean(step * step)))
            if np.isfinite(step_rms) and step_rms > 1.0:
                step = step * (1.0 / step_rms)
            step = np.clip(step, -2.0, 2.0)
            c = c - step

            cmax = float(np.max(np.abs(c)))
            if np.isfinite(cmax) and cmax > 10.0:
                c = c * (10.0 / cmax)

        if verbose:
            print(f"fourier init {ii}: obj={best_obj:.9f}")

    best_obj = objective_value(best_a)
    return best_a, best_obj


def optimize_length(n, base, time_budget, rng, label=""):
    n = int(n)
    if time_budget <= 0.0:
        a = resample_sequence(base, n) if base is not None else np.ones(n)
        return a, objective_value(a)

    base_n = resample_sequence(base, n) if base is not None else np.ones(n)
    local_a = base_n
    local_obj = objective_value(local_a)

    starts = []
    starts.append(("base", 0, base_n))

    noise = rng.normal(0.0, 0.07, n)
    starts.append(("base_noise", 1, base_n * np.exp(noise)))

    rand = candidate_random_smooth(n, 0.7, rng)
    mix = 0.6 * base_n + 0.4 * rand
    starts.append(("mix_rand", 2, mix))

    f1 = candidate_fourier_odd(n, 0.7, rng)
    starts.append(("fourier1", 3, f1))
    f2 = candidate_fourier_odd(n, 1.4, rng)
    starts.append(("fourier2", 4, f2))

    sk1 = candidate_skew(n, 0.45)
    starts.append(("skew1", 5, sk1))
    sk2 = candidate_skew(n, 0.7)
    starts.append(("skew2", 6, sk2))

    rev = base_n[::-1].copy()
    sym = 0.5 * (base_n + rev)
    starts.append(("sym", 7, sym))

    roll1 = np.roll(base_n, max(1, n // 7))
    starts.append(("roll1", 8, roll1))
    roll2 = np.roll(base_n, -max(1, n // 7))
    starts.append(("roll2", 9, roll2))

    scored = []
    for name, idx, sa in starts:
        a = prepare_a(sa)
        obj = objective_value(a)
        scored.append((obj, idx, name, a))
    scored.sort()

    if len(scored) > 0:
        first_obj = scored[0][0]
        if first_obj < local_obj:
            local_obj = first_obj
            local_a = scored[0][3].copy()

    if time_budget > 300.0:
        topk = 4
        fracs = (0.36, 0.28, 0.20, 0.16)
    elif time_budget > 150.0:
        topk = 3
        fracs = (0.45, 0.32, 0.23)
    elif time_budget > 70.0:
        topk = 2
        fracs = (0.62, 0.38)
    else:
        topk = 1
        fracs = (1.0,)

    topk = min(topk, len(scored))

    if topk == 4:
        gamma_arr = (0.40, 0.25, 0.15, 0.08)
        lr_arr = (0.03, 0.04, 0.05, 0.02)
    elif topk == 3:
        gamma_arr = (0.35, 0.20, 0.10)
        lr_arr = (0.03, 0.04, 0.05)
    elif topk == 2:
        gamma_arr = (0.30, 0.15)
        lr_arr = (0.03, 0.05)
    else:
        gamma_arr = (0.25,)
        lr_arr = (0.03,)

    local_start = time.monotonic()
    for i in range(topk):
        elapsed_local = time.monotonic() - local_start
        remaining_local = float(time_budget) - elapsed_local
        if remaining_local < 8.0:
            break
        alloc = float(fracs[i]) * float(time_budget)
        alloc = min(alloc, remaining_local * 0.92)
        if alloc < 4.0:
            break

        a_start = scored[i][3]
        start_obj = scored[i][0]
        if start_obj < local_obj:
            local_obj = start_obj
            local_a = a_start.copy()

        a_new, obj_new = smooth_optimize(
            a_start,
            alloc,
            gamma_start=float(gamma_arr[i]),
            gamma_end=0.004,
            initial_lr=float(lr_arr[i]),
            label=label + ":" + scored[i][2],
            verbose=False
        )
        if obj_new < local_obj - 1e-13:
            local_obj = obj_new
            local_a = a_new.copy()
            print(f"{label} improved with {scored[i][2]}: {obj_new:.9f}")

    return local_a, local_obj


def select_n_options(n_base):
    base = int(n_base)
    raw = []
    for mult in (1.0, 1.2, 1.5, 2.0, 2.5, 3.0):
        raw.append(int(round(base * mult)))
    for extra in (1024, 1536, 2048, 2560, 3072):
        raw.append(extra)
    raw.sort()

    max_n = min(4096, int(3.0 * base) + 64)
    out = []
    for n in raw:
        n = int(n)
        if n < 128 or n > max_n:
            continue
        if not out:
            out.append(n)
            continue
        diff = n - out[-1]
        if diff <= 0:
            continue
        if diff < max(24, 0.08 * float(out[-1])):
            continue
        out.append(n)

    if len(out) == 0:
        out = [max(128, base)]
    if len(out) > 8:
        out = out[:8]
    return out


def build_conv_matrix(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_polish(a0, time_budget, seed=123, verbose=False):
    a = prepare_a(a0)
    n = len(a)
    if time_budget < 30.0 or n > 1200:
        return a, objective_value(a)

    best_a = a.copy()
    best_obj = objective_value(a)

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + float(time_budget)
    stall = 0

    while time.monotonic() < deadline - 5.0:
        A = build_conv_matrix(a)
        c = np.zeros(n + 1, dtype=np.float64)
        c[-1] = 1.0

        A_ub = np.zeros((2 * n - 1, n + 1), dtype=np.float64)
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1, dtype=np.float64)

        A_eq = np.ones((1, n + 1), dtype=np.float64)
        A_eq[0, n] = 0.0
        b_eq = np.array([1.0], dtype=np.float64)

        bounds = [(0.0, None)] * n + [(0.0, None)]

        tlimit = max(2.0, min(60.0, deadline - time.monotonic() - 3.0))
        res = None
        try:
            res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options={"time_limit": tlimit}
            )
        except Exception:
            try:
                res = linprog(
                    c,
                    A_ub=A_ub,
                    b_ub=b_ub,
                    A_eq=A_eq,
                    b_eq=b_eq,
                    bounds=bounds,
                    method="highs"
                )
            except Exception:
                res = None

        if res is None or getattr(res, "status", -1) != 0:
            break

        x = getattr(res, "x", None)
        if x is None:
            break
        g = np.asarray(x[:n], dtype=np.float64)
        g = np.maximum(g, 0.0)
        gs = float(g.sum())
        if not np.isfinite(gs) or gs <= 1e-12:
            break
        g = g / gs

        best_trial_obj = np.inf
        best_trial = None
        for tt in (0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0):
            trial = (1.0 - tt) * a + tt * g
            trial = prepare_a(trial)
            obj = objective_value(trial)
            if obj < best_trial_obj:
                best_trial_obj = obj
                best_trial = trial

        if best_trial is not None and best_trial_obj < best_obj - 1e-13:
            best_obj = best_trial_obj
            best_a = best_trial.copy()
            a = best_trial.copy()
            stall = 0
        else:
            stall += 1
            if stall > 3:
                noise = rng.normal(0.0, 0.12, n)
                a = prepare_a(best_a * np.exp(noise))
                stall = 0

        if verbose:
            remaining = deadline - time.monotonic()
            print(f"lp_polish n={n} obj={best_obj:.9f} remaining={remaining:.1f}s")

    final_obj = objective_value(best_a)
    if final_obj < best_obj:
        best_obj = final_obj
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    start = time.monotonic()
    deadline = start + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = None

    if parent is None or parent.size == 0:
        parent = np.ones(800, dtype=np.float64)
    parent = prepare_a(parent)

    best_a = parent.copy()
    best_score = objective_value(best_a)
    print(f"initial parent n={len(best_a)} score={best_score:.9f}")

    # Low-dimensional smooth profile search.
    total_budget = float(budget_s) - 20.0
    fp_budget = min(80.0, max(0.0, total_budget * 0.055))
    if fp_budget > 20.0 and time.monotonic() < deadline - 25.0:
        fp_a, fp_obj = fourier_profile(
            len(parent),
            fp_budget,
            seed,
            max_modes=16,
            verbose=True
        )
        if fp_obj < best_score - 1e-12:
            best_score = fp_obj
            best_a = fp_a.copy()
            print(f"fourier profile improved: {best_score:.9f}")

    n_options = select_n_options(len(best_a))
    option_budgets = (380.0, 220.0, 180.0, 150.0, 120.0, 90.0, 60.0, 40.0)

    for idx, n in enumerate(n_options):
        remaining = deadline - time.monotonic()
        if remaining < 70.0:
            break
        base_budget = float(option_budgets[idx]) if idx < len(option_budgets) else 40.0
        alloc = min(base_budget, max(30.0, remaining - 90.0))
        if alloc < 30.0:
            break
        best_a, best_score = optimize_length(
            n,
            best_a,
            alloc,
            rng,
            label=f"n={n}"
        )
        print(f"after n={n}: best={best_score:.9f} length={len(best_a)} "
              f"remaining={deadline - time.monotonic():.1f}s")

    remaining = deadline - time.monotonic()
    if remaining > 50.0:
        fb = min(110.0, remaining * 0.45)
        a1, o1 = smooth_optimize(
            best_a,
            fb,
            gamma_start=0.20,
            gamma_end=0.003,
            initial_lr=0.03,
            label="final_smooth"
        )
        if o1 < best_score - 1e-13:
            best_score = o1
            best_a = a1.copy()
            print(f"final_smooth improved: {best_score:.9f}")

    remaining = deadline - time.monotonic()
    if remaining > 90.0 and len(best_a) <= 1200:
        lb = min(90.0, remaining * 0.45)
        a2, o2 = lp_polish(best_a, lb, seed=seed + 7, verbose=True)
        if o2 < best_score - 1e-13:
            best_score = o2
            best_a = a2.copy()
            print(f"lp_polish improved: {best_score:.9f}")

    remaining = deadline - time.monotonic()
    if remaining > 30.0:
        fb2 = min(55.0, remaining * 0.65)
        a3, o3 = smooth_optimize(
            best_a,
            fb2,
            gamma_start=0.10,
            gamma_end=0.002,
            initial_lr=0.02,
            label="final_smooth2"
        )
        if o3 < best_score - 1e-13:
            best_score = o3
            best_a = a3.copy()
            print(f"final_smooth2 improved: {best_score:.9f}")

    result = []
    for x in best_a:
        x = float(x)
        if not np.isfinite(x) or x < 0.0:
            x = 0.0
        if x > 1000.0:
            x = 1000.0
        result.append(x)

    s = float(sum(result))
    if not np.isfinite(s) or s < 0.01:
        result = [1.0] * 100
        s = 100.0

    scaled = []
    for x in result:
        scaled.append(float(x) / s)
    result = scaled

    try:
        val = float(evaluate_sequence(result))
    except Exception:
        val = best_score

    if not np.isfinite(val):
        result = [1.0] * 100
        try:
            val = float(evaluate_sequence(result))
        except Exception:
            val = 2.0

    print(f"FINAL length={len(result)} internal={best_score:.9f} "
          f"evaluate={val:.9f} time={time.monotonic() - start:.1f}s")
    return result
```
