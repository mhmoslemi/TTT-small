"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using
LP alternating, smooth-max gradient descent, coordinate descent polish,
multi-n exploration, and randomized restarts.
Key improvements over previous: better active-set LP, adaptive smooth
max with multiple temperatures, Nelder-Mead style local search,
and aggressive final polishing.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp_for_direction(a, active_indices=None, max_cons=None):
    """
    Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A_full = build_conv_matrix(a_norm)

    if active_indices is not None and len(active_indices) < 2 * n - 1:
        A = A_full[active_indices, :]
    elif max_cons is not None and max_cons < 2 * n - 1:
        # Use most important constraints
        conv_a = A_full @ a_norm
        top_idx = np.argsort(conv_a)[::-1][:max_cons]
        A = A_full[top_idx, :]
    else:
        A = A_full

    num_cons = A.shape[0]

    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((num_cons, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_cons)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 5000, 'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                g_norm = g_val / s
                # Verify against ALL constraints
                conv_check = A_full @ g_norm
                true_M = float(np.max(conv_check))
                return g_norm, true_M
    except Exception:
        pass
    return None, None


def line_search_blend(a, g, n_trials=20):
    """
    Find best t in [0,1] for a_new = (1-t)*a + t*g.
    Uses golden-section search on the non-convex objective.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)

    # Precompute convolutions
    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(np.maximum(val, 0)))

    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = obj_at_t(c1)
    f2 = obj_at_t(c2)

    for _ in range(n_trials):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = obj_at_t(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = obj_at_t(c2)

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.5, a_norm
    a_new = a_new / s
    return t_opt, a_new


def lp_alternating(a, time_budget=200.0, verbose=False):
    """

    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    active_set = None

    max_iters = 1000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Determine active set: top 30% of convolution peaks
        conv_cur = compute_conv(a)
        max_conv = np.max(conv_cur)
        num_active = max(n, min(2 * n - 1, int(0.3 * (2 * n - 1))))
        active_set = np.argsort(conv_cur)[::-1][:num_active]
