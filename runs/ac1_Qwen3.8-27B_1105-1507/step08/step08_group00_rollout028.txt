def compute_conv(a):

    while size < 2 * n - 1:

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)

    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Build row k of convolution matrix: conv(a, g)[k] = sum_i A[k,i]*g[i]."""
    n = len(a)
    row = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            row[i] = a[j]
    return row


def solve_dual_lp(a, active_lags):
    """
    Solve the dual LP:
        max t
        s.t. A^T @ lambda >= t * 1  (n constraints)
             1^T lambda = 1
             lambda >= 0
    where A[k, i] = a[k-i] for valid indices.
    Returns (lambda_opt, t_opt) or (None, None) on failure.
    """
    n = len(a)
    K = len(active_lags)
    if K == 0:
        return None, None

    # Build A matrix (K x n)
    A = np.zeros((K, n))
    for idx_k, k in enumerate(active_lags):
        A[idx_k, :] = build_conv_matrix_row(a, k)

    lam = cp.Variable(K, nonneg=True)
    t = cp.Variable()

    constraints = [
        A.T @ lam >= t,
        cp.sum(lam) == 1
    ]

    prob = cp.Problem(cp.Maximize(t), constraints)

    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if lam.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                lam_val = np.maximum(lam.value, 0)
                s = float(np.sum(lam_val))
                if s > 1e-15:
                    lam_val = lam_val / s
                t_val = float(t.value)
                return lam_val, t_val
        except Exception:
            continue
    return None, None

def fft_subgradient_descent(a, time_budget=200.0, K=15, verbose=False):
    """
    Nesterov accelerated subgradient descent on the simplex.
    Uses top-K peaks with exponential weighting.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.5 / np.sqrt(n)
    momentum = 0.85
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break

        # Softmax weights
        temperature = max_pv / 5.0
        weights = np.exp((peak_vals - max_pv) / temperature)
        weights /= weights.sum()

        # Compute subgradient
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        # Nesterov step
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            a = a_new  # still move for exploration
            if stall > 100:
                lr *= 0.7
                v *= 0.5
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.999

        # Adaptive K: increase if stalled
        if stall > 50 and K < 40:
            K = min(40, K + 5)

        it += 1
        if verbose and it % 5000 == 0:
            print(f"      SGD it{it}: obj={best_obj:.8f}, lr={lr:.6f}, K={K}, t={time.monotonic()-start:.0f}s")

    return best_a


def multiplicative_update(a, time_budget=100.0, verbose=False):
    """
    Multiplicative weight updates (natural for non-negative variables).
    a[i] *= exp(-lr * grad[i]), then renormalize.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.5 / np.sqrt(n)
    K = 10
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break

        weights = np.exp((peak_vals - max_pv) * 8.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        # Multiplicative update with clipping
        update = np.exp(-lr * g * 10.0)  # scale gradient for effect
        update = np.clip(update, 0.5, 2.0)
        a_new = a * update
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.8
                stall = 0
                if lr < 1e-12:
                    break

        it += 1

    return best_a


def smooth_init(n, mode=0, rng=None):
    """Generate diverse smooth initializations."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = 0.5 + 0.5 * np.cos(np.pi * x)  # raised cosine (1 to 0)
    elif mode == 2:
        a = 0.5 - 0.5 * np.cos(2 * np.pi * x)  # Hann window
    elif mode == 3:
        a = 1.0 - (2 * x - 1) ** 2  # parabolic
    elif mode == 4:
        c1 = 0.25 + rng.uniform(-0.05, 0.05)
        c2 = 0.75 + rng.uniform(-0.05, 0.05)
        w = 0.15 + rng.uniform(-0.03, 0.03)
        a = np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / w) ** 2)
    elif mode == 5:
        a = 1.0 / (np.sqrt(1 - (2 * x - 1) ** 2 + 1e-6))  # arcsine-like
    elif mode == 6:
        freq = rng.integers(2, 5)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x)
        a = np.maximum(a, 0.01)
    elif mode == 7:
        a = x ** rng.uniform(0.5, 3.0)
    elif mode == 8:
        # Bimodal with flat center
        a = np.ones(n)
        a[:n//4] *= 1.5
        a[3*n//4:] *= 1.5
    else:
        a = rng.uniform(0.5, 1.5, n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def local_search(a, time_budget=60.0, max_iter=50):
    """
    Simple local search: try small random perturbations and keep improvements.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    it = 0
    while time.monotonic() - start < time_budget and it < max_iter:
        # Random perturbation
        scale = 0.02 * max(1.0, (1.0 - (time.monotonic() - start) / time_budget))
        a_try = a * np.exp(rng.normal(0, scale, n))
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            it += 1
            continue
        a_try = a_try / s_try

        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
        it += 1

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: dual-based active-set LP + FFT subgradient + multi-n + restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: FFT Subgradient Descent (main workhorse) ===
    print(f"\n=== Phase 1: Subgradient Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.20, 250.0)
    if p1_budget > 30:
        a_sgd = fft_subgradient_descent(best_seq, time_budget=p1_budget, K=15, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            n_current = len(best_seq)
            print(f"  Phase1 SGD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 SGD: obj={obj_sgd:.8f}")

    # === Phase 2: Active-Set LP Polishing ===
    print(f"\n=== Phase 2: Active-Set LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 500.0)
    if p2_budget > 30:
        # Multiple rounds with perturbation between
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            alloc = min(p2_budget / (4 - rnd), remaining * 0.6)
            if alloc < 20:
                break
            a_lp = active_set_lp_polish(best_seq, time_budget=alloc,
                                        max_active=60, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                n_current = len(best_seq)
                print(f"  AS-LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  AS-LP round {rnd}: obj={obj_lp:.8f}")
                if rnd < 3:
                    # Perturb to escape local minimum
                    scale = 0.02 * (rnd + 1)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()

    # === Phase 3: Multiplicative Updates ===
    print(f"\n=== Phase 3: Multiplicative Updates ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.10, 100.0)
    if p3_budget > 20:
        a_mu = multiplicative_update(best_seq, time_budget=p3_budget)
        obj_mu = compute_obj(a_mu)
        if obj_mu < best_score:
            best_score = obj_mu
            best_seq = a_mu.copy()
            n_current = len(best_seq)
            print(f"  Phase3 MU: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 MU: obj={obj_mu:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Generate candidate lengths
    n_options = sorted(set([
        max(300, n_current - 200), max(300, n_current - 100),
        n_current + 50, n_current + 100, n_current + 200, n_current + 400,
        500, 600, 800, 1000, 1200, 1400
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500 and abs(n - n_current) > 30]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick subgradient
        alloc1 = min(remaining * 0.20, 60.0)
        if alloc1 > 15:
            a_try = fft_subgradient_descent(a_try, time_budget=alloc1, K=10)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.20, 60.0)
        if alloc2 > 15:
            a_try = active_set_lp_polish(a_try, time_budget=alloc2, max_active=50)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break

        mode = r % 9
        a_r = smooth_init(n_best, mode=mode, rng=rng)
        # Blend with current best
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # Optimize
        alloc1 = min(remaining * 0.20, 80.0)
        if alloc1 > 20:
            a_r = fft_subgradient_descent(a_r, time_budget=alloc1 * 0.5, K=10)
            remaining = deadline - time.monotonic()
            alloc2 = min(remaining * 0.15, 50.0)
            if alloc2 > 15:
                a_r = active_set_lp_polish(a_r, time_budget=alloc2, max_active=40)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Active-set LP with more iterations
    if remaining > 40:
        alloc = min(remaining * 0.35, 150.0)
        a_lp = active_set_lp_polish(best_seq, time_budget=alloc, max_active=80, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final AS-LP: obj={best_score:.8f} **")
        else:
            print(f"  Final AS-LP: obj={obj_lp:.8f}")

    # Subgradient with high K
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 80.0)
        a_sgd = fft_subgradient_descent(best_seq, time_budget=alloc, K=30)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final SGD: obj={best_score:.8f} **")

    # Local search
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.20, 50.0)
        a_ls = local_search(best_seq, time_budget=alloc)
        obj_ls = compute_obj(a_ls)
        if obj_ls < best_score:
            best_score = obj_ls
            best_seq = a_ls.copy()
            print(f"  Local search: obj={best_score:.8f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.40, 60.0)
        a_lp = active_set_lp_polish(best_seq, time_budget=alloc, max_active=100)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
