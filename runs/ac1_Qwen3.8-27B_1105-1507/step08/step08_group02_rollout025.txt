from scipy.sparse import csr_matrix

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:

    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a):

    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):

        k_start = j

    # Actually: sum_j w[i+j] * a[j] = sum_j w'[j] * a[j] where w'[j] = w[i+j]
    # This is the inner product of a with a shifted version of w.
    # cross_correlation(w, a)[i] = sum_j w[j+i] * a[j]
    # = correlation of w with a at lag i (sliding a over w)
    
    # Simpler: just compute it as a convolution
    # grad[i] = sum_{j=0}^{n-1} w[i+j] * a[j]  for i where i+j < 2n-1
    # Let's pad w to length 3n and compute
    size = 1
    while size < 3 * n:
        size <<= 1
    fw = rfft(weights, size)
    fa = rfft(a, size)
    # cross_correlation: IFFT(conj(fw) * fa) gives sum_j w[j-i] * a[j]
    # We want sum_j w[j+i] * a[j] = IFFT(fw * conj(fa))... 
    # Let me just use the direct formula with a different approach
    
    # grad[i] = sum_{k} w[k] * a[k-i] + w[k] * a[i-k]
    # First term: (w * a)[i] where * is cross-correlation = IFFT(conj(FW) * FA)
    # Second term: (w * a)[i] where * is convolution = IFFT(FW * FA)
    # But w has length 2n-1 and a has length n, so we need to be careful
    
    # Let's just do it directly but vectorized:
    # grad[i] = sum_{j=0}^{n-1, i+j < 2n-1} w[i+j] * a[j]
    # For each i, the valid j range is 0 <= j < min(n, 2n-1-i)
    
    # The most efficient way: create a Toeplitz-like product
    # Actually for n=800, a simple loop with numpy dot is fine (~0.5ms)
    g = np.zeros(n)
    for i in range(n):
        j_max = min(n, 2 * n - 1 - i)
        if j_max > 0:
            g[i] = 2.0 * np.dot(w_interesting(weights, i, j_max), a[:j_max])
    return g


def w_interesting(weights, i, j_max):
    """Get weights[i:i+j_max] for the gradient computation."""
    return weights[i:i + j_max]


def smooth_optimize(a, time_budget=100.0, beta_start=30.0, beta_end=2000.0):
    """Smooth optimization using log-sum-exp surrogate with mirror descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    it = 0
    lr = 1.0 / np.sqrt(n)
    
    while time.monotonic() - start < time_budget:
        if it % 100 == 0:
            frac = min(1.0, it / (time_budget * 5))
            beta = beta_start * (beta_end / beta_start) ** frac
        else:
            beta = beta_start * (beta_end / beta_start) ** min(1.0, it / (time_budget * 5))
        
        # Compute smooth gradient
        conv = compute_conv(a)
        m = conv.max()
        shifted = beta * (conv - m)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Gradient: grad[i] = 2 * sum_j weights[i+j] * a[j]
        g = np.zeros(n)
        for i in range(n):
            j_max = min(n, 2 * n - 1 - i)
            if j_max > 0:
                g[i] = 2.0 * np.dot(weights[i:i + j_max], a[:j_max])
        
        # Mirror descent (exponential update)
        step = lr / np.sqrt(beta)
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - step * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = max(0.001, lr * 0.999)
        else:
            # Try smaller step
            log_a_new2 = log_a - step * 0.3 * g
            log_a_new2 -= log_a_new2.max()
            a_new2 = np.exp(log_a_new2)
            a_new2 = a_new2 / a_new2.sum()
            obj_new2 = compute_obj(a_new2)
            if obj_new2 < best_obj - 1e-12:
                best_obj = obj_new2
                best_a = a_new2.copy()
                a = a_new2.copy()
            else:
                lr = max(0.001, lr * 0.95)
        
        it += 1
    
    return best_a


def peak_shave(a, time_budget=30.0):
    """
    Peak shaving: iteratively move mass from peak-contributing positions
    to non-contributing positions to reduce the maximum convolution value.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    conv = compute_conv(a)
    start = time.monotonic()
    
    max_rounds = 200
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]
        
        # Find contributors to the peak
        contribs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                c = a[i] * a[j]
                if c > 1e-12:
                    contribs.append((c, i, j))
        
        if not contribs:
            break
        contribs.sort(reverse=True)
        
        # Find good target positions (not contributing to peak)
        contrib_positions = set()
        for _, i, j in contribs[:20]:
            contrib_positions.add(i)
            contrib_positions.add(j)
        
        targets = [i for i in range(n) if i not in contrib_positions and a[i] < 0.3 * a.mean() * 3]
        if not targets:
            targets = [i for i in range(n) if i not in contrib_positions]
        if not targets:
            break
        
        # Randomly shuffle targets
        rng = np.random.default_rng(rnd + n)
        rng.shuffle(targets)
        
        improved = False
        for contrib, i, j in contribs[:15]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break
            
            # Source: the larger of a[i], a[j]
            if i == j:
                src = i
                max_delta = a[i] * 0.5
            else:
                src = i if a[i] >= a[j] else j
                max_delta = min(a[i], a[j]) * 0.3
            
            for target in targets[:5]:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = frac * max_delta
                    if delta < 1e-12 or delta >= a[src]:
                        continue
                    
                    a[src] -= delta
                    a[target] += delta
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s < 1e-10:
                        a[src] += delta
                        a[target] -= delta
                        continue
                    a = a / s
                    
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        a = a / a.sum()
                        break
        
        if not improved:
            # Try more aggressive: reduce the top contributor significantly
            if contribs and contribs[0][0] > 1e-8:
                _, i, j = contribs[0]
                src = i if a[i] >= a[j] else j
                for target in targets[:10]:
                    for frac in [0.3, 0.5]:
                        delta = frac * a[src]
                        a[src] -= delta
                        a[target] += delta
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        new_conv = compute_conv(a)
                        new_obj = 2.0 * n * float(np.max(new_conv))
                        if new_obj < best_obj - 1e-12:
                            best_obj = new_obj
                            best_a = a.copy()
                            conv = new_conv
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[target] -= delta
                            a = a / a.sum()
                    if improved:
                        break
            if not improved:
                break
    
    return best_a


def lp_anneal_optimize(a, time_budget=200.0, seed=42):
    """
    LP alternating with simulated annealing for escaping local minima.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    patience = 30
    temperature = 0.1
    iteration = 0
    last_good_t = 0.5
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed / time_budget > 0.9:
            break
        
        # Solve LP
        g_lp, M_lp = solve_lp_fast(a)
        if g_lp is None:
            g_lp = a.copy()
        
        # Quick line search
        current_conv_max = float(np.max(compute_conv(a / a.sum())))
        
        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = best_a.copy()
        
        # Try a few t values
        candidate_ts = [1.0, 0.7, 0.5, 0.3, 0.1]
        if last_good_t != 0.5:
            candidate_ts.append(max(0.05, min(1.0, last_good_t)))
            candidate_ts.append(max(0.05, min(1.0, last_good_t * 0.7)))
            candidate_ts.append(max(0.05, min(1.0, last_good_t * 1.3)))
        
        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        iteration += 1
        if best_t_obj < best_obj - 1e-11:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local.copy()
            last_good_t = best_t_val
            stall = 0
            temperature = max(0.01, temperature * 0.98)
        else:
            stall += 1
            if stall >= patience:
                # Annealing restart
                noise_scale = temperature * (1.0 + 0.5 * stall / patience)
                a = best_a * (1.0 + noise_scale * rng.normal(0, 1, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                
                # Sometimes blend with a random direction
                if rng.random() < 0.4:
                    random_dir = rng.dirichlet(np.ones(n) * 0.5)
                    blend = 0.1 + 0.3 * rng.random()
                    a = (1 - blend) * best_a + blend * random_dir
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                
                best_obj = compute_obj(a)
                if best_obj < compute_obj(best_a):
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                stall = 0
                last_good_t = 0.5
                temperature = min(0.4, temperature * 2.5)
                patience = max(15, patience // 2)
            else:
                if stall > patience // 2:
                    # Smaller perturbations
                    a = best_a * (1.0 + 0.02 * rng.normal(0, 1, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                temperature = max(0.02, temperature * 0.95)
    
    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two peaks (anti-palindromic)
        a = np.zeros(n)
        a[:n//2] = 1.0
        a = a / a.sum()
    elif mode == 3:
        # Smooth anti-palindromic
        x = np.linspace(0, 1, n)
        a = np.exp(5.0 * x)  # Heavily weighted towards one side
        a = a / a.sum()
    elif mode == 4:
        # Gaussian at edge
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) ** 2) / 0.05)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Chirp
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.sin(20 * np.pi * x + x * 10)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 3
        a = a / a.sum()
    elif mode == 7:
        # Block anti-palindromic
        a = np.zeros(n)
        block_size = n // 4
        a[:block_size] = 1.0
        a[3*block_size:] = 1.0
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return np.maximum(a_new, 0)


def coordinate_descent_refine(a, time_budget=30.0):
    """
    Coordinate descent: adjust pairs of coordinates to reduce max convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for rnd in range(50):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find top contributors
        contribs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(reverse=True)
        
        if not contribs or contribs[0][0] < 1e-12:
            break
        
        improved = False
        for contrib, i, j in contribs[:10]:
            if time.monotonic() - start > time_budget:
                break
            src = i if a[i] >= a[j] else j if i != j else i
            
            # Find best target
            for target in range(n):
                if target in (i, j) and i != j:
                    continue
                if target == i and target == j:
                    continue
                
                delta = 0.1 * a[src]
                if delta >= a[src]:
                    continue
                
                a[src] -= delta
                a[target] += delta
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    a[src] += delta
                    a[target] -= delta
                    continue
                a = a / s
                
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[src] += delta
                    a[target] -= delta
                    a = a / a.sum()
        
        if not improved:
            break
    
    return best_a
