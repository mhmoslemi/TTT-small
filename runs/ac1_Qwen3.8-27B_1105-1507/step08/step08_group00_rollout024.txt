"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated multiplicative weight updates, intensive LP polishing with
line search, subgradient refinement, and multi-length exploration.
Key improvements over previous: multiplicative updates for fast non-negativity-
preserving optimization, more aggressive LP iteration with line search,
and adaptive peak-targeting.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time
import math

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def cross_corr_for_grad(p, a):
    """
    Compute c[m] = sum_j p[j] * a[j-m] for m = 0, ..., 2n-2.
    This is the gradient (up to factor 2) of sum_k p[k]*conv(a,a)[k] w.r.t. a.
    Uses FFT: c = conv(p, a_rev)[n-1 : 3n-2]
    """
    n = len(a)
    a_rev = a[::-1]
    N = 1
    while N < (2*n - 1) + n - 1 + 1:
        N <<= 1
    P = fft(p, N)
    A = fft(a_rev, N)
    conv_full = ifft(P * A).real
    return conv_full[n-1 : 3*n-2]  # length 2n-1

def multiplicative_update(a, tau, eta):
    """
    One step of multiplicative weight update.
    a: normalized (sum=1), non-negative
    tau: temperature for softmax (higher = more peaked)
    eta: step size
    Returns updated a (normalized, non-negative).
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Softmax weights (numerically stable)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    # Gradient: 2 * c[m] where c[m] = sum_j p[j] * a[j-m]
    c = cross_corr_for_grad(p, a)
    grad = 2.0 * c  # only need first n components
    grad_n = grad[:n]
    
    # Multiplicative update
    a_new = a * np.exp(-eta * grad_n)
    
    # Numerical safety
    a_new = np.maximum(a_new, 1e-30)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    a_new = a_new / s
    
    return a_new

def multiplicative_phase(a, time_budget=100.0, verbose=True):
    """Phase 1: Multiplicative weight updates with annealed tau and eta."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Annealing parameters
    tau_start = 2.0
    tau_end = 200.0
    eta_start = 1.0 / np.sqrt(n)
    eta_end = 0.1 / np.sqrt(n)
    
    start = time.monotonic()
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Anneal tau (exponential) and eta (linear decay)
        tau = tau_start * (tau_end / tau_start) ** progress
        eta = eta_start + (eta_end - eta_start) * progress
        
        a = multiplicative_update(a, tau, eta)
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            best_it = it
        
        it += 1
        
        if verbose and it % 500 == 0:
            print(f"    MU it{it}: obj={best_obj:.8f}, tau={tau:.1f}, eta={eta:.4f}, t={elapsed:.0f}s")
        
        # Reset if stuck
        if it - best_it > 5000:
            a = best_a.copy()
            best_it = it
            eta = eta * 0.5
    
    return best_a

def solve_lp_convolution(a, solver='cvxpy'):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g*, M*).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Build A matrix: A[i,j] = a[i-j] if 0 <= i-j <= n-1
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a_norm[:n]
    
    # Try cvxpy first
    try:
        import cvxpy as cp
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        for solver_name in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver_name)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M_var.value)
            except Exception:
                continue
    except Exception:
        pass
    
    # Fallback: scipy linprog
    try:
        # Variables: [g_0, ..., g_{n-1}, M]
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize M
        
        # A_ub @ x <= b_ub
        # conv(a,g)[k] - M <= 0 for each k
        A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
        b_ub = np.zeros(2*n-1)
        
        # A_eq @ x = b_eq
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(None, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[-1])
    except Exception:
        pass
    
    return None, None

def lp_polish_phase(a, time_budget=500.0, verbose=True):
    """Phase 2: Intensive LP polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    it = 0
    stall = 0
    rng = np.random.default_rng(42)
    last_t = 0.3
    
    while time.monotonic() - start < time_budget:
        # Solve LP
        g_lp, M_lp = solve_lp_convolution(a)
        if g_lp is None:
            break
        
        # Line search on t
        t_candidates = sorted(set([
            0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5,
            0.6, 0.7, 0.8, 0.9, 1.0,
            max(0.02, last_t * 0.5), min(1.0, last_t * 1.5),
            max(0.02, last_t * 0.7), min(1.0, last_t * 1.3)
        ]))
        
        best_t_obj = best_obj
        best_t_val = last_t
        best_a_local = best_a.copy()
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        it += 1
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.2f}, M={M_lp:.6f}, t_elapsed={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 10:
                # Perturb
                scale = 0.01 * min(stall, 20)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
            if stall > 50:
                break
    
    return best_a

def subgradient_phase(a, time_budget=150.0, K=5, verbose=True):
    """Phase 3: Subgradient method with top-K peaks."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0 / np.sqrt(n)
    momentum = 0.5
    v = np.zeros(n)
    stall = 0
    
    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        
        # Weighted subgradient
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 10.0)
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            # grad[m] = 2 * a[k-m] for valid k-m
            for m in range(n):
                if 0 <= k - m <= n - 1:
                    g[m] += w * 2.0 * a[k - m]
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        
        v = momentum * v + g
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                lr *= 0.7
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.998
        
        it += 1
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
