def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_obj_normalized(a):
    """Compute objective assuming a is already normalized (sum=1)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def smooth_grad_fft(a, beta):
    """
    Compute gradient of smooth max-conv using FFT.
    Returns (gradient, smooth_value).
    O(n log n) via FFT-based cross-correlation.
    """
    n = len(a)
    conv = compute_conv(a)
    # Log-sum-exp for numerical stability
    m = conv.max()
    shifted = beta * (conv - m)
    weights = np.exp(shifted)
    weights /= weights.sum()
    smooth_val = m + np.log(weights.sum()) / beta  # = m + log(1)/beta = m... 
    # Actually smooth_val = m + (1/beta)*log(sum(exp(beta*(conv-m)))) = m + log(sum(exp(...)))/beta
    # But sum(exp(...)) = summary of weights * sum(weights) = weights.sum() since we normalized
    # So smooth_val = m + log(1)/beta = m. That's wrong.
    # Let me redo: smooth_val = (1/beta) * log(sum_k exp(beta * conv[k]))
    # = (1/beta) * log(exp(beta*m) * sum_k exp(beta*(conv[k]-m)))
    # = m + (1/beta) * log(sum_k exp(beta*(conv[k]-m)))
    # = m + (1/beta) * log(sum_k exp(shifted[k]))
    # But we divided by sum to get weights, so sum(exp(shifted)) = 1/weights.sum() * ... 
    # Actually: weights = exp(shifted) / sum(exp(shifted))
    # So sum(exp(shifted)) = sum(exp(shifted)) -- we need to keep track
    # Let me just recompute:
    log_sum = m * beta + np.log(np.sum(np.exp(shifted)))
    smooth_val = log_sum / beta

    # Gradient: grad[i] = 2 * sum_j a[j] * weights[i+j]
    # = 2 * (a_reversed * weights)[n-1 + i] for i=0,...,n-1
    a_rev = a[::-1]
    size = 1

        size <<= 1
    A = np.fft.rfft(a_rev, size)
    W = np.fft.rfft(weights, size)
    conv_full = np.fft.irfft(A * W, size)
    grad = 2.0 * conv_full[n - 1:2 * n - 1]

    return grad, smooth_val

def solve_lp_minpeak(a, topK=1):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k (or top-K constraints relaxed),
    sum(g)=1, g>=0. Returns (g_normalized, M_opt).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix A where A[k,j] = a_norm[k-j] if 0<=k-j<n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def mirror_descent_ada(a, time_budget=300.0, beta_start=20.0, beta_end=500.0,
                       verbose=False):
    """
    AdaGrad mirror descent with Nesterov momentum on the smooth log-sum-exp objective.
    O(n log n) per iteration via FFT.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    # AdaGrad state
    G_sq = np.zeros(n)
    lr_base = 0.5
    momentum = 0.7
    v = np.zeros(n)  # Nesterov momentum

    it = 0
    stall = 0
    patience = 100

    while time.monotonic() - start < time_budget:
        # Adaptive beta
        frac = min(1.0, it / max(1, int(time_budget * 0.8)))
        beta = beta_start * (beta_end / beta_start) ** (frac ** 0.5)

        # Compute gradient
        grad, smooth_val = smooth_grad_fft(a, beta)

        # Nesterov: look-ahead step
        a_look = a - momentum * v
        a_look = np.maximum(a_look, 1e-15)
        a_look = a_look / a_look.sum()
        grad_look, _ = smooth_grad_fft(a_look, beta)

        # AdaGrad update
        G_sq += grad_look ** 2
        lr = lr_base / (np.sqrt(G_sq) + 1e-8)

        v = momentum * v + lr * grad_look

        # Mirror descent step (exponential update)
        log_a = np.log(a)
        log_a_new = log_a - v
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = best_a.copy()

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Restart with perturbation
                rng = np.random.default_rng(it * 31 + n)
                scale = 0.02 + 0.03 * (stall // patience)
                a = best_a * np.exp(rng.normal(0, min(0.15, scale), n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                v = np.zeros(n)
                G_sq = np.zeros(n)
                momentum = 0.5
                stall = 0
                patience = max(50, patience - 10)
            else:
                lr_base = max(0.05, lr_base * 0.995)

        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD-Ada it{it}: obj={best_obj:.8f}, beta={beta:.1f}, "
                  f"stall={stall}, t={elapsed:.1f}s")

    return best_a

def newton_polish(a, time_budget=60.0):
    """
    Active-set Newton refinement. Uses the special Hessian structure
    of the max-conv objective to take second-order steps.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    for it in range(50):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = conv.max()
        # Active set: peaks within 1% of max
        active = np.where(conv >= 0.995 * max_val)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])

        # Smooth approximation of active set
        # Use soft weights
        active_vals = conv[active]
        w = (active_vals - active_vals.min() + 0.01)
        w = w ** 2  # Sharper weighting
        w /= w.sum()

        # Gradient: weighted sum of subgradients
        grad = np.zeros(n)
        for wi, k in zip(w, active):
            k = int(k)
            m_start = max(0, k - (n - 1))
            m_end = min(n - 1, k)
            if m_start <= m_end:
                j_start = k - m_end
                j_end = k - m_start
                grad[m_start:m_end + 1] += 2.0 * wi * a[j_start:j_end + 1][::-1]

        # Approximate Hessian: sum of active Hessian matrices
        # H[k][i,j] = 2 if i+j=k
        # Build and solve (H + lambda*I) d = -grad
        # H is (2n-1)-banded in a specific way, but let's just build it for small n
        if n <= 2000:
            H = np.zeros((n, n))
            for wi, k in zip(w, active):
                k = int(k)
                for i in range(max(0, k - n + 1), min(n, k + 1)):
                    j = k - i
                    if 0 <= j < n:
                        H[i, j] += 2.0 * wi

            # Regularize for positive definiteness
            lam = 0.1
            try:
                d = np.linalg.solve(H + lam * np.eye(n), -grad)
            except np.linalg.LinAlgError:
                d = -grad / np.linalg.norm(grad)

            # Project gradient step onto feasible directions (sum(d) = 0 for simplex tangent)
            d -= np.mean(d)

            # Line search
            improved = False
            for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
                a_try = a + alpha * d
                a_try = np.maximum(a_try, 1e-15)
                a_try = a_try / a_try.sum()
                obj_try = compute_obj_normalized(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break

            if not improved:
                break
        else:
            # For large n, fall back to gradient step
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-10:
                break
            d = -grad / grad_norm
            d -= np.mean(d)
            for alpha in [0.01, 0.005, 0.001]:
                a_try = a + alpha * d
                a_try = np.maximum(a_try, 1e-15)
                a_try = a_try / a_try.sum()
                obj_try = compute_obj_normalized(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""

        a = a / a.sum()
    elif mode == 2:
        # Two-bump
        x = np.arange(n) / n
        a = np.exp(-20 * (x - 0.3)**2) + np.exp(-20 * (x - 0.7)**2) + 0.01
        a = a / a.sum()
    elif mode == 3:
        # Triangle
        x = np.linspace(-1, 1, n)
        a = np.maximum(0.1, 1.0 - np.abs(x))
        a = a / a.sum()
    elif mode == 4:
        # Semicircle
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(0.01, 1 - x**2))
        a = a / a.sum()
    elif mode == 5:
        # Frequency modulation
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.01) / np.maximum(a, 0.01).sum()
    elif mode == 6:
        # W-shaped
        x = np.linspace(0, 1, n)
        a = x**2 * (1-x)**2 + 0.05
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n))
    return a


def coordinate_refine(a, time_budget=30.0):
    """
    Final coordinate descent: for each coordinate, try optimal value
    given others fixed. Uses the structure that changing a_i only affects
    conv entries k in [i, i+n-1].
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    for rnd in range(10):
        if time.monotonic() - start > time_budget:
            break
        improved = False

        for i in range(n):
            if time.monotonic() - start > time_budget:
                break

            # Try changing a_i
            # The affected conv entries are k in [i, i+n-1]
            # conv[k] = sum_j a[j]*a[k-j], the terms involving a[i]:
            #   a[i]*a[k-i] for k-i < n, plus a[k-i]*a[i] (same term counted twice if i!=k-i)
            # So conv[k] has contribution: 2*a[i]*a[k-i] (if i != k-i) or a[i]^2 (if i = k-i)

            old_val = a[i]
            # Try a few candidate values
            for new_val in [0.0, old_val * 0.5, old_val * 1.5, old_val + 0.001, old_val - 0.001]:
                if new_val < 0:
                    continue
                a[i] = new_val
                s = a.sum()
                if s < 1e-10:
                    a[i] = old_val
                    continue
                a_norm = a / s
                conv = compute_conv(a_norm)
                obj = 2.0 * n * conv.max()

                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] = old_val
                    break

        if not improved:
            break

    return best_a

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 120.0)

        # Quick MD then LP
        half = alloc * 0.4
        if half > 10:
            a_try = mirror_descent_ada(a_try, time_budget=half, beta_start=30, beta_end=200)

        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.08, 60.0)
        if lp_alloc > 15:
            a_try = lp_with_gradient(a_try, time_budget=lp_alloc)

        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # --- Phase 5: Diverse Restarts ---
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_use = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        mode = r % 8
        a_r = diverse_init(n_use, rng, mode=mode)
        # Blend with best (varying amount)
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * best_seq + blend * a_r
        a_r = np.maximum(a_r, 1e-10)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.12, 100.0)
        if restart_budget > 40:
            a_r = mirror_descent_ada(a_r, time_budget=restart_budget * 0.5,
                                     beta_start=20, beta_end=300)
            remaining = deadline - time.monotonic()
            lp_b = min(remaining * 0.06, 50.0)
            if lp_b > 15:
                a_r = lp_with_gradient(a_r, time_budget=lp_b)

        obj_r = compute_obj_normalized(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # --- Phase 6: Final Intensive Polish ---
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # MD polish
    if remaining > 50:
        a6 = mirror_descent_ada(best_seq, time_budget=min(remaining * 0.25, 100.0),
                                beta_start=50, beta_end=1000)
        obj6 = compute_obj_normalized(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # LP polish
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a6 = lp_with_gradient(best_seq, time_budget=min(remaining * 0.3, 120.0))
        obj6 = compute_obj_normalized(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Newton polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a6 = newton_polish(best_seq, time_budget=min(remaining * 0.2, 40.0))
        obj6 = compute_obj_normalized(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Final Newton: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a6 = coordinate_refine(best_seq, time_budget=min(remaining * 0.4, 30.0))
        obj6 = compute_obj_normalized(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # --- Final LP one more time ---
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a6 = lp_with_gradient(best_seq, time_budget=min(remaining * 0.5, 60.0))
        obj6 = compute_obj_normalized(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    # Scale to ensure sum >= 0.01 (should already be ~1) and values in [0, 1000]
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
