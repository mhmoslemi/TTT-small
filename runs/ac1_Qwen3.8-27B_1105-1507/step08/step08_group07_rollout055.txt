```python
def subgradient_step(a, lr):
    n = len(a)
    C = compute_conv(a)
    S = np.sum(a)
    max_C = np.max(C)
    
    # Active set: indices where C is close to max
    threshold = max_C - 0.01 * max_C  # 1% below max
    A = np.where(C >= threshold)[0]
    
    # Compute subgradient of max(C): (2/|A|) * sum_{k in A} a_{k-i}
    subgrad = np.zeros(n)
    for k in A:
        # a_{k-i} for i = 0..n-1
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        subgrad[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    subgrad *= 2.0 / len(A)
    
    # Gradient of f = 2n * max_C / S^2 w.r.t. a
    # df/da_i = (2n/S^3) * [S * subgrad_i - 2 * max_C]
    grad = (2.0 * n / S**3) * (S * subgrad - 2.0 * max_C)
    
    # Step
    a_new = a - lr * grad
    
    # Project to a >= 0, sum(a) = S (maintain scale)
    a_new = np.maximum(a_new, 0)
    if np.sum(a_new) > 0:
        a_new = a_new * S / np.sum(a_new)
    
    return a_new, A
```

```python
        hi = min(n - 1, k)
        grad_a[lo:hi+1] += weights[k] * 2.0 * a[k-hi:k-lo+1][::-1]
```

```python
def spectral_flattening(a, n_iterations=100, alpha=0.1):
    """Iteratively flatten the power spectrum."""
    n = len(a)
    size = 1
    while size < 4 * n:
        size <<= 1
    a = a.copy()
    for _ in range(n_iterations):
        A = rfft(a, size)
        power = np.abs(A) ** 2
        # Flatten: reduce peaks, boost valleys
        mean_power = np.mean(power[power > 0])
        # Multiplicative update
        factor = np.ones(len(A))
        mask = power > 1e-10
        factor[mask] = (mean_power / power[mask]) ** alpha
        A_new = A * factor
        a_new = irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 0:
            a_new = a_new / s * np.sum(a)
        a = a_new
    return a
```

```python
1. L-BFGS with annealed smooth-max (log-sum-exp) approximation to the autocorrelation peak

3. Spectral flattening (iterative power spectrum equalization)

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_max_and_grad(a, beta):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Returns (smax, grad) where smax approximates max(conv(a,a))
    and grad = d(smax)/da.
    """
    n = len(a)
    C = compute_conv(a)
    max_C = float(np.max(C))
    if max_C <= 0:
        return 0.0, np.zeros(n)

    # Numerically stable softmax
    scaled = beta * C
    max_scaled = float(np.max(scaled))
    exp_terms = np.exp(scaled - max_scaled)
    sum_exp = float(np.sum(exp_terms))
    if sum_exp < 1e-300:
        weights = np.zeros(len(C))
        weights[int(np.argmax(C))] = 1.0
    else:
        weights = exp_terms / sum_exp

    smax = max_C + (1.0 / beta) * np.log(sum_exp) if sum_exp > 0 else max_C

    # Cross-correlation: cross_i = sum_k weights[k] * a[k-i] (with zero padding)
    # This is np.correlate(weights, a, mode='full')[:n]
    # But weights has length 2n-1, a has length n
    # np.correlate(weights, a, mode='full') has length (2n-1) + n - 1 = 3n-2
    # We want indices 0 to n-1
    size_corr = 1
    while size_corr < 3 * n:
        size_corr <<= 1
    W_freq = rfft(weights, size_corr)
    A_freq = rfft(a, size_corr)
    # Cross-correlation via FFT: correlate(x, y) = IFFT(FFT(x) * conj(FFT(y)))
    cross_full = irfft(W_freq * np.conj(A_freq), size_corr)
    # The correlation at lag i is sum_k W[k] * a[k-i] = cross_full[i] for i >= 0
    # Actually, correlate(W, a)[i] = sum_j W[j+i] * conj(a[j]) for the 'full' mode
    # We want: sum_k W[k] * a[k-i] = sum_j W[j+i] * a[j] = correlate(W, a)[i]
    cross = cross_full[:n]

    grad = 2.0 * cross
    return smax, grad


def objective_and_grad(a, beta):
    """
    Compute f(a) = 2n * smooth_max(conv(a,a)) / (sum(a))^2 and its gradient.
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf, np.zeros(n)

    smax, grad_smax = smooth_max_and_grad(a, beta)
    f = 2.0 * n * smax / (S * S)

    # df/da_i = 4n * (S * cross_i - smax) / S^3

    cross = grad_smax / 2.0
    grad_f = 4.0 * n * (S * cross - smax) / (S * S * S)

    return f, grad_f

            x = np.maximum(x, 0)
            f, g = objective_and_grad(x, beta)
            if np.isinf(f) or np.isnan(f):
                return np.zeros(n)
            return g

        # Run L-BFGS-B
        try:
            res = minimize(obj_func, a, jac=jac_func, method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': max_iter_per_stage,
                                    'ftol': 1e-12, 'gtol': 1e-8,
                                    'maxls': 50, 'maxfun': 10000})
            if res.x is not None:
                a_new = np.maximum(res.x, 0)
                S_new = float(np.sum(a_new))
                if S_new > 1e-10:
                    a_new = a_new / S_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new.copy()
                        if verbose:
                            elapsed = time.monotonic() - start
                            print(f"    L-BFGS stage {stage} (beta={beta:.0f}): "
                                  f"obj={best_obj:.8f}, iter={res.nit}, t={elapsed:.0f}s")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error at stage {stage}: {e}")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ z gives cross-conv a*z."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_given(a):
    """
    Solve LP: min M s.t. conv(a, z)_k <= M, sum(z)=1, z>=0.
    Returns (z_normalized, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    # Try cvxpy first
    try:
        z = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        prob = cp.Problem(cp.Minimize(M), [A @ z <= M, cp.sum(z) == 1])
        for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    z_val = np.maximum(z.value, 0)
                    zs = float(np.sum(z_val))
                    if zs > 1e-10:
                        z_val = z_val / zs
                        return z_val, float(M.value)
            except:
                continue
    except:
        pass

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n])
    except:
        pass

    return None, None


def lp_symmetric_alternating(a_init, z_init=None, time_budget=100.0, verbose=False):
    """
    Symmetric LP alternating:
    - Given z, solve LP for a: min M s.t. conv(a,z) <= M, sum(a)=1, a>=0
    - Given a, solve LP for z: min M s.t. conv(a,z) <= M, sum(z)=1, z>=0
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / np.sum(a)
    if z_init is None:
        z = a.copy()
    else:
        z = np.maximum(z_init, 0).copy()
        z = z / np.sum(z)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        # Solve for z given a
        z_new, M1 = solve_lp_given(a)
        if z_new is None:
            break

        # Solve for a given z_new
        a_new, M2 = solve_lp_given(z_new)
        if a_new is None:
            a_new = z_new

        # Check convergence
        a_old = a.copy()
        a = a_new.copy()
        z = z_new.copy()

        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Break with perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                z = a.copy()
                best_obj = compute_obj(a)
                best_a = a.copy()
                stall = 0

        if verbose:
            elapsed = time.monotonic() - start
            if int(elapsed * 2) % 10 == 0:
                print(f"    LP-sym: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def spectral_flatten(a_init, n_iterations=200, alpha=0.15, time_budget=60.0):
    """
    Iterative spectral flattening: modify the power spectrum to be flatter,
    which corresponds to a flatter autocorrelation.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 4 * n:
        size <<= 1

    for it in range(n_iterations):
        if time.monotonic() - start > time_budget:
            break

        A = rfft(a, size)
        power = np.abs(A) ** 2

        # Flatten: reduce peaks, boost troughs
        positive_mask = power > 1e-10
        if np.sum(positive_mask) > 0:
            mean_power = float(np.mean(power[positive_mask]))
            # Multiplicative spectral shaping
            factor = np.ones(len(A))
            factor[positive_mask] = (mean_power / power[positive_mask]) ** alpha
            # Apply
            A_new = A * factor
            a_new = irfft(A_new, size)[:n]
            a_new = np.maximum(a_new, 0)
            s = float(np.sum(a_new))
            if s > 1e-10:
                a_new = a_new / s
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()
                a = a_new

    return best_a


def subgradient_polish(a_init, time_budget=60.0, verbose=False):
    """
    Subgradient method for the true (non-smooth) max objective.
    Uses active set identification and adaptive step size.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.01
    stall = 0

    while time.monotonic() - start < time_budget:
        C = compute_conv(a)
        S = float(np.sum(a))
        max_C = float(np.max(C))
        obj = 2.0 * n * max_C / (S * S)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0

        # Active set: indices where C is within 0.5% of max
        threshold = max_C * 0.995
        active = np.where(C >= threshold)[0]

        # Subgradient: (2/|A|) * sum_{k in A} a_{k-i}
        subgrad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                subgrad[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
        subgrad *= 2.0 / len(active)

        # Full gradient of f = 2n*max_C/S^2
        # df/da_i = 4n*(S*cross_i - max_C) / S^3 where cross_i = subgrad_i/2
        grad = 4.0 * n * (S * subgrad / 2.0 - max_C) / (S * S * S)

        # Adaptive step size
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                a = a_new
                lr = min(lr * 1.1, 0.1)
            else:
                lr *= 0.7
                if lr < 1e-6:
                    # Perturb to escape
                    rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
                    a = best_a * np.exp(rng.normal(0, 0.02, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    lr = 0.01
                    stall = 0

        stall += 1
        if stall > 2000:
            break

    return best_a


def coordinate_descent_polish(a_init, time_budget=30.0):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, the objective as a function of a_i is:
    f(a_i) = 2n * max(C) / S^2
    where C depends quadratically on a_i.
    We can find the optimal a_i by grid search or ternary search.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    # Precompute convolution excluding contributions of each coordinate
    # C_k = a_k^2 + 2*sum_{j<k} a_j*a_{k-j} (for valid indices)
    # For coordinate i: C_k gets contribution a_i * a_{k-i} (and a_{k-i} * a_i)
    # So C_k(a_i) = C_k_base + (contribution from a_i)

    while time.monotonic() - start < time_budget:
        improved = False
        # Random order for diversity
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > time_budget:
                break

            a_save = a[i]
            if a_save < 1e-15:
                # Try making it positive
                a[i] = 0.01 * np.mean(a)
                obj_try = compute_obj(a)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = a_save
                continue

            # Try several values around current
            for frac in [0.5, 0.7, 1.0, 1.3, 1.5, 0.3, 2.0]:
                a_try_val = a_save * frac
                a[i] = a_try_val
                a_try = a.copy()
                a_try /= np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    a_save = a_try_val
                    break
            a[i] = a_save if not improved else a[i]

        if not improved:
            # Try zeroing some coordinates
            for i in order[:n // 4]:
                if time.monotonic() - start > time_budget:
                    break
                if a[i] < 0.05 * np.mean(a):
                    a_save = a[i]
                    a[i] = 0.0
                    a_try = a.copy()
                    a_try /= np.sum(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                    else:
                        a[i] = a_save

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def palindromic_optimize(a_init, time_budget=100.0):
    """
    Optimize a palindromic sequence (a_i = a_{n-1-i}).
    Reduce to n/2 free variables.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / np.sum(a)

    # Make palindromic
    a[:,] = 0  # placeholder, will redo
    a = np.maximum(a_init, 0).copy()
    a = (a + a[::-1]) / 2.0
    a = a / np.sum(a)

    half_n = n // 2
    best_a = a.copy()
    best_obj = compute_obj(a)

    # L-BFGS on half the variables
    def make_full(w_half):
        full = np.zeros(n)
        full[:half_n] = w_half
        full[half_n:] = w_half[::-1]
        if n % 2 == 1:
            full[half_n] = w_half[-1]
        return full

    def obj_half(w):
        a_full = np.maximum(make_full(w), 0)
        s = np.sum(a_full)
        if s < 1e-10:
            return 1e10
        a_full = a_full / s
        return compute_obj(a_full)

    # Start from current
    w_init = a[:half_n].copy()
    # Add small offset to avoid zeros for log optimization
    w_init = w_init + 1e-6

    def obj_log(w):
        w_half_pos = np.exp(w - np.max(w))
        return obj_half(w_half_pos)

    try:
        res = minimize(obj_log, w_init, method='L-BFGS-B',
                       options={'maxiter': 500, 'ftol': 1e-14, 'maxfun': 20000})
        w_opt = np.exp(res.x - np.max(res.x))
        a_opt = make_full(w_opt)
        a_opt = a_opt / np.sum(a_opt)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_obj:
            best_a = a_opt.copy()
            best_obj = obj_opt
    except:
        pass

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining L-BFGS, LP alternating, spectral flattening,
    subgradient polish, multi-n exploration, and random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        # Structured initializations
        for n_init in [400, 600, 800, 1000]:
            t = np.linspace(0, 1, n_init)
            windows = [
                0.5 + 0.5 * np.cos(2 * np.pi * t),
                np.sin(np.pi * t) ** 2,
                1.0 - 0.2 * np.cos(4 * np.pi * t),
            ]
            for w in windows:
                a_init = np.maximum(w, 0.01)
                a_init = a_init / np.sum(a_init)
                obj_init = compute_obj(a_init)
                if obj_init < best_score:
                    best_score = obj_init
                    best_seq = a_init.copy()

        if best_seq is None:
            best_seq = np.ones(800) / 800.0
            best_score = compute_obj(best_seq)
        print(f"Initial (n={len(best_seq)}): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: L-BFGS with annealed smooth max ===
    print(f"\n=== Phase 1: L-BFGS (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(350.0, remaining * 0.3)
    if remaining > 30:
        a_new = lbbfgs_optimize(best_seq, time_budget=alloc,
                                beta_start=15.0, beta_end=3000.0,
                                n_beta_stages=10, max_iter_per_stage=300, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 1 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 done: obj={obj:.8f}")

    # === Phase 2: Spectral Flattening ===
    print(f"\n=== Phase 2: Spectral Flatten (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(60.0, remaining * 0.08)
        a_new = spectral_flatten(best_seq, n_iterations=500, alpha=0.1, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 2 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 done: obj={obj:.8f}")

    # === Phase 3: Symmetric LP Alternating ===
    print(f"\n=== Phase 3: LP Symmetric (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(150.0, remaining * 0.15)
        a_new = lp_symmetric_alternating(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 3 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 done: obj={obj:.8f}")

    # === Phase 4: L-BFGS again (refined start) ===
    print(f"\n=== Phase 4: L-BFGS Refine (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(200.0, remaining * 0.2)
        a_new = lbbfgs_optimize(best_seq, time_budget=alloc,
                                beta_start=50.0, beta_end=5000.0,
                                n_beta_stages=8, max_iter_per_stage=400, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 4 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4 done: obj={obj:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_candidates = sorted(set([
        n_parent - 100, n_parent - 50, n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200,
        n_parent + 400, n_parent + 600,
        400, 500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600,
    ]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 3000]
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(40.0, remaining * 0.12)

        # Quick L-BFGS
        a_opt = lbbfgs_optimize(a_try, time_budget=alloc,
                                beta_start=30.0, beta_end=2000.0,
                                n_beta_stages=5, max_iter_per_stage=200, verbose=False)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Palindromic Exploration ===
    print(f"\n=== Phase 6: Palindromic (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(60.0, remaining * 0.08)
        a_new = palindromic_optimize(best_seq, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Palindromic: obj={best_score:.8f} **")
        else:
            print(f"  Palindromic: obj={obj:.8f}")

    # === Phase 7: Subgradient + Coordinate Polish ===
    print(f"\n=== Phase 7: Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc_sub = min(60.0, remaining * 0.15)
        a_new = subgradient_polish(best_seq, time_budget=alloc_sub)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Subgradient: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc_cd = min(40.0, remaining * 0.15)
            a_new = coordinate_descent_polish(best_seq, time_budget=alloc_cd)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Coord descent: obj={best_score:.8f} **")

    # === Phase 8: Random Restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 8: Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(8, int(remaining // 40))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            if r % 4 == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.03 + 0.005 * r, n_best))
            elif r % 4 == 1:
                a_r = best_seq.copy()
                n_blocks = max(5, n_best // 80)
                block_size = n_best // n_blocks
                blocks = rng.choice(n_blocks, min(4, n_blocks), replace=False)
                for b in blocks:
                    si = int(b) * block_size
                    ei = si + block_size
                    a_r[si:ei] *= rng.uniform(0.4, 1.6)
            elif r % 4 == 2:
                # Sparse perturbation
                a_r = best_seq.copy()
                n_sparse = max(10, n_best // 20)
                idx = rng.choice(n_best, n_sparse, replace=False)
                a_r[idx] *= rng.uniform(0.2, 3.0, n_sparse)
            else:
                # Frequency-domain perturbation
                size = 1
                while size < 2 * n_best:
                    size <<= 1
                A = rfft(a_r, size)
                phase_pert = rng.normal(0, 0.1, len(A))
                A *= np.exp(1j * phase_pert)
                a_r = irfft(A, size)[:n_best]

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr

            alloc = min(remaining * 0.3, 30.0)
            a_opt = lbbfgs_optimize(a_r, time_budget=alloc,
                                    beta_start=20.0, beta_end=2000.0,
                                    n_beta_stages=5, max_iter_per_stage=200, verbose=False)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 9: Final L-BFGS + LP polish on absolute best ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 9: Final Polish ===")
        alloc = min(remaining * 0.4, 40.0)
        a_final = lbbfgs_optimize(best_seq, time_budget=alloc,
                                  beta_start=100.0, beta_end=10000.0,
                                  n_beta_stages=5, max_iter_per_stage=500, verbose=False)
        obj = compute_obj(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
            print(f"  Final L-BFGS: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.4, 20.0)
            a_final = subgradient_polish(best_seq, time_budget=alloc)
            obj = compute_obj(a_final)
            if obj < best_score:
                best_score = obj
                best_seq = a_final.copy()
                print(f"  Final subgrad: obj={best_score:.8f} **")

    # Ensure valid output
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
