```python
def anderson_accelerate(history_a, history_g, m=5):
    """
    Anderson acceleration of order m.
    history_a: list of last m iterates a_t
    history_g: list of last m "updates" (best_response(a_t) - a_t)
    Returns extrapolated point.
    """
    m = min(m, len(history_a))
    if m == 0:
        return history_a[-1]
    if m == 1:
        return history_g[-1]
    
    # Build the Anderson system
    # g_i = best_response(a_i) - a_i (the "defect")
    # We want to find gamma such that a_new = a_{t} - sum_i beta_i * g_i
    # minimizes ||sum_i beta_i * g_i||^2
    # with sum_i beta_i = 1 (for m > 1) or no constraint for m = 1
    
    G = np.array([history_g[-1 - i] for i in range(m)])  # m x n matrix
    # Solve: min ||G^T beta||^2 s.t. sum beta = 1
    # This gives beta = (I - 1 1^T / m^2) ... 
    # Actually, the standard Anderson acceleration solves:
    # min ||sum_i beta_i * g_i||^2 s.t. sum beta_i = 1
    # Solution: beta = solve(1 + G G^T, 1)
    
    GGT = G @ G.T  # m x m
    ones = np.ones(m)
    # Solve (GGT + I) beta = ones (the +I is for regularization / m=1 case)
    # Actually for m > 1, we solve (GGT) beta = ones with the constraint sum beta = 1
    # The standard formulation: min beta^T GGT beta s.t. 1^T beta = 1
    # KKT: [2*GGT  1; 1^T  0] [beta; lambda] = [0; 1]
    # Or equivalently: GGT beta = lambda * 1, and 1^T beta = 1
    # So beta = lambda * 1, and lambda * m = 1, so lambda = 1/m, beta = 1/m
    # That's just uniform averaging, which is not right.
    
    # Let me use the correct formulation:
    # Anderson acceleration of order m:
    # a_{t+1} = a_t - sum_{i=0}^{m-1} beta_i * (G a_{t-i} - a_{t-i})
    # where beta minimizes ||sum beta_i * g_{t-i}||^2 s.t. sum beta_i = 1
    # 
    # The solution is: beta = argmin ||G^T beta||^2 s.t. 1^T beta = 1
    # Using Lagrange: 2 G G^T beta + lambda * 1 = 0, 1^T beta = 1
    # So G G^T beta = -lambda/2 * 1
    # beta = -lambda/2 * (GGT)^{-1} 1
    # 1^T beta = -lambda/2 * 1^T (GGT)^{-1} 1 = 1
    # lambda = -2 / (1^T (GGT)^{-1} 1)
    # beta = 1 / (1^T (GGT)^{-1} 1) * (GGT)^{-1} 1
    # 
    # Equivalently: solve (GGT + I) beta = 1 (this is a common implementation trick)
    
    try:
        A = GGT + np.eye(m)
        beta = np.linalg.solve(A, ones)
    except:
        beta = np.ones(m) / m
    
    a_new = history_a[-1] - G.T @ beta
    return a_new
```

            lr = min(lr * 1.05, 0.5)

import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_for_a(a, active_rows=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k (or active k only), sum(g)=1, g>=0.
    Returns (g_normalized, M_opt, dual_values) or (None, None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    if active_rows is not None:
        A_used = A[active_rows, :]
    else:
        A_used = A

    m = A_used.shape[0]

    # Try CLARABEL first (fastest for this type)
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A_used @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    dual = None
                    try:
                        dual = np.array(constraints[0].dual_value)
                        dual = np.maximum(dual, 0)
                    except:
                        pass
                    return g_val, float(M.value), dual
        except:
            continue

    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A_used
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                return g_val, float(res.x[n]), None
    except:
        pass

    return None, None, None


def solve_lp_full(a):
    """Solve the full LP (all constraints)."""
    return solve_lp_for_a(a, active_rows=None)


def anderson_extrapolate(history_a, history_defect, m=5):
    """
    Anderson acceleration of order m.
    history_a: list of recent iterates (most recent last)
    history_defect: list of recent defects g_t - a_t (most recent last)
    Returns extrapolated point.
    """
    m = min(m, len(history_a))
    if m == 0:
        return history_a[-1]
    if m == 1:
        return history_a[-1] - history_defect[-1]

    # Stack defects as rows: G is m x n
    G = np.array(history_defect[-m:])
    n_dim = G.shape[1]

    # Solve: min ||G^T beta||^2 s.t. sum(beta) = 1
    # KKT: G G^T beta = lambda * 1,  1^T beta = 1
    GGT = G @ G.T
    ones = np.ones(m)

    try:
        # Solve augmented system
        # [2*GGT  1] [beta]   [0]
        # [1^T    0] [lam ] = [1]
        KKT = np.zeros((m + 1, m + 1))
        KKT[:m, :m] = 2.0 * GGT + 1e-10 * np.eye(m)
        KKT[:m, m] = 1.0
        KKT[m, :m] = 1.0
        rhs = np.zeros(m + 1)
        rhs[m] = 1.0
        sol = np.linalg.solve(KKT, rhs)
        beta = sol[:m]
    except:
        beta = ones / m

    # New point
    a_new = history_a[-1] - G.T @ beta
    return a_new

def subgradient_peak_shave(a, time_budget=50.0):
    """
    Subgradient-based peak shaving: iteratively reduce the maximum convolution value.
    Uses active set of near-maximal lags for smoother gradients.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.3
    patience = 0

    while time.monotonic() - start < time_budget:
        c = compute_conv(a)
        max_c = np.max(c)
        # Active set: lags where c[k] > 0.95 * max_c
        active_threshold = 0.95 * max_c
        active_lags = np.where(c >= active_threshold)[0]

        # Average gradient over active lags
        grad = np.zeros(n)
        for k in active_lags:
            # grad from lag k: grad[i] = 2*a[k-i] if 0 <= k-i < n
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                grad[i] += 2.0 * a[k - i]
        grad /= len(active_lags)

        # Normalize gradient scale
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm * max_c

        # Update
        a_try = a - lr * grad
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            lr *= 0.5
            continue
        a_try = a_try / st

        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            patience = 0
            lr = min(lr * 1.02, 0.8)
        else:
            patience += 1
            lr *= 0.95
            if patience > 50:
                # Try smaller active set
                active_lags = np.where(c >= 0.99 * max_c)[0]
                if len(active_lags) == 0:
                    active_lags = [np.argmax(c)]
                grad = np.zeros(n)
                for k in active_lags:
                    lo = max(0, k - n + 1)
                    hi = min(n, k + 1)
                    for i in range(lo, hi):
                        grad[i] += 2.0 * a[k - i]
                grad /= len(active_lags)
                grad_norm = np.linalg.norm(grad)
                if grad_norm > 1e-15:
                    a_try = a - 0.1 * grad / grad_norm * max_c
                    a_try = np.maximum(a_try, 0)
                    st = float(np.sum(a_try))
                    if st > 1e-10:
                        a_try = a_try / st
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            patience = 0

            if patience > 100:
                break

    return best_a


def coordinate_descent_polish(a, time_budget=30.0):
    """
    Coordinate descent: optimize one coordinate at a time.
    For each i, find optimal a[i] given others fixed.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    # Precompute: for coordinate i, the contribution to conv[k] is a[i]*a[k-i]
    # If we change a[i] to a[i]+delta, conv[k] changes by delta*a[k-i] for each k
    # So the new max conv is max_k (old_conv[k] + delta*a[k-i])
    # We want to minimize max_k (old_conv[k] + delta*a[k-i]) subject to sum = 1, a >= 0

    # This is a 1D minimization for each coordinate
    # Bisection or golden section search

    improved_any = True
    while improved_any and time.monotonic() - start < time_budget:
        improved_any = False
        order = rng.permutation(n)

        for idx in order:
            if time.monotonic() - start > time_budget:
                break
            i = order[idx]

            # Current conv
            # If we set a[i] = 0, compute conv of (a without i)
            # Then adding a[i] = x changes conv[k] by x * a[k-i]
            a_no_i = a.copy()
            ai_val = a_no_i[i]
            a_no_i[i] = 0.0
            s_no_i = float(np.sum(a_no_i))
            if s_no_i < 1e-15:
                continue

            # conv_base = convolve(a_no_i, a_no_i)
            # But we need to be careful about normalization
            # Let's work with unnormalized, then normalize at the end

            conv_base = compute_conv(a_no_i)
            # Contribution vector: for each k, adding x at position i adds x*a_no_i[k-i]
            contrib = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                j = k - i
                if 0 <= j < n:
                    contrib[k] = a_no_i[j]

            # Total conv with a[i] = x: conv_base + x * contrib (approximately, since
            # conv(a) = conv(a_no_i) + a[i]*a_shifted + a_shifted_rev*a[i] ... )
            # Actually it's more complex. Let me just do a direct search.

            # Try a few values of a[i]
            current_x = ai_val
            sum_others = s_no_i
            best_local_obj = np.inf
            best_local_x = current_x

            # Candidate values: current, 0, and interpolated
            candidates = [0.0, current_x * 0.5, current_x * 0.7, current_x * 1.3,
                          current_x * 1.5, current_x * 0.3, current_x * 2.0]
            # Add values that make sum = 1
            # If a[i] = x, total sum = x + sum_others, normalized a[i] = x/(x+sum_others)
            # Objective with normalized: 2n * max(conv_normalized)
            # conv_normalized = conv / (sum)^2

            for x in candidates:
                if x < 0:
                    continue
                total_sum = x + sum_others
                if total_sum < 1e-15:
                    continue
                # Build full vector
                a_trial = a_no_i.copy()
                a_trial[i] = x
                a_trial = a_trial / total_sum
                obj_trial = compute_obj(a_trial)
                if obj_trial < best_local_obj:
                    best_local_obj = obj_trial
                    best_local_x = x

            if best_local_x != current_x and best_local_obj < best_obj - 1e-14:
                a[i] = best_local_x
                a = a / np.sum(a)
                best_obj = compute_obj(a)
                best_a = a.copy()
                improved_any = True

    return best_a


def pairwise_exchange(a, time_budget=40.0):
    """
    Pairwise exchange polish: try transferring mass between pairs.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for _ in range(30):
            if time.monotonic() - start > time_budget:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue

            for frac in [0.03, 0.07, 0.15, 0.02, 0.05]:
                delta = frac * min(a[i], a[j])
                if delta <= 0 or delta > 0.4 * min(a[i], a[j]):
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < -1e-15 or a_try[j] < -1e-15:

                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))

                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()

                    improved = True
                    break

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_structured_init(n, rng, style=0):

    elif style == 6:
        # Triangle inward (high at edges, low in middle)
        a = 1.0 - np.abs(2 * t - 1)
    elif style == 7:
        # Power law
        a = (t * (1 - t)) ** 0.5
    elif style == 8:

    else:
        a = np.ones(n)

    a = np.maximum(a, 0.001)
    a = a / np.sum(a)
    return a

            n_candidates.add(n_try)
    # Fixed values
    for n_fix in [400, 500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600, 1800, 2000, 2500]:
        n_candidates.add(n_fix)
    n_candidates = sorted([n for n in n_candidates if 300 <= n <= 3000])
    # Sort by proximity to current best
    n_candidates.sort(key=lambda x: abs(x - n_best))

    n_tried = 0
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == n_best and n_tried > 0:
            continue

        # Resample or generate new
        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick LP optimization
        alloc = min(40.0, remaining * 0.1)
        if alloc < 10:
            alloc = 10
        a_opt = lp_alternating_enhanced(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        # Quick subgradient
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20 and obj_try < best_score + 0.01:
            a_opt2 = subgradient_peak_shave(a_opt, time_budget=min(15.0, remaining2 * 0.3))
            obj_try2 = compute_obj(a_opt2)
            if obj_try2 < obj_try:
                a_opt = a_opt2
                obj_try = obj_try2

        n_tried += 1
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            if n_tried % 3 == 0:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Structured initializations at best n (100s) ===
    print(f"\n=== Phase 4: Structured Inits at n={n_best} ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for style in range(10):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_init = generate_structured_init(n_best, rng, style)
            # Small perturbation
            a_init = a_init * np.exp(rng.normal(0, 0.05, n_best))
            a_init = np.maximum(a_init, 0)
            a_init = a_init / np.sum(a_init)

            alloc = min(20.0, remaining * 0.15)
            a_opt = lp_alternating_enhanced(a_init, time_budget=alloc)
            obj_try = compute_obj(a_opt)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                print(f"  Style {style} (n={n_best}): obj={best_score:.8f} **")

    # === Phase 5: Final LP-Anderson on best (150s) ===
    print(f"\n=== Phase 5: Final LP (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(150.0, remaining * 0.3)
        a_new = lp_alternating_enhanced(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj:.8f}")

    # === Phase 6: Subgradient + Coordinate polish (60s) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc_sg = min(30.0, remaining * 0.3)
        a_new = subgradient_peak_shave(best_seq, time_budget=alloc_sg)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  SG polish: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc_cd = min(20.0, remaining * 0.3)
            a_new = coordinate_descent_polish(best_seq, time_budget=alloc_cd)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  CD polish: obj={best_score:.8f} **")

    # === Phase 7: Pairwise exchange (30s) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_pw = min(30.0, remaining * 0.3)
        a_new = pairwise_exchange(best_seq, time_budget=alloc_pw)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  PW polish: obj={best_score:.8f} **")

    # === Phase 8: Diverse restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 8: Restarts (remaining={remaining:.0f}s) ===")
        n_cur = len(best_seq)
        max_restarts = min(8, int(remaining // 40))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            if r % 4 == 0:
                # Gaussian perturbation
                a_r = best_seq * np.exp(rng.normal(0, 0.04 + 0.015 * r, n_cur))
            elif r % 4 == 1:
                # Block modulation
                a_r = best_seq.copy()
                n_blocks = max(5, n_cur // 80)
                bs = n_cur // n_blocks
                blocks = rng.choice(n_blocks, min(4, n_blocks), replace=False)
                for b in blocks:
                    si = int(b) * bs
                    ei = si + bs
                    a_r[si:ei] *= rng.uniform(0.5, 1.6)
            elif r % 4 == 2:
                # Sparse perturbation
                a_r = best_seq.copy()
                n_sparse = max(10, n_cur // 20)
                idx = rng.choice(n_cur, n_sparse, replace=False)
                a_r[idx] *= rng.uniform(0.3, 2.5, n_sparse)
            else:
                # Different n (resample best to new n, then perturb)
                n_new = n_cur + int(rng.integers(-200, 200))
                n_new = max(300, min(2500, n_new))
                a_r = resample_sequence(best_seq, n_new)
                a_r = a_r * np.exp(rng.normal(0, 0.03, n_new))

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr
            else:
                continue

            alloc_r = min(remaining * 0.3, 30.0)
            a_opt = lp_alternating_enhanced(a_r, time_budget=alloc_r)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 9: Final micro-polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_final = subgradient_peak_shave(best_seq, time_budget=min(10.0, remaining * 0.5))
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
