subgradient polishing, structured initialization, and systematic n exploration.

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def cross_conv(a, g):
    """Compute linear cross-convolution a*g (not a*a)."""
    n = len(a)
    m = len(g)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:n + m - 1]

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = cross_conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        r = min(n, 2*n - 1 - j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp(a, g_init=None):
    """Solve LP: min M s.t. cross_conv(a, g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_opt, M_opt)."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2*n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_polish(a, time_budget=60.0, K=5):
    """Subgradient method using top-K peaks."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 1.5 / np.sqrt(n)
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break
        
        # Weights based on peak values
        weights = np.exp(peak_vals / max_val - 1.0)
        weights /= weights.sum()
        
        # Subgradient
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                k = int(k)
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                g[i_start:i_end+1] += w * a[k - i_end : k - i_start + 1][::-1]
        
        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + (t_k / t_new) * (a_new - a_prev)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                t_k = 1.0
                stall = 0
        
        a_prev = a_new.copy()
        t_k = t_new
    
    return best_a

def resample_to_n(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def compute_conv(a):

    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

def cross_conv(a, g):
    """Compute linear cross-convolution a*g via FFT."""
    n = len(a)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
