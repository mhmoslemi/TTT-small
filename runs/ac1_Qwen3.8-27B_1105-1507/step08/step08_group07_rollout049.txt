    where Q is built from weighted autocorrelation."""

    conv = compute_conv(a)
    max_conv = max(conv)
    # Weights emphasizing peaks
    w = np.exp(beta * (conv / max_conv - 1.0))  # w[k] is small for small conv, ~1 for max conv

    # Actually, (a*a)[k] = sum_i a_i a_{i+k} for k >= 0
    # sum_k w[k] (a*a)[k] = sum_k w[k] sum_i a_i a_{i+k}

    # (using the positive lags of the weighted convolution)
    
    # Actually, let me just build it directly for simplicity (n=800 -> 800x800 is fine)
    Q = np.zeros((n, n))
    for k in range(n):

        for i in range(n - k):
            Q[i, i+k] += wk / 2
            Q[i+k, i] += wk / 2
    Q += w[0] / 2 * np.eye(n)  # k=0 term
    
    # Solve QP
    a_var = cp.Variable(n, nonneg=True)
    obj = cp.quad_form(a_var, cp.psd_wrap(Q))
    constraints = [cp.sum(a_var) == 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if a_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.maximum(a_var.value, 0)
        except:
            continue
    return None

def soft_subgradient_step(a, n_steps=200, eta0=0.05, decay=0.995, top_fraction=0.1):
    """
    Soft subgradient/penalty method to reduce peak of a*a.
    Uses weighted subgradient where weights = (conv[k] - threshold) for exceeding lags.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for step in range(n_steps):
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Adaptive threshold: top fraction of lags
        n_lags = len(conv)
        n_top = max(1, int(n_top_fraction * n_lags))
        threshold = np.sort(conv)[::-1][n_top - 1] if n_top < n_lags else 0.0
        
        # Soft weights: how much each lag exceeds threshold
        weights = np.maximum(conv - threshold, 0)
        total_w = float(np.sum(weights))
        if total_w < 1e-15:
            break
        weights = weights / total_w  # normalize to sum to 1
        
        # Compute weighted subgradient: g_j = 2 * sum_k weights[k] * a_{k-j}
        # Vectorized: T[j,m] = weights[j+m], g = 2 * T @ a
        j_idx = np.arange(n)
        T = weights[j_idx[:, None] + j_idx[None, :]]  # n x n matrix, T[j,m] = weights[j+m]
        g = 2.0 * (T @ a)
        
        # Step size with decay
        eta = eta0 * (decay ** step)
        
        # Update and project
        a_try = a - eta * g
        a_try = project_simplex(a_try, 1.0)
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
        a = a_try
    
    return best_a, best_obj

    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()  # Nesterov momentum
    beta = 0.9
    stall = 0
    prev_a = a.copy()
    prev_obj = best_obj

    while time.monotonic() - start < time_budget:
        result = solve_lp(y)
        if result[0] is None:
            break
        g_lp, M_lp = result

        # Try multiple mixing ratios
        best_cand_obj = np.inf
        best_cand = a.copy()
        for mix in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - mix) * a + mix * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_cand_obj:
                best_cand_obj = obj_try
                best_cand = a_try.copy()

        if best_cand_obj < best_obj - 1e-12:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Perturb to escape
                rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = project_simplex(a, 1.0)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                y = a.copy()
                beta = 0.5
            else:
                a = a + 0.1 * (g_lp - a)
                a = project_simplex(a, 1.0)

        # Nesterov update
        t_k = 2.0 / (1.0 + np.sqrt(1.0 + 4.0 * beta * beta))
        y = a + t_k * (a - prev_a)
        y = np.maximum(y, 0)
        ys = float(np.sum(y))
        if ys > 1e-10:
            y = y / ys
        prev_a = a.copy()
        prev_obj = compute_obj(prev_a)
        beta = min(beta + 0.01, 0.95)

        if verbose and int((time.monotonic() - start) * 2) % 20 == 0:
            print(f"    LP-acc: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s, stall={stall}")

    return best_a, best_obj


def symmetric_lp_alternating(a, time_budget=80.0):
    """LP alternating restricted to symmetric sequences."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    # Make symmetric
    if n % 2 == 0:
        n_half = n // 2
        a_sym = 0.5 * (a[:n_half] + a[n_half-1::-1])
        a_sym = np.maximum(a_sym, 0)
        a = np.concatenate([a_sym, a_sym[::-1]])
    else:
        n_half = (n - 1) // 2
        a_sym = 0.5 * (a[:n_half] + a[n-2-n_half::-1])
        a_mid = a[n_half]
        a_sym = np.maximum(a_sym, 0)
        a = np.concatenate([a_sym, [a_mid], a_sym[::-1]])
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        if n % 2 == 0:
            result = solve_lp_symmetric(a, n // 2)
        else:
            result = solve_lp_symmetric(a, (n - 1) // 2 + 1)
        
        if result[0] is None:
            # Fallback to regular LP
            result = solve_lp(a)
            if result[0] is None:
                break
            g_lp, _ = result
            # Symmetrize
            if n % 2 == 0:
                g_lp = 0.5 * (g_lp[:n//2] + g_lp[n//2-1::-1])
                g_lp = np.concatenate([g_lp, g_lp[::-1]])
            else:
                nh = (n-1)//2
                g_lp = 0.5 * (g_lp[:nh] + g_lp[n-2-nh::-1])
                g_lp = np.concatenate([g_lp, [g_lp[nh]], g_lp[::-1]])
            g_lp = np.maximum(g_lp, 0)
            g_lp = g_lp / np.sum(g_lp)
        else:
            g_lp, _ = result
        
        a_try = 0.5 * a + 0.5 * g_lp
        a_try = np.maximum(a_try, 0)
        a_try = a_try / np.sum(a_try)
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_polish(a, time_budget=30.0):
    """Coordinate descent polish: optimize one coordinate at a time."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    improved = True
    rounds = 0
    while improved and time.monotonic() - start < time_budget and rounds < 10:
        improved = False
        rounds += 1
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            # Try different values for a[i]
            # The optimal a[i] given others balances the convolution peaks
            for frac in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                a_try = a.copy()
                delta = (frac - 1.0) * a[i]
                a_try[i] += delta
                if a_try[i] < 0:
                    continue
                s = float(np.sum(a_try))
                if s < 1e-10:
                    continue
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

    return best_a, best_obj


def diverse_initialization(n, seed=42):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    t = np.linspace(0, 1, n)
    
    candidates = []
    # Raised cosine
    candidates.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t), 0.01))
    # Sine squared
    candidates.append(np.maximum(np.sin(np.pi * t) ** 2, 0.01))
    # Gaussian-like
    candidates.append(np.exp(-4 * (t - 0.5) ** 2))
    # Bump with modulation
    candidates.append(np.maximum(np.sin(np.pi * t) * (1 + 0.2 * np.cos(6 * np.pi * t)), 0.01))
    # Flat with slightly raised edges
    candidates.append(1.0 + 0.1 * np.cos(2 * np.pi * t))
    # Triangle
    candidates.append(np.maximum(1 - 2 * np.abs(t - 0.5), 0.01))
    # Random smooth
    freqs = rng.choice([1, 2, 3, 4], size=3)
    amps = rng.uniform(0.1, 0.3, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    w = 1.0
    for f, a_, p in zip(freqs, amps, phases):
        w += a_ * np.cos(2 * np.pi * f * t + p)
    candidates.append(np.maximum(w, 0.01))
    # Fourier-based: sum of first few cosine modes
    w2 = np.ones(n)
    for k in range(1, 4):
        w2 += 0.15 / k * np.cos(2 * np.pi * k * t)
    candidates.append(np.maximum(w2, 0.01))

    best = None
    best_obj = np.inf
    for c in candidates:
        c = c / np.sum(c)
        obj = compute_obj(c)
        if obj < best_obj:
            best_obj = obj
            best = c.copy()
    return best, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating with acceleration,
    soft subgradient escape, symmetric exploration, multi-n search, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # If no parent, try diverse initializations
    if best_seq is None:
        print("Generating diverse initializations...")
        for n_init in [600, 800, 1000, 1200]:
            a_init, obj_init = diverse_initialization(n_init, seed=seed + n_init)
            if obj_init < best_score:
                best_score = obj_init
                best_seq = a_init.copy()
                print(f"  Init n={n_init}: obj={obj_init:.8f}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"\nStarting optimization: n={n_parent}, obj={best_score:.8f}")

    # === Phase 1: LP Alternating with Acceleration ===
    print(f"\n=== Phase 1: LP Accelerated (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(350.0, remaining * 0.3)
    if alloc > 30:
        a_new, obj_new = lp_alternating_accelerated(best_seq, time_budget=alloc, verbose=True)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: no improvement (obj={obj_new:.8f})")

    # === Phase 2: Soft Subgradient Escape ===
    print(f"\n=== Phase 2: Soft Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(150.0, remaining * 0.15)
        # Try different parameters
        for top_frac, eta0, n_steps in [(0.05, 0.03, 100), (0.1, 0.05, 150), (0.2, 0.04, 200)]:
            if time.monotonic() > deadline:
                break
            a_sub, obj_sub = soft_subgradient_step(
                best_seq, n_steps=n_steps, eta0=eta0, decay=0.997, top_fraction=top_frac)
            if obj_sub < best_score:
                best_score = obj_sub
                best_seq = a_sub.copy()
                print(f"  Subgrad (frac={top_frac}, eta={eta0}): obj={best_score:.8f} **")
        
        # Follow up with LP if subgradient helped
        remaining = deadline - time.monotonic()
        if remaining > 60:
            alloc = min(100.0, remaining * 0.15)
            a_new, obj_new = lp_alternating_accelerated(best_seq, time_budget=alloc)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Post-subgrad LP: obj={best_score:.8f} **")

    # === Phase 3: Symmetric LP ===
    print(f"\n=== Phase 3: Symmetric LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(80.0, remaining * 0.08)
        a_sym, obj_sym = symmetric_lp_alternating(best_seq, time_budget=alloc)
        if obj_sym < best_score:
            best_score = obj_sym
            best_seq = a_sym.copy()
            print(f"  Symmetric: obj={best_score:.8f} **")
        else:
            print(f"  Symmetric: no improvement (obj={obj_sym:.8f})")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_candidates = sorted(set([
        n_parent - 100, n_parent - 50, n_parent - 200,
        n_parent + 50, n_parent + 100, n_parent + 200, n_parent + 400,
        500, 600, 700, 900, 1000, 1100, 1200, 1400,
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 2000]
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(40.0, remaining * 0.12)
        a_opt, obj_opt = lp_alternating_accelerated(a_try, time_budget=alloc)
        
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 5: LP + Subgradient alternation ===
    print(f"\n=== Phase 5: Hybrid LP+Subgrad ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        # Alternate between LP and subgradient
        a_curr = best_seq.copy()
        for round_num in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            
            # LP step
            alloc_lp = min(40.0, remaining * 0.2)
            a_curr, obj_lp = lp_alternating_accelerated(a_curr, time_budget=alloc_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_curr.copy()
            
            # Subgradient step
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_curr, obj_sg = soft_subgradient_step(
                a_curr, n_steps=80, eta0=0.03, decay=0.995, top_fraction=0.08)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_curr.copy()
                print(f"  Hybrid round {round_num}: obj={best_score:.8f} **")

    # === Phase 6: Coordinate Polish ===
    print(f"\n=== Phase 6: Coordinate Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(40.0, remaining * 0.15)
        a_pol, obj_pol = coordinate_polish(best_seq, time_budget=alloc)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Polish: obj={best_score:.8f} **")

    # === Phase 7: Diverse Restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Diverse Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(5, int(remaining // 45))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            if r == 0:
                # Small gaussian perturbation
                a_r = best_seq * np.exp(rng.normal(0, 0.03, n_best))
            elif r == 1:
                # Medium perturbation
                a_r = best_seq * np.exp(rng.normal(0, 0.08, n_best))
            elif r == 2:
                # Reshape: multiply by a smooth random function
                t = np.linspace(0, 1, n_best)
                mod = 1.0 + 0.3 * np.cos(2 * np.pi * rng.uniform(1, 5) * t + rng.uniform(0, 2*np.pi))
                a_r = best_seq * mod
            elif r == 3:
                # New structured init at same n
                a_r, _ = diverse_initialization(n_best, seed=seed + 100 + r)
            else:
                # New structured init at different n
                n_alt = min(2000, n_best + rng.integers(-200, 200))
                a_r, _ = diverse_initialization(max(300, n_alt), seed=seed + 200 + r)
                if len(a_r) != len(best_seq):
                    n_best = len(a_r)

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr < 1e-10:
                continue
            a_r = a_r / sr

            alloc = min(remaining * 0.3, 35.0)
            a_opt, obj_r = lp_alternating_accelerated(a_r, time_budget=alloc)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                n_best = len(best_seq)
                print(f"  Restart {r} (n={n_best}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 8: Final LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 30.0)
        a_final, obj_final = lp_alternating_accelerated(best_seq, time_budget=alloc)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
