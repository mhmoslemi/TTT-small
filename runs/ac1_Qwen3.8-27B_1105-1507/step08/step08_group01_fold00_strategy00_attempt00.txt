def simplex_project(v):
    # Standard projection onto the simplex {x>=0, sum(x)=1}
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / float(rho + 1)
    return np.maximum(v - theta, 0)
