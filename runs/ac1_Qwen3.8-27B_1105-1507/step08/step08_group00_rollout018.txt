import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv_fft(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_reduced_lp(a, K=50, solver=None):
    """
    Solve reduced LP: min M s.t. top-K convolution constraints, sum(g)=1, g>=0.
    Returns (g, M).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv = compute_conv_fft(a_norm)
    
    # Get top-K indices of the current convolution
    top_idx = np.argsort(conv)[::-1][:K]
    top_idx = np.sort(top_idx)
    
    # Build constraint matrix for selected constraints
    # (A@g)[k] = sum_j g[j] * a_norm[k-j] for valid j
    A_sel = np.zeros((K, n))
    for ii, k in enumerate(top_idx):
        k = int(k)
        for j in range(n):
            if 0 <= k - j <= 2*n - 2:  # always true for k in [0, 2n-2]
                pass
            # a_norm[k-j] is valid when 0 <= k-j < n
            jj = k - j
            if 0 <= jj < n:
                A_sel[ii, j] = a_norm[jj]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_sel @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver is None:
        solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    else:
        solvers = [solver]
    
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def solve_full_lp(a, solver=None):
    """Solve full LP with all convolution constraints."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build full convolution matrix
    # (A@g)[k] = sum_j g[j] * a_norm[k-j]
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # For lag k, the contribution of g[j] is a_norm[k-j]
        # k-j must be in [0, n-1], so k in [j, j+n-1]
        for k in range(j, min(j+n, 2*n-1)):
            A[k, j] = a_norm[k-j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver is None:
        solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    else:
        solvers = [solver]

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def fw_line_search(a, w, max_iters=50):
    """
    Find optimal step size t for a_new = (1-t)*a + t*w
    by minimizing max_k conv(a_new, a_new)[k] subject to sum=1.
    Uses ternary search on t in [0, 1].
    """
    # The function f(t) = max_k conv((1-t)*a+t*w, (1-t)*a+t*w)[k]
    # is convex in t (max of quadratics with PSD matrices)
    # Use ternary search
    
    a_n = a / np.sum(a)
    w_n = w / np.sum(w)
    
    def f(t):
        x = (1-t)*a_n + t*w_n
        conv = compute_conv_fft(x)
        return float(np.max(conv))
    
    lo, hi = 0.0, 1.0
    for _ in range(max_iters):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if f(m1) <= f(m2):
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-6:
            break
    
    t = (lo + hi) / 2
    return t

        if s_new < 1e-10:
            t = 0.1

            a_new = np.maximum(a_new, 0)

def compute_conv_fft(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

    n = len(a)

        return np.inf

    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_norm, selected_lags=None):
    """
    Build matrix A (len(selected_lags) x n) such that A @ g gives selected lags of conv(a_norm, g).
    If selected_lags is None, use all lags.
    """
    n = len(a_norm)
    if selected_lags is None:
        selected_lags = np.arange(2 * n - 1)

    A = np.zeros((num_lags, n))
    for ii, k in enumerate(selected_lags):
        k = int(k)
        # (A@g)[k] = sum_j g[j] * a_norm[k-j] where 0 <= k-j < n

        if j_start <= j_end:
            A[ii, j_start:j_end+1] = a_norm[k-j_end:k-j_start+1][::-1]
    
    return A


def solve_lp(a, K=None, solver=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for selected k, sum(g)=1, g>=0.
    If K is given, only use top-K lags of current conv(a,a).

    conv = compute_conv_fft(a_norm)

        top_idx = np.argsort(conv)[::-1][:K]

    else:
        selected_lags = np.arange(2 * n - 1)

                s = float(np.sum(g_val))
                if s > 1e-10:

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""

        else:
            stall += 1
            if stall > 20:
                # Perturb to escape
                rng = np.random.default_rng(it * 7 + 3)
                scale = 0.01 + 0.005 * min(stall, 20)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if elapsed > time_budget * 0.92:
                    break
            elif stall > 10:
                # Try very small step
                t = 0.03
                a_try = (1 - t) * a + t * w
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    stall = 0

        # Periodic pruning
        if it > 0 and it % prune_every == 0:
            mask = a > prune_thresh
            if mask.sum() < n and mask.sum() > 10:
                a_pruned = np.zeros(n)
                a_pruned[mask] = a[mask]
                a_pruned = a_pruned / a_pruned.sum()
                obj_pruned = compute_obj(a_pruned)
                if obj_pruned < best_obj:
                    a = a_pruned
                    best_obj = obj_pruned
                    best_a = a.copy()

        it += 1
        if verbose and (it % 5 == 0):
            print(f"    FW it{it}: obj={best_obj:.8f}, K={K}, t={best_t:.3f}, elapsed={elapsed:.0f}s")

    return best_a


def smoothed_nesterov(a, time_budget=100.0, tau_init=5.0, tau_end=400.0, verbose=False):
    """
    Nesterov smoothed gradient descent on the log-sum-exp smoothed objective.
    Operates on the simplex (sum=1, a>=0).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    lr = 2.5 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    # FFT size for cross-correlation
    fft_size = 1
    while fft_size < 3 * n:
        fft_size <<= 1

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient of smoothed objective at y
        conv = compute_conv_fft(y)
        max_c = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_c))
        p = exp_vals / float(np.sum(exp_vals))

        # grad_i = 2 * sum_k p[k] * y[k-i] (cross-correlation)
        P = fft(p, fft_size)
        Y = fft(y, fft_size)
        g_full = ifft(P * np.conj(Y)).real
        g = 2.0 * g_full[:n]

        # Update
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        # Decay LR
        lr = max(1e-9, lr * 0.9997)

        # Reset if stuck
        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.5 / np.sqrt(n) * 0.5
            best_it = it

        if verbose and it % 5000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, float(a[0]) if n == 1 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
