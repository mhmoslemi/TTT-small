from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:

def line_search(a, direction, obj_current, max_steps=10):
    """
    Line search: find best t in (0, 1) such that obj((1-t)*a + t*direction) is minimized.
    Uses parabolic interpolation.
    """
    n = len(a)
    best_t = 0.0
    best_obj = obj_current
    
    # Initial candidates
    ts = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
    objs = []
    
    for t in ts:
        a_try = (1.0 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        objs.append((t, obj_try))
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
    
    if not objs:
        return best_t, best_obj
    
    # Parabolic interpolation on the best 3
    objs.sort(key=lambda x: x[1])
    if len(objs) >= 3:
        t1, f1 = objs[1]
        t2, f2 = objs[0]
        t3, f3 = objs[2]
        # Parabolic fit
        if t1 != t2 and t2 != t3 and t1 != t3:
            denom = (t1 - t2) * (t1 - t3) * (t2 - t3)
            if abs(denom) > 1e-15:
                t_par = (t1*t1*(f2-f3) + t2*t2*(f3-f1) + t3*t3*(f1-f2)) / (
                    2*(t1*(f2-f3) + t2*(f3-f1) + t3*(f1-f2)))
                if 0.01 < t_par < 0.99:
                    a_try = (1.0 - t_par) * a + t_par * direction
                    a_try = np.maximum(a_try, 0)
                    a_try = a_try / a_try.sum()
                    f_par = compute_obj(a_try)
                    if f_par < best_obj:
                        best_obj = f_par
                        best_t = t_par

def coordinate_descent(a, time_budget=100.0):
    """Coordinate descent: move mass from peak contributors to non-contributors."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    max_rounds = 20
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find top contributors to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)
        
        non_contrib = set(range(n))
        for _, i, j in contributors[:20]:
            non_contrib.discard(i)
            if i != j:
                non_contrib.discard(j)
        non_contrib_list = sorted(non_contrib, key=lambda x: a[x])[:10]
        
        if not non_contrib_list:
            break
        
        improved = False
        for contrib, i, j in contributors[:15]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break
            src = i if a[i] >= a[j] else j
            for target in non_contrib_list[:5]:
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    delta = frac * a[src]
                    if delta >= a[src] * 0.9:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                if improved:
                    break
        
        if not improved:
            break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        c1, c2 = n // 3, 2 * n // 3
        w = n // 10
        x = np.arange(n)
        a = np.exp(-((x - c1)**2) / (2*w**2)) + np.exp(-((x - c2)**2) / (2*w**2))
        a = a / a.sum()
    elif mode == 3:
        # Triangle
        x = np.linspace(-1, 1, n)
        a = np.maximum(1.0 - np.abs(x), 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Arcsine-like
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x**2)
        a = a / a.sum()
    elif mode == 5:
        # Multi-frequency
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3*np.sin(2*np.pi*3*x) + 0.2*np.sin(2*np.pi*7*x + 1)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Quartic
        x = np.linspace(0, 1, n)
        a = x**4 + (1-x)**4 + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

"""
Optimized step function search using LP-based direction finding with:
- Efficient convolution matrix construction
- Dual-guided subgradient for stalling recovery
- Adaptive line search with parabolic interpolation
- Population-based exploration
- Systematic dimension scanning
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    n = len(a)
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                g_val = g_val / s
            return g_val, float(res.x[n])
    except Exception:
        pass
    return None, None

    n = len(a)
    best_t = 0.0
    best_obj = obj_cur
    
    # Analytical estimate of optimal t

    results = []
    for t in ts:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:

        a_try = a_try / s_try
        obj_try = compute_obj(a_try)

            best_t = t
