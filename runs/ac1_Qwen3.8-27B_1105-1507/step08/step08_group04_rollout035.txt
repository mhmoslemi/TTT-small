import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time
import math


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def logsumexp(obj, tau):
    """Compute (1/tau)*log(sum(exp(tau*x))) as smoothed max."""
    x = tau * obj
    x_max = np.max(x)
    if x_max > 700:
        x = x - x_max
    return x_max / tau + np.log(np.sum(np.exp(x))) / tau


def compute_smoothed_obj(a, tau):
    """Compute smoothed objective with log-sum-exp."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf, None
    a_norm = a / s
    conv = compute_conv(a_norm)
    smoothed = logsumexp(conv, tau)
    return 2.0 * n * smoothed, conv


def smoothed_gradient(a, tau, top_k=50):
    """
    Compute gradient of 2*n*logsumexp(conv(a,a), tau) with sum(a)=1 constraint.
    Returns centered gradient on the simplex.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.zeros(n)
    a_norm = a / s
    conv = compute_conv(a_norm)

    # weights = exp(tau * conv) / sum(exp(tau * conv))
    tau_conv = tau * conv
    tau_conv_max = np.max(tau_conv)
    weights = np.exp(tau_conv - tau_conv_max)
    weights = weights / np.sum(weights)

    # grad[j] = 2n * sum_k weights[k] * (a_norm[k-j] + a_norm[j-k] valid terms)
    # Actually: d conv[k]/d a[j] = a[k-j] + a[j-k] (with valid index checks)
    # For self-convolution: d conv[k]/d a[j] = a[k-j] if 0<=k-j<n, plus a[j-k] if 0<=j-k<n
    # But since it's self-conv, d conv[k]/d a[j] = a[k-j] * (1 if 0<=k-j<n else 0)
    #   + a[j-k] * (1 if 0<=j-k<n else 0)
    # Wait, conv[k] = sum_j a[j]*a[k-j]. d/d a[j'] = a[k-j'] + a[j'-k] if both valid.
    # Actually for self: d conv[k]/d a[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    # But a[k-j] and a[j-k] are the same value when both valid (just different index mapping).
    # More carefully: conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
    # d conv[k]/d a[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    # The first term exists when j<=k and k-j<n, i.e., max(0,k-n+1)<=j<=min(k,n-1)
    # The second term exists when j>k and j-k<n, i.e., k<j<=k+n-1 and j<n

    # Use FFT for efficiency
    n_fft = 1
    while n_fft < 4 * n:
        n_fft <<= 1
    A_fft = rfft(a_norm, n_fft)
    W_fft = rfft(weights, n_fft)

    # R[l] = sum_m a[m] * w[m+l] (circular), computed as irfft(conj(A)*W)
    R = irfft(np.conj(A_fft) * W_fft, n_fft)

    # grad[j] = sum_k w[k] * (a[k-j] + a[j-k]) for valid indices
    # First part: sum_k w[k]*a[k-j] = R[(n_fft-j) % n_fft] (circular correlation)
    # Second part: sum_k w[k]*a[j-k] = sum_k w[k]*a[(-k+j)] = R[(n_fft+j) % n_fft]

    grad = np.zeros(n)
    for j in range(n):
        # First term: sum_k w[k] * a[k-j] where k-j in [0,n-1]
        # = sum_{m=0}^{n-1} a[m] * w[m+j] where m+j in [0, 2n-2]
        idx1 = (n_fft - j) % n_fft
        # Second term: sum_k w[k] * a[j-k] where j-k in [0,n-1]
        # = sum_{m=0}^{n-1} a[m] * w[j-m] where j-m in [0, 2n-2]
        idx2 = (n_fft + j) % n_fft
        grad[j] = R[idx1] + R[idx2]

    grad *= 2.0 * n
    # Center for simplex (remove component along all-ones direction)
    grad -= np.mean(grad)
    return grad


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_start = j
        k_end = min(m - 1, j + n - 1)
        length = k_end - k_start + 1
        A[k_start:k_end + 1, j] = a[:length]
    return A


def solve_lp(a, max_time=30.0):
    """
    Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_value, dual_y) or (None, None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    m = 2 * n - 1

    t_start = time.monotonic()

    # Use scipy linprog with HiGHS (most reliable for this size)
    # Variables: g[0..n-1], M
    # Minimize: M
    # Subject to: A@g - M*1 <= 0 (m constraints)
    #             sum(g) = 1 (1 constraint)
    #             g >= 0, M free (but M >= 0 implied)

    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # A@g - M <= 0
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]  # M >= 0

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if time.monotonic() - t_start > max_time:
            return None, None, None
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_norm = g_val / gs
                M_val = float(res.x[n])
                # Extract dual variables for the inequality constraints
                y_dual = None
                if res.ineqlin and len(res.ineqlin) == m:
                    y_dual = np.maximum(res.ineqlin, 0)
                    y_sum = np.sum(y_dual)
                    if y_sum > 1e-10:
                        y_dual = y_dual / y_sum
                return g_norm, M_val, y_dual
    except Exception:
        pass

    return None, None, None


def line_search_blend(a, g, max_iter=20):
    """
    Find best blending parameter t in [0,1] for a_new = (1-t)*a + t*g.
    Uses a fine grid followed by golden section search.
    """
    best_t = 0.3
    best_obj = compute_obj(a)  # t=0

    # Coarse grid
    t_candidates = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]
    for t in t_candidates:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    # Fine search around best
    if 0 < best_t < 1:
        lo = max(0, best_t - 0.1)
        hi = min(1, best_t + 0.1)
        for _ in range(8):
            m1 = lo + (hi - lo) * 0.382
            m2 = lo + (hi - lo) * 0.618
            a1 = np.maximum((1 - m1) * a + m1 * g, 0)
            a2 = np.maximum((1 - m2) * a + m2 * g, 0)
            s1, s2 = float(np.sum(a1)), float(np.sum(a2))
            if s1 < 1e-10 or s2 < 1e-10:
                break
            o1 = compute_obj(a1 / s1)
            o2 = compute_obj(a2 / s2)
            if o1 < o2:
                hi = m2
            else:
                lo = m1
            if o1 < best_obj:
                best_obj = o1
                best_t = m1
            if o2 < best_obj:
                best_obj = o2
                best_t = m2

    return best_t, best_obj


def lp_alternate(a, time_budget=200.0, verbose=False):
    """LP alternating with line search and dual-based perturbation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp, y_dual = solve_lp(a)
        if g_lp is None:
            break

        # Line search for best blend
        best_t, best_local_obj = line_search_blend(a, g_lp)

        if best_local_obj < best_obj - 1e-12:
            a_new = (1.0 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_local_obj
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.3f}, time={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 5:
                # Dual-based perturbation: use dual weights to guide perturbation
                if y_dual is not None:
                    # Compute "pressure" on each position
                    n_fft = 1
                    while n_fft < 4 * n:
                        n_fft <<= 1
                    A_fft = rfft(a, n_fft)
                    Y_fft = rfft(y_dual, n_fft)
                    pressure = irfft(np.conj(A_fft) * Y_fft, n_fft)
                    pressure = pressure[:n] + pressure[n_fft - n:][::-1] if n_fft > n else pressure[:n]
                    # Actually just use the circular correlation
                    R = irfft(np.conj(A_fft) * Y_fft, n_fft)
                    pressure = np.zeros(n)
                    for j in range(n):
                        pressure[j] = R[(n_fft - j) % n_fft] + R[(n_fft + j) % n_fft]

                    # Perturb: push mass away from high-pressure positions
                    scale = min(0.03 + 0.01 * (stall // 5), 0.15)
                    a_pert = a * np.exp(-scale * pressure)
                    a_pert = np.maximum(a_pert, 0)
                    s_p = float(np.sum(a_pert))
                    if s_p > 1e-10:
                        a_pert = a_pert / s_p
                        obj_p = compute_obj(a_pert)
                        if obj_p < best_obj:
                            best_obj = obj_p
                            best_a = a_pert.copy()
                        a = a_pert
                else:
                    # Random perturbation
                    rng = np.random.default_rng(hashing_int(stall))
                    scale = min(0.03 + 0.01 * (stall // 5), 0.15)
                    a_pert = a * np.exp(rng.normal(0, scale, n))
                    a_pert = np.maximum(a_pert, 0)
                    s_p = float(np.sum(a_pert))
                    if s_p > 1e-10:
                        a_pert = a_pert / s_p
                        obj_p = compute_obj(a_pert)
                        if obj_p < best_obj:
                            best_obj = obj_p
                            best_a = a_pert.copy()
                        a = a_pert
                stall = 0

    return best_a


def hashing_int(x):
    """Simple hash to get reproducible seed from integer."""
    return (x * 2654435761 + 12345) % (2**31)


def smoothed_gradient_optimize(a, time_budget=100.0, tau_start=50.0, tau_end=500.0, verbose=False):
    """
    Optimize using log-sum-exp smoothed objective with adaptive tau.
    Uses momentum-based gradient descent on the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Start with moderate tau, increase over time (sharper -> closer to true max)
    n_steps = 0
    lr = 0.1 / n
    momentum = np.zeros(n)
    beta = 0.9
    x = a.copy()  # Nesterov lookahead

    tau = tau_start
    steps_per_tau = 200

    while time.monotonic() - start < time_budget:
        if n_steps % steps_per_tau == 0 and tau < tau_end:
            tau = min(tau * 1.5, tau_end)

        grad = smoothed_gradient(x, tau)

        # Nesterov momentum step
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Nesterov update
        t_k = 2.0 / (n_steps + 2.0)
        x = a_new + (1 - t_k) * (a_new - a)
        x = np.maximum(x, 1e-15)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        a = a_new
        n_steps += 1

        # Periodically check true objective
        if n_steps % 50 == 0:
            obj = compute_obj(a)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                if verbose:
                    print(f"    SmoothGrad: step={n_steps}, tau={tau:.0f}, obj={best_obj:.8f}")

            # Adaptive learning rate
            lr = max(0.01 / n, lr * 0.999)

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation with refinement."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def resample_seq_smooth(a, new_n):
    """Smoother resampling using cubic-like interpolation via FFT."""
    n = len(a)
    if n == new_n:
        return a.copy()

    # Zero-pad and interpolate in frequency domain
    n_fft = 1
    while n_fft < max(n, new_n) * 2:
        n_fft <<= 1

    fa = rfft(a, n_fft)
    # Low-pass filter: keep only first new_n/2 frequencies (Nyquist for new_n)
    keep = new_n // 2 + 1
    fa_filtered = np.zeros_like(fa)
    fa_filtered[:keep] = fa[:keep]
    a_interp = irfft(fa_filtered, n_fft)

    # Take every n_fft/new_n sample
    indices = np.linspace(0, n_fft - 1, new_n, dtype=int)
    a_new = a_interp[indices]
    return np.maximum(a_new, 0)


def diverse_initialization(n, rng, method=None):
    """Generate diverse structured initializations."""
    if method is None:
        method = rng.integers(0, 8)

    x = np.linspace(0, 1, n)

    if method == 0:
        # Smooth bump
        a = np.exp(-30 * (x - 0.5) ** 2)
    elif method == 1:
        # Multi-frequency oscillation
        freqs = rng.uniform(1, 5, 3)
        a = np.ones(n)
        for f in freqs:
            a *= 1.0 + 0.5 * np.cos(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
    elif method == 2:
        # Two bumps
        c1, c2 = rng.uniform(0.2, 0.35), rng.uniform(0.65, 0.8)
        w = rng.uniform(30, 60)
        a = np.exp(-w * (x - c1) ** 2) + np.exp(-w * (x - c2) ** 2)
    elif method == 3:
        # Random smooth (sum of Gaussians)
        a = np.zeros(n)
        for _ in range(rng.integers(2, 5)):
            center = rng.uniform(0.1, 0.9)
            width = rng.uniform(0.05, 0.3)
            amplitude = rng.uniform(0.5, 2.0)
            a += amplitude * np.exp(-((x - center) / width) ** 2)
    elif method == 4:
        # Power law
        a = x ** rng.uniform(0.5, 3.0)
        if rng.random() > 0.5:
            a = (1 - x) ** rng.uniform(0.5, 3.0)
    elif method == 5:
        # Periodic
        period = rng.integers(3, 15)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * x * period + rng.uniform(0, 2))
    elif method == 6:
        # Sigmoid blend
        t = rng.uniform(5, 20)
        a = 1.0 / (1.0 + np.exp(-t * (x - 0.5)))
    else:
        # Near-uniform with small random perturbation
        a = np.ones(n) * (1.0 + 0.1 * rng.standard_normal(n))

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def structured_perturbation(a, rng, strength=0.1):
    """Apply structured perturbation: smooth random field multiplication."""
    n = len(a)
    # Generate smooth random perturbation via random Fourier features
    n_freq = min(10, n // 4)
    freqs = rng.uniform(1, 5, n_freq)
    phases = rng.uniform(0, 2 * np.pi, n_freq)
    amps = rng.uniform(0.3, 1.0, n_freq)

    x = np.linspace(0, 1, n)
    pert = np.zeros(n)
    for f, p, amp in zip(freqs, phases, amps):
        pert += amp * np.cos(2 * np.pi * f * x + p)

    a_new = a * np.exp(strength * pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating, smoothed gradient,
    n-value exploration, and diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        n_parent = len(a)
        print(f"Parent loaded: n={n_parent}, obj={compute_obj(a):.8f}")
    except Exception:
        a = np.ones(1000) / 1000
        n_parent = 1000
        print(f"No parent, using uniform: n={n_parent}, obj={compute_obj(a):.8f}")

    best_a = a.copy()
    best_obj = compute_obj(a)

    # === Phase 1: LP Alternating from parent ===
    print(f"\n=== Phase 1: LP Alternating (n={len(best_a)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    a_p1 = lp_alternate(best_a, time_budget=p1_budget, verbose=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1
        print(f"  Phase 1 result: obj={best_obj:.8f} **")
    else:
        print(f"  Phase 1 result: obj={obj_p1:.8f}")

    # === Phase 2: Smoothed gradient optimization ===
    print(f"\n=== Phase 2: Smoothed Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        p2_budget = min(remaining * 0.15, 150.0)
        alloc1 = p2_budget * 0.6
        alloc2 = p2_budget * 0.4
        a_sg1 = smoothed_gradient_optimize(best_a, time_budget=alloc1,
                                            tau_start=30.0, tau_end=200.0)
        obj_sg1 = compute_obj(a_sg1)
        if obj_sg1 < best_obj:
            best_obj = obj_sg1
            best_a = a_sg1
            print(f"  SmoothGrad (tau 30-200): obj={best_obj:.8f} **")
        else:
            print(f"  SmoothGrad (tau 30-200): obj={obj_sg1:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_sg2 = smoothed_gradient_optimize(best_a, time_budget=alloc2,
                                                tau_start=100.0, tau_end=500.0)
            obj_sg2 = compute_obj(a_sg2)
            if obj_sg2 < best_obj:
                best_obj = obj_sg2
                best_a = a_sg2
                print(f"  SmoothGrad (tau 100-500): obj={best_obj:.8f} **")
            else:
                print(f"  SmoothGrad (tau 100-500): obj={obj_sg2:.8f}")

    # === Phase 3: LP Alternating Round 2 ===
    print(f"\n=== Phase 3: LP Alternating Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        p3_budget = min(remaining * 0.20, 200.0)
        a_p3 = lp_alternate(best_a, time_budget=p3_budget, verbose=True)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3
            print(f"  Phase 3 result: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 3 result: obj={obj_p3:.8f}")

    # === Phase 4: n-value exploration ===
    print(f"\n=== Phase 4: n-value Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    # Generate candidate n values around and beyond current best
    n_candidates = set()
    for delta in [-400, -300, -200, -100, 0, 100, 200, 300, 400, 600, 800, 1200]:
        n_try = n_best + delta
        if 300 <= n_try <= 5000:
            n_candidates.add(n_try)
    n_candidates = sorted(n_candidates)

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        if n_try == len(best_a):
            a_try = best_a.copy()
        else:
            # Try both resampling methods
            a_try = resample_seq(best_a, n_try)
            a_try = a_try / np.sum(a_try)

        # Quick LP optimization
        alloc = min(35.0, remaining * 0.2)
        a_opt = lp_alternate(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse initializations with LP ===
    print(f"\n=== Phase 5: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    # Try diverse inits at the best n and a few nearby n values
    n_for_diverse = [n_best, n_best + 200, n_best - 200, n_best + 400]
    n_for_diverse = [n for n in n_for_diverse if 300 <= n <= 4000]

    for n_div in n_for_diverse:
        for d in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 35:
                break

            method = (d * len(n_for_diverse) + n_div) % 8
            a_div = diverse_initialization(n_div, rng, method=method)

            # Quick LP
            alloc = min(30.0, remaining * 0.15)
            a_opt = lp_alternate(a_div, time_budget=alloc)
            obj_div = compute_obj(a_opt)

            if obj_div < best_obj:
                best_obj = obj_div
                best_a = a_opt.copy()
                print(f"  ** n={n_div}, method={method}: obj={best_obj:.8f} **")
            else:
                print(f"  n={n_div}, method={method}: obj={obj_div:.8f}")

    # === Phase 6: Structured perturbation + LP from best ===
    print(f"\n=== Phase 6: Perturbation + LP ===")
    remaining = deadline - time.monotonic()

    for pert_i in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        strength = 0.05 + 0.03 * pert_i
        a_pert = structured_perturbation(best_a, rng, strength=strength)

        # Quick LP from perturbed point
        alloc = min(25.0, remaining * 0.15)
        a_opt = lp_alternate(a_pert, time_budget=alloc)
        obj_pert = compute_obj(a_opt)

        if obj_pert < best_obj:
            best_obj = obj_pert
            best_a = a_opt.copy()
            print(f"  ** Perturb {pert_i} (str={strength:.2f}): obj={best_obj:.8f} **")
        else:
            print(f"  Perturb {pert_i} (str={strength:.2f}): obj={obj_pert:.8f}")

    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        # LP polish
        alloc_lp = min(remaining * 0.35, 80.0)
        a_final = lp_alternate(best_a, time_budget=alloc_lp, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final LP: obj={best_obj:.8f} **")

    # Smoothed gradient micro-polish
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc_sg = min(remaining * 0.25, 50.0)
        a_sg = smoothed_gradient_optimize(best_a, time_budget=alloc_sg,
                                           tau_start=200.0, tau_end=1000.0)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"  Final SmoothGrad: obj={best_obj:.8f} **")

    # One more LP if time allows
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_last = min(remaining * 0.5, 40.0)
        a_last = lp_alternate(best_a, time_budget=alloc_last)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Last LP: obj={best_obj:.8f} **")

    # === Phase 8: Try slightly different n from final best ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n_final = len(best_a)
        for dn in [-50, 50, -100, 100]:
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            n_try = n_final + dn
            if n_try < 200 or n_try > 5000:
                continue
            a_try = resample_seq(best_a, n_try)
            a_try = a_try / np.sum(a_try)
            alloc = min(20.0, remaining * 0.3)
            a_opt = lp_alternate(a_try, time_budget=alloc)
            obj_try = compute_obj(a_opt)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_opt.copy()
                print(f"  ** Final n={n_try}: obj={best_obj:.8f} **")
            else:
                print(f"  Final n={n_try}: obj={obj_try:.8f}")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    # Ensure all values are valid
    best_a = np.maximum(best_a, 0)
    best_a = np.minimum(best_a, 1000.0)
    s = float(np.sum(best_a))
    if s < 0.01:
        best_a = np.ones(100)
        s = 100.0

    result = [float(x) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_obj:
            best_obj = val
    else:
        # Fallback
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
