import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    m = 2 * n - 1
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack((A, -np.ones((m, 1))))
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
    except Exception:
        return None, None

    if res.status == 0:
        g = np.maximum(res.x[:n], 0.0)
        gs = float(np.sum(g))
        if gs > 1e-12:
            g = g / gs
            M_val = float(res.x[n])
            return g, M_val
    return None, None


def solve_lp_dual(a):
    """Solve LP and extract active set information. Returns (g, M, active_idx)."""
    n = len(a)
    g, M_val = solve_lp(a)
    if g is None:
        return None, None, None
    A = build_conv_matrix(a)
    conv_val = A @ g
    tol = 1e-7 * max(1.0, abs(M_val))
    active = np.where(conv_val >= M_val - tol)[0]
    return g, M_val, active

def multi_peak_subgradient(a, time_budget, n_peaks=5):
    """Subgradient descent targeting multiple peaks simultaneously."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.05
    momentum = np.zeros(n)
    beta = 0.7

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        threshold = max_val * 0.85
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > n_peaks:
            active = np.argsort(conv)[::-1][:n_peaks]

        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += a[k - indices]
        grad /= len(active)
        grad_centered = grad - np.mean(grad)

        momentum = beta * momentum - lr * grad_centered
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            a = best_a  # revert
            momentum *= 0.8

        if it % 500 == 499:
            lr = max(lr * 0.92, 0.003)

    return best_a

    rng = np.random.default_rng(42)

    for iteration in range(20000):
        if time.monotonic() - start > time_budget:
            break

        k_star = int(np.argmax(conv))
        improved = False

        contributions = np.zeros(n)

            indices = np.arange(i_min, i_max + 1)
            contributions[indices] = a[indices] * a[k_star - indices]

        sorted_idx = np.argsort(contributions)[::-1]
        top = sorted_idx[:min(30, n)]

        for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
            done = False
            for i in top:
                if done or time.monotonic() - start > time_budget:
                    break
                i = int(i)
                if a[i] < 1e-10:
                    continue
                # Try moving mass from i to a "far" position
                target = (k_star - i + n // 2) % n
                if target == i:
                    target = (i + 1) % n
                if a[target] > a[i]:
                    continue
                delta = frac * a[i]
                if delta <= 0:
                    continue
                a[i] -= delta
                a[target] += delta
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                    done = True
                else:
                    a[i] += delta
                    a[target] -= delta
            if done:
                break

        if not improved:
            # Random swap
            for _ in range(3):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.02 * min(a[i], a[j])
                if delta <= 0:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s < 1e-10:
                    a = best_a.copy()
                    break
                a = a / s
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()

        if not improved and iteration > 500:
            break

    return best_a


def generate_initialization(kind, n, rng):
    """Generate diverse initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'bilinear':
        a = np.linspace(0.5, 1.5, n)
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 10)
        a[:edge] = 0.3
        a[-edge:] = 0.3
        return a / np.sum(a)
    elif kind == 'two_tone':
        x = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * x / n)
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif kind == 'fourier_mix':
        x = np.arange(n)
        a = 1.0
        for freq in [1, 2, 3]:
            a += 0.2 * np.cos(2.0 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif kind == 'barker_like':
        # Approximate Barker sequence structure
        a = np.ones(n) * 0.8
        # Add structured peaks
        n_peaks = max(2, n // 20)
        positions = rng.choice(n, size=n_peaks, replace=False)
        for p in positions:
            a[int(p)] += 1.0
        # Smooth
        kernel = np.ones(3) / 3
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining LP alternating, GS spectral
    balancing, subgradient descent, coordinate polish, and length sweep.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 40.0
    rng = np.random.default_rng(seed)

    # Load parent sequence
    best_seq = None
    best_score = np.inf

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent_arr = parent_arr / s
                best_seq = parent_arr
                best_score = compute_obj(parent_arr)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_cur = len(best_seq)

    # === Phase 1: Diverse Initializations (80s) ===
    print(f"\n=== Phase 1: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    init_budget = min(80.0, remaining * 0.05)
    init_kinds = ['cosine', 'sine_taper', 'gaussian', 'random', 'plateau',
                  'bilinear', 'two_tone', 'fourier_mix', 'barker_like']

    for kind in init_kinds:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        a_init = generate_initialization(kind, n_cur, rng)
        obj_init = compute_obj(a_init)
        marker = " **" if obj_init < best_score else ""
        print(f"  {kind}: obj={obj_init:.8f}{marker}")
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()

    # Extra random inits
    for _ in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        a_init = generate_initialization('random', n_cur, rng)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Random: obj={obj_init:.8f} **")

    a = best_seq.copy()
    print(f"  Best after Phase 1: {best_score:.8f}")

    # === Phase 2: LP Alternating (main workhorse, ~800s) ===
    print(f"\n=== Phase 2: LP Alternating with Momentum ===")
    remaining = deadline - time.monotonic()
    lp_total = min(800.0, remaining * 0.55)

    momentum_schedules = [
        (0.3, 0.3), (0.5, 0.3), (0.7, 0.2), (0.5, 0.2),
        (0.9, 0.15), (0.5, 0.15)
    ]
    per_round = lp_total / len(momentum_schedules)

    for round_idx, (beta, extra) in enumerate(momentum_schedules):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_round, remaining * 0.7)
        if alloc < 10:
            break
        print(f"  LP Round {round_idx} (beta={beta}): ", end="")
        a_new = lp_alternate(a, time_budget=alloc, beta_momentum=beta, verbose=True)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"** obj={best_score:.8f}")
        else:
            print(f"obj={obj_new:.8f} (no improvement)")
            # Perturb for next round
            scale = 0.02 + 0.01 * round_idx
            a = best_seq * np.exp(rng.normal(0, scale, n_cur))
            a = np.maximum(a, 0)
            a = a / np.sum(a)

    # === Phase 3: GS Spectral Balancing (100s) ===
    print(f"\n=== Phase 3: GS Spectral Balancing ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        gs_budget = min(100.0, remaining * 0.1)
        # Try from current best and from a perturbed version
        for try_idx in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            if try_idx == 0:
                a_gs = best_seq.copy()
            else:
                a_gs = best_seq * np.exp(rng.normal(0, 0.05, n_cur))
                a_gs = np.maximum(a_gs, 0)
                a_gs = a_gs / np.sum(a_gs)
            a_gs = gs_spectral_balance(a_gs, time_budget=min(gs_budget / 3, remaining * 0.5))
            obj_gs = compute_obj(a_gs)
            if obj_gs < best_score:
                best_score = obj_gs
                best_seq = a_gs.copy()
                print(f"  GS try {try_idx}: obj={best_score:.8f} **")
            else:
                print(f"  GS try {try_idx}: obj={obj_gs:.8f}")

    # === Phase 4: Multi-Peak Subgradient (100s) ===
    print(f"\n=== Phase 4: Multi-Peak Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        sg_budget = min(100.0, remaining * 0.1)
        for n_peaks in [3, 5, 8]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_sg = multi_peak_subgradient(best_seq, time_budget=sg_budget / 3, n_peaks=n_peaks)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  Subgrad (K={n_peaks}): obj={best_score:.8f} **")
            else:
                print(f"  Subgrad (K={n_peaks}): obj={obj_sg:.8f}")

    # === Phase 5: LP + Polish combo on best (150s) ===
    print(f"\n=== Phase 5: LP + Polish Combo ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        combo_budget = min(150.0, remaining * 0.15)
        # LP then subgradient then coordinate
        a_c = lp_alternate(best_seq, time_budget=combo_budget * 0.5, beta_momentum=0.5)
        obj_c = compute_obj(a_c)
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_c.copy()
            print(f"  LP combo: obj={best_score:.8f} **")
        a_c = multi_peak_subgradient(a_c, time_budget=combo_budget * 0.25, n_peaks=5)
        obj_c = compute_obj(a_c)
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_c.copy()
            print(f"  SG combo: obj={best_score:.8f} **")
        a_c = coordinate_polish(a_c, time_budget=combo_budget * 0.25)
        obj_c = compute_obj(a_c)
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_c.copy()
            print(f"  CD combo: obj={best_score:.8f} **")

    # === Phase 6: Length Sweep (200s) ===
    print(f"\n=== Phase 6: Length Sweep ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_options = sorted(set([
        max(100, n_best - 200),
        max(100, n_best - 100),
        max(100, n_best - 50),
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        n_best + 400,
    ]))
    n_options = [nn for nn in n_options if 200 <= nn <= 4000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc_lp = min(25.0, remaining * 0.3)
        alloc_sg = min(15.0, remaining * 0.15)

        a_opt = lp_alternate(a_try, time_budget=alloc_lp, beta_momentum=0.5)
        a_opt = multi_peak_subgradient(a_opt, time_budget=alloc_sg, n_peaks=5)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            n_cur = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final intensive LP rounds (remaining) ===
    print(f"\n=== Phase 7: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    a_final = best_seq.copy()
    n_final = len(a_final)

    rounds_done = 0
    while remaining > 25 and rounds_done < 20:
        alloc = min(remaining * 0.3, 50.0)
        if alloc < 10:
            break
        beta = 0.5 if rounds_done % 2 == 0 else 0.7
        a_new = lp_alternate(a_final, time_budget=alloc, beta_momentum=beta)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_new.copy()
            a_final = a_new.copy()
            print(f"  Final LP {rounds_done}: obj={best_score:.8f} **")
        else:
            # Perturb
            scale = 0.01 + 0.005 * (rounds_done % 5)
            a_final = best_seq * np.exp(rng.normal(0, scale, n_final))
            a_final = np.maximum(a_final, 0)
            a_final = a_final / np.sum(a_final)
            print(f"  Final LP {rounds_done}: obj={obj:.8f} (perturbed)")
        rounds_done += 1
        remaining = deadline - time.monotonic()

    # === Phase 8: Very final coordinate polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd = coordinate_polish(best_seq, time_budget=min(remaining * 0.5, 40.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_cd:.8f}")

    # Final multi-peak subgradient
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_sg = multi_peak_subgradient(best_seq, time_budget=min(remaining * 0.6, 30.0), n_peaks=3)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    # Clip to valid range
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s_final = float(np.sum(best_seq))
    if s_final < 0.01:
        best_seq = np.ones(100)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
