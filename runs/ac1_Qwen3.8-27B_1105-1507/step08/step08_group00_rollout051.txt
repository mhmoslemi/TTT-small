def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)

    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

    a_norm = a / np.sum(a)

    M = cp.Variable()
    dual_var = cp.Variable(2 * n - 1, nonneg=True)
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]

    return None, None, None


def solve_lp_maximize_sum(a, M_bound):

def spectral_diversify(a, rng, strength=0.3):
    """
    Create a diversified version of a by adding structured noise in frequency domain.
    This targets specific frequency components that contribute to autocorrelation peaks.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    # Get FFT
    A = fft(a_norm, n)
    # Add phase noise to identify/modify specific frequencies
    phase_noise = rng.normal(0, strength, n)
    # Also add amplitude noise
    amp_noise = 1.0 + rng.normal(0, strength * 0.5, n)
    A_new = A * amp_noise * np.exp(1j * phase_noise)
    # Inverse FFT
    a_new = ifft(A_new).real
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a_norm.copy()
    return a_new / s


def subgradient_refine(a, time_budget=100.0, K=8, verbose=False):
    """Subgradient method with top-K peaks and momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 3.0 / np.sqrt(n)
    momentum = 0.75
    v = np.zeros(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 8.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.99

    return best_a

            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9997)

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 4.0 / np.sqrt(n) * 0.5
            best_it = it

        if verbose and it % 3000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Concentrated at center
        x = np.arange(n)
        a = np.exp(-((x - n // 2) ** 2) / (2 * (n // 4) ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Two peaks
        c1 = n // 3
        c2 = 2 * n // 3
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Lopsided (bias toward one end)
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(1, 4)
        a = a / a.sum()
    elif mode == 5:
        # Cosine window
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n)
        a = a + 1e-10
        a = a / a.sum()
    elif mode == 6:
        # Triangular
        a = np.abs(1 - 2 * np.arange(n) / n)
        a = a + 1e-10
        a = a / a.sum()
    elif mode == 7:
        # Random spectral
        z = rng.normal(0, 1, n)
        a = softmax(z)
    else:
        a = np.ones(n) / n
    return a


def kkt_refine(a, time_budget=50.0):
    """
    KKT-based refinement: check which constraints are active and use the
    KKT multipliers to guide optimization direction.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        # Active constraints: those close to the max
        active = np.where(conv >= max_conv * 0.999)[0]
        if len(active) == 0:
            break

        # Compute gradient direction: average subgradient of active constraints
        g = np.zeros(n)
        for k in active:
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += a[j_start:j_end + 1][::-1]
        g /= len(active)

        # Project gradient to tangent space of simplex (remove mean component)
        g_tangent = g - np.mean(g)

        # Take small step
        lr = 0.1 / np.sqrt(n)
        a_new = a - lr * g_tangent
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try opposite direction (the max might be at a "valley")
            a_new2 = a + lr * g_tangent
            a_new2 = project_simplex(a_new2)
            obj_new2 = compute_obj(a_new2)
            if obj_new2 < best_obj - 1e-15:
                best_obj = obj_new2
                best_a = a_new2.copy()
                a = a_new2
            else:
                lr *= 0.5
                if lr < 1e-10:
                    break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining L-BFGS, aggressive LP polishing, spectral diversification,
    multi-length exploration, and KKT refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Nesterov smoothed gradient descent ===
    print(f"\n=== Phase 1: Nesterov Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 250.0)
    if p1_budget > 30:
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                          tau_init=5.0, tau_end=600.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 NSD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 NSD: obj={obj_sgd:.8f}")

    # === Phase 2: L-BFGS with softmax parameterization ===
    print(f"\n=== Phase 2: L-BFGS (softmax) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 30:
        a_lbfgs = lbfgs_softmax(best_seq, time_budget=p2_budget,
                                tau_init=10.0, tau_end=600.0, verbose=True)
        obj_lbfgs = compute_obj(a_lbfgs)
        if obj_lbfgs < best_score:
            best_score = obj_lbfgs
            best_seq = a_lbfgs.copy()
            print(f"  Phase2 L-BFGS: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 L-BFGS: obj={obj_lbfgs:.8f}")

    # === Phase 3: Aggressive LP Polishing ===
    print(f"\n=== Phase 3: Aggressive LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.3, 450.0)
    if p3_budget > 30:
        for rnd in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(p3_budget / (6 - rnd), remaining * 0.5)
            if alloc < 15:
                break
            a_lp = lp_polish_aggressive(best_seq, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                if rnd < 5:
                    scale = 0.02 * (rnd + 1)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()

    # === Phase 4: Spectral diversification ===
    print(f"\n=== Phase 4: Spectral Diversification ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for d in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Create diverse candidate via spectral perturbation
        strength = 0.15 + 0.08 * d
        a_div = spectral_diversify(best_seq, rng, strength=strength)
        # Blend with best
        blend = 0.6 + 0.05 * d
        a_div = blend * best_seq + (1 - blend) * a_div
        a_div = np.maximum(a_div, 0)
        a_div = a_div / a_div.sum()

        # Quick optimize: NSD then LP
        alloc1 = min(remaining * 0.15, 60.0)
        if alloc1 > 15:
            a_div = nesterov_smoothed_descent(a_div, time_budget=alloc1,
                                              tau_init=20.0, tau_end=400.0)
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 60.0)
        if alloc2 > 15:
            a_div = lp_polish_aggressive(a_div, time_budget=alloc2)

        obj_div = compute_obj(a_div)
        if obj_div < best_score:
            best_score = obj_div
            best_seq = a_div.copy()
            print(f"  Spectral div {d} (s={strength:.2f}): obj={best_score:.8f} **")
        else:
            print(f"  Spectral div {d} (s={strength:.2f}): obj={obj_div:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 200), max(500, n_current - 100),
        n_current + 100, n_current + 200, n_current + 400,
        max(400, n_current // 2), min(2000, n_current * 2)
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_current) > 50]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick NSD
        alloc1 = min(remaining * 0.15, 50.0)
        if alloc1 > 10:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=300.0)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.1, 50.0)
        if alloc2 > 10:
            a_try = lp_polish_aggressive(a_try, time_budget=alloc2)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.15, 80.0)
        if alloc > 20:
            half = alloc / 2
            a_r = nesterov_smoothed_descent(a_r, time_budget=half,
                                            tau_init=10.0, tau_end=300.0)
            remaining = deadline - time.monotonic()
            lp_alloc = min(remaining * 0.1, 50.0)
            if lp_alloc > 10:
                a_r = lp_polish_aggressive(a_r, time_budget=lp_alloc)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: KKT refinement ===
    print(f"\n=== Phase 7: KKT Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.1, 60.0)
        a_kkt = kkt_refine(best_seq, time_budget=alloc)
        obj_kkt = compute_obj(a_kkt)
        if obj_kkt < best_score:
            best_score = obj_kkt
            best_seq = a_kkt.copy()
            print(f"  KKT: obj={best_score:.8f} **")
        else:
            print(f"  KKT: obj={obj_kkt:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau NSD
    if remaining > 30:
        alloc = min(remaining * 0.25, 100.0)
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                          tau_init=50.0, tau_end=1200.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final NSD: obj={best_score:.8f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 120.0)
        a_lp = lp_polish_aggressive(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Subgradient
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 50.0)
        a_sg = subgradient_refine(best_seq, time_budget=alloc, K=10)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # One more LP if time
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 60.0)
        a_lp = lp_polish_aggressive(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
