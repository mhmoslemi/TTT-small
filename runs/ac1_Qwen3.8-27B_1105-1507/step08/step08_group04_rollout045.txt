    n = len(v)

Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2.
Combines LP alternating, log-sum-exp FISTA polished with adaptive temperature,
tilting perturbations, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(n, a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 2, j + n - 1)
        A[lo:hi + 1, j] = a[:hi - lo + 1]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(n, a_norm)

    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.ones((1, n + 1))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g_val))
        if gs > 1e-10:
            return g_val / gs, float(res.x[n])
    return None, None


def smoothed_obj_grad(a, tau):
    """Compute log-sum-exp smoothed objective and gradient.
    f(a) = (1/tau) * log(sum_k exp(tau * conv(a,a)[k]))
    grad[m] = (2/tau) * sum_j a[j] * p[m+j] where p = softmax(tau*conv)
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = tau * (conv - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    p = exp_shifted / sum_exp

    f = max_c + np.log(sum_exp) / tau

    # grad[m] = (2/tau) * sum_{j=0}^{n-1} a[j] * p[m+j]
    # Using FFT cross-correlation:
    # Let a_flip[j] = a[n-1-j], then (a_flip * p)[m+n-1] = sum_i a_flip[i]*p[m+n-1-i]
    # = sum_i a[n-1-i]*p[m+n-1-i]. Let j=n-1-i: = sum_j a[j]*p[m+j]. Yes!
    N = 1
    while N < 3 * n:
        N <<= 1
    a_flip = a[::-1]
    Af = rfft(a_flip, N)
    Pf = rfft(p, N)
    conv_fp = irfft(Af * Pf, N)
    grad = (2.0 / tau) * conv_fp[n - 1:2 * n - 1]

    return f, grad


def fista_smoothed(a, time_budget=60.0, verbose=False):
    """FISTA on smoothed objective with adaptive temperature."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()
    y = a.copy()
    t = 1.0
    step = 1.0 / (4.0 * n)  # initial step

    # Temperature schedule: start smooth, get sharper
    tau_start = max(5.0, n / 10.0)
    tau_end = max(50.0, n)
    total_its = 3000

    for it in range(total_its):
        if time.monotonic() - start > time_budget:
            break

        # Adaptive temperature
        frac = it / total_its
        tau = tau_start * (tau_end / tau_start) ** frac

        f_val, grad = smoothed_obj_grad(y, tau)

        # Backtracking line search
        step_i = step
        x_new = project_simplex(y - step_i * grad)
        f_new, _ = smoothed_obj_grad(x_new, tau)
        backtracks = 0
        while f_new > f_val + 0.5 * step_i * np.dot(grad, x_new - y) and backtracks < 10:
            step_i *= 0.5
            x_new = project_simplex(y - step_i * grad)
            f_new, _ = smoothed_obj_grad(x_new, tau)
            backtracks += 1
        step = step_i * 1.5  # increase step for next iteration (with cap)
        step = min(step, 5.0 / n)

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + (t - 1.0) / t_new * (x_new - x)
        y = project_simplex(y)
        x = x_new
        t = t_new

        # Track best using true objective
        obj_true = compute_obj(x)
        if obj_true < best_obj - 1e-12:
            best_obj = obj_true
            best_a = x.copy()
            if verbose:
                print(f"    FISTA: it={it}, tau={tau:.1f}, obj={best_obj:.8f}")

    return best_a

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        a_fista = fista_smoothed(best_a, time_budget=p2_budget, verbose=True)
        obj_fista = compute_obj(a_fista)
        if obj_fista < best_obj:
            best_obj = obj_fista
            best_a = a_fista
            print(f"  Phase 2: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_fista:.8f}")

    # === Phase 3: Perturbation + LP ===
    print(f"\n=== Phase 3: Perturb+LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.20, 150.0)
        a_pert = perturb_and_lp(best_a, n_perturb=30, time_budget=p3_budget, verbose=True)
        obj_pert = compute_obj(a_pert)
        if obj_pert < best_obj:
            best_obj = obj_pert
            best_a = a_pert
            print(f"  Phase 3: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 3: obj={obj_pert:.8f}")

    # === Phase 4: LP again from improved point ===
    print(f"\n=== Phase 4: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = min(remaining * 0.20, 200.0)
        a_p4 = lp_alternate(best_a, time_budget=p4_budget, verbose=True)
        obj_p4 = compute_obj(a_p4)
        if obj_p4 < best_obj:
            best_obj = obj_p4
            best_a = a_p4
            print(f"  Phase 4: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 4: obj={obj_p4:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([
        max(200, n_best - 100), max(200, n_best - 50),
        n_best,
        n_best + 50, n_best + 100, n_best + 200, n_best + 400,
    ]))
    n_options = [x for x in n_options if 200 <= x <= 3000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == n_best:
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try

        alloc = min(40.0, remaining * 0.2)
        a_opt = lp_alternate(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse initializations ===
    print(f"\n=== Phase 6: Diverse Inits ===")
    remaining = deadline - time.monotonic()
    n_use = len(best_a)

    for d in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_div = diverse_initialization(n_use, rng)
        alloc = min(30.0, remaining * 0.2)
        a_opt = lp_alternate(a_div, time_budget=alloc)
        obj_div = compute_obj(a_opt)

        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Diverse {d}: obj={obj_div:.8f}")

    # === Phase 7: Final FISTA polish ===
    print(f"\n=== Phase 7: Final FISTA ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 60.0)
        a_final = fista_smoothed(best_a, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final FISTA: obj={best_obj:.8f} **")
        else:
            print(f"  Final FISTA: obj={obj_final:.8f}")

    # === Phase 8: Final LP polish ===
    print(f"\n=== Phase 8: Final LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 40.0)
        a_last = lp_alternate(best_a, time_budget=alloc)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Final LP: obj={best_obj:.8f} **")
        else:
            print(f"  Final LP: obj={obj_last:.8f}")

    # === Validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
