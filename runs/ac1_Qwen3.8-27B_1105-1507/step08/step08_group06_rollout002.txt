Nesterov momentum, periodic LP corrections, adaptive active-set selection,

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_for_response(a):

        return None, None

    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    while time.monotonic() - start < time_budget:
        # Compute convolution
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        obj = 2.0 * n * max_val

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive active set: peaks within tolerance of max
        threshold = max_val * active_tol
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > max_active:
            # Keep the top max_active peaks
            idx = np.argsort(conv)[::-1][:max_active]
            active = idx

        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            # d/da[i] conv[k] = 2*a[k-i] if 0 <= k-i < n, else 0
            # (with the convention that when i == k-i, it's still 2*a[i])
            # Actually: d/da[i] conv[k] = a[k-i] + a[i-k] where a[m]=0 if m out of range
            # For i in [0, n-1]: a[k-i] is valid if 0 <= k-i < n, i.e., k-n+1 <= i <= k
            #                    a[i-k] is valid if 0 <= i-k < n, i.e., k <= i <= k+n-1
            i_lo_1 = max(0, k - n + 1)
            i_hi_1 = min(n - 1, k)
            i_lo_2 = max(0, k)
            i_hi_2 = min(n - 1, k + n - 1)

            if i_hi_1 >= i_lo_1:
                grad[i_lo_1:i_hi_1 + 1] += a[k - np.arange(i_lo_1, i_hi_1 + 1)]
            if i_hi_2 >= i_lo_2:
                grad[i_lo_2:i_hi_2 + 1] += a[np.arange(i_lo_2, i_hi_2 + 1) - k]

        grad /= len(active)

        # Momentum update
        v = beta * v - eta * grad

        # Multiplicative update
        a = a * np.exp(v)
        a = np.maximum(a, 1e-15)  # prevent underflow
        s = float(np.sum(a))
        if s < 1e-12:
            a = best_a.copy()
            v = np.zeros(n)
            continue
        a = a / s

        # Decay learning rate
        eta = max(eta_decay * eta, min_eta)

        it += 1

        # Periodic LP correction
        if lp_every > 0 and it % lp_every == 0 and time.monotonic() - start < time_budget - 5:
            g_lp, M_lp = solve_lp_for_response(a)
            if g_lp is not None:
                # Blend with LP solution
                t_lp = 0.3
                a_blend = (1 - t_lp) * a + t_lp * g_lp
                a_blend = np.maximum(a_blend, 0)
                sb = float(np.sum(a_blend))
                if sb > 1e-10:
                    a_blend = a_blend / sb
                    obj_blend = compute_obj(a_blend)
                    if obj_blend < best_obj:
                        best_obj = obj_blend
                        best_a = a_blend.copy()
                        a = a_blend.copy()
                # Reset momentum after LP
                v = np.zeros(n)
                eta = max(eta * 1.5, min_eta)

        # Stall detection: perturb if stuck
        if stall_count > patience:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            scale = 0.02 + 0.01 * min(5, stall_count // patience)
            a_perturb = best_a * np.exp(rng.normal(0, scale, n))
            a_perturb = np.maximum(a_perturb, 0)
            sp = float(np.sum(a_perturb))
            if sp > 1e-10:
                a_perturb = a_perturb / sp
                obj_perturb = compute_obj(a_perturb)
                if obj_perturb < best_obj * 1.05:  # allow slightly worse for exploration
                    a = a_perturb.copy()
                    v = np.zeros(n)
                    eta = eta0 * 0.3
                    stall_count = 0

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MWU: it={it}, obj={best_obj:.8f}, eta={eta:.4f}, "
                  f"active={len(active)}, time={elapsed:.0f}s")

    return best_a


def lp_alternating_fast(a, time_budget=100.0, verbose=False):
    """
    Fast LP alternating without momentum - just simple alternating with line search.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    it = 0
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_for_response(a)
        if g_lp is None:
            break

        # Try several step sizes
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            if verbose:
                print(f"    LP-alt: it={it}, obj={best_obj:.8f}")
        else:
            # Stall: perturb
            rng = np.random.default_rng(int(time.monotonic() * 1e4) % (2**31))
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)

        it += 1

    return best_a


def subgradient_polish(a, time_budget=100.0, verbose=False):
    """
    Subgradient method with gradient projection and adaptive step.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.1
    momentum = np.zeros(n)
    beta = 0.8
    obj_history = []

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        obj = 2.0 * n * max_val

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Active set
        threshold = max_val * 0.92
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 25:
            active = np.argsort(conv)[::-1][:25]

        # Subgradient
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_lo_1 = max(0, k - n + 1)
            i_hi_1 = min(n - 1, k)
            i_lo_2 = max(0, k)
            i_hi_2 = min(n - 1, k + n - 1)
            if i_hi_1 >= i_lo_1:
                grad[i_lo_1:i_hi_1 + 1] += a[k - np.arange(i_lo_1, i_hi_1 + 1)]
            if i_hi_2 >= i_lo_2:
                grad[i_lo_2:i_hi_2 + 1] += a[np.arange(i_lo_2, i_hi_2 + 1) - k]
        grad /= len(active)

        # Center gradient (project to sum=0 hyperplane)
        grad_centered = grad - np.mean(grad)

        # Momentum
        momentum = beta * momentum - lr * grad_centered

        # Update
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / sn

        a = a_new
        obj_history.append(obj)

        # Adaptive learning rate
        if len(obj_history) > 500:
            recent = obj_history[-200:]
            if np.all(np.diff(recent) > -1e-10):
                lr = max(lr * 0.9, 0.001)
            elif np.mean(np.diff(recent)) < -1e-6:
                lr = min(lr * 1.05, 0.5)

    return best_a


def generate_initialization(kind, n, rng):
    """Generate a diverse initialization of length n."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.005
        return a / np.sum(a)
    elif kind == 'bilinear':
        a = np.linspace(1, 2, n)
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 10)
        a[:edge] = 0.2
        a[-edge:] = 0.2
        return a / np.sum(a)
    elif kind == 'two_peak':
        a = np.ones(n) * 0.3
        a[n//4] = 2.0
        a[3*n//4] = 2.0
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'fourier':
        # Based on Fejér kernel shape
        x = np.arange(n)
        k = n // 2
        a = np.sinc((x - k) / 5.0) ** 2 + 0.01
        return np.maximum(a, 0) / np.sum(np.maximum(a, 0))
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining MWU, LP-alternating, subgradient polish,
    diverse initializations, and length sweep.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Try diverse initializations at parent length ===
    print(f"\n=== Phase 1: Diverse Initializations (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    init_budget = min(60.0, remaining * 0.04)

    init_kinds = ['cosine', 'sine_taper', 'gaussian', 'plateau', 'two_peak', 'fourier', 'random']
    for kind in init_kinds:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        a_init = generate_initialization(kind, n_parent, rng)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init '{kind}': obj={obj_init:.8f} **")
        else:
            print(f"  Init '{kind}': obj={obj_init:.8f}")

    # Also try random inits
    for _ in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 8:
            break
        a_init = generate_initialization('random', n_parent, rng)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Random init: obj={obj_init:.8f} **")

    print(f"  Best after Phase 1: obj={best_score:.8f}")

    # === Phase 2: MWU optimization (main workhorse) ===
    print(f"\n=== Phase 2: MWU Optimization ===")
    a = best_seq.copy()
    remaining = deadline - time.monotonic()
    mwu_budget = min(500.0, remaining * 0.35)

    # Multiple MWU rounds with different hyperparameters
    mwu_configs = [
        {'beta': 0.7, 'eta0': 0.5, 'active_tol': 0.95, 'lp_every': 300},
        {'beta': 0.8, 'eta0': 0.3, 'active_tol': 0.90, 'lp_every': 200},
        {'beta': 0.6, 'eta0': 0.8, 'active_tol': 0.98, 'lp_every': 400},
        {'beta': 0.75, 'eta0': 0.4, 'active_tol': 0.92, 'lp_every': 250},
    ]

    for cfg_idx, cfg in enumerate(mwu_configs):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(mwu_budget / (len(mwu_configs) - cfg_idx), remaining * 0.7)
        print(f"  MWU Round {cfg_idx} (beta={cfg['beta']}, eta0={cfg['eta0']}):")
        a_new = mwu_optimize(a, time_budget=alloc, **cfg, verbose=True, seed=seed + cfg_idx)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  ** MWU R{cfg_idx}: obj={best_score:.8f} **")
        else:
            print(f"  MWU R{cfg_idx}: obj={obj:.8f}")
            # Perturb for next round
            a = best_seq * np.exp(rng.normal(0, 0.02, n_parent))
            a = np.maximum(a, 0)
            a = a / np.sum(a)

    # === Phase 3: LP Alternating (targeted) ===
    print(f"\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(200.0, remaining * 0.15)
    if lp_budget > 15:
        a_lp = lp_alternating_fast(a, time_budget=lp_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp.copy()
            print(f"  ** LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 4: Subgradient Polish ===
    print(f"\n=== Phase 4: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        sg_budget = min(150.0, remaining * 0.12)
        a_sg = subgradient_polish(a, time_budget=sg_budget, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a = a_sg.copy()
            print(f"  ** Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    # === Phase 5: MWU again from best (with different params) ===
    print(f"\n=== Phase 5: MWU Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        mwu2_budget = min(200.0, remaining * 0.15)
        a_m2 = mwu_optimize(best_seq, time_budget=mwu2_budget,
                            beta=0.85, eta0=0.2, eta_decay=0.9998,
                            min_eta=0.002, active_tol=0.90, max_active=30,
                            lp_every=0, verbose=True, seed=seed + 100)
        obj_m2 = compute_obj(a_m2)
        if obj_m2 < best_score:
            best_score = obj_m2
            best_seq = a_m2.copy()
            a = a_m2.copy()
            print(f"  ** MWU2: obj={best_score:.8f} **")
        else:
            print(f"  MWU2: obj={obj_m2:.8f}")

    # === Phase 6: Length Sweep ===
    print(f"\n=== Phase 6: Length Sweep ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    print(f"  Current best length: n={n_best}")

    # Try a range of lengths
    n_options = sorted(set([
        max(200, n_best - 200),
        max(200, n_best - 100),
        max(200, n_best - 50),
        max(200, n_best - 25),
        n_best,
        n_best + 25,
        n_best + 50,
        n_best + 100,
        n_best + 200,
    ]))
    n_options = [n for n in n_options if 200 <= n <= 3000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick MWU optimization
        alloc = min(60.0, remaining * 0.5)
        a_opt = mwu_optimize(a_try, time_budget=alloc,
                             beta=0.75, eta0=0.4, active_tol=0.93,
                             lp_every=150, verbose=False, seed=seed + n_try)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # MWU with aggressive params
        alloc = min(remaining * 0.5, 120.0)
        a_final = mwu_optimize(best_seq, time_budget=alloc,
                               beta=0.8, eta0=0.3, eta_decay=0.9997,
                               min_eta=0.003, active_tol=0.91, max_active=35,
                               lp_every=200, verbose=True, seed=seed + 200)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  ** Final MWU: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            # Quick subgradient
            a_sg2 = subgradient_polish(best_seq, time_budget=min(remaining * 0.5, 30.0))
            obj_sg2 = compute_obj(a_sg2)
            if obj_sg2 < best_score:
                best_score = obj_sg2
                best_seq = a_sg2.copy()
                print(f"  ** Final Subgrad: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
