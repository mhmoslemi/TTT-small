from scipy.optimize import linprog

    size = 2*n - 1
    # Use FFT for efficiency
    fsize = 1
    while fsize < 2*n - 1:
        fsize <<= 1
    fa = rfft(a, fsize)

    s = a.sum()
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * conv.max()

    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        end = min(n, 2*n-1-j)
        A[j:j+end, j] = a[:end]
    return A

def solve_lp(a):

    n = len(a)
    a_n = a / a.sum()
    A = build_A(a_n)
    
    # LP: min M
    # vars: g[0..n-1], M
    # obj: [0,...,0, 1]
    # A_ub: [A | -I_{2n-1}] @ [g; M] <= 0
    # A_eq: [1,...,1, 0] @ [g; M] = 1

    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = g.sum()

"""
Hybrid LP-alternating + fictitious play + subgradient mirror descent
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast HiGHS LP solves, tracks fictitious play averages for accelerated
convergence, and employs multi-dimension exploration with adaptive
perturbation to escape local stationary points.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    fsize = 1
    while fsize < 2 * n - 1:
        fsize <<= 1
    fa = rfft(a, fsize)
    return irfft(fa * fa, fsize)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2 (scale-invariant)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def build_A(a):
    """Build matrix A where (A@g)[k] = (a*g)[k] for linear convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(n, 2 * n - 1 - j)
        A[j:j + end, j] = a[:end]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_A(a_n)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status == 0:
        g = np.maximum(res.x[:n], 0.0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s
    return None


def solve_lp_dual_info(a):
    """Solve LP and return (g, M, y) where y are dual variables (peak weights)."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_A(a_n)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)

    return None, None, None


def subgradient_step(a, a_n, eta, topk=1):
    """
    Mirror descent step using subgradient of max_k (a*a)[k].
    a_n is normalized a (sum=1).
    """
    n = len(a_n)
    conv = compute_conv(a_n)
    
    if topk == 1:
        k_star = int(np.argmax(conv))
        s = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                s[i] = 2.0 * a_n[j]
    else:
        top_idx = np.argsort(conv)[::-1][:topk]
        max_val = conv[top_idx[0]]
        if max_val < 1e-15:
            return a_n.copy()
        # Soft weights
        weights = np.exp(3.0 * (conv[top_idx] - max_val) / max_val)
        weights /= weights.sum()
        s = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    s[i] += w * 2.0 * a_n[j]

    # Mirror descent (multiplicative update)
    log_a = np.log(np.maximum(a_n, 1e-30))
    log_a_new = log_a - eta * s
    log_a_new -= np.max(log_a_new)
    a_new = np.exp(log_a_new)
    a_new = np.maximum(a_new, 0.0)
    s_new = float(np.sum(a_new))
    if s_new > 1e-10:
        a_new = a_new / s_new
    else:
        a_new = a_n.copy()
    return a_new


def smooth_optimize(a, time_budget=60.0, beta_start=30.0, beta_end=2000.0):
    """Smooth optimization using log-sum-exp surrogate with mirror descent."""
    n = len(a)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / max(1.0, time_budget))
        beta = beta_start * (beta_end / beta_start) ** frac

        conv = compute_conv(a)
        m = conv.max()
        shifted = beta * (conv - m)
        weights = np.exp(shifted)
        weights /= weights.sum()

        # Gradient of smooth max
        g = np.zeros(n)
        for i in range(n):
            m_start = max(0, i - (n - 1))
            m_end = min(n - 1, 2 * n - 2 - i)
            if m_start <= m_end:
                g[i] = 2.0 * np.dot(a[m_start:m_end + 1],
                                     weights[i + m_start:i + m_end + 1])

        eta = 1.0 / np.sqrt(beta * n)
        log_a = np.log(a)
        log_a_new = log_a - eta * g
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 0.0)
        sn = float(np.sum(a_new))
        if sn > 1e-10:
            a_new = a_new / sn
        else:
            a_new = a.copy()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

        it += 1

    return best_a


def lp_fictitious_play(a_init, time_budget=400.0, verbose=False):
    """
    LP alternating with fictitious play tracking.
    Main workhorse for convergence.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    Q = a.copy()  # Fictitious play state
    start = time.monotonic()
    t = 0
    stall = 0
    patience = 30

    while time.monotonic() - start < time_budget:
        # Solve LP: best response to current a
        g = solve_lp(a)
        if g is None:
            g = a.copy()

        # Compute objectives
        obj_g = compute_obj(g)
        if obj_g < best_obj - 1e-12:
            best_obj = obj_g
            best_a = g.copy()
            stall = 0
        else:
            stall += 1

        # Fictitious play update
        Q = ((t + 1) * Q + g) / (t + 2)
        obj_Q = compute_obj(Q)
        if obj_Q < best_obj - 1e-12:
            best_obj = obj_Q
            best_a = Q.copy()
            stall = 0

        # Update a: direct jump to g
        a = g.copy()

        # If stalled, add perturbation
        if stall > patience:
            rng = np.random.default_rng(t * 31 + n)
            noise = 0.05 * rng.normal(0, 1, n)
            a = best_a * (1.0 + noise)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            Q = a.copy()
            stall = 0
            patience = max(15, patience // 2)

        t += 1
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP-FP it{t}: obj={best_obj:.8f}, t={elapsed:.1f}s, stall={stall}")

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def coordinate_refine(a, time_budget=30.0):

            break

        contribs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:

                    delta = frac * best_a[src]
                    if delta >= best_a[src]:
                        continue
                    a_try = best_a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    a_try = np.maximum(a_try, 0.0)
                    st = float(np.sum(a_try))
                    if st < 1e-10:
                        continue
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initial sequences."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps
        c1, c2 = n // 4, 3 * n // 4
        w = n // 10
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Triangle
        x = np.linspace(-1, 1, n)
        a = np.maximum(0.1, 1.0 - np.abs(x))
        a = a / a.sum()
    elif mode == 4:
        # Semicircle
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(0.01, 1 - x**2))
        a = a / a.sum()
    elif mode == 5:
        # Cosine bump
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Chirp envelope
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(2 * np.pi * (2 + 3 * x) * x)) + 0.05
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP alternating with fictitious play as the
    primary method, supplemented by subgradient mirror descent, smooth
    optimization, multi-dimension exploration, and coordinate refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0  # 20s safety buffer

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Initialize from parent ---
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nStarting: n={n_current}, obj={best_score:.8f}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # --- Phase 1: LP Fictitious Play (main workhorse, ~50% time) ---
    print(f"\n=== Phase 1: LP Fictitious Play (n={n_current}) ===")
    p1_budget = min((deadline - time.monotonic()) * 0.45, 600.0)
    if p1_budget > 60:
        # Run in multiple sub-iterations, checking time
        a_fp = lp_fictitious_play(best_seq, time_budget=p1_budget, verbose=True)
        obj_fp = compute_obj(a_fp)
        if obj_fp < best_score:
            best_score = obj_fp
            best_seq = a_fp.copy()
            n_current = len(best_seq)
            print(f"  ** FP result: obj={best_score:.8f}")
        else:
            print(f"  FP result: obj={obj_fp:.8f} (no improvement)")

    # --- Phase 2: LP with Blending (~20% time) ---
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 2: LP Blending (remaining={remaining:.0f}s) ===")
    p2_budget = min(remaining * 0.20, 280.0)
    if p2_budget > 40:
        a_bl = lp_with_blending(best_seq, time_budget=p2_budget, verbose=True)
        obj_bl = compute_obj(a_bl)
        if obj_bl < best_score:
            best_score = obj_bl
            best_seq = a_bl.copy()
            n_current = len(best_seq)
            print(f"  ** Blend result: obj={best_score:.8f}")
        else:
            print(f"  Blend result: obj={obj_bl:.8f} (no improvement)")

    # --- Phase 3: Smooth Optimization (~10% time) ---
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 3: Smooth Optimization (remaining={remaining:.0f}s) ===")
    p3_budget = min(remaining * 0.10, 120.0)
    if p3_budget > 30:
        a_sm = smooth_optimize(best_seq, time_budget=p3_budget,
                               beta_start=50.0, beta_end=3000.0)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            n_current = len(best_seq)
            print(f"  ** Smooth result: obj={best_score:.8f}")
        else:
            print(f"  Smooth result: obj={obj_sm:.8f} (no improvement)")

    # --- Phase 4: Multi-Dimension Exploration (~15% time) ---
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 4: Multi-Dimension Exploration (remaining={remaining:.0f}s) ===")
    n_base = len(best_seq)
    n_options = sorted(set([
        max(300, n_base - 200), max(300, n_base - 100),
        n_base + 100, n_base + 200, n_base + 400,
        500, 700, 900, 1100, 1300
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and n != n_base]

    alloc_per_n = min(remaining * 0.12, 80.0)
    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < alloc_per_n + 10:
            break

        a_try = resample_sequence(best_seq, n_try)
        # Quick LP optimization
        a_opt = lp_fictitious_play(a_try, time_budget=alloc_per_n * 0.7)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # --- Phase 5: Diverse Restarts (~10% time) ---
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Diverse Restarts (remaining={remaining:.0f}s) ===")
    n_use = len(best_seq)
    restart_budget = min(remaining * 0.10, 80.0)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        mode = r % 7
        a_r = diverse_init(n_use, rng, mode=mode)
        # Blend with current best
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0.0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.25, restart_budget)
        a_opt = lp_fictitious_play(a_r, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** Restart {r} (mode={mode}): obj={best_score:.8f}")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_opt:.8f}")

    # --- Phase 6: Intensive Final Polish (~remaining time) ---
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 6: Final Polish (remaining={remaining:.0f}s) ===")

    # 6a: Another round of LP FP on best
    if remaining > 60:
        alloc = min(remaining * 0.4, 200.0)
        a_final = lp_fictitious_play(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final LP-FP: obj={best_score:.8f}")

    # 6b: Subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 60.0)
        a_sg = subgradient_mirror_descent(best_seq, time_budget=alloc, topk=3)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  ** Subgradient: obj={best_score:.8f}")

    # 6c: Coordinate refinement
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.2, 30.0)
        a_cd = coordinate_refine(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  ** Coord refine: obj={best_score:.8f}")

    # 6d: One final short LP round
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 40.0)
        a_last = lp_fictitious_play(best_seq, time_budget=alloc)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  ** Last LP: obj={best_score:.8f}")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(500)
        best_score = compute_obj(best_seq)

    # Ensure valid output format
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 500
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
